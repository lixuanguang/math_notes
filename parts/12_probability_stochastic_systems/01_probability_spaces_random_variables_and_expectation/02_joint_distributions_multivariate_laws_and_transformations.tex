\subsection{Joint Distributions, Multivariate Laws, and Transformations}

\begin{definition} \label{def:joint-distributions-and-marginals} \hlt{Joint and Marginal Distributions}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{joint mass function} of discrete $X,Y$ is $f_{X,Y}(x,y)\defeq\mathbb{P}(X=x,Y=y)$.
\item The \hlt{discrete marginals} are $f_X(x)=\sum_y f_{X,Y}(x,y)$ and $f_Y(y)=\sum_x f_{X,Y}(x,y)$.
\item The \hlt{continuous marginals} are $f_X(x)=\int_{-\infty}^{\infty}f_{X,Y}(x,y)\,dy$, and $f_Y(y)=\int_{-\infty}^{\infty}f_{X,Y}(x,y)\,dx$.\\
$(X,Y)$ has joint density $f_{X,Y}$.
\item The \hlt{joint CDF} is $F_{X,Y}(x,y)\defeq\mathbb{P}(X\leq x,Y\leq y)$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:joint-cdf-rectangular-increments-and-density} \hlt{Rectangular Increments of a Joint CDF}\\
Let $F(x,y)\defeq\mathbb{P}(X\leq x,Y\leq y)$.\\
For $x_1<x_2$ and $y_1<y_2$, $\mathbb{P}(x_1<X\leq x_2,\ y_1<Y\leq y_2)=F(x_2,y_2)-F(x_1,y_2)-F(x_2,y_1)+F(x_1,y_1)$.\\
If $(X,Y)$ has density $f_{X,Y}$ and $F\in C^2$, then $f_{X,Y}(x,y)=\partial_x\partial_yF(x,y)$ and $F(x,y)=\int_{-\infty}^{x}\int_{-\infty}^{y}f_{X,Y}(u,v)\,dv\,du$.
\begin{proof} Let $A_{ij}=\{X\leq x_i,Y\leq y_j\}$ and $R=\{x_1<X\leq x_2,\ y_1<Y\leq y_2\}$.\\
Pointwise, $\mathbf{1}_R=\mathbf{1}_{A_{22}}-\mathbf{1}_{A_{12}}-\mathbf{1}_{A_{21}}+\mathbf{1}_{A_{11}}$; taking expectations gives the four-corner identity.\\
If $F(x,y)=\int_{-\infty}^{x}\int_{-\infty}^{y}f_{X,Y}(u,v)\,dv\,du$ and $F\in C^2$, the two-variable FTC gives $\partial_x\partial_yF=f_{X,Y}$.\\
\hlt{Structural Commentary}: Joint CDFs encode southwest rectangles; finite rectangles are four-corner increments.\\
\hlt{Key Insight}: Mixed derivatives are infinitesimal rectangular increments.
\end{proof}
\end{proposition}

\begin{example} \label{ex:three-toss-joint-mass-table} \hlt{A Three-Toss Joint Mass Table}\\
Let $\Omega=\{H,T\}^3$ with $\mathbb{P}(\{\omega\})=2^{-3}$, set $X=\mathbf{1}_{\{\omega_1=H\}}$, $Y=\sum_{i=1}^{3}\mathbf{1}_{\{\omega_i=H\}}$, and write $p_{xy}\defeq\mathbb{P}(X=x,Y=y)$.
\[
\begin{array}{c|cccc}
&Y=0&Y=1&Y=2&Y=3\\ \hline
X=0&1/8&2/8&1/8&0\\
X=1&0&1/8&2/8&1/8
\end{array}
\]
Thus $p_{0,2}=1/8$, $\sum_y p_{0y}=\sum_y p_{1y}=1/2$, and $\sum_x p_{xj}=\binom{3}{j}2^{-3}$ for $j=0,\ldots,3$.
\end{example}

\begin{definition} \label{def:independent-random-variables-and-iid-samples} \hlt{Independent Random Variables and IID Samples}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Random variables $X_1,\ldots,X_n$ are \hlt{independent}\\
if $\mathbb{P}(X_1\in B_1,\ldots,X_n\in B_n)=\prod_{i=1}^{n}\mathbb{P}(X_i\in B_i)$ for all $B_i\in\mathcal{B}(\R)$.
\item They are \hlt{identically distributed} if $\mathcal{L}(X_1)=\cdots=\mathcal{L}(X_n)$.
\item They are \hlt{iid} if they are independent and identically distributed.
\item The \hlt{random vector} $X=(X_1,\ldots,X_n)$ is a \hlt{random sample} when $X_1,\ldots,X_n$ are iid from a common law.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:factorization-tests-for-random-variable-independence} \hlt{Factorization Tests for Independence}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Discrete factorization}: discrete $X,Y$ are independent iff $f_{X,Y}(x,y)=f_X(x)f_Y(y)$ for all $x,y$.
\item \hlt{Density factorization}: if $(X,Y)$ has joint density $f_{X,Y}$ and marginals $f_X,f_Y$,\\
then $X,Y$ are independent iff $f_{X,Y}(x,y)=f_X(x)f_Y(y)$ a.e.
\item \hlt{Rectangular support test}: if $f_{X,Y}(x,y)=g(x)h(y)\mathbf{1}_{A\times B}(x,y)$ for measurable $A,B$,\\
then $X,Y$ are independent after normalization.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) If $X\perp\!\!\!\perp Y$, then $f_{X,Y}(x,y)=\mathbb{P}(X=x,Y=y)=\mathbb{P}(X=x)\mathbb{P}(Y=y)=f_X(x)f_Y(y)$.
\item[$(i)$] ($\Leftarrow$) If $f_{X,Y}=f_Xf_Y$, then for $A,B$, $\mathbb{P}(X\in A,Y\in B)=\sum_{x\in A}\sum_{y\in B}f_X(x)f_Y(y)=\mathbb{P}(X\in A)\mathbb{P}(Y\in B)$.
\item[$(ii)$] ($\Rightarrow$) If $X\perp\!\!\!\perp Y$, the product law has density $f_Xf_Y$, so uniqueness of densities gives $f_{X,Y}=f_Xf_Y$ a.e.
\item[$(ii)$] ($\Leftarrow$) If $f_{X,Y}=f_Xf_Y$ a.e., then for Borel $A,B$,\\
$\mathbb{P}(X\in A,Y\in B)=\int_A\int_Bf_X(x)f_Y(y)\,dy\,dx=\mathbb{P}(X\in A)\mathbb{P}(Y\in B)$.
\item[$(iii)$] Let $G=\int_Ag$ and $H=\int_Bh$. Normalization gives $GH=1$, while $f_X(x)=g(x)H\mathbf{1}_A(x)$ and\\
$f_Y(y)=h(y)G\mathbf{1}_B(y)$, hence $f_X(x)f_Y(y)=f_{X,Y}(x,y)$ a.e.; apply (ii).
\end{enumerate}
\hlt{Structural Commentary}: Independence is product structure of the joint law.\\
\hlt{Key Insight}: Independence means the joint law factors.
\end{proof}
\end{proposition}

\begin{example} \label{ex:bivariate-density-nonrectangular-region} \hlt{A Non-Rectangular Bivariate Density}\\
Let $S=\{(x,y):x^2\leq y\leq1\}$ and $f_{X,Y}(x,y)=cx^2y\mathbf{1}_S(x,y)$.\\
Since $S=\{(x,y):-1\leq x\leq1,\ x^2\leq y\leq1\}$, $1=c\int_{-1}^{1}\int_{x^2}^{1}x^2y\,dy\,dx=4c/21$, so $c=21/4$.\\
For $E=\{X\geq Y\}$, $S\cap E=\{(x,y):0\leq x\leq1,\ x^2\leq y\leq x\}$, hence $\mathbb{P}(E)=(21/4)\int_0^1\int_{x^2}^{x}x^2y\,dy\,dx=3/20$.\\
\hlt{Key Insight}: In bivariate integration, pick one variable first and let the other range over the corresponding slice.
\end{example}

\begin{definition} \label{def:multinomial-and-multivariate-normal} \hlt{Two Multivariate Families}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Multinomial}: for $n\in\N$ and $p\in[0,1]^k$ with $\sum_{j=1}^{k}p_j=1$, $X\sim\operatorname{Multinomial}(n,p)$\\
if $\mathbb{P}(X=x)=n!(x_1!\cdots x_k!)^{-1}\prod_{j=1}^{k}p_j^{x_j}$ for $x\in\N_0^k$ with $\sum_{j=1}^{k}x_j=n$; then $X_j\sim\operatorname{Binomial}(n,p_j)$.
\item \hlt{Multivariate normal}: for $\mu\in\R^k$ and symmetric positive definite $\Sigma\in\R^{k\times k}$, $X\sim N(\mu,\Sigma)$\\
if $f_X(x)=((2\pi)^k\det\Sigma)^{-1/2}\exp(-(x-\mu)^t\Sigma^{-1}(x-\mu)/2)$ for $x\in\R^k$.\\
$N(0,I_k)$ has independent $N(0,1)$ coordinates.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:multivariate-normal-affine-standardization} \hlt{Multivariate Normal Affine Standardization}\\
Let $k\in\N$, $\mu\in\R^k$, $\Sigma\in\R^{k\times k}$ with $\Sigma\succ0$, and $A=\Sigma^{1/2}$. For $X\in\R^k$, $X\sim N(\mu,\Sigma)$ iff $A^{-1}(X-\mu)\sim N(0,I_k)$.\\
Equivalently, if $Z\sim N(0,I_k)$, then $\mu+AZ\sim N(\mu,\Sigma)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Set $Z=\Sigma^{-1/2}(X-\mu)$, so $x=\mu+\Sigma^{1/2}z$ and $|\det\Sigma^{1/2}|=(\det\Sigma)^{1/2}$.\\
By change of variables, $f_Z(z)=f_X(\mu+\Sigma^{1/2}z)(\det\Sigma)^{1/2}=(2\pi)^{-k/2}\exp(-z^tz/2)$, hence $Z\sim N(0,I_k)$.
\item[$(\Leftarrow)$] Let $X=\mu+\Sigma^{1/2}Z$ with $Z\sim N(0,I_k)$. Since $z=\Sigma^{-1/2}(x-\mu)$ and $|\det\Sigma^{-1/2}|=(\det\Sigma)^{-1/2}$,\\
$f_X(x)=f_Z(\Sigma^{-1/2}(x-\mu))(\det\Sigma)^{-1/2}=((2\pi)^k\det\Sigma)^{-1/2}\exp(-(x-\mu)^t\Sigma^{-1}(x-\mu)/2)$, so $X\sim N(\mu,\Sigma)$.
\end{enumerate}
\hlt{Structural Commentary}: Positive definite covariance is an affine deformation of the standard normal.\\
\hlt{Key Insight}: Standardize by $\Sigma^{-1/2}$; reconstruct by $\Sigma^{1/2}$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:multivariate-normal-basic-calculus} \hlt{Basic Multivariate Normal Calculus}\\
Let $X=(X_a,X_b)\sim N(\mu,\Sigma)$ with $X_a\in\R^r$, $X_b\in\R^s$, $\mu=(\mu_a,\mu_b)$, and $\Sigma=\begin{pmatrix}\Sigma_{aa}&\Sigma_{ab}\\ \Sigma_{ba}&\Sigma_{bb}\end{pmatrix}\succ0$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Marginal}: $X_a\sim N(\mu_a,\Sigma_{aa})$.
\item \hlt{Conditional}: $X_b\mid X_a=x_a\sim N(\mu_b+\Sigma_{ba}\Sigma_{aa}^{-1}(x_a-\mu_a),\Sigma_{bb}-\Sigma_{ba}\Sigma_{aa}^{-1}\Sigma_{ab})$.
\item \hlt{Linear form}: for any vector $a$, $a^tX\sim N(a^t\mu,a^t\Sigma a)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $P_aX=X_a$. By Proposition \ref{prop:multivariate-normal-affine-standardization}, $X=\mu+\Sigma^{1/2}Z$ with $Z\sim N(0,I)$.\\
Hence $X_a=P_a\mu+P_a\Sigma^{1/2}Z\sim N(\mu_a,P_a\Sigma P_a^t)=N(\mu_a,\Sigma_{aa})$.
\item Set $S\defeq\Sigma_{bb}-\Sigma_{ba}\Sigma_{aa}^{-1}\Sigma_{ab}$ and $m(x_a)\defeq\mu_b+\Sigma_{ba}\Sigma_{aa}^{-1}(x_a-\mu_a)$.\\
The block determinant and inverse identities give $\det\Sigma=\det\Sigma_{aa}\det S$ and\\
$(x-\mu)^t\Sigma^{-1}(x-\mu)=(x_a-\mu_a)^t\Sigma_{aa}^{-1}(x_a-\mu_a)+(x_b-m(x_a))^tS^{-1}(x_b-m(x_a))$.\\
Thus $f_{b\mid a}(x_b\mid x_a)=f_{a,b}(x_a,x_b)/f_a(x_a)$ is the $N(m(x_a),S)$ density.
\item With $X=\mu+\Sigma^{1/2}Z$, $a^tX=a^t\mu+(\Sigma^{1/2}a)^tZ$.\\
Since $Z$ has independent $N(0,1)$ coordinates, $(\Sigma^{1/2}a)^tZ\sim N(0,a^t\Sigma a)$, hence $a^tX\sim N(a^t\mu,a^t\Sigma a)$.
\end{enumerate}
\hlt{Structural Commentary}: Marginals, conditionals, and linear projections of Gaussian vectors remain Gaussian.\\
\hlt{Key Insight}: Gaussian calculus is affine calculus plus Schur complements.
\end{proof}
\end{proposition}

\begin{definition} \label{def:copulas-and-copula-densities} \hlt{Copulas}\\
A \hlt{copula} is a joint CDF $C:[0,1]^2\to[0,1]$ with uniform marginals: $C(u,1)=u$ and $C(1,v)=v$ for $u,v\in[0,1]$.\\
If $C\in C^2$, its \hlt{copula density} is $c(u,v)\defeq\partial_u\partial_vC(u,v)$.
\end{definition}

\begin{proposition} \label{prop:copula-density-construction} \hlt{Constructing a Joint Law from Marginals and a Copula}\\
Let $F_X,F_Y$ be continuous CDFs with densities $f_X,f_Y$, and let $C$ be a copula. Define $F_{X,Y}(x,y)\defeq C(F_X(x),F_Y(y))$. Then $F_{X,Y}$ has marginals $F_X,F_Y$. If $C\in C^2$ with density $c=\partial_u\partial_vC$, then $f_{X,Y}(x,y)=c(F_X(x),F_Y(y))f_X(x)f_Y(y)$.
\begin{proof} $F_{X,Y}(x,\infty)=C(F_X(x),1)=F_X(x)$ and $F_{X,Y}(\infty,y)=C(1,F_Y(y))=F_Y(y)$.\\
If densities exist, $\partial_x\partial_y C(F_X(x),F_Y(y))=(\partial_u\partial_vC)(F_X(x),F_Y(y))f_X(x)f_Y(y)$ by the chain rule.\\
\hlt{Structural Commentary}: Marginals and dependence are separated into $F_X,F_Y$ and $C$.\\
\hlt{Key Insight}: Marginals alone do not determine a joint law; the copula supplies the dependence.
\end{proof}
\end{proposition}

\begin{example} \label{ex:farlie-morgenstern-copulas} \hlt{Farlie--Morgenstern Copulas}\\
For $|\alpha|\leq1$, $C_\alpha(u,v)=uv\{1+\alpha(1-u)(1-v)\}$ is a copula with density $c_\alpha(u,v)=1+\alpha(1-2u)(1-2v)$.\\
Here $\alpha=0$ gives the independence copula $uv$, while $\alpha=-1$ and $\alpha=1$ give densities $2u+2v-4uv$ and $2-2u-2v+4uv$ on $[0,1]^2$, respectively; all have uniform marginals.
\end{example}

\begin{method} \label{meth:one-variable-transformation-cdf-method} \hlt{One-Variable Transformations}\\
Let $Y=r(X)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Discrete pushforward}: if $X$ is discrete, then $f_Y(y)=\mathbb{P}(Y=y)=\mathbb{P}(X\in r^{-1}(\{y\}))=\sum_{x:r(x)=y}f_X(x)$.
\item \hlt{CDF method}: if $X$ has density $f_X$, then $F_Y(y)=\mathbb{P}(r(X)\leq y)=\int_{\{x:r(x)\leq y\}}f_X(x)\,dx$ and $f_Y=F_Y'$ where differentiable.
\item \hlt{Monotone change}: if $r$ is strictly monotone on $\operatorname{supp}X$ with inverse $s$, then $f_Y(y)=f_X(s(y))\abs{s'(y)}\mathbf{1}_{r(\operatorname{supp}X)}(y)$.
\end{enumerate}
\hlt{Key Insight}: The CDF method works because it translates the transformed event back into an event about $X$.
\end{method}

\begin{example} \label{ex:log-transform-exponential} \hlt{Log Transform of an Exponential Variable}\\
Let $X\sim\operatorname{Exp}(1)$ and $Y=\log X$. Since $r(x)=\log x$ has inverse $s(y)=e^y$ with $s'(y)=e^y$, the change-of-variables formula gives $f_Y(y)=f_X(e^y)e^y=e^ye^{-e^y}$ for $y\in\R$.
\end{example}

\begin{example} \label{ex:square-transform-nonmonotone-uniform} \hlt{A Nonmonotone Square Transform}\\
Let $X\sim\operatorname{Uniform}(-1,3)$ and $Y=X^2$. Since $f_X=1/4$ on $[-1,3]$, $r^{-1}(\{y\})\cap[-1,3]=\{-\sqrt y,\sqrt y\}$ for $0<y<1$ and $\{\sqrt y\}$ for $1<y<9$. Hence $f_Y(y)=1/(4\sqrt y)\mathbf{1}_{(0,1)}(y)+1/(8\sqrt y)\mathbf{1}_{(1,9)}(y)$.\\
\hlt{Key Insight}: Non-one-to-one transformations require splitting the range into cases.
\end{example}

\begin{example} \label{ex:standard-normal-square-chi-square} \hlt{Squaring a Standard Normal}\\
Let $Z\sim N(0,1)$ and $X=Z^2$. For $x>0$, $F_X(x)=\mathbb{P}(|Z|\leq\sqrt{x})=\Phi(\sqrt{x})-\Phi(-\sqrt{x})$,\\
so $f_X(x)=x^{-1/2}\phi(\sqrt{x})=(2\pi)^{-1/2}x^{-1/2}e^{-x/2}$. Hence $X\sim\chi_1^2=\operatorname{Gamma}(1/2,1/2)$ in the rate convention.
\end{example}

\begin{example} \label{ex:reciprocal-uniform-density} \hlt{Reciprocal of a Uniform Variable}\\
Let $U\sim\operatorname{Uniform}(0,1)$ and $V=1/U$. Then $F_V(v)=0$ for $v<1$, while for $v\geq1$, $F_V(v)=\mathbb{P}(U\geq1/v)=1-1/v$.\\
Hence $f_V(v)=v^{-2}\mathbf{1}_{[1,\infty)}(v)$.
\end{example}

\begin{proposition} \label{prop:probability-integral-and-inverse-transform} \hlt{Probability Integral Transform and Inverse Transform}\\
Let $F:I\to(0,1)$ be a continuous strictly increasing CDF with inverse $F^{-1}:(0,1)\to I$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $X\sim F$, then $F(X)\sim\operatorname{Uniform}(0,1)$.
\item If $U\sim\operatorname{Uniform}(0,1)$ and $Y=F^{-1}(U)$, then $F_Y=F$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $0<u<1$, $\mathbb{P}(F(X)\leq u)=\mathbb{P}(X\leq F^{-1}(u))=F(F^{-1}(u))=u$, with endpoint values $0,1$.
\item For $x\in I$, $\mathbb{P}(Y\leq x)=\mathbb{P}(F^{-1}(U)\leq x)=\mathbb{P}(U\leq F(x))=F(x)$.
\end{enumerate}
\hlt{Structural Commentary}: $F$ maps its law to uniform scale; $F^{-1}$ maps uniform scale back.\\
\hlt{Key Insight}: Uniform random numbers become target laws by applying inverse CDFs.
\end{proof}
\end{proposition}

\begin{method} \label{meth:inverse-transform-simulation-examples} \hlt{Inverse-Transform Generation}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Exponential}: if $U\sim\operatorname{Uniform}(0,1)$ and $\lambda>0$, then $T=-\lambda^{-1}\log(1-U)\sim\operatorname{Exp}(\lambda)$.\\
Equivalently $T=-\lambda^{-1}\log U$.
\item \hlt{Weibull}: if $F(x)=1-\exp(-(x/\alpha)^\beta)$ for $x\geq0$ and $\alpha,\beta>0$,\\
then $F^{-1}(u)=\alpha[-\log(1-u)]^{1/\beta}$ and $W=\alpha[-\log(1-U)]^{1/\beta}\sim F$.
\item \hlt{Integer-valued inverse CDF}: if $F$ is integer-valued and $U\sim\operatorname{Uniform}(0,1)$, set $Y=\inf\{k\in\Z:F(k)\geq U\}$.\\
Equivalently $F(k-1)<U\leq F(k)$. Then $F_Y=F$.
\end{enumerate}
\hlt{Key Insight}: The simulation problem is reduced to solving $u=F(x)$ for $x$.
\end{method}

\begin{method} \label{meth:several-variable-transformation-cdf-method} \hlt{Several-Variable Transformations}\\
For $Z=r(X,Y)$ with joint density $f_{X,Y}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Sublevel set}: define $A_z\defeq\{(x,y):r(x,y)\leq z\}$.
\item \hlt{CDF integral}: compute $F_Z(z)=\mathbb{P}(Z\leq z)=\iint_{A_z}f_{X,Y}(x,y)\,dx\,dy$.
\item \hlt{Density}: differentiate where possible, $f_Z(z)=F_Z'(z)$.
\end{enumerate}
\end{method}

\begin{proposition} \label{prop:two-dimensional-density-change-of-variables} \hlt{Two-Dimensional Density Change of Variables}\\
Let $(X,Y)$ have density $f_{X,Y}$ on $D\subseteq\R^2$, and let $T:D\to E$ be one-to-one $C^1$ with inverse $h:E\to D$ and $\det DT\neq0$ on $D$. If $(U,V)=T(X,Y)$, then $f_{U,V}(u,v)=f_{X,Y}(h(u,v))\abs{\det Dh(u,v)}\mathbf{1}_E(u,v)$.
\begin{proof}
For Borel $B\subseteq E$, $\mathbb{P}((U,V)\in B)=\mathbb{P}((X,Y)\in h(B))$\\
$=\int_{h(B)}f_{X,Y}(x,y)\,dx\,dy=\int_B f_{X,Y}(h(u,v))\abs{\det Dh(u,v)}\,du\,dv$ (by Theorem \ref{thm:change-of-variables-rk})\\
\hlt{Structural Commentary}: Transforming a density means preserving probability mass while rescaling area.\\
\hlt{Key Insight}: The absolute Jacobian accounts for local volume distortion.
\end{proof}
\end{proposition}

\begin{example} \label{ex:sum-and-quotient-density-formulas} \hlt{Sums and Quotients}\\
Let $(X,Y)$ have joint density $f_{X,Y}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Sum: for $S=X+Y$, $f_S(s)=\int_\R f_{X,Y}(x,s-x)\,dx$; if $X\perp\!\!\!\perp Y$, then $f_S(s)=\int_\R f_X(x)f_Y(s-x)\,dx$.
\item Quotient: for $Z=Y/X$, $f_Z(z)=\int_\R\abs{x}f_{X,Y}(x,xz)\,dx$; if $X\perp\!\!\!\perp Y$, then $f_Z(z)=\int_\R\abs{x}f_X(x)f_Y(xz)\,dx$.
\end{enumerate}
\hlt{Key Insight}: Add one auxiliary coordinate, change variables, integrate it out.
\end{example}

\begin{example} \label{ex:sum-of-two-independent-uniforms} \hlt{Sum of Two Independent Uniforms}\\
Let $X_1,X_2\sim\operatorname{Uniform}(0,1)$ be independent and $Y=X_1+X_2$.\\
Since $f_{X_1,X_2}=\mathbf{1}_{[0,1]^2}$, $F_Y(y)$ is the area of $A_y=\{(x_1,x_2)\in[0,1]^2:x_1+x_2\leq y\}$,\\
so $F_Y(y)=(y^2/2)\mathbf{1}_{[0,1]}(y)+(1-(2-y)^2/2)\mathbf{1}_{(1,2]}(y)+\mathbf{1}_{(2,\infty)}(y)$. Hence $f_Y(y)=y\mathbf{1}_{(0,1)}(y)+(2-y)\mathbf{1}_{(1,2)}(y)$.
\end{example}

\begin{example} \label{ex:polar-transform-standard-normal} \hlt{Polar Transform of Two Standard Normals}\\
Let $X,Y\overset{\mathrm{iid}}{\sim}N(0,1)$, $R=(X^2+Y^2)^{1/2}$, and $\Theta=\operatorname{atan2}(Y,X)$.\\
Polar change of variables gives $f_{R,\Theta}(r,\theta)=(2\pi)^{-1}re^{-r^2/2}\mathbf{1}_{[0,\infty)}(r)\mathbf{1}_{[0,2\pi)}(\theta)$,\\
so $R\perp\!\!\!\perp\Theta$, $\Theta\sim\operatorname{Uniform}(0,2\pi)$, and $R^2\sim\operatorname{Exp}(1/2)$.\\
Hence for iid $U_1,U_2\sim\operatorname{Uniform}(0,1)$, $(\sqrt{-2\log U_1}\cos(2\pi U_2),\sqrt{-2\log U_1}\sin(2\pi U_2))\sim N(0,I_2)$.
\end{example}
