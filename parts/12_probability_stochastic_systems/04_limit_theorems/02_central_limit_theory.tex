\subsection{Central Limit Theory}
% Characteristic functions, CLT, Lindeberg/Feller flavor, delta method preview, weak convergence.

The law of large numbers says that $\overline{X}_n$ is close to $\mu$ for large
$n$. The central limit theorem describes the scale and shape of the remaining
fluctuation.

\begin{theorem} \label{thm:central-limit-theorem-iid} \hlt{Central Limit Theorem}\\
Let $(X_i)_{i\geq1}$ be IID real-valued random variables, $\mathbb{E}X_1=\mu$, $\operatorname{Var}(X_1)=\sigma^2\in(0,+\infty)$.\\
Let $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$, $Z_n\defeq\sigma^{-1}\sqrt{n}(\overline{X}_n-\mu)$.\\
Then $Z_n\rightsquigarrow N(0,1)$, equivalently $\mathbb{P}(Z_n\leq z)\to\Phi(z)\defeq\int_{-\infty}^{z}(2\pi)^{-\frac{1}{2}}e^{-\frac{x^2}{2}}\,dx$ $\forall z\in\mathbb{R}$.
\begin{proof} Set $Y_i\defeq\frac{X_i-\mu}{\sigma}$, so $\mathbb{E}Y_i=0$, $\operatorname{Var}(Y_i)=1$, and $Z_n=n^{-\frac{1}{2}}\sum\limits_{i=1}^{n}Y_i$.\\
For $\varphi(t)\defeq\mathbb{E}e^{\mathrm{i}tY_1}$, the finite second moment expansion gives $\varphi(u)=1-\frac{u^2}{2}+o(u^2)$ as $u\to0$. \\
By independence, $\varphi_{Z_n}(t)=[\varphi(\frac{t}{\sqrt{n}})]^n=[1-\frac{t^2}{2n}+o(\frac{1}{n})]^n\to e^{-\frac{t^2}{2}}$ for every $t\in\mathbb{R}$. \\
Since $e^{-\frac{t^2}{2}}$ is the characteristic function of $N(0,1)$, Levy's continuity theorem gives $Z_n\rightsquigarrow N(0,1)$; continuity of $\Phi$ gives the displayed c.d.f. convergence.\\
\hlt{Structural Commentary}: Centering and scaling turn the sum into unit-variance IID increments.\\
\hlt{Key Insight}: Characteristic functions turn normalized sums into powers, whose limit is Gaussian.
\end{proof}
\end{theorem}

\begin{method} \label{meth:clt-normal-approximation-forms} \hlt{Equivalent CLT Approximation Forms}\\
Let $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$ with mean $\mu$ and variance $\sigma^2\in(0,+\infty)$, and set $Z_n\defeq\sigma^{-1}\sqrt{n}(\overline{X}_n-\mu)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Standardized form: $Z_n\approx N(0,1)$.
\item Centered form: $\sqrt{n}(\overline{X}_n-\mu)\approx N(0,\sigma^2)$.
\item Sample-mean form: $\overline{X}_n\approx N(\mu,\frac{\sigma^2}{n})$.
\end{enumerate}
Use any equivalent form to approximate probabilities involving $\overline{X}_n$.
\end{method}

\begin{example} \label{ex:clt-measurement-error-interval} \hlt{Measurement Error and Sample Size}\\
Let $X_1,\ldots,X_n$ be independent measurements with $\mathbb{E}X_i=\mu$ and $\operatorname{Var}(X_i)=\sigma^2$.\\
For $c>0$, the CLT gives $\mathbb{P}(\abs{\overline{X}_n-\mu}<c)\approx\Phi(\frac{c\sqrt{n}}{\sigma})-\Phi(\frac{-c\sqrt{n}}{\sigma})$.\\
For $(n,\sigma,c)=(16,1,0.5)$, this is $\Phi(2)-\Phi(-2)\approx0.954$.
\end{example}

\begin{lemma} \label{lem:mgf-convergence-criterion} \hlt{MGF Continuity Theorem}\\
Let $Z_n,Z$ be real-valued with MGFs $\psi_n(t)\defeq\mathbb{E}e^{tZ_n}$ and $\psi(t)\defeq\mathbb{E}e^{tZ}$ finite on $(-a,a)$ for some $a>0$.\\
If $\psi_n(t)\to\psi(t)$ for all $t\in(-a,a)$, then $Z_n\rightsquigarrow Z$.
\begin{proof}
For each $s\in\mathbb{R}$, the characteristic functions satisfy $\varphi_n(s)=\mathbb{E}e^{\mathrm{i}sZ_n}=\psi_n(\mathrm{i}s)$ by analytic continuation on the strip $\{z:\abs{\operatorname{Re}z}<a\}$. Pointwise MGF convergence near $0$ gives $\varphi_n(s)\to\varphi(s)$, the characteristic function of $Z$; Levy's continuity theorem yields $Z_n\rightsquigarrow Z$.\\
\hlt{Structural Commentary}: Local MGF convergence determines local characteristic-function convergence.\\
\hlt{Key Insight}: MGFs near $0$ identify the weak limit.
\end{proof}
\end{lemma}

\begin{proposition} \label{prop:mgf-proof-clt-near-zero} \hlt{MGF Proof of the IID CLT}\\
Let $(X_i)_{i\geq1}$ be iid with $\mathbb{E}X_1=0$, $\operatorname{Var}(X_1)=\sigma^2\in(0,+\infty)$, and MGF $M(t)\defeq\mathbb{E}e^{tX_1}$ finite near $0$.\\
Set $S_n\defeq\sum\limits_{i=1}^{n}X_i$ and $Z_n\defeq \frac{S_n}{\sigma\sqrt{n}}$. Then $Z_n\rightsquigarrow N(0,1)$, equivalently $\mathbb{P}(Z_n\leq x)\to\Phi(x)$ for all $x\in\mathbb{R}$.
\begin{proof}
For fixed $t$ near $0$, independence gives $M_{Z_n}(t)=\mathbb{E}e^{\frac{tS_n}{\sigma\sqrt{n}}}=\prod\limits_{i=1}^{n}\mathbb{E}e^{\frac{tX_i}{\sigma\sqrt{n}}}=[M(\frac{t}{\sigma\sqrt{n}})]^n$.\\
Since $M(0)=1$, $M'(0)=0$, and $M''(0)=\sigma^2$, Taylor expansion gives $M(s)=1+\frac{\sigma^2s^2}{2}+o(s^2)$ as $s\to0$.\\
Hence $M_{Z_n}(t)=(1+\frac{t^2}{2n}+o(\frac{1}{n}))^n\to e^{\frac{t^2}{2}}$, the MGF of $N(0,1)$.\\
Lemma \ref{lem:mgf-convergence-criterion} gives $Z_n\rightsquigarrow N(0,1)$, and continuity of $\Phi$ gives the c.d.f. convergence.\\
\hlt{Structural Commentary}: The MGF proof needs the stronger near-zero MGF assumption.\\
\hlt{Key Insight}: Standardization gives $\frac{t^2}{2n}$; independence exponentiates.
\end{proof}
\end{proposition}

\begin{example} \label{ex:clt-poisson-program-errors} \hlt{A Poisson Error-Count Approximation}\\
Let $X_1,\ldots,X_{125}\sim\operatorname{Poisson}(5)$ be IID and set $\overline{X}_{125}=125^{-1}\sum\limits_{i=1}^{125}X_i$.\\
Since $\mu=5$ and $\sigma^2=5$, the CLT gives $\mathbb{P}(\overline{X}_{125}<5.5)\approx\Phi(\frac{\sqrt{125}(5.5-5)}{\sqrt5})=\Phi(2.5)\approx0.9938$.
\end{example}

\begin{example} \label{ex:standardized-poisson-normal-limit} \hlt{Normal Approximation to a Poisson Count}\\
Let $X_\lambda\sim\operatorname{Poisson}(\lambda)$, $Z_\lambda\defeq\frac{X_\lambda-\lambda}{\sqrt{\lambda}}$. Then $M_{Z_\lambda}(t)=\exp(-t\sqrt{\lambda}+\lambda(e^{\frac{t}{\sqrt{\lambda}}}-1))$ and $\log M_{Z_\lambda}(t)=\frac{t^2}{2}+o(1)$, so $Z_\lambda\rightsquigarrow N(0,1)$. If $X\sim\operatorname{Poisson}(900)$, then $\mathbb{P}(X>950)=\mathbb{P}(\frac{X-900}{30}>\frac{5}{3})\approx1-\Phi(\frac{5}{3})\approx0.04779$, close to $0.04712$.
\end{example}

\begin{example} \label{ex:normal-approximation-binomial-fair-coin} \hlt{Normal Approximation to the Binomial}\\
If $X\sim\operatorname{Binomial}(n,p)$, $np>5$, $n(1-p)>5$, then $X\approx N(np,np(1-p))$; $\frac{X-np}{\sqrt{np(1-p)}}\approx N(0,1)$.\\
For $X\sim\operatorname{Binomial}(100,\frac{1}{2})$, $\mathbb{P}(X\geq60)\approx1-\Phi(\frac{60-50}{5})=1-\Phi(2)\approx0.0228$.
\end{example}

\begin{example} \label{ex:gamma-skewness-clt-approximation} \hlt{Skewed Summands Need More Averaging}\\
Let $E_1,\ldots,E_n$ be iid $\operatorname{Exp}(1)$ and $G_n\defeq\sum\limits_{i=1}^{n}E_i\sim\operatorname{Gamma}(n,1)$. Then $\frac{G_n-n}{\sqrt{n}}\rightsquigarrow N(0,1)$.\\
Skewness makes the normal approximation slower for small $n$.
\end{example}

\begin{example} \label{ex:products-and-lognormal-clt} \hlt{Products and Lognormal Shape}\\
Let $Y_n=y_0\prod\limits_{i=1}^{n}X_i$ with iid $X_i>0$, and suppose $\mu_\ell\defeq\mathbb{E}\log X_1$ and $\sigma_\ell^2\defeq\operatorname{Var}(\log X_1)\in(0,+\infty)$.\\
Since $\log Y_n=\log y_0+\sum\limits_{i=1}^{n}\log X_i$, the CLT gives $\log Y_n\approx N(\log y_0+n\mu_\ell,n\sigma_\ell^2)$, so $Y_n$ is approximately lognormal. Likewise $V_n=v_0\prod\limits_{i=1}^{n}R_i$ when iid log-returns $\log R_i$ have finite positive variance.
\end{example}

\begin{theorem} \label{thm:studentized-central-limit-theorem} \hlt{Studentized Central Limit Theorem}\\
Let $(X_i)_{i\geq1}$ be IID real-valued random variables with $\mathbb{E}X_1=\mu$ and $\operatorname{Var}(X_1)=\sigma^2\in(0,+\infty)$.\\
Set $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$, $S_n^2\defeq(n-1)^{-1}\sum\limits_{i=1}^{n}(X_i-\overline{X}_n)^2$, and $T_n\defeq S_n^{-1}\sqrt{n}(\overline{X}_n-\mu)$.
Then $T_n\rightsquigarrow N(0,1)$.
\begin{proof}
The CLT gives $Z_n\defeq\sigma^{-1}\sqrt{n}(\overline{X}_n-\mu)\rightsquigarrow N(0,1)$, and Proposition \ref{prop:sample-variance-consistency} gives $S_n^2\xrightarrow{\mathbb{P}}\sigma^2$, hence $S_n\xrightarrow{\mathbb{P}}\sigma$.\\
Thus $\frac{\sigma}{S_n}\xrightarrow{\mathbb{P}}1$, and Slutsky's rule from Theorem \ref{thm:continuous-mapping-and-slutsky} gives $T_n=Z_n(\frac{\sigma}{S_n})\rightsquigarrow N(0,1)$.\\
\hlt{Structural Commentary}: Studentization replaces the unknown $\sigma$ by the consistent estimate $S_n$.\\
\hlt{Key Insight}: Slutsky permits multiplying the CLT statistic by $\frac{\sigma}{S_n}\to1$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:berry-esseen-rate} \hlt{Berry--Esseen Rate}\\
Let $(X_i)_{i\geq1}$ be IID real-valued random variables, $\mathbb{E}X_1=\mu$, $\operatorname{Var}(X_1)=\sigma^2\in(0,+\infty)$, $\rho_3\defeq\mathbb{E}\abs{X_1-\mu}^3<+\infty$.\\
Let $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$, $Z_n\defeq\sigma^{-1}\sqrt{n}(\overline{X}_n-\mu)$, $\Phi(z)\defeq\int_{-\infty}^{z}(2\pi)^{-\frac{1}{2}}e^{-\frac{x^2}{2}}\,dx$; $\sup\limits_{z\in\mathbb{R}}\abs{\mathbb{P}(Z_n\leq z)-\Phi(z)}\leq\frac{33}{4}\frac{\rho_3}{\sigma^3\sqrt{n}}$.
\begin{proof} Set $Y_i\defeq\frac{X_i-\mu}{\sigma}$, $\rho\defeq\mathbb{E}\abs{Y_1}^3=\frac{\rho_3}{\sigma^3}$, and $W_n\defeq n^{-\frac{1}{2}}\sum\limits_{i=1}^{n}Y_i=Z_n$.\\
For $\varphi(t)\defeq\mathbb{E}e^{\mathrm{i}tY_1}$, Taylor's formula with $\mathbb{E}Y_1=0$, $\mathbb{E}Y_1^2=1$, $\mathbb{E}\abs{Y_1}^3=\rho$ gives $\varphi(t)=1-\frac{t^2}{2}+O(\rho\abs{t}^3)$ near $0$.\\
Thus $\varphi_{W_n}(t)=[\varphi(\frac{t}{\sqrt{n}})]^n$ differs from $e^{-\frac{t^2}{2}}$ by $O(\rho n^{-\frac{1}{2}})$ in Esseen's smoothing integral.\\
Standard optimised smoothing bound gives $\sup\limits_z\abs{\mathbb{P}(W_n\leq z)-\Phi(z)}\leq\frac{33}{4}\frac{\rho}{\sqrt{n}}$. Substitute $\rho=\frac{\rho_3}{\sigma^3}$, QED.\\
\hlt{Structural Commentary}: Berry--Esseen bounds the CLT error by standardised third absolute moment.\\
\hlt{Key Insight}: Normalise, compare characteristic functions, then apply Esseen smoothing.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:multivariate-central-limit-theorem} \hlt{Multivariate Central Limit Theorem}\\
Let $(X_i)_{i\geq1}$ be IID random vectors in $\mathbb{R}^k$ with $\mathbb{E}X_1=\mu\in\mathbb{R}^k$ and $\operatorname{Cov}(X_1)=\Sigma$, and set $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$.\\
Then $\sqrt{n}(\overline{X}_n-\mu)\rightsquigarrow N_k(0,\Sigma)$.
\begin{proof} For $a\in\mathbb{R}^k$, set $Y_{i,a}\defeq a^t(X_i-\mu)$ and $s_a^2\defeq a^t\Sigma a$.\\
Then $(Y_{i,a})_{i\geq1}$ are IID, $\mathbb{E}Y_{1,a}=0$, $\operatorname{Var}(Y_{1,a})=s_a^2$, and $a^t\sqrt{n}(\overline{X}_n-\mu)=n^{-\frac{1}{2}}\sum\limits_{i=1}^{n}Y_{i,a}$.\\
If $s_a^2>0$, the scalar CLT gives $a^t\sqrt{n}(\overline{X}_n-\mu)\rightsquigarrow N(0,s_a^2)$; if $s_a^2=0$, then $\mathbb{E}Y_{1,a}^2=0$, so the same convergence holds with degenerate limit $N(0,0)$. Since $a^tZ\sim N(0,s_a^2)$ for $Z\sim N_k(0,\Sigma)$, by Cramer--Wold, QED.\\
\hlt{Structural Commentary}: Multivariate convergence is checked through one-dimensional projections.\\
\hlt{Key Insight}: Project first, apply the scalar CLT, then use Cramer--Wold.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:delta-method-univariate} \hlt{Delta Method}\\
Let $(Y_n)_{n\geq1}$ be real-valued random variables, let $\mu\in\mathbb{R}$ and $\sigma>0$, and assume $\sigma^{-1}\sqrt{n}(Y_n-\mu)\rightsquigarrow N(0,1)$.\\
If $g:\mathbb{R}\to\mathbb{R}$ is differentiable at $\mu$ and $g'(\mu)\neq0$, then $(\abs{g'(\mu)}\sigma)^{-1}\sqrt{n}(g(Y_n)-g(\mu))\rightsquigarrow N(0,1)$.\\
Equivalently $g(Y_n)\approx N(g(\mu),\frac{(g'(\mu))^2\sigma^2}{n})$.
\begin{proof} Set $A_n\defeq\sigma^{-1}\sqrt{n}(Y_n-\mu)$ and define $\rho(\mu)\defeq0$, $\rho(y)\defeq\frac{g(y)-g(\mu)}{y-\mu}-g'(\mu)$ for $y\neq\mu$.\\
Then $A_n\rightsquigarrow Z\sim N(0,1)$, $A_n=O_{\mathbb{P}}(1)$, and $Y_n-\mu=\frac{\sigma A_n}{\sqrt{n}}\xrightarrow{\mathbb{P}}0$.\\
Differentiability gives $\rho(y)\to0$ as $y\to\mu$, hence $\rho(Y_n)\xrightarrow{\mathbb{P}}0$.\\
For all $n$, $g(Y_n)-g(\mu)=(g'(\mu)+\rho(Y_n))(Y_n-\mu)$. Thus, with $T_n\defeq(\abs{g'(\mu)}\sigma)^{-1}\sqrt{n}(g(Y_n)-g(\mu))$,\\
$T_n=\operatorname{sgn}(g'(\mu))A_n(1+\frac{\rho(Y_n)}{g'(\mu)})\rightsquigarrow \operatorname{sgn}(g'(\mu))Z\sim N(0,1)$ by Theorem \ref{thm:continuous-mapping-and-slutsky}.\\
\hlt{Structural Commentary}: The delta method is first-order Taylor expansion at the limit point.\\
\hlt{Key Insight}: The derivative rescales the fluctuation; the remainder is negligible in probability.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:delta-method-multivariate} \hlt{Multivariate Delta Method}\\
Let $(Y_n)_{n\geq1}$ be random vectors in $\mathbb{R}^k$ with $\sqrt{n}(Y_n-\mu)\rightsquigarrow N_k(0,\Sigma)$, and let $g:\mathbb{R}^k\to\mathbb{R}$ be differentiable at $\mu$ with gradient $\nabla g(\mu)$. Then $\sqrt{n}(g(Y_n)-g(\mu))\rightsquigarrow N(0,\nabla g(\mu)^t\Sigma\nabla g(\mu))$.
\begin{proof} Set $A_n\defeq\sqrt{n}(Y_n-\mu)$, so $A_n\rightsquigarrow N_k(0,\Sigma)$ and $Y_n\xrightarrow{\mathbb{P}}\mu$.\\
Differentiability gives $g(Y_n)-g(\mu)=\nabla g(\mu)^t(Y_n-\mu)+r(Y_n)\norm{Y_n-\mu}$ with $r(Y_n)\xrightarrow{\mathbb{P}}0$.\\
Since $A_n=O_{\mathbb{P}}(1)$, $\sqrt{n}r(Y_n)\norm{Y_n-\mu}=r(Y_n)\norm{A_n}\xrightarrow{\mathbb{P}}0$.\\
Thus $\sqrt{n}(g(Y_n)-g(\mu))=\nabla g(\mu)^tA_n+o_{\mathbb{P}}(1)\rightsquigarrow N(0,\nabla g(\mu)^t\Sigma\nabla g(\mu))$ by Theorem \ref{thm:continuous-mapping-and-slutsky}.\\
\hlt{Structural Commentary}: Multivariate delta method linearizes $g$ by its gradient.\\
\hlt{Key Insight}: The gradient projection carries the Gaussian limit; the differentiability remainder vanishes.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:delta-method} \hlt{Delta Method}\\
If $\sqrt{n}(T_n-\theta)\Rightarrow N(0,\Sigma)$ and $g:\R^d\to\R^k$ is differentiable at $\theta$ with derivative $Dg(\theta)$,\\
then $\sqrt{n}(g(T_n)-g(\theta))\Rightarrow N(0,Dg(\theta)\Sigma Dg(\theta)^\top)$.
\begin{proof}
Differentiability gives $g(T_n)-g(\theta)=Dg(\theta)(T_n-\theta)+r_n$ with $\frac{\norm{r_n}}{\norm{T_n-\theta}}\to0$ in probability.\\
Since $\sqrt{n}(T_n-\theta)=O_P(1)$, $\sqrt{n}r_n=o_P(1)$; $\sqrt{n}(g(T_n)-g(\theta))=Dg(\theta)\sqrt{n}(T_n-\theta)+o_P(1)$, by Slutsky QED.\\
\hlt{Structural Commentary}: The derivative transports the limiting normal fluctuation through $g$.\\
\hlt{Key Insight}: First-order linearisation determines asymptotic variance.
\end{proof}
\end{proposition}

\begin{example} \label{ex:delta-method-exponential-sample-mean} \hlt{Exponentiating a Sample Mean}\\
Let $(X_i)_{i=1}^{n}$ be IID with mean $\mu$ and variance $\sigma^2<+\infty$, and set $W_n\defeq e^{\overline{X}_n}$.\\
For $g(s)=e^s$, $g'(\mu)=e^\mu$, so the delta method gives $W_n\approx N(e^\mu,\frac{e^{2\mu}\sigma^2}{n})$.
\end{example}

\begin{example} \label{ex:multivariate-delta-method-product-and-ratio} \hlt{Product and Ratio of Sample Means}\\
Let $X_i=(X_{1i},X_{2i})^t$ be IID with mean $\mu=(\mu_1,\mu_2)^t$ and covariance $\Sigma=(\sigma_{ij})$, and set $\overline{X}_{jn}\defeq n^{-1}\sum\limits_{i=1}^{n}X_{ji}$.\\
For $g(x_1,x_2)=x_1x_2$, $\nabla g(\mu)=(\mu_2,\mu_1)^t$, so $\sqrt{n}(\overline{X}_{1n}\overline{X}_{2n}-\mu_1\mu_2)\rightsquigarrow N(0,\mu_2^2\sigma_{11}+2\mu_1\mu_2\sigma_{12}+\mu_1^2\sigma_{22})$.\\
If $\mu_2\neq0$, then for $h(x_1,x_2)=\frac{x_1}{x_2}$ and $a\defeq\nabla h(\mu)=(\frac{1}{\mu_2},-\frac{\mu_1}{\mu_2^2})^t$, $\sqrt{n}(\frac{\overline{X}_{1n}}{\overline{X}_{2n}}-\frac{\mu_1}{\mu_2})\rightsquigarrow N(0,a^t\Sigma a)$.
\end{example}
