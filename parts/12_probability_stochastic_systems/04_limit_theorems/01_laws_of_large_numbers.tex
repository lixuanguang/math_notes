\subsection{Laws of Large Numbers}
% Weak LLN, strong LLN, Borel-Cantelli, Kolmogorov inequalities, sums of independent random variables.

The law of large numbers formalizes the idea that a large sample average is
close to the mean of the distribution.

\begin{perspective} \label{per:sampling-distribution-of-sample-mean} \hlt{Law of the Sample Mean}\\
If $X_1,\ldots,X_n$ are IID, mean $\mu$, variance $\sigma^2$, then $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$ is a random variable with law $\mathcal{L}(\overline{X}_n)$.\\
By Theorem \ref{thm:sample-mean-and-variance-moments}, $\mathbb{E}\overline{X}_n=\mu$ and $\operatorname{Var}(\overline{X}_n)=\frac{\sigma^2}{n}$, so $\mathcal{L}(\overline{X}_n)$ concentrates around $\mu$ as $n\to\infty$.
\end{perspective}

\begin{theorem} \label{thm:weak-law-large-numbers-finite-variance} \hlt{Weak Law of Large Numbers}\\
Let $(X_i)_{i\geq1}$ be IID real-valued random variables with $\mathbb{E}X_1=\mu$, $\operatorname{Var}(X_1)=\sigma^2<+\infty$. Set $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$.\\
Then $\overline{X}_n\xrightarrow{\mathbb{P}}\mu$, equivalently $\mathbb{P}(\abs{\overline{X}_n-\mu}>\varepsilon)\to0$ for every $\varepsilon>0$.
\begin{proof} For $n\geq1$, $\mathbb{E}\overline{X}_n=n^{-1}\sum\limits_{i=1}^{n}\mathbb{E}X_i=\mu$ and, by independence, $\operatorname{Var}(\overline{X}_n)=n^{-2}\sum\limits_{i=1}^{n}\operatorname{Var}(X_i)=\frac{\sigma^2}{n}$.\\
For every $\varepsilon>0$, Chebyshev gives $\mathbb{P}(\abs{\overline{X}_n-\mu}>\varepsilon)\leq\frac{\operatorname{Var}(\overline{X}_n)}{\varepsilon^2}=\frac{\sigma^2}{n\varepsilon^2}\to0$, hence $\overline{X}_n\xrightarrow{\mathbb{P}}\mu$.\\
\hlt{Structural Commentary}: Averaging keeps the mean fixed and divides variance by $n$.\\
\hlt{Key Insight}: Chebyshev turns vanishing variance into convergence in probability.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:weak-law-independent-nonidentical} \hlt{Weak LLN with Vanishing Average Variance}\\
Let $(X_i)_{i\geq1}$ be independent, $X_i\in L^2$, with $\mu_i\defeq\mathbb{E}X_i$ and $\sigma_i^2\defeq\operatorname{Var}(X_i)$.\\
Set $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$ and $m_n\defeq n^{-1}\sum\limits_{i=1}^{n}\mu_i$. If $m_n\to\mu$ and $n^{-2}\sum\limits_{i=1}^{n}\sigma_i^2\to0$, then $\overline{X}_n\xrightarrow{\mathbb{P}}\mu$.
\begin{proof} $\mathbb{E}\overline{X}_n=m_n$. By independence, $\operatorname{Var}(\overline{X}_n)=n^{-2}\sum\limits_{i=1}^{n}\sigma_i^2\to0$. Fix $\varepsilon>0$. $\forall$ large $n$, $\abs{m_n-\mu}<\frac{\varepsilon}{2}$.\\
Hence $\{\abs{\overline{X}_n-\mu}>\varepsilon\}\subseteq\{\abs{\overline{X}_n-m_n}>\frac{\varepsilon}{2}\}$. By Chebyshev, $\mathbb{P}(\abs{\overline{X}_n-\mu}>\varepsilon)\leq\frac{4\operatorname{Var}(\overline{X}_n)}{\varepsilon^2}\to0$.\\
\hlt{Structural Commentary}: Identical distribution is not essential for a weak law.\\
\hlt{Key Insight}: It is enough that the mean target stabilises and the average variance vanishes.
\end{proof}
\end{proposition}

\begin{example} \label{ex:coin-proportion-weak-law-bound} \hlt{Coin Proportions and Chebyshev Scale}\\
Let $X_i\sim\operatorname{Bernoulli}(p)$ be IID head indicators, $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$. Then $\mathbb{E}\overline{X}_n=p$, $\operatorname{Var}(\overline{X}_n)=\frac{p(1-p)}{n}$, $\overline{X}_n\xrightarrow{\mathbb{P}}p$. \\
For $p=\frac{1}{2}$, Chebyshev gives $\mathbb{P}(0.4\leq\overline{X}_n\leq0.6)\geq1-\frac{1}{4n(0.1)^2}=1-\frac{25}{n}$, at least $0.7$ for $n\geq84$.\\
If $n=10000$ fair tosses produce $5067$ heads, then $\overline{X}_n=0.5067$, while $\operatorname{Var}(\overline{X}_n)=\frac{1}{4n}=2.5\cdot10^{-5}$ and $\operatorname{sd}(\overline{X}_n)=0.005$. The observed deviation is about $1.34$ standard deviations from $\frac{1}{2}$.
\end{example}

\begin{theorem} \label{thm:strong-law-large-numbers-integrable} \hlt{Strong Law of Large Numbers}\\
Let $(X_i)_{i\geq1}$ be IID real-valued random variables with $X_1\in L^1$, $\mu\defeq\mathbb{E}X_1$, and $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$.\\
Then $\overline{X}_n\xrightarrow{\mathrm{a.s.}}\mu$, equivalently $\mathbb{P}(\lim\limits_{n\to\infty}\overline{X}_n=\mu)=1$.
\begin{proof} After replacing $X_i$ by $X_i-\mu$, assume $\mu=0$. Set $Y_i\defeq X_i\mathbf{1}_{\{\abs{X_i}\leq i\}}$.\\
Since $\sum\limits_{i=1}^{\infty}\mathbb{P}(X_i\neq Y_i)=\sum\limits_{i=1}^{\infty}\mathbb{P}(\abs{X_1}>i)<+\infty$, Borel--Cantelli gives $X_i=Y_i$ eventually a.s.\\
Hence $n^{-1}\sum\limits_{i=1}^{n}(X_i-Y_i)\to0$ a.s. For $Z_i\defeq Y_i-\mathbb{E}Y_i$, independence gives\\
$\sum\limits_{i=1}^{\infty}\frac{\operatorname{Var}(Z_i)}{i^2}\leq\sum\limits_{i=1}^{\infty}\frac{\mathbb{E}(X_1^2\mathbf{1}_{\{\abs{X_1}\leq i\}})}{i^2}<+\infty$, by Kolmogorov's variance criterion, $n^{-1}\sum\limits_{i=1}^{n}Z_i\to0$ a.s.\\
By dominated convergence, $\mathbb{E}Y_i\to0$, hence Cesaro averaging gives $n^{-1}\sum\limits_{i=1}^{n}\mathbb{E}Y_i\to0$.\\
Combining $X_i-Y_i$, $Z_i$, and $\mathbb{E}Y_i$ gives $\overline{X}_n\to0$ a.s.; adding back $\mu$ completes the proof.\\
\hlt{Structural Commentary}: Truncate tails; centered truncations have summable variance.\\
\hlt{Key Insight}: Replace by an eventually equal truncated sequence, then average the centered part.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:strong-law-implies-weak-law} \hlt{Strong Implies Weak}\\
Let $(X_i)_{i\geq1}$ be IID real-valued random variables, $X_1\in L^1$, $\mu\defeq\mathbb{E}X_1$, $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$. Then $\overline{X}_n\xrightarrow{\mathbb{P}}\mu$.
\begin{proof} By Theorem \ref{thm:strong-law-large-numbers-integrable}, $\overline{X}_n\to\mu$ a.s. For $\varepsilon>0$, set $Y_n\defeq\mathbf{1}_{\{\abs{\overline{X}_n-\mu}>\varepsilon\}}$.\\
Then $Y_n\to0$ a.s. and $0\leq Y_n\leq1$, so dominated convergence gives $\mathbb{P}(\abs{\overline{X}_n-\mu}>\varepsilon)=\mathbb{E}Y_n\to0$.\\
\hlt{Structural Commentary}: Almost-sure convergence gives pointwise tail indicators tending to zero.\\
\hlt{Key Insight}: Dominate the indicators by $1$ and integrate.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:sample-variance-consistency} \hlt{Consistency of Empirical Moments and Variance}\\
Let $(X_i)_{i\geq1}$ be iid, $X_1\in L^2$, $\mu\defeq\mathbb{E}X_1$, and $\sigma^2\defeq\operatorname{Var}(X_1)$.\\
For $n\geq2$, set $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$ and $S_n^2\defeq(n-1)^{-1}\sum\limits_{i=1}^{n}(X_i-\overline{X}_n)^2$. Then $S_n^2\xrightarrow{\mathbb{P}}\sigma^2$.\\
More generally, if $X_1\in L^r$ for an integer $r\geq1$, then $n^{-1}\sum\limits_{i=1}^{n}X_i^r\xrightarrow{\mathbb{P}}\mathbb{E}X_1^r$.
\begin{proof}
Since $(X_i^r)_{i\geq1}$ is iid and integrable, the strong law gives $n^{-1}\sum\limits_{i=1}^{n}X_i^r\to\mathbb{E}X_1^r$ a.s., hence in probability.\\
Let $m_2\defeq\mathbb{E}X_1^2=\sigma^2+\mu^2$. The cases $r=1,2$ give $\overline{X}_n\to\mu$ and $n^{-1}\sum\limits_{i=1}^{n}X_i^2\to m_2$ a.s.\\
Since $S_n^2=\frac{n}{n-1}(n^{-1}\sum\limits_{i=1}^{n}X_i^2-\overline{X}_n^2)$, continuous mapping gives $S_n^2\to m_2-\mu^2=\sigma^2$ a.s., hence in probability.\\
\hlt{Structural Commentary}: Sample moments are averages of transformed observations.\\
\hlt{Key Insight}: Apply the strong law to $X_i^r$, then pass through the continuous variance formula.
\end{proof}
\end{proposition}

\begin{example} \label{ex:ion-channel-current-moment-estimation} \hlt{Moment Estimation for Small Open Probabilities}\\
Let $N\sim\operatorname{Binomial}(m,p)$ be the number of open channels and let $S=cN$ be total current.\\
Then $\mathbb{E}S=cmp$, $\operatorname{Var}(S)=c^2mp(1-p)$, and $\frac{\operatorname{Var}(S)}{\mathbb{E}S}=c(1-p)\approx c$ for small $p$.\\
Thus independent measurements estimate $c$ by the empirical ratio $\frac{\widehat{\operatorname{Var}}(S)}{\widehat{\mathbb{E}}S}$, without knowing $m$.
\end{example}

\begin{remark} \label{rem:cauchy-averages-and-lln-hypotheses} \hlt{Why Integrability Matters}\\
If $X_1\notin L^1$, the mean $\mathbb{E}X_1$ need not exist and sample averages need not converge to any finite deterministic limit; integrability is the condition that makes a law of large numbers meaningful.
\end{remark}
