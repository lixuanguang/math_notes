\subsection{Proof by Contradiction}

\begin{definition} \label{def:contradiction} \hlt{Contradiction}\\
A \hlt{contradiction} is a mathematical statement whose truth conditions cannot be satisfied.\\
For a mathematical statement $c$, the expression $c\land\neg c$ is the canonical contradiction form.\\
By Definition \ref{def:truth-functional-connectives}, its two conjuncts cannot be true simultaneously.
\end{definition}

\begin{proofmethod} \label{meth:proof-by-contradiction} \hlt{Proof by Contradiction}\\
Let $a$ and $b$ be mathematical statements.
To prove $a\Rightarrow b$ by classical contradiction, discharge the following obligations.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Temporary assumption}: Assume $\neg(a\Rightarrow b)$.
By Proposition \ref{prop:negating-truth-functional-connectives}, this is equivalent to assuming $a\land\neg b$.
\item \hlt{Contradiction}: Identify a mathematical statement $c$.
Derive both $c$ and $\neg c$ from the hypotheses and temporary assumption.
\item \hlt{Conclusion}: Since the temporary assumption yields a contradiction, reject it and conclude $a\Rightarrow b$.
\end{enumerate}
Justify each inference by the hypotheses, definitions, earlier results, or temporary assumption.
\end{proofmethod}
