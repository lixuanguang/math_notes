\subsection{Extensionality, Specification, and Replacement}

Variables range over sets; formulas use first-order logic with equality and the sole nonlogical symbol $\in$.\\
For $y'$ fresh in $\varphi(y)$, use $(\exists!y)\,\varphi(y)\defeq(\exists y)[\varphi(y)\land(\forall y')(\varphi(y')\Rightarrow y'=y)]$.

\begin{assumption} \label{assump:formal-axiom-of-extensionality} \hlt{Axiom of Extensionality}\\
Assume $(\forall A)(\forall B)[(\forall x)(x\in A\Longleftrightarrow x\in B)\Rightarrow A=B]$.\\
The converse is equality substitution, so this axiom is the formal source of Assumption \ref{assump:set-extensionality}.
\end{assumption}

\begin{assumption} \label{assump:formal-separation-schema} \hlt{Separation/Specification Schema}\\
Fix $n\in\N_0$, put $\mathbf{z}=(z_1,\ldots,z_n)$, and let $\varphi(x,\mathbf{z})$ have no other free variables.\\
Require $B$ not free in $\varphi$; the associated axiom is the universal closure in $A,\mathbf{z}$ of the next formula. $(\exists B)(\forall x)[x\in B\Longleftrightarrow(x\in A\land\varphi(x,\mathbf{z}))]$.\\
In set-builder notation, its witness is $B=\{x\in A:\varphi(x,\mathbf{z})\}\subseteq A$.
\end{assumption}

\begin{assumption} \label{assump:formal-replacement-schema} \hlt{Replacement Schema}\\
Fix $n\in\N_0$, put $\mathbf{z}=(z_1,\ldots,z_n)$, and let $\varphi(x,y,\mathbf{z})$ have no other free variables.\\
Require $B$ not free in $\varphi$; the associated axiom is the universal closure in $A,\mathbf{z}$ of the next implication.\\
$[(\forall x\in A)(\exists!y)\,\varphi(x,y,\mathbf{z})]\Rightarrow(\exists B)(\forall y)[y\in B\Longleftrightarrow(\exists x\in A)\,\varphi(x,y,\mathbf{z})]$.\\
The antecedent makes $\varphi$ functional on $A$; the consequent collects its full range into a set.
\end{assumption}

\begin{perspective} \label{per:separation-versus-replacement} \hlt{Selection versus Functional Image}\\
Assumption \ref{assump:formal-separation-schema} selects elements of a bounding set $A$, so its witness satisfies $B\subseteq A$.\\
Assumption \ref{assump:formal-replacement-schema} collects the range of a definable function on $A$, whose values may lie outside $A$.\\
An axiom schema comprises one first-order axiom per permitted formula and finite parameter list.
\end{perspective}
