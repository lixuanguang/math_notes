\subsection{Baire and Cantor Spaces}

\begin{definition} \label{def:baire-cantor-sequences-and-cylinders} \hlt{Sequence Spaces and Cylinders}\\
Let $A\in\{2,\omega\}$, where $2=\{0,1\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $n<\omega$, let $A^n$ be the set of functions $n\to A$, and let $A^\omega$ be the set of functions $\omega\to A$. Set $A^{<\omega}\defeq\bigcup\limits_{n<\omega}A^n$.
\item For $s\in A^{<\omega}$, its \hlt{length} is $\abs{s}\defeq\operatorname{dom}s$. For $u\in A^{<\omega}\cup A^\omega$, write $s\sqsubseteq u$ iff $s=u|_{\abs{s}}$; then $s$ is an \hlt{initial segment of $u$}.
\item Strings $s,t\in A^{<\omega}$ are \hlt{compatible} iff $s\sqsubseteq t$ or $t\sqsubseteq s$; otherwise they are \hlt{incompatible}.\\
The \hlt{basic cylinder} determined by $s$ is $N_s^A\defeq\{x\in A^\omega:s\sqsubseteq x\}$.
\item A set $U\subseteq A^\omega$ is \hlt{open} iff it is a union of basic cylinders. It is \hlt{closed} iff $A^\omega\setminus U$ is open and \hlt{clopen} iff it is both open and closed.
\item The spaces $2^\omega$ and $\omega^\omega$ are \hlt{Cantor space} and \hlt{Baire space}, respectively. Write $N_s$ for $N_s^A$ when $A$ is fixed.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:basic-cylinders-form-clopen-basis} \hlt{Basic Cylinders Form a Clopen Basis}\\
Let $A\in\{2,\omega\}$, let $s,t\in A^{<\omega}$, and let $U\subseteq A^\omega$ be open.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $s\sqsubseteq t$, then $N_s^A\cap N_t^A=N_t^A$. If $t\sqsubseteq s$, then $N_s^A\cap N_t^A=N_s^A$. If $s,t$ are incompatible, then $N_s^A\cap N_t^A=\emptyset$.
\item The cylinder $N_s^A$ is clopen.
\item $(\forall x\in U)(\exists n<\omega)\,x\in N_{x|_n}^A\subseteq U$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $s\sqsubseteq t$, then $(\forall x\in N_t^A)\,s=t|_{\abs{s}}=x|_{\abs{s}}$; hence $N_t^A\subseteq N_s^A$ and $N_s^A\cap N_t^A=N_t^A$.\\
If $t\sqsubseteq s$, then $(\forall x\in N_s^A)\,t=s|_{\abs{t}}=x|_{\abs{t}}$; hence $N_s^A\subseteq N_t^A$ and $N_s^A\cap N_t^A=N_s^A$.\\
If $x\in N_s^A\cap N_t^A$, then $s,t$ are finite initial segments of $x$ and are comparable by length.\\
Thus incompatible $s,t$ have disjoint cylinders.
\item The cylinder $N_s^A$ is open because it is a union of itself. Fix $x\notin N_s^A$. Since $s\neq x|_{\abs{s}}$, choose $j<\abs{s}$ with $x(j)\neq s(j)$.\\
If $y\in N_{x|_{j+1}}^A$, then $y(j)=x(j)\neq s(j)$. Therefore $N_{x|_{j+1}}^A\subseteq A^\omega\setminus N_s^A$.\\
The complement is a union of cylinders and is open, so $N_s^A$ is clopen.
\item Fix $x\in U$. Since $U$ is a union of cylinders, choose $r\in A^{<\omega}$ with $x\in N_r^A\subseteq U$. Then $r=x|_{\abs{r}}$, so $n\defeq\abs{r}$ has required property.
\end{enumerate}
\hlt{Structural Commentary}: Comparability controls intersections; first disagreements construct complementary cylinders.\\
\hlt{Key Insight}: Finite prefixes determine all basic neighborhoods.
\end{proof}
\end{proposition}

\begin{definition} \label{def:baire-cantor-prefix-metric-and-polish-space} \hlt{Prefix Metric and Polish Space}\\
Let $A\in\{2,\omega\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For distinct $x,y\in A^\omega$, set $\Delta_A(x,y)\defeq\min\{n<\omega:x(n)\neq y(n)\}$. Define $d_A(x,x)\defeq0$ and $d_A(x,y)\defeq2^{-(\Delta_A(x,y)+1)}$ for $x\neq y$.
\item A metric $d$ on $X$ is an \hlt{ultrametric} iff $(\forall x,y,z\in X)\,d(x,z)\leq\max\{d(x,y),d(y,z)\}$.
\item A sequence $(x_k)_{k<\omega}\in X^\omega$ is \hlt{$d$-Cauchy} iff $(\forall\varepsilon>0)(\exists K<\omega)(\forall k,\ell\geq K)\,d(x_k,x_\ell)<\varepsilon$.\\
It \hlt{converges to $x\in X$} iff $(\forall\varepsilon>0)(\exists K<\omega)(\forall k\geq K)\,d(x_k,x)<\varepsilon$.\\
The metric $d$ is \hlt{complete} iff each $d$-Cauchy sequence in $X$ converges to a point of $X$.
\item A set $D\subseteq A^\omega$ is \hlt{dense} iff $(\forall s\in A^{<\omega})\,D\cap N_s^A\neq\emptyset$.
\item A topological space is \hlt{Polish} iff its topology is induced by some complete metric and it has a countable dense subset.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:baire-and-cantor-spaces-are-polish} \hlt{Baire and Cantor Spaces Are Polish}\\
For $A\in\{2,\omega\}$, set $E_A\defeq\{x\in A^\omega:(\exists n<\omega)(\forall k\in\omega\setminus n)\,x(k)=0\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The prefix metric $d_A$ is an ultrametric.
\item The metric $d_A$ is complete.
\item The metric $d_A$ induces the cylinder topology on $A^\omega$.
\item The eventually-zero set $E_A$ is countable.
\item The set $E_A$ is dense in $A^\omega$.
\item The space $A^\omega$ is Polish.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Symmetry of the disagreement set makes $d_A$ symmetric. By definition, $(\forall x,y\in A^\omega)\,[d_A(x,y)\geq0\land(d_A(x,y)=0\Longleftrightarrow x=y)]$.\\
Fix $x,y,z\in A^\omega$ and set $r\defeq\max\{d_A(x,y),d_A(y,z)\}$. If $r=0$, then $x=y=z$. If $r>0$, choose $m<\omega$ with $r=2^{-(m+1)}$.\\
Both pairs $x,y$ and $y,z$ agree below $m$, so $d_A(x,z)\leq r$. Thus the strong triangle inequality holds.\\
The ordinary triangle inequality follows from $r\leq d_A(x,y)+d_A(y,z)$, so $d_A$ is an ultrametric.
\item Let $(x_k)_{k<\omega}$ be $d_A$-Cauchy. For $m<\omega$, choose $K_m<\omega$ such that $(\forall k,\ell\geq K_m)\,d_A(x_k,x_\ell)<2^{-m}$.\\
Set $s_m\defeq x_{K_m}|_m\in A^m$. If $k\geq K_m$, then $x_k|_m=s_m$. If $m\leq n$, comparison at an index $k\geq\max\{K_m,K_n\}$ gives $s_m=s_n|_m$.\\
Hence $x\defeq\bigcup\limits_{m<\omega}s_m$ is a function $\omega\to A$. Given $\varepsilon>0$, choose $m<\omega$ with $2^{-m}<\varepsilon$.\\
For $k\geq K_m$, the sequences $x_k,x$ agree below $m$, so $d_A(x_k,x)<2^{-m}<\varepsilon$. Thus $x_k\to x$, and $d_A$ is complete.
\item For $x\in A^\omega$ and $m<\omega$, $N_{x|_m}^A=\{y\in A^\omega:d_A(x,y)<2^{-m}\}$.\\
Both sides require agreement below $m$, so each cylinder containing $x$ is a metric ball.\\
Conversely, given $\varepsilon>0$, choose $m<\omega$ with $2^{-m}<\varepsilon$.\\
Then $N_{x|_m}^A\subseteq\{y\in A^\omega:d_A(x,y)<\varepsilon\}$, so the metric and cylinder topologies coincide.
\item The map $k\mapsto k-1$ is a bijection $\N\to\omega$, so $\omega$ is countable.\\
For $n\in\N$, the coordinate map $f\mapsto(f(0),\ldots,f(n-1))$ bijects $\omega^n$ with the $n$-tuples over $\omega$.\\
Hence $\omega^n$ is countable by Theorem \ref{thm:finite-tuples-over-countable-set-countable}. Since $A^n\subseteq\omega^n$, the set $A^n$ is at most countable by Proposition \ref{prop:subsets-and-images-at-most-countable}(i).\\
The set $A^0=\{\emptyset\}$ is finite, so $A^{<\omega}$ is at most countable by Corollary \ref{cor:at-most-countable-union-at-most-countable}.\\
For $s\in A^{<\omega}$, let $\widehat{s}\in A^\omega$ extend $s$ and equal $0$ on $\omega\setminus\abs{s}$. Then $E_A=\{\widehat{s}:s\in A^{<\omega}\}$.\\
Thus $E_A$ is at most countable by Proposition \ref{prop:subsets-and-images-at-most-countable}(ii), applied to the image of $s\mapsto\widehat{s}$.\\
For $j<\omega$, let $e_j\in A^\omega$ equal $1$ at $j$ and $0$ elsewhere.\\
Distinct indices give distinct $e_j\in E_A$, so $E_A$ is infinite and therefore countable.
\item The relation $(\forall s\in A^{<\omega})\,\widehat{s}\in E_A\cap N_s^A$ holds, so $E_A$ is dense.
\item The complete metric $d_A$ induces the cylinder topology by (ii) and (iii).\\
The set $E_A$ is a countable dense subset by (iv) and (v). Hence $A^\omega$ is Polish.
\end{enumerate}
\hlt{Structural Commentary}: Prefix stabilization proves completeness; finite supports supply a countable dense set.\\
\hlt{Key Insight}: Finite information controls both distance and density.
\end{proof}
\end{theorem}

\begin{definition} \label{def:baire-cantor-products-projections-and-continuity} \hlt{Products, Projections, and Continuity}\\
Let $A,B\in\{2,\omega\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{product topology} on $A^\omega\times B^\omega$ has basis $\{N_s^A\times N_t^B:s\in A^{<\omega},\ t\in B^{<\omega}\}$. Its open sets are the unions of these rectangles.
\item For $E\subseteq A^\omega\times B^\omega$, its first-coordinate projection is $\pi_A(E)\defeq\{x\in A^\omega:(\exists y\in B^\omega)\,(x,y)\in E\}$.
\item A function $f:A^\omega\to B^\omega$ is \hlt{continuous} iff $(\forall t\in B^{<\omega})\,f^{-1}(N_t^B)$ is open.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:continuity-by-finite-prefixes} \hlt{Continuity by Finite Prefixes}\\
Let $A,B\in\{2,\omega\}$ and let $f:A^\omega\to B^\omega$. Then $f$ is continuous iff the following finite-prefix condition holds.\\
$(\forall x\in A^\omega)(\forall m<\omega)(\exists n<\omega)(\forall y\in A^\omega)\,[y|_n=x|_n\Longrightarrow f(y)|_m=f(x)|_m]$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $f$ is continuous, and fix $x\in A^\omega$ and $m<\omega$. Set $V\defeq f^{-1}(N_{f(x)|_m}^B)$. Then $V$ is open and $x\in V$.\\
Choose $n<\omega$ with $x\in N_{x|_n}^A\subseteq V$ by Proposition \ref{prop:basic-cylinders-form-clopen-basis}(iii). If $y|_n=x|_n$, then $y\in V$ and $f(y)|_m=f(x)|_m$.
\item[$(\Leftarrow)$] Assume the prefix condition, fix $t\in B^{<\omega}$, and set $m\defeq\abs{t}$.\\
For $x\in f^{-1}(N_t^B)$, choose $n<\omega$ such that $(\forall y\in A^\omega)\,[y|_n=x|_n\Rightarrow f(y)|_m=f(x)|_m]$ by the prefix condition.\\
If $y\in N_{x|_n}^A$, then $f(y)|_m=f(x)|_m=t$, so $N_{x|_n}^A\subseteq f^{-1}(N_t^B)$.\\
Thus $f^{-1}(N_t^B)$ is a union of cylinders and is open; hence $f$ is continuous.
\end{enumerate}
\hlt{Structural Commentary}: Each direction translates target prefixes into source-cylinder neighborhoods.\\
\hlt{Key Insight}: Finite output information depends on finite input information.
\end{proof}
\end{proposition}

\begin{definition} \label{def:trees-and-branches-on-countable-alphabets} \hlt{Trees and Infinite Branches}\\
Let $A\in\{2,\omega\}$, let $T$ be a set, and let $C\subseteq A^\omega$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{tree on $A$} is a possibly empty set $T\subseteq A^{<\omega}$ satisfying the next condition; write $\operatorname{Tree}_A(T)$. $(\forall t\in T)(\forall s\in A^{<\omega})\,[s\sqsubseteq t\Longrightarrow s\in T]$.
\item The \hlt{body} of $T$ is $[T]\defeq\{x\in A^\omega:(\forall n<\omega)\,x|_n\in T\}$.
\item The \hlt{tree of initial segments} of $C$ is $T_C\defeq\{x|_n:x\in C,\ n<\omega\}$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:closed-sets-are-tree-bodies} \hlt{Closed Sets Are Exactly Tree Bodies}\\
Let $A\in\{2,\omega\}$, let $T\subseteq A^{<\omega}$, and let $C\subseteq A^\omega$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\operatorname{Tree}_A(T)$, then $[T]$ is closed.
\item $\operatorname{Tree}_A(T_C)$.
\item If $C$ is closed, then $C=[T_C]$.
\item The set $C$ is closed iff $(\exists S\subseteq A^{<\omega})\,[\operatorname{Tree}_A(S)\land C=[S]]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] Fix $x\notin[T]$ and choose $n<\omega$ with $x|_n\notin T$. The cylinder $N_{x|_n}^A$ lies in $A^\omega\setminus[T]$ because each $y\in N_{x|_n}^A$ satisfies $y|_n=x|_n\notin T$.\\
Thus the complement of $[T]$ is open.
\item[$(ii)$] By definition, $T_C\subseteq A^{<\omega}$. If $t=x|_n\in T_C$ and $s\in A^{<\omega}$ satisfies $s\sqsubseteq t$, then $s=x|_{\abs{s}}\in T_C$; hence $\operatorname{Tree}_A(T_C)$.
\item[$(iii)$] By definition, $C\subseteq[T_C]$. Fix $x\notin C$ and choose $n<\omega$ with $x\in N_{x|_n}^A\subseteq A^\omega\setminus C$ by Proposition \ref{prop:basic-cylinders-form-clopen-basis}(iii).\\
If $x|_n\in T_C$, then some $y\in C$ extends $x|_n$, contradicting $N_{x|_n}^A\cap C=\emptyset$. Hence $x\notin[T_C]$, so $[T_C]\subseteq C$.
\item[$(iv)$] ($\Rightarrow$) If $C$ is closed, then $\operatorname{Tree}_A(T_C)$ by (ii) and $C=[T_C]$ by (iii).
\item[$(iv)$] ($\Leftarrow$) If $\operatorname{Tree}_A(S)$ and $C=[S]$, then $C$ is closed by (i) with $T=S$.
\end{enumerate}
\hlt{Structural Commentary}: Complement cylinders prove closure; realized prefixes construct the representing tree.\\
\hlt{Key Insight}: Closed membership is determined by all finite approximations.
\end{proof}
\end{theorem}
