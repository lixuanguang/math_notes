\subsection{Functions, Images, and Preimages}

\begin{definition} \label{def:functions-images-preimages} \hlt{Functions, Images, and Preimages}\\
Let $f:X\to Y$ be a \hlt{function}, assigning each $x\in X$ a unique $f(x)\in Y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $X=\operatorname{dom}(f)$ is the \hlt{domain} and $Y$ is the \hlt{codomain};
\item $\operatorname{range}(f)\defeq f(X)\defeq\{f(x):x\in X\}$ is the \hlt{range} of $f$;
\item for $A\subseteq X$, $f(A)\defeq\{f(x):x\in A\}$ is the \hlt{image} of $A$;
\item for $B\subseteq Y$, $f^{-1}(B)\defeq\{x\in X:f(x)\in B\}$ is the \hlt{preimage} of $B$; no inverse function is required.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:image-preimage-calculus} \hlt{Image and Preimage Calculus}\\
Let $f:X\to Y$, let $I\neq\emptyset$, and let $A_i\subseteq X$ and $B_i\subseteq Y$ for each $i\in I$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f\left(\bigcup\limits_{i\in I}A_i\right)=\bigcup\limits_{i\in I}f(A_i)$.
\item $f\left(\bigcap\limits_{i\in I}A_i\right)\subseteq\bigcap\limits_{i\in I}f(A_i)$.
\item $f^{-1}\left(\bigcup\limits_{i\in I}B_i\right)=\bigcup\limits_{i\in I}f^{-1}(B_i)$.
\item $f^{-1}\left(\bigcap\limits_{i\in I}B_i\right)=\bigcap\limits_{i\in I}f^{-1}(B_i)$.
\item For every $B\subseteq Y$, $f^{-1}(Y\setminus B)=X\setminus f^{-1}(B)$.
\item For all $B,C\subseteq Y$, $f^{-1}(B\setminus C)=f^{-1}(B)\setminus f^{-1}(C)$.
\item For every $B\subseteq Y$, $f(f^{-1}(B))=B\cap\operatorname{range}(f)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $y\in Y$, $y\in f\left(\bigcup\limits_{i\in I}A_i\right)\Longleftrightarrow\exists i\in I,\ \exists x\in A_i,\ y=f(x)\Longleftrightarrow y\in\bigcup\limits_{i\in I}f(A_i)$.
\item If $y\in f\left(\bigcap\limits_{i\in I}A_i\right)$, then $\exists x\in\bigcap\limits_{i\in I}A_i$ such that $y=f(x)$. Hence $x\in A_i$ for every $i\in I$, so $y\in\bigcap\limits_{i\in I}f(A_i)$.
\item For $x\in X$, $x\in f^{-1}\left(\bigcup\limits_{i\in I}B_i\right)\Longleftrightarrow\exists i\in I,\ f(x)\in B_i\Longleftrightarrow x\in\bigcup\limits_{i\in I}f^{-1}(B_i)$.
\item For $x\in X$, $x\in f^{-1}\left(\bigcap\limits_{i\in I}B_i\right)\Longleftrightarrow\forall i\in I,\ f(x)\in B_i\Longleftrightarrow x\in\bigcap\limits_{i\in I}f^{-1}(B_i)$.
\item For $x\in X$, $x\in f^{-1}(Y\setminus B)\Longleftrightarrow f(x)\notin B\Longleftrightarrow x\notin f^{-1}(B)\Longleftrightarrow x\in X\setminus f^{-1}(B)$.
\item For $x\in X$, $x\in f^{-1}(B\setminus C)\Longleftrightarrow f(x)\in B,\ f(x)\notin C\Longleftrightarrow x\in f^{-1}(B)\setminus f^{-1}(C)$.
\item For $y\in Y$, $y\in f(f^{-1}(B))\Longleftrightarrow\exists x\in X,\ f(x)\in B,\ y=f(x)\Longleftrightarrow y\in B\cap\operatorname{range}(f)$.
\end{enumerate}
\hlt{Structural Commentary}: Preimages preserve Boolean operations.\\
\hlt{Key Insight}: Membership pulls back through $f$.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Support of a Function}\\
Let $f:X\to Y$. If $Y$ has distinguished $0$, $\operatorname{supp}(f)\defeq\{x\in X:f(x)\neq0\}$.
\end{definition}
