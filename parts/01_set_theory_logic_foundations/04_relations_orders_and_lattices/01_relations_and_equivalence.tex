\subsection{Relations and Equivalence}

\begin{definition} \label{def:relations-and-equivalence} \label{def:equivalence-relation} \label{def:equivalence-classes-invariants-canonical-forms} \hlt{Relations and Equivalence}\\
A \hlt{relation} from $X$ to $Y$ is a subset $R\subseteq X\times Y$. If $(x,y)\in R$, one often writes $xRy$.\\
A relation $\sim$ on $X$ is an \hlt{equivalence relation} if, for all $x,y,z\in X$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $x\sim x$;
\item $x\sim y$ implies $y\sim x$;
\item $x\sim y$ and $y\sim z$ imply $x\sim z$.
\end{enumerate}
The equivalence class of $x$ is $[x]\defeq\{y\in X:y\sim x\}$.\\
A function $f:X\to Y$ is an \hlt{invariant} of $\sim$ if $x\sim y\Rightarrow f(x)=f(y)$, and a \hlt{complete invariant} if $x\sim y\Longleftrightarrow f(x)=f(y)$.\\
A subset $\mathcal{C}\subseteq X$ is an \hlt{set of canonical forms} for $\sim$ if each equivalence class meets $\mathcal{C}$ in exactly one element.
\end{definition}
