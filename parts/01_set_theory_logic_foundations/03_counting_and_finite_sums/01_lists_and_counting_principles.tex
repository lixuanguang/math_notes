\subsection{Lists and Counting Principles}

% Source integration: attached probability chapter screenshots, Section 1.4,
% on finite sample spaces and counting methods.

Counting arguments turn structured finite choices into cardinalities. In probability, this is the bridge from a uniform finite sample space to numerical probabilities.

\begin{definition} \label{def:lists-words-and-product-sets} \hlt{Lists, Words, and Product Sets}
Let $S_1,\ldots,S_p$ be finite sets.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{list} of length $p$ with $i$th entry in $S_i$ is an element $(s_1,\ldots,s_p)\in S_1\times\cdots\times S_p$.
\item If $S_1=\cdots=S_p=S$, a list in $S^p$ may be called a \hlt{word} of length $p$ over the alphabet $S$.
\item A counting model is \hlt{ordered} when changing the order of the selected objects changes the outcome.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:multiplication-principle-finite-products} \hlt{Multiplication Principle}\\
Let $p\geq1$ and let $S_1,\ldots,S_p$ be finite sets with $\abs{S_i}=n_i$. Then $\abs{S_1\times\cdots\times S_p}=\prod_{i=1}^{p}n_i$.\\
Equivalently, a $p$-stage experiment whose $i$th stage has $n_i$ possible choices after every history has $\prod_{i=1}^{p}n_i$ ordered outcomes; in particular, two stages of sizes $m,n$ give $mn$ ordered outcomes.
\begin{proof}
The case $p=1$ is immediate.\\
For $p=2$, write $\abs{S_1}=m$ and $\abs{S_2}=n$, and choose enumerations $S_1=\{a_1,\ldots,a_m\}$ and $S_2=\{b_1,\ldots,b_n\}$.\\
The map $\{1,\ldots,m\}\times\{1,\ldots,n\}\to S_1\times S_2$, $(i,j)\mapsto(a_i,b_j)$, is a bijection, so $\abs{S_1\times S_2}=mn$.\\
For $p>2$, use induction: $\abs{S_1\times\cdots\times S_p}=\abs{(S_1\times\cdots\times S_{p-1})\times S_p}=(\prod_{i=1}^{p-1}n_i)n_p=\prod_{i=1}^{p}n_i$.\\
\hlt{Structural Commentary}: Outcomes form a finite product, or a constant-branching tree.\\
\hlt{Key Insight}: Counts multiply because every history has equally many extensions.
\end{proof}
\end{theorem}

\begin{example} \label{ex:cards-binary-and-dna-counting} \hlt{Finite Words by the Multiplication Principle}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A playing card can be described by a suit and a face value, so there are $4\cdot13=52$ suit-value combinations.
\item There are $2^8=256$ binary words of length $8$.
\item A DNA word of length $N$ over the alphabet $\{A,C,G,T\}$ has $4^N$ possible sequences.\\
A codon of length $3$ therefore has $4^3=64$ possible words.
\end{enumerate}
\hlt{Key Insight}: Independent slots multiply; word counts grow exponentially with length.
\end{example}

\begin{proposition} \label{prop:addition-subtraction-and-division-principles} \hlt{Addition, Subtraction, and Division Principles}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Addition}: if finite sets $A_1,\ldots,A_r$ are pairwise disjoint, then $\abs{\bigcup_{i=1}^{r}A_i}=\sum_{i=1}^{r}\abs{A_i}$.
\item \hlt{Subtraction}: if $B\subseteq A$ and $A$ is finite, then $\abs{A\setminus B}=\abs{A}-\abs{B}$.
\item \hlt{Division}: if $f:Y\to X$ is surjective, $X,Y$ are finite, $d\geq1$, and $\abs{f^{-1}(\{x\})}=d$ for every $x\in X$.\\
Then $\abs{Y}=d\abs{X}$ and $\abs{X}=\abs{Y}/d$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $n_i=\abs{A_i}$ and enumerate $A_i=\{a_{i,1},\ldots,a_{i,n_i}\}$. Since the $A_i$ are pairwise disjoint, $(i,k)\mapsto a_{i,k}$ is a bijection $\{(i,k):1\leq i\leq r,\ 1\leq k\leq n_i\}\to\bigcup_{i=1}^{r}A_i$, so $\abs{\bigcup_{i=1}^{r}A_i}=\sum_{i=1}^{r}n_i=\sum_{i=1}^{r}\abs{A_i}$.
\item Since $A$ is the disjoint union of $B$ and $A\setminus B$, part (i) gives $\abs{A}=\abs{B}+\abs{A\setminus B}$, so $\abs{A\setminus B}=\abs{A}-\abs{B}$.
\item For $x\neq x'$, $f^{-1}(\{x\})\cap f^{-1}(\{x'\})=\varnothing$, and $Y=\bigcup_{x\in X}f^{-1}(\{x\})$. Thus the fibers partition $Y$, so by (i), $\abs{Y}=\sum_{x\in X}\abs{f^{-1}(\{x\})}=\sum_{x\in X}d=d\abs{X}$; since $d\geq1$, $\abs{X}=\abs{Y}/d$.
\end{enumerate}
\hlt{Structural Commentary}: Count parts; quotient equal fibers.\\
\hlt{Key Insight}: Make representations disjoint, then add their fiber sizes.
\end{proof}
\end{proposition}
