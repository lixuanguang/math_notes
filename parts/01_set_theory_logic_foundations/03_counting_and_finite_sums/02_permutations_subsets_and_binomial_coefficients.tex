\begin{definition} \label{def:permutations-and-sign} \hlt{Permutations and Sign}\\
Let $n\in\N$. A permutation of $\{1,\ldots,n\}$ is a bijection from this set to itself, and $\mathfrak{S}_n$ denotes the set of all such permutations.\\
The \hlt{sign} of $\sigma\in\mathfrak{S}_n$ is $\sign(\sigma)=1$ if the number of pairs $(j,k)$ with $1\leq j<k\leq n$ and $\sigma(j)>\sigma(k)$ is even, and $\sign(\sigma)=-1$ otherwise.
\end{definition}

\begin{definition} \label{def:factorials-binomial-coefficients} \hlt{Factorials and Binomial Coefficients}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{factorial} is defined by $0!\defeq1$ and, for $n\in\N$ with $n\geq1$, $n!\defeq n(n-1)\cdots3\cdot2\cdot1$.
\item For integers $0\leq r\leq n$, the \hlt{falling factorial} is $(n)_r\defeq n(n-1)\cdots(n-r+1)$, with $(n)_0\defeq1$.
\item For integers $0\leq k\leq n$, the \hlt{binomial coefficient} is $\binom{n}{k}\defeq\frac{n!}{k!(n-k)!}$.
\item The symbol $\binom{n}{k}$ is read \hlt{$n$ choose $k$}.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:ordered-sampling-with-and-without-replacement} \hlt{Ordered Sampling With and Without Replacement}\\
Let $n,r\in\N_0$ with $r\leq n$, let $S$ be finite with $\abs{S}=n$, and set $D_r(S)\defeq\{(s_1,\ldots,s_r)\in S^r:s_i\neq s_j$ for all $1\leq i<j\leq r\}$.\\
Use $S^0=\{()\}$, with empty products and zeroth powers equal to $1$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{With replacement}: the ordered samples are $S^r$, and $\abs{S^r}=n^r$.
\item \hlt{Without replacement}: the ordered samples are $D_r(S)$, and $\abs{D_r(S)}=(n)_r$,\\
where $(n)_0=1$ and $(n)_r=n(n-1)\cdots(n-r+1)$ for $r\geq1$.
\item \hlt{Full orderings}: $\abs{D_n(S)}=n!$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $r=0$, then $\abs{S^0}=1=n^0$; if $r\geq1$, Theorem \ref{thm:multiplication-principle-finite-products} gives $\abs{S^r}=\prod_{k=1}^{r}n=n^r$.
\item For $k=1,\ldots,r$, after choosing distinct $s_1,\ldots,s_{k-1}$, the entry $s_k$ has $n-k+1$ choices,\\
so Theorem \ref{thm:multiplication-principle-finite-products} gives $\abs{D_r(S)}=\prod_{k=1}^{r}(n-k+1)=(n)_r$.
\item Taking $r=n$ in (ii) gives $\abs{D_n(S)}=(n)_n=n!$.
\end{enumerate}
\hlt{Structural Commentary}: Replacement counts all tuples; no replacement counts injective tuples.\\
\hlt{Key Insight}: Order gives slots; replacement determines whether choices stay fixed or decrease.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:unordered-samples-binomial-coefficients} \hlt{Unordered Samples Without Replacement}\\
Let $n,r\in\N_0$ with $r\leq n$, let $S$ be finite with $\abs{S}=n$, and set $\mathcal{P}_r(S)\defeq\{A\subseteq S:\abs{A}=r\}$.\\
Then $\abs{\mathcal{P}_r(S)}=\binom{n}{r}=\frac{(n)_r}{r!}=\frac{n!}{(n-r)!r!}$.
\begin{proof}
Let $\pi:D_r(S)\to\mathcal{P}_r(S)$ be $\pi(s_1,\ldots,s_r)=\{s_1,\ldots,s_r\}$.\\
This map is surjective, and for each $A\in\mathcal{P}_r(S)$, the fiber $\pi^{-1}(\{A\})$ is the set of full orderings of $A$, so Proposition \ref{prop:ordered-sampling-with-and-without-replacement} gives it cardinality $r!$.\\
The same proposition gives $\abs{D_r(S)}=(n)_r$; Proposition \ref{prop:addition-subtraction-and-division-principles} then gives $\abs{\mathcal{P}_r(S)}=(n)_r/r!$.\\
Since $(n)_r=n!/(n-r)!$, the claimed identity follows.\\
\hlt{Structural Commentary}: Unordered samples are ordered injective samples modulo reordering.\\
\hlt{Key Insight}: Count ordered first, then divide by the $r!$ orders of each subset.
\end{proof}
\end{proposition}

\begin{example} \label{ex:committee-binomial-coefficient} \hlt{Committees}\\
The number of ways to choose a committee of $3$ people from a class of $20$ people is $\binom{20}{3}=\frac{20!}{3!17!}=\frac{20\cdot19\cdot18}{3\cdot2\cdot1}=1140$.
\end{example}

\begin{proposition} \label{prop:binomial-coefficient-edge-and-symmetry} \hlt{Edge Values and Symmetry}\\
For $n,k\in\N_0$ with $k\leq n$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Edge values}: $\binom{n}{0}=\binom{n}{n}=1$.
\item \hlt{Symmetry}: $\binom{n}{k}=\binom{n}{n-k}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $0!\defeq1$, $\binom{n}{0}=\frac{n!}{0!n!}=1$ and $\binom{n}{n}=\frac{n!}{n!0!}=1$.
\item Since $0\leq n-k\leq n$, $\binom{n}{n-k}=\frac{n!}{(n-k)!(n-(n-k))!}=\frac{n!}{(n-k)!k!}=\frac{n!}{k!(n-k)!}=\binom{n}{k}$.
\end{enumerate}
\hlt{Structural Commentary}: Choosing the selected $k$ objects is equivalent to choosing the omitted $n-k$ objects.\\
\hlt{Key Insight}: Complements turn $k$-subsets into $(n-k)$-subsets.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:binomial-theorem-and-subset-count} \hlt{Binomial Theorem and Subset Count}\\
For $n\in\N$ and commuting $a,b$, $(a+b)^n=\sum_{k=0}^{n}\binom{n}{k}a^kb^{n-k}$.\\
If $S$ is finite with $\abs{S}=n$, then $\abs{\mathcal{P}(S)}=2^n=\sum_{k=0}^{n}\binom{n}{k}$.
\begin{proof} Write $(a+b)^n=\prod_{j=1}^{n}(a+b)$.\\
For each $I\subseteq\{1,\ldots,n\}$, choose $a$ from factors in $I$ and $b$ from the rest, giving $a^{\abs{I}}b^{n-\abs{I}}$.\\
Grouping by $k=\abs{I}$ gives $(a+b)^n=\sum_{k=0}^{n}\binom{n}{k}a^kb^{n-k}$.\\
For $\abs{S}=n$, the sets $\mathcal{P}_k(S)\defeq\{A\subseteq S:\abs{A}=k\}$ partition $\mathcal{P}(S)$, so Proposition \ref{prop:addition-subtraction-and-division-principles} gives $\abs{\mathcal{P}(S)}=\sum_{k=0}^{n}\abs{\mathcal{P}_k(S)}=\sum_{k=0}^{n}\binom{n}{k}$.\\
Theorem \ref{thm:multiplication-principle-finite-products} gives $\abs{\mathcal{P}(S)}=2^n$ by choosing in/out for each element.\\
\hlt{Structural Commentary}: Expansion and subset counting both group binary choices by size.\\
\hlt{Key Insight}: A term is determined by which positions choose $a$.
\end{proof}
\end{proposition}

\begin{definition} \label{def:multisets-and-multiplicity} \hlt{Multisets}\\
Let $S$ be a set. A \hlt{multiset on $S$} is a pair $(S_0,m)$ such that
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $S_0\subseteq S$ is the \hlt{underlying set};
\item $m:S_0\to\N$ is the \hlt{multiplicity function};
\item $m(s)$ is the \hlt{multiplicity} of $s\in S_0$;
\item if $S_0$ is finite, the \hlt{size} is $\lvert m\rvert\defeq\sum_{s\in S_0}m(s)$.
\end{enumerate}
Equivalently, record $(S_0,m)$ as $\{(s,m(s)):s\in S_0\}$.
\end{definition}

\begin{example} \label{ex:multiset-eigenvalue-bookkeeping} \hlt{Multiset Bookkeeping}\\
The list $(a,a,b,b,b,c)$ determines the multiset on $\{a,b,c\}$ with multiplicities $2,3,1$.
This language is useful whenever order is irrelevant but repetition matters, such as eigenvalues counted with algebraic multiplicity.
\end{example}

\begin{definition} \label{def:multinomial-coefficients} \hlt{Multinomial Coefficients}\\
Let $r\in\N$ and $n,n_1,\ldots,n_r\in\N_0$ with $n_1+\cdots+n_r=n$. Define $\binom{n}{n_1\,n_2\,\cdots\,n_r}\defeq \frac{n!}{n_1!n_2!\cdots n_r!}$.
\end{definition}

\begin{proposition} \label{prop:multinomial-coefficient-count-and-expansion} \hlt{Multinomial Count and Expansion}\\
Let $r\in\N$, let $\mathbf{n}=(n_1,\ldots,n_r)\in\N_0^r$ with $n=\sum_{i=1}^{r}n_i$, and let $x_1,\ldots,x_r$ commute.\\
Let $\mathcal{M}_{\mathbf{n}}$ be the set of tuples $(A_1,\ldots,A_r)$ of pairwise disjoint subsets of $\{1,\ldots,n\}$ whose union is $\{1,\ldots,n\}$ and which satisfy $\abs{A_i}=n_i$. Then\\
$\abs{\mathcal{M}_{\mathbf{n}}}=\binom{n}{n_1\,n_2\,\cdots\,n_r}=\frac{n!}{n_1!n_2!\cdots n_r!}$, and $(x_1+\cdots+x_r)^n=\sum_{\substack{m_1,\ldots,m_r\in\N_0\\m_1+\cdots+m_r=n}}\binom{n}{m_1\,m_2\,\cdots\,m_r}x_1^{m_1}\cdots x_r^{m_r}$.
\begin{proof}
Choose the parts successively. After $A_1,\ldots,A_{j-1}$ are chosen, there are $n-\sum_{i=1}^{j-1}n_i$ positions left, so $A_j$ has $\binom{n-\sum_{i=1}^{j-1}n_i}{n_j}$ choices. Thus $\abs{\mathcal{M}_{\mathbf{n}}}=\prod_{j=1}^{r}\binom{n-\sum_{i=1}^{j-1}n_i}{n_j}=\frac{n!}{n_1!\cdots n_r!}$. For the expansion, write each term in $\prod_{k=1}^{n}(x_1+\cdots+x_r)$ as a function $\phi:\{1,\ldots,n\}\to\{1,\ldots,r\}$, giving $\prod_{k=1}^{n}x_{\phi(k)}$. If $m_i=\abs{\phi^{-1}(\{i\})}$, then the fibers form a tuple in $\mathcal{M}_{(m_1,\ldots,m_r)}$, so the count just proved gives $\binom{n}{m_1\,m_2\,\cdots\,m_r}$ such functions; grouping by $(m_1,\ldots,m_r)$ gives the identity.\\
\hlt{Structural Commentary}: Multinomial coefficients count labeled partitions of positions.\\
\hlt{Key Insight}: Group product terms by how many times each label is chosen.
\end{proof}
\end{proposition}

\begin{example} \label{ex:multinomial-arrangements-and-pairings} \hlt{Repeated Letters, Committees, and Pairings}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The number of arrangements of the letters in the list $(A,A,G,G,G,G,C,C,C)$ is $\binom{9}{2\,4\,3}=9!/(2!4!3!)$.
\item A committee of seven split into labeled subcommittees of sizes $3,2,2$ can be done in $\binom{7}{3\,2\,2}=210$ ways.
\item If $m\in\N$ and $2m$ people are paired without assigning the pairs to labeled positions, the count is $(2m)!/(m!2^m)$.
\end{enumerate}
\end{example}
