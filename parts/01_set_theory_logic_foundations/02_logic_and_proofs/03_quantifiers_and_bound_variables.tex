\subsection{Quantifiers and Bound Variables}

\begin{definition} \label{def:bound-and-free-variable-scope} \hlt{Quantifiers, Bound Variables, and Scope}\\
Let $X$ be a set and let $P(x)$ be a condition on its elements.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Universal quantifier}: $(\forall x\in X)\,P(x)$ means $\{x\in X:\neg P(x)\}=\emptyset$.
\item \hlt{Existential quantifier}: $(\exists x\in X)\,P(x)$ means $\{x\in X:P(x)\}\neq\emptyset$.
\item \hlt{Binding and scope}: A binding declaration governs a region, called the variable's \hlt{scope}.
\item \hlt{Bound occurrence}: An occurrence governed by a binding declaration is \hlt{bound}.
\item \hlt{Free occurrence}: An occurrence outside every binding declaration is \hlt{free}.
\item \hlt{Standard binders}: Quantifiers, summation signs, integrals, and set-builder notation bind variables.
\item \hlt{Renaming}: A bound variable may be renamed consistently within its scope to a fresh symbol.\\
The statement remains unchanged provided no free occurrence is captured.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:quantifier-order-can-change-truth-value} \hlt{Quantifier Order Can Change Truth Value}\\
There exist sets $X,Y$ and a condition $P(x,y)$ on $X\times Y$ such that
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall x\in X)(\exists y\in Y)\,P(x,y)$ is true;
\item $(\exists y\in Y)(\forall x\in X)\,P(x,y)$ is false.
\end{enumerate}
Consequently, the two quantifier orders are not logically equivalent.
\begin{proof} Take $X=Y=\{0,1\}$ and let $P(x,y)$ be the condition $x=y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The choice $y=x$ verifies $(\forall x\in X)(\exists y\in Y)\,P(x,y)$.
\item If $(\exists y\in Y)(\forall x\in X)\,P(x,y)$ were true, choose its witness $y\in Y$. Then $P(0,y)\land P(1,y)$, so $y=0$ and $y=1$, contradicting $0\neq1$.
\end{enumerate} Thus (i) and (ii) hold, so the quantifier orders are not logically equivalent.\\
\hlt{Structural Commentary}: The proof separates the orders using equality on a two-point set.\\
\hlt{Key Insight}: Pointwise witnesses exist, but no single witness works uniformly.
\end{proof}
\end{proposition}

\begin{remark} \label{rem:bound-index-variables} \hlt{Index Variables}\\
Let $X$ and $I$ be sets, and let $\{A_i\}_{i\in I}$ be an indexed family in $X$. For $x\in X$, $x\in\bigcup\limits_{i\in I}A_i\Longleftrightarrow(\exists i\in I)\,x\in A_i$.\\
Likewise, $x\in\bigcap\limits_{i\in I}A_i\Longleftrightarrow(\forall i\in I)\,x\in A_i$. The indexed operator binds $i$ in both formulas.\\
The index may be renamed consistently to any fresh symbol without changing either formula.
\end{remark}
