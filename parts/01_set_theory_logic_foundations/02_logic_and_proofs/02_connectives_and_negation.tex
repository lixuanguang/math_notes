\subsection{Connectives and Negation}

\begin{definition} \label{def:truth-functional-connectives} \hlt{Truth-Functional Connectives}\\
A connective is \hlt{truth functional} when its output depends only on input truth values. Let $a$ and $b$ be mathematical statements.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Negation}: $\neg a$ is true iff $a$ is false.
\item \hlt{Conjunction}: $a\land b$ is true iff both $a$ and $b$ are true.
\item \hlt{Disjunction}: $a\lor b$ is true iff at least one of $a$ and $b$ is true; both may be true.
\item \hlt{Implication}: $a\Rightarrow b$ is false iff $a$ is true and $b$ is false. It is true in the remaining cases.
\item \hlt{Equivalence}: $a\Longleftrightarrow b$ is true iff $a$ and $b$ have the same truth value.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:negating-truth-functional-connectives} \hlt{Negating Truth-Functional Connectives}\\
For mathematical statements $a$ and $b$, the following equivalences hold.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\neg(a\land b)\Longleftrightarrow(\neg a\lor\neg b)$.
\item $\neg(a\lor b)\Longleftrightarrow(\neg a\land\neg b)$.
\item $\neg(a\Rightarrow b)\Longleftrightarrow(a\land\neg b)$.
\end{enumerate}
\begin{proof}
Expand the truth conditions in Definition \ref{def:truth-functional-connectives}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The statement $\neg(a\land b)$ is true iff $a\land b$ is false. This occurs iff at least one of $a$ and $b$ is false, precisely when $\neg a\lor\neg b$ is true.
\item The statement $\neg(a\lor b)$ is true iff $a\lor b$ is false. This occurs iff both $a$ and $b$ are false, precisely when $\neg a\land\neg b$ is true.
\item The statement $\neg(a\Rightarrow b)$ is true iff $a\Rightarrow b$ is false. This occurs iff $a$ is true and $b$ is false, precisely when $a\land\neg b$ is true.
\end{enumerate}
Thus each pair has the same truth value, proving all three equivalences.\\
\hlt{Structural Commentary}: Each item reduces both sides to one constituent truth-value condition.\\
\hlt{Key Insight}: Negation selects the cases excluded by the original connective.\\
\end{proof}
\end{proposition}
