\subsection{Schroeder--Bernstein and Cardinal Comparison}

\begin{theorem} \label{thm:schroeder-bernstein} \hlt{Schroeder--Bernstein Theorem}\\
Let $A,B$ be sets, and let $f:A\to B$ and $g:B\to A$ be injections. Then $A\sim_{\#}B$.
\begin{proof} Set $A_0\defeq A\setminus g(B)$. For $n\geq0$, define $B_n\defeq f(A_n)$ and $A_{n+1}\defeq g(B_n)$.\\
Put $U_A\defeq\bigcup\limits_{n\geq0}A_n$, $U_B\defeq\bigcup\limits_{n\geq0}B_n$, $A_*\defeq A\setminus U_A$, and $B_*\defeq B\setminus U_B$.\\
The sets $A_n$ are pairwise disjoint: if $A_i\cap A_j\neq\emptyset$ with $i<j$, then $i=0$ contradicts $A_j\subseteq g(B)$.\\
If $i>0$, injectivity of $g$ and $f$ gives $A_{i-1}\cap A_{j-1}\neq\emptyset$, and iteration reduces to the $i=0$ contradiction.\\
Hence the sets $B_n=f(A_n)$ are pairwise disjoint.\\
The restriction $g_*\defeq g|_{B_*}:B_*\to A_*$ is bijective. Indeed, $g(B_*)\subseteq A_*$; conversely, if $a\in A_*$, then $a\notin A_0$, so $a=g(b)$ for a unique $b\in B$, and $b\notin U_B$, since $b\in B_n$ would imply $a\in A_{n+1}$.\\
Define $h:A\to B$ by $h(x)\defeq f(x)$ for $x\in U_A$ and $h(x)\defeq g_*^{-1}(x)$ for $x\in A_*$. Since $A=U_A\sqcup A_*$, $B=U_B\sqcup B_*$, $f$ maps each $A_n$ bijectively onto $B_n$, and $g_*^{-1}$ maps $A_*$ bijectively onto $B_*$, the map $h$ is bijective.\\
\hlt{Structural Commentary}: Matched layers plus matched remainder.\\
\hlt{Key Insight}: Use $f$ on layers, $g_*^{-1}$ on the remainder.
\end{proof}
\end{theorem}

\begin{definition} \label{def:cardinal-comparison-and-arithmetic} \hlt{Cardinal Comparison and Arithmetic}\\
Let $A,B$ be sets, and let $\kappa=\abs{A}$, $\lambda=\abs{B}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Cardinal comparison}: $\abs{A}\leq\abs{B}$ iff $\exists$ an injection $A\to B$.
\item \hlt{Strict cardinal comparison}: $\abs{A}<\abs{B}$ iff $\abs{A}\leq\abs{B}$ and $A\not\sim_{\#}B$.
\item If $A\cap B=\emptyset$, the \hlt{cardinal sum} is $\kappa+\lambda\defeq\abs{A\cup B}$.
\item The \hlt{cardinal product} is $\kappa\lambda\defeq\abs{A\times B}$.
\item The \hlt{function set} is $A^B\defeq\{f:B\to A\}$, and the \hlt{cardinal power} is $\kappa^\lambda\defeq\abs{A^B}$.
\end{enumerate}
\end{definition}
