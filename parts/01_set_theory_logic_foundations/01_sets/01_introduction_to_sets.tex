\subsection{Introduction to Sets}

\begin{definition} \label{def:sets-and-membership} \hlt{Sets and Membership}\\
A \hlt{set} is a collection. Write $x\in E$, $x\notin E$, and $\emptyset$.
\end{definition}

\begin{remark} \label{rem:set-builder-notation} \hlt{Set-Builder Notation}\\
$\{x\in X:P(x)\}$ means all $x\in X$ satisfying $P(x)$.\\
\hlt{Key Insight}: Expand membership conditions.
\end{remark}
