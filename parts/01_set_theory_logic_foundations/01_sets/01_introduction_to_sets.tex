\subsection{Introduction to Sets}

\begin{definition} \label{def:sets-and-membership} \hlt{Sets and Membership}\\
Sets are treated as primitive objects.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Membership}: For an object $x$ and a set $E$, $x\in E$ asserts that $x$ is an element of $E$; $x\notin E$ is its negation.
\item \hlt{Empty set}: $\emptyset$ is a set with no elements.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:set-builder-notation} \hlt{Set-Builder Notation}\\
For a set $X$ and a condition $P(x)$ on $X$, $\{x\in X:P(x)\}$ consists precisely of the $x\in X$ satisfying $P(x)$.
\end{definition}
