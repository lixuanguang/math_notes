\subsection{Sequential Experiments, Filtering, and Online Inference}

Sequential methods update decisions as observations arrive.  They must account for optional stopping, adaptive sampling, and computational constraints.

\begin{definition} \label{def:sequential-test-stopping-time} \hlt{Sequential Test and Stopping Time}\\
Let $(\Omega,\mathcal{F},\mathbb{P})$ be a probability space, let $I\subseteq\mathbb{N}_0$ be a time index, let $(\mathcal{F}_t)_{t\in I}$ be a filtration, and let $(D,\mathcal{D})$ be a decision space.
For any $\tau:\Omega\to I\cup\{\infty\}$, set $\mathcal{F}_\tau\defeq\{A\in\mathcal{F}:A\cap\{\tau\leq t\}\in\mathcal{F}_t\ \forall t\in I\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Stopping time}: $\tau:\Omega\to I\cup\{\infty\}$ satisfies $\{\tau\leq t\}\in\mathcal{F}_t$ $\forall t\in I$.
\item \hlt{Sequential test}: $(\tau,d)$ with $\tau$ a stopping time and $d:(\Omega,\mathcal{F}_\tau)\to(D,\mathcal{D})$ measurable.
\end{enumerate}
\hlt{Key Insight}: Stop and decide using only pre-stopping information.
\end{definition}

\begin{definition} \label{def:sprt} \hlt{Sequential Probability Ratio Test}\\
Let $P_0$ and $P_1$ be simple iid models with one-observation densities $p_0$ and $p_1$ for $(X_t)_{t\geq1}$,\\
let $\Lambda_0\defeq1$, $\Lambda_t\defeq\prod_{i=1}^{t}p_1(X_i)/p_0(X_i)$, and choose $0<A<B<\infty$.\\
$\tau\defeq\inf\{t\geq1:\Lambda_t\in(0,A]\cup[B,\infty)\}$, with terminal decision $d_\tau=P_0$ on $\{\Lambda_\tau\leq A\}$ and $d_\tau=P_1$ on $\{\Lambda_\tau\geq B\}$.\\
\hlt{Key Insight}: First exit of $\log\Lambda_t$ from $(\log A,\log B)$ converts cumulative likelihood evidence into a stopping rule.
\end{definition}

\begin{definition} \label{def:online-estimator-recursion} \hlt{Online Estimator}\\
Let $(X_t)_{t\geq1}$ be observations in $\mathcal{X}$, let $\widehat{\theta}_0\in\Theta\subseteq\mathbb{R}^d$, let $(a_t)_{t\geq0}$ be positive step sizes, and let $H:\Theta\times\mathcal{X}\to\mathbb{R}^d$.\\
$\widehat{\theta}_{t+1}\defeq\widehat{\theta}_t+a_tH(\widehat{\theta}_t,X_{t+1})$ for $t\geq0$.\\
\hlt{Structural Commentary}: $(\widehat{\theta}_t)$ is adapted to $\mathcal{F}_t\defeq\sigma(\widehat{\theta}_0,X_1,\ldots,X_t)$.\\
\hlt{Key Insight}: Under stationarity, analyze the recursion through $h(\theta)\defeq\mathbb{E}H(\theta,X)$.
\end{definition}

\begin{definition} \label{def:filtering-problem} \hlt{Filtering Problem}\\
Let $(Z_t,Y_t)_{t\geq1}$ be a hidden-state model with state space $(E,\mathcal{E})$, and set $\mathcal{F}_t^Y\defeq\sigma(Y_1,\ldots,Y_t)$.\\
For $A\in\mathcal{E}$, $\pi_t(A)\defeq\mathbb{P}(Z_t\in A\mid\mathcal{F}_t^Y)$; the task is recursive propagation or approximation of $(\pi_t)_{t\geq1}$.\\
\hlt{Structural Commentary}: $(\pi_t)$ is an $\mathcal{F}_t^Y$-adapted probability-kernel-valued process.\\
\hlt{Key Insight}: Conditioning converts hidden dynamics into an observable belief-state recursion.
\end{definition}

\begin{remark} \label{rem:sequential-cross-references} \hlt{Sequential Cross-References}\\
Sequential inference uses filtrations and martingales from stochastic processes, stability of recursions from dynamical systems, and simulation methods from Monte Carlo computation.
\end{remark}
