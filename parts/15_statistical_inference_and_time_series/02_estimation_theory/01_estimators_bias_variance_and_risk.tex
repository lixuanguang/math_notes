\subsection{Estimators, Bias, Variance, and Risk}

An estimator is a statistic interpreted as an answer to a target question.  Its quality depends on the target, the loss function, and the model law under which performance is measured.

\begin{definition} \label{def:estimator-loss-risk} \hlt{Estimator, Loss, and Risk}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A target parameter is a map $\psi:\Theta\to\Psi$.
\item An \hlt{estimator} of $\psi(\theta)$ is a statistic $\delta(X)\in\mathcal{D}$.
\item A \hlt{loss function} is $L:\Theta\times\mathcal{D}\to[0,\infty]$.
\item The \hlt{risk} of $\delta$ is $R(\theta,\delta)\defeq\mathbb{E}_\theta L(\theta,\delta(X))$.
\item For squared-error loss $L(\theta,d)=\norm{d-\psi(\theta)}^2$, $R(\theta,\delta)$ is mean squared error.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:bias-variance-consistency} \hlt{Bias, Variance, and Consistency}\\
For real-valued $\widehat{\psi}$, $\operatorname{Bias}_\theta(\widehat{\psi})\defeq\mathbb{E}_\theta\widehat{\psi}-\psi(\theta)$, and $\operatorname{Var}_\theta(\widehat{\psi})\defeq\mathbb{E}_\theta(\widehat{\psi}-\mathbb{E}_\theta\widehat{\psi})^2$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\widehat{\psi}$ is \hlt{unbiased} if $\operatorname{Bias}_\theta(\widehat{\psi})=0$ for all $\theta$.
\item $\widehat{\psi}_n$ is \hlt{consistent} at $\theta$ if $\widehat{\psi}_n\to\psi(\theta)$ in $P_\theta$-probability.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:mse-bias-variance-decomposition} \hlt{Bias-Variance Decomposition}\\
Let $\widehat{\psi}\in L^2(P_\theta)$ be real-valued. Under squared-error loss, $\operatorname{MSE}_\theta(\widehat{\psi})\defeq\mathbb{E}_\theta(\widehat{\psi}-\psi(\theta))^2=\operatorname{Var}_\theta(\widehat{\psi})+\operatorname{Bias}_\theta(\widehat{\psi})^2$.
\begin{proof} Write $\widehat{\psi}-\psi(\theta)=(\widehat{\psi}-\mathbb{E}_\theta\widehat{\psi})+(\mathbb{E}_\theta\widehat{\psi}-\psi(\theta))$. Square, take $\mathbb{E}_\theta$ gives the identity since $\mathbb{E}_\theta(\widehat{\psi}-\mathbb{E}_\theta\widehat{\psi})=0$.\\
\hlt{Structural Commentary}: MSE splits into random fluctuation plus squared systematic error.\\
\hlt{Key Insight}: Accuracy requires controlling both variance and bias.
\end{proof}
\end{proposition}

\begin{example} \label{ex:sample-mean-estimator} \hlt{Sample Mean}\\
If $X_1,\ldots,X_n$ are independent with $\mathbb{E}X_i=\mu$ and $\operatorname{Var}(X_i)=\sigma^2<\infty$,\\
then $\overline{X}_n\defeq n^{-1}\sum\limits_{i=1}^{n}X_i$ satisfies $\mathbb{E}\overline{X}_n=\mu$, $\operatorname{Var}(\overline{X}_n)=\frac{\sigma^2}{n}$, and $\overline{X}_n\xrightarrow{P}\mu$.
\end{example}

\begin{definition} \label{def:plug-in-estimator-and-m-estimator} \hlt{Plug-In and M-Estimators}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{plug-in estimator} estimates $T(P)$ by $T(\widehat{P}_n)$, where $\widehat{P}_n$ is empirical or fitted.
\item An \hlt{$M$-estimator} satisfies $\widehat{\theta}_n\in\arg\min\limits_{\theta\in\Theta}M_n(\theta)$ for an empirical criterion $M_n$.
\item Common choices are $M_n=$ negative log-likelihood, squared loss, or absolute loss.
\end{enumerate}
\end{definition}
