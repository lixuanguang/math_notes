\subsection{Empirical Processes, Density, and Regression Smoothing}

Nonparametric statistics replaces finite-dimensional parameter assumptions by smoothness, shape, sparsity, or complexity restrictions.

\begin{definition} \label{def:empirical-measure-and-process} \hlt{Empirical Measure and Empirical Process}\\
Let $X_1,\ldots,X_n$ be observations in $(\mathcal{X},\mathcal{A})$ with reference law $P$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Empirical measure}: $\widehat{P}_n\defeq n^{-1}\sum_{i=1}^n\delta_{X_i}$, so $\widehat{P}_nf=n^{-1}\sum_{i=1}^nf(X_i)$.
\item \hlt{Empirical process}: For measurable $\mathcal{F}$, $\mathbb{G}_n(f)\defeq\sqrt{n}(\widehat{P}_nf-Pf)$ for $f\in\mathcal{F}$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:glivenko-cantelli} \hlt{Glivenko--Cantelli Theorem}\\
Let $X_1,X_2,\ldots$ be i.i.d. real variables with CDF $F$, and set $\widehat{F}_n(t)\defeq n^{-1}\sum_{i=1}^n\mathbf{1}_{\{X_i\leq t\}}$.\\
Then $\sup_{t\in\R}\abs{\widehat{F}_n(t)-F(t)}\to0$ almost surely.
\begin{proof} Add independent $R_i\sim\operatorname{Unif}[0,1]$ and set $U_i\defeq F(X_i-)+R_i\{F(X_i)-F(X_i-)\}$.\\
Then $U_i$ are i.i.d. $\operatorname{Unif}[0,1]$, and with $H_n(u)\defeq n^{-1}\sum_i\mathbf{1}_{\{U_i\leq u\}}$ one has $\widehat{F}_n(t)=H_n(F(t))$ a.s.\\
Hence $\sup_t\abs{\widehat{F}_n(t)-F(t)}\leq\sup_{0\leq u\leq1}\abs{H_n(u)-u}$.\\
Fix $k\geq1$, $u_j\defeq j/k$, let $M_{n,k}\defeq\max_{0\leq j\leq k}\abs{H_n(u_j)-u_j}$.\\
By the SLLN, $M_{n,k}\to0$ a.s. If $u_{j-1}\leq u\leq u_j$, monotonicity gives $H_n(u_{j-1})-u_j\leq H_n(u)-u\leq H_n(u_j)-u_{j-1}$,\\
Hence $\sup_{0\leq u\leq1}\abs{H_n(u)-u}\leq M_{n,k}+k^{-1}$, and $\limsup_{n\to\infty}\sup_{0\leq u\leq1}\abs{H_n(u)-u}\leq k^{-1}$ a.s. $\forall k$. Let $k\to\infty$.\\
\hlt{Structural Commentary}: Finite-grid SLLN plus monotonicity gives uniform CDF control.\\
\hlt{Key Insight}: Monotonicity bounds between-grid drift by the mesh.
\end{proof}
\end{theorem}

\begin{definition} \label{def:kernel-density-estimator} \hlt{Kernel Density Estimator}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Kernel and bandwidth}: Let $\int K(u)\,du=1$ and $h>0$.
\item \hlt{Estimator}: For data $X_1,\ldots,X_n$, $\widehat{f}_h(x)\defeq(nh)^{-1}\sum_{i=1}^nK((x-X_i)/h)$.
\item \hlt{Smoothing}: Larger $h$ lowers variance and raises bias; smaller $h$ does the reverse.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:nonparametric-regression-smoothing} \hlt{Nonparametric Regression}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Model}: $Y=m(X)+\varepsilon$ with $\mathbb{E}[\varepsilon\mid X]=0$, so $m(x)=\mathbb{E}[Y\mid X=x]$.
\item \hlt{Weights}: For kernel $K$ and bandwidth $h>0$, set $w_i(x)\defeq K((x-X_i)/h)$.
\item \hlt{Estimator}: If $\sum_iw_i(x)>0$, define $\widehat{m}_h(x)\defeq\sum_{i=1}^nw_i(x)Y_i/\sum_{i=1}^nw_i(x)$.
\end{enumerate}
\end{definition}

\begin{method} \label{meth:local-linear-regression-smoothing} \hlt{Local Linear Regression Smoothing}\\
Fix $x_0$ and bandwidth $h>0$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Weighted fit}: Set $(\widehat{\beta}_0(x_0),\widehat{\beta}_1(x_0))\in\operatorname*{argmin}_{\beta_0,\beta_1}\sum_{i=1}^n(Y_i-\beta_0-\beta_1(X_i-x_0))^2K((X_i-x_0)/h)$.
\item \hlt{Estimate}: Define $\widehat{m}_h(x_0)\defeq\widehat{\beta}_0(x_0)$.
\item \hlt{Smoothing}: Smaller $h$ gives a more variable fit; larger $h$ gives more bias.
\end{enumerate}
\end{method}

\begin{method} \label{meth:local-linear-bandwidth-cross-validation} \hlt{Bandwidth Choice by Leave-One-Out Cross-Validation}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Candidates}: Fix bandwidths $h_1,\ldots,h_M>0$.
\item \hlt{Leave-one-out score}: Let $\widehat{m}_{h_k,-i}$ be the local linear fit excluding $(X_i,Y_i)$.\\
Set $\operatorname{CV}(h_k)\defeq\sum_{i=1}^n(Y_i-\widehat{m}_{h_k,-i}(X_i))^2$.
\item \hlt{Selection}: Choose $\widehat{h}\in\operatorname*{argmin}_{1\leq k\leq M}\operatorname{CV}(h_k)$, with a visual smoothness check.
\end{enumerate}
\end{method}

\begin{remark} \label{rem:complexity-controls-nonparametric-estimation} \hlt{Complexity Control}\\
For a function class $\mathcal{F}$, estimation is stable when $\sup_{f\in\mathcal{F}}\abs{\widehat{P}_nf-Pf}$ is small with high probability or a.s.\\
Smoothness and entropy control this supremum in place of finite dimension.
\end{remark}
