\subsection{Stationary Processes and Ergodicity}
% Stationarity, ergodic theorems, mixing, spectral representation preview.

\begin{definition} \label{def:strict-and-weak-stationarity} \hlt{Strict and Weak Stationarity}\\
Let $(X_t)_{t\in\Z}$ be a real-valued stochastic process.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Strict stationarity}: $\forall k\in\N,\ t_1,\ldots,t_k,h\in\Z$, $(X_{t_1},\ldots,X_{t_k})\disteq(X_{t_1+h},\ldots,X_{t_k+h})$.
\item \hlt{Weak stationarity}: $\exists\mu\in\R$ and $\gamma:\Z\to\R$ with $\forall t,h\in\Z$, $X_t\in L^2$, $\mathbb{E}X_t=\mu$, and $\operatorname{Cov}(X_t,X_{t-h})=\gamma(h)$.
\end{enumerate}
\hlt{Key Insight}: (i) plus $L^2$ implies (ii); (ii) fixes only $\mu$ and $\gamma$, not finite-dimensional laws.
\end{definition}
