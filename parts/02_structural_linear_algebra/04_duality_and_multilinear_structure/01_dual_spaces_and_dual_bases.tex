\subsection{Dual Spaces and Dual Bases}
% **Role:** algebraic language of linear structure.
%
% Dual spaces, dual maps, annihilators, transpose as dual map, bilinear forms, tensor products as later extension.

Duality studies vectors through scalar-valued tests. Functionals, dual maps, annihilators, transposes, bilinear forms, and tensor products extend linear structure into multilinear language.

\begin{definition} \hlt{Linear Functional and Dual Space}\\
A \hlt{linear functional} on $\mathcal{V}$ is a linear map from $\mathcal{V}$ to $\mathbb{F}$. The \hlt{dual space} of $\mathcal{V}$ is $\mathcal{V}' \defeq \LL(\mathcal{V},\mathbb{F})$.
\end{definition}

\begin{example} \label{ex:evaluation-and-integral-functionals} \hlt{Evaluation and Integral Functionals}\\
The evaluation map $\operatorname{ev}_0:\mathbb{F}[x]\to\mathbb{F}$, $p\mapsto p(0)$ is a linear functional, since $\operatorname{ev}_0(\alpha p+\beta q)=\alpha p(0)+\beta q(0)$ for all $\alpha,\beta\in\mathbb{F}$ and $p,q\in\mathbb{F}[x]$.
If $C[a,b]$ is the real vector space of continuous functions on $[a,b]$, then $I:C[a,b]\to\mathbb{R}$, $I(f)\defeq\int_a^b f(x)\,dx$ is a linear functional, since $I(\alpha f+\beta g)=\alpha I(f)+\beta I(g)$ $\forall \alpha,\beta\in\mathbb{R}$ and $f,g\in C[a,b]$.
\end{example}

\begin{definition} \hlt{Dual Basis}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis of $\mathcal{V}$. The \hlt{dual basis} of $\beta$ is the list $\beta'=(\varphi_1,\ldots,\varphi_n)$ in $\mathcal{V}'$ defined by
\begin{align}
\varphi_j(\mathbf{v}_k)=
\begin{cases}
1, & k=j,\\
0, & k\neq j.
\end{cases}
\nonumber
\end{align}
Existence and uniqueness of each $\varphi_j$ follow from Proposition \ref{prop:linear-map-determined-by-basis}.
\end{definition}

\begin{theorem} \label{thm:dual-basis-and-dimension} \hlt{Dual Basis and Dimension of the Dual Space}\\
Let $\mathcal{V}$ be finite-dimensional with basis $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$, and let $\beta'=(\varphi_1,\ldots,\varphi_n)$ be the dual basis. Then $\beta'$ is a basis of $\mathcal{V}'$. In particular, $\dim(\mathcal{V}')=\dim(\mathcal{V})$.
\begin{proof}
First suppose $a_1\varphi_1+\cdots+a_n\varphi_n=0$. Evaluating at $\mathbf{v}_j$ gives $a_j=0$ for each $j$, so $\beta'$ is linearly independent. Now let $\varphi\in\mathcal{V}'$. For $\mathbf{v}=c_1\mathbf{v}_1+\cdots+c_n\mathbf{v}_n$, $\left(\sum\limits_{j=1}^n \varphi(\mathbf{v}_j)\varphi_j\right)(\mathbf{v}) =\sum\limits_{j=1}^n \varphi(\mathbf{v}_j)c_j =\varphi(\mathbf{v})$.\\
Thus $\varphi=\sum_{j=1}^n \varphi(\mathbf{v}_j)\varphi_j$, so $\beta'$ spans $\mathcal{V}'$.\\
\hlt{Structural Commentary}: A basis of $\mathcal{V}$ induces the dual coordinate system.\\
\hlt{Key Insight}: A linear functional is determined exactly by its values on a basis.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:coordinate-functionals-for-hamel-basis} \hlt{Coordinate Functionals for a Hamel Basis}\\
Let $\mathcal{B}=\{\mathbf{b}_i\}_{i\in I}$ be a Hamel basis of $\mathcal{V}$, define $\varepsilon_i\in\mathcal{V}'$ by $\varepsilon_i\left(\sum\limits_{j\in F}a_j\mathbf{b}_j\right)\defeq a_i$, $F\subseteq I$ finite, $a_i=0$ if $i\notin F$.\\
Then $\{\varepsilon_i\}_{i\in I}$ is linearly independent in $\mathcal{V}'$. Moreover, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{span}\{\varepsilon_i:i\in I\}=\mathcal{V}'$;
\item $\abs{I}<\infty$;
\item $\dim(\mathcal{V})<\infty$.
\end{enumerate}
\begin{proof}
If $\sum\limits_{r=1}^n c_r\varepsilon_{i_r}=0$ with distinct $i_r$, then $0=\left(\sum\limits_{r=1}^n c_r\varepsilon_{i_r}\right)(\mathbf{b}_{i_s})=c_s$ $\forall s$. Hence $\{\varepsilon_i\}_{i\in I}$ linearly independent.
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow$ $(ii)$. Suppose $\operatorname{span}\{\varepsilon_i:i\in I\}=\mathcal{V}'$. Put $\operatorname{supp}_{\mathcal{B}}(\psi)\defeq\{i\in I:\psi(\mathbf{b}_i)\neq0\}$.\\
If $I$ were infinite, define $\varphi\in\mathcal{V}'$ by $\varphi(\mathbf{b}_i)=1$ for all $i\in I$.\\
Then $\operatorname{supp}_{\mathcal{B}}(\varphi)=I$, while $\operatorname{supp}_{\mathcal{B}}(\psi)$ is finite for every $\psi\in\operatorname{span}\{\varepsilon_i:i\in I\}$, a contradiction.
\item[$(ii)$] $\Rightarrow$ $(iii)$. Since $\mathcal{B}$ is a basis, $\dim(\mathcal{V})=\abs{I}<\infty$.
\item[$(iii)$] $\Rightarrow$ $(i)$. If $\dim(\mathcal{V})<\infty$, then Theorem \ref{thm:dual-basis-and-dimension} gives $\operatorname{span}\{\varepsilon_i:i\in I\}=\mathcal{V}'$.
\end{enumerate}
\hlt{Structural Commentary}: Coordinate spans are finite-support.\\
\hlt{Key Insight}: Infinite-dimensional coordinate functionals are independent, not spanning.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-algebraic-dual} \hlt{Dimension of the Algebraic Dual}\\
For every $\mathbb{F}$-vector space $\mathcal{V}$, $\dim(\mathcal{V})\leq\dim(\mathcal{V}')$. Moreover, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\dim(\mathcal{V})=\dim(\mathcal{V}')$;
\item $\dim(\mathcal{V})<\infty$.
\end{enumerate}
\begin{proof}
Let $\mathcal{B}=\{\mathbf{b}_i\}_{i\in I}$ be a Hamel basis. By Proposition \ref{prop:coordinate-functionals-for-hamel-basis}, $\{\varepsilon_i\}_{i\in I}\subseteq\mathcal{V}'$ is linearly independent; hence $\dim(\mathcal{V})=\abs{I}\leq\dim(\mathcal{V}')$. For the equality criterion:
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow$ $(ii)$. Suppose $\dim(\mathcal{V})=\dim(\mathcal{V}')$. If $I$ were infinite, then $\mathcal{V}\cong(\mathbb{F}^{I})_0$ and $\mathcal{V}'\cong\mathbb{F}^{I}$.\\
Let $\mathbb{F}_0\subseteq\mathbb{F}$ be the prime subfield. Since $\mathbb{F}_0$ is countable, $\dim_{\mathbb{F}_0}\big((\mathbb{F}_0^{I})_0\big)=\abs{I}$ and $\abs{\mathbb{F}_0^{I}}>\abs{I}$ by Cantor, so $\dim_{\mathbb{F}_0}(\mathbb{F}_0^{I})>\abs{I}$; hence $\dim_{\mathbb{F}_0}\big((\mathbb{F}_0^{I})_0\big)=\abs{I}< \dim_{\mathbb{F}_0}(\mathbb{F}_0^{I})$.\\
Every $\mathbb{F}_0$-linearly independent family in $\mathbb{F}_0^{I}$ is $\mathbb{F}$-linearly independent in $\mathbb{F}^{I}$, by expanding coefficients in an $\mathbb{F}_0$-basis of $\mathbb{F}$. Thus $\dim_{\mathbb{F}}(\mathcal{V}) =\dim_{\mathbb{F}}\big((\mathbb{F}^{I})_0\big) < \dim_{\mathbb{F}}(\mathbb{F}^{I}) =\dim_{\mathbb{F}}(\mathcal{V}')$, a contradiction. Hence $\dim(\mathcal{V})<\infty$.
\item[$(ii)$] $\Rightarrow$ $(i)$. If $\dim(\mathcal{V})<\infty$, then Theorem \ref{thm:dual-basis-and-dimension} gives $\dim(\mathcal{V})=\dim(\mathcal{V}')$.
\end{enumerate}
\hlt{Structural Commentary}: Algebraic duals have arbitrary support.\\
\hlt{Key Insight}: Infinite algebraic duals are bigger than their original spaces.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:functionals-separate-points} \hlt{Functionals Separate Points}\\
Let $\mathcal{V}$ be a vector space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathbf{v}\neq\mathbf{0}$, then there exists $\varphi\in\mathcal{V}'$ such that $\varphi(\mathbf{v})\neq0$.
\item A vector $\mathbf{v}\in\mathcal{V}$ is zero if and only if $\varphi(\mathbf{v})=0$ for every $\varphi\in\mathcal{V}'$.
\item If $\mathcal{U}\leq\mathcal{V}$ and $\mathbf{x}\notin\mathcal{U}$, then there exists $\varphi\in\mathcal{V}'$ such that $\varphi|_{\mathcal{U}}=0$ and $\varphi(\mathbf{x})=1$.
\item If $\varphi\in\mathcal{V}'$ and $\varphi(\mathbf{x})\neq0$, then $\mathcal{V}=\operatorname{span}(\mathbf{x})\oplus\ker(\varphi)$.
\item If nonzero $\varphi,\psi\in\mathcal{V}'$ have the same kernel, then $\varphi=a\psi$ for some nonzero $a\in\mathbb{F}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Extend $\{\mathbf{v}\}$ to a Hamel basis $\mathcal{B}$ of $\mathcal{V}$ by Theorem \ref{thm:hamel-basis-extension}. Define $\varphi$ on $\mathcal{B}$ by $\varphi(\mathbf{v})=1$ and $\varphi(\mathbf{b})=0$ for $\mathbf{b}\in\mathcal{B}\setminus\{\mathbf{v}\}$, then extend linearly to finite basis expansions.
\item If $\mathbf{v}=\mathbf{0}$, then $\varphi(\mathbf{v})=0$ for every $\varphi\in\mathcal{V}'$. Conversely, if $\varphi(\mathbf{v})=0$ for every $\varphi\in\mathcal{V}'$ and $\mathbf{v}\neq\mathbf{0}$, then (i) gives some $\psi\in\mathcal{V}'$ with $\psi(\mathbf{v})\neq0$, a contradiction.
\item Choose a Hamel basis $\mathcal{B}_{\mathcal{U}}$ of $\mathcal{U}$. Since $\mathbf{x}\notin\mathcal{U}=\operatorname{span}(\mathcal{B}_{\mathcal{U}})$, the set $\mathcal{B}_{\mathcal{U}}\cup\{\mathbf{x}\}$ is linearly independent. Extend it to a Hamel basis $\mathcal{B}$ of $\mathcal{V}$ by Theorem \ref{thm:hamel-basis-extension}. Define $\varphi$ on $\mathcal{B}$ by $\varphi(\mathbf{x})=1$ and $\varphi(\mathbf{b})=0$ for every $\mathbf{b}\in\mathcal{B}\setminus\{\mathbf{x}\}$, then extend linearly to finite basis expansions.
\item Since $\varphi(\mathbf{x})\neq0$, $\operatorname{span}(\mathbf{x})\cap\ker(\varphi)=\{\mathbf{0}\}$.\\
For any $\mathbf{v}\in\mathcal{V}$, $\mathbf{v}=\frac{\varphi(\mathbf{v})}{\varphi(\mathbf{x})}\mathbf{x}
       +\left(\mathbf{v}-\frac{\varphi(\mathbf{v})}{\varphi(\mathbf{x})}\mathbf{x}\right)$, and the second summand lies in $\ker(\varphi)$.
\item Let $\mathcal{K}=\ker(\varphi)=\ker(\psi)$ and choose $\mathbf{x}\notin\mathcal{K}$. By (iv), every $\mathbf{v}\in\mathcal{V}$ can be written $\mathbf{v}=c\mathbf{x}+\mathbf{k}$ with $\mathbf{k}\in\mathcal{K}$. Setting $a=\varphi(\mathbf{x})/\psi(\mathbf{x})$ gives $\varphi(\mathbf{v})=c\varphi(\mathbf{x})=a\,c\psi(\mathbf{x})=a\psi(\mathbf{v})$.
\end{enumerate}
\hlt{Structural Commentary}: Dual vectors separate points and define hyperplanes.\\
\hlt{Key Insight}: A nonzero functional splits the space into one visible direction plus its kernel.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:nonzero-functional-hyperplane-kernel} \hlt{Kernels of Nonzero Functionals}\\
If $0\neq\varphi\in\mathcal{V}'$, then $\varphi$ is surjective onto $\mathbb{F}$, $\mathcal{V}/\ker(\varphi)\cong\mathbb{F}$, and $\operatorname{codim}_{\mathcal{V}}(\ker\varphi)=1$.
\begin{proof}
Choose $\mathbf{x}$ with $\varphi(\mathbf{x})\neq0$. For each $a\in\mathbb{F}$, $\varphi\bigl(a\varphi(\mathbf{x})^{-1}\mathbf{x}\bigr)=a$, so $\operatorname{im}\varphi=\mathbb{F}$.\\
Hence, by Theorem \ref{thm:quotient-by-kernel-is-range}, $\mathcal{V}/\ker(\varphi)\cong\operatorname{im}\varphi=\mathbb{F}$ and $\operatorname{codim}_{\mathcal{V}}(\ker\varphi)=\dim(\mathcal{V}/\ker\varphi)=1$.\\
\hlt{Structural Commentary}: A nonzero functional sees exactly one scalar direction and kills a hyperplane.\\
\hlt{Key Insight}: Kernels of nonzero functionals have codimension one.
\end{proof}
\end{corollary}

\begin{definition} \label{def:algebraic-double-dual-and-natural-map} \hlt{Algebraic Double Dual and Natural Map}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{algebraic double dual} of $\mathcal{V}$ is $\mathcal{V}''\defeq(\mathcal{V}')'$.
\item The \hlt{natural evaluation map} is $\Lambda_{\mathcal{V}}:\mathcal{V}\to\mathcal{V}''$; $\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)=\varphi(\mathbf{v})$.
\item $\mathcal{V}$ is \hlt{algebraically reflexive} if $\Lambda_{\mathcal{V}}$ is surjective.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:algebraic-reflexivity-natural-map} \hlt{Algebraic Reflexivity}\\
Natural map $\Lambda_{\mathcal{V}}\in\LL(\mathcal{V},\mathcal{V}'')$ is linear and injective. It is an isomorphism if and only if $\mathcal{V}$ is finite-dimensional.
\begin{proof}
For $\mathbf{v}\in\mathcal{V}$, $\Lambda_{\mathcal{V}}(\mathbf{v}):\varphi\mapsto\varphi(\mathbf{v})$ is linear on $\mathcal{V}'$.\\
For $a,b\in\mathbb{F}$ and $\mathbf{u},\mathbf{v}\in\mathcal{V}$, $\Lambda_{\mathcal{V}}(a\mathbf{u}+b\mathbf{v})(\varphi) =a\Lambda_{\mathcal{V}}(\mathbf{u})(\varphi)+b\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)$, $\varphi\in\mathcal{V}'$, so $\Lambda_{\mathcal{V}}\in\LL(\mathcal{V},\mathcal{V}'')$.\\
If $\mathbf{v}\in\ker(\Lambda_{\mathcal{V}})$, then $\varphi(\mathbf{v})=0$ for all $\varphi\in\mathcal{V}'$, hence $\mathbf{v}=\mathbf{0}$ by Proposition \ref{prop:functionals-separate-points}. Thus $\ker(\Lambda_{\mathcal{V}})=\{\mathbf{0}\}$.\\
For the isomorphism criterion:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\Lambda_{\mathcal{V}}$ is an isomorphism and $\dim(\mathcal{V})=\infty$, then Theorem \ref{thm:dimension-of-algebraic-dual} gives $\dim(\mathcal{V}'')>\dim(\mathcal{V}')>\dim(\mathcal{V})$, contradicting $\mathcal{V}\cong\mathcal{V}''$. Hence $\dim(\mathcal{V})<\infty$.
\item[$(\Leftarrow)$] If $\dim(\mathcal{V})<\infty$, then $\dim(\mathcal{V}'')=\dim(\mathcal{V}')=\dim(\mathcal{V})$ by Theorem \ref{thm:dual-basis-and-dimension}.\\
Since $\Lambda_{\mathcal{V}}$ is injective, it is an isomorphism.
\end{enumerate}
\hlt{Structural Commentary}: $\Lambda_{\mathcal{V}}$ embeds vectors as evaluation functionals.\\
\hlt{Key Insight}: Finite-dimensional spaces are exactly the algebraically reflexive vector spaces.
\end{proof}
\end{theorem}

\begin{definition} \label{def:dual-map} \hlt{Dual Map}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$, the \hlt{dual map} of $T$ is the map $T':\mathcal{W}'\to\mathcal{V}'$ defined by $T'\varphi \defeq \varphi T$ for every $\varphi\in\mathcal{W}'$.\\
Equivalently, $(T'\varphi)(\mathbf{v})=\varphi(T\mathbf{v})$.
\end{definition}

\begin{proposition} \label{prop:dual-map-is-linear} \hlt{Dual Maps Are Linear}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$, then $T'\in\LL(\mathcal{W}',\mathcal{V}')$.
\begin{proof}
For $\varphi,\psi\in\mathcal{W}'$, $a\in\mathbb{F}$, and $\mathbf{v}\in\mathcal{V}$, $\left(T'(a\varphi+\psi)\right)(\mathbf{v}) =(a\varphi+\psi)(T\mathbf{v}) =a\varphi(T\mathbf{v})+\psi(T\mathbf{v}) =\left(aT'\varphi+T'\psi\right)(\mathbf{v})$.\\
\hlt{Structural Commentary}: The dual map pulls functionals back along $T$ and preserves linear structure.\\
\hlt{Key Insight}: Dual maps reverse direction because a functional can test $T\mathbf{v}$ only after $T$ is applied.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:algebra-of-dual-maps} \hlt{Algebra of Dual Maps}\\
The dual operation has the following properties:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $S,T\in\LL(\mathcal{V},\mathcal{W})$ and $a\in\mathbb{F}$, then $(aS+T)'=aS'+T'$.
\item If $T\in\LL(\mathcal{U},\mathcal{V})$ and $S\in\LL(\mathcal{V},\mathcal{W})$, then $(ST)'=T'S'$.
\item $(I_{\mathcal{V}})'=I_{\mathcal{V}'}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\varphi\in\mathcal{W}'$ and $\mathbf{v}\in\mathcal{V}$, $\left((aS+T)'\varphi\right)(\mathbf{v}) =\varphi(aS\mathbf{v}+T\mathbf{v}) =a(S'\varphi)(\mathbf{v})+(T'\varphi)(\mathbf{v})$, so $(aS+T)'=aS'+T'$.
\item For $\varphi\in\mathcal{W}'$ and $\mathbf{u}\in\mathcal{U}$, $\left((ST)'\varphi\right)(\mathbf{u}) =\varphi(S(T\mathbf{u})) =\left(T'(S'\varphi)\right)(\mathbf{u})$, so $(ST)'=T'S'$.
\item For $\varphi\in\mathcal{V}'$ and $\mathbf{v}\in\mathcal{V}$, $(I_{\mathcal{V}}'\varphi)(\mathbf{v})=\varphi(I_{\mathcal{V}}\mathbf{v})=\varphi(\mathbf{v})$, so $(I_{\mathcal{V}})'=I_{\mathcal{V}'}$.
\end{enumerate}
\hlt{Structural Commentary}: Duality is linear on maps and contravariant under composition.\\
\hlt{Key Insight}: Pullbacks compose in the opposite order.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:double-dual-map-identifies-with-original-map} \hlt{Double Dual Maps Identify with Original Maps}\\
Let $\dim\mathcal{V},\dim\mathcal{W}<\infty$ and $T\in\LL(\mathcal{V},\mathcal{W})$.\\
Under $\Lambda_{\mathcal{V}}:\mathcal{V}\cong\mathcal{V}''$ and $\Lambda_{\mathcal{W}}:\mathcal{W}\cong\mathcal{W}''$, $T''\Lambda_{\mathcal{V}}=\Lambda_{\mathcal{W}}T$,  so $T''$ identifies with $T$.
\begin{proof}
For $\mathbf{v}\in\mathcal{V}$ and $\psi\in\mathcal{W}'$, $\bigl(T''\Lambda_{\mathcal{V}}(\mathbf{v})\bigr)(\psi) =\Lambda_{\mathcal{V}}(\mathbf{v})(T'\psi) =(T'\psi)(\mathbf{v}) =\psi(T\mathbf{v}) =\Lambda_{\mathcal{W}}(T\mathbf{v})(\psi)$.\\
Hence $T''\Lambda_{\mathcal{V}}=\Lambda_{\mathcal{W}}T$.\\
\hlt{Structural Commentary}: Double dualisation commutes with natural evaluation.\\
\hlt{Key Insight}: The double dual square commutes because both sides evaluate every functional the same way.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:dual-map-inverse} \hlt{Dual Map of an Inverse}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$ is an isomorphism, then $T'$ is an isomorphism and $(T^{-1})'=(T')^{-1}$
\begin{proof}
Since $T^{-1}\in\LL(\mathcal{W},\mathcal{V})$, Theorem \ref{thm:algebra-of-dual-maps} gives $T'(T^{-1})'=(T^{-1}T)'=(I_{\mathcal{V}})'=I_{\mathcal{V}'}$\\
and $(T^{-1})'T'=(TT^{-1})'=(I_{\mathcal{W}})'=I_{\mathcal{W}'}$. Thus $(T^{-1})'$ is inverse of $T'$, $T'$ is isomorphism and $(T^{-1})'=(T')^{-1}$.\\
\hlt{Structural Commentary}: Duality preserves inverses contravariantly.\\
\hlt{Key Insight}: Invertibility survives duality.
\end{proof}
\end{corollary}
