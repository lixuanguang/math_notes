\subsection{Annihilators and Transposes}
\begin{definition} \hlt{Annihilator}\\
If $\mathcal{U}\subseteq\mathcal{V}$, the \hlt{annihilator} of $\mathcal{U}$ is $\mathcal{U}^0\defeq\{\varphi\in\mathcal{V}':\varphi(\mathbf{u})=0 \text{ for every } \mathbf{u}\in\mathcal{U}\}$.
\end{definition}

\begin{proposition} \label{prop:annihilator-is-subspace} \hlt{Annihilators Are Subspaces}\\
If $\mathcal{U}\subseteq\mathcal{V}$, then $\mathcal{U}^0$ is a subspace of $\mathcal{V}'$.
\begin{proof}
The zero functional lies in $\mathcal{U}^0$. If $\varphi,\psi\in\mathcal{U}^0$, $a\in\mathbb{F}$, and $\mathbf{u}\in\mathcal{U}$, then $(a\varphi+\psi)(\mathbf{u})=a\varphi(\mathbf{u})+\psi(\mathbf{u})=0$.\\
Thus $a\varphi+\psi\in\mathcal{U}^0$, and Theorem \ref{thm:subspacetest} shows that $\mathcal{U}^0$ is a subspace of $\mathcal{V}'$.\\
\hlt{Structural Commentary}: The annihilator collects all linear tests that vanish on a chosen set. Closure follows because vanishing is preserved by linear combinations of tests.\\
\hlt{Key Insight}: Annihilators are kernels of evaluation constraints.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-annihilator} \hlt{Dimension of the Annihilator}\\
Let $\mathcal{V}$ be finite-dimensional and let $\mathcal{U}$ be a subspace of $\mathcal{V}$. Then $\dim(\mathcal{U})+\dim(\mathcal{U}^0)=\dim(\mathcal{V})$.
\begin{proof}
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ of $\mathcal{U}$ and extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_k)$ of $\mathcal{V}$.\\
Let $(\varphi_1,\ldots,\varphi_m,\psi_1,\ldots,\psi_k)$ be the corresponding dual basis.\\
By Theorem \ref{thm:dual-basis-and-dimension}, every $\varphi\in\mathcal{V}'$ has the expansion $\varphi=\sum\limits_{j=1}^m \varphi(\mathbf{u}_j)\varphi_j+\sum\limits_{\ell=1}^k \varphi(\mathbf{w}_\ell)\psi_\ell$.\\
If $\varphi\in\mathcal{U}^0$, then $\varphi(\mathbf{u}_j)=0$ for every $j$, so $\varphi\in\operatorname{span}(\psi_1,\ldots,\psi_k)$.\\
Conversely, each $\psi_\ell$ vanishes on every $\mathbf{u}_j$, hence on $\mathcal{U}=\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_m)$. Therefore $\mathcal{U}^0=\operatorname{span}(\psi_1,\ldots,\psi_k)$.\\
The list $(\psi_1,\ldots,\psi_k)$ is linearly independent as a sublist of the dual basis, so $\dim(\mathcal{U}^0)=k=\dim(\mathcal{V})-\dim(\mathcal{U})$.\\
\hlt{Structural Commentary}: Extending a basis of $\mathcal{U}$ separates directions inside $\mathcal{U}$ from complementary directions. The annihilator is dual to the complementary part.\\
\hlt{Key Insight}: To vanish on $\mathcal{U}$, a functional can only use dual coordinates attached to directions outside $\mathcal{U}$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:annihilator-lattice-identities} \hlt{Annihilator Lattice Identities}\\
Let $\mathcal{V}$ be finite-dimensional, and let $\mathcal{S},\mathcal{T}$ be subspaces of $\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathcal{S}\subseteq\mathcal{T}$, then $\mathcal{T}^0\subseteq\mathcal{S}^0$.
\item $(\mathcal{S}+\mathcal{T})^0=\mathcal{S}^0\cap\mathcal{T}^0$.
\item $(\mathcal{S}\cap\mathcal{T})^0=\mathcal{S}^0+\mathcal{T}^0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\varphi\in\mathcal{T}^0$ and $\mathbf{s}\in\mathcal{S}$, then $\mathbf{s}\in\mathcal{T}$, so $\varphi(\mathbf{s})=0$. Hence $\varphi\in\mathcal{S}^0$.
\item For $\varphi\in\mathcal{V}'$, $\varphi\in(\mathcal{S}+\mathcal{T})^0$ if and only if $\varphi(\mathbf{s}+\mathbf{t})=0$ for all $\mathbf{s}\in\mathcal{S}$ and $\mathbf{t}\in\mathcal{T}$, if and only if $\varphi\in\mathcal{S}^0\cap\mathcal{T}^0$.
\item Since $\mathcal{S}\cap\mathcal{T}\subseteq\mathcal{S},\mathcal{T}$, part (i) gives $\mathcal{S}^0+\mathcal{T}^0\subseteq(\mathcal{S}\cap\mathcal{T})^0$. By part (ii), Theorem \ref{thm:dimension-formula-for-sum}, and Theorem \ref{thm:dimension-of-annihilator},\\
$\dim(\mathcal{S}^0+\mathcal{T}^0) =\dim(\mathcal{S}^0)+\dim(\mathcal{T}^0)-\dim(\mathcal{S}^0\cap\mathcal{T}^0) =\dim(\mathcal{V})-\dim(\mathcal{S}\cap\mathcal{T}) =\dim\big((\mathcal{S}\cap\mathcal{T})^0\big)$.\\
The inclusion is therefore equality.
\end{enumerate}
\hlt{Structural Commentary}: Annihilation is an order-reversing lattice duality.\\
\hlt{Key Insight}: Vanishing on more vectors gives fewer functionals.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:double-annihilator-finite-dimensional} \hlt{Double Annihilator}\\
Let $\dim\mathcal{V}<\infty$ and $\emptyset\neq M\subseteq\mathcal{V}$. Under $\Lambda_{\mathcal{V}}:\mathcal{V}\cong\mathcal{V}''$, $M^{00}=\Lambda_{\mathcal{V}}(\operatorname{span}M)$.\\
In particular, if $\mathcal{S}\leq\mathcal{V}$, then $\mathcal{S}^{00}=\Lambda_{\mathcal{V}}(\mathcal{S})\cong\mathcal{S}$.
\begin{proof}
If $\mathbf{v}\in\operatorname{span}M$ and $\varphi\in M^0$, then $\varphi(\mathbf{v})=0$, so $\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)=0$. Hence $\Lambda_{\mathcal{V}}(\operatorname{span}M)\subseteq M^{00}$.\\
Conversely, let $\eta\in M^{00}$. By Theorem \ref{thm:algebraic-reflexivity-natural-map}, $\eta=\Lambda_{\mathcal{V}}(\mathbf{v})$ for a unique $\mathbf{v}\in\mathcal{V}$.\\
If $\mathbf{v}\notin\operatorname{span}M$, Proposition \ref{prop:functionals-separate-points}(iii) gives $\varphi\in\mathcal{V}'$ with $\varphi|_{\operatorname{span}M}=0$ and $\varphi(\mathbf{v})=1$.\\
Then $\varphi\in M^0$ but $\eta(\varphi)=\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)=1$, contradicting $\eta\in M^{00}$. Thus $\mathbf{v}\in\operatorname{span}M$, so $M^{00}\subseteq\Lambda_{\mathcal{V}}(\operatorname{span}M)$.\\
\hlt{Structural Commentary}: Double annihilation returns the span under $\Lambda_{\mathcal{V}}$.\\
\hlt{Key Insight}: Linear functionals separate any vector outside a finite-dimensional span.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:direct-sum-dual-annihilator} \hlt{Direct Sums and Annihilators}\\
Let $\mathcal{V}$ be finite-dimensional, and suppose $\mathcal{V}=\mathcal{S}\oplus\mathcal{T}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Extension by zero on $\mathcal{S}$ gives an isomorphism $\mathcal{T}'\cong\mathcal{S}^0$.
\item $\mathcal{V}'=\mathcal{S}^0\oplus\mathcal{T}^0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Define $E:\mathcal{T}'\to\mathcal{S}^0$ by $(E\varphi)(\mathbf{s}+\mathbf{t})=\varphi(\mathbf{t})$. Since $\mathcal{V}=\mathcal{S}\oplus\mathcal{T}$, this is well-defined and linear. Restriction $R:\mathcal{S}^0\to\mathcal{T}'$, $R\psi=\psi|_{\mathcal{T}}$, satisfies $RE=I_{\mathcal{T}'}$ and $ER=I_{\mathcal{S}^0}$, so $E$ is an isomorphism.
\item By Theorem \ref{thm:annihilator-lattice-identities}, $\mathcal{S}^0+\mathcal{T}^0=(\mathcal{S}\cap\mathcal{T})^0=\{\mathbf{0}\}^0=\mathcal{V}'$, and $\mathcal{S}^0\cap\mathcal{T}^0=(\mathcal{S}+\mathcal{T})^0=\mathcal{V}^0=\{0\}$.\\
Thus $\mathcal{V}'=\mathcal{S}^0\oplus\mathcal{T}^0$.
\end{enumerate}
\hlt{Structural Commentary}: A direct sum decomposes the dual by complementary annihilators.\\
\hlt{Key Insight}: Direct sums split the dual space by annihilators.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:dual-kernel-range-annihilator} \hlt{Kernel and Range of the Dual Map}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\ker(T')=(\operatorname{range}(T))^0$;
\item $\dim(\ker(T'))=\dim(\ker(T))+\dim(\mathcal{W})-\dim(\mathcal{V})$;
\item $\operatorname{range}(T')=(\ker(T))^0$;
\item $\dim(\operatorname{range}(T'))=\dim(\operatorname{range}(T))$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\varphi\in\mathcal{W}'$, $T'\varphi=0$ if and only if $\varphi(T\mathbf{v})=0$ for every $\mathbf{v}\in\mathcal{V}$, if and only if $\varphi$ vanishes on $\operatorname{range}(T)$.
\item By (i), Theorem \ref{thm:dimension-of-annihilator}, and rank-nullity,\\
$\dim(\ker(T'))=\dim(\mathcal{W})-\dim(\operatorname{range}(T)) =\dim(\ker(T))+\dim(\mathcal{W})-\dim(\mathcal{V})$.
\item If $\varphi=T'\psi$ and $\mathbf{v}\in\ker(T)$, then $\varphi(\mathbf{v})=\psi(T\mathbf{v})=0$, so $\operatorname{range}(T')\subseteq(\ker(T))^0$. Also, by rank-nullity for $T'$, (ii), Theorem \ref{thm:dual-basis-and-dimension}, and Theorem \ref{thm:dimension-of-annihilator}, $\dim(\operatorname{range}(T'))=\dim(\mathcal{V})-\dim(\ker(T)) =\dim\big((\ker(T))^0\big)$.\\
Thus $\operatorname{range}(T')=(\ker(T))^0$.
\item By (iii), Theorem \ref{thm:dimension-of-annihilator}, and rank-nullity, $\dim(\operatorname{range}(T'))=\dim(\mathcal{V})-\dim(\ker(T)) =\dim(\operatorname{range}(T))$
\end{enumerate}
\hlt{Structural Commentary}: Duality exchanges kernels and ranges through annihilators.\\
\hlt{Key Insight}: A functional dies after pulling back by $T$ exactly when it vanishes on everything $T$ can reach.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:surjective-iff-dual-injective} \hlt{Surjectivity and Dual Injectivity}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then $T$ is surjective if and only if $T'$ is injective.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $T$ is surjective. Then $\operatorname{range}(T)=\mathcal{W}$, so Theorem \ref{thm:dual-kernel-range-annihilator} gives $\ker(T')=\mathcal{W}^0=\{0\}$.
\item[$(\Leftarrow)$] Suppose $T'$ is injective. Then $\ker(T')=\{0\}$, so Theorem \ref{thm:dual-kernel-range-annihilator} gives $(\operatorname{range}(T))^0=\{0\}$. By Theorem \ref{thm:dimension-of-annihilator}, $\dim(\operatorname{range}(T))=\dim(\mathcal{W})$, hence $\operatorname{range}(T)=\mathcal{W}$.
\end{enumerate}
\hlt{Structural Commentary}: A map is onto exactly when no nonzero functional can vanish on its range. Dual injectivity detects this absence of invisible codomain directions.\\
\hlt{Key Insight}: Surjectivity of $T$ becomes injectivity of the pullback map $T'$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:injective-iff-dual-surjective} \hlt{Injectivity and Dual Surjectivity}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then $T$ is injective if and only if $T'$ is surjective.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $T$ is injective. Then $\ker(T)=\{\mathbf{0}\}$, so Theorem \ref{thm:dual-kernel-range-annihilator} gives $\operatorname{range}(T')=(\ker(T))^0=\mathcal{V}'$.
\item[$(\Leftarrow)$] Suppose $T'$ is surjective. Then $\operatorname{range}(T')=\mathcal{V}'$, so Theorem \ref{thm:dual-kernel-range-annihilator} gives $(\ker(T))^0=\mathcal{V}'$. By Theorem \ref{thm:dimension-of-annihilator}, this forces $\dim(\ker(T))=0$, hence $\ker(T)=\{\mathbf{0}\}$.
\end{enumerate}
\hlt{Structural Commentary}: Injectivity says there are no nonzero domain directions killed by $T$. The dual statement says every test on the domain comes by pulling back a test from the codomain.\\
\hlt{Key Insight}: No kernel directions means no obstruction to extending functionals backward through $T$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:quotient-dual-annihilator-isomorphism} \hlt{Dual of a Quotient as an Annihilator}\\
Let $\mathcal{V}$ be finite-dimensional, let $\mathcal{U}\leq\mathcal{V}$, and let $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ be the quotient map.\\
Then $\ker(\pi')=\{0\}$ and $\operatorname{range}(\pi')=\mathcal{U}^0$. Hence $\pi'$ identifies $(\mathcal{V}/\mathcal{U})'$ with $\mathcal{U}^0$, so $(\mathcal{V}/\mathcal{U})'\cong\mathcal{U}^0$.
\begin{proof}
By Theorem \ref{thm:dimension-of-quotient}, $\mathcal{V}/\mathcal{U}$ is finite-dimensional. Since $\pi$ is surjective and $\ker(\pi)=\mathcal{U}$, Corollary \ref{cor:surjective-iff-dual-injective} gives $\ker(\pi')=\{0\}$, and Theorem \ref{thm:dual-kernel-range-annihilator} gives $\operatorname{range}(\pi')=(\ker(\pi))^0=\mathcal{U}^0$.\\
\hlt{Structural Commentary}: A functional on the quotient is exactly a functional on $\mathcal{V}$ that cannot see $\mathcal{U}$.\\
\hlt{Key Insight}: Pulling back along the quotient map identifies $(\mathcal{V}/\mathcal{U})'$ with $\mathcal{U}^0$.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:range-membership-dual-alternative} \hlt{Range Membership Dual Alternative}\\
Let $\mathcal{V},\mathcal{W}$ be finite-dimensional, $T\in\LL(\mathcal{V},\mathcal{W})$, and $\mathbf{w}\in\mathcal{W}$. Exactly one holds:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathbf{w}\in\operatorname{range}(T)$;
\item there exists $\psi\in\mathcal{W}'$ such that $\psi T=0$ and $\psi(\mathbf{w})\neq0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The alternatives are incompatible: if $\mathbf{w}=T\mathbf{v}$ and $\psi T=0$, then $\psi(\mathbf{w})=\psi(T\mathbf{v})=0$.
\item Suppose $\mathbf{w}\notin\operatorname{range}(T)$. By Proposition \ref{prop:functionals-separate-points}(iii), there exists $\psi\in\mathcal{W}'$ such that $\psi|_{\operatorname{range}(T)}=0$ and $\psi(\mathbf{w})=1$.\\
Then $\psi$ vanishes on $\operatorname{range}(T)$, so $\psi T=0$, while $\psi(\mathbf{w})=1$.
\end{enumerate}
\hlt{Structural Commentary}: A missing range vector is detected by a dual test that vanishes on the whole range.\\
\hlt{Key Insight}: Either a vector is reached by the map, or a functional separates it from the range.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:standard-coordinate-dual-identification} \hlt{Standard Coordinate Dual Identification}\\
For each positive integer $m$, the map
\[
\Theta:\mathbb{F}^m\to(\mathbb{F}^m)',\qquad
\Theta(\mathbf{y})(\mathbf{z})=\mathbf{y}^t\mathbf{z},
\]
is an isomorphism. Hence every $\psi\in(\mathbb{F}^m)'$ has a unique form $\psi(\mathbf{z})=\mathbf{y}^t\mathbf{z}$ with $\mathbf{y}\in\mathbb{F}^m$.
\begin{proof}
Linearity of $\Theta$ follows from bilinearity of the coordinate dot product. If $\Theta(\mathbf{y})=0$, then evaluating at the standard basis vector $\mathbf{e}_j$ gives $y_j=0$ for every $j$, so $\mathbf{y}=\mathbf{0}$. Thus $\Theta$ is injective. Since $\dim(\mathbb{F}^m)=m=\dim((\mathbb{F}^m)')$ by Theorem \ref{thm:dual-basis-and-dimension}, Corollary \ref{cor:equal-dimension-kernel-range} gives that $\Theta$ is an isomorphism.\\
\hlt{Structural Commentary}: Coordinate column vectors represent all coordinate linear functionals by row multiplication.\\
\hlt{Key Insight}: The standard dual of $\mathbb{F}^m$ is represented by transposed coordinate vectors.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:matrix-equation-left-nullspace-alternative} \hlt{Matrix Equation Alternative}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{b}\in\mathbb{F}^m$. Exactly one holds:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\mathbf{x}=\mathbf{b}$ for some $\mathbf{x}\in\mathbb{F}^n$;
\item $\mathbf{y}^tA=\mathbf{0}^t$ and $\mathbf{y}^t\mathbf{b}\neq0$ for some $\mathbf{y}\in\mathbb{F}^m$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:range-membership-dual-alternative}, for $T_A:\mathbb{F}^n\to\mathbb{F}^m$ and $\mathbf{w}=\mathbf{b}$, $\mathbf{b}\in\operatorname{range}(T_A)$ is exactly $A\mathbf{x}=\mathbf{b}$ for some $\mathbf{x}\in\mathbb{F}^n$.
\item By Proposition \ref{prop:standard-coordinate-dual-identification}, $\psi\in(\mathbb{F}^m)'$ has the form $\psi(\mathbf{z})=\mathbf{y}^t\mathbf{z}$ for a unique $\mathbf{y}\in\mathbb{F}^m$.\\
Thus $\psi T_A=0$ iff $\mathbf{y}^tA=\mathbf{0}^t$, and $\psi(\mathbf{b})\neq0$ iff $\mathbf{y}^t\mathbf{b}\neq0$.
\end{enumerate}
\hlt{Structural Commentary}: Inconsistency is certified by a left-null vector.\\
\hlt{Key Insight}: Left null vectors are the coordinate form of dual separating functionals.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:matrix-of-dual-map-is-transpose} \hlt{Matrix of the Dual Map Is the Transpose}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional, let $T\in\LL(\mathcal{V},\mathcal{W})$, and let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_m)$ be bases of $\mathcal{V}$ and $\mathcal{W}$. If $\beta'$ and $\gamma'$ are the corresponding dual bases, then $[T']_{\beta'\leftarrow\gamma'}=\left([T]_{\gamma\leftarrow\beta}\right)^t$.
\begin{proof}
Let $A=[T]_{\gamma\leftarrow\beta}$ and $C=[T']_{\beta'\leftarrow\gamma'}$. Write $\gamma'=(\psi_1,\ldots,\psi_m)$ and $\beta'=(\varphi_1,\ldots,\varphi_n)$.\\
For $1\leq j\leq m$, $T'\psi_j=\sum\limits_{r=1}^n C_{r,j}\varphi_r$.
Evaluating at $\mathbf{v}_k$ gives $C_{k,j}=(T'\psi_j)(\mathbf{v}_k)=\psi_j(T\mathbf{v}_k)$.\\
Since $T\mathbf{v}_k=\sum\limits_{r=1}^m A_{r,k}\mathbf{w}_r$, we get $\psi_j(T\mathbf{v}_k)=A_{j,k}$. Hence $C_{k,j}=A_{j,k}$ for all $j,k$, so $C=A^t$.\\
\hlt{Structural Commentary}: Duality turns the matrix of a map into the matrix of its pullback on coordinate functionals. The row-column indexing is exactly transposition.\\
\hlt{Key Insight}: Coefficients of $T\mathbf{v}_k$ become values of the dual basis functionals on $T\mathbf{v}_k$.
\end{proof}
\end{theorem}
