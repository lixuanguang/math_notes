\subsection{Multilinear Structure}
Throughout this subsection, all vector spaces are over the same fixed field $\mathbb{F}$.

\begin{definition} \label{def:bilinear-form-and-matrix} \hlt{Bilinear Pairings, Matrices, and Induced Functions}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A map $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ is \hlt{bilinear} if $(\forall\,a,b\in\mathbb{F})(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})(\forall\,\mathbf{w}\in\mathcal{W})\,B(a\mathbf{u}+b\mathbf{v},\mathbf{w})=aB(\mathbf{u},\mathbf{w})+bB(\mathbf{v},\mathbf{w})$,\\
and $(\forall\,a,b\in\mathbb{F})(\forall\,\mathbf{v}\in\mathcal{V})(\forall\,\mathbf{x},\mathbf{y}\in\mathcal{W})\,B(\mathbf{v},a\mathbf{x}+b\mathbf{y})=aB(\mathbf{v},\mathbf{x})+bB(\mathbf{v},\mathbf{y})$.
\item Suppose $\dim(\mathcal{V})=m\geq1$ and $\dim(\mathcal{W})=n\geq1$, with ordered bases $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$.\\
The \hlt{matrix of $B$ in $\beta,\gamma$} is $[B]_{\beta,\gamma}\defeq[B(\mathbf{v}_i,\mathbf{w}_j)]_{1\leq i\leq m,\ 1\leq j\leq n}\in\operatorname{Mat}(m,n,\mathbb{F})$.
\item For bilinear $B$, define $\Phi_B$ by $(\forall\,\mathbf{v}\in\mathcal{V})(\forall\,\mathbf{w}\in\mathcal{W})\,(\Phi_B\mathbf{v})(\mathbf{w})\defeq B(\mathbf{v},\mathbf{w})$.\\
Define $\Psi_B$ by $(\forall\,\mathbf{w}\in\mathcal{W})(\forall\,\mathbf{v}\in\mathcal{V})\,(\Psi_B\mathbf{w})(\mathbf{v})\defeq B(\mathbf{v},\mathbf{w})$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:bilinear-coordinate-evaluation} \hlt{Coordinate Evaluation of a Bilinear Pairing}\\
Let $\dim(\mathcal{V})=m\geq1$ and $\dim(\mathcal{W})=n\geq1$, let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear.\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ be ordered bases of $\mathcal{V}$ and $\mathcal{W}$, respectively, and set $M\defeq[B]_{\beta,\gamma}$.\\
For $\mathbf{v}\in\mathcal{V}$ and $\mathbf{w}\in\mathcal{W}$, set $\mathbf{x}\defeq[\mathbf{v}]_{\beta}$ and $\mathbf{y}\defeq[\mathbf{w}]_{\gamma}$. Then $B(\mathbf{v},\mathbf{w})=\sum\limits_{i=1}^{m}\sum\limits_{j=1}^{n}x_iB(\mathbf{v}_i,\mathbf{w}_j)y_j=\mathbf{x}^tM\mathbf{y}$.
\begin{proof}
The identities $\mathbf{v}=\sum\limits_{i=1}^{m}x_i\mathbf{v}_i$ and $\mathbf{w}=\sum\limits_{j=1}^{n}y_j\mathbf{w}_j$ hold by Definition \ref{def:coordinate-vector-and-coordinate-map}.\\
Hence $B(\mathbf{v},\mathbf{w})=\sum\limits_{i=1}^{m}\sum\limits_{j=1}^{n}x_iy_jB(\mathbf{v}_i,\mathbf{w}_j)$ by Definition \ref{def:bilinear-form-and-matrix}(i). The sum equals $\mathbf{x}^tM\mathbf{y}$ by Definition \ref{def:bilinear-form-and-matrix}(ii) and Proposition \ref{prop:matrix-product-entry-row-column}.\\
\hlt{Structural Commentary}: Two slotwise expansions convert basis coefficients into a matrix product.\\
\hlt{Key Insight}: A pairing matrix records every basis-vector interaction.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:bilinear-forms-linearize-through-duals} \hlt{Bilinear Pairings Linearize through Duals}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\Phi_B:\mathcal{V}\to\mathcal{W}'$ and $\Psi_B:\mathcal{W}\to\mathcal{V}'$ are well-defined linear maps;
\item if $\dim(\mathcal{V})=m\geq1$ and $\dim(\mathcal{W})=n\geq1$, $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ are ordered bases of $\mathcal{V}$ and $\mathcal{W}$ with dual bases $\beta',\gamma'$, and $M\defeq[B]_{\beta,\gamma}$, then $[\Phi_B]_{\gamma'\leftarrow\beta}=M^t$ and $[\Psi_B]_{\beta'\leftarrow\gamma}=M$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Definition \ref{def:bilinear-form-and-matrix}(i),(iii), $(\forall\,\mathbf{v}\in\mathcal{V})\,\Phi_B\mathbf{v}\in\mathcal{W}'$ and $(\forall\,\mathbf{w}\in\mathcal{W})\,\Psi_B\mathbf{w}\in\mathcal{V}'$.\\
For $a,b\in\mathbb{F}$, $\mathbf{u},\mathbf{v}\in\mathcal{V}$, and $\mathbf{w}\in\mathcal{W}$, $(\Phi_B(a\mathbf{u}+b\mathbf{v}))(\mathbf{w})=aB(\mathbf{u},\mathbf{w})+bB(\mathbf{v},\mathbf{w})=(a\Phi_B\mathbf{u}+b\Phi_B\mathbf{v})(\mathbf{w})$.\\
Hence $\Phi_B(a\mathbf{u}+b\mathbf{v})=a\Phi_B\mathbf{u}+b\Phi_B\mathbf{v}$ by Assumption \ref{assump:function-extensionality}. For $\mathbf{x},\mathbf{y}\in\mathcal{W}$, the identity $\Psi_B(a\mathbf{x}+b\mathbf{y})=a\Psi_B\mathbf{x}+b\Psi_B\mathbf{y}$ holds by Definition \ref{def:bilinear-form-and-matrix}(i) and Assumption \ref{assump:function-extensionality}. Thus both maps are linear.
\item By Definition \ref{def:matrix-representation-of-linear-map}, \ref{def:dual-basis}, and \ref{def:bilinear-form-and-matrix}(ii), $(\forall\,i\in\{1,\ldots,m\})(\forall\,j\in\{1,\ldots,n\})\,([\Phi_B]_{\gamma'\leftarrow\beta})_{j,i}=(\Phi_B\mathbf{v}_i)(\mathbf{w}_j)=B(\mathbf{v}_i,\mathbf{w}_j)=M_{i,j}$.\\
Hence $[\Phi_B]_{\gamma'\leftarrow\beta}=M^t$. Likewise, $(\forall\,i\in\{1,\ldots,m\})(\forall\,j\in\{1,\ldots,n\})\,([\Psi_B]_{\beta'\leftarrow\gamma})_{i,j}=(\Psi_B\mathbf{w}_j)(\mathbf{v}_i)=M_{i,j}$ by Definition \ref{def:matrix-representation-of-linear-map}, \ref{def:dual-basis}, and \ref{def:bilinear-form-and-matrix}(ii), so $[\Psi_B]_{\beta'\leftarrow\gamma}=M$.
\end{enumerate}
\hlt{Structural Commentary}: Frozen-slot linearity establishes the maps; dual-basis evaluation identifies their matrices.\\
\hlt{Key Insight}: Freezing one slot turns bilinearity into dual-valued linearity.
\end{proof}
\end{proposition}

\begin{definition} \label{def:bilinear-left-right-radicals} \hlt{Radicals and Nonsingularity of a Bilinear Pairing}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{left} and \hlt{right radicals} are $\operatorname{rad}_{L}(B)\defeq\{\mathbf{v}\in\mathcal{V}:(\forall\,\mathbf{w}\in\mathcal{W})\,B(\mathbf{v},\mathbf{w})=0\}$ and $\operatorname{rad}_{R}(B)\defeq\{\mathbf{w}\in\mathcal{W}:(\forall\,\mathbf{v}\in\mathcal{V})\,B(\mathbf{v},\mathbf{w})=0\}$.
\item For $\mathcal{S}\subseteq\mathcal{V}$ and $\mathcal{T}\subseteq\mathcal{W}$, the \hlt{right} and \hlt{left orthogonals} are $\mathcal{S}^{\perp_R}\defeq\{\mathbf{w}\in\mathcal{W}:(\forall\,\mathbf{s}\in\mathcal{S})\,B(\mathbf{s},\mathbf{w})=0\}$\\
and ${}^{\perp_L}\mathcal{T}\defeq\{\mathbf{v}\in\mathcal{V}:(\forall\,\mathbf{t}\in\mathcal{T})\,B(\mathbf{v},\mathbf{t})=0\}$.
\item $B$ is \hlt{left nondegenerate} if $\operatorname{rad}_{L}(B)=\{\mathbf{0}_{\mathcal{V}}\}$ and \hlt{right nondegenerate} if $\operatorname{rad}_{R}(B)=\{\mathbf{0}_{\mathcal{W}}\}$.
\item $B$ is \hlt{nonsingular} or \hlt{perfect} if $\Phi_B:\mathcal{V}\to\mathcal{W}'$ and $\Psi_B:\mathcal{W}\to\mathcal{V}'$ are isomorphisms.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:bilinear-radicals-matrix-and-dual-maps} \hlt{Radicals, Matrices, and Dual Maps}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\ker(\Phi_B)=\operatorname{rad}_{L}(B)$ and $\ker(\Psi_B)=\operatorname{rad}_{R}(B)$;
\item if $\dim(\mathcal{V})=m\geq1$ and $\dim(\mathcal{W})=n\geq1$, with ordered bases $\beta$ of $\mathcal{V}$ and $\gamma$ of $\mathcal{W}$ and with $M\defeq[B]_{\beta,\gamma}$,\\
then $\{[\mathbf{v}]_{\beta}:\mathbf{v}\in\operatorname{rad}_{L}(B)\}=\ker(M^t)$ and $\{[\mathbf{w}]_{\gamma}:\mathbf{w}\in\operatorname{rad}_{R}(B)\}=\ker(M)$;
\item under the hypotheses of (ii), $\dim(\operatorname{rad}_{L}(B))=m-\operatorname{rank}(M)$ and $\dim(\operatorname{rad}_{R}(B))=n-\operatorname{rank}(M)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Definition \ref{def:bilinear-left-right-radicals}(i), \ref{def:bilinear-form-and-matrix}(iii), Assumption \ref{assump:function-extensionality}, $(\forall\,\mathbf{v}\in\mathcal{V})\,\mathbf{v}\in\ker(\Phi_B)\Longleftrightarrow(\forall\,\mathbf{w}\in\mathcal{W})\,B(\mathbf{v},\mathbf{w})=0\Longleftrightarrow\mathbf{v}\in\operatorname{rad}_{L}(B)$.\\
By Definition \ref{def:bilinear-left-right-radicals}(i), \ref{def:bilinear-form-and-matrix}(iii),  Assumption \ref{assump:function-extensionality}, $(\forall\,\mathbf{w}\in\mathcal{W})\,\mathbf{w}\in\ker(\Psi_B)\Longleftrightarrow(\forall\,\mathbf{v}\in\mathcal{V})\,B(\mathbf{v},\mathbf{w})=0\Longleftrightarrow\mathbf{w}\in\operatorname{rad}_{R}(B)$ 
\item For $\mathbf{v}\in\mathcal{V}$ and $\mathbf{w}\in\mathcal{W}$, set $\mathbf{x}\defeq[\mathbf{v}]_{\beta}$ and $\mathbf{y}\defeq[\mathbf{w}]_{\gamma}$. By Proposition \ref{prop:matrix-representation-action} and \ref{prop:bilinear-forms-linearize-through-duals}(ii), $[\Phi_B\mathbf{v}]_{\gamma'}=M^t\mathbf{x}$ and $[\Psi_B\mathbf{w}]_{\beta'}=M\mathbf{y}$.\\
Hence $\mathbf{v}\in\operatorname{rad}_{L}(B)\Longleftrightarrow\mathbf{x}\in\ker(M^t)$ and $\mathbf{w}\in\operatorname{rad}_{R}(B)\Longleftrightarrow\mathbf{y}\in\ker(M)$ by (i) and Proposition \ref{prop:coordinate-map-isomorphism}(ii).\\
These equivalences prove both coordinate-set identities.
\item The coordinate-map restrictions to the two radicals are linear and injective by Proposition \ref{prop:coordinate-map-isomorphism}(i)--(ii), and they are surjective onto $\ker(M^t)$ and $\ker(M)$ by (ii). Thus their corestrictions are isomorphisms by Proposition \ref{prop:linear-map-invertible-iff-bijective}, so the radical dimensions equal the corresponding kernel dimensions.\\
By Corollary \ref{cor:matrix-rank-nullity}(i) and Theorem \ref{thm:row-rank-equals-column-rank}, $\dim(\ker(M^t))=m-\operatorname{rank}(M)$ and $\dim(\ker(M))=n-\operatorname{rank}(M)$.
\end{enumerate}
\hlt{Structural Commentary}: Kernel identities pass through coordinate isomorphisms; rank-nullity then determines both dimensions.\\
\hlt{Key Insight}: Radicals are the two matrix nullspaces in intrinsic form.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:square-bilinear-pairing-equivalences} \hlt{Square Pairing Nonsingularity Criteria}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear with $\dim(\mathcal{V})=\dim(\mathcal{W})=n\geq1$, let $\beta$ and $\gamma$ be ordered bases of $\mathcal{V}$ and $\mathcal{W}$, respectively.\\
Set $M\defeq[B]_{\beta,\gamma}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $B$ is left nondegenerate;
\item $B$ is right nondegenerate;
\item $M$ is invertible;
\item $B$ is perfect.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ If $\operatorname{rad}_{L}(B)=\{\mathbf{0}_{\mathcal{V}}\}$, then $\dim(\operatorname{rad}_{R}(B))=n-\operatorname{rank}(M)=\dim(\operatorname{rad}_{L}(B))=0$ by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}(iii).\\
Hence $\operatorname{rad}_{R}(B)=\{\mathbf{0}_{\mathcal{W}}\}$ by Proposition \ref{prop:dimension-bounds-and-basis-criteria}(ii).
\item[$(ii)$] $\Rightarrow (iii)$ If $\operatorname{rad}_{R}(B)=\{\mathbf{0}_{\mathcal{W}}\}$, then $0=\dim(\operatorname{rad}_{R}(B))=n-\operatorname{rank}(M)$ by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}(iii); $M$ invertible by Corollary \ref{cor:square-matrix-invertible-iff-full-rank}.
\item[$(iii)$] $\Rightarrow (iv)$ If $M$ is invertible, then $\operatorname{rank}(M)=n$ by Corollary \ref{cor:square-matrix-invertible-iff-full-rank}. Hence both radicals have dimension zero by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}(iii), so they are zero subspaces by Proposition \ref{prop:dimension-bounds-and-basis-criteria}(ii). Therefore $\ker(\Phi_B)=\{\mathbf{0}_{\mathcal{V}}\}$ and $\ker(\Psi_B)=\{\mathbf{0}_{\mathcal{W}}\}$ by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}(i).\\
Their domains have dimension $n$ by hypothesis, and their codomains have dimension $n$ by Theorem \ref{thm:dual-basis-and-dimension}(ii), so $\Phi_B$ and $\Psi_B$ are isomorphisms by Corollary \ref{cor:equal-dimension-kernel-range}(i), Proposition \ref{prop:injective-iff-kernel-zero}, and Proposition \ref{prop:linear-map-invertible-iff-bijective}. Thus $B$ is perfect by Definition \ref{def:bilinear-left-right-radicals}(iv).
\item[$(iv)$] $\Rightarrow (i)$ If $B$ is perfect, then $\Phi_B$ is an isomorphism by Definition \ref{def:bilinear-left-right-radicals}(iv) and hence injective by Proposition \ref{prop:linear-map-invertible-iff-bijective}.\\
Thus $\ker(\Phi_B)=\{\mathbf{0}_{\mathcal{V}}\}$ by Proposition \ref{prop:injective-iff-kernel-zero}, and $\operatorname{rad}_{L}(B)=\{\mathbf{0}_{\mathcal{V}}\}$ by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}(i).
\end{enumerate}
\hlt{Structural Commentary}: Equal nullities link the radicals; full rank then identifies perfectness.\\
\hlt{Key Insight}: Square pairings have one shared rank obstruction.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:perfect-pairing-represents-functionals} \hlt{Perfect Pairings Represent Functionals}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be perfect. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,\psi\in\mathcal{W}')(\exists!\,\mathbf{v}\in\mathcal{V})(\forall\,\mathbf{w}\in\mathcal{W})\,\psi(\mathbf{w})=B(\mathbf{v},\mathbf{w})$;
\item $(\forall\,\varphi\in\mathcal{V}')(\exists!\,\mathbf{w}\in\mathcal{W})(\forall\,\mathbf{v}\in\mathcal{V})\,\varphi(\mathbf{v})=B(\mathbf{v},\mathbf{w})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The map $\Phi_B:\mathcal{V}\to\mathcal{W}'$ is an isomorphism by Definition \ref{def:bilinear-left-right-radicals}(iv), and hence a bijection by Proposition \ref{prop:linear-map-invertible-iff-bijective}.\\
Thus $(\forall\,\psi\in\mathcal{W}')(\exists!\,\mathbf{v}\in\mathcal{V})\,\psi=\Phi_B\mathbf{v}$. For this $\mathbf{v}$, $(\forall\,\mathbf{w}\in\mathcal{W})\,\psi(\mathbf{w})=B(\mathbf{v},\mathbf{w})$ by Definition \ref{def:bilinear-form-and-matrix}(iii).
\item The map $\Psi_B:\mathcal{W}\to\mathcal{V}'$ is an isomorphism by Definition \ref{def:bilinear-left-right-radicals}(iv), and hence a bijection by Proposition \ref{prop:linear-map-invertible-iff-bijective}.\\
Thus $(\forall\,\varphi\in\mathcal{V}')(\exists!\,\mathbf{w}\in\mathcal{W})\,\varphi=\Psi_B\mathbf{w}$. For this $\mathbf{w}$, $(\forall\,\mathbf{v}\in\mathcal{V})\,\varphi(\mathbf{v})=B(\mathbf{v},\mathbf{w})$ by Definition \ref{def:bilinear-form-and-matrix}(iii).
\end{enumerate}
\hlt{Structural Commentary}: Bijectivity of the two induced maps supplies the two unique representatives.\\
\hlt{Key Insight}: Perfect pairings turn functionals into uniquely paired vectors.
\end{proof}
\end{corollary}

\begin{definition} \label{def:matrix-congruence} \hlt{Matrix Congruence}\\
For $n\geq1$, matrices $M,N\in\operatorname{Mat}(n,n,\mathbb{F})$ are \hlt{congruent} if $(\exists\,P\in\operatorname{Mat}(n,n,\mathbb{F}))[P\text{ is invertible}\land N=P^tMP]$.
\end{definition}

\begin{proposition} \label{prop:bilinear-pairing-change-of-bases} \hlt{Two-Sided Basis Change for Pairings}\\
Let $\dim(\mathcal{U})=p\geq1$ and $\dim(\mathcal{W})=q\geq1$, let $B:\mathcal{U}\times\mathcal{W}\to\mathbb{F}$ be bilinear, let $\alpha,\widetilde{\alpha}$ be ordered bases of $\mathcal{U}$, let $\beta,\widetilde{\beta}$ be ordered bases of $\mathcal{W}$.\\
Set $M\defeq[B]_{\alpha,\beta}$, $\widetilde{M}\defeq[B]_{\widetilde{\alpha},\widetilde{\beta}}$, $P\defeq C_{\alpha\leftarrow\widetilde{\alpha}}$, and $Q\defeq C_{\beta\leftarrow\widetilde{\beta}}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\widetilde{M}=P^tMQ$;
\item if $\mathcal{U}=\mathcal{W}$, $\alpha=\beta$, and $\widetilde{\alpha}=\widetilde{\beta}$, then $\widetilde{M}$ is congruent to $M$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{W}$, set $\mathbf{x}\defeq[\mathbf{u}]_{\widetilde{\alpha}}$ and $\mathbf{y}\defeq[\mathbf{w}]_{\widetilde{\beta}}$. Then $[\mathbf{u}]_{\alpha}=P\mathbf{x}$ and $[\mathbf{w}]_{\beta}=Q\mathbf{y}$ by Definition \ref{def:change-of-basis-matrix}.\\
By Proposition \ref{prop:bilinear-coordinate-evaluation} and Proposition \ref{prop:transpose-reverses-products}, $\mathbf{x}^t\widetilde{M}\mathbf{y}=B(\mathbf{u},\mathbf{w})=(P\mathbf{x})^tM(Q\mathbf{y})=\mathbf{x}^tP^tMQ\mathbf{y}$.\\
The coordinate maps are surjective by Proposition \ref{prop:coordinate-map-isomorphism}(iii), so $(\forall\,\mathbf{x}\in\mathbb{F}^p)(\forall\,\mathbf{y}\in\mathbb{F}^q)\,\mathbf{x}^t\widetilde{M}\mathbf{y}=\mathbf{x}^tP^tMQ\mathbf{y}$.\\
The relation $(\forall\,i\in\{1,\ldots,p\})(\forall\,j\in\{1,\ldots,q\})\,\widetilde{M}_{i,j}=(P^tMQ)_{i,j}$ holds by substituting the standard coordinate columns.\\
Hence $\widetilde{M}=P^tMQ$ by Definition \ref{def:matrices-and-matrix-operations}(i).
\item Under the stated equalities of spaces and bases, $P=Q$ and $P$ is invertible by Theorem \ref{thm:change-of-basis-matrices}(i).\\
Thus $\widetilde{M}=P^tMP$ by (i), so $\widetilde{M}$ is congruent to $M$ by Definition \ref{def:matrix-congruence}.
\end{enumerate}
\hlt{Structural Commentary}: Coordinate evaluation transforms each slot; entry comparison proves the two-sided formula.\\
\hlt{Key Insight}: Each input slot contributes its own basis-change factor.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:square-bilinear-form-change-of-basis-congruence} \hlt{Basis Change for a Square Bilinear Form}\\
Let $\dim(\mathcal{V})=n\geq1$, let $B:\mathcal{V}\times\mathcal{V}\to\mathbb{F}$ be bilinear, and let $\beta,\gamma$ be ordered bases of $\mathcal{V}$.\\
Set $M_{\beta}\defeq[B]_{\beta,\beta}$, $M_{\gamma}\defeq[B]_{\gamma,\gamma}$, and $C\defeq C_{\beta\leftarrow\gamma}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $M_{\gamma}=C^tM_{\beta}C$;
\item $M_{\beta}$ and $M_{\gamma}$ are congruent.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item With $\mathcal{U}=\mathcal{W}=\mathcal{V}$, old basis $\beta$ in both slots, and new basis $\gamma$ in both slots, $M_{\gamma}=C^tM_{\beta}C$ by Proposition \ref{prop:bilinear-pairing-change-of-bases}(i).
\item The matrix $C$ is invertible by Theorem \ref{thm:change-of-basis-matrices}(i), and $M_{\gamma}=C^tM_{\beta}C$ by (i). Hence $M_{\beta}$ and $M_{\gamma}$ are congruent by Definition \ref{def:matrix-congruence}.
\end{enumerate}
\hlt{Structural Commentary}: The square case specializes both slot changes to one invertible matrix.\\
\hlt{Key Insight}: Reusing one basis change produces congruence.
\end{proof}
\end{proposition}
