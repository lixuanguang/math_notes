\subsection{Multilinear Structure}
\begin{definition} \label{def:bilinear-form-and-matrix} \hlt{Bilinear Forms and Matrices}\\
A map $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ is \hlt{bilinear} if it is linear in each variable separately.\\
For bases $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ of $\mathcal{V}$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ of $\mathcal{W}$, the \hlt{matrix of $B$ in the bases $\beta,\gamma$} is $[B]_{\beta,\gamma}=[B(\mathbf{v}_i,\mathbf{w}_j)]$. If $\mathbf{x}=[\mathbf{v}]_{\beta}$ and $\mathbf{y}=[\mathbf{w}]_{\gamma}$, then $B(\mathbf{v},\mathbf{w})=\mathbf{x}^t [B]_{\beta,\gamma}\mathbf{y}$.
\end{definition}

\begin{proposition} \label{prop:bilinear-forms-linearize-through-duals} \hlt{Bilinear Forms Linearize through Duals}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear, and define $\Phi_B:\mathcal{V}\to\mathcal{W}'$ where $(\Phi_B\mathbf{v})(\mathbf{w})=B(\mathbf{v},\mathbf{w})$,\\
and $\Psi_B:\mathcal{W}\to\mathcal{V}'$ where $(\Psi_B\mathbf{w})(\mathbf{v})=B(\mathbf{v},\mathbf{w})$. Then $\Phi_B,\Psi_B$ are linear.\\
For bases $\beta,\gamma$ and $M=[B]_{\beta,\gamma}$, $[\Phi_B]_{\gamma'\leftarrow\beta}=M^t$; and $[\Psi_B]_{\beta'\leftarrow\gamma}=M$.
\begin{proof}
For fixed $\mathbf{v}$ and $\mathbf{w}$, bilinearity gives $\Phi_B\mathbf{v}\in\mathcal{W}'$ and $\Psi_B\mathbf{w}\in\mathcal{V}'$. For $\mathbf{v}_1,\mathbf{v}_2\in\mathcal{V}$, $a\in\mathbb{F}$, and every $\mathbf{w}\in\mathcal{W}$, $(\Phi_B(a\mathbf{v}_1+\mathbf{v}_2))(\mathbf{w}) =B(a\mathbf{v}_1+\mathbf{v}_2,\mathbf{w}) =a(\Phi_B\mathbf{v}_1)(\mathbf{w})+(\Phi_B\mathbf{v}_2)(\mathbf{w})$, so $\Phi_B$ is linear.\\
For $\mathbf{w}_1,\mathbf{w}_2\in\mathcal{W}$, $a\in\mathbb{F}$, and every $\mathbf{v}\in\mathcal{V}$, $(\Psi_B(a\mathbf{w}_1+\mathbf{w}_2))(\mathbf{v}) =B(\mathbf{v},a\mathbf{w}_1+\mathbf{w}_2) =a(\Psi_B\mathbf{w}_1)(\mathbf{v})+(\Psi_B\mathbf{w}_2)(\mathbf{v})$,  so $\Psi_B$ is linear. Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$.\\
The $j$th $\gamma'$-coordinate of $\Phi_B(\mathbf{v}_i)$ is $(\Phi_B\mathbf{v}_i)(\mathbf{w}_j)=B(\mathbf{v}_i,\mathbf{w}_j)=M_{i,j}$, so column $i$ of $[\Phi_B]_{\gamma'\leftarrow\beta}$ is row $i$ of $M$ transposed, hence $[\Phi_B]_{\gamma'\leftarrow\beta}=M^t$. Likewise, the $i$th $\beta'$-coordinate of $\Psi_B(\mathbf{w}_j)$ is $M_{i,j}$, so $[\Psi_B]_{\beta'\leftarrow\gamma}=M$.\\
\hlt{Structural Commentary}: A bilinear form becomes linear once one input is frozen. Tracking both freezes gives two dual-valued linear maps whose matrices are transposes of one another.\\
\hlt{Key Insight}: Bilinearity is linearity into a dual space.
\end{proof}
\end{proposition}

\begin{definition} \label{def:bilinear-left-right-radicals} \hlt{Radicals and Nonsingularity of a Bilinear Pairing}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{left} and \hlt{right radicals} are
\[
\begin{aligned}
\operatorname{rad}_{L}(B)&\defeq\{\mathbf{v}\in\mathcal{V}:B(\mathbf{v},\mathbf{w})=0 \text{ for every } \mathbf{w}\in\mathcal{W}\},\\
\operatorname{rad}_{R}(B)&\defeq\{\mathbf{w}\in\mathcal{W}:B(\mathbf{v},\mathbf{w})=0 \text{ for every } \mathbf{v}\in\mathcal{V}\}.
\end{aligned}
\]
\item $B$ is \hlt{left nondegenerate} if $\operatorname{rad}_{L}(B)=\{\mathbf{0}\}$ and \hlt{right nondegenerate} if $\operatorname{rad}_{R}(B)=\{\mathbf{0}\}$.
\item $B$ is \hlt{nonsingular} or \hlt{perfect} if $\Phi_B:\mathcal{V}\to\mathcal{W}'$ and $\Psi_B:\mathcal{W}\to\mathcal{V}'$ from Proposition \ref{prop:bilinear-forms-linearize-through-duals} are isomorphisms.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:bilinear-radicals-matrix-and-dual-maps} \hlt{Radicals, Matrices, and Dual Maps}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear, $\dim\mathcal{V}=m$, $\dim\mathcal{W}=n$, and $M=[B]_{\beta,\gamma}$ in ordered bases $\beta,\gamma$.\\
Then $\{[\mathbf{v}]_{\beta}:\mathbf{v}\in\operatorname{rad}_{L}(B)\}=\ker(M^t)$ and $\{[\mathbf{w}]_{\gamma}:\mathbf{w}\in\operatorname{rad}_{R}(B)\}=\ker(M)$,\\
therefore $\dim\operatorname{rad}_{L}(B)=m-\operatorname{rank}(M)$ and $\dim\operatorname{rad}_{R}(B)=n-\operatorname{rank}(M)$.\\
Moreover, $\ker(\Phi_B)=\operatorname{rad}_{L}(B)$ and $\ker(\Psi_B)=\operatorname{rad}_{R}(B)$.
\begin{proof}
For $\mathbf{x}=[\mathbf{v}]_{\beta}$ and $\mathbf{y}=[\mathbf{w}]_{\gamma}$, $B(\mathbf{v},\mathbf{w})=\mathbf{x}^{t}M\mathbf{y}$.\\
Thus $\mathbf{v}\in\operatorname{rad}_{L}(B)$ if and only if $\mathbf{x}^{t}M\mathbf{y}=0$ for all $\mathbf{y}$, if and only if $M^t\mathbf{x}=0$. Similarly, $\mathbf{w}\in\operatorname{rad}_{R}(B)$ if and only if $M\mathbf{y}=0$. This proves the coordinate-set identities. Rank-nullity gives the dimension formulas. Finally, $\mathbf{v}\in\ker(\Phi_B)$ $\Longleftrightarrow$ $B(\mathbf{v},\mathbf{w})=0$ for every $\mathbf{w}\in\mathcal{W}$ $\Longleftrightarrow$ $\mathbf{v}\in\operatorname{rad}_{L}(B)$; similarly $\ker(\Psi_B)=\operatorname{rad}_{R}(B)$.\\
\hlt{Structural Commentary}: Radicals are the kernel obstructions in each slot.\\
\hlt{Key Insight}: In coordinates, radicals are matrix nullspaces.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:square-bilinear-pairing-equivalences} \hlt{Square Pairing Nonsingularity Criteria}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear with $\dim\mathcal{V}=\dim\mathcal{W}=n$, and let $M=[B]_{\beta,\gamma}$ in ordered bases $\beta,\gamma$.\\
Then the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $B$ is left nondegenerate;
\item $B$ is right nondegenerate;
\item $M$ is invertible;
\item $B$ is perfect.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow$ $(ii)$. If $\operatorname{rad}_{L}(B)=0$, then by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}, $\ker(M^t)=0$ $\Rightarrow$ $\operatorname{rank}(M)=n$ $\Rightarrow$ $\ker(M)=0$.
\item[$(ii)$] $\Rightarrow$ $(iii)$. If $\operatorname{rad}_{R}(B)=0$, then $\ker(M)=0$ by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}; since $M\in\operatorname{Mat}(n,n,\mathbb{F})$, $M$ is invertible.
\item[$(iii)$] $\Rightarrow$ $(iv)$. If $M$ is invertible, then $\ker(M)=\ker(M^t)=0$, so $\operatorname{rad}_{L}(B)=\operatorname{rad}_{R}(B)=0$.\\
Hence $\ker(\Phi_B)=\ker(\Psi_B)=0$ by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}.\\
Since $\dim\mathcal{V}=\dim\mathcal{W}'=\dim\mathcal{W}=\dim\mathcal{V}'=n$, $\Phi_B$ and $\Psi_B$ are isomorphisms; hence $B$ is perfect.
\item[$(iv)$] $\Rightarrow$ $(i)$. If $B$ is perfect, then $\Phi_B$ is an isomorphism. Thus $\operatorname{rad}_{L}(B)=\ker(\Phi_B)=0$.
\end{enumerate}
\hlt{Structural Commentary}: In square pairings, one-sided nondegeneracy is invertibility.\\
\hlt{Key Insight}: Perfect pairings are exactly invertible matrices viewed without coordinates.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:perfect-pairing-represents-functionals} \hlt{Perfect Pairings Represent Functionals}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be perfect.\\
Then $\forall\psi\in\mathcal{W}'\ \exists!\mathbf{v}\in\mathcal{V}: \psi(\mathbf{w})=B(\mathbf{v},\mathbf{w})$, $\mathbf{w}\in\mathcal{W}$; and $\forall\varphi\in\mathcal{V}'\ \exists!\mathbf{w}\in\mathcal{W}: \varphi(\mathbf{v})=B(\mathbf{v},\mathbf{w})$, $\mathbf{v}\in\mathcal{V}$.
\begin{proof}
Since $B$ is perfect, $\Phi_B:\mathcal{V}\to\mathcal{W}'$ and $\Psi_B:\mathcal{W}\to\mathcal{V}'$ are isomorphisms.\\
Thus $\psi=\Phi_B(\mathbf{v})$ and $\varphi=\Psi_B(\mathbf{w})$ for unique $\mathbf{v},\mathbf{w}$, which gives the two formulas.\\
\hlt{Structural Commentary}: Perfect pairings identify each space with the other's dual.\\
\hlt{Key Insight}: Functionals are represented by vectors exactly when the appropriate pairing map is onto.
\end{proof}
\end{corollary}

\begin{definition} \label{def:matrix-congruence} \hlt{Matrix Congruence}\\
Matrices $M,N\in\operatorname{Mat}(n,n,\mathbb{F})$ are \hlt{congruent} if $\exists\,P\in\operatorname{Mat}(n,n,\mathbb{F})$ invertible such that $N=P^tMP$.
\end{definition}

\begin{proposition} \label{prop:square-bilinear-form-change-of-basis-congruence} \hlt{Basis Change for a Square Bilinear Form}\\
Let $\mathcal{V}$ be finite-dimensional, let $B:\mathcal{V}\times\mathcal{V}\to\mathbb{F}$ be bilinear, and let $\beta,\gamma$ be ordered bases of $\mathcal{V}$.\\
Set $M_{\beta}\defeq [B]_{\beta,\beta}$; $M_{\gamma}\defeq [B]_{\gamma,\gamma}$, and $C\defeq C_{\beta\leftarrow\gamma}$.\\
Then $M_{\gamma}=C^tM_{\beta}C$. Hence $M_{\beta}$ and $M_{\gamma}$ are congruent.
\begin{proof}
For $\mathbf{x},\mathbf{y}\in\mathcal{V}$, $[\mathbf{x}]_{\beta}=C[\mathbf{x}]_{\gamma}$ and $[\mathbf{y}]_{\beta}=C[\mathbf{y}]_{\gamma}$. Therefore $B(\mathbf{x},\mathbf{y}) =[\mathbf{x}]_{\beta}^{t}M_{\beta}[\mathbf{y}]_{\beta} =[\mathbf{x}]_{\gamma}^{t}C^tM_{\beta}C[\mathbf{y}]_{\gamma}$.\\
By uniqueness of the representing matrix in the basis $\gamma$, $M_{\gamma}=C^tM_{\beta}C$.\\
Since $C$ is invertible by Theorem \ref{thm:change-of-basis-matrices}, Definition \ref{def:matrix-congruence} gives congruence.\\
\hlt{Structural Commentary}: Operators change by similarity; square bilinear forms change by congruence because both input slots are rewritten.\\
\hlt{Key Insight}: A basis change acts on a square bilinear form from both sides.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:bilinear-forms-and-inner-products-scope} \hlt{Bilinear Forms, Inner Products, and Tensor Scope}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Bilinear forms: $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$; inner products add $\mathbf{v}\neq\mathbf{0}\Rightarrow B(\mathbf{v},\mathbf{v})>0$ and, over $\mathbb{C}$, $B(\mathbf{v},\mathbf{w})=\overline{B(\mathbf{w},\mathbf{v})}$.
\item $\Phi_B:\mathcal{V}\to\mathcal{W}'$ and $\Psi_B:\mathcal{W}\to\mathcal{V}'$ encode $B$ in dual spaces, with $[\Phi_B]_{\gamma'\leftarrow\beta}=[B]_{\beta,\gamma}^t$ and $[\Psi_B]_{\beta'\leftarrow\gamma}=[B]_{\beta,\gamma}$.
\item $\mathcal{V}\oplus\mathcal{W}$ and $\mathcal{V}\times\mathcal{W}$ use single-coordinate slots; the finite tensor scaffold uses paired slots $(i,j)$.
\end{enumerate}
\end{perspective}
