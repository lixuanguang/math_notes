\subsection{Double Duals and Dual Maps}
Throughout this subsection, all vector spaces are over the same fixed field $\mathbb{F}$.

\begin{definition} \label{def:algebraic-double-dual-and-natural-map} \hlt{Algebraic Double Dual and Natural Map}\\
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{algebraic double dual} of $\mathcal{V}$ is $\mathcal{V}''\defeq(\mathcal{V}')'$.
\item For $\mathbf{v}\in\mathcal{V}$, the rule
\[
\Lambda_{\mathcal{V}}(\mathbf{v}):\mathcal{V}'\to\mathbb{F},
\qquad
\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)\defeq\varphi(\mathbf{v}),
\]
satisfies
\[
\Lambda_{\mathcal{V}}(\mathbf{v})(a\varphi+b\psi)
=a\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)
+b\Lambda_{\mathcal{V}}(\mathbf{v})(\psi)
\]
for $a,b\in\mathbb{F}$ and $\varphi,\psi\in\mathcal{V}'$, because Proposition \ref{prop:linear-maps-form-vector-space} defines the operations on $\mathcal{V}'$ pointwise. Hence $\Lambda_{\mathcal{V}}(\mathbf{v})\in\mathcal{V}''$, and $\Lambda_{\mathcal{V}}:\mathcal{V}\to\mathcal{V}''$ is the \hlt{natural evaluation map}.
\item $\mathcal{V}$ is \hlt{algebraically reflexive} if $\Lambda_{\mathcal{V}}$ is surjective.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:algebraic-reflexivity-natural-map} \hlt{Algebraic Reflexivity}\\
For every $\mathcal{V}$, the natural map $\Lambda_{\mathcal{V}}:\mathcal{V}\to\mathcal{V}''$ is linear. If $\mathcal{V}$ is finite-dimensional, then $\Lambda_{\mathcal{V}}$ is an isomorphism. Assuming the choice principles in Definition \ref{def:hamel-dimension}, $\Lambda_{\mathcal{V}}$ is injective for every $\mathcal{V}$, and the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\Lambda_{\mathcal{V}}$ is an isomorphism;
\item $\mathcal{V}$ is algebraically reflexive;
\item $\dim(\mathcal{V})<\infty$.
\end{enumerate}
\begin{proof}
For $a,b\in\mathbb{F}$, $\mathbf{u},\mathbf{v}\in\mathcal{V}$, and $\varphi\in\mathcal{V}'$,
\[
\Lambda_{\mathcal{V}}(a\mathbf{u}+b\mathbf{v})(\varphi)
=\varphi(a\mathbf{u}+b\mathbf{v})
=\bigl(a\Lambda_{\mathcal{V}}(\mathbf{u})+b\Lambda_{\mathcal{V}}(\mathbf{v})\bigr)(\varphi).
\]
Assumption \ref{assump:function-extensionality} gives
\[
\Lambda_{\mathcal{V}}(a\mathbf{u}+b\mathbf{v})
=a\Lambda_{\mathcal{V}}(\mathbf{u})+b\Lambda_{\mathcal{V}}(\mathbf{v}),
\]
so $\Lambda_{\mathcal{V}}\in\LL(\mathcal{V},\mathcal{V}'')$.

Suppose $\mathcal{V}$ is finite-dimensional, let $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis, and let $(\varphi_1,\ldots,\varphi_n)$ be its dual basis. If $\mathbf{v}=\sum_{j=1}^n c_j\mathbf{v}_j$ lies in $\ker(\Lambda_{\mathcal{V}})$, then
\[
c_k=\varphi_k(\mathbf{v})=\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi_k)=0
\qquad(1\leq k\leq n).
\]
Thus $\ker(\Lambda_{\mathcal{V}})=\{\mathbf{0}\}$. Theorem \ref{thm:dual-basis-and-dimension} gives
\[
\dim(\mathcal{V}'')=\dim(\mathcal{V}')=\dim(\mathcal{V}),
\]
so Corollary \ref{cor:equal-dimension-kernel-range} and Proposition \ref{prop:linear-map-invertible-iff-bijective} make $\Lambda_{\mathcal{V}}$ an isomorphism.

Now assume the choice principles in Definition \ref{def:hamel-dimension}. If $\mathbf{v}\in\ker(\Lambda_{\mathcal{V}})$, then $\varphi(\mathbf{v})=0$ for every $\varphi\in\mathcal{V}'$, so Proposition \ref{prop:functionals-separate-points} gives $\mathbf{v}=\mathbf{0}$. Hence $\Lambda_{\mathcal{V}}$ is injective for arbitrary $\mathcal{V}$.
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow$ $(ii)$. An isomorphism is surjective, so Definition \ref{def:algebraic-double-dual-and-natural-map} makes $\mathcal{V}$ algebraically reflexive.
\item[$(ii)$] $\Rightarrow$ $(i)$. Algebraic reflexivity gives surjectivity, and the preceding injectivity gives bijectivity. Proposition \ref{prop:linear-map-invertible-iff-bijective} makes $\Lambda_{\mathcal{V}}$ an isomorphism.
\item[$(i)$] $\Rightarrow$ $(iii)$. Suppose instead that $\dim(\mathcal{V})$ is infinite. Theorem \ref{thm:dimension-of-algebraic-dual}, applied first to $\mathcal{V}$ and then to $\mathcal{V}'$, gives
\[
\dim(\mathcal{V}'')>\dim(\mathcal{V}')>\dim(\mathcal{V}).
\]
But an isomorphism sends a Hamel basis bijectively onto a Hamel basis: linearity and injectivity preserve independence, and surjectivity preserves spanning. Definition \ref{def:hamel-dimension} would therefore give $\dim(\mathcal{V}'')=\dim(\mathcal{V})$, a contradiction.
\item[$(iii)$] $\Rightarrow$ $(i)$. This is the finite-dimensional case proved above.
\end{enumerate}
\hlt{Structural Commentary}: $\Lambda_{\mathcal{V}}$ embeds vectors as evaluation functionals.\\
\hlt{Key Insight}: Finite-dimensional spaces are exactly the algebraically reflexive vector spaces.
\end{proof}
\end{theorem}

\begin{perspective} \hlt{Why the Double Dual Is Natural}\\
A finite-dimensional space $\mathcal{V}$ is isomorphic to $\mathcal{V}'$, but such an isomorphism usually requires choosing a basis. The map $\Lambda_{\mathcal{V}}:\mathcal{V}\to\mathcal{V}''$ requires no choice of basis.
\end{perspective}

\begin{definition} \label{def:dual-map} \hlt{Dual Map}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$, the \hlt{dual map} of $T$ is
\[
T':\mathcal{W}'\to\mathcal{V}',
\qquad
T'\varphi\defeq\varphi T.
\]
Proposition \ref{prop:linear-map-product-algebra} gives $\varphi T\in\LL(\mathcal{V},\mathbb{F})=\mathcal{V}'$, so the definition is well-typed. Equivalently,
\[
(T'\varphi)(\mathbf{v})=\varphi(T\mathbf{v})
\qquad(\varphi\in\mathcal{W}',\ \mathbf{v}\in\mathcal{V}).
\]
\end{definition}

\begin{proposition} \label{prop:dual-map-is-linear} \hlt{Dual Maps Are Linear}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$, then $T'\in\LL(\mathcal{W}',\mathcal{V}')$.
\begin{proof}
For $a,b\in\mathbb{F}$, $\varphi,\psi\in\mathcal{W}'$, and $\mathbf{v}\in\mathcal{V}$,
\[
\bigl(T'(a\varphi+b\psi)\bigr)(\mathbf{v})
=(a\varphi+b\psi)(T\mathbf{v})
=\bigl(aT'\varphi+bT'\psi\bigr)(\mathbf{v}).
\]
Assumption \ref{assump:function-extensionality} gives $T'(a\varphi+b\psi)=aT'\varphi+bT'\psi$, so Definition \ref{def:linear-map} gives $T'\in\LL(\mathcal{W}',\mathcal{V}')$.\\
\hlt{Structural Commentary}: The dual map pulls functionals back along $T$ and preserves linear structure.\\
\hlt{Key Insight}: Dual maps reverse direction because a functional can test $T\mathbf{v}$ only after $T$ is applied.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:algebra-of-dual-maps} \hlt{Algebra of Dual Maps}\\
The dual operation has the following properties:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $S,T\in\LL(\mathcal{V},\mathcal{W})$ and $a,b\in\mathbb{F}$, then $(aS+bT)'=aS'+bT'$.
\item If $T\in\LL(\mathcal{U},\mathcal{V})$ and $S\in\LL(\mathcal{V},\mathcal{W})$, then $(ST)'=T'S'$.
\item $(I_{\mathcal{V}})'=I_{\mathcal{V}'}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\varphi\in\mathcal{W}'$ and $\mathbf{v}\in\mathcal{V}$,
\[
\bigl((aS+bT)'\varphi\bigr)(\mathbf{v})
=\varphi(aS\mathbf{v}+bT\mathbf{v})
=\bigl(aS'\varphi+bT'\varphi\bigr)(\mathbf{v}).
\]
Assumption \ref{assump:function-extensionality}, first on $\mathcal{V}$ and then on $\mathcal{W}'$, gives $(aS+bT)'=aS'+bT'$.
\item For $\varphi\in\mathcal{W}'$ and $\mathbf{u}\in\mathcal{U}$,
\[
\bigl((ST)'\varphi\bigr)(\mathbf{u})
=\varphi(S(T\mathbf{u}))
=\bigl(T'(S'\varphi)\bigr)(\mathbf{u}).
\]
Function extensionality gives $(ST)'=T'S'$.
\item For $\varphi\in\mathcal{V}'$ and $\mathbf{v}\in\mathcal{V}$,
\[
\bigl((I_{\mathcal{V}})'\varphi\bigr)(\mathbf{v})
=\varphi(I_{\mathcal{V}}\mathbf{v})
=\varphi(\mathbf{v}).
\]
Function extensionality gives $(I_{\mathcal{V}})'=I_{\mathcal{V}'}$.
\end{enumerate}
\hlt{Structural Commentary}: Duality is linear on maps and contravariant under composition.\\
\hlt{Key Insight}: Pullbacks compose in the opposite order.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:dual-map-inverse} \hlt{Dual Map of an Inverse}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$ is an isomorphism, then $T'$ is an isomorphism and $(T^{-1})'=(T')^{-1}$.
\begin{proof}
Since $T^{-1}\in\LL(\mathcal{W},\mathcal{V})$, Theorem \ref{thm:algebra-of-dual-maps} gives
\[
T'(T^{-1})'=(T^{-1}T)'=(I_{\mathcal{V}})'=I_{\mathcal{V}'},
\]
and
\[
(T^{-1})'T'=(TT^{-1})'=(I_{\mathcal{W}})'=I_{\mathcal{W}'}.
\]
Thus $(T^{-1})'$ is the inverse of $T'$, so $T'$ is an isomorphism and $(T^{-1})'=(T')^{-1}$.\\
\hlt{Structural Commentary}: Duality preserves inverses contravariantly.\\
\hlt{Key Insight}: Invertibility survives duality.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:double-dual-map-identifies-with-original-map} \hlt{Double Dual Maps Identify with Original Maps}\\
Let $T\in\LL(\mathcal{V},\mathcal{W})$ and set $T''\defeq(T')'$. Then
\[
T''\Lambda_{\mathcal{V}}=\Lambda_{\mathcal{W}}T.
\]
If $\mathcal{V}$ and $\mathcal{W}$ are finite-dimensional, Theorem \ref{thm:algebraic-reflexivity-natural-map} makes $\Lambda_{\mathcal{V}}$ and $\Lambda_{\mathcal{W}}$ isomorphisms, so this identity identifies $T''$ with $T$.
\begin{proof}
For $\mathbf{v}\in\mathcal{V}$ and $\psi\in\mathcal{W}'$,
\[
\bigl(T''\Lambda_{\mathcal{V}}(\mathbf{v})\bigr)(\psi)
=\Lambda_{\mathcal{V}}(\mathbf{v})(T'\psi)
=(T'\psi)(\mathbf{v})
=\psi(T\mathbf{v})
=\Lambda_{\mathcal{W}}(T\mathbf{v})(\psi).
\]
Assumption \ref{assump:function-extensionality}, applied first on $\mathcal{W}'$ and then on $\mathcal{V}$, gives $T''\Lambda_{\mathcal{V}}=\Lambda_{\mathcal{W}}T$.\\
\hlt{Structural Commentary}: Double dualisation commutes with natural evaluation.\\
\hlt{Key Insight}: The double dual square commutes because both sides evaluate every functional the same way.
\end{proof}
\end{corollary}
