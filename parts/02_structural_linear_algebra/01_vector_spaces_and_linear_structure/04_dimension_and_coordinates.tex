\subsection{Dimension and Coordinates}

\begin{definition} \label{def:dimension} \hlt{Dimension}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathcal{V}=\{\mathbf{0}\}$, then $\dim(\mathcal{V})\defeq0$.
\item If $\mathcal{V}$ has a basis of length $n\geq1$, then $\mathcal{V}$ is \hlt{$n$-dimensional} and $\dim(\mathcal{V})\defeq n$.
\end{enumerate}
Theorem \ref{thm:basis-length-well-defined} makes $\dim(\mathcal{V})$ well-defined.
\end{definition}

\begin{proposition} \label{prop:dimension-bounds-and-basis-criteria} \hlt{Subspace Dimension Criterion}\\
Let $\mathcal{V}$ be finite-dimensional and let $\mathcal{U}\leq\mathcal{V}$. Then $\dim(\mathcal{U})\leq\dim(\mathcal{V})$, and equality holds iff $\mathcal{U}=\mathcal{V}$.
\begin{proof}
Let $\delta=(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ be a basis of $\mathcal{U}$ and let $\beta$ be a basis of $\mathcal{V}$, where $n=\dim(\mathcal{V})$.\\
Since $\delta$ is independent in $\mathcal{V}$ and $\beta$ spans $\mathcal{V}$, Theorem \ref{thm:finitedimlenofvec} gives $m\leq n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathcal{U}=\mathcal{V}$, then any basis of $\mathcal{U}$ is a basis of $\mathcal{V}$, so $\dim(\mathcal{U})=\dim(\mathcal{V})$.
\item[$(\Leftarrow)$] If $\dim(\mathcal{U})=\dim(\mathcal{V})$, then $\delta$ is independent in $\mathcal{V}$ of length $n$. By Theorem \ref{thm:extending-independent-lists}, extending $\delta$ to a basis of $\mathcal{V}$ adds no vector, since every basis has length $n$. Hence $\mathcal{U}=\operatorname{span}(\delta)=\mathcal{V}$.
\end{enumerate}
\hlt{Structural Commentary}: Dimension bounds subspaces; equality forces fullness.\\
\hlt{Key Insight}: Equal dimension subspaces fill the ambient space.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:full-dimension-basis-criteria} \hlt{Full-Dimension Basis Criteria}\\
Let $\mathcal{V}$ be finite-dimensional and set $n\defeq\dim(\mathcal{V})$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item if $\gamma$ spans $\mathcal{V}$ and has length $n$, then $\gamma$ is a basis;
\item if $\alpha$ is linearly independent in $\mathcal{V}$ and has length $n$, then $\alpha$ is a basis.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] By Theorem \ref{thm:reducing-spanning-lists}, $\gamma$ contains a basis sublist. Since every basis has length $n$, no proper sublist can have length $n$; hence $\gamma$ itself is a basis.
\item[$(ii)$] Extend $\alpha$ to a basis of $\mathcal{V}$ by Theorem \ref{thm:extending-independent-lists}. Since every basis has length $n$ and $\alpha$ already has length $n$, no vector is added; hence $\alpha$ is a basis.
\end{enumerate}
\hlt{Structural Commentary}: Full-length spanning or independent lists cannot be shortened or extended.\\
\hlt{Key Insight}: In dimension $n$, $n$ independent or $n$ spanning vectors already form a basis.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:basis-equivalent-characterizations} \hlt{Basis Characterizations}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a finite list in $\mathcal{V}$, and write $\beta^\frown\mathbf{x}=(\mathbf{v}_1,\ldots,\mathbf{v}_n,\mathbf{x})$. Then the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\beta$ is a basis of $\mathcal{V}$;
\item each $\mathbf{x}\in\mathcal{V}$ has a unique coefficient vector $(a_1,\ldots,a_n)\in\mathbb{F}^n$ with $\mathbf{x}=\sum\limits_{j=1}^{n}a_j\mathbf{v}_j$;
\item $\operatorname{span}(\beta)=\mathcal{V}$, and no proper sublist of $\beta$ spans $\mathcal{V}$;
\item $\beta$ is linearly independent, and $\beta^\frown\mathbf{x}$ is linearly dependent for each $\mathbf{x}\in\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ This is Proposition \ref{prop:basis-unique-coordinate-representation}.
\item[$(ii)$]$\Rightarrow (iii)$ Existence gives $\operatorname{span}(\beta)=\mathcal{V}$.\\
If a proper sublist $\gamma\subsetneq\beta$ spanned $\mathcal{V}$ and omitted the occurrence $\mathbf{v}_j$, then $\mathbf{v}_j$ would have both its standard $\beta$-coordinate representation and one using only $\gamma$, contradicting uniqueness.
\item[$(iii)$]$\Rightarrow (iv)$ If $\beta$ were dependent, Proposition \ref{prop:adjoiningandremovingvectors} would remove some $\mathbf{v}_j$ without changing $\operatorname{span}(\beta)$, contradicting minimality. Thus $\beta$ is independent.\\
Since $\operatorname{span}(\beta)=\mathcal{V}$, writing $\mathbf{x}$ as a linear combination of $\beta$ gives a nontrivial relation on $\beta^\frown\mathbf{x}$.
\item[$(iv)$]$\Rightarrow (i)$ If $\operatorname{span}(\beta)\neq\mathcal{V}$, choose $\mathbf{x}\notin\operatorname{span}(\beta)$. Then Proposition \ref{prop:adjoiningandremovingvectors} makes $\beta^\frown\mathbf{x}$ independent, contradicting (iv). Hence $\beta$ spans $\mathcal{V}$, and with independence from (iv), it is a basis.
\end{enumerate}
\hlt{Structural Commentary}: A basis is simultaneously just enough to span and too large to extend independently.\\
\hlt{Key Insight}: Spanning and independence are dual minimality and maximality conditions.
\end{proof}
\end{theorem}

\begin{definition} \label{def:hamel-basis} \hlt{Hamel Basis}\\
A subset $\mathcal{B}\subseteq\mathcal{V}$ is a \hlt{Hamel basis} of $\mathcal{V}$ if it is linearly independent and $\operatorname{span}(\mathcal{B})=\mathcal{V}$. Equivalently, each $\mathbf{v}\in\mathcal{V}$ has a unique finitely supported coefficient function $a:\mathcal{B}\to\mathbb{F}$ with $\mathbf{v}=\sum\limits_{\mathbf{b}\in\mathcal{B}}a(\mathbf{b})\mathbf{b}$.\\
Finite ordered bases are orderings of finite Hamel bases.
\end{definition}

\begin{theorem} \label{thm:hamel-basis-extension} \hlt{Hamel Basis Extension}\\
Let $I\subseteq S\subseteq\mathcal{V}$, with $I$ linearly independent and $\operatorname{span}(S)=\mathcal{V}$.\\
Assuming Zorn's lemma, there exists a Hamel basis $\mathcal{B}$ of $\mathcal{V}$ such that $I\subseteq\mathcal{B}\subseteq S$. Consequently, every nonzero vector space has a basis; every independent set extends to a basis; every spanning set contains a basis.
\begin{proof}
Let $\mathscr{P}=\{E:I\subseteq E\subseteq S,\ E\ \text{linearly independent}\}$, ordered by inclusion. Then $I\in\mathscr{P}$.\\
If $\mathcal{C}\subseteq\mathscr{P}$ is a nonempty chain, put $U=\bigcup\limits_{E\in\mathcal{C}}E$. In any finite relation $\sum\limits_{k=1}^{m}a_k\mathbf{u}_k=\mathbf{0}$ in $U$, the finitely many vectors $\mathbf{u}_k$ lie in one member of the chain; independence there gives $a_k=0$ for all $k$.\\
Thus $U\in\mathscr{P}$ and is an upper bound for $\mathcal{C}$. The empty chain is bounded by $I$.\\
By Zorn's lemma, Principle \ref{princ:zorns-lemma}, choose maximal $\mathcal{B}\in\mathscr{P}$.\\
If $s\in S\setminus\operatorname{span}(\mathcal{B})$, then $\mathcal{B}\cup\{s\}$ is linearly independent, contradicting maximality. Hence $S\subseteq\operatorname{span}(\mathcal{B})$, so $\operatorname{span}(\mathcal{B})=\mathcal{V}$ and $\mathcal{B}$ is a Hamel basis. The consequences follow by taking $S=\mathcal{V}$ or $I=\emptyset$.\\
\hlt{Structural Commentary}: Finite extension uses exchange; arbitrary extension uses maximality.\\
\hlt{Key Insight}: Maximal independence inside a spanning set forces spanning.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:arbitrary-bases-same-cardinality} \hlt{Dimension as Basis Cardinality}\\
If $\mathcal{B}$ and $\mathcal{C}$ are Hamel bases of $\mathcal{V}$, then $\abs{\mathcal{B}}=\abs{\mathcal{C}}$.
\begin{proof}
If one basis is finite, say $\abs{\mathcal{B}}=n$, then $\mathcal{C}$ cannot contain $n+1$ distinct vectors, since such a finite sublist would be independent while $\mathcal{B}$ spans $\mathcal{V}$, contradicting Theorem \ref{thm:finitedimlenofvec}. Hence $\mathcal{C}$ is finite, and Theorem \ref{thm:basis-length-well-defined} gives $\abs{\mathcal{C}}=n$. Thus assume both bases are infinite. For $\mathbf{c}\in\mathcal{C}$, let $\operatorname{supp}_{\mathcal{B}}(\mathbf{c})\subseteq\mathcal{B}$ be its finite $\mathcal{B}$-support.\\
Every $\mathbf{b}\in\mathcal{B}$ appears in some $\operatorname{supp}_{\mathcal{B}}(\mathbf{c})$: otherwise all vectors in $\operatorname{span}(\mathcal{C})$ would have zero $\mathbf{b}$-coordinate, contradicting $\operatorname{span}(\mathcal{C})=\mathcal{V}$. Hence $\mathcal{B}\subseteq\bigcup\limits_{\mathbf{c}\in\mathcal{C}}\operatorname{supp}_{\mathcal{B}}(\mathbf{c})$. Thus $\abs{\mathcal{B}}\leq\sum\limits_{\mathbf{c}\in\mathcal{C}}\abs{\operatorname{supp}_{\mathcal{B}}(\mathbf{c})}\leq\abs{\mathcal{C}}\aleph_0=\abs{\mathcal{C}}$ by Theorem \ref{thm:infinite-cardinal-absorption-reference}. Interchanging $\mathcal{B}$ and $\mathcal{C}$ gives $\abs{\mathcal{C}}\leq\abs{\mathcal{B}}$; Theorem \ref{thm:schroeder-bernstein} gives equality.\\
\hlt{Structural Commentary}: Even in infinite dimensions, a basis size is intrinsic.\\
\hlt{Key Insight}: Each basis vector is detected inside finite supports relative to the other basis.
\end{proof}
\end{theorem}

\begin{definition} \label{def:hamel-dimension} \hlt{Hamel Dimension}\\
For any Hamel basis $\mathcal{B}$ of $\mathcal{V}$, set $\dim(\mathcal{V}):=\abs{\mathcal{B}}$. This is well-defined by Theorem \ref{thm:arbitrary-bases-same-cardinality}.
\end{definition}
