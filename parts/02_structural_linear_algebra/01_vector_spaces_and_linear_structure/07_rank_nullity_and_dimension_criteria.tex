\subsection{Rank-Nullity and Dimension Criteria}
Throughout this subsection, all vector spaces are over the same fixed field $\mathbb{F}$.

\begin{theorem} \label{thm:range-spanned-by-images-of-basis} \hlt{Range from Basis Images}\\
Let $\mathcal{V}$ be finite-dimensional with basis $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{range}(T)=\operatorname{span}(T\mathbf{v}_1,\ldots,T\mathbf{v}_n)$.
\item $\operatorname{range}(T)$ is finite-dimensional.
\item $\dim(\operatorname{range}(T))\leq n$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$(\Rightarrow)$ For $\mathbf{y}\in\operatorname{range}(T)$, choose $\mathbf{v}\in\mathcal{V}$ with $\mathbf{y}=T\mathbf{v}$ and write $\mathbf{v}=\sum\limits_{j=1}^{n}c_j\mathbf{v}_j$.\\
Then $\mathbf{y}=\sum\limits_{j=1}^{n}c_jT\mathbf{v}_j$ by linearity, so $\operatorname{range}(T)\subseteq\operatorname{span}(T\mathbf{v}_1,\ldots,T\mathbf{v}_n)$. 
\item[$(i)$]$(\Leftarrow)$ $\sum\limits_{j=1}^{n}a_jT\mathbf{v}_j=T(\sum\limits_{j=1}^{n}a_j\mathbf{v}_j)\in\operatorname{range}(T)$ for $a_1,\ldots,a_n\in\mathbb{F}$, so the reverse inclusion holds.
\item[$(ii)$] The finite list in (i) spans $\operatorname{range}(T)$, so the range is finite-dimensional.
\item A basis sublist of the list in (i) exists by Theorem \ref{thm:reducing-spanning-lists}. Its length is at most $n$, and therefore $\dim(\operatorname{range}(T))\leq n$.
\end{enumerate}
\hlt{Structural Commentary}: Two inclusions identify the range; basis reduction then converts its generating list into a dimension bound.\\
\hlt{Key Insight}: A linear map transports every basis expansion to its image.
\end{proof}
\end{theorem}

\begin{definition} \label{def:rank-and-nullity-linear-map} \hlt{Rank and Nullity}\\
Let $\mathcal{V}$ be finite-dimensional and let $T\in\LL(\mathcal{V},\mathcal{W})$. Define $\operatorname{rank}(T)\defeq\dim(\operatorname{range}(T))$ and $\operatorname{nullity}(T)\defeq\dim(\ker(T))$.\\
The rank is finite by Theorem \ref{thm:range-spanned-by-images-of-basis}(ii); the nullity is finite by Proposition \ref{prop:kernel-and-range-are-subspaces} and Corollary \ref{cor:finite-dimensional-spaces-and-subspaces-have-bases}.
\end{definition}

\begin{theorem} \label{thm:rank-nullity} \hlt{Rank-Nullity Theorem}\\
If $\mathcal{V}$ is finite-dimensional and $T\in\LL(\mathcal{V},\mathcal{W})$, then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{range}(T)$ is finite-dimensional.
\item $\dim(\mathcal{V})=\operatorname{nullity}(T)+\operatorname{rank}(T)$.
\end{enumerate}
\begin{proof}
The kernel has a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ by Proposition \ref{prop:kernel-and-range-are-subspaces} and Corollary \ref{cor:finite-dimensional-spaces-and-subspaces-have-bases}.\\
Extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_r)$ of $\mathcal{V}$ by Theorem \ref{thm:extending-independent-lists}.\\
For $\mathbf{v}=\sum\limits_{i=1}^m a_i\mathbf{u}_i+\sum\limits_{j=1}^r b_j\mathbf{w}_j$, $T\mathbf{v}=\sum\limits_{j=1}^r b_jT\mathbf{w}_j$ by linearity, so $(T\mathbf{w}_1,\ldots,T\mathbf{w}_r)$ spans $\operatorname{range}(T)$.\\
If $\sum\limits_{j=1}^r c_jT\mathbf{w}_j=\mathbf{0}_{\mathcal{W}}$, then $\mathbf{x}\defeq\sum\limits_{j=1}^r c_j\mathbf{w}_j\in\ker(T)=\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_m)$.\\
Thus $\mathbf{x}$ also has an extended-basis representation with every $\mathbf{w}_j$-coefficient zero, so $c_1=\cdots=c_r=0$ by Proposition \ref{prop:basis-unique-coordinate-representation}.\\
Hence $(T\mathbf{w}_1,\ldots,T\mathbf{w}_r)$ is a basis of $\operatorname{range}(T)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The basis $(T\mathbf{w}_1,\ldots,T\mathbf{w}_r)$ is finite, so $\operatorname{range}(T)$ is finite-dimensional.
\item The three bases have lengths $m$, $r$, and $m+r$, respectively. Hence $\operatorname{nullity}(T)=m$, $\operatorname{rank}(T)=r$, and $\dim(\mathcal{V})=m+r$.
\end{enumerate}
\hlt{Structural Commentary}: Extending a kernel basis produces complementary vectors whose images form a basis of the range.\\
\hlt{Key Insight}: Every domain basis direction is either killed or detected.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:map-to-smaller-space-not-injective} \hlt{Maps to Smaller Spaces Are Not Injective}\\
If finite-dimensional vector spaces $\mathcal{V}$ and $\mathcal{W}$ satisfy $\dim(\mathcal{V})>\dim(\mathcal{W})$, then no $T\in\LL(\mathcal{V},\mathcal{W})$ is injective.
\begin{proof}
Fix $T\in\LL(\mathcal{V},\mathcal{W})$. By Proposition \ref{prop:kernel-and-range-are-subspaces} and Proposition \ref{prop:dimension-bounds-and-basis-criteria}, $\operatorname{rank}(T)\leq\dim(\mathcal{W})$.\\
By Theorem \ref{thm:rank-nullity}, $\operatorname{nullity}(T)=\dim(\mathcal{V})-\operatorname{rank}(T)\geq\dim(\mathcal{V})-\dim(\mathcal{W})>0$.\\
Thus $\ker(T)\neq\{\mathbf{0}_{\mathcal{V}}\}$, so $T$ is not injective by Proposition \ref{prop:injective-iff-kernel-zero}.\\
\hlt{Structural Commentary}: The range bound feeds rank-nullity, forcing a nontrivial kernel.\\
\hlt{Key Insight}: Excess domain dimension must collapse.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:map-to-larger-space-not-surjective} \hlt{Maps to Larger Spaces Are Not Surjective}\\
If finite-dimensional vector spaces $\mathcal{V}$ and $\mathcal{W}$ satisfy $\dim(\mathcal{V})<\dim(\mathcal{W})$, then no $T\in\LL(\mathcal{V},\mathcal{W})$ is surjective.
\begin{proof}
Fix $T\in\LL(\mathcal{V},\mathcal{W})$. By Theorem \ref{thm:rank-nullity}, $\operatorname{rank}(T)=\dim(\mathcal{V})-\operatorname{nullity}(T)\leq\dim(\mathcal{V})<\dim(\mathcal{W})$. Since $\operatorname{range}(T)\leq\mathcal{W}$ by Proposition \ref{prop:kernel-and-range-are-subspaces}, $\operatorname{range}(T)\neq\mathcal{W}$ by Proposition \ref{prop:dimension-bounds-and-basis-criteria}. Hence $T$ is not surjective.\\
\hlt{Structural Commentary}: Rank-nullity bounds the range below the codomain dimension.\\
\hlt{Key Insight}: A smaller domain cannot cover a larger codomain.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:equal-dimension-kernel-range} \hlt{Equal-Dimension Kernel-Range Criterion}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional vector spaces with $\dim(\mathcal{V})=\dim(\mathcal{W})$, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\ker(T)=\{\mathbf{0}_{\mathcal{V}}\}$ iff $\operatorname{range}(T)=\mathcal{W}$.
\item $T$ is injective iff $T$ is surjective.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) If $\ker(T)=\{\mathbf{0}_{\mathcal{V}}\}$, then $\operatorname{nullity}(T)=0$ and $\operatorname{rank}(T)=\dim(\mathcal{V})=\dim(\mathcal{W})$ by Theorem \ref{thm:rank-nullity}.\\
Since $\operatorname{range}(T)\leq\mathcal{W}$ by Proposition \ref{prop:kernel-and-range-are-subspaces}, $\operatorname{range}(T)=\mathcal{W}$ by Proposition \ref{prop:dimension-bounds-and-basis-criteria}.
\item[$(i)$] ($\Leftarrow$) If $\operatorname{range}(T)=\mathcal{W}$, then $\operatorname{rank}(T)=\dim(\mathcal{W})=\dim(\mathcal{V})$ and $\operatorname{nullity}(T)=0$ by Theorem \ref{thm:rank-nullity}.\\
The zero subspace and $\ker(T)$ both have dimension zero, so they are equal by Proposition \ref{prop:dimension-bounds-and-basis-criteria}.
\item[$(ii)$] ($\Rightarrow$) If $T$ is injective, then $\ker(T)=\{\mathbf{0}_{\mathcal{V}}\}$ by Proposition \ref{prop:injective-iff-kernel-zero}. By (i), $\operatorname{range}(T)=\mathcal{W}$, so $T$ is surjective.
\item[$(ii)$] ($\Leftarrow$) If $T$ is surjective, then $\operatorname{range}(T)=\mathcal{W}$. By (i), $\ker(T)=\{\mathbf{0}_{\mathcal{V}}\}$, so $T$ is injective by Proposition \ref{prop:injective-iff-kernel-zero}.
\end{enumerate}
\hlt{Structural Commentary}: Rank-nullity exchanges kernel loss with range deficit; the kernel criterion translates this to maps.\\
\hlt{Key Insight}: Equal dimensions leave no room for one-sided failure.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:finite-dimensional-operator-invertible-equivalences} \hlt{Finite-Dimensional Invertibility Criteria}\\
If $\mathcal{V}$ is finite-dimensional and $T\in\LL(\mathcal{V})$, then the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is invertible;
\item $T$ is injective;
\item $T$ is surjective.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ The map $T$ is injective by Proposition \ref{prop:linear-map-invertible-iff-bijective}.
\item[$(ii)$] $\Rightarrow (iii)$ The map $T$ is surjective by Corollary \ref{cor:equal-dimension-kernel-range}, applied with the same domain and codomain.
\item[$(iii)$] $\Rightarrow (i)$ The map $T$ is injective by Corollary \ref{cor:equal-dimension-kernel-range}, hence bijective and therefore invertible by Proposition \ref{prop:linear-map-invertible-iff-bijective}.
\end{enumerate}
\hlt{Structural Commentary}: Bijectivity supplies the outer implications; equal dimension links injectivity and surjectivity.\\
\hlt{Key Insight}: A nonzero kernel is the sole finite-dimensional obstruction to operator invertibility.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:isomorphisms-preserve-bases} \hlt{Isomorphisms Preserve Bases}\\
Let $T:\mathcal{V}\to\mathcal{W}$ be an isomorphism and let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a finite list in $\mathcal{V}$. Set $T\beta=(T\mathbf{v}_1,\ldots,T\mathbf{v}_n)$.\\
The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\beta$ is a basis of $\mathcal{V}$.
\item $T\beta$ is a basis of $\mathcal{W}$.
\end{enumerate}
\begin{proof}
The map $T$ is injective and surjective by Proposition \ref{prop:linear-map-invertible-iff-bijective}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ For $\mathbf{w}\in\mathcal{W}$, choose $\mathbf{v}\in\mathcal{V}$ with $\mathbf{w}=T\mathbf{v}$ and write $\mathbf{v}=\sum\limits_{j=1}^{n}a_j\mathbf{v}_j$. Then $\mathbf{w}=\sum\limits_{j=1}^{n}a_jT\mathbf{v}_j$ by linearity, so $T\beta$ spans $\mathcal{W}$.\\
If $\sum\limits_{j=1}^{n}a_jT\mathbf{v}_j=\mathbf{0}_{\mathcal{W}}$, then $T(\sum\limits_{j=1}^{n}a_j\mathbf{v}_j)=\sum\limits_{j=1}^{n}a_jT\mathbf{v}_j=\mathbf{0}_{\mathcal{W}}=T\mathbf{0}_{\mathcal{V}}$ by linearity and Proposition \ref{prop:linear-maps-send-zero-to-zero}.\\
Injectivity and independence of $\beta$ imply $a_1=\cdots=a_n=0$. Hence $T\beta$ is a basis of $\mathcal{W}$.
\item[$(ii)$] $\Rightarrow (i)$ For $\mathbf{v}\in\mathcal{V}$, write $T\mathbf{v}=\sum\limits_{j=1}^{n}a_jT\mathbf{v}_j=T(\sum\limits_{j=1}^{n}a_j\mathbf{v}_j)$ by linearity. By injectivity, $\mathbf{v}=\sum\limits_{j=1}^{n}a_j\mathbf{v}_j$, so $\beta$ spans $\mathcal{V}$.\\
If $\sum\limits_{j=1}^{n}a_j\mathbf{v}_j=\mathbf{0}_{\mathcal{V}}$, then $\sum\limits_{j=1}^{n}a_jT\mathbf{v}_j=\mathbf{0}_{\mathcal{W}}$ by linearity and Proposition \ref{prop:linear-maps-send-zero-to-zero}.\\
Independence of $T\beta$ implies $a_1=\cdots=a_n=0$. Hence $\beta$ is a basis of $\mathcal{V}$.
\end{enumerate}
\hlt{Structural Commentary}: Surjectivity transports spanning representations; injectivity preserves and reflects linear relations.\\
\hlt{Key Insight}: Isomorphisms preserve both halves of the basis condition.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:finite-dimensional-isomorphic-iff-same-dimension} \hlt{Isomorphism and Dimension}\\
Finite-dimensional vector spaces $\mathcal{V}$ and $\mathcal{W}$ are isomorphic iff $\dim(\mathcal{V})=\dim(\mathcal{W})$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Let $T:\mathcal{V}\to\mathcal{W}$ be an isomorphism. Choose a basis $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ of $\mathcal{V}$ by Corollary \ref{cor:finite-dimensional-spaces-and-subspaces-have-bases}.\\
Then $T\beta$ is a basis of $\mathcal{W}$ by Proposition \ref{prop:isomorphisms-preserve-bases}, so $\dim(\mathcal{V})=n=\dim(\mathcal{W})$.
\item[$(\Leftarrow)$] Suppose $\dim(\mathcal{V})=\dim(\mathcal{W})=n$, and choose bases $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ by Corollary \ref{cor:finite-dimensional-spaces-and-subspaces-have-bases}.\\
Choose $T\in\LL(\mathcal{V},\mathcal{W})$ satisfying $(\forall\,j\in\{1,\ldots,n\})\,T\mathbf{v}_j=\mathbf{w}_j$ by Proposition \ref{prop:linear-map-determined-by-basis}. By Theorem \ref{thm:range-spanned-by-images-of-basis}(i), $\operatorname{range}(T)=\operatorname{span}(\gamma)=\mathcal{W}$. \\
or $\mathbf{v}=\sum\limits_{j=1}^{n}a_j\mathbf{v}_j\in\ker(T)$, $\mathbf{0}_{\mathcal{W}}=T\mathbf{v}=\sum\limits_{j=1}^{n}a_j\mathbf{w}_j$, so $a_1=\cdots=a_n=0$ by independence of $\gamma$.\\
Thus $\ker(T)=\{\mathbf{0}_{\mathcal{V}}\}$, and $T$ is injective by Proposition \ref{prop:injective-iff-kernel-zero}.\\
It is surjective by its range identity, hence invertible by Proposition \ref{prop:linear-map-invertible-iff-bijective} and therefore an isomorphism.
\end{enumerate}
\hlt{Structural Commentary}: The forward direction transports a basis; the reverse prescribes an isomorphism on matching bases.\\
\hlt{Key Insight}: Dimension counts the coordinates preserved by an isomorphism.
\end{proof}
\end{theorem}
