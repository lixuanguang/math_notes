\subsection{Vector Spaces}

% **Role:** algebraic language of linear structure.
%
% Fields, vector spaces, subspaces, span, independence, bases, dimension, coordinates.

Vector spaces encode addition and scalar multiplication. Subspaces, spans, bases, dimension, and coordinates describe how linear structure is generated and measured.

\begin{definition} \label{def:vector-space} \hlt{Vector Space}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.vector_space_axioms}\leanok Let $\mathbb{F}$ be a field. A vector space over $\mathbb{F}$ is a set $\mathcal{V}$ with the following structure.\\
The operations are $+\colon\mathcal{V}\times\mathcal{V}\to\mathcal{V}$ and scalar multiplication $\mathbb{F}\times\mathcal{V}\to\mathcal{V}$. The scalar action is written $(a,\mathbf{v})\mapsto a\mathbf{v}$, and the axioms are:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})\,\mathbf{u}+\mathbf{v}=\mathbf{v}+\mathbf{u}$;
\item $(\forall\,\mathbf{u},\mathbf{v},\mathbf{w}\in\mathcal{V})\,(\mathbf{u}+\mathbf{v})+\mathbf{w}=\mathbf{u}+(\mathbf{v}+\mathbf{w})$;
\item an element $\mathbf{0}\in\mathcal{V}$ satisfies $(\forall\,\mathbf{v}\in\mathcal{V})\,\mathbf{0}+\mathbf{v}=\mathbf{v}$;
\item $(\forall\,\mathbf{v}\in\mathcal{V})(\exists\,\mathbf{w}\in\mathcal{V})\,\mathbf{v}+\mathbf{w}=\mathbf{0}$;
\item $(\forall\,\mathbf{v}\in\mathcal{V})\,1\mathbf{v}=\mathbf{v}$;
\item $(\forall\,a,b\in\mathbb{F})(\forall\,\mathbf{v}\in\mathcal{V})\,(ab)\mathbf{v}=a(b\mathbf{v})$;
\item $(\forall\,a\in\mathbb{F})(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})\,a(\mathbf{u}+\mathbf{v})=a\mathbf{u}+a\mathbf{v}$;
\item $(\forall\,a,b\in\mathbb{F})(\forall\,\mathbf{v}\in\mathcal{V})\,(a+b)\mathbf{v}=a\mathbf{v}+b\mathbf{v}$.
\end{enumerate}
If $\mathcal{V}=\{\mathbf{0}\}$, then $\mathcal{V}$ is a \hlt{zero vector space}; otherwise $\mathcal{V}$ is a \hlt{nonzero vector space}.
\end{definition}

\begin{theorem} \label{thm:basicvectorspaceconsequences} \hlt{Basic Consequences of Vector Space Axioms}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.basic_vector_space_consequences}\leanok Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item the additive identity is unique;
\item each $\mathbf{v}\in\mathcal{V}$ has a unique additive inverse, denoted $-\mathbf{v}$;
\item $(\forall\,\mathbf{v}\in\mathcal{V})\,0\mathbf{v}=\mathbf{0}$;
\item $(\forall\,a\in\mathbb{F})\,a\mathbf{0}=\mathbf{0}$;
\item $(\forall\,a\in\mathbb{F})(\forall\,\mathbf{v}\in\mathcal{V})\,[a\mathbf{v}=\mathbf{0}\iff(a=0\lor\mathbf{v}=\mathbf{0})]$.
\end{enumerate}
\begin{proof}
\leanok
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] If $\mathbf{0}$ and $\mathbf{0}'$ are additive identities, then $\mathbf{0}=\mathbf{0}+\mathbf{0}'=\mathbf{0}'$.
\item[$(ii)$] If $\mathbf{w},\mathbf{w}'$ are additive inverses of $\mathbf{v}$, then $\mathbf{w}=\mathbf{w}+\mathbf{0}=\mathbf{w}+(\mathbf{v}+\mathbf{w}') =(\mathbf{w}+\mathbf{v})+\mathbf{w}'=\mathbf{0}+\mathbf{w}'=\mathbf{w}'$.
\item[$(iii)$] $0\mathbf{v}=(0+0)\mathbf{v}=0\mathbf{v}+0\mathbf{v}$; adding $-(0\mathbf{v})$ gives $0\mathbf{v}=\mathbf{0}$.
\item[$(iv)$] $a\mathbf{0}=a(\mathbf{0}+\mathbf{0})=a\mathbf{0}+a\mathbf{0}$; adding $-(a\mathbf{0})$ gives $a\mathbf{0}=\mathbf{0}$.
\item[$(v)$] ($\Rightarrow$) Suppose $a\mathbf{v}=\mathbf{0}$. If $a=0$, then $a=0\lor\mathbf{v}=\mathbf{0}$. If $a\neq0$, then $\mathbf{v}=1\mathbf{v}=(a^{-1}a)\mathbf{v}=a^{-1}(a\mathbf{v})=a^{-1}\mathbf{0}=\mathbf{0}$ by (iv).
\item[$(v)$] ($\Leftarrow$) If $a=0$ or $\mathbf{v}=\mathbf{0}$, then (iii) or (iv) gives $a\mathbf{v}=\mathbf{0}$.
\end{enumerate}
\hlt{Structural Commentary}: The proof derives cancellation, zero actions, and inverse uniqueness from the axioms.\\
\hlt{Key Insight}: Zero and inverse behavior is forced by the axioms.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:negativeoneactsasinverse} \hlt{Scalar Multiplication by $-1$ Is Negation}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.negative_one_smul_eq_neg_vector}\leanok Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then $(\forall\,\mathbf{v}\in\mathcal{V})\,(-1)\mathbf{v}=-\mathbf{v}$.
\begin{proof}
\leanok For $\mathbf{v}\in\mathcal{V}$, $\mathbf{v}+(-1)\mathbf{v}=1\mathbf{v}+(-1)\mathbf{v}=(1-1)\mathbf{v}=0\mathbf{v}$. Hence $\mathbf{v}+(-1)\mathbf{v}=\mathbf{0}$ (Theorem \ref{thm:basicvectorspaceconsequences}(iii)); $(-1)\mathbf{v}=-\mathbf{v}$ (Theorem \ref{thm:basicvectorspaceconsequences}(ii)).\\
\hlt{Structural Commentary}: The proof identifies $(-1)\mathbf{v}$ as the unique additive inverse.\\
\hlt{Key Insight}: Negation is scalar action by $-1$.
\end{proof}
\end{corollary}

\begin{example} \label{ex:roman-vector-space-examples} \hlt{Function-Space Model}\\
Let $\mathbb{F}$ be a field and let $X$ be a set. Set $\mathbb{F}^{X}\defeq\{f:X\to\mathbb{F}\}$. For $f,g\in\mathbb{F}^{X}$ and $a\in\mathbb{F}$, define $f+g$ and $af$ pointwise.\\
Thus $(f+g)(x)=f(x)+g(x)$ and $(af)(x)=af(x)$ for $x\in X$. Then $\mathbb{F}^{X}$ is an $\mathbb{F}$-vector space.\\
Evaluating each axiom at $x\in X$ reduces it to the corresponding field identity, and $\mathbf{0}$ is the constant $0$ function.\\
\hlt{Key Insight}: Pointwise operations inherit linear structure from the scalar field.
\end{example}
