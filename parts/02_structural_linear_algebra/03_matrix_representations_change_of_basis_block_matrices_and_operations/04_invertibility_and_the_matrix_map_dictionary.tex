\subsection{Invertibility and the Matrix-Map Dictionary}
Throughout this subsection, all vector spaces are over the same fixed field $\mathbb{F}$, and every matrix size is a positive integer.

\begin{theorem} \label{thm:invertible-matrix-columns-basis} \hlt{Columns of an Invertible Matrix Form a Basis}\\
Let $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(n,n,\mathbb{F})$ be invertible. Then $(\mathbf{a}_1,\ldots,\mathbf{a}_n)$ is a basis of $\mathbb{F}^n$.
\begin{proof}
Fix $\mathbf{y}\in\mathbb{F}^n$ and set $\mathbf{x}=(x_1,\ldots,x_n)^t\defeq A^{-1}\mathbf{y}$.\\
Then $\mathbf{y}=I_n\mathbf{y}=(AA^{-1})\mathbf{y}=A\mathbf{x}=\sum\limits_{j=1}^{n}x_j\mathbf{a}_j$ by Proposition \ref{prop:matrix-product-associativity-and-identities}, the inverse relation $AA^{-1}=I_n$, and Proposition \ref{prop:matrix-vector-product-column-combination}.\\
If $\mathbf{c}=(c_1,\ldots,c_n)^t$ satisfies $\mathbf{y}=\sum\limits_{j=1}^{n}c_j\mathbf{a}_j=A\mathbf{c}$, then $\mathbf{c}=I_n\mathbf{c}=(A^{-1}A)\mathbf{c}=A^{-1}\mathbf{y}=\mathbf{x}$ by Proposition \ref{prop:matrix-product-associativity-and-identities} and $A^{-1}A=I_n$.\\
Hence $(\forall\,\mathbf{y}\in\mathbb{F}^n)(\exists!\,\mathbf{x}\in\mathbb{F}^n)\,\mathbf{y}=\sum\limits_{j=1}^{n}x_j\mathbf{a}_j$, so $(\mathbf{a}_1,\ldots,\mathbf{a}_n)$ is a basis by Proposition \ref{prop:basis-unique-coordinate-representation}.\\
\hlt{Structural Commentary}: The inverse constructs the unique coefficient column required by the basis criterion.\\
\hlt{Key Insight}: Invertibility turns matrix multiplication into unique coordinate representation.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:square-one-sided-inverse-is-two-sided} \hlt{One-Sided Inverses of Square Matrices}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$. If $AB=I_n$, then $BA=I_n$.
\begin{proof}
The induced map $T_B:\mathbb{F}^n\to\mathbb{F}^n$ is linear by Proposition \ref{prop:matrix-induced-map-linear}.\\
If $\mathbf{x}\in\ker(T_B)$, then $B\mathbf{x}=\mathbf{0}$ and $\mathbf{x}=I_n\mathbf{x}=(AB)\mathbf{x}=A(B\mathbf{x})=\mathbf{0}$ by Proposition \ref{prop:matrix-product-associativity-and-identities}; hence $T_B$ is injective by Proposition \ref{prop:injective-iff-kernel-zero}.\\
It is surjective by Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences}, $(ii)\Rightarrow(iii)$. Fix $\mathbf{y}\in\mathbb{F}^n$ and choose $\mathbf{x}\in\mathbb{F}^n$ satisfying $\mathbf{y}=B\mathbf{x}$.\\
Then $BA\mathbf{y}=BA(B\mathbf{x})=B((AB)\mathbf{x})=B\mathbf{x}=\mathbf{y}$ by Proposition \ref{prop:matrix-product-associativity-and-identities}.\\
Substituting the columns of $I_n$ for $\mathbf{y}$, $BA=I_n$ by Proposition \ref{prop:matrix-product-by-columns} and Proposition \ref{prop:matrix-product-associativity-and-identities}.\\
\hlt{Structural Commentary}: Kernel triviality and equal dimension convert a left inverse into the missing right inverse.\\
\hlt{Key Insight}: Surjectivity lets the known inverse relation reach every vector.
\end{proof}
\end{theorem}

\begin{motivation} \hlt{Coordinate Encoding}\\
Fixing bases encodes a linear map by a matrix. The next results verify composition compatibility and exact matrix recovery.
\end{motivation}

\begin{proposition} \label{prop:composition-and-matrix-multiplication} \hlt{Composition and Matrix Multiplication}\\
Let $\alpha=(\mathbf{u}_1,\ldots,\mathbf{u}_{\ell})$, $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$, and $\delta=(\mathbf{w}_1,\ldots,\mathbf{w}_m)$ be bases of $\mathcal{U}$, $\mathcal{V}$, and $\mathcal{W}$, respectively.\\
If $T\in\LL(\mathcal{U},\mathcal{V})$ and $S\in\LL(\mathcal{V},\mathcal{W})$, then $[ST]_{\delta\leftarrow\alpha}=[S]_{\delta\leftarrow\beta}[T]_{\beta\leftarrow\alpha}$.
\begin{proof}
$ST\in\LL(\mathcal{U},\mathcal{W})$ by Proposition \ref{prop:linear-map-product-algebra}(i). Fix $j\in\{1,\ldots,\ell\}$.\\
Then $[(ST)\mathbf{u}_j]_{\delta}=[S(T\mathbf{u}_j)]_{\delta}=[S]_{\delta\leftarrow\beta}[T\mathbf{u}_j]_{\beta}$ by Proposition \ref{prop:matrix-representation-action}, with input $T\mathbf{u}_j$.\\
The first and last terms are $j$th columns of $[ST]_{\delta\leftarrow\alpha}$ and $[S]_{\delta\leftarrow\beta}[T]_{\beta\leftarrow\alpha}$, respectively, by Definition \ref{def:matrix-representation-of-linear-map} and Proposition \ref{prop:matrix-product-by-columns}.\\
Since $j$ was arbitrary, the matrices are equal by Definition \ref{def:matrices-and-matrix-operations}(i).\\
\hlt{Structural Commentary}: Comparing basis-vector images identifies the corresponding columns of the composition and product matrices.\\
\hlt{Key Insight}: Composition order becomes matrix multiplication order.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:linear-maps-and-matrices-same-data} \hlt{Linear Maps and Matrices Are the Same Data}\\
Let $\dim(\mathcal{V})=n\geq1$ and $\dim(\mathcal{W})=m\geq1$, and fix bases $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_m)$.\\
Define $\Phi:\LL(\mathcal{V},\mathcal{W})\to\operatorname{Mat}(m,n,\mathbb{F})$ by $\Phi(T)\defeq[T]_{\gamma\leftarrow\beta}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\Phi$ is an isomorphism.
\item $\dim(\LL(\mathcal{V},\mathcal{W}))=mn$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Both spaces are vector spaces by Proposition \ref{prop:linear-maps-form-vector-space} and Proposition \ref{prop:matrix-spaces-are-vector-spaces}(i), and $\Phi$ is linear by Proposition \ref{prop:matrix-representation-respects-linear-combinations}.\\
Suppose $\Phi(S)=\Phi(T)$. Then $(\forall\,j\in\{1,\ldots,n\})\,[S\mathbf{v}_j]_{\gamma}=[T\mathbf{v}_j]_{\gamma}$ by Definition \ref{def:matrix-representation-of-linear-map}, so $(\forall\,j\in\{1,\ldots,n\})\,S\mathbf{v}_j=T\mathbf{v}_j$ by Proposition \ref{prop:coordinate-map-isomorphism}(ii). Hence $S=T$ by Proposition \ref{prop:linear-map-determined-by-basis}, and $\Phi$ is injective.\\
Fix $A=[a_{i,j}]\in\operatorname{Mat}(m,n,\mathbb{F})$ and set $\mathbf{z}_j\defeq\sum\limits_{i=1}^{m}a_{i,j}\mathbf{w}_i$ $(j\in\{1,\ldots,n\})$.\\
The relation $(\exists!\,R\in\LL(\mathcal{V},\mathcal{W}))(\forall\,j\in\{1,\ldots,n\})\,R\mathbf{v}_j=\mathbf{z}_j$ holds by Proposition \ref{prop:linear-map-determined-by-basis}; fix this $R$.\\
Then $(\forall\,j\in\{1,\ldots,n\})\,[R\mathbf{v}_j]_{\gamma}=(a_{1,j},\ldots,a_{m,j})^t$ by Definition \ref{def:coordinate-vector-and-coordinate-map}, so $\Phi(R)=A$ by Definition \ref{def:matrix-representation-of-linear-map}.\\
Thus $\Phi$ is surjective and therefore an isomorphism by Proposition \ref{prop:linear-map-invertible-iff-bijective}.
\item The matrix units form a basis of $\operatorname{Mat}(m,n,\mathbb{F})$ with $mn$ members by Proposition \ref{prop:matrix-spaces-are-vector-spaces}(ii).\\
Their inverse images under $\Phi$ form a basis of $\LL(\mathcal{V},\mathcal{W})$ by part (i) and Proposition \ref{prop:isomorphisms-preserve-bases}. Hence $\dim(\LL(\mathcal{V},\mathcal{W}))=mn$.
\end{enumerate}
\hlt{Structural Commentary}: Basis images establish the isomorphism; transporting matrix units determines the dimension.\\
\hlt{Key Insight}: A matrix records exactly the images of a domain basis.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:finite-support-coordinate-model} \hlt{Finite-Support Coordinate Model}\\
Proposition \ref{prop:coordinate-map-isomorphism} and Theorem \ref{thm:linear-maps-and-matrices-same-data} identify finite-dimensional spaces and their maps with coordinate columns and matrices after bases are chosen. Definition \ref{def:finitely-nonzero-sequence-space} supplies the countable finite-support model $\mathbb{F}^{(\mathbb{N})}$; the analogy stops before infinite linear combinations.
\end{perspective}
