\subsection{Change of Basis and Coordinate Transformations}
Throughout this subsection, all vector spaces are over the same fixed field $\mathbb{F}$.

\begin{definition} \label{def:change-of-basis-matrix} \hlt{Change-of-Basis Matrix}\\
Let $n\geq1$, and let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ be bases of the same vector space $\mathcal{V}$.\\
The \hlt{change-of-basis matrix from $\beta$-coordinates to $\gamma$-coordinates} is $C_{\gamma\leftarrow\beta}\defeq[I_{\mathcal{V}}]_{\gamma\leftarrow\beta}\in\operatorname{Mat}(n,n,\mathbb{F})$.\\
The identity $(\forall\,\mathbf{v}\in\mathcal{V})\,[\mathbf{v}]_{\gamma}=C_{\gamma\leftarrow\beta}[\mathbf{v}]_{\beta}$ holds by Proposition \ref{prop:matrix-representation-action}, with $T=I_{\mathcal{V}}$.
\end{definition}

\begin{theorem} \label{thm:change-of-basis-matrices} \hlt{Change-of-Basis Matrices}\\
Let $\mathcal{V}$ be $n$-dimensional, where $n\geq1$, and let $\beta$ be a basis of $\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\gamma$ is a basis of $\mathcal{V}$, then $C_{\gamma\leftarrow\beta}$ is invertible with $(C_{\gamma\leftarrow\beta})^{-1}=C_{\beta\leftarrow\gamma}$.
\item If $S\in\operatorname{Mat}(n,n,\mathbb{F})$ is invertible, then $S=C_{\gamma\leftarrow\beta}$ for a basis $\gamma$ of $\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The identity $[I_{\mathcal{V}}]_{\beta\leftarrow\beta}=I_n=[I_{\mathcal{V}}]_{\gamma\leftarrow\gamma}$ holds by Definition \ref{def:coordinate-vector-and-coordinate-map} and Definition \ref{def:matrix-representation-of-linear-map}.\\
Hence $C_{\beta\leftarrow\gamma}C_{\gamma\leftarrow\beta}=[I_{\mathcal{V}}I_{\mathcal{V}}]_{\beta\leftarrow\beta}=I_n$ and $C_{\gamma\leftarrow\beta}C_{\beta\leftarrow\gamma}=[I_{\mathcal{V}}I_{\mathcal{V}}]_{\gamma\leftarrow\gamma}=I_n$ by Definition \ref{def:change-of-basis-matrix} and Proposition \ref{prop:composition-and-matrix-multiplication}.\\
Thus $C_{\gamma\leftarrow\beta}$ is invertible with inverse $C_{\beta\leftarrow\gamma}$.
\item Write $S^{-1}=[\mathbf{s}_1\ \cdots\ \mathbf{s}_n]$. The relation $(\forall\,j\in\{1,\ldots,n\})(\exists!\,\mathbf{w}_j\in\mathcal{V})\,[\mathbf{w}_j]_{\beta}=\mathbf{s}_j$ holds by Proposition \ref{prop:coordinate-map-isomorphism}(ii),(iii).\\
Set $\gamma\defeq(\mathbf{w}_1,\ldots,\mathbf{w}_n)$. The matrix $S^{-1}$ is invertible with inverse $S$, so $(\mathbf{s}_1,\ldots,\mathbf{s}_n)$ is a basis of $\mathbb{F}^n$ by Theorem \ref{thm:invertible-matrix-columns-basis}.\\
The map $[\cdot]_{\beta}^{-1}:\mathbb{F}^n\to\mathcal{V}$ is an isomorphism by Proposition \ref{prop:coordinate-map-isomorphism}(iii), so $\gamma$ is a basis of $\mathcal{V}$ by Proposition \ref{prop:isomorphisms-preserve-bases}.\\
The relation $(\forall\,j\in\{1,\ldots,n\})\,(C_{\beta\leftarrow\gamma})_{\cdot,j}=[\mathbf{w}_j]_{\beta}=\mathbf{s}_j$ holds by Definition \ref{def:change-of-basis-matrix} and Definition \ref{def:matrix-representation-of-linear-map}; hence $C_{\beta\leftarrow\gamma}=S^{-1}$ by Definition \ref{def:matrices-and-matrix-operations}(i). Therefore $C_{\gamma\leftarrow\beta}=S$ by part (i) and Lemma \ref{lem:two-sided-inverse-uniqueness}.
\end{enumerate}
\hlt{Structural Commentary}: Composition proves reciprocity; inverse-coordinate columns realize an arbitrary invertible matrix.\\
\hlt{Key Insight}: Coordinate isomorphisms transport bases in both directions.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:two-sided-change-of-basis-for-linear-maps} \hlt{Change of Bases for a Linear Map}\\
Let $T\in\LL(\mathcal{V},\mathcal{W})$. Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ and $\beta'=(\mathbf{v}'_1,\ldots,\mathbf{v}'_n)$ be bases of $\mathcal{V}$, and let $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_m)$ and $\gamma'=(\mathbf{w}'_1,\ldots,\mathbf{w}'_m)$ be bases of $\mathcal{W}$, where $n,m\geq1$. Then $[T]_{\gamma'\leftarrow\beta'}=C_{\gamma'\leftarrow\gamma}[T]_{\gamma\leftarrow\beta}C_{\beta\leftarrow\beta'}$.
\begin{proof}
$T=I_{\mathcal{W}}TI_{\mathcal{V}}$ by Proposition \ref{prop:linear-map-product-algebra}(ii),(iii). Hence $[T]_{\gamma'\leftarrow\beta'}=[I_{\mathcal{W}}TI_{\mathcal{V}}]_{\gamma'\leftarrow\beta'}=[I_{\mathcal{W}}]_{\gamma'\leftarrow\gamma}[T]_{\gamma\leftarrow\beta}[I_{\mathcal{V}}]_{\beta\leftarrow\beta'}$ by Proposition \ref{prop:composition-and-matrix-multiplication} twice. The outer matrices are $C_{\gamma'\leftarrow\gamma}$ and $C_{\beta\leftarrow\beta'}$ by Definition \ref{def:change-of-basis-matrix}.\\
\hlt{Structural Commentary}: Identity maps insert the codomain and domain coordinate transitions around the fixed representation.\\
\hlt{Key Insight}: Output coordinates transform on the left; input coordinates transform on the right.
\end{proof}
\end{theorem}
