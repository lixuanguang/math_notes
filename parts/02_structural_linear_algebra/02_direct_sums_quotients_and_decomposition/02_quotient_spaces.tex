\subsection{Quotient Spaces}
Quotients collapse a prescribed subspace while retaining the linear structure that is independent of the collapsed directions.

Throughout this subsection, all vector spaces are over the same fixed field $\mathbb{F}$.

\begin{definition} \label{def:affine-subset-parallel-subspace} \hlt{Affine Subset Parallel to a Subspace}\\
Let $\mathcal{U}\leq\mathcal{V}$. For $\mathbf{v}\in\mathcal{V}$, the subset $\mathbf{v}+\mathcal{U}\defeq\{\mathbf{v}+\mathbf{u}:\mathbf{u}\in\mathcal{U}\}$ is the \hlt{affine subset} through $\mathbf{v}$ parallel to $\mathcal{U}$.
\end{definition}

\begin{proposition} \label{prop:parallel-affine-subsets-equal-or-disjoint} \hlt{Parallel Affine Subsets Are Equal or Disjoint}\\
Let $\mathcal{U}\leq\mathcal{V}$ and $\mathbf{v},\mathbf{w}\in\mathcal{V}$. The following conditions are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathbf{v}-\mathbf{w}\in\mathcal{U}$;
\item $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$;
\item $(\mathbf{v}+\mathcal{U})\cap(\mathbf{w}+\mathcal{U})\neq\emptyset$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Suppose $\mathbf{v}-\mathbf{w}\in\mathcal{U}$. For $\mathbf{u}\in\mathcal{U}$, $\mathbf{v}+\mathbf{u}=\mathbf{w}+((\mathbf{v}-\mathbf{w})+\mathbf{u})\in\mathbf{w}+\mathcal{U}$, so $\mathbf{v}+\mathcal{U}\subseteq\mathbf{w}+\mathcal{U}$.\\
Since $\mathbf{w}-\mathbf{v}=-(\mathbf{v}-\mathbf{w})\in\mathcal{U}$, exchanging $\mathbf{v}$ and $\mathbf{w}$ establishes the reverse inclusion.
\item[$(ii)$] $\Rightarrow (iii)$ If $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$, then $\mathbf{v}\in(\mathbf{v}+\mathcal{U})\cap(\mathbf{w}+\mathcal{U})$.
\item[$(iii)$] $\Rightarrow (i)$ Choose $\mathbf{x}\in(\mathbf{v}+\mathcal{U})\cap(\mathbf{w}+\mathcal{U})$ and $\mathbf{u}_1,\mathbf{u}_2\in\mathcal{U}$ with $\mathbf{x}=\mathbf{v}+\mathbf{u}_1=\mathbf{w}+\mathbf{u}_2$. Then $\mathbf{v}-\mathbf{w}=\mathbf{u}_2-\mathbf{u}_1\in\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: A cyclic implication proof connects differences, equality, and intersection.\\
\hlt{Key Insight}: A common point exposes the difference of representatives.
\end{proof}
\end{proposition}

\begin{definition} \label{def:quotient-space} \hlt{Underlying Quotient Set}\\
For $\mathcal{U}\leq\mathcal{V}$, the \hlt{quotient set} is $\mathcal{V}/\mathcal{U}\defeq\{\mathbf{v}+\mathcal{U}:\mathbf{v}\in\mathcal{V}\}$.
\end{definition}

\begin{proposition} \label{prop:quotient-relation-congruence} \hlt{Quotient Relation Is a Linear Congruence}\\
Let $\mathcal{U}\leq\mathcal{V}$ and define a relation on $\mathcal{V}$ by $\mathbf{v}\equiv_{\mathcal{U}}\mathbf{w}\Longleftrightarrow\mathbf{v}-\mathbf{w}\in\mathcal{U}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,\mathbf{v}\in\mathcal{V})\,\mathbf{v}\equiv_{\mathcal{U}}\mathbf{v}$.
\item $(\forall\,\mathbf{v},\mathbf{w}\in\mathcal{V})[\mathbf{v}\equiv_{\mathcal{U}}\mathbf{w}\Rightarrow\mathbf{w}\equiv_{\mathcal{U}}\mathbf{v}]$.
\item $(\forall\,\mathbf{u},\mathbf{v},\mathbf{w}\in\mathcal{V})[(\mathbf{u}\equiv_{\mathcal{U}}\mathbf{v}\land\mathbf{v}\equiv_{\mathcal{U}}\mathbf{w})\Rightarrow\mathbf{u}\equiv_{\mathcal{U}}\mathbf{w}]$.
\item $(\forall\,a,b\in\mathbb{F})(\forall\,\mathbf{v}_1,\mathbf{v}_2,\mathbf{w}_1,\mathbf{w}_2\in\mathcal{V})[(\mathbf{v}_1\equiv_{\mathcal{U}}\mathbf{v}_2\land\mathbf{w}_1\equiv_{\mathcal{U}}\mathbf{w}_2)\Rightarrow a\mathbf{v}_1+b\mathbf{w}_1\equiv_{\mathcal{U}}a\mathbf{v}_2+b\mathbf{w}_2]$.
\end{enumerate}
Thus $\equiv_{\mathcal{U}}$ is an equivalence relation compatible with linear combinations.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{v}\in\mathcal{V}$, $\mathbf{v}-\mathbf{v}=\mathbf{0}\in\mathcal{U}$.
\item If $\mathbf{v}-\mathbf{w}\in\mathcal{U}$, then $\mathbf{w}-\mathbf{v}=-(\mathbf{v}-\mathbf{w})\in\mathcal{U}$.
\item If $\mathbf{u}-\mathbf{v},\mathbf{v}-\mathbf{w}\in\mathcal{U}$, then $\mathbf{u}-\mathbf{w}=(\mathbf{u}-\mathbf{v})+(\mathbf{v}-\mathbf{w})\in\mathcal{U}$.
\item If $\mathbf{v}_1-\mathbf{v}_2,\mathbf{w}_1-\mathbf{w}_2\in\mathcal{U}$, then $(a\mathbf{v}_1+b\mathbf{w}_1)-(a\mathbf{v}_2+b\mathbf{w}_2)=a(\mathbf{v}_1-\mathbf{v}_2)+b(\mathbf{w}_1-\mathbf{w}_2)\in\mathcal{U}$ for $a,b\in\mathbb{F}$.
\end{enumerate}
\hlt{Structural Commentary}: The four items verify the equivalence axioms and linear compatibility.\\
\hlt{Key Insight}: Linear combinations preserve differences lying in the subspace.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:quotient-cosets-as-equivalence-classes} \hlt{Cosets as Equivalence Classes}\\
The equivalence class of $\mathbf{v}$ under $\equiv_{\mathcal{U}}$ is $\mathbf{v}+\mathcal{U}$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}. Hence the quotient set in Definition \ref{def:quotient-space} is the abstract quotient set from Definition \ref{def:quotient-set-by-equivalence-relation}. It remembers a vector only up to addition by an element of $\mathcal{U}$.
\end{perspective}

\begin{proposition} \label{prop:quotient-space-structure} \hlt{Quotient-Space Structure}\\
Let $\mathcal{U}\leq\mathcal{V}$. For $\mathbf{v},\mathbf{w}\in\mathcal{V}$ and $a\in\mathbb{F}$, set $(\mathbf{v}+\mathcal{U})+(\mathbf{w}+\mathcal{U})\defeq(\mathbf{v}+\mathbf{w})+\mathcal{U}$ and $a(\mathbf{v}+\mathcal{U})\defeq a\mathbf{v}+\mathcal{U}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item These operations are independent of the chosen representatives.
\item With these operations, $\mathcal{V}/\mathcal{U}$ is a vector space, called the \hlt{quotient space} of $\mathcal{V}$ by $\mathcal{U}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Suppose $\mathbf{v}+\mathcal{U}=\mathbf{v}'+\mathcal{U}$ and $\mathbf{w}+\mathcal{U}=\mathbf{w}'+\mathcal{U}$. Then $\mathbf{v}-\mathbf{v}',\mathbf{w}-\mathbf{w}'\in\mathcal{U}$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}.\\
Hence $(\mathbf{v}+\mathbf{w})-(\mathbf{v}'+\mathbf{w}')\in\mathcal{U}$ by Proposition \ref{prop:quotient-relation-congruence}(iv), and $a\mathbf{v}-a\mathbf{v}'=a(\mathbf{v}-\mathbf{v}')\in\mathcal{U}$.\\
The resulting sum and scalar-multiple cosets are equal by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}.
\item Write $[\mathbf{v}]\defeq\mathbf{v}+\mathcal{U}$. For $\mathbf{u},\mathbf{v},\mathbf{w}\in\mathcal{V}$ and $a,b\in\mathbb{F}$, $([\mathbf{u}]+[\mathbf{v}])+[\mathbf{w}]=[\mathbf{u}+\mathbf{v}+\mathbf{w}]=[\mathbf{u}]+([\mathbf{v}]+[\mathbf{w}])$, $[\mathbf{v}]+[\mathbf{w}]=[\mathbf{w}]+[\mathbf{v}]$, $[\mathbf{0}]=\mathcal{U}$ is an additive identity, and $[-\mathbf{v}]$ is the additive inverse of $[\mathbf{v}]$.\\
The scalar identities $1[\mathbf{v}]=[\mathbf{v}]$, $a(b[\mathbf{v}])=(ab)[\mathbf{v}]$, $a([\mathbf{v}]+[\mathbf{w}])=a[\mathbf{v}]+a[\mathbf{w}]$, and $(a+b)[\mathbf{v}]=a[\mathbf{v}]+b[\mathbf{v}]$ hold by the vector-space identities in $\mathcal{V}$. Thus all vector-space axioms hold.
\end{enumerate}
\hlt{Structural Commentary}: Representative independence precedes direct verification of the vector-space axioms.\\
\hlt{Key Insight}: Coset arithmetic inherits every identity from representatives.
\end{proof}
\end{proposition}

\begin{definition} \label{def:quotient-map} \hlt{Quotient Map}\\
For $\mathcal{U}\leq\mathcal{V}$, the \hlt{quotient map} is $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ defined by $\pi(\mathbf{v})\defeq\mathbf{v}+\mathcal{U}$.
\end{definition}

\begin{proposition} \label{prop:quotient-map-properties} \hlt{Properties of the Quotient Map}\\
Let $\mathcal{U}\leq\mathcal{V}$ and define $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ by $\pi(\mathbf{v})=\mathbf{v}+\mathcal{U}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\pi$ is linear.
\item $\pi$ is surjective.
\item $\ker(\pi)=\mathcal{U}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{v},\mathbf{w}\in\mathcal{V}$ and $a,b\in\mathbb{F}$, $\pi(a\mathbf{v}+b\mathbf{w})=(a\mathbf{v}+b\mathbf{w})+\mathcal{U}=a\pi(\mathbf{v})+b\pi(\mathbf{w})$.
\item If $\mathbf{x}\in\mathcal{V}/\mathcal{U}$, choose $\mathbf{v}\in\mathcal{V}$ with $\mathbf{x}=\mathbf{v}+\mathcal{U}=\pi(\mathbf{v})$ by the definition of $\mathcal{V}/\mathcal{U}$.
\item For $\mathbf{v}\in\mathcal{V}$, $\pi(\mathbf{v})=\mathcal{U}\Longleftrightarrow\mathbf{v}\in\mathcal{U}$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}. Therefore $\ker(\pi)=\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: The proof verifies linearity, surjectivity, and the kernel in statement order.\\
\hlt{Key Insight}: The zero coset is exactly the collapsed subspace.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-quotient} \hlt{Dimension of a Quotient}\\
Let $\mathcal{V}$ be finite-dimensional and $\mathcal{U}\leq\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{V}/\mathcal{U}$ is finite-dimensional.
\item $\dim(\mathcal{V}/\mathcal{U})=\dim(\mathcal{V})-\dim(\mathcal{U})$.
\end{enumerate}
\begin{proof}
Let $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ be the quotient map. Its range is $\mathcal{V}/\mathcal{U}$ and its kernel is $\mathcal{U}$ by Proposition \ref{prop:quotient-map-properties}(ii)--(iii).
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The range $\mathcal{V}/\mathcal{U}$ is finite-dimensional by Theorem \ref{thm:rank-nullity}(i).
\item The formula in (ii) holds by Theorem \ref{thm:rank-nullity}(ii) with $\ker(\pi)=\mathcal{U}$ and $\operatorname{range}(\pi)=\mathcal{V}/\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: Quotient-map properties followed by rank-nullity handle finiteness and dimension.\\
\hlt{Key Insight}: Quotient dimension is the rank of the quotient map.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:subspace-correspondence-quotient} \hlt{Subspace Correspondence for Quotients}\\
Let $\mathcal{U}\leq\mathcal{V}$, let $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ be the quotient map, and set $\mathfrak{A}\defeq\{\mathcal{T}\leq\mathcal{V}:\mathcal{U}\leq\mathcal{T}\}$ and $\mathfrak{B}\defeq\{\mathcal{Y}:\mathcal{Y}\leq\mathcal{V}/\mathcal{U}\}$. For $\mathcal{T}\in\mathfrak{A}$ and $\mathcal{Y}\in\mathfrak{B}$, set $\Phi(\mathcal{T})\defeq\mathcal{T}/\mathcal{U}=\pi(\mathcal{T})$ and $\Psi(\mathcal{Y})\defeq\pi^{-1}(\mathcal{Y})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\Phi:\mathfrak{A}\to\mathfrak{B}$ and $\Psi:\mathfrak{B}\to\mathfrak{A}$ are well-defined.
\item $\Phi$ and $\Psi$ are inverse bijections: $\Psi\Phi=I_{\mathfrak{A}}$ and $\Phi\Psi=I_{\mathfrak{B}}$.
\item Both maps preserve inclusion.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\mathcal{T}\in\mathfrak{A}$. The set $\mathcal{T}/\mathcal{U}$ contains $\mathcal{U}=\mathbf{0}+\mathcal{U}$, and $a(\mathbf{t}_1+\mathcal{U})+b(\mathbf{t}_2+\mathcal{U})=(a\mathbf{t}_1+b\mathbf{t}_2)+\mathcal{U}\in\mathcal{T}/\mathcal{U}$ for $\mathbf{t}_1,\mathbf{t}_2\in\mathcal{T}$ and $a,b\in\mathbb{F}$.\\
Hence $\Phi(\mathcal{T})\leq\mathcal{V}/\mathcal{U}$ by Theorem \ref{thm:subspacetest}. Let $\mathcal{Y}\in\mathfrak{B}$. The set $\pi^{-1}(\mathcal{Y})$ contains $\mathbf{0}_{\mathcal{V}}$, and $\pi(a\mathbf{v}+b\mathbf{w})=a\pi(\mathbf{v})+b\pi(\mathbf{w})\in\mathcal{Y}$ for $\mathbf{v},\mathbf{w}\in\pi^{-1}(\mathcal{Y})$ and $a,b\in\mathbb{F}$; hence $\pi^{-1}(\mathcal{Y})\leq\mathcal{V}$ by Theorem \ref{thm:subspacetest}. Moreover, $\mathcal{U}=\ker(\pi)\subseteq\pi^{-1}(\mathcal{Y})$ by Proposition \ref{prop:quotient-map-properties}(iii).
\item Let $\mathcal{T}\in\mathfrak{A}$. For $\mathbf{t}\in\mathcal{T}$, $\pi(\mathbf{t})\in\pi(\mathcal{T})$, so $\mathcal{T}\subseteq\pi^{-1}(\pi(\mathcal{T}))$. If $\mathbf{v}\in\pi^{-1}(\pi(\mathcal{T}))$, choose $\mathbf{t}\in\mathcal{T}$ with $\mathbf{v}+\mathcal{U}=\mathbf{t}+\mathcal{U}$.\\
Then $\mathbf{v}-\mathbf{t}\in\mathcal{U}\subseteq\mathcal{T}$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}, so $\mathbf{v}\in\mathcal{T}$. Thus $\Psi(\Phi(\mathcal{T}))=\mathcal{T}$. Let $\mathcal{Y}\in\mathfrak{B}$.\\
If $\mathbf{v}\in\pi^{-1}(\mathcal{Y})$, then $\pi(\mathbf{v})\in\mathcal{Y}$, so $\pi(\pi^{-1}(\mathcal{Y}))\subseteq\mathcal{Y}$. If $\mathbf{y}\in\mathcal{Y}$, choose $\mathbf{v}\in\mathcal{V}$ with $\pi(\mathbf{v})=\mathbf{y}$ by Proposition \ref{prop:quotient-map-properties}(ii).\\
Then $\mathbf{v}\in\pi^{-1}(\mathcal{Y})$, so $\mathbf{y}\in\pi(\pi^{-1}(\mathcal{Y}))$. Therefore $\Phi(\Psi(\mathcal{Y}))=\mathcal{Y}$.
\item If $\mathcal{T}_1\subseteq\mathcal{T}_2$ in $\mathfrak{A}$, then $\pi(\mathcal{T}_1)\subseteq\pi(\mathcal{T}_2)$. If $\mathcal{Y}_1\subseteq\mathcal{Y}_2$ in $\mathfrak{B}$, then $\pi^{-1}(\mathcal{Y}_1)\subseteq\pi^{-1}(\mathcal{Y}_2)$.
\end{enumerate}
\hlt{Structural Commentary}: Well-defined assignments, inverse identities, and monotonicity establish the correspondence.\\
\hlt{Key Insight}: Images and preimages undo the quotient construction exactly.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:quotient-universal-property} \hlt{Universal Property of the Quotient Map}\\
Let $\mathcal{U}\leq\mathcal{V}$, let $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ be the quotient map, and let $T\in\LL(\mathcal{V},\mathcal{W})$ satisfy $\mathcal{U}\leq\ker(T)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\exists!\,\overline{T}\in\LL(\mathcal{V}/\mathcal{U},\mathcal{W}))[\overline{T}\pi=T\land(\forall\,\mathbf{v}\in\mathcal{V})\,\overline{T}(\mathbf{v}+\mathcal{U})=T\mathbf{v}]$.
\item The map $\overline{T}$ from (i) satisfies $\ker(\overline{T})=\ker(T)/\mathcal{U}$.
\item The map $\overline{T}$ from (i) satisfies $\operatorname{range}(\overline{T})=\operatorname{range}(T)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Define $\overline{T}(\mathbf{v}+\mathcal{U})\defeq T\mathbf{v}$. If $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$, then $\mathbf{v}-\mathbf{w}\in\mathcal{U}\leq\ker(T)$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}, so $T\mathbf{v}-T\mathbf{w}=T(\mathbf{v}-\mathbf{w})=\mathbf{0}$.\\
Hence $\overline{T}$ is well-defined. For $a,b\in\mathbb{F}$ and $\mathbf{v},\mathbf{w}\in\mathcal{V}$, $\overline{T}(a(\mathbf{v}+\mathcal{U})+b(\mathbf{w}+\mathcal{U}))=T(a\mathbf{v}+b\mathbf{w})=aT\mathbf{v}+bT\mathbf{w}$, so $\overline{T}$ is linear.\\
Also, $(\overline{T}\pi)(\mathbf{v})=\overline{T}(\mathbf{v}+\mathcal{U})=T\mathbf{v}$ for $\mathbf{v}\in\mathcal{V}$, so $\overline{T}\pi=T$.\\
If $S\in\LL(\mathcal{V}/\mathcal{U},\mathcal{W})$ satisfies $S\pi=T$, then $(\forall\,\mathbf{v}\in\mathcal{V})\,S(\mathbf{v}+\mathcal{U})=S\pi(\mathbf{v})=T\mathbf{v}=\overline{T}(\mathbf{v}+\mathcal{U})$. Since $\mathcal{V}/\mathcal{U}=\{\mathbf{v}+\mathcal{U}:\mathbf{v}\in\mathcal{V}\}$, $S=\overline{T}$.
\item For $\mathbf{v}\in\mathcal{V}$, $\mathbf{v}+\mathcal{U}\in\ker(\overline{T})\Longleftrightarrow T\mathbf{v}=\mathbf{0}\Longleftrightarrow\mathbf{v}\in\ker(T)$. Hence $\ker(\overline{T})=\{\mathbf{v}+\mathcal{U}:\mathbf{v}\in\ker(T)\}=\ker(T)/\mathcal{U}$.
\item The equality $\operatorname{range}(\overline{T})=\{\overline{T}(\mathbf{v}+\mathcal{U}):\mathbf{v}\in\mathcal{V}\}=\{T\mathbf{v}:\mathbf{v}\in\mathcal{V}\}=\operatorname{range}(T)$ is immediate from the defining formula.
\end{enumerate}
\hlt{Structural Commentary}: Construction establishes factorisation; set calculations identify kernel and range.\\
\hlt{Key Insight}: Killing $\mathcal{U}$ is precisely the condition for descent.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:quotient-by-kernel-is-range} \hlt{Quotient by the Kernel}\\
Let $T\in\LL(\mathcal{V},\mathcal{W})$ and define $\widetilde{T}(\mathbf{v}+\ker(T))\defeq T\mathbf{v}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The assignment defines a well-defined linear map $\widetilde{T}:\mathcal{V}/\ker(T)\to\mathcal{W}$.
\item $\widetilde{T}$ is injective.
\item $\operatorname{range}(\widetilde{T})=\operatorname{range}(T)$.
\item The codomain restriction $\widehat{T}:\mathcal{V}/\ker(T)\to\operatorname{range}(T)$ is an isomorphism; hence $\mathcal{V}/\ker(T)\cong\operatorname{range}(T)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The claimed well-defined linear map exists by Theorem \ref{thm:quotient-universal-property}(i) with $\mathcal{U}=\ker(T)$.
\item The identity $\ker(\widetilde{T})=\ker(T)/\ker(T)=\{\ker(T)\}$ holds by Theorem \ref{thm:quotient-universal-property}(ii). Thus $\widetilde{T}$ is injective by Proposition \ref{prop:injective-iff-kernel-zero}.
\item The range identity holds by Theorem \ref{thm:quotient-universal-property}(iii).
\item The codomain restriction $\widehat{T}$ is linear and injective by parts (i)--(ii), and it is surjective by part (iii).\\
Therefore $\widehat{T}$ is invertible by Proposition \ref{prop:linear-map-invertible-iff-bijective}, and $\mathcal{V}/\ker(T)\cong\operatorname{range}(T)$.
\end{enumerate}
\hlt{Structural Commentary}: The universal property constructs the map; kernel and range identities produce the isomorphism.\\
\hlt{Key Insight}: Removing the kernel makes the induced map injective.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:second-isomorphism-vector-spaces} \hlt{Second Isomorphism Theorem}\\
If $\mathcal{S},\mathcal{T}\leq\mathcal{V}$, then $\mathcal{S}/(\mathcal{S}\cap\mathcal{T})\cong(\mathcal{S}+\mathcal{T})/\mathcal{T}$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Construction}: The sum $\mathcal{S}+\mathcal{T}$ is a subspace containing $\mathcal{T}$ by Proposition \ref{prop:sum-subspaces-smallest-containing-subspace}(i). Define $L:\mathcal{S}\to(\mathcal{S}+\mathcal{T})/\mathcal{T}$ by $L\mathbf{s}\defeq\mathbf{s}+\mathcal{T}$.\\
For $\mathbf{s},\mathbf{r}\in\mathcal{S}$ and $a,b\in\mathbb{F}$, $L(a\mathbf{s}+b\mathbf{r})=(a\mathbf{s}+b\mathbf{r})+\mathcal{T}=aL\mathbf{s}+bL\mathbf{r}$, so $L$ is linear.
\item \hlt{Kernel}: For $\mathbf{s}\in\mathcal{S}$, $L\mathbf{s}=\mathcal{T}\Longleftrightarrow\mathbf{s}\in\mathcal{T}$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}. Hence $\ker(L)=\mathcal{S}\cap\mathcal{T}$.
\item \hlt{Surjectivity}: Each coset equals $(\mathbf{s}+\mathbf{t})+\mathcal{T}=\mathbf{s}+\mathcal{T}=L\mathbf{s}$ for suitable $\mathbf{s}\in\mathcal{S}$ and $\mathbf{t}\in\mathcal{T}$; hence $L$ is surjective.
\item \hlt{Conclusion}: The equality $\operatorname{range}(L)=(\mathcal{S}+\mathcal{T})/\mathcal{T}$ holds by part (iii).\\
Therefore $\mathcal{S}/(\mathcal{S}\cap\mathcal{T})\cong(\mathcal{S}+\mathcal{T})/\mathcal{T}$ by Theorem \ref{thm:quotient-by-kernel-is-range} with the linearity and kernel identity from parts (i)--(ii).
\end{enumerate}
\hlt{Structural Commentary}: Construction, kernel, surjectivity, and the first isomorphism theorem form four operative stages.\\
\hlt{Key Insight}: The target forgets exactly $\mathcal{S}\cap\mathcal{T}$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:third-isomorphism-vector-spaces} \hlt{Third Isomorphism Theorem}\\
If $\mathcal{U}\leq\mathcal{T}\leq\mathcal{V}$, then $(\mathcal{V}/\mathcal{U})/(\mathcal{T}/\mathcal{U})\cong\mathcal{V}/\mathcal{T}$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Construction}: The inclusion $\mathcal{T}/\mathcal{U}\leq\mathcal{V}/\mathcal{U}$ holds by Theorem \ref{thm:subspace-correspondence-quotient}(i). Define $L:\mathcal{V}/\mathcal{U}\to\mathcal{V}/\mathcal{T}$ by $L(\mathbf{v}+\mathcal{U})\defeq\mathbf{v}+\mathcal{T}$.
\item \hlt{Well-Definedness and Linearity}: If $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$, then $\mathbf{v}-\mathbf{w}\in\mathcal{U}\subseteq\mathcal{T}$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}, so $\mathbf{v}+\mathcal{T}=\mathbf{w}+\mathcal{T}$ by the same proposition. For $a,b\in\mathbb{F}$, $L(a(\mathbf{v}+\mathcal{U})+b(\mathbf{w}+\mathcal{U}))=(a\mathbf{v}+b\mathbf{w})+\mathcal{T}=aL(\mathbf{v}+\mathcal{U})+bL(\mathbf{w}+\mathcal{U})$.
\item \hlt{Kernel and Range}: For $\mathbf{v}\in\mathcal{V}$, $L(\mathbf{v}+\mathcal{U})=\mathcal{T}\Longleftrightarrow\mathbf{v}\in\mathcal{T}$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}; hence $\ker(L)=\mathcal{T}/\mathcal{U}$.\\
Every $\mathbf{v}+\mathcal{T}\in\mathcal{V}/\mathcal{T}$ equals $L(\mathbf{v}+\mathcal{U})$, so $L$ is surjective.
\item \hlt{Conclusion}: The equality $\operatorname{range}(L)=\mathcal{V}/\mathcal{T}$ holds by part (iii).\\
Therefore $(\mathcal{V}/\mathcal{U})/(\mathcal{T}/\mathcal{U})\cong\mathcal{V}/\mathcal{T}$ by Theorem \ref{thm:quotient-by-kernel-is-range} with the linearity and kernel identity from parts (ii)--(iii).
\end{enumerate}
\hlt{Structural Commentary}: Construction, kernel computation, and the first isomorphism theorem identify the iterated quotient.\\
\hlt{Key Insight}: The nested collapse has kernel $\mathcal{T}/\mathcal{U}$.
\end{proof}
\end{theorem}

Quotient spaces retain precisely the structure invariant under translation by a chosen subspace. Their correspondence, universal property, and isomorphism theorems formalise which data survive that collapse.
