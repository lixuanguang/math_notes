\subsection{Decomposition and Complementation}
\begin{theorem} \label{thm:dimension-formula-for-sum} \hlt{Dimension Formula for a Sum}\\
If $\mathcal{U}, \mathcal{W}$ are finite-dimensional subspaces of $\mathcal{V}$, then $\dim(\mathcal{U} + \mathcal{W}) = \dim(\mathcal{U}) + \dim(\mathcal{W}) - \dim(\mathcal{U} \cap \mathcal{W})$.
\begin{proof}
Choose a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r)$ of $\mathcal{U} \cap \mathcal{W}$, extend it to a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p)$ of $\mathcal{U}$, and to a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{w}_1, \ldots, \mathbf{w}_q)$ of $\mathcal{W}$. Then $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p, \mathbf{w}_1, \ldots, \mathbf{w}_q) $ spans $\mathcal{U}+\mathcal{W}$ because every vector in the sum is the sum of one vector from $\mathcal{U}$ and one vector from $\mathcal{W}$.\\
To prove linear independence, suppose $a_1\mathbf{e}_1+\cdots+a_r\mathbf{e}_r+b_1\mathbf{u}_1+\cdots+b_p\mathbf{u}_p+c_1\mathbf{w}_1+\cdots+c_q\mathbf{w}_q=\mathbf{0}$.
Then $a_1\mathbf{e}_1+\cdots+a_r\mathbf{e}_r+b_1\mathbf{u}_1+\cdots+b_p\mathbf{u}_p$ lies in $\mathcal{U}\cap\mathcal{W}$. Since $(\mathbf{e}_1,\ldots,\mathbf{e}_r)$ is a basis of $\mathcal{U}\cap\mathcal{W}$ and $(\mathbf{e}_1,\ldots,\mathbf{e}_r,\mathbf{u}_1,\ldots,\mathbf{u}_p)$ is a basis of $\mathcal{U}$, we get $b_1=\cdots=b_p=0$. The remaining relation lies in the basis $(\mathbf{e}_1,\ldots,\mathbf{e}_r,\mathbf{w}_1,\ldots,\mathbf{w}_q)$ of $\mathcal{W}$, so $a_1=\cdots=a_r=c_1=\cdots=c_q=0$. Thus the displayed list is a basis of $\mathcal{U}+\mathcal{W}$.\\
Therefore $\dim(\mathcal{U} + \mathcal{W}) = r + p + q = (r+p) + (r+q) - r$.\\
\hlt{Structural Commentary}: The formula corrects double-counting: vectors in $\mathcal{U} \cap \mathcal{W}$ are counted once in each subspace. Extending an intersection basis separates shared directions from new directions.\\
\hlt{Key Insight}: The dimension of a sum is the sum of dimensions minus the overlap.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:dimension-direct-sum} \hlt{Dimension of a Direct Sum}\\
If $\mathcal{U}$ and $\mathcal{W}$ are finite-dimensional subspaces of $\mathcal{V}$ with $\mathcal{U} \cap \mathcal{W} = \{ \mathbf{0} \}$, then $\dim(\mathcal{U} \oplus \mathcal{W}) = \dim(\mathcal{U}) + \dim(\mathcal{W})$.
\begin{proof}
Since $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$, the sum is direct and $\dim(\mathcal{U}\cap\mathcal{W})=0$. The result follows from Theorem \ref{thm:dimension-formula-for-sum}.\\
\hlt{Structural Commentary}: When the overlap is trivial, there is no double-counting in the dimension formula. Direct sums are exactly the case where the overlap term vanishes.\\
\hlt{Key Insight}: Dimensions add exactly for direct sums.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:large-subspaces-intersect} \hlt{Large Subspaces Must Intersect}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of a finite-dimensional vector space $\mathcal{V}$, and let $k$ be a positive integer.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\dim(\mathcal{U})+\dim(\mathcal{W})>\dim(\mathcal{V})$, then $\mathcal{U}\cap\mathcal{W}$ contains a nonzero vector.
\item If $\dim(\mathcal{U})+\dim(\mathcal{W})\geq\dim(\mathcal{V})+k$, then $\mathcal{U}\cap\mathcal{W}$ contains $k$ linearly independent vectors.
\end{enumerate}
\begin{proof} Since $\mathcal{U}+\mathcal{W}\subseteq\mathcal{V}$, Theorem \ref{thm:dimension-formula-for-sum} gives\\
$\dim(\mathcal{U}\cap\mathcal{W})=\dim(\mathcal{U})+\dim(\mathcal{W})-\dim(\mathcal{U}+\mathcal{W})\geq\dim(\mathcal{U})+\dim(\mathcal{W})-\dim(\mathcal{V})$.\\
If $\dim(\mathcal{U})+\dim(\mathcal{W})>\dim(\mathcal{V})$, the right-hand side is at least $1$, so the intersection contains a nonzero vector.\\
If $\dim(\mathcal{U})+\dim(\mathcal{W})\geq\dim(\mathcal{V})+k$, then $\dim(\mathcal{U}\cap\mathcal{W})\geq k$, so intersection contains $k$ linearly independent vectors.\\
\hlt{Structural Commentary}: Excess dimension appears as overlap.\\
\hlt{Key Insight}: Two large subspaces cannot fit inside $\mathcal{V}$ without sharing directions.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:direct-sum-iff-dimensions-add} \hlt{Direct Sums and Dimension}\\
Suppose $\mathcal{V}$ is finite-dimensional and $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a direct sum if and only if $\dim(\mathcal{U}_1+\cdots+\mathcal{U}_m) = \dim(\mathcal{U}_1)+\cdots+\dim(\mathcal{U}_m)$.
\begin{proof}
For each $j$, choose a basis $\beta_j$ of $\mathcal{U}_j$, and concatenate these bases into a list $\beta$.\\
This list spans $\mathcal{U}_1+\cdots+\mathcal{U}_m$ and has length $\dim(\mathcal{U}_1)+\cdots+\dim(\mathcal{U}_m)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If the sum is direct, any linear relation among the vectors in $\beta$ gives a zero decomposition $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ with $\mathbf{u}_j\in\mathcal{U}_j$. Directness gives $\mathbf{u}_j=\mathbf{0}$ for every $j$, and independence of each $\beta_j$ gives all coefficients zero. Thus $\beta$ is a basis of the sum, and the dimensions add.
\item[$(\Leftarrow)$] If the dimensions add, then $\beta$ is a spanning list whose length equals the dimension of the sum, so Proposition \ref{prop:full-dimension-basis-criteria} makes $\beta$ a basis. A zero decomposition $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ expands to a zero linear relation in this basis, so every $\mathbf{u}_j=\mathbf{0}$. Hence the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: Direct sums add dimensions without hidden overlap.\\
\hlt{Key Insight}: Equality of dimensions means the concatenated bases have no linear redundancy.
\end{proof}
\end{theorem}

\begin{definition} \label{def:adapted-basis-direct-sum} \hlt{Basis Adapted to a Direct Sum}\\
Suppose $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_m$ and each $\beta_j$ is a basis of $\mathcal{U}_j$. The concatenated basis $\beta=(\beta_1,\ldots,\beta_m)$ of $\mathcal{V}$ is called a \hlt{basis adapted to the direct-sum decomposition}.\\
Equivalently, the vectors in an adapted basis are grouped by the summands whose direct sum is $\mathcal{V}$.
\end{definition}

\begin{proposition} \label{prop:complements-exist-finite-dimensions} \hlt{Complements Exist in Finite Dimensions}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U} \subseteq \mathcal{V}$ is a subspace, then there exists a subspace $\mathcal{W}$ such that $\mathcal{V} = \mathcal{U} \oplus \mathcal{W}$.
\begin{proof}
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ of $\mathcal{U}$ and extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_n)$ of $\mathcal{V}$.\\
Let $\mathcal{W}=\operatorname{span}(\mathbf{w}_1,\ldots,\mathbf{w}_n)$. Every vector in $\mathcal{V}$ is a linear combination of this extended basis, so $\mathcal{V}=\mathcal{U}+\mathcal{W}$. If $\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$, then $\mathbf{v}$ has one representation using only $\mathbf{u}_1,\ldots,\mathbf{u}_m$ and another using only $\mathbf{w}_1,\ldots,\mathbf{w}_n$.\\
Uniqueness of coefficients in the basis of $\mathcal{V}$ forces $\mathbf{v}=\mathbf{0}$. Hence $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.\\
\hlt{Structural Commentary}: Complements split a space into a subspace and the remaining basis directions.\\
\hlt{Key Insight}: Extend a basis of $\mathcal{U}$ to split $\mathcal{V}$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:complements-exist-arbitrary-vector-spaces} \hlt{Complements Exist}\\
Assuming Zorn's lemma, for every $\mathcal{U}\leq\mathcal{V}$, $\exists\,\mathcal{W}\leq\mathcal{V}$ such that $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.
\begin{proof}
By Theorem \ref{thm:hamel-basis-extension}, choose a Hamel basis $\mathcal{B}_{\mathcal{U}}$ of $\mathcal{U}$ and a Hamel basis $\mathcal{B}\supseteq\mathcal{B}_{\mathcal{U}}$ of $\mathcal{V}$.\\
Put $\mathcal{C}=\mathcal{B}\setminus\mathcal{B}_{\mathcal{U}}$ and $\mathcal{W}=\operatorname{span}(\mathcal{C})$. For $\mathbf{v}\in\mathcal{V}$, its finite $\mathcal{B}$-expansion splits as $\mathbf{v}=\sum\limits_{b\in F_{\mathcal{U}}}a_b b+\sum\limits_{c\in F_{\mathcal{C}}}a_c c\in\mathcal{U}+\mathcal{W}$, with finite $F_{\mathcal{U}}\subseteq\mathcal{B}_{\mathcal{U}}$ and $F_{\mathcal{C}}\subseteq\mathcal{C}$.
If $\mathbf{x}\in\mathcal{U}\cap\mathcal{W}$, then its $\mathcal{B}$-coordinates are supported in both $\mathcal{B}_{\mathcal{U}}$ and $\mathcal{C}$; uniqueness gives $\mathbf{x}=\mathbf{0}$. Hence $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.\\
\hlt{Structural Commentary}: Extend a basis of $\mathcal{U}$; span the new directions.\\
\hlt{Key Insight}: The uniqueness of basis expansion becomes the uniqueness of direct-sum decomposition.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:rank-nullity-complement-view} \hlt{Rank-Nullity as a Complement Isomorphism}\\
Let $T\in\LL(\mathcal{V},\mathcal{W})$, and suppose $\mathcal{V}=\ker(T)\oplus\mathcal{M}$. Then $T|_{\mathcal{M}}:\mathcal{M}\to\operatorname{range}(T)$ is an isomorphism.
\begin{proof}
Indeed, $\ker(T|_{\mathcal{M}})=\mathcal{M}\cap\ker(T)=\{\mathbf{0}\}$, and every $T\mathbf{v}$ equals $T\mathbf{m}$ for the decomposition $\mathbf{v}=\mathbf{k}+\mathbf{m}$.\\
Thus rank-nullity counts $\mathcal{V}=\ker(T)\oplus\mathcal{M}$ with $\mathcal{M}\cong\operatorname{range}(T)$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:finite-dimensional-one-sided-inverses-split} \hlt{One-Sided Inverses Split in Finite Dimensions}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional vector spaces and let $T\in\LL(\mathcal{V},\mathcal{W})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $T$ is injective, then there exists $L\in\LL(\mathcal{W},\mathcal{V})$ such that $LT=I_{\mathcal{V}}$.
\item If $T$ is surjective, then there exists $R\in\LL(\mathcal{W},\mathcal{V})$ such that $TR=I_{\mathcal{W}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $\mathcal{H}\subseteq\mathcal{W}$ with $\mathcal{W}=\operatorname{range}(T)\oplus\mathcal{H}$. For $\mathbf{w}=T\mathbf{v}+\mathbf{h}$, define $L\mathbf{w}=\mathbf{v}$. This is well-defined since the direct-sum decomposition is unique and $T$ is injective. It is linear, and $LT\mathbf{v}=\mathbf{v}$ for every $\mathbf{v}\in\mathcal{V}$.
\item Choose $\mathcal{H}\subseteq\mathcal{V}$ with $\mathcal{V}=\ker(T)\oplus\mathcal{H}$. Then $T|_{\mathcal{H}}:\mathcal{H}\to\mathcal{W}$ is injective since $\mathcal{H}\cap\ker(T)=\{\mathbf{0}\}$, and surjective since each $\mathbf{v}=\mathbf{k}+\mathbf{h}$ satisfies $T\mathbf{v}=T\mathbf{h}$. Let $R=(T|_{\mathcal{H}})^{-1}$, viewed as a map $\mathcal{W}\to\mathcal{V}$.\\
Then $R$ is linear and $TR=I_{\mathcal{W}}$.
\end{enumerate}
\hlt{Structural Commentary}: Complements supply the missing directions on which a one-sided inverse is chosen.\\
\hlt{Key Insight}: Finite-dimensional vector spaces split kernels and ranges.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:quotient-by-complement-is-complement} \hlt{Quotient by a Complemented Subspace}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$ with $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$. Then $J:\mathcal{W}\to\mathcal{V}/\mathcal{U}$, $J(\mathbf{w})=\mathbf{w}+\mathcal{U}$ is an isomorphism.\\
Hence every complement of $\mathcal{U}$ is isomorphic to $\mathcal{V}/\mathcal{U}$.
\begin{proof}
The map is linear by the quotient operations from Proposition \ref{prop:quotient-space-structure}.\\
If $J(\mathbf{w})=\mathcal{U}$, then $\mathbf{w}\in\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$, so $J$ is injective.\\
If $\mathbf{v}+\mathcal{U}\in\mathcal{V}/\mathcal{U}$, write $\mathbf{v}=\mathbf{u}+\mathbf{w}$ with $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{W}$. Then $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}=J(\mathbf{w})$, so $J$ is surjective.\\
\hlt{Structural Commentary}: A complement gives canonical representatives for quotient cosets.\\
\hlt{Key Insight}: Collapsing $\mathcal{U}$ leaves exactly the complementary directions visible.
\end{proof}
\end{proposition}

\begin{definition} \label{def:codimension-subspace} \hlt{Codimension}\\
If $\mathcal{U}$ is a subspace of $\mathcal{V}$ and $\dim(\mathcal{V}/\mathcal{U})<\infty$, define $\operatorname{codim}_{\mathcal{V}}(\mathcal{U})\defeq\dim(\mathcal{V}/\mathcal{U})$.\\
Thus, if $\mathcal{V}$ is finite-dimensional, Theorem \ref{thm:dimension-of-quotient} gives $\operatorname{codim}_{\mathcal{V}}(\mathcal{U})=\dim(\mathcal{V})-\dim(\mathcal{U})$.\\
If $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, Proposition \ref{prop:quotient-by-complement-is-complement} gives $\operatorname{codim}_{\mathcal{V}}(\mathcal{U})=\dim(\mathcal{W})$.
\end{definition}

\begin{perspective} \label{per:quotients-split-over-fields} \hlt{Quotients Split Over Fields}\\
In finite-dimensional vector spaces, choosing a complement $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$ identifies $\mathcal{V}/\mathcal{U}$ with $\mathcal{W}$. The quotient map collapses $\mathcal{U}$ and keeps the complementary directions.
\end{perspective}

\begin{proposition} \label{prop:quotient-of-direct-sum-by-direct-sum} \hlt{Quotient of a Direct Sum}\\
If $\mathcal{V}=\mathcal{V}_1\oplus\mathcal{V}_2$ and $\mathcal{U}_j\leq\mathcal{V}_j$ for $j=1,2$, then $\frac{\mathcal{V}}{\mathcal{U}_1\oplus\mathcal{U}_2} \cong \frac{\mathcal{V}_1}{\mathcal{U}_1}\oplus \frac{\mathcal{V}_2}{\mathcal{U}_2}$.
\begin{proof}
Define, using the unique decomposition\\
$\mathbf{v}=\mathbf{v}_1+\mathbf{v}_2$,
$\Phi:\mathcal{V}\to (\mathcal{V}_1/\mathcal{U}_1)\oplus(\mathcal{V}_2/\mathcal{U}_2)$ and $\Phi(\mathbf{v}_1+\mathbf{v}_2) =(\mathbf{v}_1+\mathcal{U}_1,\mathbf{v}_2+\mathcal{U}_2)$.\\
For $\mathbf{x}=\mathbf{x}_1+\mathbf{x}_2$, $\mathbf{y}=\mathbf{y}_1+\mathbf{y}_2$, and $a,b\in\mathbb{F}$, $\Phi(a\mathbf{x}+b\mathbf{y}) =a\Phi(\mathbf{x})+b\Phi(\mathbf{y})$, so $\Phi$ is linear.\\
It is surjective since $(\mathbf{v}_1+\mathcal{U}_1,\mathbf{v}_2+\mathcal{U}_2)=\Phi(\mathbf{v}_1+\mathbf{v}_2)$. Finally, $\ker(\Phi)=\{\mathbf{v}_1+\mathbf{v}_2:\mathbf{v}_1\in\mathcal{U}_1,\ \mathbf{v}_2\in\mathcal{U}_2\} =\mathcal{U}_1\oplus\mathcal{U}_2$.\\
The result follows from Theorem \ref{thm:quotient-by-kernel-is-range}.\\
\hlt{Structural Commentary}: Quotienting respects direct-sum coordinates.\\
\hlt{Key Insight}: Direct-sum coordinates make quotienting componentwise.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:quotients-and-complements-are-not-functorial-warnings} \hlt{Quotient and Complement Warnings}\\
If $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, then $\mathcal{W}\cong\mathcal{V}/\mathcal{U}$, but complements are not functorial in $\mathcal{U}$.\\
For $\mathcal{V}$ countably infinite-dimensional with basis $(\mathbf{e}_n)_{n\geq1}$, put $\mathcal{S}=\operatorname{span}(\mathbf{e}_n:n\geq2)$.\\
Then $\mathcal{S}\cong\mathcal{V}$ and $\mathcal{V}=\mathcal{V}\oplus\{\mathbf{0}\} =\mathcal{S}\oplus\mathbb{F}\mathbf{e}_1$; $\mathcal{V}/\mathcal{V}=\{\mathbf{0}\}$, $\mathcal{V}/\mathcal{S}\cong\mathbb{F}\mathbf{e}_1$.
\end{perspective}
