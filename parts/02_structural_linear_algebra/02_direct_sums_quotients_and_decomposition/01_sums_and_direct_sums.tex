\subsection{Sums and Direct Sums}
Sums combine subspaces inside a common ambient space, while products and external sums assemble separate spaces. Directness records when these decompositions are unique.

Throughout this subsection, all vector spaces are over the same fixed field $\mathbb{F}$.

\begin{definition} \label{def:finite-sums-and-direct-sums-subspaces} \hlt{Sums of Subspaces}\\
Let $m\in\N$ and let $\mathcal{U}_1,\ldots,\mathcal{U}_m\leq\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Their \hlt{sum} is $\mathcal{U}_1+\cdots+\mathcal{U}_m\defeq\left\{\sum\limits_{j=1}^{m}\mathbf{u}_j:(\forall\,j\in\{1,\ldots,m\})\,\mathbf{u}_j\in\mathcal{U}_j\right\}$.
\item For $\mathcal{U},\mathcal{W}\leq\mathcal{V}$, $\mathcal{U}+\mathcal{W}=\{\mathbf{u}+\mathbf{w}:\mathbf{u}\in\mathcal{U},\ \mathbf{w}\in\mathcal{W}\}$.
\item The sum is \hlt{direct} iff $(\forall\,\mathbf{v}\in\mathcal{U}_1+\cdots+\mathcal{U}_m)(\exists!\,(\mathbf{u}_1,\ldots,\mathbf{u}_m)\in\mathcal{U}_1\times\cdots\times\mathcal{U}_m)\,\mathbf{v}=\sum\limits_{j=1}^{m}\mathbf{u}_j$. A direct sum is denoted $\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_m$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:sum-subspaces-smallest-containing-subspace} \hlt{Sum of Subspaces Is the Smallest Containing Subspace}\\
Let $m\in\N$, let $\mathcal{U}_1,\ldots,\mathcal{U}_m\leq\mathcal{V}$, and set $S\defeq\sum\limits_{j=1}^{m}\mathcal{U}_j$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $S\leq\mathcal{V}$, $(\forall\,j\in\{1,\ldots,m\})\,\mathcal{U}_j\subseteq S$, and $(\forall\,\mathcal{W}\leq\mathcal{V})[(\forall\,j\in\{1,\ldots,m\})\,\mathcal{U}_j\subseteq\mathcal{W}\Rightarrow S\subseteq\mathcal{W}]$.
\item $S=\operatorname{span}(\bigcup\limits_{j=1}^{m}\mathcal{U}_j)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,j\in\{1,\ldots,m\})\,\mathbf{0}_{\mathcal{V}}\in\mathcal{U}_j$, so $\mathbf{0}_{\mathcal{V}}\in S$. Let $\mathbf{x},\mathbf{y}\in S$ and $c\in\mathbb{F}$.\\
Choose $(\mathbf{u}_1,\ldots,\mathbf{u}_m),(\mathbf{v}_1,\ldots,\mathbf{v}_m)\in\mathcal{U}_1\times\cdots\times\mathcal{U}_m$ with $\mathbf{x}=\sum\limits_{j=1}^{m}\mathbf{u}_j$ and $\mathbf{y}=\sum\limits_{j=1}^{m}\mathbf{v}_j$.\\
Then $c\mathbf{x}+\mathbf{y}=\sum\limits_{j=1}^{m}(c\mathbf{u}_j+\mathbf{v}_j)\in S$ by $(\forall\,j\in\{1,\ldots,m\})\,c\mathbf{u}_j+\mathbf{v}_j\in\mathcal{U}_j$. Hence $S\leq\mathcal{V}$ by Theorem \ref{thm:subspacetest}.\\
For $j\in\{1,\ldots,m\}$ and $\mathbf{u}\in\mathcal{U}_j$, the sum with $j$th term $\mathbf{u}$ and all other terms $\mathbf{0}_{\mathcal{V}}$ belongs to $S$, so $\mathcal{U}_j\subseteq S$.\\
If $\mathcal{W}\leq\mathcal{V}$ and $(\forall\,j\in\{1,\ldots,m\})\,\mathcal{U}_j\subseteq\mathcal{W}$, then each defining finite sum for $S$ belongs to $\mathcal{W}$ by additive closure; hence $S\subseteq\mathcal{W}$.
\item Set $E\defeq\bigcup\limits_{j=1}^{m}\mathcal{U}_j$. Then $\operatorname{span}(E)\subseteq S$ by Theorem \ref{thm:span-smallest-containing-subspace}(iii) and part (i).\\
The inclusions $E\subseteq\operatorname{span}(E)\leq\mathcal{V}$ hold by Theorem \ref{thm:span-smallest-containing-subspace}(i)--(ii), so $S\subseteq\operatorname{span}(E)$ by part (i). Thus $S=\operatorname{span}(E)$.
\end{enumerate}
\hlt{Structural Commentary}: Closure and minimality establish the sum; span minimality identifies the same subspace.\\
\hlt{Key Insight}: Both constructions are forced by the same generators.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:sum-not-union} \hlt{Sum Is Not Union}\\
Proposition \ref{prop:sum-subspaces-smallest-containing-subspace} identifies $\mathcal{U}+\mathcal{W}$ with the linear closure of $\mathcal{U}\cup\mathcal{W}$. The union itself contains only vectors already in one summand.\\
If neither summand contains the other, choose $\mathbf{u}\in\mathcal{U}\setminus\mathcal{W}$ and $\mathbf{w}\in\mathcal{W}\setminus\mathcal{U}$.\\
Then $\mathbf{u}+\mathbf{w}\notin\mathcal{U}\cup\mathcal{W}$, since membership in either summand would place the other chosen vector there. Thus the sum supplies exactly the mixed vectors missing from the union; this perspective is explanatory and is not used as proof authority.
\end{perspective}

\begin{theorem} \label{thm:finite-union-proper-subspaces-infinite-field} \hlt{Finite Unions of Proper Subspaces}\\
Let $n\in\N$ and let $\mathcal{V}\neq\{\mathbf{0}_{\mathcal{V}}\}$ be over an infinite field $\mathbb{F}$. If $\mathcal{U}_1,\ldots,\mathcal{U}_n\subsetneq\mathcal{V}$ are subspaces, then $\mathcal{V}\neq\bigcup\limits_{j=1}^{n}\mathcal{U}_j$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Minimal cover}: Assume $\mathcal{V}=\bigcup\limits_{j=1}^{n}\mathcal{U}_j$. Choose a minimal subcover $\mathcal{W}_1,\ldots,\mathcal{W}_k$ and set $K\defeq\{1,\ldots,k\}$.\\
Then $(\forall\,i\in K)\,\mathcal{W}_i\not\subseteq\bigcup\limits_{j\in K\setminus\{i\}}\mathcal{W}_j$ by minimality. The case $k=1$ contradicts $\mathcal{W}_1\subsetneq\mathcal{V}$; hence $k\geq2$.
\item \hlt{Affine line}: Choose $\mathbf{w}\in\mathcal{W}_1\setminus\bigcup\limits_{j=2}^{k}\mathcal{W}_j$ and $\mathbf{v}\in\mathcal{V}\setminus\mathcal{W}_1$. Since $k\geq2$ and $\mathbf{0}_{\mathcal{V}}\in\mathcal{W}_2$, $\mathbf{w}\neq\mathbf{0}_{\mathcal{V}}$. Set $A\defeq\{\mathbf{v}+r\mathbf{w}:r\in\mathbb{F}\}$.\\
The parametrization $r\mapsto\mathbf{v}+r\mathbf{w}$ is injective, so $A$ is infinite by the infinitude of $\mathbb{F}$.
\item \hlt{Intersections}: If $\mathbf{v}+r\mathbf{w}\in\mathcal{W}_1$, then $\mathbf{v}=(\mathbf{v}+r\mathbf{w})-r\mathbf{w}\in\mathcal{W}_1$, a contradiction; hence $A\cap\mathcal{W}_1=\emptyset$.\\
For $j\in\{2,\ldots,k\}$, if $r_1\neq r_2$ and $\mathbf{v}+r_1\mathbf{w},\mathbf{v}+r_2\mathbf{w}\in\mathcal{W}_j$, then $(r_1-r_2)\mathbf{w}\in\mathcal{W}_j$, so $\mathbf{w}\in\mathcal{W}_j$, a contradiction. Thus $\abs{A\cap\mathcal{W}_j}\leq1$.
\item \hlt{Contradiction}: The set $A\cap\bigcup\limits_{j=1}^{k}\mathcal{W}_j$ is finite by part (iii), whereas $A\subseteq\mathcal{V}=\bigcup\limits_{j=1}^{k}\mathcal{W}_j$ would make it equal to the infinite set $A$.\\
Therefore the assumed cover does not exist.
\end{enumerate}
\hlt{Structural Commentary}: A minimal cover selects a direction whose affine line meets every summand finitely.\\
\hlt{Key Insight}: An infinite affine line defeats a finite cover.
\end{proof}
\end{theorem}

\begin{definition} \label{def:subspace-meet-and-join} \hlt{Subspace Poset, Meet, and Join}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{subspace poset} is $\operatorname{Sub}(\mathcal{V})\defeq\{\mathcal{U}\subseteq\mathcal{V}:\mathcal{U}\leq\mathcal{V}\}$, ordered by inclusion.
\item For $\{\mathcal{U}_i\}_{i\in I}\subseteq\operatorname{Sub}(\mathcal{V})$, its \hlt{meet} is $\bigwedge\limits_{i\in I}\mathcal{U}_i\defeq\bigcap\limits_{i\in I}\mathcal{U}_i$.
\item For $\{\mathcal{U}_i\}_{i\in I}\subseteq\operatorname{Sub}(\mathcal{V})$, its \hlt{join} is $\bigvee\limits_{i\in I}\mathcal{U}_i\defeq\sum\limits_{i\in I}\mathcal{U}_i\defeq\operatorname{span}(\bigcup\limits_{i\in I}\mathcal{U}_i)$.
\item The empty meet is $\mathcal{V}$, and the empty join is $\{\mathbf{0}_{\mathcal{V}}\}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:subspaces-form-complete-lattice} \hlt{Subspace Meet and Join}\\
Let $\{\mathcal{U}_i\}_{i\in I}\subseteq\operatorname{Sub}(\mathcal{V})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\bigwedge\limits_{i\in I}\mathcal{U}_i=\inf\limits\{\mathcal{U}_i:i\in I\}$.
\item $\bigvee\limits_{i\in I}\mathcal{U}_i=\sup\limits\{\mathcal{U}_i:i\in I\}$.
\item $\bigvee\limits_{i\in I}\mathcal{U}_i=\left\{\sum\limits_{i\in F}\mathbf{u}_i:F\subseteq I,\ \abs{F}<\infty,\ (\forall\,i\in F)\,\mathbf{u}_i\in\mathcal{U}_i\right\}$, where the empty sum is $\mathbf{0}_{\mathcal{V}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $I=\emptyset$, the empty meet is $\mathcal{V}$, the greatest lower bound of the empty family. Assume $I\neq\emptyset$ and set $M\defeq\bigcap\limits_{i\in I}\mathcal{U}_i$.\\
Then $M\leq\mathcal{V}$ by Proposition \ref{prop:intersectionsubspaces}, $(\forall\,i\in I)\,M\subseteq\mathcal{U}_i$, and $(\forall\,\mathcal{W}\leq\mathcal{V})[(\forall\,i\in I)\,\mathcal{W}\subseteq\mathcal{U}_i\Rightarrow\mathcal{W}\subseteq M]$. Hence $M=\inf\limits\{\mathcal{U}_i:i\in I\}$.
\item Set $J\defeq\operatorname{span}(\bigcup\limits_{i\in I}\mathcal{U}_i)$. Then $J\leq\mathcal{V}$ and $(\forall\,i\in I)\,\mathcal{U}_i\subseteq J$ by Theorem \ref{thm:span-smallest-containing-subspace}(i)--(ii).\\
If $\mathcal{W}\leq\mathcal{V}$ and $(\forall\,i\in I)\,\mathcal{U}_i\subseteq\mathcal{W}$, then $\bigcup\limits_{i\in I}\mathcal{U}_i\subseteq\mathcal{W}$, so $J\subseteq\mathcal{W}$ by Theorem \ref{thm:span-smallest-containing-subspace}(iii).\\
Thus $J$ is the least upper bound, including $J=\{\mathbf{0}_{\mathcal{V}}\}$ when $I=\emptyset$.
\item Every stated finite sum is a linear combination of vectors in $\bigcup\limits_{i\in I}\mathcal{U}_i$, so it belongs to $J$.\\
Conversely, each $\mathbf{x}\in J$ is a finite linear combination of vectors from that union.\\
Group terms assigned to the same index; linear closure of $\mathcal{U}_i$ replaces each nonempty group by one vector in $\mathcal{U}_i$.\\
The empty combination represents $\mathbf{0}_{\mathcal{V}}$, so the displayed representation set equals $J$.
\end{enumerate}
\hlt{Structural Commentary}: Order bounds identify meet and join; grouping span terms establishes finite representation.\\
\hlt{Key Insight}: Intersection preserves constraints; span combines directions.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:subspace-lattice-complete-modular} \hlt{Subspace Lattice}\\
The poset $\operatorname{Sub}(\mathcal{V})$ has the following properties.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item It is a complete lattice. For $\{\mathcal{U}_i\}_{i\in I}\subseteq\operatorname{Sub}(\mathcal{V})$, $\bot=\{\mathbf{0}_{\mathcal{V}}\}$, $\top=\mathcal{V}$, $\bigwedge\limits_{i\in I}\mathcal{U}_i=\bigcap\limits_{i\in I}\mathcal{U}_i$, and $\bigvee\limits_{i\in I}\mathcal{U}_i=\sum\limits_{i\in I}\mathcal{U}_i=\operatorname{span}(\bigcup\limits_{i\in I}\mathcal{U}_i)$.
\item It is modular: $(\forall\,\mathcal{U},\mathcal{S},\mathcal{T}\leq\mathcal{V})[\mathcal{U}\leq\mathcal{S}\Rightarrow\mathcal{S}\cap(\mathcal{T}+\mathcal{U})=(\mathcal{S}\cap\mathcal{T})+\mathcal{U}]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The arbitrary meet and join formulas hold by Proposition \ref{prop:subspaces-form-complete-lattice}(i)--(ii).\\
The empty formulas identify $\top=\mathcal{V}$ and $\bot=\{\mathbf{0}_{\mathcal{V}}\}$, so $\operatorname{Sub}(\mathcal{V})$ is complete.
\item Let $\mathcal{U}\leq\mathcal{S}$. If $\mathbf{x}\in(\mathcal{S}\cap\mathcal{T})+\mathcal{U}$, write $\mathbf{x}=\mathbf{y}+\mathbf{u}$ with $\mathbf{y}\in\mathcal{S}\cap\mathcal{T}$ and $\mathbf{u}\in\mathcal{U}$. Then $\mathbf{x}\in\mathcal{S}\cap(\mathcal{T}+\mathcal{U})$.\\
Conversely, if $\mathbf{x}\in\mathcal{S}\cap(\mathcal{T}+\mathcal{U})$, write $\mathbf{x}=\mathbf{t}+\mathbf{u}$ with $\mathbf{t}\in\mathcal{T}$ and $\mathbf{u}\in\mathcal{U}$. Then $\mathbf{t}=\mathbf{x}-\mathbf{u}\in\mathcal{S}\cap\mathcal{T}$ by $\mathbf{x},\mathbf{u}\in\mathcal{S}$, so $\mathbf{x}\in(\mathcal{S}\cap\mathcal{T})+\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: Completeness follows from the bound formulas; two inclusions establish modularity.\\
\hlt{Key Insight}: A contained summand survives intersection unchanged.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:uniqueness-direct-sum-two-subspaces} \hlt{Two-Subspace Directness Criterion}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$. Then $\mathcal{U}+\mathcal{W}$ is direct iff $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}_{\mathcal{V}}\}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $\mathcal{U}+\mathcal{W}$ is direct. For $\mathbf{x}\in\mathcal{U}\cap\mathcal{W}$, directness of $\mathbf{x}=\mathbf{x}+\mathbf{0}_{\mathcal{V}}=\mathbf{0}_{\mathcal{V}}+\mathbf{x}$ forces $\mathbf{x}=\mathbf{0}_{\mathcal{V}}$. Hence $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}_{\mathcal{V}}\}$.
\item[$(\Leftarrow)$] Assume $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}_{\mathcal{V}}\}$ and let $\mathbf{u}_1+\mathbf{w}_1=\mathbf{u}_2+\mathbf{w}_2$, where $\mathbf{u}_1,\mathbf{u}_2\in\mathcal{U}$ and $\mathbf{w}_1,\mathbf{w}_2\in\mathcal{W}$.\\
Then $\mathbf{u}_1-\mathbf{u}_2=\mathbf{w}_2-\mathbf{w}_1\in\mathcal{U}\cap\mathcal{W}$, so $\mathbf{u}_1=\mathbf{u}_2$ and $\mathbf{w}_1=\mathbf{w}_2$.\\
Every vector in the sum has a decomposition by its definition; therefore the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: The two directions translate decomposition equality into intersection and back.\\
\hlt{Key Insight}: Trivial intersection prevents cross-summand cancellation.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:sum-versus-direct-sum-existence} \hlt{Sums Always Exist; Directness Is Extra}\\
For $\mathcal{U},\mathcal{W}\leq\mathcal{V}$, the sum $\mathcal{U}+\mathcal{W}$ exists as the smallest subspace containing both by Proposition \ref{prop:sum-subspaces-smallest-containing-subspace}.\\
The notation $\mathcal{U}\oplus\mathcal{W}$ records the additional equivalent conditions in Proposition \ref{prop:uniqueness-direct-sum-two-subspaces}: $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}_{\mathcal{V}}\}$ and uniqueness of $\mathbf{u}+\mathbf{w}$.\\
Thus $+$ records generation, whereas $\oplus$ records generation without ambiguity; this comparison is not proof authority.
\end{perspective}

\begin{proposition} \label{prop:direct-sum-equivalent-conditions} \hlt{Equivalent Conditions for a Direct Sum}\\
For $m\in\N$ and subspaces $\mathcal{U}_1,\ldots,\mathcal{U}_m\leq\mathcal{V}$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_m$;
\item $(\forall\,\mathbf{v}\in\mathcal{V})(\exists!\,(\mathbf{u}_1,\ldots,\mathbf{u}_m)\in\mathcal{U}_1\times\cdots\times\mathcal{U}_m)\,\mathbf{v}=\sum\limits_{j=1}^{m}\mathbf{u}_j$;
\item $\mathcal{V}=\sum\limits_{j=1}^{m}\mathcal{U}_j$ and $(\forall\,(\mathbf{u}_1,\ldots,\mathbf{u}_m)\in\mathcal{U}_1\times\cdots\times\mathcal{U}_m)[\sum\limits_{j=1}^{m}\mathbf{u}_j=\mathbf{0}_{\mathcal{V}}\Rightarrow(\forall\,j\in\{1,\ldots,m\})\,\mathbf{u}_j=\mathbf{0}_{\mathcal{V}}]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Condition (i) asserts that the sum equals $\mathcal{V}$ and that its decompositions are unique, which is condition (ii).
\item[$(ii)$] $\Rightarrow (iii)$ Existence in (ii) implies $\mathcal{V}=\sum\limits_{j=1}^{m}\mathcal{U}_j$.\\
If $\sum\limits_{j=1}^{m}\mathbf{u}_j=\mathbf{0}_{\mathcal{V}}$, uniqueness identifies this representation with $\sum\limits_{j=1}^{m}\mathbf{0}_{\mathcal{V}}$, so $(\forall\,j\in\{1,\ldots,m\})\,\mathbf{u}_j=\mathbf{0}_{\mathcal{V}}$.
\item[$(iii)$] $\Rightarrow (i)$ The equality in (iii) supplies a decomposition of every $\mathbf{v}\in\mathcal{V}$. If $\sum\limits_{j=1}^{m}\mathbf{u}_j=\sum\limits_{j=1}^{m}\mathbf{w}_j$, then $\sum\limits_{j=1}^{m}(\mathbf{u}_j-\mathbf{w}_j)=\mathbf{0}_{\mathcal{V}}$.\\
Hence $(\forall\,j\in\{1,\ldots,m\})\,\mathbf{u}_j=\mathbf{w}_j$ by the zero-relation condition, so the sum is direct and equals $\mathcal{V}$.
\end{enumerate}
\hlt{Structural Commentary}: The implication cycle separates existence of decompositions from the zero-relation obstruction.\\
\hlt{Key Insight}: Uniqueness is equivalent to trivial zero relations.
\end{proof}
\end{proposition}

\begin{definition} \label{def:finite-product-vector-spaces} \hlt{Product of Vector Spaces}\\
Let $m\in\N$ and let $\mathcal{V}_1,\ldots,\mathcal{V}_m$ be $\mathbb{F}$-vector spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Their \hlt{product} is $\mathcal{V}_1\times\cdots\times\mathcal{V}_m\defeq\{(\mathbf{v}_1,\ldots,\mathbf{v}_m):(\forall\,j\in\{1,\ldots,m\})\,\mathbf{v}_j\in\mathcal{V}_j\}$.
\item Addition is $(\mathbf{u}_1,\ldots,\mathbf{u}_m)+(\mathbf{v}_1,\ldots,\mathbf{v}_m)\defeq(\mathbf{u}_1+\mathbf{v}_1,\ldots,\mathbf{u}_m+\mathbf{v}_m)$.
\item Scalar multiplication is $a(\mathbf{v}_1,\ldots,\mathbf{v}_m)\defeq(a\mathbf{v}_1,\ldots,a\mathbf{v}_m)$ for $a\in\mathbb{F}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:product-vector-space} \hlt{Products Are Vector Spaces}\\
If $m\in\N$ and $\mathcal{V}_1,\ldots,\mathcal{V}_m$ are vector spaces over $\mathbb{F}$, then $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ is a vector space under componentwise operations.
\begin{proof}
Set $P\defeq\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ and let $\mathbf{0}_j$ be the zero vector of $\mathcal{V}_j$. The operations are closed on $P$ by componentwise closure.\\
The tuple $(\mathbf{0}_1,\ldots,\mathbf{0}_m)$ is an additive identity, and $(-\mathbf{v}_1,\ldots,-\mathbf{v}_m)$ is the additive inverse of $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$.\\
For arbitrary tuples $\mathbf{u},\mathbf{v},\mathbf{w}\in P$, scalars $a,b\in\mathbb{F}$, and $j\in\{1,\ldots,m\}$, the $j$th coordinates on both sides of each commutativity, associativity, scalar-identity, scalar-associativity, and distributivity axiom are equal by the corresponding axiom in $\mathcal{V}_j$.\\
Coordinatewise equality proves each tuple identity, so $P$ is an $\mathbb{F}$-vector space.\\
\hlt{Structural Commentary}: Closure, identities, and all algebraic laws are verified coordinatewise.\\
\hlt{Key Insight}: Tuple equality reduces every axiom to its components.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-product} \hlt{Dimension of a Product}\\
Let $m\in\N$ and let $\mathcal{V}_1,\ldots,\mathcal{V}_m$ be finite-dimensional $\mathbb{F}$-vector spaces.\\
Then $P\defeq\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ is finite-dimensional and $\dim(P)=\sum\limits_{j=1}^{m}\dim(\mathcal{V}_j)$.
\begin{proof}
For $j\in\{1,\ldots,m\}$, let $(\mathbf{v}_{j,1},\ldots,\mathbf{v}_{j,n_j})$ be a basis of $\mathcal{V}_j$.\\
For $j\in\{1,\ldots,m\}$ and $k\in\{1,\ldots,n_j\}$, let $\mathbf{e}_{j,k}\in P$ have $j$th component $\mathbf{v}_{j,k}$ and zero in every other component.\\
The list of all $\mathbf{e}_{j,k}$ spans $P$ because each component has an expansion in its chosen basis.\\
If $\sum\limits_{j=1}^{m}\sum\limits_{k=1}^{n_j}a_{j,k}\mathbf{e}_{j,k}=\mathbf{0}$, then $(\forall\,j\in\{1,\ldots,m\})\,\sum\limits_{k=1}^{n_j}a_{j,k}\mathbf{v}_{j,k}=\mathbf{0}_{\mathcal{V}_j}$.\\
Hence $(\forall\,j\in\{1,\ldots,m\})(\forall\,k\in\{1,\ldots,n_j\})\,a_{j,k}=0$ by linear independence of each factor basis.\\
Thus the list is a basis, and its length $\sum\limits_{j=1}^{m}n_j=\sum\limits_{j=1}^{m}\dim(\mathcal{V}_j)$ proves the formula.\\
\hlt{Structural Commentary}: Embedded factor bases span componentwise and are independent componentwise.\\
\hlt{Key Insight}: Each factor contributes its own coordinate directions.
\end{proof}
\end{theorem}

\begin{definition} \label{def:graph-of-a-function} \hlt{Graph of a Function}\\
For $T:\mathcal{V}\to\mathcal{W}$, its \hlt{graph} is $\operatorname{graph}(T)\defeq\{(\mathbf{v},T\mathbf{v}):\mathbf{v}\in\mathcal{V}\}\subseteq\mathcal{V}\times\mathcal{W}$.
\end{definition}

\begin{proposition} \label{prop:graph-subspace-characterizes-linearity} \hlt{Linearity and Graphs}\\
A function $T:\mathcal{V}\to\mathcal{W}$ is linear iff $\operatorname{graph}(T)\leq\mathcal{V}\times\mathcal{W}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $T$ is linear. Then $(\mathbf{0}_{\mathcal{V}},\mathbf{0}_{\mathcal{W}})\in\operatorname{graph}(T)$ by Proposition \ref{prop:linear-maps-send-zero-to-zero}.\\
For $(\mathbf{u},T\mathbf{u}),(\mathbf{v},T\mathbf{v})\in\operatorname{graph}(T)$ and $a,b\in\mathbb{F}$, $a(\mathbf{u},T\mathbf{u})+b(\mathbf{v},T\mathbf{v})=(a\mathbf{u}+b\mathbf{v},T(a\mathbf{u}+b\mathbf{v}))\in\operatorname{graph}(T)$ by linearity.\\
Hence $\operatorname{graph}(T)\leq\mathcal{V}\times\mathcal{W}$ by Theorem \ref{thm:subspacetest}.
\item[$(\Leftarrow)$] Assume $\operatorname{graph}(T)\leq\mathcal{V}\times\mathcal{W}$. For $\mathbf{u},\mathbf{v}\in\mathcal{V}$ and $a,b\in\mathbb{F}$, $(a\mathbf{u}+b\mathbf{v},aT\mathbf{u}+bT\mathbf{v})=a(\mathbf{u},T\mathbf{u})+b(\mathbf{v},T\mathbf{v})\in\operatorname{graph}(T)$.\\
Membership in the graph forces $T(a\mathbf{u}+b\mathbf{v})=aT\mathbf{u}+bT\mathbf{v}$, so $T$ is linear.
\end{enumerate}
\hlt{Structural Commentary}: Each direction translates the linear-combination law into graph closure.\\
\hlt{Key Insight}: Graph closure is the linearity identity in product form.
\end{proof}
\end{proposition}

\begin{definition} \label{def:external-product-family} \hlt{External Product}\\
Let $\{\mathcal{V}_i:i\in I\}$ be $\mathbb{F}$-vector spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{external direct product} is $\prod\limits_{i\in I}\mathcal{V}_i\defeq\{(\mathbf{v}_i)_{i\in I}:(\forall i\in I)\,\mathbf{v}_i\in\mathcal{V}_i\}$.
\item Addition is defined by $(\forall\,i\in I)\,(\mathbf{u}+\mathbf{v})_i\defeq\mathbf{u}_i+\mathbf{v}_i$.
\item Scalar multiplication is defined by $(\forall\,i\in I)\,(a\mathbf{v})_i\defeq a\mathbf{v}_i$ for $a\in\mathbb{F}$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:support-and-finite-family-sums} \hlt{Support and Finite Sums of a Family}\\
Let $I$ be an index set, and let $\mathcal{X}_i$ be a vector space with zero vector $\mathbf{0}_i$ for $i\in I$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{support} of $\mathbf{x}=(\mathbf{x}_i)_{i\in I}\in\prod\limits_{i\in I}\mathcal{X}_i$ is $\operatorname{supp}(\mathbf{x})\defeq\{i\in I:\mathbf{x}_i\neq\mathbf{0}_i\}$.
\item The family $\mathbf{x}$ is \hlt{finitely supported} iff $\abs{\operatorname{supp}(\mathbf{x})}<\infty$.
\item If $(\forall\,i\in I)\,\mathcal{X}_i\leq\mathcal{V}$ and $\mathbf{x}$ is finitely supported, its \hlt{finite sum} is $\sum\limits_{i\in I}\mathbf{x}_i\defeq\sum\limits_{i\in\operatorname{supp}(\mathbf{x})}\mathbf{x}_i$; the empty sum is $\mathbf{0}_{\mathcal{V}}$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:external-products-and-direct-sums-family} \hlt{External Direct Sum}\\
Let $\{\mathcal{V}_i:i\in I\}$ be $\mathbb{F}$-vector spaces. The \hlt{external direct sum} is $\bigoplus\limits_{i\in I}^{\mathrm{ext}}\mathcal{V}_i\defeq\{\mathbf{v}\in\prod\limits_{i\in I}\mathcal{V}_i:\abs{\operatorname{supp}(\mathbf{v})}<\infty\}$.
\end{definition}

\begin{proposition} \label{prop:external-direct-sum-family-vector-space} \hlt{External Sums Are Finite-Support Products}\\
Let $\{\mathcal{V}_i\}_{i\in I}$ be $\mathbb{F}$-vector spaces and set $P\defeq\prod\limits_{i\in I}\mathcal{V}_i$ and $S\defeq\bigoplus\limits_{i\in I}^{\mathrm{ext}}\mathcal{V}_i$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $P$ is an $\mathbb{F}$-vector space under componentwise operations.
\item $S\leq P$.
\item If $\abs{I}<\infty$, then $S=P$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Componentwise closure keeps addition and scalar multiplication in $P$.\\
The family $(\mathbf{0}_i)_{i\in I}$ is its zero, and $(-\mathbf{v}_i)_{i\in I}$ is the inverse of $\mathbf{v}$. For arbitrary elements, scalars, and $i\in I$, the relevant identity at coordinate $i$ holds by the corresponding identity in $\mathcal{V}_i$; coordinatewise equality proves the identity in $P$.
\item The zero family has empty support. For $\mathbf{u},\mathbf{v}\in S$ and $a\in\mathbb{F}$, $\operatorname{supp}(a\mathbf{u}+\mathbf{v})\subseteq\operatorname{supp}(\mathbf{u})\cup\operatorname{supp}(\mathbf{v})$, a finite set.\\
Hence $a\mathbf{u}+\mathbf{v}\in S$, so $S\leq P$ by Theorem \ref{thm:subspacetest}.
\item If $\abs{I}<\infty$ and $\mathbf{v}\in P$, then $\operatorname{supp}(\mathbf{v})\subseteq I$ is finite. Thus $P\subseteq S$, while $S\subseteq P$ by construction, so $S=P$.
\end{enumerate}
\hlt{Structural Commentary}: Coordinatewise axioms build the product; support containment proves the remaining claims.\\
\hlt{Key Insight}: Finite linear combinations preserve finite support.
\end{proof}
\end{proposition}

\begin{definition} \label{def:finitely-nonzero-sequence-space} \hlt{Finitely Nonzero Sequence Space}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.FinitelyNonzeroSequenceSpace}\leanok The \hlt{finitely nonzero sequence space} is $\mathbb{F}^{(\mathbb{N})}\defeq\bigoplus\limits_{j\in\mathbb{N}}^{\mathrm{ext}}\mathbb{F}=\{(x_j)_{j\in\mathbb{N}}\in\mathbb{F}^{\mathbb{N}}:\abs{\{j\in\mathbb{N}:x_j\neq0\}}<\infty\}$, with componentwise operations.
\end{definition}

\begin{proposition} \label{prop:finitely-nonzero-sequences-vector-space} \hlt{Finitely Nonzero Sequences Form a Vector Space}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.finitely_nonzero_sequences_vector_space}\leanok The set $\mathbb{F}^{(\mathbb{N})}$ is a vector space over $\mathbb{F}$ under componentwise addition and scalar multiplication.
\begin{proof}
\leanok The space $\mathbb{F}^{(\mathbb{N})}$ is the external direct sum of the constant family $\mathcal{V}_j=\mathbb{F}$ by its definition, so it is a subspace of the product $\mathbb{F}^{\mathbb{N}}$ by Proposition \ref{prop:external-direct-sum-family-vector-space}(ii). Hence it is an $\mathbb{F}$-vector space under the inherited componentwise operations.\\
\hlt{Structural Commentary}: The result specializes the external direct-sum subspace construction to a constant countable family.\\
\hlt{Key Insight}: Finitely supported sequences are a canonical external sum.
\end{proof}
\end{proposition}

Set $\operatorname{Seq}(\mathbb{F})\defeq\mathbb{F}^{\mathbb{N}}$. Then $\mathbb{F}^{(\mathbb{N})}\leq\operatorname{Seq}(\mathbb{F})$ by Proposition \ref{prop:finitely-nonzero-sequences-vector-space}.

\begin{definition} \label{def:internal-direct-sum-family} \hlt{Internal Direct Sum of a Family}\\
For $\{\mathcal{U}_i\}_{i\in I}\subseteq\operatorname{Sub}(\mathcal{V})$, the internal sum $S\defeq\sum\limits_{i\in I}\mathcal{U}_i$ is \hlt{direct} iff $(\forall\,\mathbf{v}\in S)(\exists!\,\mathbf{u}\in\bigoplus\limits_{i\in I}^{\mathrm{ext}}\mathcal{U}_i)\,\mathbf{v}=\sum\limits_{i\in I}\mathbf{u}_i$.\\
A direct internal sum is denoted $\bigoplus\limits_{i\in I}\mathcal{U}_i$; the equality $\mathcal{V}=\bigoplus\limits_{i\in I}\mathcal{U}_i$ additionally asserts $\mathcal{V}=S$.
\end{definition}

\begin{proposition} \label{prop:internal-direct-sum-family-criterion} \hlt{Family Direct-Sum Criterion}\\
For $\{\mathcal{U}_i\}_{i\in I}\subseteq\operatorname{Sub}(\mathcal{V})$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\sum\limits_{i\in I}\mathcal{U}_i$ is direct;
\item $(\forall\,F\subseteq I)[\abs{F}<\infty\Rightarrow(\forall\,(\mathbf{u}_i)_{i\in F}\in\prod\limits_{i\in F}\mathcal{U}_i)(\sum\limits_{i\in F}\mathbf{u}_i=\mathbf{0}_{\mathcal{V}}\Rightarrow(\forall\,i\in F)\,\mathbf{u}_i=\mathbf{0}_{\mathcal{V}})]$;
\item $(\forall\,i\in I)\,\mathcal{U}_i\cap\sum\limits_{j\in I\setminus\{i\}}\mathcal{U}_j=\{\mathbf{0}_{\mathcal{V}}\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Let $F\subseteq I$ be finite and let $(\mathbf{u}_i)_{i\in F}\in\prod\limits_{i\in F}\mathcal{U}_i$ satisfy $\sum\limits_{i\in F}\mathbf{u}_i=\mathbf{0}_{\mathcal{V}}$. Extend this family by zero outside $F$.\\
Directness identifies the resulting decomposition with the all-zero decomposition, so $(\forall\,i\in F)\,\mathbf{u}_i=\mathbf{0}_{\mathcal{V}}$.
\item[$(ii)$] $\Rightarrow (iii)$ Fix $i\in I$ and let $\mathbf{x}\in\mathcal{U}_i\cap\sum\limits_{j\in I\setminus\{i\}}\mathcal{U}_j$.\\
Then $\mathbf{x}=\sum\limits_{j\in F}\mathbf{u}_j$ for a finite $F\subseteq I\setminus\{i\}$ with $(\forall\,j\in F)\,\mathbf{u}_j\in\mathcal{U}_j$ by Proposition \ref{prop:subspaces-form-complete-lattice}(iii).\\
The relation on $F\cup\{i\}$ with $i$th term $\mathbf{x}$ and $j$th term $-\mathbf{u}_j$ is zero. It is trivial by (ii), so $\mathbf{x}=\mathbf{0}_{\mathcal{V}}$.
\item[$(iii)$] $\Rightarrow (i)$ Let $F\subseteq I$ be finite and suppose $\sum\limits_{i\in F}\mathbf{u}_i=\mathbf{0}_{\mathcal{V}}$ with $(\forall\,i\in F)\,\mathbf{u}_i\in\mathcal{U}_i$.\\
For $k\in F$, $\mathbf{u}_k=-\sum\limits_{i\in F\setminus\{k\}}\mathbf{u}_i\in\mathcal{U}_k\cap\sum\limits_{j\in I\setminus\{k\}}\mathcal{U}_j=\{\mathbf{0}_{\mathcal{V}}\}$ by (iii), so every finite zero relation is trivial.\\
Each vector in the sum has a finite decomposition by Proposition \ref{prop:subspaces-form-complete-lattice}(iii).\\
The difference of two decompositions is a finite zero relation, so they coincide and the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: The cycle converts unique decompositions into zero relations and collective intersection tests.\\
\hlt{Key Insight}: Directness forbids cancellation across distinct summands.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:direct-sum-map-injective} \hlt{Direct Sum Map}\\
Let $m\in\N$ and let $\mathcal{U}_1,\ldots,\mathcal{U}_m\leq\mathcal{V}$. Define $\Gamma:\prod\limits_{j=1}^{m}\mathcal{U}_j\to\sum\limits_{j=1}^{m}\mathcal{U}_j$ by $\Gamma((\mathbf{u}_j)_{j=1}^{m})\defeq\sum\limits_{j=1}^{m}\mathbf{u}_j$. The following hold:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\Gamma\in\LL(\mathcal{U}_1\times\cdots\times\mathcal{U}_m,\mathcal{U}_1+\cdots+\mathcal{U}_m)$;
\item $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is direct iff $\Gamma$ is injective.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] For $\mathbf{u},\mathbf{v}\in\mathcal{U}_1\times\cdots\times\mathcal{U}_m$ and $a,b\in\mathbb{F}$, $\Gamma(a\mathbf{u}+b\mathbf{v})=\sum\limits_{j=1}^{m}(a\mathbf{u}_j+b\mathbf{v}_j)=a\Gamma(\mathbf{u})+b\Gamma(\mathbf{v})$. Hence $\Gamma$ is linear.
\item[$(ii)$] ($\Rightarrow$) Assume  sum is direct. If $\Gamma(\mathbf{u})=\Gamma(\mathbf{v})$, then $\mathbf{u}$ and $\mathbf{v}$ are decompositions of same vector, so $\mathbf{u}=\mathbf{v}$ by directness. QED.
\item[$(ii)$] ($\Leftarrow$) Assume $\Gamma$ is injective. If $\sum\limits_{j=1}^{m}\mathbf{u}_j=\sum\limits_{j=1}^{m}\mathbf{v}_j$, then $\Gamma(\mathbf{u})=\Gamma(\mathbf{v})$, so $\mathbf{u}=\mathbf{v}$ by injectivity.\\
Thus every vector in the sum has a unique decomposition, and the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: Linearity is coordinatewise; injectivity translates equality of sums into equality of decompositions.\\
\hlt{Key Insight}: Directness is injectivity of the addition map.
\end{proof}
\end{proposition}
