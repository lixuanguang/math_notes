\subsection{Polynomial Functional Calculus}

\begin{theorem} \label{thm:polynomial-functional-calculus-diagonalizable-matrix} \hlt{Polynomial Functional Calculus for Diagonalizable Matrices}\\
Let $n\geq1$, $\lambda_1,\ldots,\lambda_n\in\mathbb{F}$, and set $\Lambda\defeq\operatorname{diag}(\lambda_1,\ldots,\lambda_n)$. Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ and $S\in\operatorname{Mat}(n,n,\mathbb{F})^{\times}$.\\
If $A=S\Lambda S^{-1}$, then $(\forall\,p\in\mathbb{F}[z])\,p(A)=S\operatorname{diag}(p(\lambda_1),\ldots,p(\lambda_n))S^{-1}$.
\begin{proof}
Set $P_j\defeq(A^j=S\Lambda^jS^{-1})$, $j\in\mathbb{Z}_{\geq0}$. Base case $P_0$ holds by Definition \ref{def:polynomial-evaluation-on-square-matrix} and inverse relations: $A^0=I_n=SI_nS^{-1}=S\Lambda^0S^{-1}$.\\
Fix $j\in\mathbb{Z}_{\geq0}$ and suppose $P_j$. The identity $A^{j+1}=A^jA=S\Lambda^jS^{-1}S\Lambda S^{-1}$ holds by Definition \ref{def:polynomial-evaluation-on-square-matrix} and $P_j$.\\
The identity $S\Lambda^jS^{-1}S\Lambda S^{-1}=S\Lambda^{j+1}S^{-1}$ holds by Proposition \ref{prop:matrix-product-associativity-and-identities}(i),(ii) and $S^{-1}S=I_n$.\\
Thus $P_{j+1}$ holds, so $(\forall\,j\in\mathbb{Z}_{\geq0})\,P_j$ by Theorem \ref{thm:mathematical-induction}. Fix $p\in\mathbb{F}[z]$ and write $p(z)=\sum\limits_{j=0}^{d}c_jz^j$ with $d\in\mathbb{Z}_{\geq0}$ and $c_0,\ldots,c_d\in\mathbb{F}$.\\
The identity $p(A)=\sum\limits_{j=0}^{d}c_jS\Lambda^jS^{-1}$ holds by Definition \ref{def:polynomial-evaluation-on-square-matrix} and the power identity.\\
The identity $\sum\limits_{j=0}^{d}c_jS\Lambda^jS^{-1}=S\left(\sum\limits_{j=0}^{d}c_j\Lambda^j\right)S^{-1}$ holds by Proposition \ref{prop:matrix-product-associativity-and-identities}(iii),(iv). Hence $p(A)=Sp(\Lambda)S^{-1}$ by Definition \ref{def:polynomial-evaluation-on-square-matrix}.\\
The identity $(\forall\,j\in\mathbb{Z}_{\geq0})(\forall\,i,k\in J_n)\,(\Lambda^j)_{ik}=\delta_{ik}\lambda_i^j$ holds by induction on $j$ and Definition \ref{def:matrices-and-matrix-operations}(iv),(ix),(xii).\\
Hence $(\forall\,i,k\in J_n)\,p(\Lambda)_{ik}=\delta_{ik}p(\lambda_i)$ by Definition \ref{def:polynomial-evaluation-on-square-matrix}. Therefore $p(\Lambda)=\operatorname{diag}(p(\lambda_1),\ldots,p(\lambda_n))$ by Definition \ref{def:matrices-and-matrix-operations}(i),(iv).\\
The claimed identity holds by substituting this formula into $p(A)=Sp(\Lambda)S^{-1}$.\\
\hlt{Structural Commentary}: Power induction transfers conjugation to polynomial sums; diagonal powers reduce entrywise.\\
\hlt{Key Insight}: Conjugation cancels between powers, while diagonal entries evolve independently.
\end{proof}
\end{theorem}
