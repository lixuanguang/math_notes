\subsection{Diagonalizability and Minimal Polynomial Tests}
\begin{theorem} \label{thm:diagonalizable-iff-minimal-polynomial-squarefree} \hlt{Diagonalizability and the Minimal Polynomial}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have distinct eigenvalues $\lambda_1,\ldots,\lambda_d$.\\
Then $A$ is diagonalizable if and only if $\mu_A(z)=(z-\lambda_1)\cdots(z-\lambda_d)$.\\
Equivalently, $A$ is diagonalizable if and only if $\mu_A$ is square-free.
\begin{proof}
Set $p(z)\defeq\prod_{i=1}^{d}(z-\lambda_i)$.
By Theorem \ref{thm:minimal-polynomial-basic-properties}, $\mu_A(z)=\prod_{i=1}^{d}(z-\lambda_i)^{q_i}$ with $q_i\geq1$, so $\mu_A$ is square-free if and only if $\mu_A=p$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $A=S\Lambda S^{-1}$ is diagonalizable, then $p(\Lambda)=0$, so $p(A)=0$.\\
Theorem \ref{thm:minimal-polynomial-basic-properties} gives $\mu_A\mid p$; since every factor $z-\lambda_i$ occurs in $\mu_A$, we have $\mu_A=p$.
\item[$(\Leftarrow)$] Suppose $\mu_A=p$. For $i\in\{1,\ldots,d\}$, set $\ell_i(z)\defeq\prod\limits_{j\neq i}\frac{z-\lambda_j}{\lambda_i-\lambda_j}$ and $\rho_i\defeq\ell_i(A)$.\\
The polynomials $\ell_i\ell_j$ for $i\neq j$, $\sum_i\ell_i-1$, and $z-\sum_i\lambda_i\ell_i$ vanish at $\lambda_1,\ldots,\lambda_d$, so each is divisible by $p$. Since $p(A)=0$, $\rho_i\rho_j=0$ for $i\neq j$, $\sum_i\rho_i=I_n$, and $A=\sum_i\lambda_i\rho_i$. Thus $\rho_i^2=\rho_i$.\\
If $\rho_i=0$, then $\prod_{j\neq i}(A-\lambda_jI_n)=0$, contradicting the minimality of the degree-$d$ polynomial $\mu_A=p$. Hence every $\rho_i\neq0$.\\
Theorem \ref{thm:algebraic-spectral-resolution-diagonalizable}, applied to the operator induced by $A$, gives that $A$ is diagonalizable.
\end{enumerate}
\hlt{Structural Commentary}: Lagrange polynomials turn square-free annihilation into a spectral resolution.\\
\hlt{Key Insight}: Distinct roots produce complementary eigenspace projections.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:squarefree-annihilator-implies-diagonalizable} \hlt{Square-Free Annihilator Criterion}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. If a polynomial $f$ with no repeated roots satisfies $f(A)=0$, then $A$ is diagonalizable.
\begin{proof}
Theorem \ref{thm:operator-annihilator-iff-multiple-minimal-polynomial} gives $\mu_A\mid f$. Thus $\mu_A$ is square-free, so Theorem \ref{thm:diagonalizable-iff-minimal-polynomial-squarefree} applies.\\
\hlt{Structural Commentary}: A square-free annihilator forces square-free $\mu_A$.\\
\hlt{Key Insight}: Square-free $\mu_A$ means diagonalizable.
\end{proof}
\end{corollary}

\begin{method} \label{meth:operator-minimal-polynomial-from-matrix} \hlt{Computing $\mu_T$ from a Matrix}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ be the matrix of $T$ in some basis.
Propositions \ref{prop:matrix-representation-respects-linear-combinations} and \ref{prop:composition-and-matrix-multiplication} give $[p(T)]=p(A)$ for every $p\in\mathbb{F}[z]$; Theorem \ref{thm:linear-maps-and-matrices-same-data} then gives $\mu_T=\mu_A$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $m=1,\ldots,n^2$, test the monic candidate $p_m(z)=a_0+a_1z+\cdots+a_{m-1}z^{m-1}+z^m$.
\item Solve linear system by comparing matrix entries in $a_0I+a_1A+\cdots+a_{m-1}A^{m-1}=-A^m$ for $a_0,\ldots,a_{m-1}$.
\item If no solution exists, increase $m$. The first solvable candidate is $\mu_T$.
\end{enumerate}
Proposition \ref{prop:finite-dimensional-operators-have-annihilating-polynomials} guarantees termination by $m=n^2$.
\end{method}

\begin{method} \label{meth:minimal-polynomial-testing-from-characteristic-polynomial} \hlt{Testing a Minimal Polynomial from $p_A$}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$, and suppose $p_A(z)=\prod_i(z-\lambda_i)^{a_i}$ is known.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Write each candidate as $m_{\mathbf{q}}(z)=\prod_i(z-\lambda_i)^{q_i}$, where $1\leq q_i\leq a_i$.
\item Test exponent vectors $\mathbf{q}=(q_i)$ in increasing total degree $\sum_iq_i$.
\item For each $\mathbf{q}$, compute $m_{\mathbf{q}}(A)=\prod_i(A-\lambda_iI)^{q_i}$.
\item The first candidate with $m_{\mathbf{q}}(A)=0$ is $\mu_A$, by Theorem \ref{thm:minimal-polynomial-basic-properties}.
\end{enumerate}
\end{method}
