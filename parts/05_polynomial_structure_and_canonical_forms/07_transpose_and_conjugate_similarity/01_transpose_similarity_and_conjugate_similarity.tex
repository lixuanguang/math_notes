\subsection{Transpose Similarity and Conjugate Similarity}

\begin{theorem} \label{thm:conjugate-similarity-jordan-block-criterion} \hlt{Conjugate Similarity Criterion}\\
Let $n\in\mathbb{Z}_{>0}$ and let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Write $\sim$ for similarity on $\operatorname{Mat}(n,n,\mathbb{C})$.\\
For $(\lambda,k)\in\mathbb{C}\times\mathbb{Z}_{>0}$, let $m_A(\lambda,k)$ count the $J_k(\lambda)$ blocks in any Jordan form of $A$.\\
Then $A\sim\overline{A}\iff(\forall\,(\lambda,k)\in\mathbb{C}\times\mathbb{Z}_{>0})\,m_A(\lambda,k)=m_A(\overline{\lambda},k)$.
\begin{proof}
The function $m_A$ is well-defined by Corollary \ref{cor:jordan-form-unique-up-to-block-permutation}(i), with $m_A(\lambda,k)=0$ for $\lambda\notin\operatorname{spec}(A)$.\\
By Theorem \ref{thm:complex-jordan-form}, choose $S\in\operatorname{Mat}(n,n,\mathbb{C})^\times$ and a Jordan matrix $J$ satisfying $A=SJS^{-1}$.\\
Then $\overline{A}=\overline{S}\,\overline{J}\,\overline{S}^{-1}$, and the $J_k(\lambda)$-multiplicity in $\overline{J}$ is $m_A(\overline{\lambda},k)$ by blockwise conjugation.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $A\sim\overline{A}$. The chain $J\sim A\sim\overline{A}\sim\overline{J}$ holds by Proposition \ref{prop:similarity-equivalence-relation}.\\
By Corollary \ref{cor:jordan-form-unique-up-to-block-permutation}(ii), choose a block-permutation matrix $P$ satisfying $\overline{J}=P^{-1}JP$.\\
The equality $(\forall\,(\lambda,k)\in\mathbb{C}\times\mathbb{Z}_{>0})\,m_A(\lambda,k)=m_A(\overline{\lambda},k)$ holds by $\overline{J}=P^{-1}JP$ and the conjugate-block identity.
\item[$(\Leftarrow)$] Assume $(\forall\,(\lambda,k)\in\mathbb{C}\times\mathbb{Z}_{>0})\,m_A(\lambda,k)=m_A(\overline{\lambda},k)$.\\
By the hypothesis and conjugate-block identity, choose a block-permutation matrix $P$ satisfying $\overline{J}=P^{-1}JP$.\\
Set $Q\defeq\overline{S}P^{-1}S^{-1}\in\operatorname{Mat}(n,n,\mathbb{C})^\times$. Then $\overline{A}=\overline{S}P^{-1}JP\overline{S}^{-1}=QAQ^{-1}$, so $A\sim\overline{A}$.
\end{enumerate}
\hlt{Structural Commentary}: Conjugation relabels Jordan blocks; block-permutation uniqueness converts their counts into similarity.\\
\hlt{Key Insight}: $J_k(\lambda)$ becomes $J_k(\overline{\lambda})$ without changing $k$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:reversal-matrix} \hlt{Reversal Matrix}\\
For $k\in\mathbb{Z}_{>0}$, the \hlt{reversal matrix} $R_k\in\operatorname{Mat}(k,k,\mathbb{C})$ is defined by $(\forall\,i,j\in\{1,\ldots,k\})\,(R_k)_{ij}\defeq\delta_{i+j,k+1}$.\\
The identities $R_k^t=R_k$ and $(R_k^2)_{ij}=\sum_{r=1}^k\delta_{i+r,k+1}\delta_{r+j,k+1}=\delta_{ij}$ hold by the entry rule. Hence $R_k^2=I_k$.
\end{definition}

\begin{lemma} \label{lem:jordan-block-transpose-reversal-similarity} \hlt{Reversal Conjugates a Jordan Block to Its Transpose}\\
Let $k\in\mathbb{Z}_{>0}$ and $\lambda\in\mathbb{C}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $J_k(\lambda)^t=R_kJ_k(\lambda)R_k$;
\item $R_kJ_k(\lambda)$ and $J_k(\lambda)R_k$ are symmetric.
\end{enumerate}
\begin{proof}
Set $N\defeq J_k(0)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $i,j\in\{1,\ldots,k\}$, $(R_kNR_k)_{ij}=N_{k+1-i,k+1-j}=\delta_{i,j+1}=(N^t)_{ij}$ by the entry rules for $R_k$ and $N$.\\
Thus $R_kJ_k(\lambda)R_k=\lambda R_k^2+R_kNR_k=\lambda I_k+N^t=J_k(\lambda)^t$.
\item The identities $\bigl(R_kJ_k(\lambda)\bigr)^t=J_k(\lambda)^tR_k=R_kJ_k(\lambda)$ and $\bigl(J_k(\lambda)R_k\bigr)^t=R_kJ_k(\lambda)^t=J_k(\lambda)R_k$ hold by part (i) and $R_k^2=I_k$.
\end{enumerate}
\hlt{Structural Commentary}: The reversal entry rule turns the superdiagonal shift into its transposed subdiagonal shift.\\
\hlt{Key Insight}: Reversal reads each Jordan chain in the opposite order.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:matrix-similar-to-transpose-and-symmetric-factorizations} \hlt{Transpose Similarity and Symmetric Factorisations}\\
Let $n\in\mathbb{Z}_{>0}$ and let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\exists P\in\operatorname{Mat}(n,n,\mathbb{C})^\times)\ P^t=P\ \land\ PAP^{-1}=A^t$.
\item $(\exists B,C,D,E\in\operatorname{Mat}(n,n,\mathbb{C}))\ A=BC=DE$.\\
The witnesses satisfy $B,E\in\operatorname{Mat}(n,n,\mathbb{C})^\times$ and $B^t=B$, $C^t=C$, $D^t=D$, and $E^t=E$.
\end{enumerate}
\begin{proof}
By Theorem \ref{thm:complex-jordan-form}, choose $S\in\operatorname{Mat}(n,n,\mathbb{C})^\times$ and a Jordan matrix $J$ satisfying $A=SJS^{-1}$.\\
Let $R$ be the direct sum of the reversal matrices matching the blocks of $J$.\\
The identities $J^t=RJR$, $R^t=R$, and $R^2=I_n$ hold blockwise by Lemma \ref{lem:jordan-block-transpose-reversal-similarity}(i), while $RJ$ and $JR$ are symmetric by Lemma \ref{lem:jordan-block-transpose-reversal-similarity}(ii).
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $P\defeq(S^{-1})^tRS^{-1}$. Then $P^t=P$ and $P^{-1}=SRS^t$ by $R^t=R=R^{-1}$.\\
The identity $PAP^{-1}=(S^{-1})^tRJRS^t=(S^{-1})^tJ^tS^t=A^t$ holds by $A=SJS^{-1}$ and $J^t=RJR$.
\item Set $B\defeq SRS^t$ and $C\defeq(S^{-1})^tRJS^{-1}$. Set $D\defeq SJRS^t$ and $E\defeq(S^{-1})^tRS^{-1}$.\\
The matrices $B,C,D,E$ are symmetric by congruence of $R,RJ,JR,R$, respectively, and $B,E$ are invertible.\\
The identities $BC=SJS^{-1}=A$ and $DE=SJS^{-1}=A$ hold by $R^2=I_n$.
\end{enumerate}
\hlt{Structural Commentary}: Jordan reversal transfers through congruence to a symmetric similarity and two symmetric factor pairs.\\
\hlt{Key Insight}: The reversal involution both transposes blocks and cancels between factors.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:hermitian-products-and-conjugate-products} \hlt{Hermitian Products and Conjugate Products}\\
Let $n\in\mathbb{Z}_{>0}$ and let $A,H,K\in\operatorname{Mat}(n,n,\mathbb{C})$, where $H$ and $K$ are Hermitian. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $HK$ is similar to $KH$.
\item $A\overline{A}$ is similar to $\overline{A}A$.
\end{enumerate}
\begin{proof}
For $M\in\operatorname{Mat}(n,n,\mathbb{C})$, $\ker(\overline{M})=\{\overline{\mathbf{v}}:\mathbf{v}\in\ker(M)\}$ by $\overline{M}\,\overline{\mathbf{v}}=\overline{M\mathbf{v}}$ and involutivity of conjugation.\\
The identity $\operatorname{rank}(M^*)=\operatorname{rank}(\overline{M}^{\,t})=\operatorname{rank}(\overline{M})=\operatorname{rank}(M)$ holds by Theorem \ref{thm:row-rank-equals-column-rank}, Corollary \ref{cor:matrix-rank-nullity}(i), and the kernel identity.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The equality $(\forall\,p\in\{1,\ldots,n\})\,\bigl((HK)^p\bigr)^*=\bigl((HK)^*\bigr)^p=(KH)^p$ holds by $H^*=H$ and $K^*=K$.\\
Thus $(\forall\,p\in\{1,\ldots,n\})\,\operatorname{rank}((HK)^p)=\operatorname{rank}((KH)^p)$, so $HK$ is similar to $KH$ by Corollary \ref{cor:ab-ba-similar-rank-power-criterion}.
\item The equality $(\forall\,p\in\{1,\ldots,n\})\,\overline{(A\overline{A})^p}=(\overline{A}A)^p$ holds by entrywise conjugation of products.\\
Thus $(\forall\,p\in\{1,\ldots,n\})\,\operatorname{rank}((A\overline{A})^p)=\operatorname{rank}((\overline{A}A)^p)$, so $A\overline{A}$ is similar to $\overline{A}A$ by Corollary \ref{cor:ab-ba-similar-rank-power-criterion}.
\end{enumerate}
\hlt{Structural Commentary}: Adjoint and conjugation preserve every required power rank before the $AB$--$BA$ criterion.\\
\hlt{Key Insight}: Equal power ranks convert reordered products into similar matrices.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:conjugate-similarity-hermitian-product-equivalences} \hlt{Conjugate Similarity and Hermitian Products}\\
Let $n\in\mathbb{Z}_{>0}$ and let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ is similar to $\overline{A}$.
\item $A$ is similar to $A^*$.
\item $(\exists R\in\operatorname{Mat}(n,n,\mathbb{C})^\times)\ R^*=R\ \land\ A=RA^*R^{-1}$.
\item $(\exists H,K,L,M\in\operatorname{Mat}(n,n,\mathbb{C}))\ A=HK=LM$.\\
The witnesses satisfy $H,M\in\operatorname{Mat}(n,n,\mathbb{C})^\times$ and $H^*=H$, $K^*=K$, $L^*=L$, and $M^*=M$.
\item $(\exists H,K\in\operatorname{Mat}(n,n,\mathbb{C}))\ H^*=H\ \land\ K^*=K\ \land\ A=HK$.
\end{enumerate}
\begin{proof}
Write $\sim$ for similarity on $\operatorname{Mat}(n,n,\mathbb{C})$, which is transitive by Proposition \ref{prop:similarity-equivalence-relation}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ $A\sim\overline{A}\sim\overline{A}^{\,t}=A^*$ by part (i) and Theorem \ref{thm:matrix-similar-to-transpose-and-symmetric-factorizations}(i).
\item[$(ii)$]$\Rightarrow (iii)$ Choose $S\in\operatorname{Mat}(n,n,\mathbb{C})^\times$ satisfying $A=SA^*S^{-1}$, equivalently $AS=SA^*$, and set $Q\defeq S^{-1}S^*$.\\
For $\theta\in\mathbb{R}$, set $S_{\theta}\defeq e^{i\theta}S$ and $H_{\theta}\defeq S_{\theta}+S_{\theta}^*$.\\
The identities $AH_{\theta}=H_{\theta}A^*$, $H_{\theta}^*=H_{\theta}$, and $S_{\theta}^{-1}H_{\theta}=I_n+e^{-2i\theta}Q$ hold by $AS=SA^*$, its adjoint, and the definitions.\\
For $\theta\in\mathbb{R}$, $H_{\theta}\notin\operatorname{Mat}(n,n,\mathbb{C})^\times\iff-e^{2i\theta}\in\operatorname{spec}(Q)$ by the last identity and invertibility of $S_{\theta}$.\\
The existence $(\exists\theta\in\mathbb{R})\,-e^{2i\theta}\notin\operatorname{spec}(Q)$ holds by Corollary \ref{cor:number-of-distinct-eigenvalues} and infinitude of $\{-e^{2i\phi}:\phi\in\mathbb{R}\}$.\\
For such $\theta$, $H_{\theta}\in\operatorname{Mat}(n,n,\mathbb{C})^\times$, $H_{\theta}^*=H_{\theta}$, and $A=H_{\theta}A^*H_{\theta}^{-1}$ by the preceding identities.
\item[$(iii)$]$\Rightarrow (iv)$ Choose $R\in\operatorname{Mat}(n,n,\mathbb{C})^\times$ satisfying $R^*=R$ and $A=RA^*R^{-1}$, and set $(H,K,L,M)\defeq(R,A^*R^{-1},RA^*,R^{-1})$.\\
$A=HK=LM$ and $H,M\in\operatorname{Mat}(n,n,\mathbb{C})^\times$ by the definitions and part (iii).\\
$(H^*,K^*,L^*,M^*)=(H,K,L,M)$ by $R^*=R$, $AR=RA^*$, and $R^{-1}A=A^*R^{-1}$.
\item[$(iv)$]$\Rightarrow (v)$ The witnesses $(H,K)$ from part (iv) satisfy $H^*=H\ \land\ K^*=K\ \land\ A=HK$.
\item[$(v)$]$\Rightarrow (i)$ Choose $H,K\in\operatorname{Mat}(n,n,\mathbb{C})$ satisfying $H^*=H$, $K^*=K$, and $A=HK$ by part (v).\\
$A=HK\sim KH\sim(KH)^t=H^tK^t=\overline{H}\,\overline{K}=\overline{A}$ by Corollary \ref{cor:hermitian-products-and-conjugate-products}(i), Theorem \ref{thm:matrix-similar-to-transpose-and-symmetric-factorizations}(i), and Hermitianity of $H,K$.
\end{enumerate}
\hlt{Structural Commentary}: Transpose similarity starts the cycle; phase averaging Hermitianizes the adjoint-similarity witness.\\
\hlt{Key Insight}: A phase avoids the finitely many singular Hermitian averages.
\end{proof}
\end{theorem}
