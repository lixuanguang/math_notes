\subsection{Normal Primary Structure}

\begin{corollary} \label{cor:normal-kernels-of-powers-and-squarefree-minimal-polynomial} \hlt{Normal Operators Have No Nilpotent Primary Part}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be a finite-dimensional complex inner-product space and let $T\in\LL(\mathcal{V})$ be normal.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $S=p(T)$ for some $p\in\mathbb{C}[z]$, then $\ker(S^k)=\ker(S)$ for every $k\geq1$.
\item The minimal polynomial of $T$ is square-free, where $\mu_T(z)=\prod\limits_{\lambda\in\operatorname{spec}(T)}(z-\lambda)$.
\end{enumerate}
\begin{proof}
By Theorem \ref{thm:complex-spectral-theorem}, choose an orthonormal eigenbasis $\mathbf{e}_1,\ldots,\mathbf{e}_n$ with $T\mathbf{e}_j=\lambda_j\mathbf{e}_j$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $S=p(T)$, $k\geq1$, $S^k\mathbf{e}_j=p(\lambda_j)^k\mathbf{e}_j$; $\ker(S^k)=\operatorname{span}\{\mathbf{e}_j:p(\lambda_j)^k=0\} =\operatorname{span}\{\mathbf{e}_j:p(\lambda_j)=0\} =\ker(S)$.
\item Let $m(z)=\prod_{\lambda\in\operatorname{spec}(T)}(z-\lambda)$. Since $m(T)\mathbf{e}_j=0$ for all $j$, $m(T)=0$, so $\mu_T\mid m$. Conversely, every $\lambda\in\operatorname{spec}(T)$ is a root of $\mu_T$ by Theorem \ref{thm:operator-minimal-polynomial-roots-eigenvalues}, hence $m\mid\mu_T$. Since both polynomials are monic, $\mu_T=m$.
\end{enumerate}
\hlt{Structural Commentary}: Normality makes every primary component an eigenspace.\\
\hlt{Key Insight}: A diagonal operator has no hidden nilpotent part.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:normal-coprime-invariant-subspaces-orthogonal} \hlt{Coprime Normal Primary Pieces Are Orthogonal}\\
Let $\mathcal{V}$ be a finite-dimensional complex inner-product space, let $T\in\LL(\mathcal{V})$ be normal, and let $\mathcal{S},\mathcal{R}\leq\mathcal{V}$ be $T$-invariant.\\
If $p,q\in\mathbb{C}[z]$, $(p,q)=1$, $p(T)|_{\mathcal{S}}=0$, and $q(T)|_{\mathcal{R}}=0$, then $\mathcal{S}\perp\mathcal{R}$.
\begin{proof}
Choose $a,b\in\mathbb{C}[z]$ with $ap+bq=1$. For $\mathbf{s}\in\mathcal{S}$, $\mathbf{s}=a(T)p(T)\mathbf{s}+b(T)q(T)\mathbf{s}=b(T)q(T)\mathbf{s}$.\\
By Corollary \ref{cor:normal-operator-invariant-subspaces-reduce}, $\mathcal{R}$ reduces $T$, so it reduces $b(T)$ and $q(T)$. Hence $b(T)^*\mathcal{R}\subseteq\mathcal{R}$ and $q(T)^*|_{\mathcal{R}}=(q(T)|_{\mathcal{R}})^*=0$.\\
Thus, for $\mathbf{r}\in\mathcal{R}$, $\innerprod{\mathbf{s},\mathbf{r}} =\innerprod{b(T)q(T)\mathbf{s},\mathbf{r}} =\innerprod{\mathbf{s},q(T)^*b(T)^*\mathbf{r}} =0$.\\
\hlt{Structural Commentary}: Coprime annihilators orthogonally separate normal invariant pieces.\\
\hlt{Key Insight}: Normality makes invariant subspaces reduce, so annihilation also holds after taking adjoints.
\end{proof}
\end{proposition}
