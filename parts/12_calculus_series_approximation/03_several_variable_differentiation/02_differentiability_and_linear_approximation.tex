\subsection{Differentiability and Linear Approximation}

\begin{theorem} \label{thm:bounded-derivative-convex-mean-value} \hlt{Mean-Value Inequality on Convex Sets}\\
Let $E\subseteq\R^n$ be open and convex, and let $\f:E\to\R^m$ be differentiable.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $M\geq0$ and $\normop{D\f(\x)}\leq M$ for all $\x\in E$, then $\norm{\f(\b)-\f(\a)}_2\leq M\norm{\b-\a}_2$ for all $\a,\b\in E$.
\item If $D\f(\x)=0$ for all $\x\in E$, then $\f$ is constant on $E$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $\a,\b\in E$, set $\gamma(t)\defeq(1-t)\a+t\b$, and put $\mathbf{g}\defeq\f\circ\gamma$. Convexity gives $\gamma([0,1])\subseteq E$, and the chain rule gives $\mathbf{g}'(t)=D\f(\gamma(t))(\b-\a)$, so $\norm{\mathbf{g}'(t)}_2\leq M\norm{\b-\a}_2$.\\
Theorem \ref{thm:vector-valued-mean-value-inequality} gives $\norm{\f(\b)-\f(\a)}_2=\norm{\mathbf{g}(1)-\mathbf{g}(0)}_2\leq M\norm{\b-\a}_2$.
\item If $D\f(\x)=0$ on $E$, apply (i) with $M=0$: for all $\a,\b\in E$, $\norm{\f(\b)-\f(\a)}_2=0$, hence $\f(\a)=\f(\b)$.
\end{enumerate}
\hlt{Structural Commentary}: Line restriction gives a one-variable estimate.\\
\hlt{Key Insight}: Derivative bounds control secants.
\end{proof}
\end{theorem}

\begin{definition} \label{def:c1-map} \hlt{$C^1$ Mapping}\\
Let $E\subseteq\R^n$ be open. A differentiable map $\f:E\to\R^m$ is \hlt{continuously differentiable}, or $C^1$, if $\x\mapsto D\f(\x)$ is continuous from $E$ into $\LL(\R^n,\R^m)$ in operator norm.
\end{definition}

\begin{perspective} \label{per:c1-operator-norm-continuity} \hlt{$C^1$ Uses the Operator-Norm Metric}\\
$C^1$ continuity measures $\x\mapsto D\f(\x)$ in operator norm. In coordinates this is spectral-norm continuity, equivalent here to continuity of the derivative matrix entries.
\end{perspective}

\begin{theorem} \label{thm:c1-iff-continuous-partials} \hlt{$C^1$ Means Continuous Partials}\\
Let $E\subseteq\R^n$ be open and let $\f=(f_1,\ldots,f_m):E\to\R^m$.\\
Then $\f$ is $C^1$ on $E$ iff each $D_jf_i$ exists and is continuous on $E$, for $1\leq i\leq m$ and $1\leq j\leq n$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\f$ is $C^1$, then $D\f:E\to\LL(\R^n,\R^m)$ is continuous. By Theorem \ref{thm:total-derivative-matrix-partials}, $(D_jf_i)(\x)$ is the $(i,j)$ matrix entry of $D\f(\x)$; matrix-entry projections are continuous, so each $D_jf_i$ is continuous on $E$.
\item[$(\Leftarrow)$] By Theorem \ref{thm:several-variable-componentwise-differentiability}, it suffices to treat $m=1$. Fix $\x\in E$ and small $\h$ with coordinate polygon in $E$.\\
Set $\x^{(0)}\defeq\x$ and $\x^{(j)}\defeq\x+\sum\limits_{k=1}^{j}h_k\mathbf{e}_k$. Telescoping and the one-variable MVT give\\
$f(\x+\h)-f(\x)=\sum\limits_{j=1}^{n}h_j(D_jf)(\xi_j)$ for some $\xi_j\in[\x^{(j-1)},\x^{(j)}]$. Let $A_\x\h\defeq\sum\limits_{j=1}^{n}h_j(D_jf)(\x)$.\\
Since $\xi_j\to\x$ as $\h\to\ze$, $\abs{f(\x+\h)-f(\x)-A_\x\h}\leq\sum\limits_{j=1}^{n}\abs{h_j}\abs{(D_jf)(\xi_j)-(D_jf)(\x)}=o(\norm{\h}_2)$.\\
Thus $Df(\x)=A_\x$, i.e. $Df(\x)=[(D_1f)(\x)\ \cdots\ (D_nf)(\x)]$, and continuity of the entries gives $\f\in C^1$.
\end{enumerate}
\hlt{Structural Commentary}: Continuous partials form the derivative matrix.\\
\hlt{Key Insight}: Coordinate telescoping gives the linear approximation.
\end{proof}
\end{theorem}

Higher derivatives repeat the same idea: once $D\f$ is itself a map into the finite-dimensional vector space $\LL(\R^n,\R^m)$, we differentiate it again. The linear-algebra viewpoint matters here because the second derivative is naturally bilinear, not merely an array of second partial derivatives.

\begin{definition} \label{def:higher-order-partials-and-c2} \hlt{Higher Derivatives and $C^r$}\\
Let $E\subseteq\R^n$ be open, $\f:E\to\R^m$, $r\geq1$, and let $\LL^r(\R^n,\R^m)$ denote the space of $r$-linear maps $(\R^n)^r\to\R^m$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{$r$th derivative} is recursive: $D^1\f\defeq D\f$.\\
For $r\geq2$, if $D^{r-1}\f$ is differentiable at $\x$, then $D^r\f(\x)(\mathbf{v}_1,\ldots,\mathbf{v}_r)\defeq\bigl(D(D^{r-1}\f)(\x)\mathbf{v}_1\bigr)(\mathbf{v}_2,\ldots,\mathbf{v}_r)$.
\item For real-valued $f:E\to\R$, the mixed second partial is $D_{ij}f\defeq D_i(D_jf)$ where defined.
\item $\f$ is \hlt{$C^r$} on $E$ iff $D^r\f:E\to\LL^r(\R^n,\R^m)$ exists and is continuous; $\f$ is \hlt{$C^\infty$} iff $\f$ is $C^r$ for every $r\geq1$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:second-derivative-partials-and-continuity} \hlt{Second Derivatives and Second Partials}\\
Let $\f=(f_1,\ldots,f_m):E\subseteq\R^n\to\R^m$ and $p\in E$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $D^2\f(p)$ exists, then each $D_{ij}f_\ell(p)$ exists and $D^2 f_\ell(p)(\mathbf{e}_i,\mathbf{e}_j)=D_{ij}f_\ell(p)$.
\item[$(\Leftarrow)$] If all $D_{ij}f_\ell$ exist near $p$ and are continuous at $p$, then $D^2\f(p)$ exists.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Apply Theorem \ref{thm:several-variable-componentwise-differentiability} to $D\f:E\to\LL(\R^n,\R^m)$. The entries of $D\f$ are $D_jf_\ell$ by Theorem \ref{thm:total-derivative-matrix-partials}; differentiating the $(\ell,j)$ entry in the $\mathbf{e}_i$ direction gives $D_{ij}f_\ell(p)$ and the displayed formula.
\item[$(\Leftarrow)$] Matrix entries of $D\f$ are functions $D_jf_\ell$. Since first partials $D_i(D_jf_\ell)=D_{ij}f_\ell$ exist near $p$ and are continuous at $p$, coordinate criterion at $p$ gives differentiability of $D\f$ there. Hence $D^2\f(p)$ exists.
\end{enumerate}
\hlt{Structural Commentary}: Second partials are matrix entries of the derivative of the first-derivative matrix.\\
\hlt{Key Insight}: Continuous second partials make the first-derivative matrix differentiable.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:second-derivative-symmetric} \hlt{Symmetry of the Second Derivative}\\
If $D^2\f(p)$ exists, then $D^2\f(p)(\mathbf{v},\mathbf{w})=D^2\f(p)(\mathbf{w},\mathbf{v})$ for all $\mathbf{v},\mathbf{w}\in\R^n$.
\begin{proof}
Reduce to scalar $f$ by Theorem \ref{thm:several-variable-componentwise-differentiability}.\\
For small $t$, set $\Delta(t,\mathbf{v},\mathbf{w})\defeq f(p+t\mathbf{v}+t\mathbf{w})-f(p+t\mathbf{v})-f(p+t\mathbf{w})+f(p)$.\\
Then $\Delta(t,\mathbf{v},\mathbf{w})=\Delta(t,\mathbf{w},\mathbf{v})$. The one-variable mean value theorem and differentiability of $Df$ at $p$ give $D^2 f(p)(\mathbf{v},\mathbf{w}) =\lim\limits_{t\to0}\frac{\Delta(t,\mathbf{v},\mathbf{w})}{t^2}$. The same formula with $\mathbf{v},\mathbf{w}$ reversed proves symmetry.\\
\hlt{Structural Commentary}: A pointwise second derivative is already forced to be a symmetric bilinear form.\\
\hlt{Key Insight}: The parallelogram difference is symmetric in the two directions.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:mixed-partials-from-second-derivative} \hlt{Mixed Partials from $D^2$}\\
If $D^2\f(p)$ exists, then $D_{ij}f_\ell(p)=D_{ji}f_\ell(p)$ for all components and all $i,j$.
\begin{proof}
By Theorem \ref{thm:second-derivative-partials-and-continuity}, $D_{ij}f_\ell(p)=D^2f_\ell(p)(\mathbf{e}_i,\mathbf{e}_j)$; and $D_{ji}f_\ell(p)=D^2f_\ell(p)(\mathbf{e}_j,\mathbf{e}_i)$.\\
Theorem \ref{thm:second-derivative-symmetric} makes the two right-hand sides equal.\\
\hlt{Structural Commentary}: Mixed partials are coordinate values of one bilinear map.\\
\hlt{Key Insight}: Symmetry of $D^2f_\ell(p)$ gives equality of mixed entries.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:higher-derivatives-symmetric} \hlt{Symmetry of Higher Derivatives}\\
Let $E\subseteq\R^n$ be open, $\f:E\to\R^m$, $p\in E$, and $r\geq2$.\\
If $D^r\f(p)$ exists, then for every $\sigma\in S_r$ and $\mathbf{v}_1,\ldots,\mathbf{v}_r\in\R^n$, $D^r\f(p)(\mathbf{v}_{\sigma(1)},\ldots,\mathbf{v}_{\sigma(r)})=D^r\f(p)(\mathbf{v}_1,\ldots,\mathbf{v}_r)$.
\begin{proof} Induct on $r$; $r=2$ is Theorem \ref{thm:second-derivative-symmetric}. For $r>2$, the induction hypothesis gives symmetry in slots $2,\ldots,r$ of $D^r\f(p)$. Applying Theorem \ref{thm:second-derivative-symmetric} to $D^{r-2}\f:E\to\LL^{r-2}(\R^n,\R^m)$ and evaluating at $\mathbf{v}_3,\ldots,\mathbf{v}_r$ gives symmetry in slots $1,2$. Hence every adjacent transposition fixes $D^r\f(p)$, so every $\sigma\in S_r$ fixes it.\\
\hlt{Structural Commentary}: Symmetry propagates from $D^2$.\\
\hlt{Key Insight}: Adjacent swaps generate $S_r$.
\end{proof}
\end{corollary}

The next rectangle argument is the classical coordinate route to equality of mixed partials. It is weaker in hypothesis than global $C^2$, but it proves the same compatibility of coordinate second derivatives.

\begin{theorem} \label{thm:rectangle-mean-value-mixed-partial} \hlt{Rectangle Mean Value Formula}\\
Let $E\subseteq\R^2$ be open, let $f:E\to\R$, and assume $D_1f$ and $D_{21}f=D_2(D_1f)$ exist on $E$. Let $h,k\neq0$, let $Q\subseteq E$ be the closed coordinate rectangle with opposite vertices $(a,b)$ and $(a+h,b+k)$, and set $\Delta(f,Q)\defeq f(a+h,b+k)-f(a+h,b)-f(a,b+k)+f(a,b)$.\\
Then there exists $(x,y)\in\operatorname{int}Q$ such that $\Delta(f,Q)=hk(D_{21}f)(x,y)$.
\begin{proof}
Let $I,J$ be the open intervals with endpoints $a,a+h$ and $b,b+k$. Set $u(t)\defeq f(t,b+k)-f(t,b)$.\\
By Corollary \ref{cor:mean-value-theorem}, $\Delta(f,Q)=u(a+h)-u(a)=h[(D_1f)(x,b+k)-(D_1f)(x,b)]$ for some $x\in I$. Applying the same result to $s\mapsto(D_1f)(x,s)$ gives $(D_1f)(x,b+k)-(D_1f)(x,b)=k(D_{21}f)(x,y)$ for some $y\in J$.\\
Thus $(x,y)\in I\times J=\operatorname{int}Q$ and $\Delta(f,Q)=hk(D_{21}f)(x,y)$.\\
\hlt{Structural Commentary}: Two MVT steps factor the rectangle increment.\\
\hlt{Key Insight}: $D_{21}f$ tracks vertical change of $D_1f$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:equality-of-mixed-partials} \hlt{Equality of Mixed Partials}\\
Let $(a,b)$ lie in an open set $E\subseteq\R^2$. If $D_1f$, $D_2f$, and $D_{21}f$ exist on $E$, and $D_{21}f$ is continuous at $(a,b)$, then $D_{12}f(a,b)$ exists and $(D_{12}f)(a,b)=(D_{21}f)(a,b)$. Hence $D_{12}f=D_{21}f$ when $f\in C^2(E)$.
\begin{proof} Let $A=(D_{21}f)(a,b)$, $\varepsilon>0$.\\
Continuity of $D_{21}f$, Theorem \ref{thm:rectangle-mean-value-mixed-partial} gives $\forall$ small $h,k\neq0$, $\abs{\frac{\Delta(f,Q)}{hk}-A}<\varepsilon$.\\
Fix such an $h$ and let $k\to0$. Since $D_2f$ exists at $(a+h,b)$ and $(a,b)$, $\frac{\Delta(f,Q)}{hk}\to \frac{(D_2f)(a+h,b)-(D_2f)(a,b)}{h}$.\\
Thus this last quotient differs from $A$ by at most $\varepsilon$ $\forall$ sufficiently small $h\neq0$.\\
Since $\varepsilon$ was arbitrary, the quotient tends to $A$ as $h\to0$, so $D_{12}f(a,b)$ exists and equals $A$.\\
\hlt{Structural Commentary}: Continuity of $D_{21}f$ controls local rectangle quotients.\\
\hlt{Key Insight}: The missing mixed partial is recovered by taking the two one-variable limits in the reverse order.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:mixed-partials-need-hypotheses} \hlt{Mixed Partials Need Hypotheses}\\
$D_{12}f$ and $D_{21}f$ are different iterated limits. Theorems \ref{thm:second-derivative-symmetric} and \ref{thm:equality-of-mixed-partials} give hypotheses for equality.
\end{perspective}
