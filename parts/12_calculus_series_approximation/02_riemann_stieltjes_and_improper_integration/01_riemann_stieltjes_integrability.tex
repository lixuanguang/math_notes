\subsection{Riemann--Stieltjes Integrability}

\begin{definition} \label{def:riemann-stieltjes-integral} \hlt{Riemann--Stieltjes Integral}\\
Let $\alpha:[a,b]\to\R$ be increasing, and let $\Pi=\{x_0,\ldots,x_n\}$ be a partition of $[a,b]$.\\
Let $f:[a,b]\to\R$ be bounded. Set $M_i\defeq\sup_{x\in[x_{i-1},x_i]}f(x)$ and $m_i\defeq\inf_{x\in[x_{i-1},x_i]}f(x)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{Stieltjes increment} on $[x_{i-1},x_i]$ is $\Delta\alpha_i\defeq\alpha(x_i)-\alpha(x_{i-1})\geq0$ for $1\leq i\leq n$.
\item The \hlt{upper Riemann--Stieltjes sum} is $U(\Pi,f,\alpha)\defeq\sum\limits_{i=1}^{n}M_i\Delta\alpha_i$.\\
The \hlt{lower Riemann--Stieltjes sum} is $L(\Pi,f,\alpha)\defeq\sum\limits_{i=1}^{n}m_i\Delta\alpha_i$.
\item The \hlt{upper Riemann--Stieltjes integral} is $\overline{\int_a^b f\,d\alpha}\defeq\inf_{\Pi}U(\Pi,f,\alpha)$.\\
The \hlt{lower Riemann--Stieltjes integral} is $\underline{\int_a^b f\,d\alpha}\defeq\sup_{\Pi}L(\Pi,f,\alpha)$.\\
Both extrema range over all partitions of $[a,b]$.
\item For a tagging $t=(t_i)_{i=1}^{n}$ of $\Pi$, the \hlt{tagged Riemann--Stieltjes sum} is $S_\alpha(f;\Pi,t)\defeq\sum\limits_{i=1}^{n}f(t_i)\Delta\alpha_i$.
\item The function $f$ is \hlt{Riemann integrable with respect to $\alpha$} iff $\overline{\int_a^b f\,d\alpha}=\underline{\int_a^b f\,d\alpha}$.\\
In this case, write $f\in\mathcal{R}(\alpha)$ and let $\int_a^b f\,d\alpha$ denote their common value.
\end{enumerate}
For $\alpha(x)=x$, these objects reduce to the ordinary Riemann objects in Definitions \ref{def:partition-and-riemann-upper-lower-sums} and \ref{def:tagged-partitions-and-riemann-sums}.
\end{definition}

\begin{definition} \label{def:partition-refinement} \hlt{Refinements}\\
Let $\Pi,\Pi^\ast,\Pi_1,\Pi_2$ be partitions of $[a,b]$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\Pi^\ast$ is a \hlt{refinement} of $\Pi$ iff $\Pi\subseteq\Pi^\ast$.
\item The \hlt{common refinement} of $\Pi_1$ and $\Pi_2$ is $\Pi_1\cup\Pi_2$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:refinement-monotonicity-riemann-stieltjes} \hlt{Refinement Monotonicity}\\
Let $\alpha:[a,b]\to\R$ be increasing, let $f:[a,b]\to\R$ be bounded, and let $\Pi^\ast$ refine $\Pi$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $L(\Pi,f,\alpha)\leq L(\Pi^\ast,f,\alpha)$.
\item $U(\Pi^\ast,f,\alpha)\leq U(\Pi,f,\alpha)$.
\end{enumerate}
\begin{proof} Prove both claims when $\Pi^\ast$ inserts one point $y\in(x_{j-1},x_j)$ into $\Pi$; every refinement obtained by finitely many such insertions.\\
Let $m'_j,m''_j$ and $M'_j,M''_j$ be the infima and suprema of $f$ on $[x_{j-1},y]$ and $[y,x_j]$, respectively.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item As $m'_j,m''_j\geq m_j$, $\alpha$ increasing, $L(\Pi^\ast,f,\alpha)-L(\Pi,f,\alpha) =(m'_j-m_j)\bigl[\alpha(y)-\alpha(x_{j-1})\bigr] +(m''_j-m_j)\bigl[\alpha(x_j)-\alpha(y)\bigr]\geq0$
\item As $M'_j,M''_j\leq M_j$, $\alpha$ increasing, $U(\Pi,f,\alpha)-U(\Pi^\ast,f,\alpha) =(M_j-M'_j)\bigl[\alpha(y)-\alpha(x_{j-1})\bigr] +(M_j-M''_j)\bigl[\alpha(x_j)-\alpha(y)\bigr]\geq0$.
\end{enumerate}
\hlt{Structural Commentary}: One-point insertion compares local weighted extrema; finite iteration handles every refinement.\\
\hlt{Key Insight}: Subdivision raises infima and lowers suprema while preserving total Stieltjes weight.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:piecewise-constant-envelope-view} \hlt{Piecewise-Constant Envelopes}\\
Definition \ref{def:riemann-stieltjes-integral} retains the ordinary lower and upper step heights but replaces interval widths by Stieltjes increments.\\
Comparability of these weighted envelopes under subdivision holds by Theorem \ref{thm:refinement-monotonicity-riemann-stieltjes}.
\end{perspective}

\begin{theorem} \label{thm:lower-integral-below-upper-integral} \hlt{Lower Integral Below Upper Integral}\\
If $\alpha:[a,b]\to\R$ is increasing and $f:[a,b]\to\R$ is bounded, then $\underline{\int_a^b f\,d\alpha}\leq\overline{\int_a^b f\,d\alpha}$.
\begin{proof}
Let $\Pi_1,\Pi_2$ be partitions of $[a,b]$, and set $\Pi^\ast\defeq\Pi_1\cup\Pi_2$.\\
The inequalities $L(\Pi_1,f,\alpha)\leq L(\Pi^\ast,f,\alpha)\leq U(\Pi^\ast,f,\alpha)\leq U(\Pi_2,f,\alpha)$ hold by Theorem \ref{thm:refinement-monotonicity-riemann-stieltjes}.\\
Taking the supremum over $\Pi_1$ and then the infimum over $\Pi_2$ proves the claim.\\
\hlt{Structural Commentary}: A common refinement places arbitrary lower and upper sums in one comparison chain.\\
\hlt{Key Insight}: Every lower sum lies below every upper sum.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:riemann-stieltjes-integrability-criterion} \hlt{Upper--Lower Criterion}\\
Let $\alpha:[a,b]\to\R$ be increasing, $f:[a,b]\to\R$ be bounded. Then $f\in\mathcal{R}(\alpha)$ iff $(\forall\varepsilon>0)(\exists\Pi)\ U(\Pi,f,\alpha)-L(\Pi,f,\alpha)<\varepsilon$.\\
Here $\Pi$ ranges over partitions of $[a,b]$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Set $I\defeq\int_a^b f\,d\alpha$, and fix $\varepsilon>0$. Choose $\Pi_L$ with $I-\frac{\varepsilon}{2}<L(\Pi_L,f,\alpha)$ by the definition of supremum.\\
Choose $\Pi_U$ with $U(\Pi_U,f,\alpha)<I+\frac{\varepsilon}{2}$ by the definition of infimum.\\
For $\Pi\defeq\Pi_L\cup\Pi_U$, $U(\Pi,f,\alpha)-L(\Pi,f,\alpha)\leq U(\Pi_U,f,\alpha)-L(\Pi_L,f,\alpha)<\varepsilon$ by Theorem \ref{thm:refinement-monotonicity-riemann-stieltjes}.
\item[$(\Leftarrow)$] Assume right-hand condition, set $D\defeq\overline{\int_a^b f\,d\alpha}-\underline{\int_a^b f\,d\alpha}\geq0$ by Theorem \ref{thm:lower-integral-below-upper-integral}. For $\varepsilon>0$, choose corresponding partition $\Pi$.\\
Then $D\leq U(\Pi,f,\alpha)-L(\Pi,f,\alpha)<\varepsilon$ by the definitions of the Darboux integrals. Hence $D=0$.
\end{enumerate}
\hlt{Structural Commentary}: Near-extremal sums merge through refinement; the converse bounds the integral gap by one partition gap.\\
\hlt{Key Insight}: The Darboux gap is the exact quantitative defect of integrability.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:tagged-riemann-stieltjes-sums} \hlt{Tagged Sums Controlled by the Darboux Gap}\\
Let $\alpha:[a,b]\to\R$ be increasing, let $f:[a,b]\to\R$ be bounded, and let $\varepsilon>0$.\\
Suppose that a partition $\Pi=\{x_0,\ldots,x_n\}$ satisfies $U(\Pi,f,\alpha)-L(\Pi,f,\alpha)<\varepsilon$, and let $s,t$ be taggings of $\Pi$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\sum\limits_{i=1}^{n}\abs{f(s_i)-f(t_i)}\Delta\alpha_i<\varepsilon$.
\item If $f\in\mathcal{R}(\alpha)$, then $\abs{S_\alpha(f;\Pi,t)-\int_a^b f\,d\alpha}<\varepsilon$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item As $\abs{f(s_i)-f(t_i)}\leq M_i-m_i$, $\Delta\alpha_i\geq0$ for $1\leq i\leq n$, $\sum\limits_{i=1}^{n}\abs{f(s_i)-f(t_i)}\Delta\alpha_i \leq\sum\limits_{i=1}^{n}(M_i-m_i)\Delta\alpha_i =U(\Pi,f,\alpha)-L(\Pi,f,\alpha)<\varepsilon$
\item The inequalities $L(\Pi,f,\alpha)\leq S_\alpha(f;\Pi,t)\leq U(\Pi,f,\alpha)$ hold by the tag conditions.\\
The integral lies in the same interval by the definitions of the lower and upper integrals.\\
Its length is less than $\varepsilon$, so the asserted distance is less than $\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Intervalwise oscillation bounds control tag changes and locate the integral inside the same Darboux bracket.\\
\hlt{Key Insight}: A small Darboux gap makes every compatible tagged sum close.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:darboux-and-riemann-sum-definitions} \hlt{Darboux Gaps and Tagged Sums}\\
For $\alpha(x)=x$, Theorem \ref{thm:tagged-riemann-stieltjes-sums} traps every compatible tagged Riemann sum near the integral.\\
For general $\alpha$, the same mechanism measures oscillation with the weights $\Delta\alpha_i$.
\end{perspective}

\begin{theorem} \label{thm:continuous-functions-riemann-stieltjes-integrable} \hlt{Continuous Functions Are Riemann--Stieltjes Integrable}\\
If $f:[a,b]\to\R$ is continuous and $\alpha:[a,b]\to\R$ is increasing, then $f\in\mathcal{R}(\alpha)$.
\begin{proof}
The interval $[a,b]$ is compact by Theorem \ref{thm:heine-borel-rk}, so $f$ is bounded by Theorem \ref{thm:extreme-value-theorem-compact}. Set $A\defeq\alpha(b)-\alpha(a)$, and fix $\varepsilon>0$.\\
If $A=0$, every Darboux gap is zero, so $f\in\mathcal{R}(\alpha)$ by Theorem \ref{thm:riemann-stieltjes-integrability-criterion}. Assume $A>0$ and set $\eta\defeq\varepsilon/(2A)$.\\
The function $f$ is uniformly continuous by Theorem \ref{thm:compact-continuity-uniform-continuity}. Choose $\delta>0$ such that $\abs{x-y}<\delta$ implies $\abs{f(x)-f(y)}<\eta$.\\
Choose partition $\Pi$, $\operatorname{mesh}(\Pi)<\delta$. Extrema on every partition interval attained by Theorem \ref{thm:extreme-value-theorem-compact}, $M_i-m_i<\eta$ for $1\leq i\leq n$.\\
Hence $U(\Pi,f,\alpha)-L(\Pi,f,\alpha) =\sum\limits_{i=1}^{n}(M_i-m_i)\Delta\alpha_i <\eta A<\varepsilon$. Therefore $f\in\mathcal{R}(\alpha)$ by Theorem \ref{thm:riemann-stieltjes-integrability-criterion}.\\
\hlt{Structural Commentary}: Uniform continuity bounds local oscillations; total Stieltjes weight controls the Darboux gap.\\
\hlt{Key Insight}: Uniformly small oscillations remain small after finite Stieltjes weighting.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:monotone-functions-integrable-continuous-integrator} \hlt{Monotone Functions and Continuous Integrators}\\
If $f:[a,b]\to\R$ is monotone and $\alpha:[a,b]\to\R$ is continuous and increasing, then $f\in\mathcal{R}(\alpha)$.
\begin{proof}
The function $f$ is bounded. Set $V\defeq\abs{f(b)-f(a)}$, and fix $\varepsilon>0$.\\
If $V=0$, monotonicity makes $f$ constant, so $f\in\mathcal{R}(\alpha)$ by Theorem \ref{thm:riemann-stieltjes-integrability-criterion}.\\
Assume $V>0$. The function $\alpha$ is uniformly continuous by Theorem \ref{thm:compact-continuity-uniform-continuity}.\\
Choose $\delta>0$ such that $\abs{x-y}<\delta$ implies $\abs{\alpha(x)-\alpha(y)}<\varepsilon/V$. Choose a partition $\Pi$ with $\operatorname{mesh}(\Pi)<\delta$.\\
If $f$ is increasing, then $M_i-m_i=f(x_i)-f(x_{i-1})$; if $f$ is decreasing, then $M_i-m_i=f(x_{i-1})-f(x_i)$.\\
In either case, $U(\Pi,f,\alpha)-L(\Pi,f,\alpha) =\sum\limits_{i=1}^{n}(M_i-m_i)\Delta\alpha_i <\frac{\varepsilon}{V}\sum\limits_{i=1}^{n}(M_i-m_i) =\varepsilon$. Hence $f\in\mathcal{R}(\alpha)$ by Theorem \ref{thm:riemann-stieltjes-integrability-criterion}.\\
\hlt{Structural Commentary}: Uniform continuity of $\alpha$ limits every weight; monotonicity telescopes the total value oscillation.\\
\hlt{Key Insight}: Small Stieltjes weights distribute finite monotone variation below any gap threshold.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:finitely-many-discontinuities-integrable} \hlt{Finitely Many Discontinuities}\\
Let $\alpha:[a,b]\to\R$ be increasing and let $f:[a,b]\to\R$ be bounded.\\
If $E\defeq\{x\in[a,b]:f\text{ is discontinuous at }x\}$ is finite and $\alpha$ is continuous at all points of $E$, then $f\in\mathcal{R}(\alpha)$.
\begin{proof}
Set $A\defeq\alpha(b)-\alpha(a)$ and $B\defeq\sup_{x\in[a,b]}\abs{f(x)}$. Fix $\varepsilon>0$ and set $\eta\defeq\varepsilon/(A+2B+1)$.\\
If $E=\emptyset$, the claim follows by Theorem \ref{thm:continuous-functions-riemann-stieltjes-integrable}. Assume $E\neq\emptyset$ and write $E=\{c_1,\ldots,c_N\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Isolation}: Choose closed intervals $I_j=[u_j,v_j]\subseteq[a,b]$ with pairwise disjoint relative interiors.\\
Require $c_j\in\operatorname{int}_{[a,b]}I_j$, $\alpha(v_j)-\alpha(u_j)<\frac{\eta}{N}$ for $1\leq j\leq N$. These choices exist by finiteness of $E$, continuity of $\alpha$ at each $c_j$.
\item \hlt{Compact Complement}: Set $K\defeq[a,b]\setminus\bigcup_{j=1}^{N}\operatorname{int}_{[a,b]}I_j$.\\
Then $K$ is compact and contains no point of $E$, so $f|_K$ is uniformly continuous by Theorem \ref{thm:compact-continuity-uniform-continuity}.\\
Choose $\delta>0$ such that $x,y\in K$ and $\abs{x-y}<\delta$ imply $\abs{f(x)-f(y)}<\eta$.
\item \hlt{Adapted Partition}: Choose a partition $\Pi$ containing every $u_j,v_j$, and leave each $I_j$ unsubdivided.\\
Subdivide each component of $K$ into intervals of length less than $\delta$.\\
Every partition interval is then either some $I_j$ or is contained in $K$.
\item \hlt{Gap Estimate}: On intervals contained in $K$, $M_i-m_i\leq\eta$; on each $I_j$, $M_i-m_i\leq2B$.\\
Hence $U(\Pi,f,\alpha)-L(\Pi,f,\alpha) \leq\eta A+2B\sum\limits_{j=1}^{N}\bigl[\alpha(v_j)-\alpha(u_j)\bigr] <\eta(A+2B)<\varepsilon$. Therefore $f\in\mathcal{R}(\alpha)$ by Theorem \ref{thm:riemann-stieltjes-integrability-criterion}.
\end{enumerate}
\hlt{Structural Commentary}: Small Stieltjes neighborhoods isolate discontinuities; uniform continuity controls the compact complement.\\
\hlt{Key Insight}: Discontinuities carry arbitrarily small weight when $\alpha$ is continuous there.
\end{proof}
\end{theorem}

\begin{example} \label{ex:dirichlet-function-not-riemann-integrable} \hlt{Dirichlet's Function Is Not Riemann Integrable}\\
Let $g=\mathbf{1}_{\Q}$ on $[0,1]$. Every nondegenerate subinterval contains rational and irrational points by Corollary \ref{cor:rationals-and-irrationals-dense}.\\
Thus $M_i=1$ and $m_i=0$ on every partition interval. Therefore $(\forall\Pi)\ U(\Pi,g)=1$ and $L(\Pi,g)=0$, so $g\notin\mathcal{R}[0,1]$.
\end{example}

\begin{theorem} \label{thm:continuous-composition-preserves-riemann-stieltjes-integrability} \hlt{Continuous Composition Preserves Integrability}\\
Let $\alpha:[a,b]\to\R$ be increasing, let $f\in\mathcal{R}(\alpha)$, and suppose $(\forall x\in[a,b])\,m\leq f(x)\leq M$.\\
If $\phi:[m,M]\to\R$ is continuous and $h\defeq\phi\circ f$, then $h\in\mathcal{R}(\alpha)$.
\begin{proof}
The interval $[m,M]$ is compact by Theorem \ref{thm:heine-borel-rk}, so $C\defeq\sup_{y\in[m,M]}\abs{\phi(y)}$ is finite by Theorem \ref{thm:extreme-value-theorem-compact}.\\
Set $A\defeq\alpha(b)-\alpha(a)$, fix $\varepsilon>0$, and set $\eta\defeq\varepsilon/(A+2C+1)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Uniform Value Control}: The function $\phi$ is uniformly continuous by Theorem \ref{thm:compact-continuity-uniform-continuity}.\\
Choose $\gamma>0$ such that $\abs{s-t}<\gamma$ implies $\abs{\phi(s)-\phi(t)}<\eta$ for $s,t\in[m,M]$.
\item \hlt{Oscillation Split}: Choose a partition $\Pi$ satisfying $U(\Pi,f,\alpha)-L(\Pi,f,\alpha)<\gamma\eta$ by Theorem \ref{thm:riemann-stieltjes-integrability-criterion}.\\
Let $M_i^h,m_i^h$ be the suprema and infima of $h$ on $[x_{i-1},x_i]$, and set $I\defeq\{i:M_i-m_i<\gamma\}$ and $J\defeq\{i:M_i-m_i\geq\gamma\}$.
\item \hlt{Weight Control}: For $i\in I$, $M_i^h-m_i^h\leq\eta$ by uniform continuity; for $i\in J$, $M_i^h-m_i^h\leq2C$.\\
The large-oscillation indices satisfy $\gamma\sum\limits_{i\in J}\Delta\alpha_i \leq\sum\limits_{i\in J}(M_i-m_i)\Delta\alpha_i <\gamma\eta$, so $\sum\limits_{i\in J}\Delta\alpha_i<\eta$.
\item \hlt{Gap Estimate}: Darboux gap $U(\Pi,h,\alpha)-L(\Pi,h,\alpha) \leq\eta\sum\limits_{i\in I}\Delta\alpha_i +2C\sum\limits_{i\in J}\Delta\alpha_i <\eta(A+2C)<\varepsilon$; $h\in\mathcal{R}(\alpha)$ by Theorem \ref{thm:riemann-stieltjes-integrability-criterion}.
\end{enumerate}
\hlt{Structural Commentary}: An oscillation split converts uniform continuity of $\phi$ into a weighted Darboux-gap estimate.\\
\hlt{Key Insight}: Intervals with large $f$-oscillation have small total Stieltjes weight.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:riemann-stieltjes-integral-properties} \hlt{Basic Properties of the Integral}\\
Let $\alpha,\alpha_1,\alpha_2:[a,b]\to\R$ be increasing, and let $f,g:[a,b]\to\R$ be bounded.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Linearity in the integrand}: If $f,g\in\mathcal{R}(\alpha)$, $c\in\R$, then $f+g,cf\in\mathcal{R}(\alpha)$; $\int_a^b(f+g)\,d\alpha=\int_a^b f\,d\alpha+\int_a^b g\,d\alpha$; $\int_a^b cf\,d\alpha=c\int_a^b f\,d\alpha$.
\item \hlt{Order}: Assume $f,g\in\mathcal{R}(\alpha)$ and $(\forall x\in[a,b])\,f(x)\leq g(x)$. Then $\int_a^b f\,d\alpha\leq\int_a^b g\,d\alpha$.
\item \hlt{Interval additivity}: Assume $f\in\mathcal{R}(\alpha)$, $d\in(a,b)$. Then $f|_{[a,d]}\in\mathcal{R}(\alpha|_{[a,d]})$ and $f|_{[d,b]}\in\mathcal{R}(\alpha|_{[d,b]})$; $\int_a^d f\,d\alpha+\int_d^b f\,d\alpha=\int_a^b f\,d\alpha$.
\item \hlt{Uniform bound}: Assume $f\in\mathcal{R}(\alpha)$ and $M\in[0,\infty)$. If $(\forall x\in[a,b])\,\abs{f(x)}\leq M$, then $\abs{\int_a^b f\,d\alpha}\leq M[\alpha(b)-\alpha(a)]$.
\item \hlt{Linearity in the integrator}: If $f\in\mathcal{R}(\alpha_1)\cap\mathcal{R}(\alpha_2)$, then $f\in\mathcal{R}(\alpha_1+\alpha_2)$. In this case, $\int_a^b f\,d(\alpha_1+\alpha_2)=\int_a^b f\,d\alpha_1+\int_a^b f\,d\alpha_2$.\\
If $\lambda>0$ and $f\in\mathcal{R}(\alpha)$, then $f\in\mathcal{R}(\lambda\alpha)$ and $\int_a^b f\,d(\lambda\alpha)=\lambda\int_a^b f\,d\alpha$.
\end{enumerate}
\begin{proof}
For a partition $\Pi$, set $G_\beta^\Pi(h)\defeq U(\Pi,h,\beta)-L(\Pi,h,\beta)$.\\
Finitely many integrable pairs admit common partitions with arbitrarily small gaps by Theorems \ref{thm:riemann-stieltjes-integrability-criterion} and \ref{thm:refinement-monotonicity-riemann-stieltjes}.\\
An exact identity among their tagged sums passes to their integrals by Theorem \ref{thm:tagged-riemann-stieltjes-sums}(ii) and the triangle inequality.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Linearity in the integrand}: $(\forall\Pi)\,G_\alpha^\Pi(f+g)\leq G_\alpha^\Pi(f)+G_\alpha^\Pi(g)$. $(\forall\Pi)\,G_\alpha^\Pi(cf)=\abs{c}G_\alpha^\Pi(f)$; $f+g,cf\in\mathcal{R}(\alpha)$ by Theorem \ref{thm:riemann-stieltjes-integrability-criterion}.\\
The identities $S_\alpha(f+g;\Pi,t)=S_\alpha(f;\Pi,t)+S_\alpha(g;\Pi,t)$ and $S_\alpha(cf;\Pi,t)=cS_\alpha(f;\Pi,t)$ pass to the asserted integral identities.
\item \hlt{Order}: Pointwise order implies $(\forall\Pi)\,L(\Pi,f,\alpha)\leq L(\Pi,g,\alpha)$. Taking suprema proves $\int_a^b f\,d\alpha\leq\int_a^b g\,d\alpha$ by Definition \ref{def:riemann-stieltjes-integral}.
\item \hlt{Interval additivity}: For $\varepsilon>0$, choose $\Pi\ni d$ with $G_\alpha^\Pi(f)<\varepsilon$ by the common-refinement observation.\\
Set $\Pi^-\defeq\Pi\cap[a,d]$, $\Pi^+\defeq\Pi\cap[d,b]$. Then $G_\alpha^\Pi(f)=G_\alpha^{\Pi^-}(f)+G_\alpha^{\Pi^+}(f)$, both restrictions  integrable by Theorem \ref{thm:riemann-stieltjes-integrability-criterion}.\\
The tagged sum over $\Pi$ is the sum of its restrictions to $\Pi^-$ and $\Pi^+$; the integral identity follows by the preceding observation.
\item \hlt{Uniform bound}: The constant function $q$ has integral $q[\alpha(b)-\alpha(a)]$ by Definition \ref{def:riemann-stieltjes-integral}.\\
The asserted bound follows by part (ii) with $-M\leq f\leq M$.
\item \hlt{Linearity in the integrator}: $(\forall\Pi)\,G_{\alpha_1+\alpha_2}^\Pi(f)=G_{\alpha_1}^\Pi(f)+G_{\alpha_2}^\Pi(f)$. $(\forall\Pi)\,G_{\lambda\alpha}^\Pi(f)=\lambda G_\alpha^\Pi(f)$.\\
The claimed integrability follows by common refinement and Theorem \ref{thm:riemann-stieltjes-integrability-criterion}.\\
The corresponding tagged-sum identities pass to the asserted integral identities by the preceding observation.
\end{enumerate}
\hlt{Structural Commentary}: Gap identities establish integrability; tagged-sum identities then transfer to integral values.\\
\hlt{Key Insight}: Exact finite-sum algebra survives arbitrarily small Darboux gaps.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:products-and-absolute-values-riemann-stieltjes} \hlt{Products and Absolute Values}\\
Let $\alpha:[a,b]\to\R$ be increasing and let $f,g\in\mathcal{R}(\alpha)$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $fg\in\mathcal{R}(\alpha)$.
\item $\abs{f}\in\mathcal{R}(\alpha)$.
\item $\abs{\int_a^b f\,d\alpha}\leq\int_a^b\abs{f}\,d\alpha$.
\end{enumerate}
\begin{proof}
The functions $f,g$ are bounded by Definition \ref{def:riemann-stieltjes-integral}; hence $f\pm g$ are bounded.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The functions $f\pm g$ belong to $\mathcal{R}(\alpha)$ by Theorem \ref{thm:riemann-stieltjes-integral-properties}(i).\\
Choose a compact interval containing both ranges. Then $(f\pm g)^2\in\mathcal{R}(\alpha)$ by Theorem \ref{thm:continuous-composition-preserves-riemann-stieltjes-integrability} with $\phi(t)=t^2$.\\
Hence $fg\in\mathcal{R}(\alpha)$ by $4fg=(f+g)^2-(f-g)^2$ and Theorem \ref{thm:riemann-stieltjes-integral-properties}(i).
\item Choose a compact interval containing $f([a,b])$. Then $\abs{f}\in\mathcal{R}(\alpha)$ by Theorem \ref{thm:continuous-composition-preserves-riemann-stieltjes-integrability} with $\phi(t)=\abs{t}$.
\item Bounds $-\int_a^b\abs{f}\,d\alpha\leq\int_a^b f\,d\alpha\leq\int_a^b\abs{f}\,d\alpha$ hold by Theorem \ref{thm:riemann-stieltjes-integral-properties}(i)--(ii),  $-\abs{f}\leq f\leq\abs{f}$. Equivalent to claim.
\end{enumerate}
\hlt{Structural Commentary}: Continuous composition handles squares and absolute values; linearity and order complete the reductions.\\
\hlt{Key Insight}: Products reduce to squares, while $-\abs{f}\leq f\leq\abs{f}$ controls the integral.
\end{proof}
\end{theorem}
