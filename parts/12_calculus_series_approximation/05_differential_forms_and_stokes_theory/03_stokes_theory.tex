\subsection{Stokes Theory}

% **Role:** calculus as local linearization plus global geometric integration.
%
% Generalized Stokes theorem, Green's theorem, classical Stokes theorem, divergence theorem.

Stokes theory relates exterior differentiation to oriented boundary.\\
Green's theorem, classical Stokes, and the divergence theorem are its coordinate forms.

\begin{definition} \label{def:affine-chains-and-boundaries} \hlt{Chains and Boundaries}\\
Fix $n\in\N$ and $k\in\Z_{\geq0}$, set $Q^0\defeq\{0\}$, and let $V\subseteq\R^n$ be open.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $\mathbf{p}_0,\ldots,\mathbf{p}_k\in\R^n$. Set $\sigma(\mathbf{u})\defeq\mathbf{p}_0+\sum\limits_{i=1}^{k}u_i(\mathbf{p}_i-\mathbf{p}_0)$.\\
This map represents the oriented affine \hlt{$k$-simplex} $[\mathbf{p}_0,\ldots,\mathbf{p}_k]$. Work in the real vector space generated by these symbols.\\
For a permutation $\pi$ of $\{0,\ldots,k\}$, impose $[\mathbf{p}_{\pi(0)},\ldots,\mathbf{p}_{\pi(k)}]=\operatorname{sgn}(\pi)[\mathbf{p}_0,\ldots,\mathbf{p}_k]$.
\item A \hlt{$C^2$ oriented $k$-simplex} in $V$ is a map $\Phi\in C^2(Q^k,V)$. A \hlt{$C^2$ $k$-chain} is a finite sum $\Psi=\sum\limits_\ell c_\ell\Phi_\ell$ with $c_\ell\in\R$.
\item If $k\geq1$ and $\omega$ is a $k$-form on $V$, set $\int_\Psi\omega\defeq\sum\limits_\ell c_\ell\int_{\Phi_\ell}\omega$. If $k=0$ and $f$ is a $0$-form, set $\int_{\Phi}f\defeq f(\Phi(0))$ and extend linearly.
\item Let $\mathbf{q}_0=\ze$ and $\mathbf{q}_i=\mathbf{e}_i$ for $1\leq i\leq k$.\\
For $k\geq1$ and $j\in\{0,\ldots,k\}$, let $\iota_j:Q^{k-1}\to Q^k$ be the affine map underlying $[\mathbf{q}_0,\ldots,\widehat{\mathbf{q}_j},\ldots,\mathbf{q}_k]$.\\
Set $\partial_{\mathrm{ch}}\Phi\defeq\sum\limits_{j=0}^{k}(-1)^j(\Phi\circ\iota_j)$ for $k\geq1$ and extend linearly. Set $\partial_{\mathrm{ch}}\Psi\defeq0$ on $0$-chains.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:stokes-theorem-differential-forms} \hlt{Stokes' Theorem}\\
Fix $m\in\N$, $k\in J_m$. Let $V\subseteq\R^m$ be open, $\Psi$ be $C^2$ $k$-chain in $V$, $\omega$ be $C^1$ $(k-1)$-form on $V$. Then $\int_{\Psi}d\omega=\int_{\partial_{\mathrm{ch}}\Psi}\omega$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Linearity Reduction}: By linearity, take $\Psi=\Phi$. Choose a $C^2$ extension $T:U\to V$ on an open neighborhood $U$ of $Q^k$.\\
Let $\sigma:Q^k\to U$ be the standard inclusion; then $\Phi=T\circ\sigma$.
\item \hlt{Pullback}: Set $\lambda\defeq T^\ast\omega$. The identity $\int_{\Phi}d\omega=\int_{\sigma}d\lambda$ holds by Theorem \ref{thm:pullback-basic-laws}(iii),(v).\\
For $k\geq2$, pullback naturality holds facewise by Theorem \ref{thm:pullback-basic-laws}(v). Thus $\int_{\partial_{\mathrm{ch}}\Phi}\omega=\int_{\partial_{\mathrm{ch}}\sigma}\lambda$.\\
For $k=1$, the same boundary identity holds directly by the definitions of pullback and integration of $0$-forms.
\item \hlt{Coordinate Form}: By standard presentation and linearity, take one basic summand.\\
It has the form $\lambda=f\,dx_1\wedge\cdots\wedge\widehat{dx_r}\wedge\cdots\wedge dx_k$ for $r\in J_k$. For $k=1$, interpret the empty wedge as $1$.\\
Coefficientwise expansion is valid by Definition \ref{def:wedge-product-and-exterior-derivative}. Reordering contributes $(-1)^{r-1}$ by Proposition \ref{prop:basic-forms-standard-presentation}(i).\\
Hence $d\lambda=(-1)^{r-1}(D_rf)\,dx_1\wedge\cdots\wedge dx_k$.
\item \hlt{Boundary Faces}: Write $\partial_{\mathrm{ch}}\sigma=\sum\limits_{j=0}^{k}(-1)^j\iota_j$. Pullback of $\lambda$ vanishes when $j\notin\{0,r\}$. Indeed, $x_j\circ\iota_j=0$ while $dx_j$ occurs in $\lambda$.\\
The $j=0$ pullback has sign $(-1)^{r-1}$, restricts to $x_r=1-\sum\limits_{i\neq r}x_i$. The $j=r$ term has boundary sign $(-1)^r$, restricts to $x_r=0$.
\item \hlt{Conclusion}: If $k=1$, both sides equal $f(1)-f(0)$ by Theorem \ref{thm:fundamental-theorem-calculus-riemann}. Suppose $k\geq2$, $\mathbf{x}_{\widehat r}$ for coordinates other than $x_r$.\\
Set $s_r(\mathbf{x}_{\widehat r})\defeq1-\sum\limits_{i\neq r}x_i$; $g_r(\mathbf{x}_{\widehat r})\defeq f(\mathbf{x}_{\widehat r},s_r(\mathbf{x}_{\widehat r}))-f(\mathbf{x}_{\widehat r},0)$. Boundary-face integral equals $(-1)^{r-1}\int_{Q^{k-1}}g_r$.\\
The interior integral has the same value by Theorem \ref{thm:fundamental-theorem-calculus-riemann} on the $x_r$-slices.
\end{enumerate}
\hlt{Structural Commentary}: Pullback moves the chain to $Q^k$; face restrictions isolate the missing coordinate's endpoints.\\
\hlt{Key Insight}: The wedge sign $(-1)^{r-1}$ matches the orientation difference between the slanted face and $x_r=0$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:closed-and-exact-forms} \hlt{Closed and Exact Forms}\\
Fix $n\in\N$, $k\in J_n$. Let $E\subseteq\R^n$ be open, $\omega$ be a $C^1$ $k$-form on $E$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\omega$ is \hlt{exact} iff $(\exists\,\lambda\in C^1(E,\bigwedge^{k-1}((\R^n)')))\ \omega=d\lambda$.
\item $\omega$ is \hlt{closed} iff $d\omega=0$.
\end{enumerate}
\end{definition}

\begin{perspective} \label{per:closed-exact-local-global} \hlt{Closed Is Local, Exact Is Global}\\
If $\omega=d\lambda$ with $\lambda\in C^2$, then $\omega$ is closed by Theorem \ref{thm:exterior-derivative-basic-laws}(ii). Converse depends on domain.\\
On $\R^2\setminus\{\ze\}$, write $(x,y)\defeq(x_1,x_2)$ and $(dx,dy)\defeq(dx_1,dx_2)$.\ Corm $\omega=\frac{-y}{x^2+y^2}\,dx+\frac{x}{x^2+y^2}\,dy$ satisfies $d\omega=0$.\\
Set $\gamma(u)\defeq(\cos(2\pi u),\sin(2\pi u))$ for $u\in Q^1=[0,1]$; $\int_\gamma\omega=\int_0^1 2\pi\,du=2\pi$ holds by Definition \ref{def:integration-of-differential-form}.\\
As $\partial_{\mathrm{ch}}\gamma=0$, by exactness $\int_\gamma\omega=0$ by Theorem \ref{thm:stokes-theorem-differential-forms}. Holes obstruct global antiderivatives even when local derivative tests vanish.
\end{perspective}

\begin{theorem} \label{thm:poincare-lemma-convex-rudin} \hlt{Poincar\'e Lemma on Convex Sets}\\
Fix $n\in\N$ and $k\in J_n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $E\subseteq\R^n$ be open and convex, and let $\omega$ be a closed $C^1$ $k$-form on $E$. Then $\omega$ is exact.
\item Let $C,E\subseteq\R^n$ be open, $C$ be convex, $\Phi:C\to E$ be a $C^2$ diffeomorphism. If $\omega$ is a closed $C^1$ $k$-form on $E$, then $\omega$ is exact.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $\mathbf{a}\in E$. Let $H_t(\x)\defeq\mathbf{a}+t(\x-\mathbf{a})$ for $t\in[0,1]$, $\omega=\sum\limits_I b_I\,dx_I$.\\
For $I=(i_1<\cdots<i_k)$, let $I\setminus i_r$ denote increasing index obtained by omitting $i_r$. For $k=1$, set $dx_{I\setminus i_1}\defeq1$.\\
Set $A_I(\x)\defeq\int_0^1t^{k-1}b_I(H_t(\x))\,dt$;  $K\omega\defeq\sum\limits_I\sum\limits_{r=1}^{k}(-1)^{r-1}(x_{i_r}-a_{i_r})A_I(\x)\,dx_{I\setminus i_r}$, use same formula in each positive degree.\\
The coordinate difference-quotient bound holds by Corollary \ref{cor:mean-value-theorem}.\\
The derivatives are uniformly continuous on a compact tube around the segment from $\mathbf{a}$ to $\x$ by Theorem \ref{thm:compact-continuity-uniform-continuity}.\\
Hence the coordinate difference quotients of $b_I(H_t(\x))$ converge uniformly in $t$.\\
Thus, $D_jA_I(\x)=\int_0^1t^k(D_jb_I)(H_t(\x))\,dt$ by Theorem \ref{thm:riemann-stieltjes-integral-properties}(i),(iv). By linearity, consider one basic summand $b_I\,dx_I$.\\
The undifferentiated factors in $dK(b_I\,dx_I)$ contribute $kA_I$ to the $dx_I$ coefficient.\\
Terms differentiated in $i_r$ direction contribute $\sum\limits_{r=1}^{k}(x_{i_r}-a_{i_r})D_{i_r}A_I$. Sum is $\int_0^1\frac{d}{dt}[t^kb_I(H_t(\x))]\,dt=b_I(\x)$ by Theorem \ref{thm:fundamental-theorem-calculus-riemann}.\\
Coefficients containing $dx_j$ with $j\notin I$ cancel against those of $K(d(b_I\,dx_I))$. Signs are opposite by Proposition \ref{prop:basic-forms-standard-presentation}(i).\\
Hence $dK\omega+Kd\omega=\omega$. If $d\omega=0$, then $K\omega\in C^1$ and $\omega=d(K\omega)$, so $\omega$ is exact.
\item Let $\Phi:C\to E$ be as stated, $\omega$ be a closed $C^1$ $k$-form on $E$. Identity $d(\Phi^\ast\omega)=\Phi^\ast(d\omega)=0$ holds by Theorem \ref{thm:pullback-basic-laws}(iii).\\
Choose a $C^1$ $(k-1)$-form $\eta$ on $C$ such that $\Phi^\ast\omega=d\eta$ by (i). Then $\omega=(\Phi^{-1})^\ast(\Phi^\ast\omega)=d((\Phi^{-1})^\ast\eta)$ by Theorem \ref{thm:pullback-basic-laws}(iii),(iv).
\end{enumerate}
\hlt{Structural Commentary}: The contraction satisfies $dK+Kd=\operatorname{id}$; pullback transports it through a diffeomorphism.\\
\hlt{Key Insight}: If $d\omega=0$, the homotopy identity reduces to $\omega=d(K\omega)$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:vector-fields-as-differential-forms} \hlt{Vector Fields as Differential Forms}\\
Let $E\subseteq\R^3$ be open, $(x,y,z)\defeq(x_1,x_2,x_3)$, $(dx,dy,dz)\defeq(dx_1,dx_2,dx_3)$. Fix $\mathbf{H}=(H_1,H_2,H_3)\in C^1(E,\R^3)$.\\
Set $\nabla\cdot\mathbf{H}\defeq D_1H_1+D_2H_2+D_3H_3$; $\nabla\times\mathbf{H}\defeq(D_2H_3-D_3H_2,D_3H_1-D_1H_3,D_1H_2-D_2H_1)$.\\
Set $\lambda_{\mathbf{H}}\defeq H_1\,dx+H_2\,dy+H_3\,dz$; $\omega_{\mathbf{H}}\defeq H_1\,dy\wedge dz+H_2\,dz\wedge dx+H_3\,dx\wedge dy$.
\end{definition}

\begin{theorem} \label{thm:vector-analysis-forms-dictionary} \hlt{Vector Analysis Dictionary in $\R^3$}\\
Let $E\subseteq\R^3$ be open, use notation of Definition \ref{def:vector-fields-as-differential-forms}. Fix $\mathbf{F}\in C^1(E,\R^3)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $u\in C^1(E)$, $\mathbf{F}=\nabla u$ iff $\lambda_{\mathbf{F}}=du$.
\item $d\lambda_{\mathbf{F}}=\omega_{\nabla\times\mathbf{F}}$.
\item $\nabla\times\mathbf{F}=0$ iff $d\lambda_{\mathbf{F}}=0$.
\item For $\mathbf{G}\in C^1(E,\R^3)$, $\mathbf{F}=\nabla\times\mathbf{G}$ iff $\omega_{\mathbf{F}}=d\lambda_{\mathbf{G}}$.
\item $d\omega_{\mathbf{F}}=(\nabla\cdot\mathbf{F})\,dx\wedge dy\wedge dz$.
\item $\nabla\cdot\mathbf{F}=0$ iff $d\omega_{\mathbf{F}}=0$.
\item Assume $E$ is $C^2$-diffeomorphic to a convex open set and $\nabla\times\mathbf{F}=0$. Then $\exists u\in C^1(E):\mathbf{F}=\nabla u$.
\item Assume $E$ is $C^2$-diffeomorphic to a convex open set and $\nabla\cdot\mathbf{F}=0$. Then $\exists \mathbf{G}\in C^1(E,\R^3):\mathbf{F}=\nabla\times\mathbf{G}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) If $\mathbf{F}=\nabla u$, then $\lambda_{\mathbf{F}}=(D_1u)\,dx+(D_2u)\,dy+(D_3u)\,dz=du$.
\item[$(i)$] ($\Leftarrow$) If $\lambda_{\mathbf{F}}=du$, then $\mathbf{F}=\nabla u$ by equality of the $dx,dy,dz$ coefficients.
\item[$(ii)$] The $dy\wedge dz$ coefficient of $d\lambda_{\mathbf{F}}$ is $D_2F_3-D_3F_2$. Cyclic permutation produces other two curl coefficients, so $d\lambda_{\mathbf{F}}=\omega_{\nabla\times\mathbf{F}}$.
\item[$(iii)$] ($\Rightarrow$) If $\nabla\times\mathbf{F}=0$, then $d\lambda_{\mathbf{F}}=\omega_{\nabla\times\mathbf{F}}=0$ by (ii).
\item[$(iii)$] ($\Leftarrow$) If $d\lambda_{\mathbf{F}}=0$, then $\omega_{\nabla\times\mathbf{F}}=0$ by (ii). Hence $\nabla\times\mathbf{F}=0$ by Proposition \ref{prop:basic-forms-standard-presentation}(ii).
\item[$(iv)$] ($\Rightarrow$) If $\mathbf{F}=\nabla\times\mathbf{G}$, then $\omega_{\mathbf{F}}=\omega_{\nabla\times\mathbf{G}}$. This form equals $d\lambda_{\mathbf{G}}$ by (ii).
\item[$(iv)$] ($\Leftarrow$) If $\omega_{\mathbf{F}}=d\lambda_{\mathbf{G}}$, then $\omega_{\mathbf{F}}=\omega_{\nabla\times\mathbf{G}}$ by (ii). Hence $\mathbf{F}=\nabla\times\mathbf{G}$ by Proposition \ref{prop:basic-forms-standard-presentation}(ii).
\item[$(v)$] Only derivatives in missing coordinate survive in $d\omega_{\mathbf{F}}$. Thus $d\omega_{\mathbf{F}}=(D_1F_1+D_2F_2+D_3F_3)\,dx\wedge dy\wedge dz=(\nabla\cdot\mathbf{F})\,dx\wedge dy\wedge dz$.
\item[$(vi)$] ($\Rightarrow$) If $\nabla\cdot\mathbf{F}=0$, then $d\omega_{\mathbf{F}}=0$ by (v).
\item[$(vi)$] ($\Leftarrow$) If $d\omega_{\mathbf{F}}=0$, then its $dx\wedge dy\wedge dz$ coefficient is $0$. Hence $\nabla\cdot\mathbf{F}=0$ by (v).
\item[$(vii)$] If $\nabla\times\mathbf{F}=0$, then $d\lambda_{\mathbf{F}}=0$ by (iii). Choose $u\in C^1(E)$ satisfying $\lambda_{\mathbf{F}}=du$ by Theorem \ref{thm:poincare-lemma-convex-rudin}(ii). Then $\mathbf{F}=\nabla u$ by (i).
\item[$(viii)$] If $\nabla\cdot\mathbf{F}=0$, then $d\omega_{\mathbf{F}}=0$ by (vi). Choose a $C^1$ $1$-form $\eta=g_1\,dx+g_2\,dy+g_3\,dz$ satisfying $\omega_{\mathbf{F}}=d\eta$ by Theorem \ref{thm:poincare-lemma-convex-rudin}(ii).\\
Set $\mathbf{G}\defeq g_1\mathbf{e}_1+g_2\mathbf{e}_2+g_3\mathbf{e}_3$. Then $\eta=\lambda_{\mathbf{G}}$ and $\mathbf{F}=\nabla\times\mathbf{G}$ by (iv).
\end{enumerate}
\hlt{Structural Commentary}: Coefficient expansion realizes $d$ as curl and divergence; the Poincar\'e lemma supplies potentials.\\
\hlt{Key Insight}: Closed $1$- and $2$-forms correspond respectively to curl-free and divergence-free fields.
\end{proof}
\end{theorem}

\begin{definition} \label{def:vector-calculus-integral-notation} \hlt{Vector-Calculus Integral Notation}\\
In $\R^2$, write $(x,y)\defeq(x_1,x_2)$ and $(dx,dy)\defeq(dx_1,dx_2)$. In $\R^3$, use $\lambda_{\mathbf{F}}$ and $\omega_{\mathbf{F}}$ from Definition \ref{def:vector-fields-as-differential-forms}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $V\subseteq\R^3$ be open, let $\Gamma$ be a $C^2$ $1$-chain in $V$, and let $\mathbf{F}\in C^1(V,\R^3)$. Set $\int_\Gamma\mathbf{F}\cdot\mathbf{t}\,ds\defeq\int_\Gamma\lambda_{\mathbf{F}}$.
\item Let $V\subseteq\R^3$ be open, let $\Sigma$ be a $C^2$ $2$-chain in $V$, and let $\mathbf{F}\in C^1(V,\R^3)$. Set $\int_\Sigma\mathbf{F}\cdot\mathbf{n}\,dA\defeq\int_\Sigma\omega_{\mathbf{F}}$.
\item Let $V\subseteq\R^2$ be open, let $\Sigma$ be a $C^2$ $2$-chain in $V$, and let $g\in C(V)$. Set $\int_\Sigma g\,dA\defeq\int_\Sigma g\,dx\wedge dy$.
\item Let $V\subseteq\R^3$ be open, let $\Omega$ be a $C^2$ $3$-chain in $V$, and let $g\in C(V)$. Set $\int_\Omega g\,dV\defeq\int_\Omega g\,dx\wedge dy\wedge dz$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:green-stokes-divergence-special-cases} \hlt{Green, Classical Stokes, and Divergence}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $V\subseteq\R^2$ be open, let $\Sigma$ be a $C^2$ $2$-chain in $V$, and let $\alpha,\beta\in C^1(V)$. Then $\int_{\partial_{\mathrm{ch}}\Sigma}(\alpha\,dx+\beta\,dy)=\int_{\Sigma}(D_1\beta-D_2\alpha)\,dA$.
\item Let $V\subseteq\R^3$ be open, let $\Sigma$ be a $C^2$ $2$-chain in $V$, and let $\mathbf{F}\in C^1(V,\R^3)$. Then $\int_{\partial_{\mathrm{ch}}\Sigma}\mathbf{F}\cdot\mathbf{t}\,ds=\int_{\Sigma}(\nabla\times\mathbf{F})\cdot\mathbf{n}\,dA$.
\item Let $V\subseteq\R^3$ be open, let $\Omega$ be a $C^2$ $3$-chain in $V$, and let $\mathbf{F}\in C^1(V,\R^3)$. Then $\int_{\partial_{\mathrm{ch}}\Omega}\mathbf{F}\cdot\mathbf{n}\,dA=\int_{\Omega}(\nabla\cdot\mathbf{F})\,dV$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $\lambda\defeq\alpha\,dx+\beta\,dy$. Then $d\lambda=(D_1\beta-D_2\alpha)\,dx\wedge dy$. Stated identity holds by Theorem \ref{thm:stokes-theorem-differential-forms} and Definition \ref{def:vector-calculus-integral-notation}(iii).
\item Set $\lambda\defeq\lambda_{\mathbf{F}}$; $d\lambda=\omega_{\nabla\times\mathbf{F}}$ holds by Theorem \ref{thm:vector-analysis-forms-dictionary}(ii). Stated identity holds by Theorem \ref{thm:stokes-theorem-differential-forms}, Definition \ref{def:vector-calculus-integral-notation}(i),(ii).
\item Set $\omega\defeq\omega_{\mathbf{F}}$; $d\omega=(\nabla\cdot\mathbf{F})\,dx\wedge dy\wedge dz$ holds by Theorem \ref{thm:vector-analysis-forms-dictionary}(v). Stated identity holds by Theorem \ref{thm:stokes-theorem-differential-forms}, Definition \ref{def:vector-calculus-integral-notation}(ii),(iv).
\end{enumerate}
\hlt{Structural Commentary}: Stokes transfers the three exterior-derivative identities to chain integrals.\\
\hlt{Key Insight}: The same Stokes identity becomes circulation, flux, or volume balance in degrees $1,2,3$.
\end{proof}
\end{theorem}
