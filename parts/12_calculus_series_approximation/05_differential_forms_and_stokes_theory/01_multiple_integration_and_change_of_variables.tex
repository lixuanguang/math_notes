\subsection{Multiple Integration and Change of Variables}

% **Role:** calculus as local linearization plus global geometric integration.
%
% Alternating forms, pullbacks, orientation, manifolds, integration on manifolds.

This subsection develops integration on cells, compactly supported integrals, and the determinant factor in change of variables.\\
These constructions support the differential-form and manifold integrals that follow.

\begin{definition} \label{def:k-cell-and-multiple-integral} \hlt{$k$-Cells and Multiple Integrals}\\
Fix $k\in\Z_{\geq1}$ and real endpoints satisfying $(\forall\,i\in J_k)\,a_i\leq b_i$. Use the convention $\int_a^a h=0$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $I^j\defeq\prod\limits_{i=1}^{j}[a_i,b_i]\subseteq\R^j$ for $0\leq j\leq k$, with $I^0$ a point. The \hlt{$k$-cell} is $I^k$.
\item For $f\in C(I^k)$, set $f_k\defeq f$ and $f_{j-1}(x_1,\ldots,x_{j-1})\defeq\int_{a_j}^{b_j}f_j(x_1,\ldots,x_j)\,dx_j$, $j=k,\ldots,1$.\\
The \hlt{multiple integral} is $\int_{I^k}f(\x)\,d\x\defeq f_0$, also written $\int_{I^k}f$.
\end{enumerate}
\end{definition}

For bounded functions on rectangles, the Darboux envelopes from Definition \ref{def:partition-and-riemann-upper-lower-sums} generalize to grid rectangles.

\begin{definition} \label{def:rectangular-grid-riemann-sum} \hlt{Rectangular Grids and Riemann Sums}\\
Fix $m,n\in\N$. Let $R=[a,b]\times[c,d]\subseteq\R^2$, $f:R\to\R$ be bounded,  $P:a=x_0<\cdots<x_m=b$, $Q:c=y_0<\cdots<y_n=d$, $G\defeq P\times Q$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{grid rectangles} are $R_{ij}\defeq[x_{i-1},x_i]\times[y_{j-1},y_j]$, with $\abs{R_{ij}}\defeq(x_i-x_{i-1})(y_j-y_{j-1})$.
\item A \hlt{tagging} is $S=\bigl((s_{ij},t_{ij})\bigr)_{(i,j)\in J_m\times J_n}$ with $(\forall\,(i,j)\in J_m\times J_n)\,(s_{ij},t_{ij})\in R_{ij}$. \hlt{Riemann sum} is $R(f,G,S)\defeq\sum\limits_{i=1}^{m}\sum\limits_{j=1}^{n}f(s_{ij},t_{ij})\abs{R_{ij}}$.
\item Set $(m_{ij},M_{ij})\defeq(\inf\limits_{R_{ij}}f,\sup\limits_{R_{ij}}f)$. Set $L(f,G)\defeq\sum\limits_{i=1}^{m}\sum\limits_{j=1}^{n}m_{ij}\abs{R_{ij}}$ and $U(f,G)\defeq\sum\limits_{i=1}^{m}\sum\limits_{j=1}^{n}M_{ij}\abs{R_{ij}}$.
\item The \hlt{lower} and \hlt{upper integrals} are $\underline{\int_R}f\defeq\sup\limits_G L(f,G)$ and $\overline{\int_R}f\defeq\inf\limits_G U(f,G)$.\\
The extrema range over rectangular grids. If they agree, $f$ is Riemann integrable on $R$, and $\int_R f$ denotes the common value.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:fubini-riemann-rectangle-slices} \hlt{Fubini on a Rectangle, Darboux Form}\\
Let $a,b,c,d\in\R$ satisfy $a<b$ and $c<d$, $R\defeq[a,b]\times[c,d]$, $f:R\to\R$ be Riemann integrable.\\
For $y\in[c,d]$, set $f_y(x)\defeq f(x,y)$, $\underline F(y)\defeq\sup\limits_P L(P,f_y)$, $\overline F(y)\defeq\inf\limits_P U(P,f_y)$. Both extrema range over partitions $P$ of $[a,b]$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\underline F,\overline F\in\mathcal R[c,d]$.
\item $\int_R f=\int_c^d\underline F(y)\,dy=\int_c^d\overline F(y)\,dy$.
\item If $(\forall\,y\in[c,d])\,f_y\in\mathcal R[a,b]$, define $g:[c,d]\to\R$ by $g(y)\defeq\int_a^b f(x,y)\,dx$. Then $g\in\mathcal R[c,d]$ and $\int_R f=\int_c^d g(y)\,dy$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $\mu,\nu\in\R$, $(\forall\,(x,y)\in R)\,\mu\leq f(x,y)\leq\nu$. Then $(\forall\,y\in[c,d])\,\mu(b-a)\leq\underline F(y)\leq\overline F(y)\leq\nu(b-a)$ by Proposition \ref{prop:riemann-darboux-and-tagged-sum-bounds}(iii).\\
Fix $G=P\times Q$, with $P:a=x_0<\cdots<x_m=b$ and $Q:c=y_0<\cdots<y_n=d$.\\
Set $\Delta x_i\defeq x_i-x_{i-1}$, $\Delta y_j\defeq y_j-y_{j-1}$, and $R_{ij}\defeq[x_{i-1},x_i]\times[y_{j-1},y_j]$. Set $m_{ij}\defeq\inf\limits_{R_{ij}}f$ and $M_{ij}\defeq\sup\limits_{R_{ij}}f$.\\
For $j\in J_n$, set $\ell_j\defeq\sum\limits_{i=1}^{m}m_{ij}\Delta x_i$ and $u_j\defeq\sum\limits_{i=1}^{m}M_{ij}\Delta x_i$.\\
If $y\in[y_{j-1},y_j]$, then $\ell_j\leq L(P,f_y)\leq\underline F(y)\leq\overline F(y)\leq U(P,f_y)\leq u_j$ by the Darboux definitions.\\
Multiplication by $\Delta y_j$ and summation over $j$ establish $L(f,G)\leq L(Q,\underline F)\leq U(Q,\overline F)\leq U(f,G)$.\\
Fix $\varepsilon>0$ and set $I\defeq\int_R f$. Choose grids $G_L=P_L\times Q_L$ and $G_U=P_U\times Q_U$ such that $I-L(f,G_L)<\frac{\varepsilon}{2}$ and $U(f,G_U)-I<\frac{\varepsilon}{2}$.\\
Set $P_\ast\defeq P_L\cup P_U$, $Q_\ast\defeq Q_L\cup Q_U$, and $G_\ast\defeq P_\ast\times Q_\ast$. Then $U(f,G_\ast)-L(f,G_\ast)<\varepsilon$ by cellwise refinement monotonicity.\\
Set $D_-\defeq U(Q_\ast,\underline F)-L(Q_\ast,\underline F)$ and $D_+\defeq U(Q_\ast,\overline F)-L(Q_\ast,\overline F)$. Then $0\leq\max\{D_-,D_+\}\leq U(Q_\ast,\overline F)-L(Q_\ast,\underline F)<\varepsilon$ because $\underline F\leq\overline F$.\\
Thus $\underline F,\overline F\in\mathcal R[c,d]$ by Theorem \ref{thm:riemann-stieltjes-integrability-criterion} with $\alpha(t)=t$.
\item Set $I_-\defeq\int_c^d\underline F(y)\,dy$ and $I_+\defeq\int_c^d\overline F(y)\,dy$.\\
For $G=P\times Q$, $L(f,G)\leq I_-\leq I_+\leq U(f,G)$ by Part (i), the Darboux definitions, and Theorem \ref{thm:riemann-stieltjes-integral-properties}(ii) with $\alpha(t)=t$.\\
Hence $\int_R f\leq I_-\leq I_+\leq\int_R f$ by Definition \ref{def:rectangular-grid-riemann-sum}, proving the identities.
\item Under the slice hypothesis, $(\forall\,y\in[c,d])\,g(y)=\underline F(y)=\overline F(y)$ by Definition \ref{def:partition-and-riemann-upper-lower-sums}. Thus $g\in\mathcal R[c,d]$ and $\int_R f=\int_c^d g(y)\,dy$ by Parts (i)--(ii).
\end{enumerate}
\hlt{Structural Commentary}: A common product refinement transfers the rectangular Darboux gap to both slice envelopes.\\
\hlt{Key Insight}: One cross-gap controls integrability and both integral values.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:iterated-integrals-order-independent} \hlt{Order Independence on Cells}\\
Fix $k\in\Z_{\geq1}$, a $k$-cell $I^k$ as in Definition \ref{def:k-cell-and-multiple-integral}, and $f\in C(I^k)$. For $\pi\in\mathfrak S_k$, let $I_\pi(f)$ denote successive integration in $x_{\pi(k)},\ldots,x_{\pi(1)}$.\\
Then $(\forall\,\pi\in\mathfrak S_k)\,I_\pi(f)$ exists, and $(\forall\,\pi\in\mathfrak S_k)\,I_\pi(f)=\int_{I^k}f$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Existence and continuity}: Let $K$ be a cell, $u\leq v$, $h\in C(K\times[u,v])$, $H(\z)\defeq\int_u^v h(\z,t)\,dt$.\\
If $u=v$, then $H=0$ by Definition \ref{def:k-cell-and-multiple-integral}; assume $u<v$. The integral exists by Theorem \ref{thm:continuous-functions-riemann-stieltjes-integrable} with $\alpha(t)=t$.\\
The domain is compact by Theorem \ref{thm:heine-borel-rk}, so $h$ is uniformly continuous by Theorem \ref{thm:compact-continuity-uniform-continuity}.\\
For $\z,\z'\in K$, $\abs{H(\z)-H(\z')}\leq(v-u)\sup\limits_{t\in[u,v]}\abs{h(\z,t)-h(\z',t)}$ by Theorem \ref{thm:riemann-stieltjes-integral-properties}(i),(iv) with $\alpha(t)=t$.\\
Thus $H\in C(K)$. Repetition proves existence of every $I_\pi(f)$ and continuity of every intermediate integrand.
\item \hlt{Adjacent interchange}: Fix consecutive coordinates $x_p,x_q$ in an integration order and hold the remaining variables fixed.\\
If $a_p=b_p$ or $a_q=b_q$, both orders are zero by Definition \ref{def:k-cell-and-multiple-integral}.\\
Otherwise, set $S\defeq[a_p,b_p]\times[a_q,b_q]$ and let $h\in C(S)$. Then $h$ is uniformly continuous by Theorems \ref{thm:heine-borel-rk} and \ref{thm:compact-continuity-uniform-continuity}.\\
Set $A\defeq(b_p-a_p)(b_q-a_q)>0$. For $\varepsilon>0$, choose a grid $G$ whose cell oscillations are below $\varepsilon/A$.\\
Then $U(h,G)-L(h,G)<\varepsilon$, so the rectangular integral gap is zero by Definition \ref{def:rectangular-grid-riemann-sum}. The relevant two-variable slice is continuous by Part (i), and its one-variable slices are integrable by Theorem \ref{thm:continuous-functions-riemann-stieltjes-integrable} with $\alpha(t)=t$.\\
Transposed grids preserve Darboux sums. Adjacent integrals commute by Theorem \ref{thm:fubini-riemann-rectangle-slices}(iii) in both coordinate orientations.
\item \hlt{Permutation reduction}: Fix $\pi\in\mathfrak S_k$. Moving each index to its identity position factors $\pi$ into adjacent transpositions.\\
Thus $I_\pi(f)=I_{\mathrm{id}}(f)$ by Part (ii), while $I_{\mathrm{id}}(f)=\int_{I^k}f$ by Definition \ref{def:k-cell-and-multiple-integral}.
\end{enumerate}
\hlt{Structural Commentary}: Partial integration preserves continuity; rectangular Darboux control licenses each adjacent swap.\\
\hlt{Key Insight}: Adjacent transpositions connect every order to the defining order.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:fubini-hypotheses-control-order} \hlt{Order Interchange Requires Hypotheses}\\
Finite sums are order-free; limiting processes are not.\\
Theorems \ref{thm:fubini-riemann-rectangle-slices} and \ref{thm:iterated-integrals-order-independent} state hypotheses for interchanging integral order.
\end{perspective}

\begin{definition} \label{def:support-and-compact-support-integral} \hlt{Support and Simplex Integrals}\\
Fix $k\in\Z_{\geq1}$ and $\mathbb F\in\{\R,\C\}$. Integrals of complex-valued functions are taken componentwise.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Compact support}: For $f:\R^k\to\mathbb F$, set $\operatorname{supp}f\defeq\overline{\{\x\in\R^k:f(\x)\neq0\}}$.\\
Set $C_c(\R^k,\mathbb F)\defeq\{f\in C(\R^k,\mathbb F):\operatorname{supp}f\text{ is compact}\}$. For $f\in C_c(\R^k,\mathbb F)$ and a $k$-cell $I^k\supseteq\operatorname{supp}f$, set $\int_{\R^k}f\defeq\int_{I^k}f$.
\item \hlt{Standard simplex}: Set $Q^0$ to be a point and $Q^j\defeq\{\x\in[0,+\infty)^j:\sum\limits_{i=1}^{j}x_i\leq1\}$ for $j\in J_k$.\\
The \hlt{standard $k$-simplex} is $Q^k$. For $j\in J_k$ and $\z\in Q^{j-1}$, set $u_j(\z)\defeq1-\sum\limits_{i=1}^{j-1}z_i$, with the empty sum equal to $0$.\\
For $f\in C(Q^k,\mathbb F)$, set $f_k\defeq f$ and recursively set $f_{j-1}(\z)\defeq\int_0^{u_j(\z)}f_j(\z,t)\,dt$ for $j=k,\ldots,1$.\\
The \hlt{simplex integral} is $\int_{Q^k}f\defeq f_0$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:support-and-simplex-integrals-well-defined} \hlt{Well-Defined Support and Simplex Integrals}\\
The constructions in Definition \ref{def:support-and-compact-support-integral} satisfy:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $f\in C_c(\R^k,\mathbb F)$, set $\mathscr C_f\defeq\{I\subseteq\R^k:I\text{ is a $k$-cell and }\operatorname{supp}f\subseteq I\}$. Then $\mathscr C_f\neq\emptyset$ and $(\forall\,I,J\in\mathscr C_f)\,\int_I f=\int_J f$.
\item For $f\in C(Q^k,\mathbb F)$, every recursive integral exists and $(\forall\,j\in\{0,\ldots,k\})\,f_j\in C(Q^j,\mathbb F)$.
\end{enumerate}
\begin{proof}
It suffices to treat $\mathbb F=\R$; the complex case holds componentwise.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The set $K\defeq\operatorname{supp}f$ is bounded by Theorem \ref{thm:heine-borel-rk}, so a containing $k$-cell exists by Lemma \ref{lem:bounded-rk-set-contained-in-k-cell}.\\
Let $I,J\in\mathscr C_f$, and choose a cell $L\supseteq I\cup J$ by the same lemma. All three cell integrals exist by Theorem \ref{thm:iterated-integrals-order-independent}.\\
Since $f=0$ on $\R^k\setminus K$, $\int_I f=\int_L f=\int_J f$ by repeated interval additivity in Theorem \ref{thm:riemann-stieltjes-integral-properties}(iii) with $\alpha(t)=t$.
\item Proceed by downward induction from $f_k=f\in C(Q^k)$. Fix $j\in J_k$ with $f_j\in C(Q^j)$.\\
For $(\z,s)\in Q^{j-1}\times[0,1]$, set $\Phi_j(\z,s)\defeq(\z,u_j(\z)s)$ and $q_j(\z,s)\defeq u_j(\z)f_j(\Phi_j(\z,s))$.\\
For $(\z,s)\in Q^{j-1}\times[0,1]$, $\sum\limits_{i=1}^{j-1}z_i+u_j(\z)s\leq\sum\limits_{i=1}^{j-1}z_i+u_j(\z)=1$.\\
Thus $\Phi_j(Q^{j-1}\times[0,1])\subseteq Q^j$ by Definition \ref{def:support-and-compact-support-integral}(ii), and $q_j$ is continuous by Theorems \ref{thm:continuous-composition} and \ref{thm:continuous-algebra-and-components}.\\
The domain of $q_j$ is closed and bounded by Definition \ref{def:support-and-compact-support-integral}(ii), hence compact by Theorem \ref{thm:heine-borel-rk}.\\
Thus $q_j$ is uniformly continuous by Theorem \ref{thm:compact-continuity-uniform-continuity}. The slice integrals exist by Theorem \ref{thm:continuous-functions-riemann-stieltjes-integrable} with the identity integrator.\\
For $\z\in Q^{j-1}$ with $u_j(\z)>0$, $f_{j-1}(\z)=\int_0^1q_j(\z,s)\,ds$ by Theorem \ref{thm:riemann-stieltjes-change-of-variable}(v); both sides vanish when $u_j(\z)=0$.\\
For $\z,\z'\in Q^{j-1}$, $\abs{f_{j-1}(\z)-f_{j-1}(\z')}\leq\sup\limits_{s\in[0,1]}\abs{q_j(\z,s)-q_j(\z',s)}$ by Theorem \ref{thm:riemann-stieltjes-integral-properties}(i),(iv) with $\alpha(s)=s$.\\
Uniform continuity of $q_j$ implies $f_{j-1}\in C(Q^{j-1})$, completing the induction.
\end{enumerate}
\hlt{Structural Commentary}: Cell enlargement proves independence; fiber rescaling transfers continuity through the simplex recursion.\\
\hlt{Key Insight}: Rescaling converts every simplex fiber to $[0,1]$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:linear-map-volume-scaling-determinant} \hlt{Linear Determinant Change of Variables}\\
Fix $n\in\N$. Let $T\in\LL(\R^n)$ be invertible, $f\in C_c(\R^n,\R)$. Then $\int_{\R^n}f(\y)\,d\y=\abs{\det T}\int_{\R^n}f(T\x)\,d\x$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Factorization}: Set $A\defeq[T]$. Choose elementary matrices $E_1,\ldots,E_q$ with $I=E_q\cdots E_1A$ by Lemma \ref{lem:invertible-matrices-row-reduce-to-identity}(ii).\\
Set $F_r\defeq E_r^{-1}$, $S_r$ be linear map represented by $F_r$. Then $A=F_1\cdots F_q$, and every $F_r$ is elementary by Proposition \ref{prop:elementary-matrices-perform-row-operations}(ii).\\
For any invertible $S\in\LL(\R^n)$ and $g\in C_c(\R^n,\R)$, $S$ is a homeomorphism by Corollary \ref{cor:finite-dimensional-linear-maps-continuous}.\\
Thus $\operatorname{supp}(g\circ S)=S^{-1}(\operatorname{supp}g)$, which is compact by Theorem \ref{thm:continuous-image-compact}; hence $g\circ S\in C_c(\R^n,\R)$ by Theorem \ref{thm:continuous-composition}.
\item \hlt{Elementary Factors}: Fix $g\in C_c(\R^n,\R)$, and choose $M>0$ such that $\operatorname{supp}g\subseteq C\defeq[-M,M]^n$ by Theorem \ref{thm:heine-borel-rk}.\\
Let $S$ be an invertible elementary linear map. We prove $\int_{\R^n}g(S\x)\,d\x=\abs{\det S}^{-1}\int_{\R^n}g(\y)\,d\y$ by cases.\\
It suffices to compare cell integrals on cells containing the two supports by Proposition \ref{prop:support-and-simplex-integrals-well-defined}(i).
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item Suppose $S$ scales coordinate $i$ by $c\neq0$, $D_c\defeq\{\x:\abs{x_i}\leq M/\abs c,\ \abs{x_k}\leq M\ (k\neq i)\}$. Then $D_c\supseteq\operatorname{supp}(g\circ S)$.\\
Integrate in $x_i$ first by Theorem \ref{thm:iterated-integrals-order-independent}. Positive scaling satisfies the claimed identity by Theorem \ref{thm:riemann-stieltjes-change-of-variable}(v).\\
For $c<0$, reflected partitions preserve the lower and upper sums, so reflection preserves the integral.\\
Positive scaling by $\abs c$ then proves the same identity. Finally, $\abs{\det S}=\abs c$ by Theorem \ref{thm:determinant-basic-laws}(v).
\item Suppose $S$ swaps coordinates $i$ and $j$. Then $S(C)=C$, so $C$ contains both supports.\\
Renaming the two dummy variables converts the iterated integral of $g\circ S$ into order obtained by swapping $i$ and $j$ for $g$.\\
The integrals are equal by Theorem \ref{thm:iterated-integrals-order-independent}. Moreover, $\abs{\det S}=1$ by Theorem \ref{thm:determinant-basic-laws}(i),(iii),(ix).
\item Suppose $S\x=\x+cx_j\mathbf e_i$ with $i\neq j$, and set $D\defeq\{\x:\abs{x_i}\leq(1+\abs c)M,\ \abs{x_k}\leq M\ (k\neq i)\}$.\\
Then $D\supseteq\operatorname{supp}(g\circ S)$. Integrate in $x_i$ first by Theorem \ref{thm:iterated-integrals-order-independent}.\\
For fixed remaining coordinates, the translated $x_i$-interval contains $[-M,M]$, outside which the slice of $g$ vanishes. Its integral therefore equals the integral over $[-M,M]$ by Theorem \ref{thm:riemann-stieltjes-change-of-variable}(v) and interval additivity.\\
Thus the two cell integrals are equal, and $\det S=1$ by Theorem \ref{thm:determinant-basic-laws}(v).
\end{enumerate}
\item \hlt{Composition}: If $q=0$, then $T=I$ and the claim is immediate.\\
Suppose $q\geq1$, and set $g_0\defeq f$ and $g_r\defeq f\circ S_1\circ\cdots\circ S_r$ for $1\leq r\leq q$.\\
Every $g_r$ has compact support by Part (i). For $1\leq r\leq q$, $\int_{\R^n}g_r(\x)\,d\x=\abs{\det S_r}^{-1}\int_{\R^n}g_{r-1}(\y)\,d\y$ by Part (ii).\\
Since $g_q=f\circ T$, $\int_{\R^n}f(T\x)\,d\x=\abs{\det T}^{-1}\int_{\R^n}f(\y)\,d\y$ by iteration, Theorem \ref{thm:determinant-basic-laws}(vi). Rearranging proves claim.
\end{enumerate}
\hlt{Structural Commentary}: Elementary reduction isolates one-coordinate substitutions; compact support makes identities composable.\\
\hlt{Key Insight}: Each elementary factor contributes exactly its reciprocal determinant modulus.
\end{proof}
\end{theorem}

\begin{definition} \label{def:primitive-mapping-and-flip} \hlt{Primitive Mappings and Flips}\\
Fix $n\in\N$ and let $E\subseteq\R^n$ be open.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $m\in J_n$, a map $G:E\to\R^n$ is \hlt{primitive in coordinate $m$} if $(\exists\,g:E\to\R)(\forall\,\x\in E)\,G(\x)=\x+(g(\x)-x_m)\mathbf e_m$.
\item For distinct $p,q\in J_n$, let $\tau=(p\ q)\in\mathfrak S_n$. A \hlt{flip} is the unique $B\in\LL(\R^n)$ satisfying $(\forall\,i\in J_n)\,B\mathbf e_i=\mathbf e_{\tau(i)}$.\\
Existence and uniqueness hold by Proposition \ref{prop:linear-map-determined-by-basis}.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:primitive-mapping-derivative-jacobian} \hlt{Derivative and Jacobian of a Primitive Map}\\
Fix $n\in\N$, an open set $E\subseteq\R^n$, $m\in J_n$, and $\a\in E$. Let $g:E\to\R$ be differentiable at $\a$.\\
Define $G:E\to\R^n$ by $(\forall\,\x\in E)\,G(\x)=\x+(g(\x)-x_m)\mathbf e_m$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The map $G$ is differentiable at $\a$, and $[DG(\a)]$ is $I_n$ with row $m$ replaced by $\nabla g(\a)^t$.
\item $J_G(\a)=(D_mg)(\a)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By the definition of $G$, $G_m=g$ and $(\forall\,k\in J_n\setminus\{m\})(\forall\,\x\in E)\,G_k(\x)=x_k$.\\
Each map $\x\mapsto x_k$ differentiable at $\a$ with derivative $\h\mapsto h_k$ by Definition \ref{def:total-derivative-rn-rm}; $G$ differentiable at $\a$ by Theorem \ref{thm:several-variable-componentwise-differentiability}(i).\\
The rows indexed by $k\neq m$ are $\mathbf e_k^t$, and row $m$ is $\nabla g(\a)^t$, by Theorem \ref{thm:total-derivative-matrix-partials}(i).
\item By Part (i), a nonzero term in the determinant sum must satisfy $(\forall\,k\in J_n\setminus\{m\})\,\sigma(k)=k$, hence $\sigma=\operatorname{id}$.\\
Its product is $(D_mg)(\a)$, so $J_G(\a)=(D_mg)(\a)$ by Definitions \ref{def:matrix-trace-and-determinant} and \ref{def:jacobian-determinant}.
\end{enumerate}
\hlt{Structural Commentary}: One derivative row changes; fixed identity rows eliminate all nonidentity determinant terms.\\
\hlt{Key Insight}: A primitive map's Jacobian is the partial derivative in its active coordinate.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:local-decomposition-primitive-mappings} \hlt{Local Decomposition into Primitive Mappings}\\
Fix $n\in\N$, an open set $E\subseteq\R^n$, and $\F\in C^1(E,\R^n)$. Assume $\ze\in E$, $\F(\ze)=\ze$, and $D\F(\ze)$ is invertible.\\
Then families $(V_i)_{i\in J_n}$, $(G_i)_{i\in J_n}$, $(B_i)_{i\in J_{n-1}}$, and a set $U$ exist such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $i\in J_n$, $V_i\subseteq\R^n$ is open, $\ze\in V_i$, and $G_i\in C^1(V_i,\R^n)$ is primitive in coordinate $i$.\\
The identities $G_i(\ze)=\ze$ and $J_{G_i}(\ze)\neq0$ hold. For $i\in J_{n-1}$, $B_i\in\LL(\R^n)$ is a flip or $I_n$.
\item Set $U\subseteq E$ is open, $\ze\in U$. On $U$, $B_1\circ\cdots\circ B_{n-1}\circ G_n\circ\cdots\circ G_1$ is defined, equals $\F$. $B$-prefix omitted if $n=1$.
\end{enumerate}
\begin{proof}
For $0\leq r\leq n$, set $P_r\x\defeq\sum\limits_{i=1}^{r}x_i\mathbf e_i$. Set $W_1\defeq E$ and $\F_1\defeq\F$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The base relations $P_0\circ\F_1=P_0$, $\F_1(\ze)=\ze$, and invertibility of $D\F_1(\ze)$ hold by the hypotheses.\\
Suppose $1\leq m<n$, $W_m\ni\ze$ is open, $\F_m\in C^1(W_m,\R^n)$, $P_{m-1}\circ\F_m=P_{m-1}$, $\F_m(\ze)=\ze$, and $D\F_m(\ze)$ is invertible.\\
Write $\F_m=P_{m-1}+\sum\limits_{i=m}^{n}\alpha_i\mathbf e_i$ on $W_m$, where $\alpha_m,\ldots,\alpha_n\in C^1(W_m)$.\\
If $(\forall\,k\in\{m,\ldots,n\})\,(D_m\alpha_k)(\ze)=0$, then $D\F_m(\ze)\mathbf e_m=\sum\limits_{k=m}^{n}(D_m\alpha_k)(\ze)\mathbf e_k=\ze$, a contradiction.\\
Choose $k\in\{m,\ldots,n\}$ with $(D_m\alpha_k)(\ze)\neq0$. Let $B_m=I_n$ if $k=m$; otherwise let $B_m$ be the flip of coordinates $m,k$.\\
Define $G_m(\x)\defeq\x+(\alpha_k(\x)-x_m)\mathbf e_m$ on $W_m$. Then $G_m$ is primitive in coordinate $m$, belongs to $C^1(W_m,\R^n)$, and fixes $\ze$.\\
The identity $J_{G_m}(\ze)=(D_m\alpha_k)(\ze)\neq0$ holds by Proposition \ref{prop:primitive-mapping-derivative-jacobian}(ii), so $DG_m(\ze)$ is invertible by Corollary \ref{cor:jacobian-nonzero-iff-derivative-invertible}.\\
Choose open neighborhoods $V_m\subseteq W_m$, $W_{m+1}$ of $\ze$ s.t. $G_m|_{V_m}:V_m\to W_{m+1}$ is $C^1$ diffeomorphism by Theorem \ref{thm:inverse-function-theorem-rn}(i),(iii).\\
Retain this restriction as $G_m$, and define $\F_{m+1}\defeq B_m\circ\F_m\circ G_m^{-1}$ on $W_{m+1}$.\\
The map $\F_{m+1}$ is $C^1$ and fixes $\ze$ by Theorem \ref{thm:inverse-function-theorem-rn}(iii) and the definitions.\\
For $\y\in W_{m+1}$, set $\x\defeq G_m^{-1}(\y)$. Then $y_i=x_i$ for $i<m$ and $y_m=\alpha_k(\x)$ by the definition of $G_m$.\\
The first $m$ components of $B_m\F_m(\x)$ are $y_1,\ldots,y_m$ by the induction hypothesis and the definition of $B_m$; hence $P_m\circ\F_{m+1}=P_m$.\\
Since $DB_m(\ze)=B_m$, $D\F_{m+1}(\ze)=B_mD\F_m(\ze)(DG_m(\ze))^{-1}$ by Definition \ref{def:total-derivative-rn-rm} and Theorems \ref{thm:chain-rule-several-variables}(ii), \ref{thm:inverse-function-theorem-rn}(ii).\\
Every factor is invertible, so $D\F_{m+1}(\ze)$ is invertible. This completes the induction for $m=1,\ldots,n-1$.\\
Set $V_n\defeq W_n$ and $G_n\defeq\F_n$. Since $P_{n-1}\circ\F_n=P_{n-1}$, $G_n$ is primitive in coordinate $n$.\\
The identities $G_n(\ze)=\ze$ and $J_{G_n}(\ze)\neq0$ hold by the induction invariant and Corollary \ref{cor:jacobian-nonzero-iff-derivative-invertible}. Hence (i) holds.
\item For $m\in J_{n-1}$, $B_m^2=I_n$ and the recurrence imply $\F_m=B_m\circ\F_{m+1}\circ G_m$ on $V_m$. By continuity and finite repeated shrinking, choose open neighborhood $U\subseteq E$ of $\ze$ on which every partial composition $G_r\circ\cdots\circ G_1$ is defined for $r\in J_n$.\\
$\F=B_1\circ\cdots\circ B_{n-1}\circ G_n\circ\cdots\circ G_1$ holds on $U$ by successive substitution, $G_n=\F_n$; for $n=1$, it reduces to $\F=G_1$.
\end{enumerate}
\hlt{Structural Commentary}: Each primitive inverse fixes the next coordinate; a flip moves its nonzero derivative entry into that position.\\
\hlt{Key Insight}: An invertible derivative supplies a pivot at every inductive stage.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:partition-of-unity-rudin} \hlt{Partition of Unity on a Compact Set}\\
Fix $n\in\N$. Let $K\subseteq\R^n$ be nonempty and compact, $(V_\alpha)_{\alpha\in\mathcal A}$ be an open cover of $K$, $\mathcal C_K\defeq\{f\in C(\R^n):\operatorname{supp}(f)\subseteq K\}$.\\
Then $s\in\N$ and $\psi_1,\ldots,\psi_s\in C^1(\R^n)$ exist such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,i\in J_s)\,0\leq\psi_i\leq1$;
\item $(\forall\,i\in J_s)(\exists\,\alpha\in\mathcal A)\,\operatorname{supp}(\psi_i)\subseteq V_\alpha$;
\item $\sum\limits_{i=1}^s\psi_i=1$ on $K$.
\item $(\forall\,f\in\mathcal C_K)\,f=\sum\limits_{i=1}^{s}\psi_i f$;
\item $(\forall\,(f,i)\in\mathcal C_K\times J_s)(\exists\,\alpha\in\mathcal A)\,\operatorname{supp}(\psi_i f)\subseteq V_\alpha$.
\end{enumerate}
\begin{proof}
Choose $\alpha:K\to\mathcal A$ with $(\forall\,\x\in K)\,\x\in V_{\alpha(\x)}$ by Definition \ref{def:open-cover-and-compactness}(i); $R:K\to(0,+\infty)$, $(\forall\,\x\in K)\,\overline{B}_{R(\x)}(\x)\subseteq V_{\alpha(\x)}$ by openness.\\
Set $r(\x)\defeq\frac{R(\x)}{2}$. Family $(B_{r(\x)}(\x))_{\x\in K}$ is open cover of $K$. Choose $s\in\N$, $\x_1,\ldots,\x_s\in K$, $K\subseteq\bigcup\limits_{i=1}^{s}B_{r(\x_i)}(\x_i)$ by Definition \ref{def:open-cover-and-compactness}(ii).\\
For $i\in J_s$, set $\alpha_i\defeq\alpha(\x_i)$, $r_i\defeq r(\x_i)$, and $R_i\defeq R(\x_i)$. Set $p(t)\defeq1-3t^2+2t^3$.\\
Then $p(0)=1$, $p(1)=0$, $p'(0)=p'(1)=0$, and $(\forall\,t\in[0,1])\,p'(t)=6t(t-1)\leq0$.\\
Set $\eta(t)\defeq1$ for $t\leq0$, $\eta(t)\defeq p(t)$ for $0<t<1$, and $\eta(t)\defeq0$ for $t\geq1$. Thus $\eta\in C^1(\R)$ and $0\leq\eta\leq1$.\\
For $i\in J_s$, set $q_i(\x)\defeq\frac{\norm{\x-\x_i}_2^2-r_i^2}{R_i^2-r_i^2}$ and $\varphi_i\defeq\eta\circ q_i$. Set $\psi_i\defeq\prod\limits_{j=1}^{i-1}(1-\varphi_j)\varphi_i$ for $i\in J_s$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,(i,\ell,\x)\in J_s\times J_n\times\R^n)\,(D_\ell q_i)(\x)=\frac{2(x_\ell-(\x_i)_\ell)}{R_i^2-r_i^2}$. Hence $(\forall\,i\in J_s)\,q_i\in C^1(\R^n)$ by Theorem \ref{thm:c1-iff-continuous-partials}.\\
Identity $(\forall\,(i,\ell)\in J_s\times J_n)\,D_\ell\varphi_i=(\eta'\circ q_i)D_\ell q_i$ holds by Theorem \ref{thm:chain-rule-several-variables}; $(\forall\,i\in J_s)\,\varphi_i\in C^1(\R^n)$ by Theorem \ref{thm:c1-iff-continuous-partials}.\\
Relation $(\forall\,i\in J_s)\,\psi_i\in C^1(\R^n)$ holds by Theorems \ref{thm:derivative-algebra-rules}(i)--(ii) and \ref{thm:c1-iff-continuous-partials}. Since $(\forall\,j\in J_s)\,0\leq\varphi_j\leq1$, $(\forall\,i\in J_s)\,0\leq\psi_i\leq1$.
\item For $i\in J_s$, $q_i\geq1$ on $\R^n\setminus B_{R_i}(\x_i)$, so $\varphi_i=0$ there by the definition of $\eta$.\\
Thus $\operatorname{supp}(\psi_i)\subseteq\operatorname{supp}(\varphi_i)\subseteq\overline{B}_{R_i}(\x_i)$ by Definition \ref{def:support-and-compact-support-integral}(i), formula for $\psi_i$. Inclusion $\overline{B}_{R_i}(\x_i)\subseteq V_{\alpha_i}$ holds by choice of $R$.
\item Set $P_0\defeq1$ and $P_i\defeq\prod\limits_{j=1}^{i}(1-\varphi_j)$ for $i\in J_s$. Then $P_{i-1}-P_i=\psi_i$, so $\sum\limits_{i=1}^{s}\psi_i=1-P_s$.\\
If $\x\in K$, choose $i\in J_s$ with $\x\in B_{r_i}(\x_i)$ by the finite subcover. Then $q_i(\x)<0$, $\varphi_i(\x)=1$, and $P_s(\x)=0$.
\item Let $f\in\mathcal C_K$. The identity $f=\sum\limits_{i=1}^{s}\psi_i f$ holds on $K$ by Part (iii). On $\R^n\setminus K$, $f=0$ by the definition of $\mathcal C_K$ and Definition \ref{def:support-and-compact-support-integral}(i).\\
Hence $(\forall\,i\in J_s)\,\psi_i f=0$ there, so the identity holds on $\R^n$.
\item Let $(f,i)\in\mathcal C_K\times J_s$. Then $\operatorname{supp}(\psi_i f)\subseteq\operatorname{supp}(\psi_i)$ by Definition \ref{def:support-and-compact-support-integral}(i). Thus $\operatorname{supp}(\psi_i f)\subseteq V_{\alpha_i}$ by Part (ii).
\end{enumerate}
\hlt{Structural Commentary}: Finite inner balls localize cubic cutoffs; telescoping products assemble the exact sum.\\
\hlt{Key Insight}: Coverage makes one cutoff equal $1$, annihilating the residual product.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:change-of-variables-rk} \hlt{Change of Variables in $\R^k$}\\
Fix $k\in\N$, an open set $E\subseteq\R^k$, an injective map $T\in C^1(E,\R^k)$, and $f\in C_c(\R^k,\R)$. Assume $(\forall\,\x\in E)\,J_T(\x)\neq0$, $\operatorname{supp}f\subseteq T(E)$.\\
Then $\int_{\R^k}f(\y)\,d\y=\int_{\R^k}f(T(\x))\abs{J_T(\x)}\,d\x$. On the right, the integrand on $E$ is extended by $0$ to $\R^k$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Zero extension and pullback}: Let $U\subseteq\R^k$ be open, $C\subseteq U$ be compact, and $q\in C(U)$ vanish on $U\setminus C$.\\
Define $\widetilde q=q$ on $U$ and $\widetilde q=0$ on $\R^k\setminus C$. The set $C$ is closed by Theorem \ref{thm:compact-subsets-of-metric-spaces-closed}.\\
The formulas agree on $U\setminus C$. Since $U$ and $\R^k\setminus C$ are open and cover $\R^k$, $\widetilde q\in C_c(\R^k,\R)$ and $\operatorname{supp}\widetilde q\subseteq C$.\\
Set $K\defeq\operatorname{supp}f$. If $f=0$, both integrals vanish; assume $f\neq0$, so $K\neq\emptyset$.\\
For $\x\in E$, $DT(\x)$ is invertible by Corollary \ref{cor:jacobian-nonzero-iff-derivative-invertible}; hence $T$ is open by Corollary \ref{cor:local-diffeomorphism-open-map}(i).\\
If $O\subseteq E$ is open, then $(T^{-1})^{-1}(O)=T(O)$ is open in $T(E)$, so $T^{-1}:T(E)\to E$ is continuous.\\
Set $L\defeq T^{-1}(K)$ and $\rho(\x)\defeq f(T(\x))\abs{J_T(\x)}$ on $E$. Then $L$ is compact by Theorem \ref{thm:continuous-image-compact}, $\rho\in C(E)$, and $\rho=0$ on $E\setminus L$.\\
Thus the zero extension $\widetilde\rho$ belongs to $C_c(\R^k,\R)$ by the zero-extension argument above. For $u\in C_c(\R^k,\R)$, set $\mathscr I(u)\defeq\int_{\R^k}u$.
\item \hlt{Model maps}: Let $S:U\to V$ be a $C^1$ diffeomorphism between open subsets of $\R^k$.\\
Set $\mathcal H(V)\defeq\{h\in C_c(\R^k,\R):\operatorname{supp}h\subseteq V\}$. For $h\in\mathcal H(V)$, let $R_Sh$ be the zero extension of $\x\mapsto h(S(\x))\abs{J_S(\x)}$ from $U$.\\
Since $S^{-1}(\operatorname{supp}h)$ is compact by Theorem \ref{thm:continuous-image-compact}, $R_Sh\in C_c(\R^k,\R)$ by Part (i). Set $\operatorname{CV}(S)\defeq(\forall\,h\in\mathcal H(V))\,\mathscr I(h)=\mathscr I(R_Sh)$.
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item Let $(\forall\,j\in J_k)\,a_j<b_j$, $U\defeq\prod\limits_{j=1}^{k}(a_j,b_j)$. Let $G:U\to G(U)$ be a primitive $C^1$ diffeomorphism in coordinate $m$.\\
Write $G(\x)=\x+(g(\x)-x_m)\mathbf e_m$. If $h=0$, then $\mathscr I(h)=\mathscr I(R_Gh)=0$; fix $0\neq h\in\mathcal H(G(U))$.\\
The set $M\defeq G^{-1}(\operatorname{supp}h)$ is compact by Theorem \ref{thm:continuous-image-compact}. Choose a $k$-cell $I\defeq\prod\limits_{j=1}^{k}[c_j,d_j]\subseteq U$ with $M\subseteq\operatorname{int}I$.\\
Choose $A<B$ containing the $m$th-coordinate projection of $\operatorname{supp}h$.\\
Set $I'\defeq\prod\limits_{j\in J_k\setminus\{m\}}[c_j,d_j]$, with $I'$ a point if $k=1$. Write $(\z,t)$ for insertion of $t$ in coordinate $m$.\\
For $\x\in U$, $D(G^{-1}\circ G)(\x)=D(G^{-1})(G(\x))DG(\x)=I_k$ by Theorem \ref{thm:chain-rule-several-variables}(ii).\\
Thus $\ker DG(\x)=\{\ze\}$, so $DG(\x)$ is invertible by Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences}.\\
Hence $(\forall\,\x\in U)\,(D_mg)(\x)=J_G(\x)\neq0$ by Proposition \ref{prop:primitive-mapping-derivative-jacobian}(ii) and Corollary \ref{cor:jacobian-nonzero-iff-derivative-invertible}.\\
For $\z\in I'$, $g_\z(t)\defeq g(\z,t)$ has derivative of constant sign by Theorem \ref{thm:derivatives-have-intermediate-value-property}; $g_\z$ is strictly monotone by Corollary \ref{cor:mean-value-theorem}.\\
Set $A_\z\defeq\int_A^Bh(\z,y)\,dy$ and $B_\z\defeq\int_{c_m}^{d_m}h(G(\z,t))\abs{(D_mg)(\z,t)}\,dt$.\\
Then $A_\z=B_\z$ by Theorem \ref{thm:riemann-stieltjes-change-of-variable}(v) and interval additivity, with $t\mapsto c_m+d_m-t$ when $g_\z$ decreases.\\
Under the insertion notation, $\operatorname{supp}h\subseteq I'\times[A,B]$ and $\operatorname{supp}(R_Gh)\subseteq I$.\\
Thus $\mathscr I(h)=\int_{I'}A_\z\,d\z=\int_{I'}B_\z\,d\z=\mathscr I(R_Gh)$ by Theorem \ref{thm:iterated-integrals-order-independent} and Proposition \ref{prop:support-and-simplex-integrals-well-defined}(i). Hence $\operatorname{CV}(G)$.
\item Every flip $B$ satisfies $\operatorname{CV}(B)$ by Theorems \ref{thm:linear-map-volume-scaling-determinant} and \ref{thm:determinant-basic-laws}(i),(iii),(ix). For $c\in\R$ and $m\in J_k$, set $\tau_{c,m}(\x)\defeq\x+c\mathbf e_m$.\\
Fix $h\in C_c(\R^k,\R)$. The identity $\operatorname{supp}(h\circ\tau_{c,m})=\tau_{c,m}^{-1}(\operatorname{supp}h)$ makes this support compact by Theorem \ref{thm:continuous-image-compact}.\\
Choose a cell containing both supports by Lemma \ref{lem:bounded-rk-set-contained-in-k-cell}. After integration in $x_m$ first, the two cell integrals are equal by Theorem \ref{thm:riemann-stieltjes-change-of-variable}(v) and interval additivity; the remaining coordinates are unchanged.\\
Hence $\operatorname{CV}(\tau_{c,m})$ by Theorem \ref{thm:iterated-integrals-order-independent} and Proposition \ref{prop:support-and-simplex-integrals-well-defined}(i).
\end{enumerate}
\item \hlt{Restriction and composition}: Assume $\operatorname{CV}(S)$ for a diffeomorphism $S:U\to V$, and let $U_0\subseteq U$ be open.\\
For $h\in\mathcal H(S(U_0))$, $R_Sh=R_{S|_{U_0}}h$ by injectivity of $S$; hence $\operatorname{CV}(S|_{U_0})$.\\
Let $Q:U\to V$ and $P:W\to Z$ satisfy $V\subseteq W$, $\operatorname{CV}(Q)$, and $\operatorname{CV}(P)$. Fix $h\in\mathcal H(P(V))$.\\
Set $C_q\defeq P^{-1}(\operatorname{supp}h)$. Then $C_q\subseteq V$ is compact by Theorem \ref{thm:continuous-image-compact} for $P^{-1}$.\\
On $W$, set $q(\z)\defeq h(P(\z))\abs{J_P(\z)}$, and let $q_0$ be its zero extension. Since $q=0$ on $W\setminus C_q$, $q_0\in\mathcal H(V)$ by Part (i).\\
The identity $R_Qq_0=R_{P\circ Q}h$ holds by $J_{P\circ Q}=(J_P\circ Q)J_Q$, Theorem \ref{thm:chain-rule-several-variables}(ii), and Theorem \ref{thm:determinant-basic-laws}(vi).\\
Therefore $\mathscr I(h)=\mathscr I(q_0)=\mathscr I(R_Qq_0)=\mathscr I(R_{P\circ Q}h)$ by $\operatorname{CV}(P)$ and $\operatorname{CV}(Q)$; hence $\operatorname{CV}(P\circ Q)$.\\
For $\c=(c_1,\ldots,c_k)\in\R^k$, set $\tau_\c(\x)\defeq\x+\c$. Then $\operatorname{CV}(\tau_\c)$ by Part (ii)(b) and the composition result above.
\item \hlt{Localization}: Fix $\a\in E$, set $E_\a\defeq\{\h\in\R^k:\a+\h\in E\}$, and define $\F_\a:E_\a\to\R^k$ by $\F_\a(\h)\defeq T(\a+\h)-T(\a)$.\\
Then $\F_\a(\ze)=\ze$ and $D\F_\a(\ze)=DT(\a)$ is invertible by Corollary \ref{cor:jacobian-nonzero-iff-derivative-invertible}.\\
A primitive-and-flip factorization of $\F_\a$ near $\ze$ exists by Theorem \ref{thm:local-decomposition-primitive-mappings}.\\
Each primitive factor is a $C^1$ diffeomorphism on an open box about $\ze$ by Theorem \ref{thm:inverse-function-theorem-rn}(i),(iii).\\
The inverse images of these boxes under the finitely many partial compositions are open neighborhoods of $\ze$.\\
Choose an open box $B_\a$ inside their intersection and the factorization domain.\\
Then $\operatorname{CV}(\F_\a|_{B_\a})$ by Parts (ii)--(iii). Set $U_\a\defeq\a+B_\a$. Then $\operatorname{CV}(T|_{U_\a})$ by $T|_{U_\a}=\tau_{T(\a)}\circ(\F_\a|_{B_\a})\circ(\tau_{-\a}|_{U_\a})$ and Parts (ii)--(iii).\\
Since $L=T^{-1}(K)$, the family $(T(U_\a))_{\a\in L}$ is an open cover of $K$ by Part (i).
\item \hlt{Globalization}: Choose $\psi_1,\ldots,\psi_s$ with $f=\sum\limits_{i=1}^{s}\psi_i f$ by Theorem \ref{thm:partition-of-unity-rudin}(iv).\\
Choose $\a_1,\ldots,\a_s\in L$ with $(\forall\,i\in J_s)\,\operatorname{supp}(\psi_i f)\subseteq T(U_{\a_i})$ by Part (v) of same theorem. For $i\in J_s$, set $h_i\defeq\psi_i f$, $r_i\defeq R_{T|_{U_{\a_i}}}h_i$.\\
Then $\mathscr I(h_i)=\mathscr I(r_i)$ by $\operatorname{CV}(T|_{U_{\a_i}})$. If $\x\in E\setminus U_{\a_i}$, then $T(\x)\notin T(U_{\a_i})$ by injectivity, so $h_i(T(\x))=0$.\\
Thus $r_i$ is the zero extension from $E$ of $\x\mapsto h_i(T(\x))\abs{J_T(\x)}$. $\sum\limits_{i=1}^{s}h_i=f$, $\sum\limits_{i=1}^{s}r_i=\widetilde\rho$.\\
Hence $\mathscr I(f)=\sum\limits_{i=1}^{s}\mathscr I(h_i)=\sum\limits_{i=1}^{s}\mathscr I(r_i)=\mathscr I(\widetilde\rho)$ by repeated one-variable linearity in Theorem \ref{thm:riemann-stieltjes-integral-properties}(i), Proposition \ref{prop:support-and-simplex-integrals-well-defined}(i).
\end{enumerate}
\hlt{Structural Commentary}: Restriction and composition propagate $\operatorname{CV}$; a subordinate partition transfers it across compact support.\\
\hlt{Key Insight}: Compact support converts local Jacobian identities into one finite global identity.
\end{proof}
\end{theorem}

\begin{computationmethod} \label{meth:change-of-variables-jacobian} \hlt{Change-of-Variables Jacobian Method}\\
Fix $n\in\N$, an open set $\Omega\subseteq\R^n$, a map $\sigma\in C^1(\Omega,\R^n)$, and $f\in C_c(\R^n,\R)$. The method rewrites $I_{\y}\defeq\int_{\R^n}f(\y)\,d\y$ in the $\x$-coordinates.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Differential Data:} Compute $D\sigma(\x)$ and set $J_\sigma(\x)\defeq\det D\sigma(\x)$.
\item \hlt{Admissibility:} Verify that $\sigma$ is injective, $(\forall\,\x\in\Omega)\,J_\sigma(\x)\neq0$, and $\operatorname{supp}f\subseteq\sigma(\Omega)$.\\
If any condition fails, Theorem \ref{thm:change-of-variables-rk} does not justify the substitution.
\item \hlt{Support Pullback:} Set $K\defeq\operatorname{supp}f$ and $L\defeq\sigma^{-1}(K)$, and express $L$ in the $\x$-coordinates.
\item \hlt{Integrand Pullback:} On $\Omega$, set $\rho(\x)\defeq f(\sigma(\x))\abs{J_\sigma(\x)}$, and extend $\rho$ by $0$ to $\R^n$.
\item \hlt{Evaluation:} Verify that $\rho=0$ on $\R^n\setminus L$, and compute $I_{\x}\defeq\int_{\R^n}\rho(\x)\,d\x$. The output satisfies $I_{\y}=I_{\x}$ by Theorem \ref{thm:change-of-variables-rk}.
\end{enumerate}
\end{computationmethod}

\begin{example} \label{ex:polar-and-spherical-jacobians} \hlt{Polar and Spherical Jacobian Factors}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Polar coordinates}: Set $\Omega_{\mathrm p}\defeq(0,\infty)\times(0,2\pi)$ and $\sigma_{\mathrm p}(r,\theta)\defeq(r\cos\theta,r\sin\theta)$.\\
Equality of images fixes $r$ by Euclidean norms and then $\theta\in(0,2\pi)$; hence $\sigma_{\mathrm p}$ is injective and belongs to $C^1(\Omega_{\mathrm p},\R^2)$.\\
The identities $D\sigma_{\mathrm p}(r,\theta)=\left(\begin{smallmatrix}\cos\theta&-r\sin\theta\\\sin\theta&r\cos\theta\end{smallmatrix}\right)$ and $J_{\sigma_{\mathrm p}}(r,\theta)=r>0$ hold. Fix $f\in C_c(\R^2,\R)$ with $\operatorname{supp}f\subseteq\sigma_{\mathrm p}(\Omega_{\mathrm p})$.\\
Then $\int_{\R^2}f=\int_{\R^2}(f\circ\sigma_{\mathrm p})r$ by Theorem \ref{thm:change-of-variables-rk}, with the right integrand zero-extended from $\Omega_{\mathrm p}$.
\item \hlt{Spherical coordinates}: Set $\Omega_{\mathrm s}\defeq(0,\infty)\times(0,\pi)\times(0,2\pi)$. Define $\sigma_{\mathrm s}:\Omega_{\mathrm s}\to\R^3$ by $\sigma_{\mathrm s}(\rho,\varphi,\theta)\defeq(\rho\sin\varphi\cos\theta,\rho\sin\varphi\sin\theta,\rho\cos\varphi)$.\\
Equality of images successively fixes $\rho$, $\varphi\in(0,\pi)$, and $\theta\in(0,2\pi)$; hence $\sigma_{\mathrm s}$ is injective and belongs to $C^1(\Omega_{\mathrm s},\R^3)$.\\
The identity $J_{\sigma_{\mathrm s}}(\rho,\varphi,\theta)=\rho^2\sin\varphi>0$ holds by direct differentiation. Fix $f\in C_c(\R^3,\R)$ with $\operatorname{supp}f\subseteq\sigma_{\mathrm s}(\Omega_{\mathrm s})$.\\
Then $\int_{\R^3}f=\int_{\R^3}(f\circ\sigma_{\mathrm s})\rho^2\sin\varphi$ by Theorem \ref{thm:change-of-variables-rk}, with the right integrand zero-extended from $\Omega_{\mathrm s}$.
\end{enumerate}
\end{example}

\begin{proofmethod} \label{meth:multiple-integration-proof-patterns} \hlt{Multiple-Integration Proof Patterns}\\
Use method for existence or equality claims involving multiple Riemann integrals. Bind  domains, integrands, order, and exact target.\\
Parts (i), (ii), and (iii) are standalone; Parts (iv)--(vi) form the coordinate-change route.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Order permutation}: Let $I^k$ be a $k$-cell, $f\in C(I^k)$. If only the integration order changes, encode the target order by $\pi\in\mathfrak S_k$.\\
Then $I_\pi(f)$ exists and equals $\int_{I^k}f$ by Theorem \ref{thm:iterated-integrals-order-independent}.
\item \hlt{Slice existence}: Let $R\defeq[a,b]\times[c,d]$, $f:R\to\R$ be Riemann integrable. For $y\in[c,d]$, set $f_y(x)\defeq f(x,y)$.\\
Set $\underline F(y)\defeq\sup\limits_P L(P,f_y)$ and $\overline F(y)\defeq\inf\limits_P U(P,f_y)$. Common product grid transfers its Darboux gap to both slice envelopes.\\
Then $\underline F,\overline F\in\mathcal R[c,d]$ by Theorem \ref{thm:fubini-riemann-rectangle-slices}(i), and their integrals equal $\int_R f$ by Part (ii).\\
If $(\forall\,y\in[c,d])\,f_y\in\mathcal R[a,b]$, set $g(y)\defeq\int_a^b f(x,y)\,dx$. Then $g\in\mathcal R[c,d]$ and $\int_R f=\int_c^d g(y)\,dy$ by Part (iii).
\item \hlt{Compact support}: For $f\in C_c(\R^k,\mathbb F)$, choose $k$-cell $I^k\supseteq\operatorname{supp}f$; $\int_{\R^k}f=\int_{I^k}f$, independently of $I^k$, by Proposition \ref{prop:support-and-simplex-integrals-well-defined}(i).
\item \hlt{Local coordinate form}: Let $E\subseteq\R^k$ be open and $T\in C^1(E,\R^k)$ be injective with $(\forall\,\x\in E)\,J_T(\x)\neq0$.\\
At $\a\in E$, set $E_\a\defeq\{\h\in\R^k:\a+\h\in E\}$. Define $\F_\a:E_\a\to\R^k$ by $\F_\a(\h)\defeq T(\a+\h)-T(\a)$.\\
Then $\F_\a(\ze)=\ze$ and $D\F_\a(\ze)=DT(\a)$ is invertible by Corollary \ref{cor:jacobian-nonzero-iff-derivative-invertible}.\\
Factor $\F_\a$ locally into primitive maps and flips by Theorem \ref{thm:local-decomposition-primitive-mappings}.\\
Restrict each primitive factor to a box on which it is a diffeomorphism.\\
Close its integral identity by Theorem \ref{thm:riemann-stieltjes-change-of-variable}(v), and close flips by Theorem \ref{thm:linear-map-volume-scaling-determinant}.
\item \hlt{Composition}: For local factors $Q:U\to V$ and $P:V\to W$, apply their integral identities successively.\\
Track the volume factor through $J_{P\circ Q}=(J_P\circ Q)J_Q$ by Theorems \ref{thm:chain-rule-several-variables}(ii) and \ref{thm:determinant-basic-laws}(vi).
\item \hlt{Globalization}: For $f\in C_c(\R^k,\R)$ with $K\defeq\operatorname{supp}f\subseteq T(E)$, cover $K$ by local images.\\
Choose $f=\sum\limits_{i=1}^{s}\psi_i f$ subordinate to those images by Theorem \ref{thm:partition-of-unity-rudin}(iv)--(v). Apply the local identity to each $\psi_i f$ and sum.
\end{enumerate}
\hlt{Verification}: Check every intermediate integral and required zero extension exists.\\
Check each localized support lies in its chosen image and the final equality has the original domains, integrands, and order.\\
\hlt{Key Insight}: Cell identities control order; primitive factors and subordinate supports control coordinates.
\end{proofmethod}
