\subsection{Further Topics in Uniform Convergence}

\begin{method} \label{meth:uniform-convergence-proof-patterns-source-todo}\label{meth:uniform-convergence-proof-patterns} \hlt{Uniform Convergence Proof Patterns}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Pointwise}: fix $x$, then choose $N=N(x,\varepsilon)$.
\item \hlt{Uniform}: choose $N=N(\varepsilon)$; equivalently $\sup_E\abs{f_n-f}\to0$.
\item \hlt{Uniform Cauchy}: prove $\sup_E\abs{f_n-f_m}\to0$ as $m,n\to\infty$.
\item \hlt{Preservation}: use a three-term estimate and choose the uniform index first.
\item \hlt{Series of functions}: apply Weierstrass $M$-test when numerical absolute tails dominate all functional tails.
\item \hlt{Termwise operations}: require $f_n\to f$ locally uniformly, i.e. $\sup_{x\in K}\abs{f_n-f}\to0$ for each compact $K$, before $\lim_n\int_K f_n=\int_K f$; for derivatives, require $f_n'\to g$ locally uniformly and get $(\lim_n f_n)'=g$.
\end{enumerate}
\hlt{Key Insight}: Uniformity means $N$ is chosen before $x$.
\end{method}
