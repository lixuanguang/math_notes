\subsection{Further Topics in Uniform Convergence}

\begin{proofmethod} \label{meth:uniform-convergence-proof-patterns} \hlt{Uniform Convergence Proof Patterns}\\
\hlt{Use When}: Proving pointwise or uniform convergence, preserving a uniform limit, or justifying a termwise operation.\\
Let $\emptyset\ne E$ and let $f_n:E\to\C$; in branches with a specified limit, let $f:E\to\C$.\\
Select the branch matching the target and verify its stated hypotheses.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Pointwise Target}: Establish $(\forall x\in E)(\forall\varepsilon>0)(\exists N\in\N)(\forall n\geq N)\ \abs{f_n(x)-f(x)}\leq\varepsilon$.\\
The index may depend on $(x,\varepsilon)$ by Definition \ref{def:pointwise-convergence-functions}.
\item \hlt{Direct Uniform Target}: Establish $(\forall\varepsilon>0)(\exists N\in\N)(\forall n\geq N)(\forall x\in E)\ \abs{f_n(x)-f(x)}\leq\varepsilon$.\\
Its quantifier order makes $N$ independent of $x$ by Definition \ref{def:uniform-convergence-functions}.
\item \hlt{Supremum Route}: After pointwise convergence to $f$ is known, set $M_n\defeq\sup\limits_{\overline{\R}}\{\abs{f_n(x)-f(x)}:x\in E\}$.\\
Then $f_n\rightrightarrows f\Longleftrightarrow M_n\to0$ by Theorem \ref{thm:uniform-convergence-supremum-test}.
\item \hlt{Uniform-Cauchy Route}: If no candidate limit is known, establish $(\forall\varepsilon>0)(\exists N\in\N)(\forall m,n\geq N)(\forall x\in E)\ \abs{f_n(x)-f_m(x)}\leq\varepsilon$.\\
Uniform convergence follows by Theorem \ref{thm:uniform-cauchy-criterion}.
\item \hlt{Function-Series Route}: Choose $(M_n)\subseteq[0,+\infty)$ satisfying $(\forall n\in\N)(\forall x\in E)\ \abs{f_n(x)}\leq M_n$.\\
If $\sum M_n$ converges, then $\sum f_n$ converges uniformly on $E$ by Theorem \ref{thm:weierstrass-m-test}.
\item \hlt{Local-Limit Route}: Assume $f_n\rightrightarrows f$, let $E$ be a metric subspace, and let $p$ be a limit point of $E$.\\
For $n\in\N$, set $A_n\defeq\lim\limits_{t\to p}f_n(t)$ and verify $A_n\in\C$. Interchange the uniform and local limits by Theorem \ref{thm:uniform-convergence-interchanges-function-limits}.\\
If each $f_n$ is continuous, conclude continuity of $f$ by Theorem \ref{thm:uniform-convergence-preserves-continuity}.
\item \hlt{Integration Route}: Let $a<b$, let $\alpha:[a,b]\to\R$ be increasing, and let $f_n,f:[a,b]\to\R$.\\
Verify $(\forall n\in\N)\ f_n\in\mathcal{R}(\alpha)$ and uniform convergence of $\sum f_n$ to $f$. Then $\int_a^b f\,d\alpha=\sum\limits_{n=1}^{\infty}\int_a^b f_n\,d\alpha$ by Corollary \ref{cor:termwise-riemann-stieltjes-integration}.
\item \hlt{Differentiation Route}: Let $f_n:[a,b]\to\R$ be differentiable relative to $[a,b]$ and let $g:[a,b]\to\R$.\\
Choose $x_0\in[a,b]$ and verify that $(f_n(x_0))$ converges and $f_n'\rightrightarrows g$. Pass derivatives to the uniform limit by Theorem \ref{thm:uniform-derivative-convergence}.\\
For a parameterized Riemann--Stieltjes integral, verify slice integrability and uniform derivative-slice control.\\
Differentiate under the integral by Theorem \ref{thm:differentiation-under-riemann-stieltjes-integral}.
\end{enumerate}
\hlt{Verification}: Confirm the selected endpoint, its domain and codomain, and every cited hypothesis.\\
An argument supplying only an $x$-dependent index establishes at most pointwise convergence.\\
\hlt{Key Insight}: Uniformity is the quantifier order $N=N(\varepsilon)$ before $x\in E$.
\end{proofmethod}
