\subsection{Riemann Integration}

The integration side of one-variable calculus is organized by Riemann and Riemann--Stieltjes sums; the fundamental theorem connects the two halves.

This subsection develops Riemann and Riemann--Stieltjes integration by partitions, then links integrals to antiderivatives through the Fundamental Theorem of Calculus. Point partitions and interval-envelope partitions are equivalent via common refinement.

\begin{definition} \label{def:partition-and-riemann-upper-lower-sums}\label{def:riemann-stieltjes-core-source-todo} \hlt{Partitions and Riemann Upper/Lower Sums}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{partition} of $[a,b]$ is $\Pi=\{x_0,\ldots,x_n\}$ with $a=x_0\leq x_1\leq\cdots\leq x_n=b$.\\
Set $\Delta x_i\defeq x_i-x_{i-1}$ for $1\leq i\leq n$.
\item For bounded $f:[a,b]\to\R$, set $M_i\defeq\sup_{x\in[x_{i-1},x_i]}f(x)$ and $m_i\defeq\inf_{x\in[x_{i-1},x_i]}f(x)$.
\item The \hlt{upper} and \hlt{lower Darboux sums} are $U(\Pi,f)\defeq\sum\limits_{i=1}^{n}M_i\Delta x_i$ and $L(\Pi,f)\defeq\sum\limits_{i=1}^{n}m_i\Delta x_i$.
\item The \hlt{upper} and \hlt{lower Darboux integrals} are $\overline{\int_a^b f\,dx}\defeq\inf_{\Pi}U(\Pi,f)$ and $\underline{\int_a^b f\,dx}\defeq\sup_{\Pi}L(\Pi,f)$, where $\Pi$ ranges over all partitions of $[a,b]$.
\item If $\overline{\int_a^b f\,dx}=\underline{\int_a^b f\,dx}$, then $f$ is \hlt{Riemann integrable} on $[a,b]$; $f\in\mathcal{R}[a,b]$, $\int_a^b f\,dx\defeq\overline{\int_a^b f\,dx}=\underline{\int_a^b f\,dx}$.
\end{enumerate}
If $m\leq f(x)\leq M$ for $x\in[a,b]$, then $m(b-a)\leq L(\Pi,f)\leq U(\Pi,f)\leq M(b-a)$ for every partition $\Pi$.\\
Hence the upper and lower Darboux integrals are finite for bounded $f$.\\
\hlt{Key Insight}: Riemann integrability means least upper Darboux area equals greatest lower Darboux area.
\end{definition}

\begin{definition} \label{def:tagged-partitions-and-riemann-sums} \hlt{Tagged Partitions and Riemann Sums}\\
Let $\Pi=\{x_0,\ldots,x_n\}$ be a partition of $[a,b]$, and set $\Delta x_i\defeq x_i-x_{i-1}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{tagging} of $\Pi$ is $c=(c_1,\ldots,c_n)$ with $c_i\in[x_{i-1},x_i]$ for $1\leq i\leq n$; the pair $(\Pi,c)$ is a \hlt{tagged partition}.
\item For $f:[a,b]\to\R$, the associated \hlt{Riemann sum} is $R(f,\Pi,c)\defeq\sum\limits_{i=1}^{n}f(c_i)\Delta x_i$.
\item If $f$ is bounded, then $L(\Pi,f)\leq R(f,\Pi,c)\leq U(\Pi,f)$.
\end{enumerate}
\end{definition}

\begin{perspective} \label{per:gauge-integral-context} \hlt{Tagged Sums and Gauges}\\
Gauge integration keeps tagged sums but replaces one mesh bound $\delta$ with a local bound $\delta(c_i)$.\\
Thus fineness may vary by tag, giving a stronger framework for FTC-type results.
\end{perspective}
