\subsection{Categories and Functors}

\begin{definition} \label{def:linear-map-morphism-terminology} \hlt{Morphism Terminology for Linear Maps}\\
For $T\in\LL(\mathcal{V},\mathcal{W})$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is a \hlt{homomorphism}.
\item If $\mathcal{V}=\mathcal{W}$, then $T$ is an \hlt{endomorphism}.
\item If $T$ is injective, then $T$ is a \hlt{monomorphism}, or \hlt{embedding}.
\item If $T$ is surjective, then $T$ is an \hlt{epimorphism}.
\item If $T$ is bijective, then $T$ is an \hlt{isomorphism}.
\item If $T$ is a bijective endomorphism, then $T$ is an \hlt{automorphism}.
\end{enumerate}
\end{definition}
