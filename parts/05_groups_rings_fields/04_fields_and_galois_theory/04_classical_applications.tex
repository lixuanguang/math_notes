\subsection{Classical Applications}

\begin{perspective} \label{per:finite-fields-and-characteristic-two} \hlt{Finite Fields and Characteristic}\\
If $\operatorname{char}(F)=2$, then $(\forall a\in F)\ 2a=0$, so $a=-a$. A finite field has $|F|=p^n$ with $p$ prime and $n\geq1$; conversely, each prime power $q=p^n$ determines a unique field $\mathbb{F}_q$ up to isomorphism.\\
Thus the classification is field-theoretic, while the characteristic constraint is ring-theoretic.
\end{perspective}
