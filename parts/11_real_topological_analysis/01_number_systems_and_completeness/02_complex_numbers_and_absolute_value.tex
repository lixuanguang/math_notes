\subsection{Complex Numbers and Absolute Value}

\begin{definition} \label{def:complex-number-field-operations} \hlt{Complex Numbers}\\
Let $\C\defeq\R^2$. For $x=(a,b)$ and $y=(c,d)$ in $\C$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $x=y$ iff $a=c$ and $b=d$.
\item $x+y\defeq(a+c,b+d)$.
\item $xy\defeq(ac-bd,ad+bc)$.
\end{enumerate}
Elements of $\C$ are \hlt{complex numbers}.
\end{definition}

\begin{theorem} \label{thm:complex-numbers-form-field} \hlt{The Complex Numbers Form a Field}\\
Let $\C=\R^2$ with $(a,b)+(c,d)=(a+c,b+d)$ and $(a,b)(c,d)=(ac-bd,ad+bc)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\C$ is a field.
\item The additive and multiplicative identities are $(0,0)$ and $(1,0)$.
\item If $x=(a,b)\neq(0,0)$, then $x^{-1}=\left(\frac{a}{a^2+b^2},\frac{-b}{a^2+b^2}\right)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $x,y,z\in\C$, associativity, commutativity, and distributivity follow by expanding coordinates in $\R$; $-(a,b)=(-a,-b)$, and (iii) gives $x^{-1}$ for $x\neq0$.
\item Directly, $(a,b)+(0,0)=(a,b)$ and $(a,b)(1,0)=(a,b)$.
\item Since $(a,b)\neq(0,0)$, $a^2+b^2>0$. Multiplication gives $(a,b)\left(\frac{a}{a^2+b^2},\frac{-b}{a^2+b^2}\right)=(1,0)$.
\end{enumerate}
\hlt{Structural Commentary}: Inverses use conjugate-over-norm.\\
\hlt{Key Insight}: $a^2+b^2>0$ for $(a,b)\neq(0,0)$.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:real-field-embedded-in-complex-field} \hlt{Real Numbers Inside Complex Numbers}\\
Let $\iota:\R\to\C$ be $\iota(a)\defeq(a,0)$, and set $i\defeq(0,1)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\iota(a+b)=\iota(a)+\iota(b)$ and $\iota(ab)=\iota(a)\iota(b)$ for all $a,b\in\R$.
\item $\iota$ is injective, $\iota(0)=(0,0)$, and $\iota(1)=(1,0)$.
\end{enumerate}
Where $a\in\R$ with $\iota(a)\in\C$; $i$ is the \hlt{complex unit}.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\iota(a)+\iota(b)=(a,0)+(b,0)=(a+b,0)=\iota(a+b)$, and $\iota(a)\iota(b)=(a,0)(b,0)=(ab,0)=\iota(ab)$.
\item If $\iota(a)=\iota(b)$, then $(a,0)=(b,0)$, so $a=b$. Also $\iota(0)=(0,0)$ and $\iota(1)=(1,0)$.
\end{enumerate}
\hlt{Structural Commentary}: Real arithmetic embeds as the real axis.\\
\hlt{Key Insight}: $\iota$ preserves $+,\cdot,0,1$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:complex-unit-and-coordinate-form} \hlt{Coordinate Form and $i^2$}\\
For $i=(0,1)\in\C$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $i^2=-1$.
\item For all $a,b\in\R$, $(a,b)=a+bi$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $i^2=(0,1)(0,1)=(-1,0)=-1$.
\item $a+bi=(a,0)+(b,0)(0,1)=(a,0)+(0,b)=(a,b)$.
\end{enumerate}
\hlt{Structural Commentary}: Pair coordinates become $a+bi$.\\
\hlt{Key Insight}: $i=(0,1)$ encodes the second coordinate.
\end{proof}
\end{proposition}

\begin{definition} \label{def:complex-conjugate-real-imaginary-parts} \hlt{Conjugate, Real Part, and Imaginary Part}\\
Let $z=a+bi$ with $a,b\in\R$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{complex conjugate} of $z$ is $\overline{z}\defeq a-bi$.
\item The \hlt{real part} and \hlt{imaginary part} of $z$ are $\operatorname{Re}(z)\defeq a$ and $\operatorname{Im}(z)\defeq b$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:complex-conjugation-identities} \hlt{Basic Conjugation Identities}\\
If $z,w\in\C$, then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\overline{z+w}=\overline{z}+\overline{w}$;
\item $\overline{zw}=\overline{z}\,\overline{w}$;
\item $z+\overline{z}=2\operatorname{Re}(z)$ and $z-\overline{z}=2i\operatorname{Im}(z)$;
\item $z\overline{z}$ is real and positive unless $z=0$.
\end{enumerate}
\begin{proof}
Write $z=a+bi$ and $w=c+di$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\overline{z+w}=(a+c)-(b+d)i=\overline{z}+\overline{w}$.
\item $\overline{zw}=(ac-bd)-(ad+bc)i=(a-bi)(c-di)=\overline{z}\,\overline{w}$.
\item $z+\overline{z}=2a=2\operatorname{Re}(z)$ and $z-\overline{z}=2bi=2i\operatorname{Im}(z)$.
\item $z\overline{z}=a^2+b^2$, positive unless $a=b=0$, i.e. unless $z=0$.
\end{enumerate}
\hlt{Structural Commentary}: Conjugation fixes $\operatorname{Re}$ and negates $\operatorname{Im}$.\\
\hlt{Key Insight}: $z\overline{z}=a^2+b^2\geq0$.
\end{proof}
\end{proposition}

\begin{definition} \label{def:complex-absolute-value-analysis} \hlt{Absolute Value of a Complex Number}\\
For $z=a+bi\in\C$, define $\abs{z}\defeq\sqrt{z\overline{z}}=\sqrt{a^2+b^2}$, the unique $r\in[0,\infty)$ with $r^2=z\overline{z}=a^2+b^2$.
\end{definition}

\begin{theorem} \label{thm:complex-absolute-value-identities} \hlt{Absolute Value Identities}\\
For $z,w\in\C$,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\abs{z}>0$ unless $z=0$, and $\abs{0}=0$;
\item $\abs{\overline{z}}=\abs{z}$;
\item $\abs{zw}=\abs{z}\abs{w}$;
\item $\abs{\operatorname{Re}(z)}\leq\abs{z}$;
\item $\abs{z+w}\leq\abs{z}+\abs{w}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $z=a+bi$, then $\abs{z}=(a^2+b^2)^{1/2}$; this is positive unless $a=b=0$, and $\abs{0}=0$.
\item $\abs{\overline{z}}^2=\overline{z}z=z\overline{z}=\abs{z}^2$, so nonnegative square roots give $\abs{\overline{z}}=\abs{z}$.
\item $\abs{zw}^2=zw\overline{zw}=z\overline{z}w\overline{w}=\abs{z}^2\abs{w}^2$; take nonnegative square roots.
\item If $z=a+bi$, then $\abs{\operatorname{Re}(z)}=\abs{a}\leq(a^2+b^2)^{1/2}=\abs{z}$.
\item $z\overline{w}$ and $\overline{z}w$ are conjugates, $\abs{z+w}^2 =\abs{z}^2+2\operatorname{Re}(z\overline{w})+\abs{w}^2 \leq\abs{z}^2+2\abs{z\overline{w}}+\abs{w}^2 =(\abs{z}+\abs{w})^2$, using (iii) and (iv). Take nonnegative square roots.
\end{enumerate}
\hlt{Structural Commentary}: The complex absolute value behaves like length because $z\overline{z}$ is a squared length.\\
\hlt{Key Insight}: Triangle inequality is a real-part estimate plus multiplicativity.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:finite-schwarz-inequality-complex-scalars} \hlt{Finite Schwarz Inequality for Complex Scalars}\\
Let $n\in\N$ and let $a_1,\ldots,a_n,b_1,\ldots,b_n\in\C$. Then $\abs{\sum_{j=1}^{n}a_j\overline{b_j}}^2\leq \left(\sum_{j=1}^{n}\abs{a_j}^2\right)\left(\sum_{j=1}^{n}\abs{b_j}^2\right)$.
\begin{proof} Set $A\defeq\sum_{j=1}^{n}\abs{a_j}^2$, $B\defeq\sum_{j=1}^{n}\abs{b_j}^2$, and $C\defeq\sum_{j=1}^{n}a_j\overline{b_j}$. If $B=0$, then every $b_j=0$, so $C=0$.\\
If $B>0$, then $0\leq\sum_{j=1}^{n}\abs{Ba_j-Cb_j}^2=B^2A-B\overline{C}C-BC\overline{C}+\abs{C}^2B=B(AB-\abs{C}^2)$. Thus $\abs{C}^2\leq AB$.\\
\hlt{Structural Commentary}: Expand a nonnegative square sum.\\
\hlt{Key Insight}: $0\leq\sum\abs{Ba_j-Cb_j}^2$ controls $\abs{C}^2$.
\end{proof}
\end{theorem}
