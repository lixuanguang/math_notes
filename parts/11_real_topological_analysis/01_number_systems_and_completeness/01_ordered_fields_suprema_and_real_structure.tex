\subsection{Ordered Fields, Suprema, and Real Structure}

% **Role:** rigorous language of real and complex analysis.
%
% Natural numbers, integers, rationals, reals, complex numbers, ordered fields, supremum property, Archimedean property, density.

$\Q$ has gaps that completeness repairs in $\R$, while $\C$ retains field algebra without a compatible order.

\begin{definition} \label{def:field-and-ordered-field-analysis} \hlt{Ordered Fields}\\
Let $F$ be a field equipped with a total order $<$. Then $F$ is an \hlt{ordered field} iff
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall x,y,z\in F)\,[y<z\Rightarrow x+y<x+z]$;
\item $(\forall x,y\in F)\,[(0<x\land0<y)\Rightarrow0<xy]$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:field-axiom-consequences-analysis} \hlt{Basic Consequences of the Field Axioms}\\
Let $F$ be a field. Then $(\forall a,b,c,x,y\in F)$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $a+c=b+c\Rightarrow a=b$, and $c\neq0\Rightarrow(ac=bc\Rightarrow a=b)$.
\item The identities $0,1$ and the inverses $-a$, $a^{-1}$ for $a\neq0$ are unique.
\item $0x=0$.
\item $xy=0\Rightarrow(x=0\lor y=0)$.
\item $(-x)y=-(xy)=x(-y)$ and $(-x)(-y)=xy$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $a=b$ by adding $-c$ to $a+c=b+c$. If $c\neq0$, then $a=b$ by multiplying $ac=bc$ by $c^{-1}$.
\item In a group $(G,\ast,e)$, any identity $e'$ satisfies $e=e\ast e'=e'$.\\
If $u\ast a=e=a\ast v$, then $u=u\ast e=u\ast(a\ast v)=(u\ast a)\ast v=v$. These calculations apply to $(F,+,0)$ and $(F\setminus\{0\},\cdot,1)$.
\item $0x=0$ by cancellation from $0x+0x=(0+0)x=0x$.
\item If $x=0$, then $x=0\lor y=0$. If $xy=0$ and $x\neq0$, then $y=0$ by multiplication by $x^{-1}$.
\item $(-x)y=-(xy)$ by $xy+(-x)y=(x+(-x))y=0$, and $x(-y)=-(xy)$ by the same calculation.\\
The identity $(-x)(-y)=-(x(-y))=xy$ follows by the first equality.
\end{enumerate}
\hlt{Structural Commentary}: Group cancellation fixes identities and inverses; distributivity then controls zero and signs.\\
\hlt{Key Insight}: Uniqueness turns zero-sum identities into sign laws.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:ordered-field-inequality-rules-analysis} \hlt{Basic Inequality Rules in Ordered Fields}\\
Let $F$ be an ordered field. Then $(\forall c,x,y\in F)$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $x>0\Longleftrightarrow-x<0$.
\item $(c>0\land x<y)\Rightarrow cx<cy$.
\item $(c<0\land x<y)\Rightarrow cy<cx$.
\item $x\neq0\Rightarrow x^2>0$; in particular, $1>0$.
\item $0<x<y\Rightarrow0<y^{-1}<x^{-1}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) $-x<0$ by adding $-x$ to $0<x$.
\item[$(i)$] ($\Leftarrow$) $0<x$ by adding $x$ to $-x<0$.
\item[$(ii)$] The relations $0<y-x$ and $0<c(y-x)=cy-cx$ hold by additive order invariance and Definition \ref{def:field-and-ordered-field-analysis}(ii), respectively.\\
The inequality $cx<cy$ follows by adding $cx$ to $0<cy-cx$.
\item[$(iii)$] $-c>0$ by (i), and $(-c)x<(-c)y$ by (ii). The inequality $cy<cx$ follows by adding $cx+cy$ to $-cx<-cy$.
\item[$(iv)$] If $x>0$, then $x^2>0$ by Definition \ref{def:field-and-ordered-field-analysis}(ii). If $x<0$, then $-x>0$ by (i), so $x^2=(-x)^2>0$.\\
One case holds by $x\neq0$ and totality. For $x=1$, $1=1^2>0$.
\item[$(v)$] $x^{-1}=x(x^{-1})^2>0$ and $y^{-1}=y(y^{-1})^2>0$ by (iv) and Definition \ref{def:field-and-ordered-field-analysis}(ii).\\
$x^{-1}y^{-1}>0$ by Definition \ref{def:field-and-ordered-field-analysis}(ii), and $y^{-1}<x^{-1}$ by (ii) with multiplier $x^{-1}y^{-1}$.
\end{enumerate}
\hlt{Structural Commentary}: Additive order invariance reduces every inequality rule to multiplication by a positive element.\\
\hlt{Key Insight}: A negative multiplier is the negative of a positive one.
\end{proof}
\end{proposition}

\begin{motivation} \label{mot:dedekind-construction-as-completeness-proof} \hlt{Completeness Becomes a Theorem}\\
Dedekind cuts encode a real number by its rational lower side.\\
The construction turns rational arithmetic into cut arithmetic and bounded suprema into unions.
\end{motivation}

\begin{definition} \label{def:dedekind-cuts-and-order} \hlt{Dedekind Cuts and Their Order}\\
A \hlt{cut} is a set $\alpha\subseteq\Q$ such that
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\emptyset\neq\alpha\neq\Q$;
\item $(\forall p\in\alpha)(\forall q\in\Q)\,[q<p\Rightarrow q\in\alpha]$;
\item $(\forall p\in\alpha)(\exists r\in\alpha)\,p<r$.
\end{enumerate}
Let $\mathscr{D}$ be the set of cuts. For $\alpha,\beta\in\mathscr{D}$, define $\alpha\leq\beta\Longleftrightarrow\alpha\subseteq\beta$ and $\alpha<\beta\Longleftrightarrow\alpha\subsetneq\beta$.\\
Within the construction, $p,q,r,\ldots\in\Q$ and $\alpha,\beta,\gamma,\ldots\in\mathscr{D}$. A cut is a proper rational initial segment without a largest element.
\end{definition}

\begin{remark} \label{rem:left-cut-versus-two-sided-cut} \hlt{Left Cuts Versus Two-Sided Cuts}\\
Two-sided notation writes $A\mid B$ with $A=\alpha$ and $B=\Q\setminus\alpha$.\\
Definition \ref{def:dedekind-cuts-and-order} retains $A$; its lack of a largest element prevents duplicate boundary representations.
\end{remark}

\begin{definition} \label{def:cut-arithmetic} \hlt{Arithmetic of Cuts}\\
For $\alpha,\beta\in\mathscr{D}$, define $\alpha+\beta\defeq\{r+s:r\in\alpha,\ s\in\beta\}$ and $0^*\defeq\{q\in\Q:q<0\}$.\\
Define $-\alpha\defeq\{p\in\Q:(\exists r\in\Q)\,[r>0\land-p-r\notin\alpha]\}$ and $\mathscr{D}^+\defeq\{\alpha\in\mathscr{D}:0^*<\alpha\}$.\\
For $\alpha,\beta\in\mathscr{D}^+$, define $\alpha\beta\defeq\{p\in\Q:(\exists r\in\alpha)(\exists s\in\beta)\,[r>0\land s>0\land p\leq rs]\}$. Set $1^*\defeq\{q\in\Q:q<1\}$.\\
For $\alpha\in\mathscr{D}^+$, define $\alpha^{-1}\defeq\{p\in\Q:p\leq0\lor[p>0\land(\exists q\in\Q\setminus\alpha)\,(q>0\land p<q^{-1})]\}$.\\
Extend multiplication to zero factors by $\alpha0^*=0^*\alpha=0^*$.
\begin{equation*}
\alpha\beta\defeq
\begin{cases}
(-\alpha)(-\beta), & \alpha<0^*,\ \beta<0^*,\\
-\bigl((-\alpha)\beta\bigr), & \alpha<0^*,\ \beta>0^*,\\
-\bigl(\alpha(-\beta)\bigr), & \alpha>0^*,\ \beta<0^*.
\end{cases}
\end{equation*}
Positive rational representatives multiply without reversing order.\\
The sign rules reduce every remaining product to the positive case.
\end{definition}

\begin{definition} \label{def:rational-cuts} \hlt{Rational Cuts}\\
For $r\in\Q$, define the \hlt{rational cut} $r^*\defeq\{p\in\Q:p<r\}$. Define $\iota:\Q\to\mathscr{D}$ by $\iota(r)\defeq r^*$.
\end{definition}

\begin{theorem} \label{thm:real-field-existence-lub} \hlt{Existence of the Real Field}\\
The Dedekind-cut construction satisfies the following claims.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\mathscr{D},+,\cdot,<)$ is an ordered field with the least-upper-bound property.
\item The map $\iota:\Q\to\mathscr{D}$, $r\mapsto r^*$, is an order-preserving field embedding.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Field and completeness}:
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Order and suprema}: Inclusion is reflexive, antisymmetric, and transitive. If $\alpha\not\subseteq\beta$, choose $p\in\alpha\setminus\beta$.\\
The relation $(\forall q\in\beta)\,q<p$ holds by downward closure of $\beta$, and $\beta\subseteq\alpha$ by downward closure of $\alpha$. Hence inclusion is total.\\
Let $\emptyset\neq A\subseteq\mathscr{D}$ have upper bound $\beta$, and set $\gamma\defeq\bigcup\limits_{\alpha\in A}\alpha$. The set $\gamma$ is nonempty, and $\gamma\subseteq\beta\neq\Q$.\\
Downward closure and the absence of a largest element pass from the member of $A$ containing a chosen $p\in\gamma$; hence $\gamma\in\mathscr{D}$.\\
Every $\alpha\in A$ satisfies $\alpha\subseteq\gamma$, while any upper bound $\delta$ of $A$ contains every $\alpha\in A$ and hence contains $\gamma$.\\
These bound relations prove $\gamma=\sup A$.
\item \hlt{Addition}: Set $\alpha+\beta$ is nonempty. Choose $r'\notin\alpha$,  $s'\notin\beta$. Relation $r+s<r'+s'$ holds for $r\in\alpha$, $s\in\beta$, so $r'+s'\notin\alpha+\beta$.\\
Downward closure follows from $q<r+s\Rightarrow q=r+(s+q-r-s)$ because the final summand lies in $\beta$.\\
If $p=r+s$, choose $r_1\in\alpha$ with $r<r_1$; then $p<r_1+s\in\alpha+\beta$. These properties prove $\alpha+\beta\in\mathscr{D}$.\\
Commutativity and associativity follow from the rational identities. Inclusion $\alpha+0^*\subseteq\alpha$ holds by downward closure.\\
The reverse inclusion follows from $p=r+(p-r)$ with $p<r\in\alpha$, so $\alpha+0^*=\alpha$.\\
The set $-\alpha$ is nonempty by any $q\notin\alpha$ and the witness $p=-q-1$. It is proper because $-a\notin-\alpha$ for $a\in\alpha$.\\
Downward closure follows from upward closure of $\Q\setminus\alpha$, and $p\in-\alpha$ with witness $r>0$ satisfies $p<p+\frac{r}{2}\in-\alpha$.\\
Thus $-\alpha$ is a cut. For $s\in-\alpha$, relation $-s\notin\alpha$ follows by its witness, so $(\forall r\in\alpha)\,r+s<0$ and $\alpha+(-\alpha)\subseteq0^*$.\\
For the reverse inclusion, let $v<0$ and set $w\defeq-\frac{v}{2}>0$. Rational multiples of $w$ are unbounded in both directions.\\
Indeed, if $w=\frac{A}{B}$ and $q=\frac{C}{D}$ with $A,B,D\in\N$, then $n=B(\abs{C}+1)$ satisfies $nw>q$.\\
The same calculation for $-q$ supplies a negative multiple below $q$.\\
Choose $a\in\alpha$ and $b\notin\alpha$. Select $j<k$ in $\Z$ with $jw<a$ and $kw>b$ by this unboundedness.\\
The finite set $\{\ell\in\Z:j\leq\ell\leq k,\ \ell w\in\alpha\}$ has a largest element $n$, with $(n+1)w\notin\alpha$.\\
The relations $nw\in\alpha$, $-(n+2)w\in-\alpha$ with witness $w$, and $v=nw-(n+2)w$ prove $v\in\alpha+(-\alpha)$. Hence $\alpha+(-\alpha)=0^*$.
\item \hlt{Positive multiplication}: For $\alpha,\beta>0^*$, the product $\alpha\beta$ is nonempty because both cuts contain positive rationals.\\
Positive $r'\notin\alpha$ and $s'\notin\beta$ satisfy $\alpha\beta\subseteq\{p:p<r's'\}$, so the product is proper.\\
Downward closure follows from the defining inequality. If $p\leq rs$ is a defining witness, choose $p<q<rs$ when $p<rs$.\\
If $p=rs$, choose $r<r_1\in\alpha$ and then $p<q<r_1s$. Hence $\alpha\beta$ has no largest element.\\
Commutativity follows from $rs=sr$. If $p\in(\alpha\beta)\gamma$, then $p\leq ut\leq rst$ for positive $u\in\alpha\beta$, $r\in\alpha$, $s\in\beta$, and $t\in\gamma$.\\
Since $st\in\beta\gamma$, $p\in\alpha(\beta\gamma)$. The reverse inclusion is symmetric, so multiplication is associative.\\
If $p\in\alpha(\beta+\gamma)$, choose positive $r\in\alpha$, $u=s_0+t_0\in\beta+\gamma$ with $p\leq ru$, and then positive $s\in\beta$, $t\in\gamma$ with $s>s_0$, $t>t_0$.\\
The membership $p\in\alpha\beta+\alpha\gamma$ follows from $p<r(s+t)=rs+rt$.\\
Conversely, if $p=u+v$ with $u\in\alpha\beta$ and $v\in\alpha\gamma$, choose positive witnesses $u\leq r_1s$, $v\leq r_2t$, and $r\in\alpha$ with $r>\max\{r_1,r_2\}$.\\
Then $p<r(s+t)$ with $s+t\in\beta+\gamma$, so $p\in\alpha(\beta+\gamma)$. Thus positive multiplication distributes over addition.\\
If $p\leq rs$ with $r\in\alpha$, $0<s<1$, then $p<r$ and $p\in\alpha$. Conversely, for $p\in\alpha$, choose $p<r\in\alpha$ and $0<s<1$ with $p\leq rs$.\\
Such an $s$ exists when $p>0$, while any $s>0$ works when $p\leq0$. Hence $\alpha1^*=\alpha$, and $\alpha,\beta>0^*\Rightarrow\alpha\beta>0^*$.
\item \hlt{Positive reciprocals}: For $\alpha>0^*$, the set $\alpha^{-1}$ contains every nonpositive rational and is nonempty.\\
It is proper because $r^{-1}\notin\alpha^{-1}$ for any positive $r\in\alpha$.\\
Downward closure is immediate. A larger member follows from a positive defining witness by taking a midpoint.\\
A nonpositive member is smaller than $\frac{1}{2q}\in\alpha^{-1}$ for any positive $q\notin\alpha$. These properties prove $\alpha^{-1}\in\mathscr{D}^+$.\\
If $p\in\alpha\alpha^{-1}$, then $p\leq rs$ for positive $r\in\alpha$ and $s<q^{-1}$ with positive $q\notin\alpha$. Since $r<q$, $p<\frac{r}{q}<1$, so $\alpha\alpha^{-1}\subseteq1^*$.\\
For the reverse inclusion, fix $0<c<1$ and positive $r_0\in\alpha$, $q_0\notin\alpha$.\\
Repeated bisection produces $r_k\in\alpha$, $q_k\notin\alpha$ with $0<r_0\leq r_k<q_k$ and $q_k-r_k=2^{-k}(q_0-r_0)$.\\
Choose $k$ with $q_k-r_k<\frac{(1-c)r_0}{c}$; such a $k$ exists because $2^k\geq k+1$ and positive rational multiples are unbounded.\\
Then $cq_k<r_k$. Choose $s\in\Q$ with $\frac{c}{r_k}<s<\frac{1}{q_k}$. The relations $s\in\alpha^{-1}$ and $c<r_ks$ prove $c\in\alpha\alpha^{-1}$.\\
Every $c\leq0$ belongs to the product by any positive pair of witnesses. Therefore $\alpha\alpha^{-1}=1^*$.
\item \hlt{Signs and order compatibility}: Addition by a fixed cut preserves inclusion.\\
If $\beta<\gamma$, then $\alpha+\beta\subseteq\alpha+\gamma$; equality would imply $\beta=\gamma$ by adding $-\alpha$. Hence $\alpha+\beta<\alpha+\gamma$ and $\alpha>0^*\Longleftrightarrow-\alpha<0^*$.\\
Every nonzero cut has the form $\varepsilon a$ with $a>0^*$ and $\varepsilon\in\{1,-1\}$.\\
The signed definition makes products commutative and associative and sets $(-a)^{-1}=-a^{-1}$.\\
For distributivity, $a(b-c)=ab-ac$ when $a,b,c>0^*$.\\
If $b>c$, then $a(b-c)+ac=ab$ by positive distributivity and additive cancellation.\\
The cases $b=c$ and $b<c$ follow by zero and sign reversal, so this identity covers all sign patterns.\\
Since $0^*\neq1^*$, all field axioms hold. Strict addition compatibility and $\alpha,\beta>0^*\Rightarrow\alpha\beta>0^*$ make the field ordered.
\end{enumerate}
\item \hlt{Rational embedding}: Each $r^*$ is a cut. For $r,s\in\Q$, one inclusion in $r^*+s^*=(r+s)^*$ follows from $u<r$ and $v<s$.\\
For $p<r+s$, set $t\defeq\frac{r+s-p}{2}>0$. Then $r-t\in r^*$, $s-t\in s^*$, and $p=(r-t)+(s-t)$, so $p\in r^*+s^*$.\\
For $r,s>0$, the forward inclusion in $r^*s^*=(rs)^*$ follows from $0<u<r$, $0<v<s\Rightarrow uv<rs$.\\
Conversely, if $0<p<rs$, choose $0<\delta<\min\{r,s,\frac{rs-p}{r+s}\}$. Then $p<(r-\delta)(s-\delta)$, so $p\in r^*s^*$.\\
Nonpositive $p$ belongs by any positive witnesses, and the signed cases follow from the definitions.\\
Finally, $r<s\Longleftrightarrow r^*<s^*$. The forward implication follows from $r\in s^*\setminus r^*$.\\
The reverse implication follows because $r^*<s^*$ supplies $p\in s^*\setminus r^*$ with $r\leq p<s$.\\
Hence $\iota$ is injective and preserves $0,1,+,\cdot$, and order.
\end{enumerate}
\hlt{Structural Commentary}: Cut inclusion supplies suprema; rational witnesses transfer field operations and reciprocals.\\
\hlt{Key Insight}: Bounded unions and boundary approximation create completeness and inverses.
\end{proof}
\end{theorem}

Write $\R\defeq\mathscr{D}$ and identify $\Q$, together with its subsets $\Z$ and $\N$, with its image under $\iota$.

\begin{definition} \label{def:irrational-and-dense-subsets-real-line} \hlt{Irrational and Dense Subsets}\\
A real number $x$ is \hlt{irrational} iff $x\in\R\setminus\Q$. A set $D\subseteq\R$ is \hlt{dense in $\R$} iff $(\forall a,b\in\R)\,[a<b\Rightarrow(\exists d\in D)\,a<d<b]$.
\end{definition}

\begin{theorem} \label{thm:nth-roots-positive-reals} \hlt{Existence and Uniqueness of Positive $n$th Roots}\\
Let $x\in\R$ with $x>0$, and let $n\in\N$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\exists y\in\R)\,[y>0\land y^n=x]$.
\item $(\forall y,z\in\R)\,[(y>0\land z>0\land y^n=x\land z^n=x)\Rightarrow y=z]$.
\end{enumerate}
Denote this unique root by $x^{1/n}$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Existence}:
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Supremum candidate}: Set $E\defeq\{t\in\R:t>0,\ t^n<x\}$. For $t_0\defeq\frac{x}{1+x}$, $0<t_0<1$ and $t_0^n\leq t_0<x$, so $E\neq\emptyset$.\\
Every $t\in E$ satisfies $t\leq1+x$ because $t>1+x$ would imply $t^n\geq t>x$.\\
Hence $y\defeq\sup E$ exists by Theorem \ref{thm:real-field-existence-lub}, and $y\geq t_0>0$.
\item \hlt{Lower perturbation}: For $0<a<b$, the factorization satisfies $0<b^n-a^n\leq n(b-a)b^{n-1}$.\\
Suppose $y^n<x$ and choose $0<h<\min\{1,\frac{x-y^n}{n(y+1)^{n-1}}\}$.\\
The bound $(y+h)^n-y^n\leq hn(y+h)^{n-1}<x-y^n$ places $y+h$ in $E$, contradicting that $y$ is an upper bound.
\item \hlt{Upper perturbation}: Suppose $y^n>x$ and set $k\defeq\frac{y^n-x}{ny^{n-1}}$. Then $0<k<\frac{y}{n}\leq y$, and $y^n-(y-k)^n\leq kny^{n-1}=y^n-x$.\\
The inequality $(y-k)^n\geq x$ makes $y-k$ an upper bound of $E$ by monotonicity of positive integer powers.\\
This contradicts $y=\sup E$. Hence $y^n=x$ by trichotomy.
\end{enumerate}
\item \hlt{Uniqueness}: If $0<y<z$, then $z^n-y^n=(z-y)(z^{n-1}+z^{n-2}y+\cdots+y^{n-1})>0$.\\
Equal positive $n$th powers therefore have equal bases by trichotomy.
\end{enumerate}
\hlt{Structural Commentary}: Completeness selects a supremum; two perturbations exclude powers below and above $x$.\\
\hlt{Key Insight}: The supremum cannot survive either strict power inequality.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:nth-root-product-positive-reals} \hlt{Product Rule for Positive Roots}\\
For $a,b\in\R$ with $a,b>0$ and $n\in\N$, $(ab)^{1/n}=a^{1/n}b^{1/n}$.
\begin{proof}
Set $\alpha\defeq a^{1/n}$ and $\beta\defeq b^{1/n}$. Then $(\alpha\beta)^n=\alpha^n\beta^n=ab$ by commutativity; $(ab)^{1/n}=\alpha\beta$ holds by Theorem \ref{thm:nth-roots-positive-reals}(ii).\\
\hlt{Structural Commentary}: Multiplicativity of $n$th powers transfers the product to the uniqueness claim.\\
\hlt{Key Insight}: Equal positive $n$th powers have equal bases.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:rational-square-root-two-gap} \hlt{The Rational Gap at $\sqrt{2}$}\\
Let $A\defeq\{p\in\Q:p>0,\ p^2<2\}$ and $B\defeq\{p\in\Q:p>0,\ p^2>2\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\nexists p\in\Q)\,p^2=2$.
\item The set $A$ has no largest element, and $B$ has no smallest element.
\item The nonempty set $A$ is bounded above in $\Q$ but has no supremum in $\Q$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Suppose $p=\frac{m}{n}\in\Q$ satisfies $p^2=2$, where $m,n\in\Z$ and $n\neq0$. Then $m^2=2n^2$, with $m,n\neq0$.\\
The conflicting parity of the exponent of $2$ contradicts unique factorization in $\Z$ by Theorems \ref{thm:integers-and-polynomial-ring-are-pids} and \ref{thm:pid-is-ufd}.
\item For $p>0$, set $q\defeq p-\frac{p^2-2}{p+2}=\frac{2p+2}{p+2}$. Then $q-p=-\frac{p^2-2}{p+2}$ and $q^2-2=\frac{2(p^2-2)}{(p+2)^2}$.\\
Thus $p\in A\Rightarrow q\in A$ with $q>p$, while $p\in B\Rightarrow q\in B$ with $0<q<p$.
\item Relations $1\in A$ and $(\forall p\in A)\,p<2$ make $A$ nonempty, bounded above. Suppose $s=\sup\limits_{\Q}A$. Then $s\geq1$, and $s^2\neq2$ by (i).\\
If $s^2<2$, the element constructed in (ii) belongs to $A$ and exceeds $s$.\\
If $s^2>2$, the element $q$ from (ii) satisfies $0<q<s$ and is an upper bound of $A$.\\
Any $a\in A$ with $a\geq q$ would satisfy $a^2\geq q^2>2$. Both cases contradict $s=\sup\limits_{\Q}A$.
\end{enumerate}
\hlt{Structural Commentary}: Unique factorization excludes the boundary; a rational correction moves toward it from either side.\\
\hlt{Key Insight}: Every rational candidate can be improved without crossing the gap.
\end{proof}
\end{proposition}

\begin{remark} \label{rem:rational-density-does-not-fill-gaps} \hlt{Density Does Not Mean Completeness}\\
Both $\Q$ and $\R$ are ordered fields. For $r,s\in\Q$ with $r<s$, the midpoint $\frac{r+s}{2}$ proves that $\Q$ is order-dense in itself.\\
$\Q$ nevertheless lacks the least-upper-bound property by Proposition \ref{prop:rational-square-root-two-gap}(iii). The field $\R$ has this property by Theorem \ref{thm:real-field-existence-lub}(i).
\end{remark}

\begin{corollary} \label{cor:infimum-as-negative-supremum} \hlt{Infimum by Reflection}\\
Let $\emptyset\neq B\subseteq\R$ be bounded below and set $-B\defeq\{-b:b\in B\}$. Then $-B$ is bounded above and $\inf B=-\sup(-B)$.
\begin{proof}
If $m$ is a lower bound of $B$, then $-m$ is an upper bound of $-B$. Hence $s\defeq\sup(-B)$ exists by Theorem \ref{thm:real-field-existence-lub}.\\
The element $-s$ is a lower bound of $B$ by $(\forall b\in B)\,(-b\leq s\Rightarrow-s\leq b)$.\\
If $t$ is a lower bound of $B$, then $-t$ is an upper bound of $-B$, so $s\leq-t$ and $t\leq-s$. Thus $-s=\inf B$ by the definition of an infimum.\\
\hlt{Structural Commentary}: Order reversal converts lower bounds of $B$ into upper bounds of $-B$.\\
\hlt{Key Insight}: Negating the least upper bound makes the greatest lower bound.
\end{proof}
\end{corollary}

\begin{definition} \label{def:real-absolute-value-and-distance} \hlt{Real Absolute Value and Distance}\\
For $x\in\R$, define $\abs{x}\defeq\max\{x,-x\}$. For $a,b\in\R$, define their real-line distance by $d(a,b)\defeq\abs{a-b}$.
\end{definition}

\begin{proposition} \label{prop:real-absolute-value-triangle-inequality} \hlt{Real Absolute-Value Laws}\\
For $x,y,a,b,c\in\R$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\abs{x}\geq0$, $\abs{x}=0\Longleftrightarrow x=0$, and $-\abs{x}\leq x\leq\abs{x}$;
\item $\abs{xy}=\abs{x}\abs{y}$;
\item $\abs{x+y}\leq\abs{x}+\abs{y}$;
\item $\abs{a-b}\leq\abs{a-c}+\abs{c-b}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) At least one of $x$ and $-x$ is nonnegative by totality, so $\abs{x}=\max\{x,-x\}\geq0$.\\
Both $x\leq\abs{x}$ and $-x\leq\abs{x}$ hold by the definition of a maximum, and the second inequality is equivalent to $-\abs{x}\leq x$.\\
If $\abs{x}=0$, then $x=0$ in each of the cases $x\geq0$ and $x<0$ by Definition \ref{def:real-absolute-value-and-distance}.
\item[$(i)$] ($\Leftarrow$) $\abs{0}=0$ by Definition \ref{def:real-absolute-value-and-distance}.
\item[$(ii)$] If $xy>0$, then $\abs{xy}=xy=\abs{x}\abs{y}$ when $x,y>0$ or $x,y<0$. If $xy<0$, then $\abs{xy}=-xy=\abs{x}\abs{y}$; the case $xy=0$ follows directly.
\item[$(iii)$] The inequalities $x+y\leq\abs{x}+\abs{y}$ and $-(x+y)\leq\abs{x}+\abs{y}$ follow by (i). Their maximum is $\abs{x+y}$, and therefore $\abs{x+y}\leq\abs{x}+\abs{y}$.
\item[$(iv)$] $\abs{a-b}=\abs{(a-c)+(c-b)}\leq\abs{a-c}+\abs{c-b}$ by (iii).
\end{enumerate}
\hlt{Structural Commentary}: The maximum representation supplies two-sided bounds; addition then transfers them to sums and distances.\\
\hlt{Key Insight}: Bounding a number and its negative bounds its absolute value.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:number-systems-from-representatives} \hlt{Number Systems from Representatives}\\
The constructions of $\Z$ and $\Q$ by equivalence classes require operations and order to descend from representatives.\\
Thus $x\sim x'$ and $y\sim y'$ must imply $x\star y\sim x'\star y'$, with analogous compatibility for order.\\
Dedekind cuts use subsets rather than equivalence classes, so closure of cut operations replaces representative independence.\\
\hlt{Key Insight}: Each construction must transfer algebra from its underlying data.
\end{perspective}

\begin{theorem} \label{thm:archimedean-and-rational-density} \hlt{Archimedean Property and Density of $\Q$}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall x,y\in\R)\,[x>0\Rightarrow(\exists n\in\N)\,nx>y]$.
\item $(\forall x,y\in\R)\,[x<y\Rightarrow(\exists p\in\Q)\,x<p<y]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $x>0$ and set $A\defeq\{nx:n\in\N\}$. If $A$ were bounded above, then $\alpha\defeq\sup A$ would exist by Theorem \ref{thm:real-field-existence-lub}.\\
Since $\alpha-x<\alpha$, element $\alpha-x$ is not upper bound of $A$. Choose $m\in\N$ with $\alpha-x<mx$; $\alpha<(m+1)x\in A$, contradiction.
\item Fix $x<y$. Choose $n\in\N$ with $n(y-x)>1$ by (i), and choose $N\in\N$ with $N>\abs{nx}+1$ by (i) with unit step.\\
The finite set $S\defeq\{k\in\Z:-N\leq k\leq N,\ k\leq nx\}$ is nonempty; let $m\defeq\max S$.\\
Since $nx<N$, inequalities $m\leq nx<m+1$ hold by maximality; $nx<m+1\leq nx+1<ny$, so $x<\frac{m+1}{n}<y$ with $\frac{m+1}{n}\in\Q$.
\end{enumerate}
\hlt{Structural Commentary}: Completeness forbids finite supremum for integer multiples; finite integer grid then locates rational point.\\
\hlt{Key Insight}: A mesh smaller than $y-x$ must cross the interval.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:rationals-and-irrationals-dense} \hlt{Density of Rational and Irrational Points}\\
For $x,y\in\R$ with $x<y$, $(\exists r\in\Q)(\exists s\in\R\setminus\Q)\,[x<r<y\land x<s<y]$.
\begin{proof}
Choose $r\in\Q\cap(x,y)$ by Theorem \ref{thm:archimedean-and-rational-density}(ii). The number $\eta\defeq2^{1/2}>0$ exists by Theorem \ref{thm:nth-roots-positive-reals}(i) with $x=n=2$.\\
It satisfies $\eta\notin\Q$ by Proposition \ref{prop:rational-square-root-two-gap}(i). Choose $n\in\N$ with $n(y-x)>\eta$ by Theorem \ref{thm:archimedean-and-rational-density}(i), $q\in\Q\cap(x-\frac{\eta}{n},y-\frac{\eta}{n})$ by part (ii).\\
Then $s\defeq q+\frac{\eta}{n}$ lies in $(x,y)$, and $s\notin\Q$ because $s\in\Q$ would imply $\eta=n(s-q)\in\Q$.\\
\hlt{Structural Commentary}: Rational density is applied before and after translating the interval by a small irrational.\\
\hlt{Key Insight}: A rational translate of a nonzero irrational remains irrational.
\end{proof}
\end{corollary}

\begin{definition} \label{def:extended-real-number-system} \hlt{Extended Real Number System}\\
The \hlt{extended real line} is $\overline{\R}\defeq\R\cup\{-\infty,+\infty\}$. It retains the usual order on $\R$ and satisfies $(\forall x\in\R)\,-\infty<x<+\infty$.\\
For $E\subseteq\R$, extend the extrema as follows.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E=\emptyset\Rightarrow\sup E=-\infty$; if $E$ is unbounded above, $\sup E\defeq+\infty$; otherwise retain the real supremum.
\item $E=\emptyset\Rightarrow\inf E=+\infty$; if $E$ is unbounded below, $\inf E\defeq-\infty$; otherwise retain the real infimum.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:extended-real-arithmetic-conventions} \hlt{Extended Arithmetic Conventions}\\
For $x\in\R$, define the following non-indeterminate operations:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(+\infty)+(+\infty)=+\infty$ and $(-\infty)+(-\infty)=-\infty$.
\item $x+(+\infty)=(+\infty)+x=+\infty$ and $x+(-\infty)=(-\infty)+x=-\infty$.
\item $x>0\Rightarrow[x(+\infty)=(+\infty)x=+\infty\land x(-\infty)=(-\infty)x=-\infty]$.
\item $x<0\Rightarrow[x(+\infty)=(+\infty)x=-\infty\land x(-\infty)=(-\infty)x=+\infty]$.
\end{enumerate}
The forms $(+\infty)+(-\infty)$ and $0(\pm\infty)$ are undefined, so $\overline{\R}$ is not a field.
\end{definition}

\begin{definition} \label{def:unbounded-real-intervals} \hlt{Unbounded Real Intervals}\\
For $a,b\in\R$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(a,+\infty)\defeq\{x\in\R:a<x\}$ and $[a,+\infty)\defeq\{x\in\R:a\leq x\}$.
\item $(-\infty,b)\defeq\{x\in\R:x<b\}$ and $(-\infty,b]\defeq\{x\in\R:x\leq b\}$.
\end{enumerate}
Define $(-\infty,+\infty)\defeq\R$.
\end{definition}
