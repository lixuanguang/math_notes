\subsection{Density, Separability, and Closure Tests}

\begin{proposition} \label{prop:dense-closure-criterion} \hlt{Density and Closure}\\
Let $(X,d)$ be a metric space and $E\subseteq X$. Then $E$ is dense in $X$ iff $\overline{E}=X$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $E$ dense in $X$. Every $p\in X$ belongs to $E$ or is limit point of $E$, so $p\in E\cup E^{\operatorname{acc}}=\overline{E}$. Thus $X\subseteq\overline{E}\subseteq X$, $\overline{E}=X$.
\item[$(\Leftarrow)$] Suppose $\overline{E}=X$. Then every $p\in X$ belongs to $E\cup E^{\operatorname{acc}}$. Hence $p$ belongs to $E$ or is a limit point of $E$, so $E$ is dense in $X$.
\end{enumerate}
\hlt{Structural Commentary}: The density alternative is membership in $E\cup E^{\operatorname{acc}}$.\\
\hlt{Key Insight}: Closure packages the points of $E$ together with all its limit points.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:dense-closure-and-interval-test} \hlt{Real-Line Density Interval Test}\\
Let $E\subseteq\R$. Then $E$ is dense in $\R$ iff every nonempty open interval $(a,b)$ satisfies $(a,b)\cap E\neq\emptyset$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $E$ is dense in $\R$. Let $(a,b)\neq\emptyset$, choose $p\in(a,b)$, and set $r\defeq\min\{p-a,b-p\}>0$.\\
If $p\in E$, then $p\in(a,b)\cap E$. If $p\notin E$, then $p$ is a limit point of $E$, so choose $q\in B_r(p)\cap(E\setminus\{p\})$.\\
Since $B_r(p)\subseteq(a,b)$, the point $q$ belongs to $(a,b)\cap E$.
\item[$(\Leftarrow)$] Suppose every nonempty open interval meets $E$, and fix $p\in\R\setminus E$.\\
For every $\varepsilon>0$, choose $q\in(p-\varepsilon,p+\varepsilon)\cap E$. Then $q\neq p$ and $q\in B_\varepsilon(p)\cap(E\setminus\{p\})$.\\
Thus every $p\in\R\setminus E$ is a limit point of $E$, so $E$ is dense in $\R$.
\end{enumerate}
\hlt{Structural Commentary}: On $\R$, balls and open intervals test the same local condition.\\
\hlt{Key Insight}: Density means no interval can avoid the set.
\end{proof}
\end{proposition}

\begin{definition} \label{def:separable-metric-space} \hlt{Separable Metric Space}\\
A metric space $(X,d)$ is \hlt{separable} iff it contains an at most countable dense subset.
\end{definition}

\begin{example} \label{ex:separable-and-nonseparable-metric-spaces} \hlt{Separable and Nonseparable Metric Spaces}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $k\in\Z_{>0}$. The set $\Q^k$ is countable by Theorem \ref{thm:finite-tuples-over-countable-set-countable} and Corollary \ref{cor:rationals-countable}.\\
For $\mathbf{x}\in\R^k$ and $\varepsilon>0$, choose $\mathbf{q}\in\Q^k$ satisfying $(\forall\,j\in\{1,\ldots,k\})\,\abs{x_j-q_j}<\frac{\varepsilon}{\sqrt{k}}$ by Theorem \ref{thm:archimedean-and-rational-density}(ii).\\
Then $\norm{\mathbf{x}-\mathbf{q}}_2=\sqrt{\sum\limits_{j=1}^{k}\abs{x_j-q_j}^2}<\varepsilon$, so $\Q^k$ is dense in $\R^k$. Hence $\R^k$ is separable; in particular, $\R$ is separable.\\
The standard isometry $\C^k\cong\R^{2k}$ identifies $(\Q+i\Q)^k$ with $\Q^{2k}$, so $\C^k$ is separable.
\item If discrete $X$ is at most countable, then $X$ itself is a dense witness to separability.\\
Conversely, let $D\subseteq X$ be an at most countable dense subset. For every $x\in X$, the ball $B_{\frac12}(x)=\{x\}$ meets $D$, so $x\in D$.\\
Thus $D=X$, and $X$ is at most countable. Therefore $X$ is separable iff $X$ is at most countable.
\end{enumerate}
\hlt{Structural Commentary}: Rational coordinate grids are dense in Euclidean spaces. Singleton balls force $D=X$ in discrete spaces.\\
\hlt{Key Insight}: Separability depends on whether countable data can meet every open ball.
\end{example}

\begin{theorem} \label{thm:supremum-belongs-to-closure} \hlt{Suprema Belong to Closures}\\
Let $\emptyset\neq E\subseteq\R$ be bounded above and let $y\defeq\sup E$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $y\in\overline{E}$.
\item If $E$ is closed, then $y\in E$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $y\in E$, then $y\in\overline{E}$. Suppose instead that $y\notin E$.\\
Fix $h>0$. If $y-h$ were an upper bound for $E$, then minimality of $y=\sup E$ would imply $y\leq y-h$.\\
This contradicts $h>0$, so $y-h$ is not an upper bound for $E$.\\
Choose $x_h\in E$ with $y-h<x_h\leq y$. Then $x_h\neq y$, $\abs{x_h-y}<h$, and $B_h(y)\cap(E\setminus\{y\})\neq\emptyset$. Hence $y\in E^{\operatorname{acc}}\subseteq\overline{E}$.
\item If $E$ is closed, then $y\in\overline{E}$ by (i) and $\overline{E}=E$ by Theorem \ref{thm:closure-smallest-closed-containing-set}(ii). Hence $y\in E$.
\end{enumerate}
\hlt{Structural Commentary}: Minimality of $y=\sup E$ produces points of $E$ arbitrarily close below $y$.\\
\hlt{Key Insight}: A supremum is either attained or approached.
\end{proof}
\end{theorem}
