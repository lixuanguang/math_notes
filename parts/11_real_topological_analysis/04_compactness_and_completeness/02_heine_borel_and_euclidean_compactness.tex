\subsection{Heine--Borel and Euclidean Compactness}

\begin{theorem} \label{thm:nested-closed-intervals-intersect} \hlt{Nested Closed Intervals Intersect}\\
Let $(I_n)_{n\in\N}$ be nonempty closed intervals in $\R$, where $I_n=[a_n,b_n]$. If $(\forall n\in\N)\,I_{n+1}\subseteq I_n$, then $\bigcap\limits_{n=1}^{\infty}I_n\ne\emptyset$.
\begin{proof}
Set $A\defeq\{a_n:n\in\N\}$, and fix $m\in\N$. If $n\geq m$, then $a_n\in I_n\subseteq I_m$, so $a_n\leq b_m$.\\
If $n<m$, then $b_m\in I_m\subseteq I_n$, so $a_n\leq b_m$. Thus $b_m$ is an upper bound for $A$.\\
Since $A\ne\emptyset$ and $A$ is bounded above, $x\defeq\sup A$ exists by Theorem \ref{thm:real-field-existence-lub}(i).\\
Inequalities $a_m\leq x\leq b_m$ follow from $a_m\in A$ and the upper-bound property of $b_m$. Therefore $(\forall m\in\N)\,x\in I_m$, so $x\in\bigcap\limits_{n=1}^{\infty}I_n$.\\
\hlt{Structural Commentary}: Nestedness makes each right endpoint bound all left endpoints; real completeness supplies their supremum.\\
\hlt{Key Insight}: Every interval contains $\sup\{a_n:n\in\N\}$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:nested-k-cells-intersect} \hlt{Nested $k$-Cells Intersect}\\
Let $k\in\N$, $(I_n)_{n\in\N}$ be sequence of nonempty $k$-cells, where $I_n=\prod\limits_{j=1}^{k}[a_{n,j},b_{n,j}]\subseteq\R^k$. If $(\forall n\in\N)\,I_{n+1}\subseteq I_n$, then $\bigcap\limits_{n=1}^{\infty}I_n\ne\emptyset$.
\begin{proof}
For $n\in\N$ and $1\leq j\leq k$, coordinate projection of $I_{n+1}\subseteq I_n$ establishes $[a_{n+1,j},b_{n+1,j}]\subseteq[a_{n,j},b_{n,j}]$.\\
Choose $x_j^*\in\bigcap\limits_{n=1}^{\infty}[a_{n,j},b_{n,j}]$ for $1\leq j\leq k$ by Theorem \ref{thm:nested-closed-intervals-intersect}. Then $\mathbf{x}^*\defeq(x_1^*,\ldots,x_k^*)$ satisfies $(\forall n\in\N)\,\mathbf{x}^*\in I_n$, so $\mathbf{x}^*\in\bigcap\limits_{n=1}^{\infty}I_n$.\\
\hlt{Structural Commentary}: Coordinate projection reduces a nested cell sequence to $k$ nested interval sequences.\\
\hlt{Key Insight}: One persistent point in each coordinate defines a persistent vector.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:k-cells-compact} \hlt{$k$-Cells Are Compact}\\
Let $k\in\N$, and let $I=\prod\limits_{j=1}^{k}[a_j,b_j]\subseteq\R^k$ be a nonempty $k$-cell. Then $I$ is compact.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Contradictory cover}: Put $\delta\defeq\norm{(b_1-a_1,\ldots,b_k-a_k)}_2$. Suppose an open cover $\mathcal{G}=(G_\alpha)_{\alpha\in A}$ of $I$ has no finite subcover.
\item \hlt{Nested construction}: Set $I_0\defeq I$. Bisecting every coordinate interval covers $I_n$ by $2^k$ nonempty subcells.\\
If each subcell had a finite $\mathcal{G}$-subcover, their finite union would cover $I_n$. Thus some subcell has no finite subcover.\\
Choose $I_{n+1}$ as the lexicographically first such subcell. This recursion defines $I_0\supseteq I_1\supseteq\cdots$.
\item \hlt{Diameter control}: The $j$th side of $I_n$ has length $2^{-n}(b_j-a_j)$. Hence $(\forall\,\mathbf{p},\mathbf{q}\in I_n)\,\norm{\mathbf{p}-\mathbf{q}}_2\leq2^{-n}\delta$, so $\diam I_n\leq2^{-n}\delta$.
\item \hlt{Contradiction}: Choose $\mathbf{x}^*\in\bigcap\limits_{n=1}^{\infty}I_n$ by Theorem \ref{thm:nested-k-cells-intersect}. Choose $\alpha\in A$ with $\mathbf{x}^*\in G_\alpha$, and $r>0$ with $B_r(\mathbf{x}^*)\subseteq G_\alpha$.\\
The bound $(\forall n\in\N)\,2^n\geq n+1$ holds by induction: $2^1=2$, and $2^n\geq n+1$ implies $2^{n+1}\geq2(n+1)\geq n+2$.\\
Choose $n\in\N$ with $n+1>\frac{\delta}{r}$ by Theorem \ref{thm:archimedean-and-rational-density}(i). Then $2^{-n}\delta<r$.\\
Because $\mathbf{x}^*\in I_n$, the diameter bound implies $I_n\subseteq B_r(\mathbf{x}^*)\subseteq G_\alpha$. This one-member subcover contradicts the construction of $I_n$.
\end{enumerate}
\hlt{Structural Commentary}: Recursive bisection combines nested-cell intersection with geometric diameter decay.\\
\hlt{Key Insight}: An open neighborhood of the trapped point eventually contains an entire bad cell.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:bounded-rk-set-contained-in-k-cell} \hlt{Bounded Euclidean Sets Fit in $k$-Cells}\\
Let $k\in\N$ and let $E\subseteq\R^k$ be bounded. Then $E$ is contained in a $k$-cell.
\begin{proof}
If $E=\emptyset$, then $E\subseteq\prod\limits_{j=1}^{k}[-1,1]$. Suppose $E\ne\emptyset$, and choose $\mathbf{q}=(q_1,\ldots,q_k)\in\R^k$ and $M>0$ with $E\subseteq B_M(\mathbf{q})$.\\
Set $I\defeq\prod\limits_{j=1}^{k}[q_j-M,q_j+M]$. If $\mathbf{p}=(p_1,\ldots,p_k)\in E$, then $(\forall j\in\{1,\ldots,k\})\,\abs{p_j-q_j}\leq\norm{\mathbf{p}-\mathbf{q}}_2<M$. Hence $\mathbf{p}\in I$, $E\subseteq I$.\\
\hlt{Structural Commentary}: A containing Euclidean ball transfers its radius bound to every coordinate interval.\\
\hlt{Key Insight}: Euclidean distance dominates each coordinate difference.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:heine-borel-rk} \hlt{Heine--Borel in $\R^k$}\\
Let $k\in\N$ and let $E\subseteq\R^k$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E$ is closed and bounded;
\item $E$ is compact;
\item $(\forall\,S\subseteq E)\,[S\text{ infinite}\Longrightarrow S^{\operatorname{acc}}\cap E\ne\emptyset]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ Choose a $k$-cell $I$ with $E\subseteq I$ by Lemma \ref{lem:bounded-rk-set-contained-in-k-cell}. The set $I$ is compact by Theorem \ref{thm:k-cells-compact}.\\
Since $E$ is closed in $\R^k$, it is compact by Theorem \ref{thm:closed-subsets-of-compact-sets-compact} with $E\subseteq I$.\\
\item[$(ii)$]$\Rightarrow (iii)$ If $S\subseteq E$ is infinite, then $S^{\operatorname{acc}}\cap E\ne\emptyset$ by Theorem \ref{thm:infinite-subset-of-compact-has-limit-point}.\\
\item[$(iii)$]$\Rightarrow (i)$
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Boundedness}: Suppose $E$ is unbounded. Choose $(\mathbf{x}_n)\subseteq E$ with $(\forall n\in\N)\,\norm{\mathbf{x}_n}_2>n$, $S\defeq\{\mathbf{x}_n:n\in\N\}$.\\
The set $S$ is infinite because a finite subset of $\R^k$ has bounded norms. If $\mathbf{x}_n\in B_1(\mathbf{y})$, then $n<\norm{\mathbf{x}_n}_2<\norm{\mathbf{y}}_2+1$.\\
Thus $(\forall\,\mathbf{y}\in\R^k)\,[S\cap B_1(\mathbf{y})\text{ is finite}]$. Hence $S^{\operatorname{acc}}=\emptyset$ by contrapositive of Theorem \ref{thm:limit-point-neighborhoods-infinitely-many}.\\
This contradicts (iii), and hence $E$ is bounded.
\item \hlt{Closedness}: Suppose $E$ is not closed, and choose $\mathbf{x}_0\in\overline{E}\setminus E$. Then $\mathbf{x}_0\in E^{\operatorname{acc}}$ by Definition \ref{def:metric-closure}.\\
Choose $(\mathbf{x}_n)\subseteq E\setminus\{\mathbf{x}_0\}$ with $\mathbf{x}_n\to\mathbf{x}_0$ by Theorem \ref{thm:closed-sets-and-limit-points-via-sequences}(i).\\
Set $S\defeq\{\mathbf{x}_n:n\in\N\}$. If $S$ were finite, then $\rho\defeq\min\limits_{\mathbf{x}\in S}\norm{\mathbf{x}-\mathbf{x}_0}_2>0$.\\
Since $(\forall n\in\N)\,\mathbf{x}_n\in S$, this contradicts $\mathbf{x}_n\to\mathbf{x}_0$; hence $S$ is infinite. Fix $\mathbf{y}\ne\mathbf{x}_0$, and put $r\defeq\frac{1}{3}\norm{\mathbf{y}-\mathbf{x}_0}_2$.\\
The balls $B_r(\mathbf{y})$ and $B_r(\mathbf{x}_0)$ are disjoint, while all but finitely many $\mathbf{x}_n$ lie in $B_r(\mathbf{x}_0)$.\\
Hence $B_r(\mathbf{y})\cap S$ is finite, so $\mathbf{y}\notin S^{\operatorname{acc}}$ by the contrapositive of Theorem \ref{thm:limit-point-neighborhoods-infinitely-many}.\\
Therefore $S^{\operatorname{acc}}\subseteq\{\mathbf{x}_0\}$ and $S^{\operatorname{acc}}\cap E=\emptyset$, contradicting (iii); hence $E$ is closed.
\end{enumerate}
\end{enumerate}
\hlt{Structural Commentary}: Compact cells prove sufficiency; escaping and boundary-convergent sequences prove necessity.\\
\hlt{Key Insight}: Failure of either bound or closure creates an infinite set without internal accumulation.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:weierstrass-limit-point-rk} \hlt{Weierstrass Limit-Point Theorem}\\
Let $k\in\N$ and let $E\subseteq\R^k$ be bounded and infinite. Then $E^{\operatorname{acc}}\ne\emptyset$.
\begin{proof}
Choose a $k$-cell $I$ with $E\subseteq I$ by Lemma \ref{lem:bounded-rk-set-contained-in-k-cell}. Set $I$ is compact by Theorem \ref{thm:k-cells-compact}, so $E^{\operatorname{acc}}\cap I\ne\emptyset$ by Theorem \ref{thm:infinite-subset-of-compact-has-limit-point}.\\
\hlt{Structural Commentary}: Cell containment transfers the abstract compact accumulation theorem to bounded Euclidean sets.\\
\hlt{Key Insight}: A bounded infinite set is an infinite subset of one compact cell.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:bounded-euclidean-sequences-convergent-subsequence} \hlt{Bounded Euclidean Sequences Have Convergent Subsequences}\\
Let $k\in\N$ and let $(\mathbf{x}_n)$ be a bounded sequence in $\R^k$. Then $(\mathbf{x}_n)$ has a convergent subsequence.
\begin{proof}
Set $R\defeq\{\mathbf{x}_n:n\in\N\}$. Choose a $k$-cell $I$ with $R\subseteq I$ by Lemma \ref{lem:bounded-rk-set-contained-in-k-cell}.\\
The set $I$ is compact by Theorem \ref{thm:k-cells-compact}; hence $(\mathbf{x}_n)$ has a convergent subsequence by Theorem \ref{thm:compact-sequential-subsequence}.\\
\hlt{Structural Commentary}: Range containment moves the sequence into a compact cell before subsequence extraction.\\
\hlt{Key Insight}: One compact cell contains every term.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:euclidean-cauchy-convergence} \hlt{Euclidean Cauchy Convergence}\\
Let $k\in\N$ and let $(\mathbf{x}_n)$ be a Cauchy sequence in $\R^k$. Then $(\mathbf{x}_n)$ converges.
\begin{proof}
The range of $(\mathbf{x}_n)$ is bounded by Proposition \ref{prop:cauchy-sequences-bounded-distance-converges}(i). Thus $(\mathbf{x}_n)$ has a convergent subsequence by Corollary \ref{cor:bounded-euclidean-sequences-convergent-subsequence}.\\
Therefore $(\mathbf{x}_n)$ converges by Lemma \ref{lem:cauchy-sequence-with-convergent-subsequence}.\\
\hlt{Structural Commentary}: Cauchy boundedness supplies a convergent subsequence, whose limit controls the full sequence.\\
\hlt{Key Insight}: A Cauchy sequence cannot diverge from a convergent subsequence.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:heine-borel-rk-sequential-form} \hlt{Euclidean Compactness Tests}\\
Let $k\in\N$ and let $E\subseteq\R^k$, with the subspace metric. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E$ is closed and bounded;
\item $E$ is compact;
\item $(\forall\,(\mathbf{x}_n)\subseteq E)(\exists\,n_1<n_2<\cdots)(\exists\,\mathbf{x}\in E)\,\mathbf{x}_{n_j}\to\mathbf{x}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ The set $E$ is compact by Theorem \ref{thm:heine-borel-rk}, direction $(i)\Rightarrow(ii)$.
\item[$(ii)$]$\Rightarrow (iii)$ Condition (iii) holds by Theorem \ref{thm:compact-sequential-subsequence}.
\item[$(iii)$]$\Rightarrow (i)$ The metric space $E$ is sequentially compact by (iii). Hence $E$ is compact by Theorem \ref{thm:compact-complete-totally-bounded}, direction $(iii)\Rightarrow(i)$.\\
Therefore $E$ is closed and bounded by Theorem \ref{thm:heine-borel-rk}, direction $(ii)\Rightarrow(i)$.
\end{enumerate}
\hlt{Structural Commentary}: Heine--Borel and metric sequential compactness connect geometric, cover, and subsequence criteria.\\
\hlt{Key Insight}: In Euclidean space, closed bounded geometry is exactly sequential compactness.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:heine-borel-real-line-sequential-form} \hlt{Heine--Borel on the Real Line}\\
Let $E\subseteq\R$, with the usual metric. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E$ is closed and bounded;
\item $E$ is compact;
\item $(\forall\,(x_n)\subseteq E)(\exists\,n_1<n_2<\cdots)(\exists\,x\in E)\,x_{n_j}\to x$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ The set $E$ is compact by Corollary \ref{cor:heine-borel-rk-sequential-form}, direction $(i)\Rightarrow(ii)$, with $k=1$.
\item[$(ii)$]$\Rightarrow (iii)$ Condition (iii) holds by Corollary \ref{cor:heine-borel-rk-sequential-form}, direction $(ii)\Rightarrow(iii)$, with $k=1$.
\item[$(iii)$]$\Rightarrow (i)$ The set $E$ is closed and bounded by Corollary \ref{cor:heine-borel-rk-sequential-form}, direction $(iii)\Rightarrow(i)$, with $k=1$.
\end{enumerate}
\hlt{Structural Commentary}: Specializing the Euclidean compactness cycle to $k=1$ transfers all three implications.\\
\hlt{Key Insight}: Real-line compactness is the one-dimensional Euclidean case.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:nested-closed-bounded-real-sets-intersect} \hlt{Nested Closed Bounded Real Sets Intersect}\\
Let $(F_n)_{n\in\N}$ be nonempty closed subsets of $\R$ with $(\forall n\in\N)\,F_{n+1}\subseteq F_n$. If $F_1$ is bounded, then $\bigcap\limits_{n=1}^{\infty}F_n\ne\emptyset$.
\begin{proof}
The set $F_1$ is compact by Corollary \ref{cor:heine-borel-real-line-sequential-form}, direction $(i)\Rightarrow(ii)$.\\
For $n\in\N$, $F_n\subseteq F_1$ by nestedness, so $F_n$ is compact by Theorem \ref{thm:closed-subsets-of-compact-sets-compact}. Hence $\bigcap\limits_{n=1}^{\infty}F_n\ne\emptyset$ by Corollary \ref{cor:nested-nonempty-compact-intersection}.\\
\hlt{Structural Commentary}: Heine--Borel makes $F_1$ compact; closed-subset compactness transfers this to the nested tail.\\
\hlt{Key Insight}: Nested nonempty compact sets have the finite intersection property.
\end{proof}
\end{corollary}
