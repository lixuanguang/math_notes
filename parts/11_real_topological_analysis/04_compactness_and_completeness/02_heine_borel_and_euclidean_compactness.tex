\subsection{Heine--Borel and Euclidean Compactness}

\begin{theorem} \label{thm:nested-closed-intervals-intersect} \hlt{Nested Closed Intervals Intersect}\\
Let $(I_n)_{n\in\N}$ be nonempty compact intervals in $\R$, with $I_n=[a_n,b_n]$, such that $I_{n+1}\subseteq I_n$ $\forall n$. Then $\bigcap_{n=1}^{\infty}I_n\neq\emptyset$.
\begin{proof} Set $A\defeq\{a_n:n\in\N\}$. Fix $m\in\N$. If $n\geq m$, then $a_n\in I_n\subseteq I_m$, so $a_n\leq b_m$.\\
If $n<m$, then $b_m\in I_m\subseteq I_n$, so $a_n\leq b_m$. Hence $b_m$ is an upper bound for $A$.\\
Let $x\defeq\sup A$. Since $a_m\in A$, $a_m\leq x$; since $b_m$ is an upper bound for $A$, $x\leq b_m$.\\
Hence $x\in I_m$ for every $m$, and therefore $x\in\bigcap_{n=1}^{\infty}I_n$.\\
\hlt{Structural Commentary}: Nested endpoints form a LUB trap.\\
\hlt{Key Insight}: $\sup_n a_n$ lies in every interval.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:nested-k-cells-intersect} \hlt{Nested $k$-Cells Intersect}\\
Let $k\in\N$, and for each $n\in\N$ let $I_n=\prod_{j=1}^{k}[a_{n,j},b_{n,j}]\subseteq\R^k$ be a nonempty $k$-cell.\\
Assume $I_{n+1}\subseteq I_n$ for every $n$. Then $\bigcap_{n=1}^{\infty}I_n\neq\emptyset$.
\begin{proof} For each $j$, the $j$th coordinate projection of $I_{n+1}\subseteq I_n$ gives $[a_{n+1,j},b_{n+1,j}]\subseteq[a_{n,j},b_{n,j}]$.\\
By Theorem \ref{thm:nested-closed-intervals-intersect}, choose $x_j^*\in\bigcap_{n=1}^{\infty}[a_{n,j},b_{n,j}]$. Then $\mathbf{x}^*\defeq(x_1^*,\ldots,x_k^*)$ satisfies $\mathbf{x}^*\in I_n$ $\forall n$, so $\mathbf{x}^*\in\bigcap_{n=1}^{\infty}I_n$.\\
\hlt{Structural Commentary}: Cells reduce coordinatewise to intervals.\\
\hlt{Key Insight}: Pick one persistent coordinate per axis.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:k-cells-compact} \hlt{$k$-Cells Are Compact}\\
Let $k\in\N$, and let $I=\prod_{j=1}^{k}[a_j,b_j]\subseteq\R^k$ be a nonempty $k$-cell. Then $I$ is compact.
\begin{proof} Set $\delta\defeq\left(\sum_{j=1}^{k}(b_j-a_j)^2\right)^{1/2}=\diam I$. Suppose open cover $\mathcal{G}=\{G_\alpha\}_{\alpha\in A}$ of $I$ has no finite subcover.\\
Bisect each coordinate interval of $I$. If every resulting subcell had a finite $\mathcal{G}$-subcover, their union would finitely cover $I$, so choose a subcell $I_1$ with no finite $\mathcal{G}$-subcover. Repeating inside $I_{n-1}$ gives nested $k$-cells $I\supseteq I_1\supseteq I_2\supseteq\cdots$ such that $I_n$ has no finite $\mathcal{G}$-subcover and $\diam I_n\leq2^{-n}\delta$.\\
By Theorem \ref{thm:nested-k-cells-intersect}, choose $\mathbf{x}^*\in\bigcap_{n=1}^{\infty}I_n$. Since $\mathcal{G}$ covers $I$, choose $\alpha$ with $\mathbf{x}^*\in G_\alpha$. Openness gives $r>0$ such that $B_r(\mathbf{x}^*)\subseteq G_\alpha$. Choose $n$ with $2^{-n}\delta<r$. Since $\mathbf{x}^*\in I_n$ and $\diam I_n<r$, one has $I_n\subseteq B_r(\mathbf{x}^*)\subseteq G_\alpha$, contradicting the choice of $I_n$. Thus every open cover of $I$ has a finite subcover.\\
\hlt{Structural Commentary}: Bisection gives shrinking nested cells.\\
\hlt{Key Insight}: One open set eventually contains the trapped cell.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:bounded-rk-set-contained-in-k-cell} \hlt{Bounded Euclidean Sets Fit in $k$-Cells}\\
Let $k\in\N$ and let $E\subseteq\R^k$ be bounded. Then there exists a $k$-cell $I\subseteq\R^k$ such that $E\subseteq I$.
\begin{proof} If $E=\emptyset$, take $I=\prod_{j=1}^{k}[0,0]$. Assume $E\neq\emptyset$.\\
Choose $\mathbf{q}=(q_1,\ldots,q_k)\in\R^k$ and $M>0$ such that $E\subseteq B_M(\mathbf{q})$. Set $I\defeq\prod_{j=1}^{k}[q_j-M,q_j+M]$.\\
For $\mathbf{p}=(p_1,\ldots,p_k)\in E$ and $1\leq j\leq k$, $\abs{p_j-q_j}\leq\norm{\mathbf{p}-\mathbf{q}}_2<M$, hence $\mathbf{p}\in I$. Thus $E\subseteq I$.\\
\hlt{Structural Commentary}: One ball fits inside one coordinate box.\\
\hlt{Key Insight}: A radius gives coordinate bounds.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:heine-borel-rk} \hlt{Heine--Borel in $\R^k$}\\
Let $k\in\N$ and let $E\subseteq\R^k$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E$ is closed and bounded;
\item $E$ is compact;
\item every infinite subset of $E$ has a limit point in $E$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ Since $E$ is bounded, Lemma \ref{lem:bounded-rk-set-contained-in-k-cell} gives a $k$-cell $I$ with $E\subseteq I$. By Theorem \ref{thm:k-cells-compact}, $I$ is compact. Since $E$ is closed in $\R^k$, $E$ is closed in $I$, so Theorem \ref{thm:closed-subsets-of-compact-sets-compact} implies that $E$ is compact.
\item[$(ii)$]$\Rightarrow (iii)$ Theorem \ref{thm:infinite-subset-of-compact-has-limit-point} applies directly to the compact set $E$.
\item[$(iii)$]$\Rightarrow (i)$ Suppose first that $E$ is unbounded. Choose $\mathbf{x}_n\in E$ with $\norm{\mathbf{x}_n}_2>n$ and set $S\defeq\{\mathbf{x}_n:n\in\N\}$. For any $\mathbf{y}\in\R^k$, the set $S\cap B_1(\mathbf{y})$ is finite because $\mathbf{x}_n\in B_1(\mathbf{y})$ implies $n<\norm{\mathbf{y}}_2+1$. Hence $S$ has no limit point, contradicting (iii). Thus $E$ is bounded. Suppose next that $E$ is not closed. Choose $\mathbf{x}_0\in\overline{E}\setminus E$ and distinct $\mathbf{x}_n\in E$ with $0<\norm{\mathbf{x}_n-\mathbf{x}_0}_2<1/n$. Then $S\defeq\{\mathbf{x}_n:n\in\N\}$ is infinite, and every limit point of $S$ equals $\mathbf{x}_0\notin E$, contradicting (iii). Thus $E$ is closed.
\end{enumerate}
\hlt{Structural Commentary}: Compactness equals closed boundedness in $\R^k$.\\
\hlt{Key Insight}: Cells convert bounded geometry into compactness.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:weierstrass-limit-point-rk} \hlt{Weierstrass Limit-Point Theorem}\\
Let $k\in\N$ and let $E\subseteq\R^k$ be bounded and infinite. Then $E$ has a limit point in $\R^k$.
\begin{proof} By Lemma \ref{lem:bounded-rk-set-contained-in-k-cell}, choose a $k$-cell $I$ with $E\subseteq I$. By Theorem \ref{thm:k-cells-compact}, $I$ is compact.\\
Since $E$ is an infinite subset of $I$, Theorem \ref{thm:infinite-subset-of-compact-has-limit-point} gives a limit point of $E$ in $I\subseteq\R^k$.\\
\hlt{Structural Commentary}: Bounded sets sit in compact cells.\\
\hlt{Key Insight}: Infinite compact subsets accumulate.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:bounded-euclidean-sequences-convergent-subsequence} \hlt{Bounded Euclidean Sequences Have Convergent Subsequences}\\
If $(\mathbf{x}_n)$ is bounded in $\R^k$, then it has a convergent subsequence.
\begin{proof}
The range of $(\mathbf{x}_n)$ has compact closure by Theorem \ref{thm:heine-borel-rk}; apply Theorem \ref{thm:compact-sequential-subsequence} in that closure.\\
\hlt{Structural Commentary}: Heine--Borel makes the range closure compact.\\
\hlt{Key Insight}: Bounded Euclidean sequences live inside compact closed sets.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:euclidean-cauchy-convergence} \hlt{Euclidean Cauchy Convergence}\\
In $\R^k$, every Cauchy sequence converges.
\begin{proof}
For Cauchy $(\mathbf{x}_n)\subseteq\R^k$, choose $N$ with $\diam\{\mathbf{x}_j:j\geq N\}<1$.\\
Its range is bounded, hence has compact closure by Theorem \ref{thm:heine-borel-rk}; apply Theorem \ref{thm:cauchy-convergence-criterion}(ii).\\
\hlt{Structural Commentary}: A Cauchy tail has bounded compact closure.\\
\hlt{Key Insight}: In $\R^k$, Cauchy convergence follows from Heine--Borel.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:heine-borel-rk-sequential-form} \hlt{Euclidean Compactness Tests}\\
Let $k\in\N$ and let $E\subseteq\R^k$, with the subspace metric. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E$ is closed and bounded;
\item $E$ is compact;
\item every sequence in $E$ has a subsequence converging to a point of $E$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ This is Theorem \ref{thm:heine-borel-rk}.
\item[$(ii)$]$\Rightarrow (iii)$ Let $(x_n)_{n\in\N}$ be a sequence in $E$.\\
Since $E$ is compact, Theorem \ref{thm:compact-complete-totally-bounded} gives indices $n_1<n_2<\cdots$ and $x\in E$ such that $x_{n_j}\to x$.
\item[$(iii)$]$\Rightarrow (i)$ By (iii), $E$ is sequentially compact. Theorem \ref{thm:compact-complete-totally-bounded}, applied to $X=E$, gives that $E$ is compact.\\
Then Theorem \ref{thm:heine-borel-rk} implies that $E$ is closed and bounded.
\end{enumerate}
\hlt{Structural Commentary}: Euclidean compactness has three tests.\\
\hlt{Key Insight}: Heine--Borel links geometry to sequences.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:heine-borel-real-line-sequential-form} \hlt{Heine--Borel on the Real Line}\\
Let $E\subseteq\R$, with the usual metric. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E$ is closed and bounded;
\item $E$ is compact;
\item every sequence in $E$ has a subsequence converging to a point of $E$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ By (i), $E$ is closed, bounded as a subset of $\R^1$. Theorem \ref{thm:heine-borel-rk} with $k=1$ gives that $E$ is compact.
\item[$(ii)$]$\Rightarrow (iii)$ Let $(x_n)_{n\in\N}$ be a sequence in $E$.\\
By compactness of $E$, Theorem \ref{thm:compact-complete-totally-bounded} gives indices $n_1<n_2<\cdots$ and $x\in E$ such that $x_{n_j}\to x$.
\item[$(iii)$]$\Rightarrow (i)$ Let $F\subseteq E$ be infinite, and choose distinct $x_n\in F$. By (iii), some subsequence satisfies $x_{n_j}\to x\in E$. Since the $x_n$ are distinct, $x$ is a limit point of $F$. Thus every infinite subset of $E$ has a limit point in $E$, so Theorem \ref{thm:heine-borel-rk} with $k=1$ gives that $E$ is closed and bounded.
\end{enumerate}
\hlt{Structural Commentary}: On $\R$, compactness is sequential.\\
\hlt{Key Insight}: Closed bounded sets trap subsequences.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:nested-closed-bounded-real-sets-intersect} \hlt{Nested Closed Bounded Real Sets Intersect}\\
Let $(F_n)_{n\in\N}$ be nonempty closed subsets of $\R$ with $F_{n+1}\subseteq F_n$ for all $n$. If $F_1$ is bounded, then $\bigcap_{n=1}^{\infty}F_n\neq\emptyset$.
\begin{proof}
By Corollary \ref{cor:heine-borel-real-line-sequential-form}, $F_1$ is compact. For each $n$, $F_n$ is a nonempty closed subset of the compact set $F_1$, so Theorem \ref{thm:closed-subsets-of-compact-sets-compact} gives that $F_n$ is compact. Since $F_{n+1}\subseteq F_n$, Corollary \ref{cor:nested-nonempty-compact-intersection} gives $\bigcap_{n=1}^{\infty}F_n\neq\emptyset$.\\
\hlt{Structural Commentary}: Nested compact sets intersect.\\
\hlt{Key Insight}: Heine--Borel supplies compactness.
\end{proof}
\end{corollary}
