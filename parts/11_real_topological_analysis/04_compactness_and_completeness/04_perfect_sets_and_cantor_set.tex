\subsection{Perfect Sets and the Cantor Set}

\begin{theorem} \label{thm:nonempty-perfect-subset-rk-uncountable} \hlt{Perfect Sets Are Uncountable}\\
Every nonempty perfect subset of $\R^k$ is uncountable.
\begin{proof}
Assume $P=\{\mathbf{x}_1,\mathbf{x}_2,\ldots\}$. Choose a neighborhood $V_1$ with $V_1\cap P\neq\emptyset$.\\
Inductively, since $P\cap V_n$ contains infinitely many points by Theorem \ref{thm:limit-point-neighborhoods-infinitely-many}, choose $\mathbf{y}_n\in P\cap V_n$ with $\mathbf{y}_n\neq\mathbf{x}_n$ and then a neighborhood $V_{n+1}$ of $\mathbf{y}_n$ such that $\overline{V}_{n+1}\subseteq V_n$ and $\mathbf{x}_n\notin\overline{V}_{n+1}$.\\
Put $K_n\defeq\overline{V}_n\cap P$. Then $K_n$ is nonempty, compact, and $K_{n+1}\subseteq K_n$. By Corollary \ref{cor:nested-nonempty-compact-intersection}, choose $\mathbf{z}\in\bigcap\limits_{n=1}^{\infty}K_n$. For each $n$, $\mathbf{z}\in K_{n+1}$ but $\mathbf{x}_n\notin K_{n+1}$, so $\mathbf{z}\neq\mathbf{x}_n$ for all $n$, contradicting the enumeration.\\
\hlt{Structural Commentary}: Nested compact sets dodge the list.\\
\hlt{Key Insight}: Perfectness gives new points; compactness keeps one.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:closed-intervals-and-reals-uncountable} \hlt{Nondegenerate Intervals and the Real Line Are Uncountable}\\
Let $J\subseteq\R$ be an interval.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $J$ contains two distinct points, then $J$ is uncountable.
\item $\R$ is uncountable.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $u<v$ in $J$ and then $a,b$ with $u<a<b<v$. Since $J$ is an interval, $I\defeq[a,b]\subseteq J$.\\
The set $I$ is a nonempty $1$-cell, hence compact by Theorem \ref{thm:k-cells-compact} and closed by Theorem \ref{thm:compact-subsets-of-metric-spaces-closed}.\\
For every $x\in I$ and $\varepsilon>0$, $I\cap(B_\varepsilon(x)\setminus\{x\})\ne\emptyset$, so $I$ is perfect.\\
Thus $I$ is uncountable by Theorem \ref{thm:nonempty-perfect-subset-rk-uncountable}; hence $J$ is uncountable.
\item Since $\R$ is an interval and $0,1\in\R$ with $0<1$, (i) applied to $J=\R$ gives that $\R$ is uncountable.
\end{enumerate}
\hlt{Structural Commentary}: Closed subintervals are perfect compact subsets.\\
\hlt{Key Insight}: One closed subinterval already forces uncountability.
\end{proof}
\end{corollary}

\begin{example} \label{ex:cantor-set} \hlt{The Cantor Set}\\
Set $E_0=[0,1]$ and obtain $E_{n+1}$ from $E_n$ by deleting the open middle third of each component.\\
Then $E_n$ is a union of $2^n$ compact intervals of length $3^{-n}$ and $E_{n+1}\subseteq E_n$.\\
The \hlt{Cantor set} is $P\defeq\bigcap_{n=0}^{\infty}E_n$. Since $(E_n)$ is nested nonempty compact, $P\ne\emptyset$ by Corollary \ref{cor:nested-nonempty-compact-intersection}.\\
Since $P$ is closed in $E_0$, $P$ is compact. Every nonempty open interval $J\subseteq[0,1]$ contains a deleted middle third $R$ with $R\cap P=\emptyset$, so $P$ contains no nontrivial interval.If $x\in P$ and $U$ is an interval neighborhood of $x$, choose $n$ with the component $I_n(x)\subseteq U$; an endpoint $y\neq x$ of $I_n(x)$ lies in $P$, so $U\cap(P\setminus\{x\})\neq\emptyset$. Hence $P$ is perfect.\\
\hlt{Structural Commentary}: The Cantor set is built by nested compact deletion, not by listing points.\\
\hlt{Key Insight}: It is compact and perfect, but contains no interval.
\end{example}
