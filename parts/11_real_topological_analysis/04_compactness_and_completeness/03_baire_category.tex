\subsection{Baire Category}

\begin{definition} \label{def:fsigma-gdelta-nowhere-dense} \hlt{$F_\sigma$, $G_\delta$, and Nowhere-Dense Sets}\\
Let $(X,d)$ be a metric space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A set $A\subseteq X$ is an \hlt{$F_\sigma$ set} iff $A=\bigcup\limits_{n=1}^{\infty}F_n$ for a sequence $(F_n)$ of closed subsets of $X$.
\item A set $B\subseteq X$ is a \hlt{$G_\delta$ set} iff $B=\bigcap\limits_{n=1}^{\infty}G_n$ for a sequence $(G_n)$ of open subsets of $X$.
\item A set $E\subseteq X$ is \hlt{nowhere dense} iff $(\overline{E})^\circ=\emptyset$; equivalently, no nonempty open set is contained in $\overline{E}$.
\end{enumerate}
For $X=\R$, this says exactly that $\overline{E}$ contains no nonempty open interval.
\end{definition}

\begin{proposition} \label{prop:open-real-sets-are-fsigma} \hlt{Open Real Sets Are $F_\sigma$}\\
Let $G,A\subseteq\R$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $G$ is open, then $G$ is $F_\sigma$.
\item If $A$ is closed, then $A$ is $G_\delta$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Whole-line case}: If $G=\R$, then $G=\bigcup\limits_{m=1}^{\infty}[-m,m]$ by Theorem \ref{thm:archimedean-and-rational-density}(i). Each interval closed by Corollary \ref{cor:closed-balls-are-closed}.
\item \hlt{Distance control}: Suppose $G\ne\R$, and set $C\defeq\R\setminus G\ne\emptyset$.\\
For $m\in\N$, define $D_m\defeq\{x\in\R:\operatorname{dist}(x,C)\geq\frac{1}{m}\}$ and $F_m\defeq[-m,m]\cap D_m$.\\
For $x,y\in\R$ and $c\in C$, $\abs{x-c}\leq\abs{x-y}+\abs{y-c}$ by the triangle inequality.\\
Taking the infimum over $c\in C$, $\operatorname{dist}(x,C)\leq\abs{x-y}+\operatorname{dist}(y,C)$ by Definition \ref{def:set-distance-in-metric-space}(i).\\
After interchanging $x$ and $y$, $\abs{\operatorname{dist}(x,C)-\operatorname{dist}(y,C)}\leq\abs{x-y}$.
\item \hlt{Closed approximants}: If $x\notin D_m$, put $\eta\defeq\frac{1}{m}-\operatorname{dist}(x,C)>0$. Then $y\in B_\eta(x)$ implies $\operatorname{dist}(y,C)<\frac{1}{m}$, so $B_\eta(x)\subseteq D_m^c$.\\
Thus $D_m$ is closed by Corollary \ref{cor:closed-iff-complement-open}. The interval $[-m,m]$ is closed, so $F_m$ is closed by Theorem \ref{thm:metric-open-closed-set-calculus}(ii).
\item \hlt{Exhaustion}: If $x\in C$, then $\operatorname{dist}(x,C)=0$; hence $F_m\cap C=\emptyset$ and $F_m\subseteq G$. Fix $x\in G$, $r>0$ with $(x-r,x+r)\subseteq G$.\\
Then $(\forall c\in C)\,\abs{x-c}\geq r$, so $\operatorname{dist}(x,C)\geq r$. Choose $m\in\N$, $m\geq\abs{x}$, $\frac{1}{m}\leq r$ by Theorem \ref{thm:archimedean-and-rational-density}(i); $x\in F_m$, so $G=\bigcup\limits_{m=1}^{\infty}F_m$.
\end{enumerate}
\item If $A$ is closed, then $A^c=\bigcup\limits_{m=1}^{\infty}F_m$ for a sequence $(F_m)$ of closed sets by (i). Therefore $A=\bigcap\limits_{m=1}^{\infty}F_m^c$ is $G_\delta$.
\end{enumerate}
\hlt{Structural Commentary}: Distance from the complement constructs a closed exhaustion; complementation transfers it to closed sets.\\
\hlt{Key Insight}: Positive distance converts local openness into closed approximants.
\end{proof}
\end{proposition}

\begin{definition} \label{def:first-category-set} \hlt{First Category}\\
Let $(X,d)$ be metric space, $E\subseteq X$. Set $E$ is \hlt{meager} in $X$ iff $E\subseteq\bigcup\limits_{n=1}^{\infty}N_n$ for sequence $(N_n)$ of nowhere-dense subsets of $X$.\\
The terms \hlt{of first category} and \hlt{thin} are synonymous. The ambient space $X$ is part of the property.
\end{definition}

\begin{theorem} \label{thm:dense-open-baire-real-line} \hlt{Dense Open Intersections in $\R$}\\
Let $(G_n)_{n\in\N}$ be dense open subsets of $\R$. Then $\bigcap\limits_{n=1}^{\infty}G_n\ne\emptyset$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Initialization}: Choose $x_1\in G_1\cap(0,1)$ by Proposition \ref{prop:dense-closure-and-interval-test}. By openness, choose $s_1>0$ with $B_{s_1}(x_1)\subseteq G_1\cap(0,1)$.\\
Set $\rho_1\defeq\min\{\frac{s_1}{2},\frac{1}{4}\}$ and $I_1\defeq[x_1-\rho_1,x_1+\rho_1]$. Then $I_1\subseteq G_1\cap(0,1)$ and $I_1^\circ\ne\emptyset$.
\item \hlt{Recursion}: Suppose $I_n=[a_n,b_n]\subseteq G_n$ and $a_n<b_n$. Choose $x_{n+1}\in G_{n+1}\cap(a_n,b_n)$ by Proposition \ref{prop:dense-closure-and-interval-test}.\\
By openness, choose $s_{n+1}>0$ with $B_{s_{n+1}}(x_{n+1})\subseteq G_{n+1}\cap(a_n,b_n)$.\\
Put $\rho_{n+1}\defeq\frac{s_{n+1}}{2}$ and $I_{n+1}\defeq[x_{n+1}-\rho_{n+1},x_{n+1}+\rho_{n+1}]$. Then $I_{n+1}\subseteq I_n\cap G_{n+1}$ and $I_{n+1}^\circ\ne\emptyset$.
\item \hlt{Intersection}: Choose $x\in\bigcap\limits_{n=1}^{\infty}I_n$ by Theorem \ref{thm:nested-closed-intervals-intersect}. Since $(\forall n\in\N)\,I_n\subseteq G_n$, one has $x\in\bigcap\limits_{n=1}^{\infty}G_n$.
\end{enumerate}
\hlt{Structural Commentary}: Dense openness constructs nested intervals; real completeness supplies their common point.\\
\hlt{Key Insight}: Each interval can be trapped inside the next dense open set.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:dense-open-baire-complete-metric-space} \hlt{Dense Open Intersections in Complete Metric Spaces}\\
Let $(X,d)$ be complete, and let $(O_n)_{n\in\N}$ be dense open subsets of $X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\emptyset\ne U\subseteq X$ is open, then $U\cap\bigcap\limits_{n=1}^{\infty}O_n\ne\emptyset$.
\item The set $\bigcap\limits_{n=1}^{\infty}O_n$ is dense in $X$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Initialization}: Fix a nonempty open set $U\subseteq X$. Choose $x_1\in U\cap O_1$ by density of $O_1$.\\
By openness, choose $s_1>0$, $B_{s_1}(x_1)\subseteq U\cap O_1$. Set $r_1\defeq\min\{\frac{s_1}{2},\frac{1}{4}\}$, $C_1\defeq\overline{B}_{r_1}(x_1)$; $C_1\subseteq U\cap O_1$ and $0<r_1\leq\frac{1}{2}$.
\item \hlt{Recursion}: Suppose $C_n=\overline{B}_{r_n}(x_n)\subseteq U\cap\bigcap\limits_{j=1}^{n}O_j$, $0<r_n\leq\frac{1}{2n}$. Since $O_{n+1}$ is dense, choose $x_{n+1}\in B_{r_n}(x_n)\cap O_{n+1}$.\\
By openness, choose $s_{n+1}>0$ with $B_{s_{n+1}}(x_{n+1})\subseteq B_{r_n}(x_n)\cap O_{n+1}$.\\
Set $r_{n+1}\defeq\min\{\frac{s_{n+1}}{2},\frac{1}{2(n+1)}\}$ and $C_{n+1}\defeq\overline{B}_{r_{n+1}}(x_{n+1})$. Then $C_{n+1}\subseteq C_n\cap O_{n+1}$ and $0<r_{n+1}\leq\frac{1}{2(n+1)}$.
\item \hlt{Diameter decay}: Each $C_n$ is nonempty and closed by Corollary \ref{cor:closed-balls-are-closed}. For $p,q\in C_n$, $d(p,q)\leq2r_n\leq\frac{1}{n}$, so $\diam C_n\leq\frac{1}{n}$.\\
Given $\varepsilon>0$, choose $N\in\N$ with $N>\frac{1}{\varepsilon}$ by Theorem \ref{thm:archimedean-and-rational-density}(i). Then $(\forall n\geq N)\,\diam C_n\leq\frac{1}{n}<\varepsilon$, so $\diam C_n\to0$.
\item \hlt{Intersection}: Choose $x\in X$ with $\bigcap\limits_{n=1}^{\infty}C_n=\{x\}$ by Theorem \ref{thm:cantor-intersection-complete-metric-space}, direction $(\Rightarrow)$. As $C_n\subseteq U\cap O_n$, $x\in U\cap\bigcap\limits_{n=1}^{\infty}O_n$.
\end{enumerate}
\item Set $D\defeq\bigcap\limits_{n=1}^{\infty}O_n$. If $p\in X\setminus D$ and $r>0$, then the nonempty open ball $B_r(p)$ meets $D$ by (i).\\
Hence $B_r(p)\cap(D\setminus\{p\})\ne\emptyset$, so $p\in D^{\operatorname{acc}}$. Thus $X=D\cup D^{\operatorname{acc}}=\overline{D}$, and $D$ is dense by Proposition \ref{prop:dense-closure-criterion}.
\end{enumerate}
\hlt{Structural Commentary}: Dense openness builds contracting closed balls; Cantor intersection supplies the common point.\\
\hlt{Key Insight}: Completeness prevents a nested sequence of shrinking admissible balls from vanishing.
\end{proof}
\end{theorem}

\begin{definition} \label{def:residual-thick-generic-set} \hlt{Residual, Thick, and Generic Sets}\\
Let $(X,d)$ be a metric space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A set $G\subseteq X$ is \hlt{residual} in $X$ iff it contains a countable intersection of dense open subsets of $X$. Term \hlt{thick} is synonymous.
\item A property $P$ is \hlt{generic} in $X$ iff $\{x\in X:P(x)\}$ is residual in $X$.
\end{enumerate}
Thus meager sets are topologically small, whereas residual sets contain a countable dense-open core.
\end{definition}

\begin{corollary} \label{cor:residual-subsets-of-complete-metric-spaces-dense} \hlt{Residual Subsets of Complete Metric Spaces Are Dense}\\
Let $(X,d)$ be complete, and let $G\subseteq X$ be residual. Then $G$ is dense in $X$.
\begin{proof}
Choose dense open sets $(O_n)$ in $X$ with $\bigcap\limits_{n=1}^{\infty}O_n\subseteq G$ by Definition \ref{def:residual-thick-generic-set}(i).\\
Set $D\defeq\bigcap\limits_{n=1}^{\infty}O_n$. The set $D$ is dense by Theorem \ref{thm:dense-open-baire-complete-metric-space}(ii), so $\overline{D}=X$ by Proposition \ref{prop:dense-closure-criterion}.\\
Since $D\subseteq G$, one has $X=\overline{D}\subseteq\overline{G}\subseteq X$; hence $G$ is dense by Proposition \ref{prop:dense-closure-criterion}.\\
\hlt{Structural Commentary}: Baire makes the dense-open core dense; closure monotonicity transfers density to its residual superset.\\
\hlt{Key Insight}: Every superset of a dense set is dense.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:baire-category-complete-metric-space} \hlt{Baire Category Theorem for Complete Metric Spaces}\\
Let $(X,d)$ be a nonempty complete metric space, and let $(E_n)_{n\in\N}$ be nowhere-dense subsets of $X$. Then $X\ne\bigcup\limits_{n=1}^{\infty}E_n$.
\begin{proof}
Suppose $X=\bigcup\limits_{n=1}^{\infty}E_n$, $O_n\defeq X\setminus\overline{E_n}$. Then $(O_n)$ is sequence of open subsets of $X$ by Theorem \ref{thm:closure-smallest-closed-containing-set}(i), Corollary \ref{cor:closed-iff-complement-open}.\\
Fix $n\in\N$, $p\in X\setminus O_n$, $r>0$. Since $(\overline{E_n})^\circ=\emptyset$, ball $B_r(p)$ is not contained in $\overline{E_n}$. Hence $B_r(p)\cap O_n\ne\emptyset$, so $p\in O_n^{\operatorname{acc}}$.\\
Thus $(O_n)$ is a sequence of dense open subsets of $X$ by Definition \ref{def:metric-topology-core-definitions}.\\
Choose $x\in\bigcap\limits_{n=1}^{\infty}O_n$ by Theorem \ref{thm:dense-open-baire-complete-metric-space}(i) with $U=X$. Then $(\forall n\in\N)\,x\notin E_n$, contradicting $X=\bigcup\limits_{n=1}^{\infty}E_n$.\\
\hlt{Structural Commentary}: Nowhere-dense closures have dense open complements; their Baire intersection avoids the proposed cover.\\
\hlt{Key Insight}: One point can simultaneously escape every nowhere-dense layer.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:baire-category-real-line} \hlt{Baire's Theorem for $\R$}\\
If $(E_n)_{n\in\N}$ are nowhere-dense subsets of $\R$, then $\R\ne\bigcup\limits_{n=1}^{\infty}E_n$.
\begin{proof}
Suppose $\R=\bigcup\limits_{n=1}^{\infty}E_n$, $U_n\defeq\R\setminus\overline{E_n}$. Then $(U_n)$ is a sequence of open subsets of $\R$ by Theorem \ref{thm:closure-smallest-closed-containing-set}(i), Corollary \ref{cor:closed-iff-complement-open}.\\
Fix $n\in\N$ and a nonempty open interval $I$. The set $I$ meets $U_n$ because $I\nsubseteq\overline{E_n}$.\\
Hence $(U_n)$ is a sequence of dense open subsets of $\R$ by Proposition \ref{prop:dense-closure-and-interval-test}.\\
Choose $x\in\bigcap\limits_{n=1}^{\infty}U_n$ by Theorem \ref{thm:dense-open-baire-real-line}. Then $(\forall n\in\N)\,x\notin E_n$, contradicting $\R=\bigcup\limits_{n=1}^{\infty}E_n$.\\
\hlt{Structural Commentary}: Real-line interval tests turn nowhere-dense complements into the dense-open intersection theorem.\\
\hlt{Key Insight}: Countably many thin sets leave a common point uncovered.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:rationals-not-gdelta-irrationals-not-fsigma} \hlt{Rationals and Irrationals by Category}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The set $\Q$ is $F_\sigma$.
\item The set $\R\setminus\Q$ is $G_\delta$.
\item The set $\Q$ is not $G_\delta$.
\item The set $\R\setminus\Q$ is not $F_\sigma$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose an enumeration $\Q=\{q_1,q_2,\ldots\}$ by Corollary \ref{cor:rationals-countable}.\\
For $n\in\N$, the singleton $\{q_n\}$ is closed by Corollary \ref{cor:finite-point-set-no-limit-points} and Definition \ref{def:metric-topology-core-definitions}. Hence $\Q=\bigcup\limits_{n=1}^{\infty}\{q_n\}$ is $F_\sigma$.
\item Taking complements in (i), $\R\setminus\Q=\bigcap\limits_{n=1}^{\infty}(\R\setminus\{q_n\})$ is $G_\delta$.
\item
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Dense layers}: Suppose $\Q=\bigcap\limits_{n=1}^{\infty}G_n$, where $(G_n)$ is a sequence of open subsets of $\R$. Fix $n\in\N$.\\
The set $G_n$ is dense by $\Q\subseteq G_n$, Theorem \ref{thm:archimedean-and-rational-density}(ii), and Proposition \ref{prop:dense-closure-and-interval-test}.
\item \hlt{Punctured layers}: For $n\in\N$, set $H_n\defeq G_n\setminus\{q_n\}$.\\
Each $H_n$ is open by Corollary \ref{cor:closed-iff-complement-open} and Theorem \ref{thm:metric-open-closed-set-calculus}(iii). Fix $n\in\N$ and a nonempty open interval $(a,b)$.\\
Choose $y\in(a,b)\cap G_n$ by density of $G_n$. By openness, choose $\varepsilon>0$ with $(y-\varepsilon,y+\varepsilon)\subseteq(a,b)\cap G_n$.\\
At least one of $y-\frac{\varepsilon}{2}$ and $y+\frac{\varepsilon}{2}$ differs from $q_n$, so $(a,b)\cap H_n\ne\emptyset$. Thus $H_n$ is dense by Proposition \ref{prop:dense-closure-and-interval-test}.
\item \hlt{Baire contradiction}: Therefore $(H_n)$ is a sequence of dense open subsets of $\R$.\\
Choose $x\in\bigcap\limits_{n=1}^{\infty}H_n$ by Theorem \ref{thm:dense-open-baire-real-line}. Since $(\forall n\in\N)\,H_n\subseteq G_n$, one has $x\in\bigcap\limits_{n=1}^{\infty}G_n=\Q$.\\
Choose $k\in\N$, $x=q_k$ by $\Q=\{q_1,q_2,\ldots\}$. Then $x\notin H_k$, whereas $x\in H_k$ by choice of $x$, contradiction.
\end{enumerate}
\item If $\R\setminus\Q$ were $F_\sigma$, then its complement $\Q$ would be $G_\delta$, contrary to (iii).
\end{enumerate}
\hlt{Structural Commentary}: Singleton decompositions establish the positive classes; Baire excludes the reversed classifications.\\
\hlt{Key Insight}: Removing the enumerated rational from each dense open layer forces an impossible intersection.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:baire-closed-cover-has-interior} \hlt{Closed Cover Interior Test}\\
Let $X$ be a complete metric space and let $(F_n)_{n\in\N}$ be closed subsets of $X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\emptyset\ne U\subseteq X$ is open and $U\subseteq\bigcup\limits_{n=1}^{\infty}F_n$, then $(\exists\,N\in\N)\,F_N^\circ\ne\emptyset$.
\item If $X\ne\emptyset$ and $(\forall n\in\N)\,F_n^\circ=\emptyset$, then $X\ne\bigcup\limits_{n=1}^{\infty}F_n$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] Suppose conclusion fails, and $O_n\defeq X\setminus F_n$. Then $(O_n)$ is sequence of open subsets of $X$ by Corollary \ref{cor:closed-iff-complement-open}.\\
Fix $n\in\N$, $p\in F_n$, and $r>0$. Then $B_r(p)\nsubseteq F_n$ because $F_n^\circ=\emptyset$. Thus $B_r(p)\cap O_n\ne\emptyset$, so $p\in O_n^{\operatorname{acc}}$.\\
Hence $(O_n)$ is a sequence of dense open subsets of $X$ by Definition \ref{def:metric-topology-core-definitions}.\\
Choose $x\in U\cap\bigcap\limits_{n=1}^{\infty}O_n$ by Theorem \ref{thm:dense-open-baire-complete-metric-space}(i). Then $x\in U\subseteq\bigcup\limits_{n=1}^{\infty}F_n$ while $(\forall n\in\N)\,x\notin F_n$, a contradiction.
\item[$(ii)$] If $X=\bigcup\limits_{n=1}^{\infty}F_n$, then (i) with $U=X$ implies $(\exists\,N\in\N)\,F_N^\circ\ne\emptyset$, contrary to the hypothesis.
\end{enumerate}
\hlt{Structural Commentary}: Empty interiors create dense open complements; their Baire intersection contradicts the closed cover.\\
\hlt{Key Insight}: A complete space cannot be covered locally by closed sets with empty interior.
\end{proof}
\end{corollary}

\begin{perspective} \label{per:baire-category-in-function-spaces} \hlt{Category in Function Spaces}\\
Theorems \ref{thm:dense-open-baire-complete-metric-space} and \ref{thm:baire-category-complete-metric-space} depend on metric completeness rather than the order structure of $\R$.\\
They therefore transfer to later function spaces after completeness of the chosen function-space metric has been established.
\end{perspective}

\begin{proofmethod} \label{meth:baire-category-proof-patterns} \hlt{Baire-Category Proof Patterns}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Dense-open target}: to prove $\bigcap\limits_{n=1}^{\infty}O_n$ dense, fix a nonempty open set $U$. Construct a witness in $U\cap\bigcap\limits_{n=1}^{\infty}O_n$.
\item \hlt{Complete-metric construction}: choose nonempty closed balls $C_{n+1}\subseteq C_n\cap O_{n+1}$ with $\diam C_n\to0$.\\
The common point exists by Theorem \ref{thm:cantor-intersection-complete-metric-space}.
\item \hlt{Nowhere-dense input}: replace exceptional sets $E_n$ by the dense open complements $X\setminus\overline{E_n}$.
\item \hlt{Meager exceptional set}: find nowhere-dense $(E_n)$ with $E\subseteq\bigcup\limits_{n=1}^{\infty}E_n$.\\
Then choose a point in the complementary dense-open intersection.
\item \hlt{Residual property}: exhibit dense open sets $(O_n)$ with $\bigcap\limits_{n=1}^{\infty}O_n$ contained in the truth set.
\item \hlt{$F_\sigma/G_\delta$ duality}: complement a countable union of closed sets by De Morgan's laws.
\end{enumerate}
\end{proofmethod}
