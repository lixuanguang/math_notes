\subsection{Connectedness and Continuity}\label{subsec:connectedness-and-continuity}

% **Role:** abstract structure behind limits and continuity.
%
% Connectedness, path-connectedness, continuity, homeomorphisms, preservation theorems, IVT/EVT.


Connectedness prevents a space from splitting into separated pieces; continuity transports topological structure through maps. This subsection also records IVT/EVT and real-line refinements.

\begin{definition} \label{def:separated-and-connected-sets} \hlt{Separated and Connected Sets}\\
Let $X$ be a metric space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Subsets $A,B\subseteq X$ are \hlt{separated} if $A\cap\overline{B}=\overline{A}\cap B=\emptyset$.
\item A subset $E\subseteq X$ is \hlt{connected} if no nonempty separated subsets $A,B\subseteq E$ satisfy $E=A\cup B$.
\item A subset $U\subseteq X$ is \hlt{clopen} if it is both open and closed.
\end{enumerate}
\hlt{Structural Commentary}: Separated sets avoid each other's closures.\\
\hlt{Key Insight}: Disconnected means a nontrivial clopen split; continuous preimages preserve such splits.
\end{definition}

\begin{theorem} \label{thm:connected-subsets-of-real-line-interval-property} \hlt{Connected Subsets of the Real Line}\\
A set $E\subseteq\R$ is connected if and only if $x,y\in E,\ x<y$ $\Longrightarrow$ $[x,y]\subseteq E$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If the interval property fails, choose $x,y\in E$ and $x<z<y$ with $z\notin E$. Then $A\defeq E\cap(-\infty,z)$ and $B\defeq E\cap(z,\infty)$ are nonempty separated sets with $E=A\cup B$, contradicting connectedness.
\item[$(\Leftarrow)$] Suppose $E=A\cup B$ with $A,B$ nonempty separated. Choose $x\in A$, $y\in B$, $x<y$, and set $S\defeq A\cap[x,y]$, $z\defeq\sup S$. By Theorem \ref{thm:supremum-belongs-to-closure}, $z\in\overline{A}$; hence $z\notin B$ and $z<y$. If $z\notin A$, then $z\in[x,y]\setminus E$.\\
If $z\in A$, then $z\notin\overline{B}$, so choose $z_1\in(z,y)$ with $z_1\notin B$; by maximality of $z$, also $z_1\notin A$.\\
Thus $z_1\in[x,y]\setminus E$, contradicting $[x,y]\subseteq E$.
\end{enumerate}
\hlt{Structural Commentary}: On the real line, connectedness is exactly having no gaps between any two of its points.\\
\hlt{Key Insight}: A separation creates a missing point by taking a supremum at the boundary of one side.
\end{proof}
\end{theorem}

\begin{method} \label{meth:compact-connected-proof-patterns} \hlt{Compactness and Connectedness Proof Patterns}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Open-cover compactness}: for every cover $\{G_\alpha\}_{\alpha\in A}$, find finite $F\subseteq A$ with $K\subseteq\bigcup\limits_{\alpha\in F}G_\alpha$.
\item \hlt{Euclidean closed-bounded compactness}: if $E\subseteq\R^k$ is closed and bounded, choose a compact $k$-cell $I\supseteq E$ and apply Theorem \ref{thm:closed-subsets-of-compact-sets-compact}.
\item \hlt{Connectedness on $\R$}: replace connectedness of $E\subseteq\R$ by $x,y\in E, x<y$ $\Longrightarrow$ $[x,y]\subseteq E$.
\end{enumerate}
\end{method}

\begin{definition} \label{def:function-limit-metric}\label{def:continuity-core-definitions-source-todo} \hlt{Limit of a Function Between Metric Spaces}\\
Let $X,Y$ be metric spaces, $E\subseteq X$, $f:E\to Y$, and let $p$ be a limit point of $E$. For $q\in Y$, $\lim_{x\to p} f(x)=q$ means that, for every $\varepsilon>0$, there exists $\delta>0$ such that $x\in E$, $0<d_X(x,p)<\delta$ $\Longrightarrow$ $d_Y(f(x),q)<\varepsilon$.\\
The point $p$ need not belong to $E$, and $f(p)$ is irrelevant when it does.\\
\hlt{Key Insight}: Domain $E$ controls approach: $J\setminus\{a\}$ gives two-sided limit,  $(a,b)$ or $(c,a)$ gives one-sided limits.
\end{definition}

\begin{theorem} \label{thm:function-limit-sequential-characterization} \hlt{Sequential Characterization of Function Limits}\\
Let $X,Y$ be metric spaces, $E\subseteq X$, $f:E\to Y$, and let $p$ be a limit point of $E$.\\
For $q\in Y$, $\lim_{x\to p}f(x)=q$ if and only if $f(p_n)\to q$ for every sequence $(p_n)\subseteq E\setminus\{p\}$ with $p_n\to p$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $p_n\to p$ with $p_n\in E\setminus\{p\}$, then eventually $0<d_X(p_n,p)<\delta$; hence $d_Y(f(p_n),q)<\varepsilon$, so $f(p_n)\to q$.
\item[$(\Leftarrow)$] If the limit fails, then for some $\varepsilon>0$ and each $n$ there is $x_n\in E$ with $0<d_X(x_n,p)<1/n$ and $d_Y(f(x_n),q)\geq\varepsilon$. Then $x_n\to p$, $x_n\neq p$, but $f(x_n)\not\to q$.
\end{enumerate}
\hlt{Structural Commentary}: Function limits can be tested by all sequences approaching the input point.\\
\hlt{Key Insight}: The negation of an epsilon--delta limit manufactures a bad sequence.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:function-limits-are-unique} \hlt{Uniqueness of Function Limits}\\
If a function has a limit at $p$, then that limit is unique.
\begin{proof}
Let $q$ and $r$ be limits of $f$ at $p$. Choose $p_n\in E\setminus\{p\}$ with $d_X(p_n,p)<1/n$.\\
Then $p_n\to p$, so Theorem \ref{thm:function-limit-sequential-characterization} gives $f(p_n)\to q$ and $f(p_n)\to r$. By Theorem \ref{thm:metric-sequence-basic-properties}, $q=r$.\\
\hlt{Structural Commentary}: Limit uniqueness is inherited from metric sequence uniqueness.\\
\hlt{Key Insight}: One approaching sequence cannot have two different limits.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:function-limit-algebra} \hlt{Algebra of Function Limits}\\
Let $X$ be a metric space, $E\subseteq X$, $p$ be limit point of $E$.\\
For $f,g:E\to\C$, assume $\lim_{x\to p}f(x)=A$ and $\lim_{x\to p}g(x)=B$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\lim\limits_{x\to p}(f+g)(x)=A+B$;
\item $\lim\limits_{x\to p}(fg)(x)=AB$;
\item if $B\neq0$ and $f/g$ is defined near $p$, then $\lim\limits_{x\to p}(f/g)(x)=A/B$.
\end{enumerate}
For $f,g:E\to\R^k$ with limits $a,b$, the corresponding limits are $f+g\to a+b$, $\lambda f\to\lambda a$, and $f\cdot g\to a\cdot b$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For any $p_n\in E\setminus\{p\}$ with $p_n\to p$, Theorem \ref{thm:function-limit-sequential-characterization} gives $f(p_n)\to A$ and $g(p_n)\to B$.\\
Hence $(f+g)(p_n)\to A+B$ by Theorem \ref{thm:complex-sequence-limit-algebra}, so Theorem \ref{thm:function-limit-sequential-characterization} gives (i).
\item The same sequence gives $(fg)(p_n)\to AB$ by Theorem \ref{thm:complex-sequence-limit-algebra}; hence (ii).
\item If $B\neq0$ and $f/g$ is defined near $p$, then every such sequence is eventually in the domain of $f/g$, and $(f/g)(p_n)\to A/B$ by Theorem \ref{thm:complex-sequence-limit-algebra}; hence (iii).
\end{enumerate}
The vector laws follow by the same sequential characterization and Theorem \ref{thm:rk-sequence-coordinate-convergence}.\\
\hlt{Structural Commentary}: Limit algebra for functions is pulled back to sequence algebra.\\
\hlt{Key Insight}: Sequential characterization lets previous sequence theorems do the work.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:function-limits-are-local} \hlt{Function Limits Are Local}\\
Let $X,Y$ be metric spaces, let $E\subseteq X$, let $f:E\to Y$, let $p$ be a limit point of $E$, and let $q\in Y$.\\
For every $r>0$, $\lim\limits_{x\to p;\,x\in E}f(x)=q$ $\Longleftrightarrow$ $\lim\limits_{x\to p;\,x\in E\cap B_r(p)}f(x)=q$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Since $E\cap B_r(p)\subseteq E$, any $\delta$ witnessing the limit on $E$ also witnesses the limit on $E\cap B_r(p)$.
\item[$(\Leftarrow)$] Assume the restricted limit. Given $\varepsilon>0$, choose $\delta_0>0$ such that $x\in E\cap B_r(p)$ and $0<d_X(x,p)<\delta_0$ imply $d_Y(f(x),q)<\varepsilon$. Let $\delta\defeq\min\{\delta_0,r\}$.\\
If $x\in E$ and $0<d_X(x,p)<\delta$, then $x\in E\cap B_r(p)$, so $d_Y(f(x),q)<\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Values away from $p$ cannot affect the limit at $p$.\\
\hlt{Key Insight}: Shrink $\delta$ until every tested point lies in the chosen local neighborhood.
\end{proof}
\end{proposition}

\begin{definition} \label{def:continuity-at-a-point-and-on-a-set} \hlt{Continuity at a Point and on a Set}\\
Let $(X,d_X),(Y,d_Y)$ be metric spaces, $E\subseteq X$, and $f:E\to Y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $p\in E$, $f$ is \hlt{continuous at $p$} if $\forall\varepsilon>0\ \exists\delta>0\ \forall x\in E$, $d_X(x,p)<\delta\Longrightarrow d_Y(f(x),f(p))<\varepsilon$.
\item $f$ is \hlt{continuous on $E$} if $\forall p\in E$, $f$ is continuous at $p$.
\item If $p$ is isolated in $E$, then $f$ is continuous at $p$.
\end{enumerate}
\end{definition}

\begin{corollary} \label{cor:discrete-domain-maps-continuous} \hlt{Discrete Domains Make Every Map Continuous}\\
If $X$ carries the discrete metric, then every map $f:X\to Y$ into any metric space $Y$ is continuous.
\begin{proof}
For $p\in X$, $B_{1/2}(p)=\{p\}$. Take $\delta=\frac{1}{2}$. If $d_X(x,p)<\delta$, then $x=p$, so $d_Y(f(x),f(p))=0<\varepsilon$.\\
\hlt{Structural Commentary}: In the discrete metric, no distinct point can approach $p$.\\
\hlt{Key Insight}: Pointwise continuity is automatic when small enough balls are singletons.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:continuity-as-limit-at-limit-point} \hlt{Continuity as Equality with the Function Limit}\\
Let $X,Y$ be metric spaces, $E\subseteq X$, $p\in E$, and $f:E\to Y$.\\
If $p$ is a limit point of $E$, then $f$ is continuous at $p$ if and only if $\lim_{x\to p}f(x)=f(p)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $0<d_X(x,p)<\delta$, continuity at $p$ gives $d_Y(f(x),f(p))<\varepsilon$, hence $\lim_{x\to p}f(x)=f(p)$.
\item[$(\Leftarrow)$] If $\lim_{x\to p}f(x)=f(p)$, the same $\delta$ proves continuity for $x\neq p$; for $x=p$, $d_Y(f(p),f(p))=0$.
\end{enumerate}
\hlt{Structural Commentary}: Continuity is the compatibility of the value at $p$ with nearby values.\\
\hlt{Key Insight}: Limits ignore the point; continuity ties the point back in.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:adherent-function-limits-convention} \hlt{Adherent-Point Limit Convention}\\
The adherent-point convention takes $p\in\overline{E}$ and allows $x=p$ in the $\delta$-test; Definition \ref{def:function-limit-metric} instead assumes $p$ is a limit point and requires $0<d_X(x,p)$. If $p\in E$, the unpunctured convention also tests $f(p)$, forcing $q=f(p)$; in the punctured convention this is exactly continuity at $p$ by Theorem \ref{thm:continuity-as-limit-at-limit-point}.\\
\hlt{Key Insight}: Punctured limits study nearby values; continuity adds compatibility with the value at the point.
\end{remark}

\begin{remark} \label{rem:sequential-continuity-real-line} \hlt{The Sequence-First View}\\
For functions on subsets of $\R$, continuity at limit points is equivalently sequential: $x_n\to p$ in the domain implies $f(x_n)\to f(p)$; isolated points are automatic by Theorems \ref{thm:function-limit-sequential-characterization} and \ref{thm:continuity-as-limit-at-limit-point}.
\end{remark}

\begin{theorem} \label{thm:metric-continuity-sequential-characterization} \hlt{Sequential Characterization of Continuity}\\
Let $(X,d_X),(Y,d_Y)$ be metric spaces, $f:X\to Y$, and $p\in X$.\\
Then $f$ is continuous at $p$ if and only if $\forall(x_n)\subseteq X$, $x_n\to p\Longrightarrow f(x_n)\to f(p)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $f$ is continuous at $p$ and let $x_n\to p$.\\
Given $\varepsilon>0$, choose $\delta>0$ such that $d_X(x,p)<\delta\Longrightarrow d_Y(f(x),f(p))<\varepsilon$.\\
Since $x_n\to p$, eventually $d_X(x_n,p)<\delta$, hence eventually $d_Y(f(x_n),f(p))<\varepsilon$. Thus $f(x_n)\to f(p)$.
\item[$(\Leftarrow)$] Suppose the sequential condition holds. If $f$ is not continuous at $p$, then $\exists\varepsilon_0>0$ such that $\forall\delta>0$, $\exists x\in X$ with $d_X(x,p)<\delta$ and $d_Y(f(x),f(p))\geq\varepsilon_0$. Taking $\delta=1/n$ gives $x_n\in X$ with $d_X(x_n,p)<1/n$ and $d_Y(f(x_n),f(p))\geq\varepsilon_0$. Then $x_n\to p$, but $f(x_n)\nrightarrow f(p)$, a contradiction.
\end{enumerate}
\hlt{Structural Commentary}: Failure of epsilon--delta continuity manufactures a bad sequence.\\
\hlt{Key Insight}: In metric spaces, sequences detect continuity at points.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:continuous-composition} \hlt{Composition of Continuous Functions}\\
Let $E\subseteq X$ and let $f:E\to Y$, $g:f(E)\to Z$ be maps between metric spaces.\\
If $f$ is continuous at $p\in E$ and $g$ is continuous at $f(p)$, then $g\circ f$ is continuous at $p$.
\begin{proof}
Given $\varepsilon>0$, choose $\eta>0$ for $g$ at $f(p)$, then choose $\delta>0$ for $f$ at $p$ with tolerance $\eta$.\\
If $x\in E$ and $d_X(x,p)<\delta$, then $d_Y(f(x),f(p))<\eta$, hence $d_Z(g(f(x)),g(f(p)))<\varepsilon$.\\
\hlt{Structural Commentary}: The output tolerance for $g$ becomes the input tolerance for $f$.\\
\hlt{Key Insight}: Continuity composes by passing epsilon through an intermediate eta.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:distance-function-is-continuous} \hlt{The Distance Function Is Continuous}\\
Let $(X,d)$ be a metric space and $D:X\times X\to\R$, $D(x,y)\defeq d(x,y)$.\\
For $d_\times\in\{d_1,d_2,d_\infty\}$ on $X\times X$, $D:(X\times X,d_\times)\to\R$ is continuous.\\
Equivalently, $x_n\to x$ and $y_n\to y$ in $X$ imply $d(x_n,y_n)\to d(x,y)$.
\begin{proof} For $x,y,u,v\in X$, $\abs{D(x,y)-D(u,v)}=\abs{d(x,y)-d(u,v)}\leq d(x,u)+d(y,v)$. Hence $x_n\to x$ and $y_n\to y$ imply $D(x_n,y_n)\to D(x,y)$. Proposition \ref{prop:product-metric-comparison} and Theorem \ref{thm:metric-continuity-sequential-characterization} give continuity for each product metric.\\
\hlt{Structural Commentary}: Distance changes continuously when both endpoints move.\\
\hlt{Key Insight}: The triangle inequality controls distance error by endpoint errors.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:continuity-open-preimages}\label{thm:continuity-structure-results-source-todo} \hlt{Continuity via Open Preimages}\\
For metric spaces $X,Y$, map $f:X\to Y$ is continuous if and only if $f^{-1}(V)$ is open in $X \forall$ open $V\subseteq Y$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $p\in f^{-1}(V)$ and $V$ is open, choose $\varepsilon>0$ with $B_\varepsilon(f(p))\subseteq V$.\\
Continuity gives $\delta>0$ with $B_\delta(p)\subseteq f^{-1}(V)$, so $f^{-1}(V)$ is open.
\item[$(\Leftarrow)$] For $p\in X$ and $\varepsilon>0$, the ball $V=B_\varepsilon(f(p))$ is open by Theorem \ref{thm:every-neighborhood-open}.\\
Since $p\in f^{-1}(V)$ and $f^{-1}(V)$ is open, some $B_\delta(p)\subseteq f^{-1}(V)$; this is continuity at $p$.
\end{enumerate}
\hlt{Structural Commentary}: Continuity is exactly preservation of openness under inverse image.\\
\hlt{Key Insight}: Images are not Boolean-exact, but preimages are; continuity belongs to preimages.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:relative-domain-continuity-open-preimages} \hlt{Continuity on a Relative Domain}\\
Let $(X,d_X)$ and $(Y,d_Y)$ be metric spaces, let $E\subseteq X$, and let $f:E\to Y$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f:(E,d_X|_{E\times E})\to(Y,d_Y)$ is continuous.
\item For every open $V\subseteq Y$, $f^{-1}(V)$ is open relative to $E$.
\item For every open $V\subseteq Y$, there exists open $G\subseteq X$ such that $f^{-1}(V)=E\cap G$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ For open $V\subseteq Y$, Theorem \ref{thm:continuity-open-preimages} on the metric space $E$ gives $f^{-1}(V)$ open in $E$.
\item[$(ii)$]$\Rightarrow (iii)$ For open $V\subseteq Y$, relative openness of $f^{-1}(V)\subseteq E$ gives $f^{-1}(V)=E\cap G$ for some open $G\subseteq X$.
\item[$(iii)$]$\Rightarrow (i)$ For open $V\subseteq Y$, $f^{-1}(V)=E\cap G$ with $G$ open in $X$, hence $f^{-1}(V)$ is open in $E$.\\
Theorem \ref{thm:continuity-open-preimages} on $E$ gives continuity.
\end{enumerate}
\hlt{Structural Commentary}: Open preimages are relative opens in $E$.\\
\hlt{Key Insight}: Relative continuity says ordinary open sets in $X$ are seen through intersection with $E$.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:continuity-closed-preimages} \hlt{Continuity via Closed Preimages}\\
Let $X,Y$ be metric spaces, $f:X\to Y$. Then $f$ is continuous iff $f^{-1}(C)$ is closed in $X$ for every closed $C\subseteq Y$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] For closed $C\subseteq Y$, $X\setminus f^{-1}(C)=f^{-1}(Y\setminus C)$ is open by Theorem \ref{thm:continuity-open-preimages}; hence $f^{-1}(C)$ is closed.
\item[$(\Leftarrow)$] For open $V\subseteq Y$, $X\setminus f^{-1}(V)=f^{-1}(Y\setminus V)$ is closed by hypothesis; hence $f^{-1}(V)$ is open, QED.
\end{enumerate}
\hlt{Structural Commentary}: Closed-set continuity is the complement form of open-set continuity.\\
\hlt{Key Insight}: $f^{-1}(Y\setminus C)=X\setminus f^{-1}(C)$ transfers open and closed formulations.
\end{proof}
\end{corollary}

\begin{perspective} \label{per:three-views-of-continuity} \hlt{Three Views of Continuity}\\
For metric-space maps, continuity is equivalently $\varepsilon$--$\delta$ control, $x_n\to x\Rightarrow f(x_n)\to f(x)$,\\
and $V\subseteq Y$ open $\Rightarrow f^{-1}(V)$ open; see Definition \ref{def:continuity-at-a-point-and-on-a-set} and Theorems \ref{thm:metric-continuity-sequential-characterization}, \ref{thm:continuity-open-preimages}.\\
\hlt{Key Insight}: Continuity sends near points forward and pulls open sets back.
\end{perspective}

\begin{proposition} \label{prop:distance-to-set-is-continuous} \hlt{Distance-to-Set Is Continuous}\\
Let $(X,d)$ be a metric space, let $\emptyset\neq E\subseteq X$, and define $f:X\to\R$ by $f(p)\defeq\operatorname{dist}(p,E)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f$ is $1$-Lipschitz: $\abs{f(p)-f(q)}\leq d(p,q)$ for all $p,q\in X$; in particular, $f$ is continuous.
\item $f^{-1}(\{0\})=\overline{E}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $p,q\in X$ and $e\in E$, $d(p,e)\leq d(p,q)+d(q,e)$. Taking the infimum over $e\in E$ gives $f(p)\leq d(p,q)+f(q)$, hence $f(p)-f(q)\leq d(p,q)$. Hence $f(q)-f(p)\leq d(p,q)$, so $\abs{f(p)-f(q)}\leq d(p,q)$, QED.
\item For $p\in X$, $p\in f^{-1}(\{0\})$ iff $\operatorname{dist}(p,E)=0$ iff $p\in\overline{E}$ by Proposition \ref{prop:set-distance-closure-criterion}.
\end{enumerate}
\hlt{Structural Commentary}: Distance to a fixed set changes no faster than the point itself moves.\\
\hlt{Key Insight}: The triangle inequality turns nearest-set distance into a continuous detector of closure.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:open-closed-sets-by-continuous-real-functions} \hlt{Open and Closed Sets by Distance Functions}\\
Let $X$ be a metric space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A set $O\subseteq X$ is open if and only if there is a continuous $f:X\to\R$ such that $O=\{x\in X:f(x)>0\}$.
\item A set $F\subseteq X$ is closed if and only if there is a continuous $g:X\to\R$ such that $F=g^{-1}(\{0\})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $O=X$, take $f\equiv1$. Otherwise set $f(x)\defeq\operatorname{dist}(x,O^c)$. Since $O^c$ is closed, Proposition \ref{prop:distance-to-set-is-continuous} gives $\{f=0\}=O^c$, hence $O=\{f>0\}$. Conversely, $\{f>0\}=f^{-1}((0,\infty))$ is open by Theorem \ref{thm:continuity-open-preimages}.
\item If $F=\emptyset$, take $g\equiv1$. Otherwise set $g(x)\defeq\operatorname{dist}(x,F)$.\\
If $F$ closed, by Proposition \ref{prop:distance-to-set-is-continuous}, $g^{-1}(\{0\})=\overline{F}=F$. Conversely, $g^{-1}(\{0\})$ closed by Corollary \ref{cor:continuity-closed-preimages}.
\end{enumerate}
\hlt{Structural Commentary}: Metric spaces encode open and closed sets by continuous real-valued distance functions.\\
\hlt{Key Insight}: A set is closed exactly when its distance-zero locus has no extra points.
\end{proof}
\end{corollary}

\begin{definition} \label{def:homeomorphism} \hlt{Homeomorphism}\\
A \hlt{homeomorphism} from $X$ to $Y$ is a bijection $f:X\to Y$ such that both $f$ and $f^{-1}$ are continuous.\\
If such a map exists, $X$ and $Y$ are \hlt{homeomorphic}, written $X\simeq Y$.
\end{definition}

\begin{proposition} \label{prop:homeomorphisms-preserve-topology} \hlt{Homeomorphisms Preserve Topology}\\
If $f:X\to Y$ is a homeomorphism, then $U\mapsto f(U)$ bijects open subsets of $X$ with open subsets of $Y$.
\begin{proof}
If $U\subseteq X$ is open, then $f(U)=(f^{-1})^{-1}(U)$ is open in $Y$ since $f^{-1}$ is continuous. If $V\subseteq Y$ is open, then $f^{-1}(V)$ is open in $X$ since $f$ is continuous. These two assignments are inverse because $f$ is bijective.\\
\hlt{Structural Commentary}: A homeomorphism preserves open sets in both directions.\\
\hlt{Key Insight}: Topological sameness is continuity plus continuous reversibility.
\end{proof}
\end{proposition}

\begin{definition} \label{def:metric-isometry} \hlt{Metric Isometry}\\
Let $(X,d_X),(Y,d_Y)$ be metric spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A map $f:X\to Y$ is an \hlt{isometry} if $\forall x,y\in X$, $d_Y(f(x),f(y))=d_X(x,y)$.
\item $X$ and $Y$ are \hlt{isometric} if $\exists$ a bijective isometry $f:X\to Y$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:metric-isometry-basic-properties} \hlt{Basic Properties of Metric Isometries}\\
Let $f:(X,d_X)\to(Y,d_Y)$ be an isometry.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f$ is injective.
\item $f$ is continuous.
\item $f^{-1}:(f(X),d_Y|_{f(X)\times f(X)})\to(X,d_X)$ is an isometry.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $f(x)=f(y)$, then $d_X(x,y)=d_Y(f(x),f(y))=0$, hence $x=y$.
\item If $x_n\to x$ in $X$, then $d_Y(f(x_n),f(x))=d_X(x_n,x)\to0$; by Theorem \ref{thm:metric-continuity-sequential-characterization}, $f$ is continuous.
\item For $u=f(x),v=f(y)\in f(X)$, $d_X(f^{-1}(u),f^{-1}(v))=d_X(x,y)=d_Y(f(x),f(y))=d_Y(u,v)$; QED.
\end{enumerate}
\hlt{Structural Commentary}: An isometry preserves all distances, so injectivity and continuity are automatic.\\
\hlt{Key Insight}: Distance preservation is stronger than topological equivalence.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:metric-versus-linear-isometries} \hlt{Metric Versus Linear Isometries}\\
Metric isometry is purely distance-theoretic. If $T:V\to W$ is linear between inner-product spaces, then\\
$\norm{Tv}=\norm v\ \forall v\in V\Longleftrightarrow d_W(Tu,Tv)=d_V(u,v)\ \forall u,v\in V$.
\end{remark}

\begin{example} \label{ex:circle-param-continuous-bijection-not-homeomorphism} \hlt{Continuous Bijection Need Not Be a Homeomorphism}\\
Let $f:[0,2\pi)\to S^1$, and $f(t)=(\cos t,\sin t)$. This continuous bijection is not a homeomorphism: points of $S^1$ approaching $(1,0)$ from below have preimages tending to $2\pi$, not $0$.
\end{example}

\begin{remark} \label{rem:continuity-domain-only} \hlt{Continuity Only Uses the Domain Metric}\\
For $f:E\to Y$ with $E\subseteq X$, continuity on $E$ depends only on the restricted metric on $E$, not on $X\setminus E$.
\end{remark}

\begin{definition} \label{def:product-map-direct-sum} \hlt{Product Map}\\
If $f:X\to Y$ and $g:X\to Z$ are maps between metric spaces, their \hlt{product map}, or direct sum, is\\
$f\oplus g:X\to Y\times Z$, $(f\oplus g)(x)\defeq(f(x),g(x))$, where $Y\times Z$ carries any of product metrics in Definition \ref{def:product-metrics}.
\end{definition}

\begin{proposition} \label{prop:product-map-continuity} \hlt{Continuity of Product Maps}\\
The product map $f\oplus g:X\to Y\times Z$ is continuous at $x_0$ iff $f$ and $g$ are continuous at $x_0$.\\
Hence it is continuous on $X$ iff both components are.
\begin{proof}
By Proposition \ref{prop:product-metric-comparison}, it suffices to use $d_{\mathrm{sum}}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Given $\varepsilon>0$, choose $\delta>0$ from continuity of $f\oplus g$ at $x_0$.\\
If $d_X(x,x_0)<\delta$, then $d_Y(f(x),f(x_0)),d_Z(g(x),g(x_0))\leq d_{\mathrm{sum}}((f\oplus g)(x),(f\oplus g)(x_0))<\varepsilon$.
\item[$(\Leftarrow)$] Given $\varepsilon>0$, choose $\delta_f,\delta_g$ for $f,g$ at tolerance $\varepsilon/2$ and set $\delta=\min\{\delta_f,\delta_g\}$.\\
Then $d_X(x,x_0)<\delta$ gives $d_{\mathrm{sum}}((f\oplus g)(x),(f\oplus g)(x_0))=d_Y(f(x),f(x_0))+d_Z(g(x),g(x_0))<\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Product continuity is exactly componentwise continuity.\\
\hlt{Key Insight}: A product metric packages two output estimates into one.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:continuous-algebra-and-components} \hlt{Continuous Algebra and Components}\\
Let $X$ be a metric space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $f,g:X\to\C$ are continuous, so are $f+g$, $fg$, and $f/g$ where $g\neq0$.
\item For $f=(f_1,\ldots,f_k):X\to\R^k$, $f$ is continuous if and only if each $f_j$ is continuous.
\item If $f,g:X\to\R^k$ are continuous, so are $f+g$ and $f\cdot g$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item At isolated points continuity is automatic; at limit points, use Theorems \ref{thm:function-limit-algebra} and \ref{thm:continuity-as-limit-at-limit-point}.
\item By Theorem \ref{thm:rk-sequence-coordinate-convergence}, convergence in $\R^k$ is exactly coordinatewise convergence; apply this to $f(x_n)\to f(x)$.
\item Apply (ii) to the coordinate functions of $f+g$ and $f\cdot g$, using the scalar algebra from (i).
\end{enumerate}
\hlt{Structural Commentary}: Continuity is stable under the algebraic operations already stable for limits.\\
\hlt{Key Insight}: Componentwise control is equivalent to Euclidean control in finite dimensions.
\end{proof}
\end{theorem}

\begin{example} \label{ex:coordinate-polynomial-rational-norm-continuity} \hlt{Coordinate, Polynomial, Rational, and Norm Maps}\\
Coordinate functions on $\R^k$ are continuous; hence polynomials are continuous, and rational functions are continuous where defined. Also $\mathbf{x}\mapsto\norm{\mathbf{x}}_2$ is continuous since $\bigl|\norm{\mathbf{x}}_2-\norm{\mathbf{y}}_2\bigr|\leq\norm{\mathbf{x}-\mathbf{y}}_2$.\\
Thus $p\mapsto\norm{f(p)}_2$ is continuous whenever $f:X\to\R^k$ is, by Theorem \ref{thm:continuous-composition}.
\end{example}

\begin{definition} \label{def:bounded-mapping-into-rk} \hlt{Bounded Mapping}\\
A mapping $f:E\to\R^k$ is \hlt{bounded} if there exists $M\in\R$ such that $\norm{f(x)}_2\leq M$ for every $x\in E$.
\end{definition}

\begin{theorem} \label{thm:continuous-image-compact} \hlt{Continuous Images of Compact Sets Are Compact}\\
If $f:X\to Y$ is continuous and $X$ is compact, then $f(X)$ is compact.
\begin{proof}
Let $\{V_\alpha\}$ cover $f(X)$. Then $\{f^{-1}(V_\alpha)\}$ is an open cover of $X$ by Theorem \ref{thm:continuity-open-preimages}.\\
Choose $\alpha_1,\ldots,\alpha_n$ with $X\subseteq\bigcup_{i=1}^n f^{-1}(V_{\alpha_i})$; then $f(X)\subseteq\bigcup_{i=1}^n V_{\alpha_i}$.\\
\hlt{Structural Commentary}: Pull back, take a finite subcover, push forward.\\
\hlt{Key Insight}: Compactness is preserved by continuous images because inverse images preserve openness.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:continuous-rk-image-compact-closed-bounded} \hlt{Continuous Images into $\R^k$ Are Closed and Bounded}\\
If $f:X\to\R^k$ is continuous and $X$ is compact, then $f(X)$ is closed and bounded. In particular, $f$ is bounded.
\begin{proof}
By Theorem \ref{thm:continuous-image-compact}, $f(X)$ is compact.\\
By Theorem \ref{thm:heine-borel-rk}, compact subsets of $\R^k$ are closed and bounded; hence $f$ is bounded.\\
\hlt{Structural Commentary}: Compactness in the domain becomes closed-bounded geometry in Euclidean range.\\
\hlt{Key Insight}: Continuous functions on compact spaces cannot escape to infinity.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:extreme-value-theorem-compact} \hlt{Extreme Value Theorem}\\
Let $f:X\to\R$ be continuous, let $X$ be compact.\\
If $M\defeq\sup_{p\in X}f(p)$, $m\defeq\inf_{p\in X}f(p)$, then $\exists$ $p,q\in X$ such that $f(p)=M$ and $f(q)=m$.
\begin{proof}
By Corollary \ref{cor:continuous-rk-image-compact-closed-bounded}, $f(X)$ is closed and bounded in $\R$. Thus $\sup f(X)$ and $\inf f(X)$ exist and belong to $f(X)$ by Theorems \ref{thm:supremum-belongs-to-closure} and \ref{thm:lub-property-gives-glb-property}. Hence $M=f(p)$ and $m=f(q)$ for some $p,q\in X$.\\
\hlt{Structural Commentary}: Compactness turns suprema and infima into attained values.\\
\hlt{Key Insight}: Closed bounded images contain their endpoints.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:compact-bijection-has-continuous-inverse} \hlt{Compact Bijective Maps Have Continuous Inverses}\\
If $f:X\to Y$ is a continuous bijection from compact $X$ to a metric space $Y$, then $f^{-1}:Y\to X$ is continuous.
\begin{proof}
It suffices, by Theorem \ref{thm:continuity-open-preimages}, to show that $f(V)$ is open in $Y$ for every open $V\subseteq X$. If $V$ is open, then $V^c$ is compact by Theorem \ref{thm:closed-subsets-of-compact-sets-compact}, so $f(V^c)$ is compact, hence closed by Theorems \ref{thm:continuous-image-compact} and \ref{thm:compact-subsets-of-metric-spaces-closed}.\\
Since $f$ is bijective, $f(V)=Y\setminus f(V^c)$ is open.\\
\hlt{Structural Commentary}: Compactness turns closed complements into closed images.\\
\hlt{Key Insight}: A continuous bijection from compact domain has no room for inverse discontinuity.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:continuity-controls-preimages-not-images} \hlt{Preimages, Not Forward Images}\\
Theorem \ref{thm:continuity-open-preimages} is about inverse images. A continuous map need not send open sets to open sets or closed sets to closed sets; compactness is one hypothesis that recovers closedness of images through Theorem \ref{thm:continuous-image-compact}.
\end{remark}

\begin{theorem} \label{thm:continuous-image-connected} \hlt{Continuous Images of Connected Sets Are Connected}\\
If $f:X\to Y$ is continuous and $E\subseteq X$ is connected, then $f(E)$ is connected.
\begin{proof}
If $f(E)=A\cup B$ with $A,B$ nonempty separated, set $G\defeq E\cap f^{-1}(A)$ and $H\defeq E\cap f^{-1}(B)$. Then $E=G\cup H$ and $G,H$ are nonempty. Since $G\subseteq f^{-1}(\overline{A})$ and $f^{-1}(\overline{A})$ is closed, $\overline{G}\subseteq f^{-1}(\overline{A})$; hence $\overline{G}\cap H=\emptyset$. Similarly $G\cap\overline{H}=\emptyset$, so $G,H$ separate $E$, a contradiction.\\
\hlt{Structural Commentary}: A separation of the image pulls back to a separation of the domain.\\
\hlt{Key Insight}: Continuous maps cannot create a gap in the image of a connected set.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:intermediate-value-theorem} \hlt{Intermediate Value Theorem}\\
If $f:[a,b]\to\R$ is continuous and $c$ lies strictly between $f(a)$ and $f(b)$, then $f(x)=c$ for some $x\in(a,b)$.
\begin{proof}
The interval $[a,b]$ is connected by Theorem \ref{thm:connected-subsets-of-real-line-interval-property}, so $f([a,b])$ is connected by Theorem \ref{thm:continuous-image-connected}.\\
Connected subsets of $\R$ contain every value between any two of their points; hence $c=f(x)$ for some $x\in[a,b]$. Since $c\neq f(a),f(b)$, $x\in(a,b)$.\\
\hlt{Structural Commentary}: The intermediate value theorem is connectedness of intervals transported by continuity.\\
\hlt{Key Insight}: Continuous real functions on intervals cannot jump over a value.
\end{proof}
\end{theorem}

\begin{method} \label{meth:ivt-bisection-nested-intervals} \hlt{Bisection Proof of the Intermediate Value Theorem}\\
Let $f:[a,b]\to\R$ be continuous and let $c$ lie between $f(a)$ and $f(b)$.\\
Replacing $(f,c)$ by $(-f,-c)$ if needed, assume $f(a)\leq c\leq f(b)$.\\
If $c=f(a)$ or $c=f(b)$, take the corresponding endpoint; otherwise $f(a)<c<f(b)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $[a_1,b_1]\defeq[a,b]$.
\item Given $[a_n,b_n]$ with $f(a_n)\leq c\leq f(b_n)$, set $m_n\defeq(a_n+b_n)/2$.\\
If $f(m_n)\geq c$, set $[a_{n+1},b_{n+1}]\defeq[a_n,m_n]$; otherwise set $[a_{n+1},b_{n+1}]\defeq[m_n,b_n]$.
\end{enumerate}
Then, for all $n$, $[a_{n+1},b_{n+1}]\subseteq[a_n,b_n]$, $f(a_n)\leq c\leq f(b_n)$, and $b_n-a_n=(b-a)/2^{n-1}$.\\
By Theorem \ref{thm:nested-closed-intervals-intersect}, choose $x_0\in\bigcap_n[a_n,b_n]$. Since $b_n-a_n\to0$, the intersection is $\{x_0\}$ and $a_n,b_n\to x_0$.\\
Continuity gives $f(a_n),f(b_n)\to f(x_0)$, so $f(x_0)\leq c\leq f(x_0)$; hence $f(x_0)=c$.\\
If $f(a)<c<f(b)$, then $x_0\in(a,b)$.\\
\hlt{Key Insight}: Bisection keeps the target value bracketed while nested intervals collapse to one point.
\end{method}

\begin{definition} \label{def:path-connected-set} \hlt{Path-Connected Set}\\
A set $E$ in a metric space is \hlt{path-connected} if for every $p,q\in E$ there is a continuous map $\gamma:[a,b]\to E$ with $\gamma(a)=p$ and $\gamma(b)=q$. Such a map $\gamma$ is a \hlt{path} from $p$ to $q$.
\end{definition}

\begin{theorem} \label{thm:path-connected-implies-connected} \hlt{Path-Connected Sets Are Connected}\\
Every path-connected set is connected.
\begin{proof}
Suppose $E=A\cup B$ with $A,B$ nonempty separated.\\
Choose $p\in A$, $q\in B$, and a path $\gamma:[a,b]\to E$ from $p$ to $q$. Since $[a,b]$ is connected, so is $\gamma([a,b])$.\\
But $\gamma([a,b])$ meets both $A$ and $B$, which separate it, a contradiction.\\
\hlt{Structural Commentary}: A path transports connectedness from an interval to the set.\\
\hlt{Key Insight}: To connect two points by a path, the path image cannot be split into two separated pieces.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:convex-sets-path-connected} \hlt{Convex Sets Are Path-Connected}\\
Every convex subset of $\R^k$ is path-connected.
\begin{proof}
For $\mathbf{x},\mathbf{y}\in E$, set $\gamma(t)=(1-t)\mathbf{x}+t\mathbf{y}$ on $[0,1]$. Convexity gives $\gamma([0,1])\subseteq E$, with $\gamma(0)=\mathbf{x}$ and $\gamma(1)=\mathbf{y}$.\\
Since the coordinates of $\gamma$ are affine in $t$, $\gamma$ is continuous. Hence $E$ is path-connected.\\
\hlt{Structural Commentary}: Convexity supplies the whole straight path between any two points.\\
\hlt{Key Insight}: In Euclidean space, line segments are continuous paths.
\end{proof}
\end{corollary}
