\subsection{Sequences, Countability, and Metrization}

Countability axioms control when topological questions can be reduced to sequences and countable tests. They also separate metric phenomena from genuinely non-metric topologies.

\begin{definition} \label{def:topological-sequence-convergence} \hlt{Sequence Convergence and Sequential Compactness}\\
Let $X$ be a topological space, and let $\mathcal{N}_X(x)$ denote the neighborhoods of $x\in X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(x_n)$ \hlt{converges} to $x$, written $x_n\to x$, iff $\forall U\in\mathcal{N}_X(x)$ $\exists N\in\N$ such that $n\geq N\Rightarrow x_n\in U$.
\item $X$ is \hlt{sequentially compact} iff every sequence $(x_n)\subseteq X$ has a convergent subsequence $x_{n_k}\to x\in X$.
\end{enumerate}
\hlt{Key Insight}: Sequential compactness is compactness tested by subsequences.
\end{definition}

\begin{proposition} \label{prop:hausdorff-sequence-limits-unique} \hlt{Hausdorff Sequence Limits Are Unique}\\
Let $X$ be a topological space and let $(x_n)\subseteq X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $X$ is Hausdorff, then $(x_n)$ has at most one limit.
\item If $X$ has the trivial topology $\{\emptyset,X\}$, then $x_n\to x$ for every $x\in X$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] If $x_n\to x$ and $x_n\to y$ with $x\neq y$, choose disjoint neighborhoods $U\in\mathcal{N}_X(x)$ and $V\in\mathcal{N}_X(y)$.\\
Eventually $x_n\in U$ and $x_n\in V$, contradicting $U\cap V=\emptyset$.
\item[$(ii)$] In the trivial topology, $\mathcal{N}_X(x)=\{X\}$ for every $x\in X$, and every sequence is eventually in $X$.
\end{enumerate}
\hlt{Structural Commentary}: Hausdorff neighborhoods separate competing limits.\\
\hlt{Key Insight}: A sequence tail cannot lie in two disjoint neighborhoods.
\end{proof}
\end{proposition}

\begin{definition} \label{def:topological-countability-axioms} \hlt{Countability Axioms}\\
Let $(X,\tau)$ be a topological space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $X$ is \hlt{first countable} iff $\forall x\in X$ $\exists$ countable $\mathcal{B}_x\subseteq\mathcal{N}_X(x)$ such that every $U\in\mathcal{N}_X(x)$ contains some $B\in\mathcal{B}_x$.
\item $X$ is \hlt{second countable} iff $\tau$ has a countable basis.
\item $X$ is \hlt{separable} iff $\exists$ countable $D\subseteq X$ with $\overline{D}=X$, equivalently every nonempty $U\in\tau$ meets $D$.
\end{enumerate}
\hlt{Key Insight}: Countability turns topology into countable local, global, or dense tests.
\end{definition}

\begin{example} \label{ex:metric-spaces-first-countable} \hlt{Metric Spaces Are First Countable}\\
Every metric space is first countable: $\forall x$, balls $\{B_{1/n}(x):n\geq1\}$ form a countable neighborhood basis at $x$.
\end{example}

\begin{proposition} \label{prop:second-countable-implies-first-countable} \hlt{Second Countable Implies First Countable}\\
Let $(X,\tau)$ be second countable. Then $X$ is first countable.
\begin{proof} Let $\mathcal{B}$ be a countable basis for $\tau$. For $x\in X$, set $\mathcal{B}_x\defeq\{B\in\mathcal{B}:x\in B\}$.\\
Then $\mathcal{B}_x$ is countable, and for every $U\in\mathcal{N}_X(x)$, some $B\in\mathcal{B}_x$ satisfies $x\in B\subseteq U$.\\
\hlt{Structural Commentary}: Restrict the global basis to sets containing $x$.\\
\hlt{Key Insight}: Countable global tests give countable local tests.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:second-countable-implies-separable} \hlt{Second Countable Consequences}\\
If $(X,\tau)$ is second countable, then there is a countable $D\subseteq X$ with $\overline{D}=X$.\\
Moreover, every subspace of a second countable space is second countable.
\begin{proof}
Let $\mathcal{B}=\{B_n:n\in I\}$ be a countable basis, with $I\subseteq\N$.\\
For each $n\in I$ with $B_n\neq\emptyset$, choose $x_n\in B_n$ and set $D=\{x_n:n\in I,\ B_n\neq\emptyset\}$. Then $D$ is countable.\\
If $U\in\tau\setminus\{\emptyset\}$, choose $x\in U$; some $B_n\in\mathcal{B}$ satisfies $x\in B_n\subseteq U$, so $x_n\in D\cap U$. Hence $\overline{D}=X$.\\
If $Y\subseteq X$, then $\{B_n\cap Y:n\in I\}$ is a countable basis for the subspace topology on $Y$.\\
\hlt{Structural Commentary}: $\mathcal{B}$ countable gives countable $D$ meeting every $U\in\tau\setminus\{\emptyset\}$.\\
\hlt{Key Insight}: $D$ is dense iff $D\cap U\neq\emptyset$ for every $U\in\tau\setminus\{\emptyset\}$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:metric-separable-countable-base} \hlt{Separable Metric Spaces and Countable Bases}\\
Let $(X,d)$ be a metric space, and let $\tau_d$ be its metric topology. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\exists$ countable $D\subseteq X$ such that $\overline{D}^{\,X}=X$.
\item $\exists$ countable $\mathcal{B}\subseteq\tau_d$ such that $\mathcal{B}$ is a basis for $\tau_d$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ Let $D\subseteq X$ be countable with $\overline{D}^{\,X}=X$, and set $\mathcal{B}\defeq\{B_q(x):x\in D,\ q\in\Q,\ q>0\}$.\\
Then $\mathcal{B}$ is countable. For $U\in\tau_d$ and $y\in U$, choose $r>0$ with $B_r(y)\subseteq U$, then choose $q\in\Q$ with $0<q<r/2$, $x\in D\cap B_q(y)$. Then $y\in B_q(x)\subseteq B_r(y)\subseteq U$, since $z\in B_q(x)$ implies $d(z,y)<2q<r$. QED.
\item[$(ii)$]$\Rightarrow (i)$ Let $\mathcal{B}=\{U_n:n\in I\}$ be a countable basis for $\tau_d$, with $I\subseteq\N$.\\
Choose $x_n\in U_n$ for each nonempty $U_n$, and set $D\defeq\{x_n:U_n\neq\emptyset\}$. Then $D$ is countable.\\
If $\emptyset\neq U\in\tau_d$, choose $y\in U$ and $n\in I$ with $y\in U_n\subseteq U$; then $x_n\in D\cap U$.\\
Hence every nonempty open set meets $D$, so $\overline{D}^{\,X}=X$.
\end{enumerate}
\hlt{Structural Commentary}: In metric spaces, one countable dense set produces all rational-radius local tests.\\
\hlt{Key Insight}: Separability and second countability coincide for metric spaces.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:subspaces-of-separable-metric-spaces-separable} \hlt{Subspaces of Separable Metric Spaces}\\
Let $(X,d)$ be a separable metric space and let $E\subseteq X$ with the subspace metric $d|_{E\times E}$. Then $E$ is separable.
\begin{proof}
By Proposition \ref{prop:metric-separable-countable-base}, the metric topology $\tau_d$ has a countable basis $\mathcal{B}$. Set $\mathcal{B}_E\defeq\{B\cap E:B\in\mathcal{B}\}$. Then $\mathcal{B}_E$ is countable. If $U$ is open in $E$ and $x\in U$, then $U=E\cap G$ for some $G\in\tau_d$; choose $B\in\mathcal{B}$ with $x\in B\subseteq G$. Hence $x\in B\cap E\subseteq U$, so $\mathcal{B}_E$ is a countable basis for $E$. By Proposition \ref{prop:metric-separable-countable-base}, $E$ is separable.\\
\hlt{Structural Commentary}: A countable ambient basis restricts to a countable subspace basis.\\
\hlt{Key Insight}: Metric separability passes to all subspaces through countable bases.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:euclidean-subsets-separable} \hlt{Euclidean Subsets Are Separable}\\
Let $k\in\N$ and let $E\subseteq\R^k$ have the subspace Euclidean metric. Then $E$ is separable.
\begin{proof} The set $\Q^k$ is countable and dense in $\R^k$.\\
For $x\in\R^k$ and $r>0$, choose $q=(q_1,\ldots,q_k)\in\Q^k$ with $\abs{x_i-q_i}<r/\sqrt{k}$ for each $i$, so $\norm{x-q}_2<r$.\\
Thus $\R^k$ is separable, and Proposition \ref{prop:subspaces-of-separable-metric-spaces-separable} gives that $E$ is separable.\\
\hlt{Structural Commentary}: Rational grids are countable dense nets.\\
\hlt{Key Insight}: Euclidean separability passes to subspaces.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:first-countable-sequential-closure} \hlt{First Countable Sequential Closure}\\
Let $X$ be first countable, let $E\subseteq X$, and let $x\in X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $x\in\overline{E}$ iff $\exists (x_n)\subseteq E$ with $x_n\to x$.
\item $E$ is closed iff every convergent sequence in $E$ has all its limits in $E$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Let $\{B_n:n\geq1\}$ be a countable neighborhood basis at $x$.\\
Since $x\in\overline{E}$ and $\bigcap\limits_{j=1}^n B_j\in\mathcal{N}_X(x)$, choose $x_n\in E\cap\bigcap\limits_{j=1}^n B_j$. Then $x_n\to x$.
\item[$(i)$] ($\Leftarrow$) If $(x_n)\subseteq E$ and $x_n\to x$, then every neighborhood of $x$ contains some $x_n\in E$, so $x\in\overline{E}$.
\item[$(ii)$] ($\Rightarrow$) If $E$ is closed and $(x_n)\subseteq E$ with $x_n\to x$, then $x\in\overline{E}=E$ by $(i)$.
\item[$(ii)$] ($\Leftarrow$) If every convergent sequence in $E$ has all its limits in $E$, then $(i)$ gives $\overline{E}\subseteq E$. Hence $E=\overline{E}$, QED.
\end{enumerate}
\hlt{Structural Commentary}: Countable local bases produce one testing sequence.\\
\hlt{Key Insight}: In first countable spaces, closure is sequential closure.
\end{proof}
\end{proposition}

\begin{example} \label{ex:uncountable-cofinite-topology-not-first-countable} \hlt{Uncountable Cofinite Topology}\\
Let $X$ be uncountable and let $\tau\defeq\{\emptyset\}\cup\{U\subseteq X:X\setminus U\text{ is finite}\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\tau$ is a topology on $X$.
\item $(X,\tau)$ is $T_1$.
\item $(X,\tau)$ is not first countable.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\emptyset,X\in\tau$. Finite intersections and arbitrary unions of cofinite sets are cofinite, so $\tau$ is a topology.
\item[$(ii)$] For $x\in X$, $X\setminus\{x\}\in\tau$, so $\{x\}$ is closed. Hence $(X,\tau)$ is $T_1$.
\item[$(iii)$] Fix $x\in X$. If $\{B_n:n\geq1\}$ were a countable neighborhood basis at $x$, then $C\defeq\{x\}\cup\bigcup_n(X\setminus B_n)$ is countable. Choose $y\in X\setminus C$. Then $U\defeq X\setminus\{y\}\in\mathcal{N}_X(x)$, but $y\in B_n$ for all $n$, so no $B_n\subseteq U$.
\end{enumerate}
\hlt{Key Insight}: Cofinite local bases cannot be countable.
\end{proof}
\end{example}

\begin{definition} \label{def:lindelof-space}\label{def:countably-compact-space} \hlt{Lindel\"of and Countably Compact Spaces}\\
Let $(X,\tau)$ be a topological space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $X$ is \hlt{Lindel\"of} iff every open cover $\{O_\alpha\}_{\alpha\in A}\subseteq\tau$ has a countable subcover $\{O_\alpha\}_{\alpha\in J}$, $J\subseteq A$.
\item $X$ is \hlt{countably compact} iff every countable open cover $\{O_n\}_{n\in\N}\subseteq\tau$ has a finite subcover $\{O_n\}_{n\in J}$, $J\subseteq\N$.
\end{enumerate}
\hlt{Key Insight}: Covers reduce to countable or finite subcovers.
\end{definition}

\begin{proposition} \label{prop:countably-compact-frechet-intersection} \hlt{Frechet Intersection Test for Countable Compactness}\\
Let $X$ be a topological space. Then $X$ is countably compact iff every descending sequence $F_1\supseteq F_2\supseteq\cdots$ of nonempty closed subsets of $X$ satisfies $\bigcap\limits_{n\geq1}F_n\neq\emptyset$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $X$ is countably compact. If $\bigcap\limits_{n\geq1}F_n=\emptyset$, then $\{X\setminus F_n:n\geq1\}$ is a countable open cover.\\
A finite subcover exists; since $F_n$ descends, some $X\setminus F_N$ covers $X$, contradicting $F_N\neq\emptyset$.
\item[$(\Leftarrow)$] Let $\{O_n:n\geq1\}$ be a countable open cover with no finite subcover.\\
Then $F_n\defeq X\setminus\bigcup\limits_{j=1}^n O_j$ is nonempty, closed, and descending, while $\bigcap\limits_{n\geq1}F_n=\emptyset$, a contradiction.
\end{enumerate}
\hlt{Structural Commentary}: Complements turn countable covers into descending closed sets.\\
\hlt{Key Insight}: Countable compactness is nested closed intersection.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dini-countably-compact} \hlt{Dini's Theorem on Countably Compact Spaces}\\
Let $X$ be countably compact, and let $f_n:X\to\R$ be continuous with $f_n(x)\downarrow0$ for every $x\in X$.\\
Then $f_n\to0$ uniformly: $\forall\varepsilon>0$ $\exists N$ such that $0\leq f_n(x)<\varepsilon$ for all $x\in X$ and $n\geq N$.
\begin{proof} Fix $\varepsilon>0$ and set $O_n\defeq\{x\in X:f_n(x)<\varepsilon\}$. Then $O_n$ is open, $O_n\subseteq O_{n+1}$, and $\bigcup\limits_{n\geq1}O_n=X$.\\
Countable compactness gives $X=\bigcup\limits_{j=1}^m O_{n_j}$; with $N\defeq\max_j n_j$, increasingness gives $O_N=X$.\\
Hence $0\leq f_n(x)\leq f_N(x)<\varepsilon$ for all $x\in X$ and $n\geq N$.\\
\hlt{Structural Commentary}: Increasing sublevel covers reduce finite subcovers to one index.\\
\hlt{Key Insight}: Monotone pointwise convergence becomes uniform convergence.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:second-countable-implies-lindelof} \hlt{Second Countable Implies Lindel\"of}\\
Let $(X,\tau)$ be second countable. Then $X$ is Lindel\"of: every open cover of $X$ has a countable subcover.
\begin{proof} Let $\mathcal{B}=\{B_n:n\in I\}$ be a countable basis, and let $\mathcal{U}\subseteq\tau$ cover $X$. For $I_0\defeq\{n\in I:\exists U\in\mathcal{U},\ B_n\subseteq U\}$, choose $U_n\in\mathcal{U}$ with $B_n\subseteq U_n$ for each $n\in I_0$. The family $\{U_n:n\in I_0\}$ is countable.\\
If $x\in X$, choose $U\in\mathcal{U}$ with $x\in U$ and $B_n\in\mathcal{B}$ with $x\in B_n\subseteq U$; then $n\in I_0$ and $x\in U_n$.\\
\hlt{Structural Commentary}: Basis elements refine arbitrary open covers.\\
\hlt{Key Insight}: A countable basis yields a countable subcover.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:second-countable-countably-compact-compact} \hlt{Second Countable Countable Compactness Is Compactness}\\
Let $X$ be second countable. Then $X$ is compact iff $X$ is countably compact.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\{O_n\}_{n\in\N}$ covers $X$, compactness gives finite $J\subseteq\N$ with $X=\bigcup\limits_{n\in J}O_n$.
\item[$(\Leftarrow)$] Let $\mathcal{U}=\{O_\alpha\}_{\alpha\in A}$ cover $X$. By Proposition \ref{prop:second-countable-implies-lindelof}, $\exists$ countable $J\subseteq A$ with $X=\bigcup\limits_{\alpha\in J}O_\alpha$.\\
Countable compactness gives finite $J_0\subseteq J$ with $X=\bigcup\limits_{\alpha\in J_0}O_\alpha$.
\end{enumerate}
\hlt{Structural Commentary}: Second countability reduces covers to countable subcovers.\\
\hlt{Key Insight}: Countable compactness then makes the subcover finite.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:second-countable-compact-iff-sequentially-compact} \hlt{Second Countable Compactness and Sequential Compactness}\\
Let $X$ be a second countable topological space. Then $X$ is compact iff $X$ is sequentially compact.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $X$ is compact. Let $(x_n)\subseteq X$ and $F_n\defeq\overline{\{x_k:k\geq n\}}$. Since $F_1\supseteq F_2\supseteq\cdots$ are nonempty closed, compactness gives $x\in\bigcap\limits_{n\geq1}F_n$. By first countability, choose a decreasing neighborhood basis $(B_j)$ at $x$. With $n_0=0$, choose $n_j>n_{j-1}$ and $x_{n_j}\in B_j$ from $x\in F_{n_{j-1}+1}$. Then $x_{n_j}\to x$.
\item[$(\Leftarrow)$] Assume $X$ is sequentially compact. By Proposition \ref{prop:second-countable-countably-compact-compact}, it is enough to prove countable compactness. If $\{O_n\}_{n\geq1}$ is a countable open cover with no finite subcover, choose $x_n\in X\setminus\bigcup\limits_{j=1}^n O_j$. Sequential compactness gives $x_{n_k}\to x\in O_N$ for some $N$. For $k$ large, $n_k\geq N$ and $x_{n_k}\in O_N$, contradicting $x_{n_k}\notin O_N$.
\end{enumerate}
\hlt{Structural Commentary}: Closed tails give a cluster point; local countability extracts a subsequence.\\
\hlt{Key Insight}: Countable covers and bases make compactness sequential.
\end{proof}
\end{theorem}

\begin{definition} \label{def:sorgenfrey-line} \hlt{Sorgenfrey Line}\\
Let $\mathcal{B}_S\defeq\{[a,b):a<b\}$ on $\R$. The \hlt{Sorgenfrey line} is $\R_S\defeq(\R,\tau_S)$, where $\tau_S$ is the topology generated by $\mathcal{B}_S$.
\end{definition}

\begin{example} \label{ex:sorgenfrey-line-countability-tests} \hlt{Sorgenfrey Line Countability Tests}\\
The Sorgenfrey line $\R_S$ is first countable and separable, but not second countable; hence it is not metrizable.
\begin{proof}
For $x\in[a,b)\cap[c,d)$, choose $e$ with $x<e\leq\min\{b,d\}$; then $x\in[x,e)\subseteq[a,b)\cap[c,d)$, so $\mathcal{B}_S$ is a basis. Also $\{[x,x+1/n):n\geq1\}\subseteq\mathcal{N}_{\R_S}(x)$ is a local basis, and $\Q\cap[a,b)\neq\emptyset$ for every $[a,b)\neq\emptyset$.\\
Hence $\R_S$ is first countable and separable.\\
If $\mathcal{C}$ is a basis for $\R_S$, choose $C_x\in\mathcal{C}$ with $x\in C_x\subseteq[x,x+1)$. For $x<y$, $x\in C_x\setminus C_y$, so $C_x\neq C_y$.\\
Thus $x\mapsto C_x$ is injective, so no countable $\mathcal{C}$ exists. Hence $\R_S$ is not second countable.\\
If $\R_S$ were metrizable, separability and Proposition \ref{prop:metric-separable-countable-base} would force second countability.\\
\hlt{Structural Commentary}: Half-open neighborhoods encode left endpoints.\\
\hlt{Key Insight}: Local countability does not imply one countable global basis.
\end{proof}
\end{example}

\begin{definition} \label{def:moore-plane} \hlt{Moore Plane}\\
Let $\mathbb{M}\defeq\{(x,y)\in\R^2:y\geq0\}$ and $L\defeq\R\times\{0\}$. The \hlt{Moore plane} is $\mathbb{M}$ with basis consisting of:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $B_{\mathrm{euc}}(p,r)\subseteq\mathbb{M}$ for $p\in\mathbb{M}\setminus L$ and $r>0$;
\item $N(x,r)\defeq\{(x,0)\}\cup B_{\mathrm{euc}}((x,r),r)$ for $(x,0)\in L$ and $r>0$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:moore-plane-countability-and-metrizability} \hlt{Moore Plane Countability Tests}\\
The Moore plane $\mathbb{M}$ is Hausdorff and separable, but not second countable; hence it is not metrizable.
\begin{proof}
The basis and Hausdorff axioms follow by shrinking Euclidean disks and tangent neighborhoods $N(x,r)$. Let $D\defeq\Q\times(\Q\cap(0,\infty))$; then every basic neighborhood meets $D$, so $\overline{D}=\mathbb{M}$. Thus $\mathbb{M}$ is separable.\\
For $(x,0)\in L$, $N(x,r)\cap L=\{(x,0)\}$, so $L$ is an uncountable discrete subspace and hence not separable. If $\mathbb{M}$ were second countable, then $L$ would be second countable and separable by Proposition \ref{prop:second-countable-implies-separable}, a contradiction. If $\mathbb{M}$ were metrizable, separability and Proposition \ref{prop:metric-separable-countable-base} would force second countability.\\
\hlt{Structural Commentary}: Dense rational interior; discrete uncountable boundary.\\
\hlt{Key Insight}: The boundary is an uncountable discrete obstruction.
\end{proof}
\end{example}

\begin{definition} \label{def:metrizable-topological-space} \hlt{Metrizable Space}\\
A topological space is \hlt{metrizable} if its topology is induced by some metric.
\end{definition}

\begin{theorem} \label{thm:urysohn-metrization-second-countable} \hlt{Urysohn Metrization Theorem}\\
Let $(X,\tau)$ be second countable. Then $\tau$ is induced by a metric iff $(X,\tau)$ is normal.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Let $d$ induce $\tau$. Then $X$ is $T_1$, since $X\setminus\{x\}=\bigcup\limits_{y\neq x}B_d\left(y,\frac{d(x,y)}{2}\right)$. For disjoint nonempty closed $F,G\subseteq X$, set $u(x)\defeq d(x,F)$ and $v(x)\defeq d(x,G)$. Since $u,v$ are continuous, $U\defeq\{u<v\}$ and $V\defeq\{v<u\}$ are disjoint open sets with $F\subseteq U$ and $G\subseteq V$. Empty cases are trivial, so $(X,\tau)$ is normal.
\item[$(\Leftarrow)$] Assume $(X,\tau)$ is normal. Let $\{U_n:n\in\N\}$ be a basis and $A\defeq\{(n,m)\in\N^2:\overline{U_n}\subseteq U_m\}$.\\
For $(n,m)\in A$, choose $f_{n,m}:X\to[0,1]$ by Theorem \ref{thm:urysohn-lemma-normal-space}, with $f_{n,m}=0$ on $\overline{U_n}$, $f_{n,m}=1$ on $X\setminus U_m$.\\
Set $\rho(x,y)\defeq\sum\limits_{(n,m)\in A}\frac{1}{2^{n+m}}\abs{f_{n,m}(x)-f_{n,m}(y)}$, a $\tau$-continuous pseudometric by uniform convergence.\\
For every $G\in\tau$ and $x\in G$, Proposition \ref{prop:normality-nested-neighborhoods} and the basis give $n,m$ with $x\in U_n\subseteq\overline{U_n}\subseteq U_m\subseteq G$.\\
If $y\notin U_m$, then $\rho(x,y)\geq\frac{1}{2^{n+m}}$, so $B_\rho\left(x,\frac{1}{2^{n+m+1}}\right)\subseteq U_m\subseteq G$.\\
Taking $G=X\setminus\{y\}$ shows $\rho(x,y)>0$ for $x\neq y$. Hence $\rho$ is a metric and its topology is $\tau$.
\end{enumerate}
\hlt{Structural Commentary}: Urysohn functions separate nested basis pairs.\\
\hlt{Key Insight}: Countably many separators recover the topology as a metric.
\end{proof}
\end{theorem}
