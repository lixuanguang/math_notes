\subsection{Continuity and Topological Constructions}

Inverse images organize topological constructions; homeomorphisms preserve open-set structure.\\
Products, quotients, and weak topologies are characterized by the maps they make continuous.

\begin{definition} \label{def:topological-continuity} \hlt{Continuity Between Topological Spaces}\\
Let $(X,\mathcal{T}_X)$ and $(Y,\mathcal{T}_Y)$ be topological spaces, and let $f:X\to Y$.\\
For $x\in X$, set $\mathcal{N}_X(x)\defeq\{U\in\mathcal{T}_X:x\in U\}$. For $y\in Y$, set $\mathcal{N}_Y(y)\defeq\{O\in\mathcal{T}_Y:y\in O\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $x_0\in X$, $f$ is \hlt{continuous at $x_0$} iff $(\forall O\in\mathcal{N}_Y(f(x_0)))(\exists U\in\mathcal{N}_X(x_0))\,f(U)\subseteq O$.
\item $f$ is \hlt{continuous} iff $(\forall x\in X)$, $f$ is continuous at $x$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:topological-continuity-open-preimages} \hlt{Continuity by Open Preimages}\\
Let $(X,\mathcal{T}_X)$ and $(Y,\mathcal{T}_Y)$ be topological spaces, and let $f:X\to Y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f$ is continuous iff $(\forall O\in\mathcal{T}_Y)\,f^{-1}(O)\in\mathcal{T}_X$.
\item Let $A\subseteq X$ carry the subspace topology $\mathcal{T}_A$. If $f$ is continuous, then $f|_A:(A,\mathcal{T}_A)\to(Y,\mathcal{T}_Y)$ is continuous.
\item Let $(Z,\mathcal{T}_Z)$ be topological space, $g:(Y,\mathcal{T}_Y)\to(Z,\mathcal{T}_Z)$ continuous. If $f$ continuous, then $g\circ f:(X,\mathcal{T}_X)\to(Z,\mathcal{T}_Z)$ continuous.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) For $O\in\mathcal{T}_Y$ and $x\in f^{-1}(O)$, choose $U\in\mathcal{N}_X(x)$ with $f(U)\subseteq O$ by continuity at $x$.\\
The relations $x\in U\subseteq f^{-1}(O)$ imply $f^{-1}(O)\in\mathcal{T}_X$ by Proposition \ref{prop:open-neighborhood-test-topological}.
\item[$(i)$] ($\Leftarrow$) Assume $(\forall O\in\mathcal{T}_Y)\,f^{-1}(O)\in\mathcal{T}_X$. For $x_0\in X$ and $O\in\mathcal{N}_Y(f(x_0))$, set $U\defeq f^{-1}(O)$.\\
Then $U\in\mathcal{N}_X(x_0)$ and $f(U)\subseteq O$, so $f$ is continuous at $x_0$ by Definition \ref{def:topological-continuity}(i). Thus $f$ is continuous by Definition \ref{def:topological-continuity}(ii).
\item[$(ii)$] For $O\in\mathcal{T}_Y$, $f^{-1}(O)\in\mathcal{T}_X$ by (i), $(f|_A)^{-1}(O)=A\cap f^{-1}(O)\in\mathcal{T}_A$ by definition of $\mathcal{T}_A$; $f|_A$ is continuous by (i).
\item[$(iii)$] For $W\in\mathcal{T}_Z$, $g^{-1}(W)\in\mathcal{T}_Y$ by (i), $(g\circ f)^{-1}(W)=f^{-1}(g^{-1}(W))\in\mathcal{T}_X$ by (i) for $f$; $g\circ f$ continuous by (i).
\end{enumerate}
\hlt{Structural Commentary}: Neighborhood witnesses build opens; intersection and nesting preserve them.\\
\hlt{Key Insight}: Continuity is exactly preservation of openness under inverse images.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:metric-and-topological-continuity-agree} \hlt{Metric and Topological Continuity Agree}\\
Let $(X,d_X),(Y,d_Y)$ be metric spaces with metric topologies $\tau_X,\tau_Y$, and let $f:X\to Y$.\\
Then $f$ is metrically continuous iff $f:(X,\tau_X)\to(Y,\tau_Y)$ is topologically continuous.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $f$ is metrically continuous, then $(\forall O\in\tau_Y)\,f^{-1}(O)\in\tau_X$ by Theorem \ref{thm:continuity-open-preimages}.\\
Hence $f:(X,\tau_X)\to(Y,\tau_Y)$ is continuous by Proposition \ref{prop:topological-continuity-open-preimages}(i).
\item[$(\Leftarrow)$] Assume $f:(X,\tau_X)\to(Y,\tau_Y)$ is continuous. Then $(\forall O\in\tau_Y)\,f^{-1}(O)\in\tau_X$ by Proposition \ref{prop:topological-continuity-open-preimages}(i).\\
Hence $f$ is metrically continuous by Theorem \ref{thm:continuity-open-preimages}.
\end{enumerate}
\hlt{Structural Commentary}: The common open-preimage condition identifies the two notions.\\
\hlt{Key Insight}: Their open-preimage conditions coincide.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:continuity-tested-on-subbasis} \hlt{Continuity Tested on a Subbasis}\\
Let $(X,\mathcal{T}_X)$ be topological space, $\mathcal{S}$ be a subbasis generating $\mathcal{T}_Y$ on $Y$. For $f:X\to Y$, $f$ is continuous iff $(\forall S\in\mathcal{S})\,f^{-1}(S)\in\mathcal{T}_X$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] For $S\in\mathcal{S}\subseteq\mathcal{T}_Y$, $f^{-1}(S)\in\mathcal{T}_X$ by Proposition \ref{prop:topological-continuity-open-preimages}(i).
\item[$(\Leftarrow)$] Assume $(\forall S\in\mathcal{S})\,f^{-1}(S)\in\mathcal{T}_X$, and let $\mathcal{B}_{\mathcal{S}}$ be the basis of finite intersections from $\mathcal{S}$.\\
For $B\in\mathcal{B}_{\mathcal{S}}$, choose $n\in\N$ and $S_1,\ldots,S_n\in\mathcal{S}$ with $B=\bigcap\limits_{j=1}^nS_j$ by Definition \ref{def:basis-for-topology}(iv).\\
Then $f^{-1}(B)=\bigcap\limits_{j=1}^nf^{-1}(S_j)\in\mathcal{T}_X$ by the hypothesis and Definition \ref{def:topological-space}(iii).\\
For $O\in\mathcal{T}_Y$, choose $\mathcal{U}\subseteq\mathcal{B}_{\mathcal{S}}$ with $O=\bigcup\mathcal{U}$ by Definition \ref{def:basis-for-topology}(iii),(iv).\\
Then $f^{-1}(O)=\bigcup\limits_{B\in\mathcal{U}}f^{-1}(B)\in\mathcal{T}_X$ by Definition \ref{def:topological-space}(ii). Hence $f$ is continuous by Proposition \ref{prop:topological-continuity-open-preimages}(i).
\end{enumerate}
\hlt{Structural Commentary}: Finite intersections reach the basis; unions reach the topology.\\
\hlt{Key Insight}: Inverse images preserve both topology-generating operations.
\end{proof}
\end{proposition}

\begin{definition} \label{def:topological-homeomorphism} \hlt{Homeomorphism}\\
Let $(X,\mathcal{T}_X)$ and $(Y,\mathcal{T}_Y)$ be topological spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A bijection $f:X\to Y$ is a \hlt{homeomorphism} iff $f$ and $f^{-1}$ are continuous.
\item The spaces $X$ and $Y$ are \hlt{homeomorphic}, written $X\simeq Y$, iff a homeomorphism $f:X\to Y$ exists.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:homeomorphism-preserves-open-and-closed-sets} \hlt{Homeomorphic Spaces Have the Same Open-Set Structure}\\
Let $(X,\mathcal{T}_X)$ and $(Y,\mathcal{T}_Y)$ be topological spaces, let $f:X\to Y$ be a homeomorphism, and let $E\subseteq X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E\in\mathcal{T}_X$ iff $f(E)\in\mathcal{T}_Y$.
\item $E$ is closed in $X$ iff $f(E)$ is closed in $Y$.
\end{enumerate}
\begin{proof}
The map $f$ is bijective, and $f,f^{-1}$ are continuous by Definition \ref{def:topological-homeomorphism}(i).
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) If $E\in\mathcal{T}_X$, then $f(E)=(f^{-1})^{-1}(E)\in\mathcal{T}_Y$ by Proposition \ref{prop:topological-continuity-open-preimages}(i) for $f^{-1}$.
\item[$(i)$] ($\Leftarrow$) If $f(E)\in\mathcal{T}_Y$, then $E=f^{-1}(f(E))$ by bijectivity. Hence $E\in\mathcal{T}_X$ by Proposition \ref{prop:topological-continuity-open-preimages}(i) for $f$.
\item[$(ii)$] ($\Rightarrow$) If $E$ is closed in $X$, then $f(X\setminus E)\in\mathcal{T}_Y$ by (i).\\
The identity $f(X\setminus E)=Y\setminus f(E)$ holds by bijectivity, so $f(E)$ is closed in $Y$ by Definition \ref{def:open-closed-neighborhood-topological}(ii).
\item[$(ii)$] ($\Leftarrow$) If $f(E)$ is closed in $Y$, then $Y\setminus f(E)\in\mathcal{T}_Y$ by Definition \ref{def:open-closed-neighborhood-topological}(ii). Thus $X\setminus E=f^{-1}(Y\setminus f(E))\in\mathcal{T}_X$ by Proposition \ref{prop:topological-continuity-open-preimages}(i).\\
Hence $E$ is closed in $X$ by Definition \ref{def:open-closed-neighborhood-topological}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Continuity of $f,f^{-1}$ transfers opens; bijectivity transfers complements.\\
\hlt{Key Insight}: A homeomorphism preserves and reflects the topology.
\end{proof}
\end{proposition}

\begin{definition} \label{def:product-topology} \hlt{Product Topology}\\
Let $(X,\mathcal{T}_X),(Y,\mathcal{T}_Y)$ and $(X_i,\mathcal{T}_i)$, $i\in I$, be topological spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $\mathcal{B}_{X\times Y}\defeq\{U\times V:U\in\mathcal{T}_X,\ V\in\mathcal{T}_Y\}$. The \hlt{product topology} on $X\times Y$ is generated by $\mathcal{B}_{X\times Y}$.
\item Set $\mathcal{B}_{\Pi}\defeq\{\prod\limits_{i\in I}U_i:(\forall i\in I)\,U_i\in\mathcal{T}_i,\ \abs{\{i\in I:U_i\neq X_i\}}<\infty\}$. The \hlt{product topology} on $\prod\limits_{i\in I}X_i$ is generated by $\mathcal{B}_{\Pi}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:product-collections-are-bases} \hlt{Product Collections Are Bases}\\
Let $(X,\mathcal{T}_X),(Y,\mathcal{T}_Y)$ and $(X_i,\mathcal{T}_i)$, $i\in I$, be topological spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{B}_{X\times Y}$ is a basis on $X\times Y$.
\item $\mathcal{B}_{\Pi}$ is a basis on $\prod\limits_{i\in I}X_i$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The rectangle $X\times Y$ belongs to $\mathcal{B}_{X\times Y}$. For $(x,y)\in(U_1\times V_1)\cap(U_2\times V_2)$, set $W\defeq(U_1\cap U_2)\times(V_1\cap V_2)$.\\
Then $W\in\mathcal{B}_{X\times Y}$ by Definitions \ref{def:topological-space}(iii) and \ref{def:product-topology}(i).\\
Refinement relation $(x,y)\in W\subseteq(U_1\times V_1)\cap(U_2\times V_2)$ holds by definition of $W$. Hence $\mathcal{B}_{X\times Y}$ is a basis by Definition \ref{def:basis-for-topology}(ii).
\item The full product $\prod\limits_{i\in I}X_i$ belongs to $\mathcal{B}_{\Pi}$. For boxes $U=\prod\limits_{i\in I}U_i$ and $V=\prod\limits_{i\in I}V_i$ in $\mathcal{B}_{\Pi}$ and $x\in U\cap V$, set $W_i\defeq U_i\cap V_i$.\\
Then $(\forall i\in I)\,W_i\in\mathcal{T}_i$ by Definition \ref{def:topological-space}(iii). The index sets satisfy $\{i\in I:W_i\neq X_i\}\subseteq\{i\in I:U_i\neq X_i\}\cup\{i\in I:V_i\neq X_i\}$.\\
Right-hand side finite by $U,V\in\mathcal{B}_{\Pi}$; $\prod\limits_{i\in I}W_i\in\mathcal{B}_{\Pi}$ by Definition \ref{def:product-topology}(ii),  $x\in\prod\limits_{i\in I}W_i\subseteq U\cap V$; $\mathcal{B}_{\Pi}$ is basis by Definition \ref{def:basis-for-topology}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Box intersections preserve finite support.\\
\hlt{Key Insight}: Intersecting product boxes introduces only finitely many restrictions.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:product-topology-coordinate-continuity} \hlt{Product Topology and Coordinate Maps}\\
Let $(Z,\mathcal{T}_Z),(X,\mathcal{T}_X),(Y,\mathcal{T}_Y)$ be topological spaces, and equip $X\times Y$ with product topology $\mathcal{T}_{X\times Y}$.\\
Define $\pi_X:X\times Y\to X$ by $\pi_X(x,y)\defeq x$. Define $\pi_Y:X\times Y\to Y$ by $\pi_Y(x,y)\defeq y$.\\
For $f:Z\to X\times Y$, $f$ is continuous iff $\pi_X\circ f$ and $\pi_Y\circ f$ are continuous.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] For $U\in\mathcal{T}_X$, $U\times Y\in\mathcal{T}_{X\times Y}$ by Definition \ref{def:product-topology}(i). Thus $(\pi_X\circ f)^{-1}(U)=f^{-1}(U\times Y)\in\mathcal{T}_Z$ by Proposition \ref{prop:topological-continuity-open-preimages}(i) for $f$.\\
For $V\in\mathcal{T}_Y$, $X\times V\in\mathcal{T}_{X\times Y}$ by Definition \ref{def:product-topology}(i). Thus $(\pi_Y\circ f)^{-1}(V)=f^{-1}(X\times V)\in\mathcal{T}_Z$ by Proposition \ref{prop:topological-continuity-open-preimages}(i) for $f$.\\
Hence both coordinate maps are continuous by Proposition \ref{prop:topological-continuity-open-preimages}(i).
\item[$(\Leftarrow)$] Let $U\times V\in\mathcal{B}_{X\times Y}$. Both coordinate preimages $(\pi_X\circ f)^{-1}(U)$ and $(\pi_Y\circ f)^{-1}(V)$ are open by Proposition \ref{prop:topological-continuity-open-preimages}(i).\\
Intersection $= f^{-1}(U\times V)$, lies in $\mathcal{T}_Z$ by Definition \ref{def:topological-space}(iii). For $O\in\mathcal{T}_{X\times Y}$, $\mathcal{U}\subseteq\mathcal{B}_{X\times Y}$ with $O=\bigcup\mathcal{U}$ by Definition \ref{def:product-topology}(i).\\
Then $f^{-1}(O)=\bigcup\limits_{B\in\mathcal{U}}f^{-1}(B)\in\mathcal{T}_Z$ by Definition \ref{def:topological-space}(ii). Hence $f$ is continuous by Proposition \ref{prop:topological-continuity-open-preimages}(i).
\end{enumerate}
\hlt{Structural Commentary}: Coordinate preimages handle rectangles; unions handle all product opens.\\
\hlt{Key Insight}: A product map is continuous exactly coordinatewise.
\end{proof}
\end{proposition}

\begin{definition} \label{def:quotient-topology} \hlt{Quotient Topology}\\
Let $q:(X,\tau_X)\twoheadrightarrow Y$. The \hlt{quotient topology} on $Y$ induced by $q$ is $\tau_q\defeq\{V\subseteq Y:q^{-1}(V)\in\tau_X\}$.
\end{definition}

\begin{proposition} \label{prop:quotient-collection-is-topology} \hlt{The Quotient Collection Is a Topology}\\
Let $q:(X,\tau_X)\twoheadrightarrow Y$, and let $\tau_q$ be the quotient topology induced by $q$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\tau_q$ is a topology on $Y$.
\item $q:(X,\tau_X)\to(Y,\tau_q)$ is continuous.
\item If $\sigma$ is a topology on $Y$ and $q:(X,\tau_X)\to(Y,\sigma)$ is continuous, then $\sigma\subseteq\tau_q$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The identities $q^{-1}(\emptyset)=\emptyset$ and $q^{-1}(Y)=X$ imply $\emptyset,Y\in\tau_q$ by Definitions \ref{def:topological-space}(i) and \ref{def:quotient-topology}.\\
For $(V_\alpha)_{\alpha\in A}\subseteq\tau_q$, $q^{-1}(\bigcup\limits_{\alpha\in A}V_\alpha)=\bigcup\limits_{\alpha\in A}q^{-1}(V_\alpha)\in\tau_X$ by Definition \ref{def:topological-space}(ii). Hence $\bigcup\limits_{\alpha\in A}V_\alpha\in\tau_q$ by Definition \ref{def:quotient-topology}.\\
If $n\geq1$ and $(V_j)_{j=1}^n\subseteq\tau_q$, then $q^{-1}(\bigcap\limits_{j=1}^nV_j)=\bigcap\limits_{j=1}^nq^{-1}(V_j)\in\tau_X$ by Definition \ref{def:topological-space}(iii).\\
Hence $\bigcap\limits_{j=1}^nV_j\in\tau_q$ by Definition \ref{def:quotient-topology}. Hence $\tau_q$ is a topology by Definition \ref{def:topological-space}.
\item For $V\in\tau_q$, $q^{-1}(V)\in\tau_X$ by Definition \ref{def:quotient-topology}. Hence $q$ is continuous by Proposition \ref{prop:topological-continuity-open-preimages}(i).
\item For $V\in\sigma$, $q^{-1}(V)\in\tau_X$ by Proposition \ref{prop:topological-continuity-open-preimages}(i), so $V\in\tau_q$ by Definition \ref{def:quotient-topology}. Thus $\sigma\subseteq\tau_q$.
\end{enumerate}
\hlt{Structural Commentary}: Inverse-image identities prove the axioms and compare admissible topologies.\\
\hlt{Key Insight}: Quotient-open sets are exactly those with open pullbacks.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:quotient-topology-universal-property} \hlt{Quotient Universal Property}\\
Let $q:(X,\tau_X)\twoheadrightarrow Y$, and equip $Y$ with the quotient topology $\tau_Y$ induced by $q$.\\
For $g:Y\to(Z,\tau_Z)$, $g:(Y,\tau_Y)\to(Z,\tau_Z)$ is continuous iff $g\circ q:(X,\tau_X)\to(Z,\tau_Z)$ is continuous.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $g$ is continuous, then $q$ is continuous by Proposition \ref{prop:quotient-collection-is-topology}(ii). Hence $g\circ q$ is continuous by Proposition \ref{prop:topological-continuity-open-preimages}(iii).
\item[$(\Leftarrow)$] Assume $g\circ q$ is continuous. For $W\in\tau_Z$, $q^{-1}(g^{-1}(W))=(g\circ q)^{-1}(W)\in\tau_X$ by Proposition \ref{prop:topological-continuity-open-preimages}(i).\\
Thus $g^{-1}(W)\in\tau_Y$ by Definition \ref{def:quotient-topology}, so $g$ is continuous by Proposition \ref{prop:topological-continuity-open-preimages}(i).
\end{enumerate}
\hlt{Structural Commentary}: Precomposition with $q$ converts continuity into the quotient test.\\
\hlt{Key Insight}: Continuity descends precisely through quotient-open pullbacks.
\end{proof}
\end{proposition}

\begin{definition} \label{def:weak-topology-induced-by-maps} \hlt{Weak Topology Induced by Maps}\\
Let $X$ be a set.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For topologies $\mathcal{T}_1,\mathcal{T}_2$ on $X$, $\mathcal{T}_2$ is \hlt{weaker} than $\mathcal{T}_1$ iff $\mathcal{T}_2\subseteq\mathcal{T}_1$. Equivalently, $\mathcal{T}_1$ is \hlt{stronger} than $\mathcal{T}_2$.
\item Let $\{(X_\alpha,\mathcal{T}_\alpha)\}_{\alpha\in A}$ be topological spaces, and let $(f_\alpha:X\to X_\alpha)_{\alpha\in A}$ be maps.\\
Set $\mathcal{S}\defeq\{X\}\cup\{f_\alpha^{-1}(O):\alpha\in A,\ O\in\mathcal{T}_\alpha\}$. The \hlt{weak topology induced by $\{f_\alpha\}_{\alpha\in A}$} is the topology generated by $\mathcal{S}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:weak-topology-universal-property} \hlt{Weak Topology Universal Property}\\
Let $X$ be a set, let $\{(X_\alpha,\mathcal{T}_\alpha)\}_{\alpha\in A}$ be topological spaces, and let $(f_\alpha:X\to X_\alpha)_{\alpha\in A}$ be maps.\\
Let $\mathcal{T}_w$ be generated by $\mathcal{S}\defeq\{X\}\cup\{f_\alpha^{-1}(O):\alpha\in A,\ O\in\mathcal{T}_\alpha\}$.\\
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{T}_w$ is a topology on $X$.
\item $(\forall\alpha\in A)(\forall O\in\mathcal{T}_\alpha)\,f_\alpha^{-1}(O)\in\mathcal{T}_w$.
\item Let $\mathcal{T}$ be a topology on $X$ satisfying $(\forall\alpha\in A)(\forall O\in\mathcal{T}_\alpha)\,f_\alpha^{-1}(O)\in\mathcal{T}$. Then the weak topology satisfies $\mathcal{T}_w\subseteq\mathcal{T}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\mathcal{B}_w$ be finite intersections of members of $\mathcal{S}$. Then $X\in\mathcal{B}_w$ and $(\forall B_1,B_2\in\mathcal{B}_w)\,B_1\cap B_2\in\mathcal{B}_w$ by definition of $\mathcal{B}_w$.\\
$\mathcal{B}_w$ is basis by Definition \ref{def:basis-for-topology}(ii); $\mathcal{T}_w=\{\bigcup\mathcal{U}:\mathcal{U}\subseteq\mathcal{B}_w\}$ by Definition \ref{def:basis-for-topology}(iii),(iv); $\mathcal{T}_w$ is topology by Proposition \ref{prop:basis-generated-topology-characterization}(i).
\item For $\alpha\in A$ and $O\in\mathcal{T}_\alpha$, $f_\alpha^{-1}(O)\in\mathcal{S}\subseteq\mathcal{T}_w$ by the definitions of $\mathcal{S}$ and $\mathcal{T}_w$.
\item If $\mathcal{T}$ satisfies the stated predicate, then $\mathcal{S}\subseteq\mathcal{T}$.\\
Every member of $\mathcal{B}_w$, every union of members lies in $\mathcal{T}$ by Definition \ref{def:topological-space}(ii),(iii). Hence $\mathcal{T}_w\subseteq\mathcal{T}$ by union representation in (i).
\end{enumerate}
\hlt{Structural Commentary}: Finite intersections and unions generate the least admissible topology.\\
\hlt{Key Insight}: Only subbasic preimages are forced to be open.
\end{proof}
\end{proposition}
