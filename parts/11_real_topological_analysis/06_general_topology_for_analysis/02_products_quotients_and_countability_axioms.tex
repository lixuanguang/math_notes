\subsection{Continuity and Topological Constructions}

General topological constructions are organized by inverse images: homeomorphisms preserve open-set structure, while products, quotients, and weak topologies are characterized by which maps they make continuous.

\begin{definition} \label{def:topological-continuity} \hlt{Continuity Between Topological Spaces}\\
Let $(X,\mathcal{T}_X)$ and $(Y,\mathcal{T}_Y)$ be topological spaces, let $f:X\to Y$.\\
For $x\in X$, $y\in Y$, set $\mathcal{N}_X(x)\defeq\{U\in\mathcal{T}_X:x\in U\}$, $\mathcal{N}_Y(y)\defeq\{O\in\mathcal{T}_Y:y\in O\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $x_0\in X$, $f$ is \hlt{continuous at $x_0$} iff $\forall O\in\mathcal{N}_Y(f(x_0))\ \exists U\in\mathcal{N}_X(x_0)$ with $f(U)\subseteq O$.
\item $f$ is \hlt{continuous} iff $\forall x\in X$, $f$ is continuous at $x$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:topological-continuity-open-preimages} \hlt{Continuity by Open Preimages}\\
Let $(X,\mathcal{T}_X)$ and $(Y,\mathcal{T}_Y)$ be topological spaces, and let $f:X\to Y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f$ is continuous iff $\forall O\in\mathcal{T}_Y$, $f^{-1}(O)\in\mathcal{T}_X$.
\item If $A\subseteq X$, $\mathcal{T}_A=\{A\cap V:V\in\mathcal{T}_X\}$, and $f$ is continuous, then $f|_A:(A,\mathcal{T}_A)\to(Y,\mathcal{T}_Y)$ is continuous.
\item If $(Z,\mathcal{T}_Z)$ is a topological space, $g:(Y,\mathcal{T}_Y)\to(Z,\mathcal{T}_Z)$ is continuous, and $f$ is continuous,\\
then $g\circ f:(X,\mathcal{T}_X)\to(Z,\mathcal{T}_Z)$ is continuous.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Let $O\in\mathcal{T}_Y$ and $x\in f^{-1}(O)$. Since $O\in\mathcal{N}_Y(f(x))$ and $f$ is continuous at $x$, choose $U_x\in\mathcal{N}_X(x)$ with $f(U_x)\subseteq O$. Then $x\in U_x\subseteq f^{-1}(O)$, so Proposition \ref{prop:open-neighborhood-test-topological} gives $f^{-1}(O)\in\mathcal{T}_X$.
\item[$(i)$] ($\Leftarrow$) Assume $\forall O\in\mathcal{T}_Y$, $f^{-1}(O)\in\mathcal{T}_X$. Let $x_0\in X$ and $O\in\mathcal{N}_Y(f(x_0))$. Then $U\defeq f^{-1}(O)\in\mathcal{N}_X(x_0)$ and $f(U)\subseteq O$, so $f$ is continuous at $x_0$. Since $x_0$ was arbitrary, $f$ is continuous.
\item[$(ii)$] Let $O\in\mathcal{T}_Y$. Then $(f|_A)^{-1}(O)=A\cap f^{-1}(O)\in\mathcal{T}_A$, so $f|_A$ is continuous by $(i)$.
\item[$(iii)$] Let $W\in\mathcal{T}_Z$. Since $g$ is continuous, $g^{-1}(W)\in\mathcal{T}_Y$.\\
Since $f$ is continuous, $(g\circ f)^{-1}(W)=f^{-1}(g^{-1}(W))\in\mathcal{T}_X$. Hence $g\circ f$ is continuous by $(i)$.
\end{enumerate}
\hlt{Structural Commentary}: Subspaces and composites preserve open preimages by intersection and nesting.\\
\hlt{Key Insight}: Continuity is an inverse-image property.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:continuity-tested-on-subbasis} \hlt{Continuity Tested on a Subbasis}\\
Let $(X,\mathcal{T}_X)$ be a topological space, let $Y$ carry the topology $\mathcal{T}_Y$ generated by a subbasis $\mathcal{S}$, and let $f:X\to Y$. Then $f$ is continuous iff $\forall S\in\mathcal{S}$, $f^{-1}(S)\in\mathcal{T}_X$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $f$ is continuous and $S\in\mathcal{S}$, then $S\in\mathcal{T}_Y$, so Proposition \ref{prop:topological-continuity-open-preimages} gives $f^{-1}(S)\in\mathcal{T}_X$.
\item[$(\Leftarrow)$] Assume $\forall S\in\mathcal{S}$, $f^{-1}(S)\in\mathcal{T}_X$. Let $\mathcal{B}$ be the basis of finite intersections from $\mathcal{S}$, including $Y$ as the empty intersection. Then every $B\in\mathcal{B}$ has $f^{-1}(B)\in\mathcal{T}_X$, since $f^{-1}(\bigcap\limits_{j=1}^n S_j)=\bigcap\limits_{j=1}^n f^{-1}(S_j)$.\\
If $O=\bigcup\limits_{\alpha\in A}B_\alpha\in\mathcal{T}_Y$, $B_\alpha\in\mathcal{B}$, then $f^{-1}(O)=\bigcup\limits_{\alpha\in A}f^{-1}(B_\alpha)\in\mathcal{T}_X$. Proposition \ref{prop:topological-continuity-open-preimages} gives continuity of $f$.
\end{enumerate}
\hlt{Structural Commentary}: Subbasis tests extend through finite intersections and unions.\\
\hlt{Key Insight}: Inverse images preserve the operations generating the topology.
\end{proof}
\end{proposition}

\begin{definition} \label{def:topological-homeomorphism} \hlt{Homeomorphism}\\
Let $(X,\mathcal{T}_X)$ and $(Y,\mathcal{T}_Y)$ be topological spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{homeomorphism} is a bijection $f:X\to Y$ such that\\
both $f:(X,\mathcal{T}_X)\to(Y,\mathcal{T}_Y)$ and $f^{-1}:(Y,\mathcal{T}_Y)\to(X,\mathcal{T}_X)$ are continuous.
\item The spaces $X$ and $Y$ are \hlt{homeomorphic}, written $X\cong_{\mathrm{top}}Y$, iff $\exists$ a homeomorphism $f:X\to Y$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:homeomorphism-preserves-open-and-closed-sets} \hlt{Homeomorphic Spaces Have the Same Open-Set Structure}\\
Let $(X,\mathcal{T}_X)$ and $(Y,\mathcal{T}_Y)$ be topological spaces, let $f:X\to Y$ be a homeomorphism, and let $E\subseteq X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E\in\mathcal{T}_X$ iff $f(E)\in\mathcal{T}_Y$.
\item $E$ is closed in $X$ iff $f(E)$ is closed in $Y$.
\end{enumerate}
\begin{proof}
Since $f$ is a homeomorphism, $f$ is bijective and both $f$ and $f^{-1}$ are continuous.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) If $E\in\mathcal{T}_X$, then $f(E)=(f^{-1})^{-1}(E)\in\mathcal{T}_Y$.
\item[$(i)$] ($\Leftarrow$) If $f(E)\in\mathcal{T}_Y$, then $E=f^{-1}(f(E))\in\mathcal{T}_X$.
\item[$(ii)$] ($\Rightarrow$) If $E$ is closed in $X$, then $X\setminus E\in\mathcal{T}_X$. By $(i)$, $f(X\setminus E)\in\mathcal{T}_Y$.\\
Since $f$ is bijective, $f(X\setminus E)=Y\setminus f(E)$, so $f(E)$ is closed in $Y$.
\item[$(ii)$] ($\Leftarrow$) If $f(E)$ is closed in $Y$, then $Y\setminus f(E)\in\mathcal{T}_Y$. Since $f$ is bijective, $Y\setminus f(E)=f(X\setminus E)$.\\
By $(i)$, $X\setminus E\in\mathcal{T}_X$, so $E$ is closed in $X$.
\end{enumerate}
\hlt{Structural Commentary}: Openness transfers through inverse images of $f$ and $f^{-1}$.\\
\hlt{Key Insight}: Homeomorphisms preserve and reflect open sets.
\end{proof}
\end{proposition}

\begin{definition} \label{def:product-topology} \hlt{Product Topology}\\
Let $(X,\mathcal{T}_X)$ and $(Y,\mathcal{T}_Y)$ be topological spaces, and let $\{(X_i,\mathcal{T}_i)\}_{i\in I}$ be a family of topological spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{product topology} on $X\times Y$ is generated by $\{U\times V:U\in\mathcal{T}_X,\ V\in\mathcal{T}_Y\}$.
\item The \hlt{product topology} on $\prod_{i\in I}X_i$ is generated by $\{\prod_{i\in I}U_i:U_i\in\mathcal{T}_i\ \forall i\in I,\ \abs{\{i\in I:U_i\neq X_i\}}<\infty\}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:product-topology-coordinate-continuity} \hlt{Product Topology and Coordinate Maps}\\
Let $(Z,\mathcal{T}_Z)$, $(X,\mathcal{T}_X)$, $(Y,\mathcal{T}_Y)$ be topological spaces. Let $\mathcal{T}_{X\times Y}$ be generated by $\mathcal{B}=\{U\times V:U\in\mathcal{T}_X,\ V\in\mathcal{T}_Y\}$.\\
For $f:Z\to X\times Y$, $f$ is continuous iff $\pi_X\circ f$ and $\pi_Y\circ f$ are continuous.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $f$ is continuous and $U\in\mathcal{T}_X$, then $U\times Y\in\mathcal{B}$ and $(\pi_X\circ f)^{-1}(U)=f^{-1}(U\times Y)\in\mathcal{T}_Z$.\\
Similarly, $(\pi_Y\circ f)^{-1}(V)=f^{-1}(X\times V)\in\mathcal{T}_Z$ for every $V\in\mathcal{T}_Y$.
\item[$(\Leftarrow)$] Suppose $\pi_X\circ f$ and $\pi_Y\circ f$ are continuous. For $U\times V\in\mathcal{B}$, $f^{-1}(U\times V)=(\pi_X\circ f)^{-1}(U)\cap(\pi_Y\circ f)^{-1}(V)\in\mathcal{T}_Z$.\\
If $O\in\mathcal{T}_{X\times Y}$, write $O=\bigcup\limits_{\alpha\in A}B_\alpha$ with $B_\alpha\in\mathcal{B}$; then $f^{-1}(O)=\bigcup\limits_{\alpha\in A}f^{-1}(B_\alpha)\in\mathcal{T}_Z$. Hence $f$ is continuous.
\end{enumerate}
\hlt{Structural Commentary}: Basis rectangles pull back to intersections of coordinate preimages.\\
\hlt{Key Insight}: Product continuity is coordinatewise continuity.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:product-topology-pointwise-convergence} \hlt{Product Convergence Is Pointwise}\\
Let $X^\Lambda=\prod_{\lambda\in\Lambda}X_\lambda$ carry the product topology from $\{(X_\lambda,\mathcal{T}_\lambda)\}_{\lambda\in\Lambda}$.\\
For $(x_n)\subseteq X^\Lambda$ and $x\in X^\Lambda$, $x_n\to x$ iff $x_n(\lambda)\to x(\lambda)$ in $X_\lambda$ for every $\lambda\in\Lambda$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $x_n\to x$, then each projection $\pi_\lambda:X^\Lambda\to X_\lambda$ is continuous, so $x_n(\lambda)=\pi_\lambda(x_n)\to\pi_\lambda(x)=x(\lambda)$ $\forall \lambda\in\Lambda$.
\item[$(\Leftarrow)$] Suppose $x_n(\lambda)\to x(\lambda)$ $\forall \lambda\in\Lambda$. Let $W=\prod_{\lambda\in\Lambda}U_\lambda$ be a basic neighborhood of $x$, with $x(\lambda)\in U_\lambda\in\mathcal{T}_\lambda$ and finite $F=\{\lambda\in\Lambda:U_\lambda\neq X_\lambda\}$. Coordinatewise convergence on the finite set $F$ gives $n_0$ such that $x_n(\lambda)\in U_\lambda$ for every $\lambda\in F$ and $n\ge n_0$. Hence $x_n\in W$ eventually, so $x_n\to x$.
\end{enumerate}
\hlt{Structural Commentary}: Product neighborhoods impose finitely many coordinate constraints.\\
\hlt{Key Insight}: Coordinate convergence satisfies every finite constraint.
\end{proof}
\end{proposition}

\begin{definition} \label{def:quotient-topology} \hlt{Quotient Topology}\\
Let $q:X\to Y$ be a surjection from a topological space $X$ onto a set $Y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{quotient topology} on $Y$ is $\{V\subseteq Y:q^{-1}(V)\text{ is open in }X\}$.
\item Equivalently, it is the strongest topology on $Y$ for which $q$ is continuous.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:quotient-topology-universal-property} \hlt{Quotient Universal Property}\\
Let $q:(X,\tau_X)\twoheadrightarrow Y$ and give $Y$ the quotient topology $\tau_Y=\{V\subseteq Y:q^{-1}(V)\in\tau_X\}$.\\
For any map $g:Y\to (Z,\tau_Z)$, $g:(Y,\tau_Y)\to (Z,\tau_Z)$ is continuous $\Longleftrightarrow$ $g\circ q:(X,\tau_X)\to (Z,\tau_Z)$ is continuous.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $g$ is continuous, then $g\circ q$ is continuous by composition.
\item[$(\Leftarrow)$] Suppose $g\circ q$ is continuous. For $W\in\tau_Z$, $q^{-1}(g^{-1}(W))=(g\circ q)^{-1}(W)\in\tau_X$.\\
Thus $g^{-1}(W)\in\tau_Y$ by the quotient topology, so $g$ is continuous.
\end{enumerate}
\hlt{Structural Commentary}: Continuity from a quotient is tested by precomposition with the quotient map.\\
\hlt{Key Insight}: Pull back through the quotient map and use the defining open-set test.
\end{proof}
\end{proposition}

\begin{definition} \label{def:weak-topology-induced-by-maps} \hlt{Weak Topology Induced by Maps}\\
Let $X$ be a set.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For topologies $\mathcal{T}_1,\mathcal{T}_2$ on $X$, $\mathcal{T}_2$ is \hlt{weaker} than $\mathcal{T}_1$ and $\mathcal{T}_1$ is \hlt{stronger} than $\mathcal{T}_2$ iff $\mathcal{T}_2\subseteq\mathcal{T}_1$.
\item For topological spaces $\{(X_\alpha,\mathcal{T}_\alpha)\}_{\alpha\in A}$ and maps $f_\alpha:X\to X_\alpha$, the \hlt{weak topology induced by $\{f_\alpha\}_{\alpha\in A}$} is generated by $\mathcal{S}\defeq\{X\}\cup\{f_\alpha^{-1}(O):\alpha\in A,\ O\in\mathcal{T}_\alpha\}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:weak-topology-universal-property} \hlt{Weak Topology Universal Property}\\
Let $f_\alpha:X\to(X_\alpha,\mathcal{T}_\alpha)$ for $\alpha\in A$, and let $\mathcal{T}_w$ be generated by $\mathcal{S}\defeq\{X\}\cup\{f_\alpha^{-1}(O):\alpha\in A,\ O\in\mathcal{T}_\alpha\}$.\\
Then $\mathcal{T}_w$ is the weakest topology on $X$ making every $f_\alpha$ continuous.
\begin{proof}
Since $f_\alpha^{-1}(O)\in\mathcal{T}_w$ for all $\alpha\in A$ and $O\in\mathcal{T}_\alpha$, every $f_\alpha:(X,\mathcal{T}_w)\to(X_\alpha,\mathcal{T}_\alpha)$ is continuous.\\
If a topology $\mathcal{T}'$ on $X$ makes every $f_\alpha$ continuous, then $\mathcal{S}\subseteq\mathcal{T}'$; hence $\mathcal{T}_w\subseteq\mathcal{T}'$.\\
\hlt{Structural Commentary}: Subbasic preimages are the forced open sets.\\
\hlt{Key Insight}: Weakest means contained in every topology passing the tests.
\end{proof}
\end{proposition}
