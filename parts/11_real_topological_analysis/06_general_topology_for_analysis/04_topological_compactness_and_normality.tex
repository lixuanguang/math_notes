\subsection{Topological Compactness and Normality Tools}

Compactness in general topology keeps the open-cover principle from metric spaces and pairs especially well with Hausdorff and normal separation. These tools also give useful continuous functions on normal spaces.

\begin{definition} \label{def:topological-compactness} \hlt{Compactness in Topological Spaces}\\
Let $X$ be a topological space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A family $\{O_\alpha\}_{\alpha\in A}$ is an \hlt{open cover} of $E\subseteq X$ if every $O_\alpha$ is open and $E\subseteq\bigcup_{\alpha\in A}O_\alpha$.
\item The space $X$ is \hlt{compact} if every open cover of $X$ has a finite subcover.
\item A subset $K\subseteq X$ is \hlt{compact} if it is compact with the subspace topology.\\
Equivalently, every cover of $K$ by open subsets of $X$ has a finite subcover.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:topological-compactness-fip} \hlt{Finite-Intersection Compactness Test}\\
Let $(X,\mathcal{T})$ be a topological space. Then $X$ is compact if and only if every closed family $\{F_\alpha\}_{\alpha\in A}$ with the finite intersection property satisfies $\bigcap\limits_{\alpha\in A}F_\alpha\neq\emptyset$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $X$ is compact and closed $\{F_\alpha\}_{\alpha\in A}$ has empty intersection, then $\{X\setminus F_\alpha\}_{\alpha\in A}$ covers $X$.\\
A finite subcover gives finite $J\subseteq A$ with $\bigcap\limits_{\alpha\in J}F_\alpha=\emptyset$, so the family is not FIP.
\item[$(\Leftarrow)$] Assume every closed FIP family intersects. If an open cover $\{O_\alpha\}_{\alpha\in A}$ has no finite subcover, then $\{X\setminus O_\alpha\}_{\alpha\in A}$ is closed and FIP, so some $x$ lies in all $X\setminus O_\alpha$, contradicting that $\{O_\alpha\}_{\alpha\in A}$ covers $X$.
\end{enumerate}
\hlt{Structural Commentary}: Complements exchange covers and intersections.\\
\hlt{Key Insight}: Compactness forbids closed FIP families from missing every point.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:closed-subsets-compact-topological} \hlt{Closed Subsets of Compact Spaces Are Compact}\\
Let $(X,\mathcal{T})$ be compact and let $K\subseteq X$ be closed. Then $K$ is compact in the subspace topology.
\begin{proof}
Let $\{O_\alpha\}_{\alpha\in A}$ cover $K$, with $O_\alpha\in\mathcal{T}$. Since $X\setminus K\in\mathcal{T}$, the family $\{O_\alpha\}_{\alpha\in A}\cup\{X\setminus K\}$ covers $X$. Compactness gives a finite subcover; discarding $X\setminus K$ if present leaves a finite subcover of $K$.\\
\hlt{Structural Commentary}: Add the open complement to cover $X$.\\
\hlt{Key Insight}: Compactness passes to closed subspaces.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:compact-subsets-hausdorff-closed} \hlt{Compact Subsets of Hausdorff Spaces Are Closed}\\
Let $(X,\mathcal{T})$ be Hausdorff and let $K\subseteq X$ be compact in the subspace topology. Then $K$ is closed in $X$.
\begin{proof} For $y\in X\setminus K$, choose disjoint open $O_x\ni x$ and $U_x\ni y$ for each $x\in K$.\\
Compactness gives $x_1,\ldots,x_n\in K$ with $K\subseteq\bigcup\limits_{j=1}^n O_{x_j}$, so $U\defeq\bigcap\limits_{j=1}^n U_{x_j}$ is an open neighborhood of $y$ disjoint from $K$. Thus $X\setminus K$ is open and $K$ is closed.\\
\hlt{Structural Commentary}: Compactness turns pointwise Hausdorff separations into a finite one.\\
\hlt{Key Insight}: Every exterior point has an open neighborhood avoiding $K$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:compact-hausdorff-normal} \hlt{Compact Hausdorff Spaces Are Normal}\\
Let $(X,\mathcal{T})$ be compact and Hausdorff. Then $X$ is normal.
\begin{proof}
Hausdorff implies $T_1$. Let $A,B\subseteq X$ be disjoint closed sets; then $A,B$ are compact. For each $b\in B$, compact-set separation gives disjoint open $U_b\ni b$ and $V_b\supseteq A$. Choose $b_1,\ldots,b_n$ with $B\subseteq\bigcup\limits_{j=1}^n U_{b_j}$.\\
Then $\bigcup\limits_{j=1}^n U_{b_j}$ and $\bigcap\limits_{j=1}^n V_{b_j}$ are disjoint open neighborhoods of $B$ and $A$, so $X$ is normal.\\
\hlt{Structural Commentary}: Compactness makes finitely many Hausdorff separations suffice.\\
\hlt{Key Insight}: Disjoint closed compact sets have disjoint neighborhoods.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:continuous-image-compact-topological} \hlt{Continuous Images of Compact Spaces Are Compact}\\
Let $f:(X,\mathcal{T}_X)\to(Y,\mathcal{T}_Y)$ be continuous. If $X$ is compact, then $f(X)$ is compact in the subspace topology.
\begin{proof}
Let $\{V_\alpha\}_{\alpha\in A}$ be an open cover of $f(X)$ by subsets open in $Y$. Then $\{f^{-1}(V_\alpha)\}_{\alpha\in A}$ covers $X$ and each $f^{-1}(V_\alpha)\in\mathcal{T}_X$. Compactness gives finite $J\subseteq A$ with $X=\bigcup\limits_{\alpha\in J}f^{-1}(V_\alpha)$, hence $f(X)\subseteq\bigcup\limits_{\alpha\in J}V_\alpha$.\\
\hlt{Structural Commentary}: Pull the cover back to the compact domain.\\
\hlt{Key Insight}: Finite subcovers descend through the image.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:extreme-value-compact-topological} \hlt{Extreme Values on Compact Spaces}\\
If $X$ is compact and $f:X\to\R$ is continuous, then $f$ attains its minimum and maximum.
\begin{proof}
By Theorem \ref{thm:continuous-image-compact-topological}, $f(X)$ is compact in $\R$. By Heine--Borel, Theorem \ref{thm:heine-borel-rk}, $f(X)$ is closed and bounded, so $\inf f(X),\sup f(X)\in f(X)$. Choose preimages $x_{\min},x_{\max}\in X$.\\
\hlt{Structural Commentary}: Compact images in $\R$ are closed and bounded.\\
\hlt{Key Insight}: Closed bounded images contain their endpoints.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:compact-hausdorff-bijection-homeomorphism} \hlt{Compact-to-Hausdorff Bijection Criterion}\\
Let $f:(X,\mathcal{T}_X)\to(Y,\mathcal{T}_Y)$ be continuous bijection. If $X$ is compact, $Y$ is Hausdorff, then $f$ is a homeomorphism.
\begin{proof} It remains to prove continuity of $f^{-1}:Y\to X$. If $F\subseteq X$ is closed, then $F$ is compact by Proposition \ref{prop:closed-subsets-compact-topological}, so $f(F)$ is compact by Theorem \ref{thm:continuous-image-compact-topological}. Since $Y$ is Hausdorff, $f(F)$ is closed by Proposition \ref{prop:compact-subsets-hausdorff-closed}.\\
Thus $(f^{-1})^{-1}(F)=f(F)$ is closed in $Y$, so $f^{-1}$ is continuous.\\
\hlt{Structural Commentary}: Closed sets map compact, then closed.\\
\hlt{Key Insight}: Compact-to-Hausdorff bijections have continuous inverses.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:compact-hausdorff-topology-rigidity} \hlt{Compact Hausdorff Topologies Are Rigid}\\
Let $X$ be a set and let $\mathcal{T},\mathcal{T}_1,\mathcal{T}_2$ be topologies on $X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $(X,\mathcal{T})$ is compact and $\mathcal{T}_1\subseteq\mathcal{T}$, then $(X,\mathcal{T}_1)$ is compact.
\item If $(X,\mathcal{T})$ is Hausdorff and $\mathcal{T}\subseteq\mathcal{T}_2$, then $(X,\mathcal{T}_2)$ is Hausdorff.
\item If $(X,\mathcal{T})$ is compact Hausdorff, then no strictly weaker topology is Hausdorff and no strictly stronger topology is compact.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] Every $\mathcal{T}_1$-open cover is a $\mathcal{T}$-open cover, so compactness of $(X,\mathcal{T})$ gives a finite subcover.
\item[$(ii)$] If $x\neq y$, choose disjoint $\mathcal{T}$-open neighborhoods of $x$ and $y$; they are also $\mathcal{T}_2$-open.
\item[$(iii)$] If $\mathcal{S}\subseteq\mathcal{T}$ is Hausdorff, Theorem \ref{thm:compact-hausdorff-bijection-homeomorphism} applied to $\operatorname{id}:(X,\mathcal{T})\to(X,\mathcal{S})$ gives $\mathcal{T}\subseteq\mathcal{S}$, hence $\mathcal{S}=\mathcal{T}$.\\
If $\mathcal{T}\subseteq\mathcal{S}$ and $(X,\mathcal{S})$ is compact, apply it to $\operatorname{id}:(X,\mathcal{S})\to(X,\mathcal{T})$ to get $\mathcal{S}\subseteq\mathcal{T}$, hence $\mathcal{S}=\mathcal{T}$.
\end{enumerate}
\hlt{Structural Commentary}: Weakening preserves covers; strengthening preserves separations.\\
\hlt{Key Insight}: Compact Hausdorff topologies cannot be enlarged or shrunk compatibly.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:maximal-fip-extension} \hlt{Maximal Finite-Intersection Families}\\
Let $X$ be a set and let $\mathcal{A}\subseteq\mathcal{P}(X)$ have the finite intersection property. Then $\exists\mathcal{M}\subseteq\mathcal{P}(X)$ such that $\mathcal{A}\subseteq\mathcal{M}$, $\mathcal{M}$ has the finite intersection property, and no proper extension of $\mathcal{M}$ has it.
\begin{proof}
Let $\mathfrak{P}=\{\mathcal{B}\subseteq\mathcal{P}(X):\mathcal{A}\subseteq\mathcal{B}\text{ and }\mathcal{B}\text{ has FIP}\}$, ordered by inclusion. If $\mathfrak{C}\subseteq\mathfrak{P}$ is a chain, then $\bigcup\mathfrak{C}$ has FIP: each finite subfamily lies in one $\mathcal{B}\in\mathfrak{C}$. Thus Zorn's lemma, Principle \ref{princ:zorns-lemma}, gives a maximal $\mathcal{M}\in\mathfrak{P}$.\\
\hlt{Structural Commentary}: Chain unions preserve FIP because tests are finite.\\
\hlt{Key Insight}: Zorn extends any FIP family to a maximal one.
\end{proof}
\end{lemma}

\begin{lemma} \label{lem:maximal-fip-family-properties} \hlt{Closure Properties of Maximal FIP Families}\\
Let $X$ be a set and let $\mathcal{M}\subseteq\mathcal{P}(X)$ be maximal with respect to the finite intersection property.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $M_1,\ldots,M_n\in\mathcal{M}$, then $\bigcap\limits_{j=1}^n M_j\in\mathcal{M}$.
\item If $C\subseteq X$ satisfies $C\cap\bigcap\limits_{j=1}^n M_j\neq\emptyset$ for all $M_1,\ldots,M_n\in\mathcal{M}$, then $C\in\mathcal{M}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] For $N=\bigcap\limits_{j=1}^n M_j$, the family $\mathcal{M}\cup\{N\}$ still has the finite intersection property, so maximality gives $N\in\mathcal{M}$.
\item[$(ii)$] The hypothesis says $\mathcal{M}\cup\{C\}$ has the finite intersection property, so maximality gives $C\in\mathcal{M}$.
\end{enumerate}
\hlt{Structural Commentary}: Maximality absorbs every FIP-compatible set.\\
\hlt{Key Insight}: Finite intersections are already forced inside $\mathcal{M}$.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:tychonoff-product-theorem} \hlt{Tychonoff Product Theorem}\\
The product $X=\prod_{\lambda\in\Lambda}X_\lambda$ of compact topological spaces is compact in the product topology.
\begin{proof} Let $\mathcal{F}$ be a closed FIP family in $X$, and extend it to maximal FIP $\mathcal{M}$ by Lemma \ref{lem:maximal-fip-extension}.\\
For each $\lambda$, compactness of $X_\lambda$ gives $x_\lambda\in\bigcap\limits_{M\in\mathcal{M}}\overline{\pi_\lambda(M)}$; put $x=(x_\lambda)_{\lambda\in\Lambda}$.\\
If $N=\pi_\lambda^{-1}(O)$ is subbasic at $x$ and $P=\bigcap\limits_{j=1}^n M_j$ with $M_j\in\mathcal{M}$, then $P\in\mathcal{M}$ and $O\cap\pi_\lambda(P)\neq\emptyset$, so $N\cap P\neq\emptyset$.\\
Lemma \ref{lem:maximal-fip-family-properties} gives $N\in\mathcal{M}$, hence every basic neighborhood of $x$ lies in $\mathcal{M}$.\\
Thus every basic neighborhood of $x$ meets each $F\in\mathcal{F}$, so $x\in\overline{F}=F$.\\
Hence $\bigcap\limits_{F\in\mathcal{F}}F\neq\emptyset$, and Proposition \ref{prop:topological-compactness-fip} gives compactness.\\
\hlt{Structural Commentary}: Coordinate compactness selects a candidate point.\\
\hlt{Key Insight}: Maximal FIP forces all basic neighborhoods of that point to remain compatible.
\end{proof}
\end{theorem}

\begin{definition} \label{def:compact-open-topology} \hlt{Compact-Open Topology}\\
Let $X,Y$ be topological spaces and let $Y^X=\{f:X\to Y\}$. The \hlt{compact-open topology} on $Y^X$ is generated by the subbasis $\mathcal{U}_{K,O}\defeq\{f\in Y^X:f(K)\subseteq O\}$, where $K\subseteq X$ is compact and $O\subseteq Y$ is open.\\
\hlt{Structural Commentary}: Subbasic opens prescribe uniform containment on compact subsets.
\end{definition}

\begin{remark} \label{rem:compact-open-versus-product-topology} \hlt{Compact-Open Versus Product Topology}\\
On $Y^X=\prod_{x\in X}Y$, the product topology tests finitely many coordinates $f(x)$.\\
The compact-open topology tests compact sets at once through $\mathcal{U}_{K,O}=\{f\in Y^X:f(K)\subseteq O\}$\\
For metric $Y$, this encodes uniform control on compact subsets.
\end{remark}

\begin{definition} \label{def:normally-ascending-family} \hlt{Normally Ascending Families}\\
Let $X$ be a topological space and let $\Lambda\subseteq\R$.\\
A family of open sets $\{O_\lambda\}_{\lambda\in\Lambda}$ is \hlt{normally ascending} iff $\overline{O_{\lambda_1}}\subseteq O_{\lambda_2}$ for all $\lambda_1,\lambda_2\in\Lambda$ with $\lambda_1<\lambda_2$.\\
\hlt{Key Insight}: Later open sets contain the closures of earlier ones.
\end{definition}

\begin{lemma} \label{lem:normally-ascending-function-continuity} \hlt{Normally Ascending Families Define Continuous Functions}\\
Let $X$ be a topological space, let $\Lambda\subseteq(a,b)$ be dense, and let $\{O_\lambda\}_{\lambda\in\Lambda}$ be normally ascending. Define $g:X\to[a,b]$ by $g(x)=\inf\{\lambda\in\Lambda:x\in O_\lambda\}$, with $\inf\emptyset=b$. Then $g$ is continuous.
\begin{proof}
For $c\in(a,b)$, $\{g<c\}=\bigcup_{\lambda<c}O_\lambda$ and $\{g>c\}=\bigcup_{\lambda>c}(X\setminus\overline{O_\lambda})$. For the second identity, $x\notin\overline{O_\lambda}$ with $\lambda>c$ gives $g(x)\geq\lambda>c$; conversely, if $g(x)>c$, choose $\mu,\lambda\in\Lambda$ with $c<\mu<\lambda<g(x)$, so $x\notin O_\lambda\supseteq\overline{O_\mu}$. Both threshold sets are open, hence $g$ is continuous.\\
\hlt{Structural Commentary}: Threshold preimages reduce to unions of nested open or closed-complement sets.\\
\hlt{Key Insight}: Normal ascent makes upper thresholds open.
\end{proof}
\end{lemma}

\begin{lemma} \label{lem:normal-dyadic-neighborhood-ladder} \hlt{Dyadic Neighborhood Ladders in Normal Spaces}\\
Let $X$ be normal, let $F\subseteq X$ be closed, and let $U$ be an open neighborhood of $F$. For any open interval $(a,b)$ there are dense $\Lambda\subseteq(a,b)$ and normally ascending open sets $\{O_\lambda\}_{\lambda\in\Lambda}$ with $F\subseteq O_\lambda\subseteq\overline{O_\lambda}\subseteq U$ for all $\lambda\in\Lambda$.
\begin{proof} By affine reparametrization, take $(a,b)=(0,1)$ and $\Lambda=\{m2^{-n}:0<m<2^n,\ n\in\N\}$.\\
Choose $O_{1/2}$ with $F\subseteq O_{1/2}\subseteq\overline{O_{1/2}}\subseteq U$ by Proposition \ref{prop:normality-nested-neighborhoods}.\\
Inductively, for each new dyadic $p<r<q$, choose $O_r$ with $\overline{O_p}\subseteq O_r\subseteq\overline{O_r}\subseteq O_q$, interprete endpoints as $F$, $U$\\
 This preserves normal ascent and containment in $U$.\\
\hlt{Structural Commentary}: Normality inserts closed-neighborhood buffers.\\
\hlt{Key Insight}: Dyadic induction builds the whole ascending ladder.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:urysohn-lemma-normal-space} \hlt{Urysohn's Lemma}\\
Let $X$ be normal, let $A,B\subseteq X$ be disjoint closed sets, and let $a<b$ in $\R$.\\
Then $\exists$ continuous $f:X\to[a,b]$ such that $f|_A=a$ and $f|_B=b$.
\begin{proof}
Apply Lemma \ref{lem:normal-dyadic-neighborhood-ladder} to $F=A$ and $U=X\setminus B$. Lemma \ref{lem:normally-ascending-function-continuity} applied to the resulting ladder gives continuous $f:X\to[a,b]$. Since $A\subseteq O_\lambda$ for all $\lambda$ and $B\cap\bigcup_\lambda O_\lambda=\emptyset$, the definition gives $f|_A=a$ and $f|_B=b$.\\
\hlt{Structural Commentary}: A nested open ladder becomes a continuous separator.\\
\hlt{Key Insight}: Normality separates closed sets by a real-valued function.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:tietze-extension-bounded-normal-space} \hlt{Tietze Extension Theorem, Bounded Form}\\
Let $X$ be normal, let $F\subseteq X$ be closed, let $a<b$, and let $f:F\to[a,b]$ be continuous.\\
Then $\exists$ continuous $g:X\to[a,b]$ with $g|_F=f$.
\begin{proof}
After an affine change of range, assume $[a,b]=[-1,1]$. The Urysohn step is: if $\abs{h}\leq A$ on $F$, then some continuous $u:X\to[-A/3,A/3]$ satisfies $\abs{h-u}\leq2A/3$ on $F$. Starting with $s_0=0$, apply this to $h=f-s_{n-1}$ to get $u_n$ with $\abs{u_n}\leq\frac13(2/3)^{n-1}$ on $X$ and $\abs{f-s_n}\leq(2/3)^n$ on $F$, where $s_n=s_{n-1}+u_n$.\\
Then $\sum_nu_n$ converges uniformly to continuous $g:X\to[-1,1]$ and $g|_F=f$. Undo the affine change of range.\\
\hlt{Structural Commentary}: Urysohn corrections shrink the residual geometrically.\\
\hlt{Key Insight}: Uniform convergence turns approximate extensions into an exact one.
\end{proof}
\end{theorem}
