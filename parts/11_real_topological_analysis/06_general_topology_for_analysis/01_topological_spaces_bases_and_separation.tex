\subsection{Topological Spaces, Bases, and Separation}

General topology retains the open-set structure underlying metric limits and continuity.\\
The basic objects here are topological spaces, bases, subspaces, closure operations, and separation axioms.

\begin{definition} \label{def:topological-space} \hlt{Topological Space}\\
A \hlt{topology} on a set $X$ is a collection $\mathcal{T}\subseteq 2^X$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\emptyset,X\in\mathcal{T}$;
\item if $\{\mathcal{O}_\alpha\}_{\alpha\in A}\subseteq\mathcal{T}$, then $\bigcup\limits_{\alpha\in A}\mathcal{O}_\alpha\in\mathcal{T}$;
\item if $\mathcal{O}_1,\ldots,\mathcal{O}_n\in\mathcal{T}$ with $n\geq1$, then $\bigcap\limits_{j=1}^n\mathcal{O}_j\in\mathcal{T}$.
\end{enumerate}
Then $(X,\mathcal{T})$ is a \hlt{topological space}.
\end{definition}

\begin{definition} \label{def:open-closed-neighborhood-topological} \hlt{Open, Closed, and Neighborhoods}\\
Let $(X,\mathcal{T})$ be a topological space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A set $U\subseteq X$ is \hlt{open} iff $U\in\mathcal{T}$.
\item A set $F\subseteq X$ is \hlt{closed} iff $X\setminus F\in\mathcal{T}$.
\item For $x\in X$, a \hlt{neighborhood} of $x$ is an open set $U\subseteq X$ with $x\in U$.
\item For $A\subseteq X$, a \hlt{neighborhood} of $A$ is an open set $U\subseteq X$ with $A\subseteq U$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:open-neighborhood-test-topological} \hlt{Open Sets by Neighborhoods}\\
Let $(X,\mathcal{T})$ be a topological space and let $E\subseteq X$. Then $E\in\mathcal{T}$ iff $(\forall x\in E)(\exists U\in\mathcal{T})\,[x\in U\subseteq E]$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $E\in\mathcal{T}$, take $U=E$ for $x\in E$.
\item[$(\Leftarrow)$] Set $\mathcal{U}_E\defeq\{U\in\mathcal{T}:U\subseteq E\}$. The hypothesis implies $E=\bigcup\mathcal{U}_E$, so $E\in\mathcal{T}$ by Definition \ref{def:topological-space}(ii).
\end{enumerate}
\hlt{Structural Commentary}: The union axiom assembles pointwise open witnesses into the entire set.\\
\hlt{Key Insight}: Local open containment is equivalent to global openness.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:metric-open-sets-form-topology} \hlt{Metric Open Sets Form a Topology}\\
Let $(X,d)$ be a metric space, $\tau_d$ be collection of metric-open sets from Definition \ref{def:metric-topology-core-definitions}. Then $\tau_d$ is a topology on $X$.
\begin{proof}
$\emptyset,X\in\tau_d$ by Definition \ref{def:metric-topology-core-definitions}. Collection $\tau_d$ is closed under arbitrary unions and finite intersections by Theorem \ref{thm:metric-open-closed-set-calculus}(i),(iii).\\
Hence $\tau_d$ is a topology by Definition \ref{def:topological-space}.\\
\hlt{Structural Commentary}: Metric union and finite-intersection calculus verifies the three topology axioms.\\
\hlt{Key Insight}: General topology keeps the open-set calculus and forgets the distance function.
\end{proof}
\end{proposition}

\begin{example} \label{ex:discrete-and-trivial-topologies} \hlt{Discrete and Trivial Topologies}\\
On a nonempty set $X$, the \hlt{discrete topology} is $2^X$, so every subset is open and closed.\\
It is induced by the discrete metric. The \hlt{trivial topology} is $\{\emptyset,X\}$, so the only neighborhood of any point is $X$.
\end{example}

\begin{definition} \label{def:topological-closure-interior-boundary} \hlt{Closure, Interior, and Boundary}\\
Let $(X,\mathcal{T})$ be a topological space and let $E\subseteq X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{closure} is $\overline{E}\defeq\{x\in X:(\forall U\in\mathcal{T})\,[x\in U\Rightarrow U\cap E\neq\emptyset]\}$.
\item The \hlt{interior} is $\operatorname{int}(E)\defeq\bigcup\{U\in\mathcal{T}:U\subseteq E\}$.
\item The \hlt{exterior} is $\operatorname{ext}(E)\defeq\operatorname{int}(X\setminus E)$.
\item The \hlt{boundary} of $E$ is $\operatorname{bd}(E)\defeq\overline{E}\setminus\operatorname{int}(E)$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:topological-closure-closed-calculus} \hlt{Closure, Interior, and Boundary Tests}\\
Let $(X,\mathcal{T})$ be a topological space and let $E\subseteq X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{int}(E)$ is the largest open subset of $E$.
\item $E$ is open iff $E=\operatorname{int}(E)$.
\item $\overline{E}$ is the smallest closed subset of $X$ containing $E$.
\item $E$ is closed iff $E=\overline{E}$.
\item For $x\in X$, $x\in\operatorname{bd}(E)$ iff $(\forall U\in\mathcal{T})\,[x\in U\Rightarrow U\cap E\neq\emptyset\land U\cap(X\setminus E)\neq\emptyset]$.
\item $\operatorname{bd}(E)=\overline{E}\cap\overline{X\setminus E}$.
\item $\operatorname{bd}(E)$ is closed.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] The set $\operatorname{int}(E)=\bigcup\{U\in\mathcal{T}:U\subseteq E\}$ is open by Definition \ref{def:topological-space}(ii).\\
It contains every open subset of $E$ and is contained in $E$, so it is the largest open subset of $E$.
\item[$(ii)$] ($\Rightarrow$) If $E$ is open, then $E\subseteq\operatorname{int}(E)\subseteq E$, so $E=\operatorname{int}(E)$.
\item[$(ii)$] ($\Leftarrow$) If $E=\operatorname{int}(E)$, then $E$ is open by (i).
\item[$(iii)$] Every $x\in E$ belongs to $\overline{E}$ by Definition \ref{def:topological-closure-interior-boundary}(i). If $x\notin\overline{E}$, choose neighborhood $U$ of $x$ with $U\cap E=\emptyset$ by same definition.\\
Then $U\subseteq X\setminus\overline{E}$. Thus $X\setminus\overline{E}$ open by Proposition \ref{prop:open-neighborhood-test-topological}, so $\overline{E}$ closed.\\
If $F$ is closed and $E\subseteq F$, then $X\setminus F$ is an open neighborhood of each $x\notin F$ and is disjoint from $E$.\\
Hence $\overline{E}\subseteq F$, proving the smallest-closed-superset claim.
\item[$(iv)$] ($\Rightarrow$) If $E$ is closed, then $\overline{E}\subseteq E$ by (iii), while $E\subseteq\overline{E}$. Hence $E=\overline{E}$.
\item[$(iv)$] ($\Leftarrow$) If $E=\overline{E}$, then $E$ is closed by (iii).
\item[$(v)$] ($\Rightarrow$) If $x\in\operatorname{bd}(E)$, then $x\in\overline{E}\setminus\operatorname{int}(E)$ by Definition \ref{def:topological-closure-interior-boundary}(iv). Thus every neighborhood of $x$ meets $E$ by Definition \ref{def:topological-closure-interior-boundary}(i).\\
No neighborhood of $x$ lies in $E$, so each neighborhood also meets $X\setminus E$.
\item[$(v)$] ($\Leftarrow$) If every neighborhood of $x$ meets both $E$ and $X\setminus E$, then $x\in\overline{E}$ and $x\notin\operatorname{int}(E)$. Hence $x\in\operatorname{bd}(E)$ by Definition \ref{def:topological-closure-interior-boundary}(iv).
\item[$(vi)$] By (v), $x\in\operatorname{bd}(E)$ iff $x\in\overline{E}\cap\overline{X\setminus E}$. Thus $\operatorname{bd}(E)=\overline{E}\cap\overline{X\setminus E}$.
\item[$(vii)$] Both factors in (vi) are closed by (iii), intersection is closed because its complement is the union of their open complements.
\end{enumerate}
\hlt{Structural Commentary}: Open complements make closure minimal; two-sided neighborhood contact identifies the boundary.\\
\hlt{Key Insight}: Boundary points persist under approaches from both $E$ and its complement.
\end{proof}
\end{proposition}

\begin{definition} \label{def:basis-for-topology} \hlt{Bases and Subbases}\\
Let $X$ be a set.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\mathcal{T}$ be a topology on $X$, let $x\in X$, and let $\mathcal{B}_x$ be a collection of neighborhoods of $x$.\\
Then $\mathcal{B}_x$ is a \hlt{neighborhood basis at $x$} iff $(\forall U\in\mathcal{T})\,[x\in U\Rightarrow(\exists B\in\mathcal{B}_x)\,B\subseteq U]$.
\item A collection $\mathcal{B}\subseteq 2^X$ is a \hlt{basis} for a topology on $X$ iff:
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item $X=\bigcup\limits_{B\in\mathcal{B}}B$;
\item $(\forall B_1,B_2\in\mathcal{B})(\forall x\in B_1\cap B_2)(\exists B_3\in\mathcal{B})\,x\in B_3\subseteq B_1\cap B_2$.
\end{enumerate}
\item If $\mathcal{B}$ is a basis on $X$, the collection generated by $\mathcal{B}$ is $\mathcal{T}_{\mathcal{B}}\defeq\{\bigcup\limits_{B\in\mathcal{U}}B:\mathcal{U}\subseteq\mathcal{B}\}$.
\item For $\mathcal{S}\subseteq 2^X$, set $\mathcal{B}_{\mathcal{S}}\defeq\{\bigcap\limits_{j=1}^{n}S_j:n\in\N,\ S_1,\ldots,S_n\in\mathcal{S}\}$. The collection $\mathcal{S}$ is a \hlt{subbasis} on $X$ iff $\bigcup\mathcal{S}=X$ and $\mathcal{B}_{\mathcal{S}}$ is a basis on $X$.
\end{enumerate}
\hlt{Key Insight}: Bases generate open sets by unions; subbases generate bases by finite intersections.
\end{definition}

\begin{proposition} \label{prop:basis-generated-topology-characterization} \hlt{Basis Characterization}\\
Let $X$ be a set and let $\mathcal{B}\subseteq 2^X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathcal{B}$ is a basis on $X$, then $\mathcal{T}_{\mathcal{B}}\defeq\{\bigcup\mathcal{U}:\mathcal{U}\subseteq\mathcal{B}\}$ is a topology on $X$.
\item Let $\mathcal{T}$ be a topology on $X$ with $\mathcal{B}\subseteq\mathcal{T}$ and $(\forall U\in\mathcal{T}\setminus\{\emptyset\})(\exists\mathcal{U}\subseteq\mathcal{B})\,U=\bigcup\mathcal{U}$. Then $\mathcal{B}$ is a basis on $X$.
\item Under the hypotheses of (ii), $\mathcal{T}=\mathcal{T}_{\mathcal{B}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $O_{\mathcal{U}}\defeq\bigcup\mathcal{U}$ for $\mathcal{U}\subseteq\mathcal{B}$. Then $X=O_{\mathcal{B}}$, $\emptyset=O_{\emptyset}$, and $\bigcup\limits_{\alpha}O_{\mathcal{U}_\alpha}=O_{\bigcup\limits_{\alpha}\mathcal{U}_\alpha}$.\\
If $x\in O_{\mathcal{U}}\cap O_{\mathcal{V}}$, choose $B_1\in\mathcal{U}$ and $B_2\in\mathcal{V}$ with $x\in B_1\cap B_2$.\\
Choose $B_3\in\mathcal{B}$ with $x\in B_3\subseteq B_1\cap B_2$ by the basis refinement property. Hence $O_{\mathcal{U}}\cap O_{\mathcal{V}}=\bigcup\{B\in\mathcal{B}:B\subseteq O_{\mathcal{U}}\cap O_{\mathcal{V}}\}$.\\
Closure under finite intersections follows from this binary identity by Theorem \ref{thm:mathematical-induction}. Thus $\mathcal{T}_{\mathcal{B}}$ is a topology.
\item If $X\neq\emptyset$, the hypothesis for $U=X$ supplies $\mathcal{U}\subseteq\mathcal{B}$ with $X=\bigcup\mathcal{U}$. If $X=\emptyset$, then $\bigcup\mathcal{B}=X$.\\
If $B_1,B_2\in\mathcal{B}$ and $x\in B_1\cap B_2$, then $B_1\cap B_2\in\mathcal{T}\setminus\{\emptyset\}$. Choose $\mathcal{V}\subseteq\mathcal{B}$ with $B_1\cap B_2=\bigcup\mathcal{V}$ by the hypothesis.\\
Then choose $B_3\in\mathcal{V}$ with $x\in B_3\subseteq B_1\cap B_2$. Hence $\mathcal{B}$ is a basis on $X$ by Definition \ref{def:basis-for-topology}(ii).
\item The inclusion $\mathcal{T}_{\mathcal{B}}\subseteq\mathcal{T}$ follows by $\mathcal{B}\subseteq\mathcal{T}$ and closure under arbitrary unions.\\
Every nonempty $U\in\mathcal{T}$ belongs to $\mathcal{T}_{\mathcal{B}}$ by the hypothesis. Since $\emptyset=\bigcup\emptyset\in\mathcal{T}_{\mathcal{B}}$, one has $\mathcal{T}\subseteq\mathcal{T}_{\mathcal{B}}$ and therefore $\mathcal{T}=\mathcal{T}_{\mathcal{B}}$.
\end{enumerate}
\hlt{Structural Commentary}: Basis refinements rewrite binary intersections as basis unions; topology axioms complete the generation.\\
\hlt{Key Insight}: Refinement is precisely the intersection mechanism missing from arbitrary unions.
\end{proof}
\end{proposition}

\begin{example} \label{ex:metric-balls-basis} \hlt{Metric Balls as a Basis}\\
In metric space $(X,d)$, every point $x$ lies in $B_1(x)$. If $z\in B_r(x)\cap B_s(y)$, $B_\varepsilon(z)\subseteq B_r(x)\cap B_s(y)$ for $\varepsilon\defeq\min\{r-d(x,z),s-d(y,z)\}>0$.\\
Hence $\{B_r(x):x\in X,\ r>0\}$ is a basis. The family $\{B_r(q):q\in\Q^k,\ r\in\Q_{>0}\}$ is countable by Theorem \ref{thm:finite-tuples-over-countable-set-countable} and Corollary \ref{cor:rationals-countable}.\\
Coordinatewise rational density and the triangle inequality make this family a basis on $\R^k$ by Theorem \ref{thm:archimedean-and-rational-density}(ii).\\
If $a<b$, then $\{[a,c):a<c<b\}\cup\{(c,b]:a<c<b\}$ is a subbasis for the usual topology on $[a,b]$.
\end{example}

\begin{definition} \label{def:topological-subspace} \hlt{Subspace Topology}\\
Let $(X,\mathcal{T})$ be topological space, $E\subseteq X$. The \hlt{subspace topology} on $E$ is $\mathcal{T}_E\defeq\{E\cap U:U\in\mathcal{T}\}$, and $(E,\mathcal{T}_E)$ is a \hlt{subspace} of $X$.
\end{definition}

\begin{proposition} \label{prop:subspace-collection-is-topology} \hlt{The Subspace Collection Is a Topology}\\
Let $(X,\mathcal{T})$ be a topological space and let $E\subseteq X$. Then $\mathcal{T}_E$ from Definition \ref{def:topological-subspace} is a topology on $E$.
\begin{proof}
The identities $\emptyset=E\cap\emptyset$ and $E=E\cap X$ put $\emptyset,E$ in $\mathcal{T}_E$. For $(U_\alpha)_{\alpha\in A}\subseteq\mathcal{T}$, $\bigcup\limits_{\alpha\in A}(E\cap U_\alpha)=E\cap\bigcup\limits_{\alpha\in A}U_\alpha\in\mathcal{T}_E$.\\
For $U_1,\ldots,U_n\in\mathcal{T}$ with $n\geq1$, $\bigcap\limits_{j=1}^n(E\cap U_j)=E\cap\bigcap\limits_{j=1}^nU_j\in\mathcal{T}_E$. Thus $\mathcal{T}_E$ is a topology on $E$ by Definition \ref{def:topological-space}.\\
\hlt{Structural Commentary}: Intersecting ambient opens with $E$ preserves the topology axioms.\\
\hlt{Key Insight}: Unions and finite intersections commute with restriction to $E$.
\end{proof}
\end{proposition}

\begin{definition} \label{def:separation-axioms} \hlt{Separation Properties}\\
Let $(X,\mathcal{T})$ be a topological space.\\
Disjoint sets $A,B\subseteq X$ are \hlt{separated by disjoint neighborhoods} iff $(\exists U,V\in\mathcal{T})\,[A\subseteq U\land B\subseteq V\land U\cap V=\emptyset]$. Space $X$ is:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{$T_1$} iff $(\forall x\in X)\,\{x\}$ is closed;
\item \hlt{Hausdorff} iff $\{x\}$ and $\{y\}$ are separated by disjoint neighborhoods whenever $x,y\in X$ and $x\neq y$;
\item \hlt{regular} iff $X$ is $T_1$ and $\{x\},F$ can be separated by disjoint neighborhoods when $F\subseteq X$ is closed and $x\in X\setminus F$;
\item \hlt{normal} iff $X$ is $T_1$ and any disjoint closed $F,G\subseteq X$ can be separated by disjoint neighborhoods.
\end{enumerate}
\hlt{Key Insight}: Successive axioms separate distinct points, point--closed-set pairs, and closed-set pairs.
\end{definition}

\begin{proposition} \label{prop:t1-iff-singletons-closed} \hlt{$T_1$ Spaces and Closed Points}\\
Let $(X,\mathcal{T})$ be a topological space. Then $X$ is $T_1$ iff $(\forall x,y\in X)\,[x\neq y\Rightarrow(\exists U\in\mathcal{T})\,[x\in U\land y\notin U]]$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $X$ is $T_1$, and fix $x,y\in X$ with $x\neq y$. Then $\{y\}$ is closed, so $U=X\setminus\{y\}\in\mathcal{T}$ satisfies $x\in U$ and $y\notin U$.
\item[$(\Leftarrow)$] Assume the stated condition. Fix $y\in X$ and set $\mathcal{U}_y\defeq\{U\in\mathcal{T}:y\notin U\}$.\\
Every $x\in X\setminus\{y\}$ belongs to a member of $\mathcal{U}_y$ by the condition, while no member contains $y$.\\
Hence $X\setminus\{y\}=\bigcup\mathcal{U}_y\in\mathcal{T}$ by Definition \ref{def:topological-space}(ii), so $\{y\}$ is closed. Thus $X$ is $T_1$ by Definition \ref{def:separation-axioms}(i).
\end{enumerate}
\hlt{Structural Commentary}: Point exclusion makes singleton complements open.\\
\hlt{Key Insight}: $T_1$ means each point can be omitted near every other point.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:metric-spaces-are-normal} \hlt{Metric Spaces Are Normal}\\
Let $(X,d)$ be a metric space with metric topology $\tau_d$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall x\in X)\,\{x\}$ is closed.
\item If $F_1,F_2\subseteq X$ are closed and $F_1\cap F_2=\emptyset$, then $(\exists O_1,O_2\in\tau_d)\,[F_1\subseteq O_1\land F_2\subseteq O_2\land O_1\cap O_2=\emptyset]$.
\item $X$ is normal.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $x\in X$. For $y\in X\setminus\{x\}$, $B_{\frac{d(x,y)}{2}}(y)\subseteq X\setminus\{x\}$. Thus $X\setminus\{x\}$ is open and $\{x\}$ is closed.
\item
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Empty cases}: If $F_1=\emptyset$, take $(O_1,O_2)=(\emptyset,X)$. If $F_1\neq\emptyset$, $F_2=\emptyset$, take $(O_1,O_2)=(X,\emptyset)$. Hence assume $F_1,F_2\neq\emptyset$.
\item \hlt{Distance construction}: For $i=1,2$, set $d_i(x)\defeq\operatorname{dist}(x,F_i)$ and $O_i\defeq\{x\in X:d_i(x)<d_{3-i}(x)\}$.\\
The complement of $F_i$ lies in $\tau_d$ by Definition \ref{def:open-closed-neighborhood-topological}(ii). It is therefore metric-open by Definition \ref{def:metric-topology-core-definitions}(iii).\\
Thus $F_i$ is metric-closed by Corollary \ref{cor:closed-iff-complement-open}. The map $d_i$ is continuous by Proposition \ref{prop:distance-to-set-is-continuous}(ii).\\
Its zero set satisfies $d_i^{-1}(\{0\})=\overline{F_i}=F_i$ by Proposition \ref{prop:distance-to-set-is-continuous}(iii) and Theorem \ref{thm:closure-smallest-closed-containing-set}(ii).
\item \hlt{Open containment}: The map $d_i-d_{3-i}$ is continuous by Theorem \ref{thm:continuous-algebra-and-components}(i).\\
Hence $O_i$ is open by Theorem \ref{thm:continuity-open-preimages}. If $x\in F_i$, then $d_i(x)=0<d_{3-i}(x)$ because $x\notin F_{3-i}=d_{3-i}^{-1}(\{0\})$. Thus $F_i\subseteq O_i$.
\item \hlt{Disjointness}: The intersection $O_1\cap O_2$ is empty because $d_1(x)<d_2(x)$ and $d_2(x)<d_1(x)$ cannot both hold.
\end{enumerate}
\item The space $X$ is normal by (i), (ii), and Definition \ref{def:separation-axioms}(iv).
\end{enumerate}
\hlt{Structural Commentary}: Distance zero sets and strict distance comparisons construct the required disjoint open neighborhoods.\\
\hlt{Key Insight}: No point can be closer to each closed set than to the other.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:normality-nested-neighborhoods} \hlt{Normality by Nested Neighborhoods}\\
Let $X$ be a $T_1$ topological space. Then $X$ is normal iff each closed--open inclusion $F\subseteq U$ admits an open $O$ with $F\subseteq O\subseteq\overline{O}\subseteq U$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $X$ is normal. Let $F\subseteq X$ be closed and let $U\subseteq X$ be open with $F\subseteq U$.\\
Choose disjoint open sets $O,V$ with $F\subseteq O$ and $X\setminus U\subseteq V$ by Definition \ref{def:separation-axioms}(iv).\\
Then $O\subseteq X\setminus V\subseteq U$. Since $X\setminus V$ is closed, $\overline{O}\subseteq X\setminus V\subseteq U$ by Proposition \ref{prop:topological-closure-closed-calculus}(iii).
\item[$(\Leftarrow)$] Assume the nested-neighborhood property. Let $A,B\subseteq X$ be closed with $A\cap B=\emptyset$.\\
By the property with $F=A$ and $U=X\setminus B$, choose open $O$ with $A\subseteq O\subseteq\overline{O}\subseteq X\setminus B$.\\
The set $\overline{O}$ is closed by Proposition \ref{prop:topological-closure-closed-calculus}(iii). Hence $O$ and $X\setminus\overline{O}$ are disjoint open neighborhoods of $A$ and $B$, respectively.\\
Since $X$ is $T_1$, $X$ is normal by Definition \ref{def:separation-axioms}(iv).
\end{enumerate}
\hlt{Structural Commentary}: Disjoint neighborhoods place $\overline{O}$ inside $U$; taking its complement reverses the construction.\\
\hlt{Key Insight}: Separating $F$ from $X\setminus U$ is equivalent to closing $O$ within $U$.
\end{proof}
\end{proposition}
