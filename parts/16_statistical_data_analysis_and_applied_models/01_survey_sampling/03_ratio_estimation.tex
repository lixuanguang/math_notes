\subsection{Ratio Estimation}
% Source integration: supplied Chapter 7 screenshots, Section 7.4 and selected
% ratio-estimation problems.

Ratio estimation uses an auxiliary variable whose population mean or total is
known.  It is especially useful when the auxiliary variable is strongly
correlated with the target variable.

\begin{definition} \label{def:finite-population-ratio} \hlt{Population and Sample Ratios}\\
Let $U\defeq\{1,\ldots,N\}$ and fixed pairs $(x_i,y_i)\in\R^2$, $i\in U$.\\
Set $\mu_x\defeq \frac{1}{N}\sum\limits_{i=1}^{N}x_i$, $\mu_y\defeq \frac{1}{N}\sum\limits_{i=1}^{N}y_i$, $\tau_x\defeq\sum\limits_{i=1}^{N}x_i$, and $\tau_y\defeq\sum\limits_{i=1}^{N}y_i$, with $\mu_x\neq0$.\\
For SRS pairs $(X_i,Y_i)_{i=1}^{n}$, set $\overline{X}\defeq \frac{1}{n}\sum\limits_{i=1}^{n}X_i$ and $\overline{Y}\defeq \frac{1}{n}\sum\limits_{i=1}^{n}Y_i$, with $\overline{X}\neq0$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Population ratio}: $r\defeq \sum\limits_{i=1}^{N}y_i/\sum\limits_{i=1}^{N}x_i =\frac{\mu_y}{\mu_x}$.
\item \hlt{Sample ratio}: $R\defeq \frac{\overline{Y}}{\overline{X}}$.
\item \hlt{Ratio mean estimator}: $\overline{Y}_R\defeq \mu_xR$ when $\mu_x$ is known.
\item \hlt{Ratio total estimator}: $T_R\defeq \tau_xR$ when $\tau_x$ is known.
\end{enumerate}
\hlt{Key Insight}: $\{i\in U:x_i=0\}=\emptyset$ does not imply $r=\frac{1}{N}\sum\limits_{i=1}^{N}\frac{y_i}{x_i}$.
\end{definition}

\begin{definition} \label{def:finite-population-covariance-correlation} \hlt{Population Covariance and Correlation}\\
Let $U\defeq\{1,\ldots,N\}$ and fixed pairs $(x_i,y_i)\in\R^2$, $i\in U$.\\
Set $\mu_x\defeq \frac{1}{N}\sum\limits_{i=1}^{N}x_i$, $\mu_y\defeq \frac{1}{N}\sum\limits_{i=1}^{N}y_i$, $\sigma_x^2\defeq \frac{1}{N}\sum\limits_{i=1}^{N}(x_i-\mu_x)^2$, and $\sigma_y^2\defeq \frac{1}{N}\sum\limits_{i=1}^{N}(y_i-\mu_y)^2$.\\
For sample pairs $(X_i,Y_i)_{i=1}^{n}$, set $\overline{X}\defeq \frac{1}{n}\sum\limits_{i=1}^{n}X_i$ and $\overline{Y}\defeq \frac{1}{n}\sum\limits_{i=1}^{n}Y_i$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Population covariance}: $\sigma_{xy}\defeq \frac{1}{N}\sum\limits_{i=1}^{N}(x_i-\mu_x)(y_i-\mu_y)$.
\item \hlt{Population correlation}: $\rho\defeq\frac{\sigma_{xy}}{\sigma_x\sigma_y}$ when $\sigma_x,\sigma_y>0$.
\item \hlt{Sample covariance}: $s_{xy}\defeq \frac{1}{n-1}\sum\limits_{i=1}^{n}(X_i-\overline{X})(Y_i-\overline{Y})$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:srs-ratio-approximate-variance-bias} \hlt{Approximate Moments of the Ratio Estimator}\\
Let $U\defeq\{1,\ldots,N\}$, $N>1$, $1\leq n\leq N$, and let $(x_a,y_a)\in\R^2$ be fixed for $a\in U$.\\
Set $\lambda_n\defeq\frac1n(1-\frac{n-1}{N-1})$, $\mu_x\defeq\frac1N\sum\limits_{a=1}^{N}x_a$, $\mu_y\defeq\frac1N\sum\limits_{a=1}^{N}y_a$, $r\defeq\frac{\mu_y}{\mu_x}$\\
with $\mu_x\neq0$, $\sigma_x^2\defeq\frac1N\sum\limits_{a=1}^{N}(x_a-\mu_x)^2$, $\sigma_y^2\defeq\frac1N\sum\limits_{a=1}^{N}(y_a-\mu_y)^2$, and $\sigma_{xy}\defeq\frac1N\sum\limits_{a=1}^{N}(x_a-\mu_x)(y_a-\mu_y)$.\\
Let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$, let $(J_1,\ldots,J_n)$ be a random ordering of $S$, set $(X_k,Y_k)\defeq(x_{J_k},y_{J_k})$, $\overline{X}\defeq\frac1n\sum\limits_{k=1}^{n}X_k$, $\overline{Y}\defeq\frac1n\sum\limits_{k=1}^{n}Y_k$, and $R\defeq\frac{\overline{Y}}{\overline{X}}$ on $\{\overline{X}\neq0\}$.
The subscript $d$ denotes design moments.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Variance}: first-order Taylor gives $\operatorname{Var}_d(R)\approx\frac{\lambda_n}{\mu_x^2}(r^2\sigma_x^2+\sigma_y^2-2r\sigma_{xy})$.
\item \hlt{Mean}: second-order Taylor gives $\mathbb{E}_dR\approx r+\frac{\lambda_n}{\mu_x^2}(r\sigma_x^2-\sigma_{xy})$.
\end{enumerate}
\begin{proof} Let $(A_q,B_q)_{q=1}^{N}$ be a uniformly random ordering of the population pairs; the ordered sample has the law of the first $n$ entries. For $q\neq r$, write $\gamma_x\defeq\operatorname{Cov}_d(A_q,A_r)$, $\gamma_y\defeq\operatorname{Cov}_d(B_q,B_r)$, and $\gamma_{xy}\defeq\operatorname{Cov}_d(A_q,B_r)$.\\
The fixed sums give $0=\operatorname{Var}_d(\sum\limits_{q=1}^{N}A_q)=N\sigma_x^2+N(N-1)\gamma_x$, $0=\operatorname{Var}_d(\sum\limits_{q=1}^{N}B_q)=N\sigma_y^2+N(N-1)\gamma_y$,\\
and $0=\operatorname{Cov}_d(\sum\limits_{q=1}^{N}A_q,\sum\limits_{q=1}^{N}B_q)=N\sigma_{xy}+N(N-1)\gamma_{xy}$.\\
Hence $\gamma_x=-\frac{\sigma_x^2}{N-1}$, $\gamma_y=-\frac{\sigma_y^2}{N-1}$, $\gamma_{xy}=-\frac{\sigma_{xy}}{N-1}$, $V_x\defeq\operatorname{Var}_d(\overline{X})=\lambda_n\sigma_x^2$, $V_y\defeq\operatorname{Var}_d(\overline{Y})=\lambda_n\sigma_y^2$,\\
and $C_{xy}\defeq\operatorname{Cov}_d(\overline{X},\overline{Y})=\lambda_n\sigma_{xy}$. Let $\Sigma\defeq\left(\begin{smallmatrix}V_x&C_{xy}\\ C_{xy}&V_y\end{smallmatrix}\right)$.\\
For $g(x,y)=\frac{y}{x}$ and $\mu\defeq(\mu_x,\mu_y)$, $g(\mu)=r$, $\nabla g(\mu)=(-\frac r{\mu_x},\frac1{\mu_x})^{\top}$, and $H_g(\mu)=\left(\begin{smallmatrix}2r/\mu_x^2&-1/\mu_x^2\\ -1/\mu_x^2&0\end{smallmatrix}\right)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Variance}: $\operatorname{Var}_d(R)\approx\nabla g(\mu)^{\top}\Sigma\nabla g(\mu)=\frac{\lambda_n}{\mu_x^2}(r^2\sigma_x^2+\sigma_y^2-2r\sigma_{xy})$.
\item \hlt{Mean}: $\mathbb{E}_dR\approx g(\mu)+\frac12\operatorname{tr}(H_g(\mu)\Sigma)=r+\frac{\lambda_n}{\mu_x^2}(r\sigma_x^2-\sigma_{xy})$.
\end{enumerate}
\hlt{Structural Commentary}: The ratio estimator is locally linear for variance and locally quadratic for bias.\\
\hlt{Key Insight}: Ratio variance is scaled residual variation in $Y-rX$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:ratio-estimated-standard-error} \hlt{Estimated Standard Error of a Ratio}\\
Let $U\defeq\{1,\ldots,N\}$, $N>1$, $1<n\leq N$, and let $(x_a,y_a)\in\R^2$ be fixed for $a\in U$.\\
Set $\lambda_n\defeq\frac1n(1-\frac{n-1}{N-1})$, $\mu_x\defeq\frac1N\sum\limits_{a=1}^{N}x_a$, $\tau_x\defeq\sum\limits_{a=1}^{N}x_a$, $\tau_y\defeq\sum\limits_{a=1}^{N}y_a$, and $r\defeq\frac{\tau_y}{\tau_x}$ with $\tau_x\neq0$.\\
Let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$. For a random ordering $(J_1,\ldots,J_n)$ of $S$,\\
set $(X_i,Y_i)\defeq(x_{J_i},y_{J_i})$, $\overline{X}\defeq\frac1n\sum\limits_{i=1}^{n}X_i$, and $\overline{Y}\defeq\frac1n\sum\limits_{i=1}^{n}Y_i$.\\
Assume $\overline{X}\neq0$ and set $R\defeq\frac{\overline{Y}}{\overline{X}}$, $s_x^2\defeq\frac{1}{n-1}\sum\limits_{i=1}^{n}(X_i-\overline{X})^2$, $s_y^2\defeq\frac{1}{n-1}\sum\limits_{i=1}^{n}(Y_i-\overline{Y})^2$, $s_{xy}\defeq\frac{1}{n-1}\sum\limits_{i=1}^{n}(X_i-\overline{X})(Y_i-\overline{Y})$, and $q_R\defeq R^2s_x^2+s_y^2-2Rs_{xy}$. If $0<\alpha<1$ and $z(\frac{\alpha}{2})\defeq\Phi^{-1}(1-\frac{\alpha}{2})$ for the standard normal cdf $\Phi$, then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Ratio variance}: $s_R^2\defeq\frac{\lambda_n}{\overline{X}^2}q_R$ estimates $\operatorname{Var}_d(R)$.
\item \hlt{Ratio interval}: $\operatorname{CI}_{1-\alpha}(r)\defeq R\pm z(\frac{\alpha}{2})s_R$.
\item \hlt{Mean and total}: for $\overline{Y}_R\defeq\mu_xR$ and $T_R\defeq\tau_xR$, $\operatorname{SE}(\overline{Y}_R)=\abs{\mu_x}s_R$ and $\operatorname{SE}(T_R)=\abs{\tau_x}s_R$ when the corresponding auxiliary constant is known.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The plug-in map $(\mu_x,r,\sigma_x^2,\sigma_y^2,\sigma_{xy})\mapsto(\overline{X},R,s_x^2,s_y^2,s_{xy})$ in Theorem \ref{thm:srs-ratio-approximate-variance-bias} yields\\
$\widehat{\operatorname{Var}}_d(R)=\frac{\lambda_n}{\overline{X}^2}(R^2s_x^2+s_y^2-2Rs_{xy})=s_R^2$.
\item If $Z_R\defeq(R-r)/s_R\approx\mathcal{N}(0,1)$, then $\mathbb{P}_d(\abs{Z_R}\leq z(\frac{\alpha}{2}))\approx1-\alpha$, equivalently $r\in R\pm z(\frac{\alpha}{2})s_R$.
\item For known $c$, $\widehat{\operatorname{Var}}_d(cR)=c^2s_R^2$, so $\operatorname{SE}(cR)=\abs{c}s_R$; set $c=\mu_x$ and $c=\tau_x$.
\end{enumerate}
\hlt{Structural Commentary}: Plug in sample moments to estimate the ratio variance.\\
\hlt{Key Insight}: Ratio uncertainty is residual uncertainty in $Y-RX$.
\end{proof}
\end{corollary}

\begin{method} \label{meth:when-ratio-estimation-helps} \hlt{When the Ratio Estimate Helps}\\
Let $U=\{1,\ldots,N\}$ contain fixed pairs $(x_i,y_i)$ with $\mu_x,\mu_y>0$, $\sigma_x,\sigma_y>0$, $\sigma_{xy}$, $\rho\defeq\frac{\sigma_{xy}}{\sigma_x\sigma_y}$, and $r\defeq\frac{\mu_y}{\mu_x}$.\\
For an SRS $(X_k,Y_k)_{k=1}^{n}$, set $\overline{X}\defeq\frac1n\sum\limits_{k=1}^{n}X_k$, $\overline{Y}\defeq\frac1n\sum\limits_{k=1}^{n}Y_k$, $R\defeq\frac{\overline{Y}}{\overline{X}}$, and $\overline{Y}_R\defeq\mu_xR$.\\
Ignoring finite-population corrections:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Variances}: $V_0\defeq\operatorname{Var}(\overline{Y})=\frac{\sigma_y^2}{n}$ and $V_R^\ast\defeq\frac1n(r^2\sigma_x^2+\sigma_y^2-2r\sigma_{xy})$ approximate $\operatorname{Var}(\overline{Y}_R)$.
\item \hlt{Improvement}: $V_R^\ast<V_0\iff 2\sigma_{xy}>r\sigma_x^2\iff 2\rho\sigma_y>r\sigma_x$.
\item \hlt{Coefficient form}: with $C_x\defeq\frac{\sigma_x}{\mu_x}$ and $C_y\defeq\frac{\sigma_y}{\mu_y}$, $V_R^\ast<V_0\iff \rho>\frac12\frac{C_x}{C_y}$.
\item \hlt{Favorable case}: large positive $\rho$ and small $C_x/C_y$.
\end{enumerate}
\end{method}

\begin{example} \label{ex:mortgage-income-ratio-estimate} \hlt{Mortgage-Payment Ratio}\\
For an SRS of $n=100$ recent house buyers, let $Y_i$ be monthly mortgage payment and $X_i$ gross income.\\
Suppose $\overline{X}=3100$, $\overline{Y}=868$, $s_x=1200$, $s_y=250$, and $\widehat\rho\defeq s_{xy}/(s_xs_y)=0.85$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Ratio}: $R=\frac{\overline{Y}}{\overline{X}}=\frac{868}{3100}=0.28$.
\item \hlt{Variance input}: $q\defeq R^2s_x^2+s_y^2-2R\widehat\rho s_xs_y\approx32596$.
\item \hlt{Standard error}: ignoring the finite-population correction, $s_R\approx\frac{\sqrt q}{\sqrt{100}\,\overline{X}}\approx0.00582$.
\item \hlt{Interval}: $\operatorname{CI}_{0.95}(r)=R\pm1.96s_R\approx0.28\pm1.96(0.00582)=(0.269,0.291)$.
\end{enumerate}
Thus the population mortgage-payment-to-income ratio is about $28\%$.
\end{example}

\begin{remark} \label{rem:ratio-bias-and-resampling-todo} \hlt{Bias and Resampling Corrections}\\
For SRS from fixed pairs $(x_i,y_i)$, set $R\defeq\overline{Y}/\overline{X}$, $\overline{Y}_R\defeq\mu_xR$, and $T_R\defeq\tau_xR$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $g(x,y)=y/x$ is nonlinear, generally $\mathbb{E}_dR\neq r$, hence $\mathbb{E}_d\overline{Y}_R\neq\mu_y$ and $\mathbb{E}_dT_R\neq\tau_y$.
\item Bias corrections target $B_d(R)\defeq\mathbb{E}_dR-r$; use resampling when Taylor bias matters.
\end{enumerate}
\end{remark}
