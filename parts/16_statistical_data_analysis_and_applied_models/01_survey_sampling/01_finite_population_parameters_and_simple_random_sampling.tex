\subsection{Finite Population Parameters and Simple Random Sampling}

Survey sampling treats a finite population as fixed and the sample-selection
mechanism as random.  The inferential target is often a population mean, total,
or proportion, while the random variables come from the sampling design.

\begin{definition} \label{def:finite-population-parameters} \hlt{Finite Population Parameters}\\
Let $U\defeq\{1,\ldots,N\}$ and fix $x_i\in\R$, $i\in U$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Population mean}: $\mu\defeq N^{-1}\sum_{i=1}^{N}x_i$.
\item \hlt{Population total}: $\tau\defeq\sum_{i=1}^{N}x_i=N\mu$.
\item \hlt{Population variance}: $\sigma^2\defeq N^{-1}\sum_{i=1}^{N}(x_i-\mu)^2=N^{-1}\sum_{i=1}^{N}x_i^2-\mu^2$.
\item \hlt{Population proportion}: $\{x_i:i\in U\}\subseteq\{0,1\}\implies p\defeq\mu=N^{-1}\sum_{i=1}^{N}x_i$ and $\sigma^2=p(1-p)$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:simple-random-sampling-without-replacement} \hlt{Simple Random Sampling}\\
Let $U\defeq\{1,\ldots,N\}$, $1\leq n\leq N$, and fix $x_i\in\R$, $i\in U$.\\
Let $S$ be a random $n$-element subset of $U$; for a random ordering $J_1,\ldots,J_n$ of $S$, set $X_k\defeq x_{J_k}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Simple random sample}: $\mathbb{P}(S=s)=\binom{N}{n}^{-1}$, $\forall s\subseteq U$ with $|s|=n$.
\item \hlt{Sample mean}: $\overline{X}\defeq n^{-1}\sum_{i=1}^{n}X_i$.
\item \hlt{Total estimator}: $T\defeq N\overline{X}$.
\item \hlt{Sample proportion}: $\{x_i:i\in U\}\subseteq\{0,1\}\implies \widehat p\defeq\overline{X}$.
\end{enumerate}
\end{definition}

\begin{lemma} \label{lem:srs-one-draw-law} \hlt{Law of One Sample Position}\\
Let $U\defeq\{1,\ldots,N\}$, $1\leq n\leq N$, fix $x_a\in\R$ for $a\in U$, set $\mu\defeq N^{-1}\sum_{a=1}^{N}x_a$ and $\sigma^2\defeq N^{-1}\sum_{a=1}^{N}(x_a-\mu)^2$, and let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$.  If $(J_1,\ldots,J_n)$ is a random ordering of $S$, set $X_k\defeq x_{J_k}$.  Let $\zeta_1,\ldots,\zeta_m$ be the distinct values in $\{x_a:a\in U\}$ and set $n_j\defeq\#\{a\in U:x_a=\zeta_j\}$.  For each $k\in\{1,\ldots,n\}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Distribution}: the selected value $X_k$ has the finite-population law $\mathbb{P}(X_k=\zeta_j)=n_j/N$ for $j=1,\ldots,m$.
\item \hlt{Moments}: the finite-population law has mean $\mathbb{E}X_k=\mu$ and variance $\operatorname{Var}(X_k)=\sigma^2$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For every $a\in U$, $\mathbb{P}(J_k=a)=1/N$, so $\mathbb{P}(X_k=\zeta_j)=\sum_{a:x_a=\zeta_j}\mathbb{P}(J_k=a)=n_j/N$.
\item From distribution in (i), $\mathbb{E}X_k=N^{-1}\sum_{a=1}^{N}x_a=\mu$. \\
By finite-population law, $\operatorname{Var}(X_k)=N^{-1}\sum_{a=1}^{N}(x_a-\mu)^2=\sigma^2$.
\end{enumerate}
\hlt{Structural Commentary}: Each sample position is marginally uniform on $U$.\\
\hlt{Key Insight}: Without replacement changes joint dependence, not the one-position marginal distribution.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:srs-unbiased-mean-and-total} \hlt{Unbiased SRS Estimators}\\
Let $U\defeq\{1,\ldots,N\}$, $1\leq n\leq N$, fix $x_a\in\R$ for $a\in U$, and set $\mu\defeq N^{-1}\sum_{a=1}^{N}x_a$ and $\tau\defeq\sum_{a=1}^{N}x_a=N\mu$.  Let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$. For a random ordering $(J_1,\ldots,J_n)$ of $S$, set $X_k\defeq x_{J_k}$, $\overline{X}\defeq n^{-1}\sum_{k=1}^{n}X_k$, and $T\defeq N\overline{X}$.  Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Mean estimator}: the sample mean is design-unbiased for the finite-population mean, $\mathbb{E}\overline{X}=\mu$.
\item \hlt{Total estimator}: the expansion estimator is design-unbiased for the finite-population total, $\mathbb{E}T=\tau$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For each $k$, $\mathbb{P}(J_k=a)=1/N$ for every $a\in U$, so $\mathbb{E}X_k=N^{-1}\sum_{a=1}^{N}x_a=\mu$.\\
Linearity gives $\mathbb{E}\overline{X}=n^{-1}\sum_{k=1}^{n}\mathbb{E}X_k=\mu$.
\item Since $T=N\overline{X}$, $\mathbb{E}T=N\mathbb{E}\overline{X}=N\mu=\tau$.
\end{enumerate}
\hlt{Structural Commentary}: $T=N\overline{X}$ rescales the mean estimate.\\
\hlt{Key Insight}: Equal position marginals give unbiasedness; independence is unnecessary.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:srs-without-replacement-covariance} \hlt{Negative Dependence Without Replacement}\\
Let $U\defeq\{1,\ldots,N\}$, $2\leq n\leq N$, fix $x_a\in\R$ for $a\in U$, set $\mu\defeq N^{-1}\sum_{a=1}^{N}x_a$, $\tau\defeq\sum_{a=1}^{N}x_a$, and $\sigma^2\defeq N^{-1}\sum_{a=1}^{N}(x_a-\mu)^2$. Let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$. For a random ordering $(J_1,\ldots,J_n)$ of $S$, set $X_k\defeq x_{J_k}$.  If $i,j\in\{1,\ldots,n\}$ and $i\neq j$, then $\operatorname{Cov}(X_i,X_j)=-\sigma^2/(N-1)$.
\begin{proof}
Let $(Y_1,\ldots,Y_N)$ be a uniformly random ordering of $(x_1,\ldots,x_N)$.  Each $Y_a$ has variance $\sigma^2$, and symmetry gives $\operatorname{Cov}(Y_a,Y_b)=\gamma$ for $a\neq b$. Since $\sum_{a=1}^{N}Y_a=\tau$ is fixed, $0=\operatorname{Var}(\sum_{a=1}^{N}Y_a)=N\sigma^2+N(N-1)\gamma$.
Thus $\gamma=-\sigma^2/(N-1)$, and $(X_i,X_j)$ has the same law as any two distinct positions in the full random ordering.
\hlt{Structural Commentary}: A fixed total forces negative pairwise covariance.\\
\hlt{Key Insight}: Without replacement, one large draw leaves less for the rest.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:srs-sample-mean-variance} \hlt{Variance and Finite Population Correction}\\
Let $U\defeq\{1,\ldots,N\}$, $N>1$, $1\leq n\leq N$, fix $x_a\in\R$ for $a\in U$.\\
Set $\mu\defeq N^{-1}\sum_{a=1}^{N}x_a$ and $\sigma^2\defeq N^{-1}\sum_{a=1}^{N}(x_a-\mu)^2$. Let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$.\\
For a random ordering $(J_1,\ldots,J_n)$ of $S$, set $X_k\defeq x_{J_k}$, $\overline{X}\defeq n^{-1}\sum_{k=1}^{n}X_k$, and $T\defeq N\overline{X}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Mean variance}: the sample mean has design variance $\operatorname{Var}(\overline{X})=\frac{\sigma^2}{n}\left(\frac{N-n}{N-1}\right)=\frac{\sigma^2}{n}\left(1-\frac{n-1}{N-1}\right)$.
\item \hlt{Total variance}: the total estimator has design variance $\operatorname{Var}(T)=N^2\operatorname{Var}(\overline{X})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Mean variance}: Each $X_k$ has variance $\sigma^2$.  Let $(Y_1,\ldots,Y_N)$ be a uniformly random ordering of $(x_1,\ldots,x_N)$, set $\tau\defeq\sum_{a=1}^{N}x_a$, and let $\gamma$ be the common covariance of $Y_a,Y_b$ for $a\neq b$.\\
Since $\sum_{a=1}^{N}Y_a=\tau$, $0=\operatorname{Var}(\sum_{a=1}^{N}Y_a)=N\sigma^2+N(N-1)\gamma$, so $\gamma=-\sigma^2/(N-1)$.\\
Therefore $\operatorname{Var}(\overline{X}) =\frac1{n^2}\left(n\sigma^2+n(n-1)\left[-\frac{\sigma^2}{N-1}\right]\right) =\frac{\sigma^2}{n}\left(1-\frac{n-1}{N-1}\right)$.
\item \hlt{Total variance}: The total estimator satisfies $T=N\overline{X}$, so $\operatorname{Var}(T)=\operatorname{Var}(N\overline{X})=N^2\operatorname{Var}(\overline{X})$.
\end{enumerate}
\hlt{Structural Commentary}: The factor $(N-n)/(N-1)$ is the \hlt{finite population correction}.\\
\hlt{Key Insight}: Averaging gives $1/n$ scaling; without replacement subtracts negative covariance.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:srs-estimated-standard-errors} \hlt{Estimated Standard Errors}\\
Let $U\defeq\{1,\ldots,N\}$, $1<n\leq N$, fix $x_a\in\R$ for $a\in U$, and let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$.  For a random ordering $(J_1,\ldots,J_n)$ of $S$, set $X_k\defeq x_{J_k}$, $\overline{X}\defeq n^{-1}\sum_{k=1}^{n}X_k$, $T\defeq N\overline{X}$, and $s^2\defeq (n-1)^{-1}\sum_{k=1}^{n}(X_k-\overline{X})^2$.  Then practical plug-in estimated variances are:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Mean}: the plug-in estimated variance of the sample mean is $s_{\overline{X}}^2=\frac{s^2}{n}\left(1-\frac{n}{N}\right)$.
\item \hlt{Total}: the plug-in estimated variance of the total estimator is $s_T^2=N^2s_{\overline{X}}^2$.
\item \hlt{Proportion}: if $x_a\in\{0,1\}$ $\forall a\in U$ and $\widehat p\defeq\overline{X}$, then plug-in estimated variance of $\widehat p$ is $s_{\widehat p}^2=\frac{\widehat p(1-\widehat p)}{n-1}\left(1-\frac{n}{N}\right)$.
\end{enumerate}
\begin{proof}
Set $c_N\defeq1-n/N$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Mean}: Replace unknown finite-population variance by $s^2$, correction by $c_N$ gives $s_{\overline{X}}^2=(s^2/n)c_N=\frac{s^2}{n}(1-\frac{n}{N})$.
\item \hlt{Total}: Since $T=N\overline{X}$, variance scales quadratically, so $s_T^2=N^2s_{\overline{X}}^2$.
\item \hlt{Proportion}: If $X_k\in\{0,1\}$ and $\widehat p=\overline{X}$,\\
then $s^2=(n-1)^{-1}\sum_{k=1}^{n}(X_k-\widehat p)^2=(n-1)^{-1}(n\widehat p-n\widehat p^2)=n\widehat p(1-\widehat p)/(n-1)$;\\
substituting this into (i) gives $s_{\widehat p}^2=\widehat p(1-\widehat p)c_N/(n-1)=\frac{\widehat p(1-\widehat p)}{n-1}(1-\frac{n}{N})$.
\end{enumerate}
\hlt{Structural Commentary}: Plug-in SEs use $s^2$ and the $1-n/N$ correction.\\
\hlt{Key Insight}: Binary-data variance depends only on $\widehat p$.
\end{proof}
\end{corollary}

\begin{example} \label{ex:srs-finite-population-correction-hospitals} \hlt{Finite Population Correction in a Hospital Survey}\\
For the hospital population in the source chapter, $N=393$ and the population
standard deviation of discharges is about $589.7$.  If $n=32$ hospitals are
sampled without replacement, then
\[
\sigma_{\overline{X}}
=\frac{589.7}{\sqrt{32}}\sqrt{1-\frac{31}{392}}
\approx 100.0.
\]
The correction is small here because $32/393$ is small; the main scale is still
the familiar $1/\sqrt n$ averaging factor.
\end{example}

\begin{method} \label{meth:reservoir-sampling-simple-random-sample} \hlt{Sequential Simple Random Sampling}\\
When the population size is not known in advance but $n\leq N$, a simple random
sample can be generated while reading the population once:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Put the first $n$ population members into the sample.
\item When the $(n+k)$th member is read, place it in the sample with probability
$n/(n+k)$.
\item If it is placed in the sample, remove one of the existing sample members
uniformly at random.
\end{enumerate}
After each step the current sample is uniformly distributed over all subsets of
the current population list with size $n$.
\end{method}
