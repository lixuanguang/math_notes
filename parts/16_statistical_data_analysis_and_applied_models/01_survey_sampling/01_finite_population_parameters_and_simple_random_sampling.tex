\subsection{Finite Population Parameters and Simple Random Sampling}

Survey sampling treats a finite population as fixed and the sample-selection mechanism as random.  The inferential target is often a population mean, total, or proportion, while the random variables come from the sampling design.

\begin{definition} \label{def:finite-population-parameters} \hlt{Finite Population Parameters}\\
Let $U\defeq\{1,\ldots,N\}$ and fix $x_i\in\R$, $i\in U$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Population mean}: $\mu\defeq \frac{1}{N}\sum\limits_{i=1}^{N}x_i$.
\item \hlt{Population total}: $\tau\defeq\sum\limits_{i=1}^{N}x_i=N\mu$.
\item \hlt{Population variance}: $\sigma^2\defeq \frac{1}{N}\sum\limits_{i=1}^{N}(x_i-\mu)^2=\frac{1}{N}\sum\limits_{i=1}^{N}x_i^2-\mu^2$.
\item \hlt{Population proportion}: $\{x_i:i\in U\}\subseteq\{0,1\}\implies p\defeq\mu=\frac{1}{N}\sum\limits_{i=1}^{N}x_i$ and $\sigma^2=p(1-p)$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:simple-random-sampling-without-replacement} \hlt{Simple Random Sampling}\\
Let $U\defeq\{1,\ldots,N\}$, $1\leq n\leq N$, and fix $x_i\in\R$, $i\in U$.\\
Let $S$ be a random $n$-element subset of $U$; for a random ordering $J_1,\ldots,J_n$ of $S$, set $X_k\defeq x_{J_k}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Simple random sample}: $\mathbb{P}(S=s)=\frac{1}{\binom{N}{n}}$, $\forall s\subseteq U$ with $|s|=n$.
\item \hlt{Sample mean}: $\overline{X}\defeq \frac{1}{n}\sum\limits_{i=1}^{n}X_i$.
\item \hlt{Total estimator}: $T\defeq N\overline{X}$.
\item \hlt{Sample proportion}: $\{x_i:i\in U\}\subseteq\{0,1\}\implies \widehat p\defeq\overline{X}$.
\end{enumerate}
\end{definition}

\begin{lemma} \label{lem:srs-one-draw-law} \hlt{Law of One Sample Position}\\
Let $U\defeq\{1,\ldots,N\}$, $1\leq n\leq N$, fix $x_a\in\R$ for $a\in U$.\\
Let $\mu\defeq \frac{1}{N}\sum\limits_{a=1}^{N}x_a$, $\sigma^2\defeq \frac{1}{N}\sum\limits_{a=1}^{N}(x_a-\mu)^2$, $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$.\\
If $(J_1,\ldots,J_n)$ is a random ordering of $S$, set $X_k\defeq x_{J_k}$.\\
Let $\zeta_1,\ldots,\zeta_m$ be distinct values in $\{x_a:a\in U\}$, let $n_j\defeq\#\{a\in U:x_a=\zeta_j\}$. For each $k\in\{1,\ldots,n\}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Distribution}: the selected value $X_k$ has the finite-population law $\mathbb{P}(X_k=\zeta_j)=\frac{n_j}{N}$ for $j=1,\ldots,m$.
\item \hlt{Moments}: the finite-population law has mean $\mathbb{E}X_k=\mu$ and variance $\operatorname{Var}(X_k)=\sigma^2$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For every $a\in U$, $\mathbb{P}(J_k=a)=\frac{1}{N}$, so $\mathbb{P}(X_k=\zeta_j)=\sum\limits_{a:x_a=\zeta_j}\mathbb{P}(J_k=a)=\frac{n_j}{N}$.
\item From (i), $\mathbb{E}X_k=\frac{1}{N}\sum\limits_{a=1}^{N}x_a=\mu$. By finite-population law, $\operatorname{Var}(X_k)=\frac{1}{N}\sum\limits_{a=1}^{N}(x_a-\mu)^2=\sigma^2$.
\end{enumerate}
\hlt{Structural Commentary}: Each sample position is marginally uniform on $U$.\\
\hlt{Key Insight}: Without replacement changes joint dependence, not the one-position marginal distribution.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:srs-unbiased-mean-and-total} \hlt{Unbiased SRS Estimators}\\
Let $U\defeq\{1,\ldots,N\}$, $1\leq n\leq N$, fix $x_a\in\R$ for $a\in U$.\\
Set $\mu\defeq \frac{1}{N}\sum\limits_{a=1}^{N}x_a$, $\tau\defeq\sum\limits_{a=1}^{N}x_a=N\mu$. Let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$.\\
Given random ordering $(J_1,\ldots,J_n)$ of $S$, set $X_k\defeq x_{J_k}$, $\overline{X}\defeq \frac{1}{n}\sum\limits_{k=1}^{n}X_k$, and $T\defeq N\overline{X}$.  Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Mean estimator}: the sample mean is design-unbiased for the finite-population mean, $\mathbb{E}\overline{X}=\mu$.
\item \hlt{Total estimator}: the expansion estimator is design-unbiased for the finite-population total, $\mathbb{E}T=\tau$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For each $k$, $\mathbb{P}(J_k=a)=\frac{1}{N}$ $\forall a\in U$, so $\mathbb{E}X_k=\frac{1}{N}\sum\limits_{a=1}^{N}x_a=\mu$. By linearity $\mathbb{E}\overline{X}=\frac{1}{n}\sum\limits_{k=1}^{n}\mathbb{E}X_k=\mu$.
\item Since $T=N\overline{X}$, $\mathbb{E}T=N\mathbb{E}\overline{X}=N\mu=\tau$.
\end{enumerate}
\hlt{Structural Commentary}: $T=N\overline{X}$ rescales the mean estimate.\\
\hlt{Key Insight}: Equal position marginals give unbiasedness; independence is unnecessary.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:srs-without-replacement-covariance} \hlt{Negative Dependence Without Replacement}\\
Let $U\defeq\{1,\ldots,N\}$, $2\leq n\leq N$, fix $x_a\in\R$ for $a\in U$.\\
Set $\mu\defeq \frac{1}{N}\sum\limits_{a=1}^{N}x_a$, $\tau\defeq\sum\limits_{a=1}^{N}x_a$, $\sigma^2\defeq \frac{1}{N}\sum\limits_{a=1}^{N}(x_a-\mu)^2$. Let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$.\\
For random ordering $(J_1,\ldots,J_n)$ of $S$, set $X_k\defeq x_{J_k}$. If $i,j\in\{1,\ldots,n\}$ and $i\neq j$, then $\operatorname{Cov}(X_i,X_j)=-\frac{\sigma^2}{N-1}$.
\begin{proof}
Let $(Y_1,\ldots,Y_N)$ be a uniformly random ordering of $(x_1,\ldots,x_N)$.\\
Each $Y_a$ has variance $\sigma^2$, and symmetry gives $\operatorname{Cov}(Y_a,Y_b)=\gamma$ for $a\neq b$.\\
Since $\sum\limits_{a=1}^{N}Y_a=\tau$ is fixed, $0=\operatorname{Var}(\sum\limits_{a=1}^{N}Y_a)=N\sigma^2+N(N-1)\gamma$.\\
Thus $\gamma=-\frac{\sigma^2}{N-1}$, and $(X_i,X_j)$ has same law as any two distinct positions in full random ordering.\\
\hlt{Structural Commentary}: A fixed total forces negative pairwise covariance.\\
\hlt{Key Insight}: Without replacement, one large draw leaves less for the rest.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:srs-sample-mean-variance} \hlt{Variance and Finite Population Correction}\\
Let $U\defeq\{1,\ldots,N\}$, $N>1$, $1\leq n\leq N$, fix $x_a\in\R$ for $a\in U$.\\
Set $\mu\defeq \frac{1}{N}\sum\limits_{a=1}^{N}x_a$ and $\sigma^2\defeq \frac{1}{N}\sum\limits_{a=1}^{N}(x_a-\mu)^2$. Let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$.\\
For a random ordering $(J_1,\ldots,J_n)$ of $S$, set $X_k\defeq x_{J_k}$, $\overline{X}\defeq \frac{1}{n}\sum\limits_{k=1}^{n}X_k$, and $T\defeq N\overline{X}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Mean variance}: the sample mean has design variance $\operatorname{Var}(\overline{X})=\frac{\sigma^2}{n}\left(\frac{N-n}{N-1}\right)=\frac{\sigma^2}{n}\left(1-\frac{n-1}{N-1}\right)$.
\item \hlt{Total variance}: the total estimator has design variance $\operatorname{Var}(T)=N^2\operatorname{Var}(\overline{X})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Each $X_k$ has variance $\sigma^2$.  Let $(Y_1,\ldots,Y_N)$ be a uniformly random ordering of $(x_1,\ldots,x_N)$, set $\tau\defeq\sum\limits_{a=1}^{N}x_a$, and let $\gamma$ be the common covariance of $Y_a,Y_b$ for $a\neq b$.\\
Since $\sum\limits_{a=1}^{N}Y_a=\tau$, $0=\operatorname{Var}(\sum\limits_{a=1}^{N}Y_a)=N\sigma^2+N(N-1)\gamma$, so $\gamma=-\frac{\sigma^2}{N-1}$.\\
Therefore $\operatorname{Var}(\overline{X}) =\frac1{n^2}\left(n\sigma^2+n(n-1)\left[-\frac{\sigma^2}{N-1}\right]\right) =\frac{\sigma^2}{n}\left(1-\frac{n-1}{N-1}\right)$.
\item The total estimator satisfies $T=N\overline{X}$, so $\operatorname{Var}(T)=\operatorname{Var}(N\overline{X})=N^2\operatorname{Var}(\overline{X})$.
\end{enumerate}
\hlt{Structural Commentary}: The factor $\frac{N-n}{N-1}$ is the \hlt{finite population correction}.\\
\hlt{Key Insight}: Averaging gives $\frac{1}{n}$ scaling; without replacement subtracts negative covariance.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:srs-estimated-standard-errors} \hlt{Estimated Standard Errors}\\
Let $U\defeq\{1,\ldots,N\}$, $1<n\leq N$, fix $x_a\in\R$ for $a\in U$, and let $S\sim\operatorname{Unif}\{s\subseteq U:|s|=n\}$. For a random ordering $(J_1,\ldots,J_n)$ of $S$, set $X_k\defeq x_{J_k}$, $\overline{X}\defeq \frac{1}{n}\sum\limits_{k=1}^{n}X_k$, $T\defeq N\overline{X}$, and $s^2\defeq \frac{1}{n-1}\sum\limits_{k=1}^{n}(X_k-\overline{X})^2$.\\
Then practical plug-in estimated variances are:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Mean}: the plug-in estimated variance of the sample mean is $s_{\overline{X}}^2=\frac{s^2}{n}\left(1-\frac{n}{N}\right)$.
\item \hlt{Total}: the plug-in estimated variance of the total estimator is $s_T^2=N^2s_{\overline{X}}^2$.
\item \hlt{Proportion}: if $x_a\in\{0,1\}$ $\forall a\in U$ and $\widehat p\defeq\overline{X}$, then plug-in estimated variance of $\widehat p$ is $s_{\widehat p}^2=\frac{\widehat p(1-\widehat p)}{n-1}\left(1-\frac{n}{N}\right)$.
\end{enumerate}
\begin{proof}
Set $c_N\defeq1-\frac{n}{N}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Replace unknown finite-population variance by $s^2$, correction by $c_N$ gives $s_{\overline{X}}^2=(\frac{s^2}{n})c_N=\frac{s^2}{n}(1-\frac{n}{N})$.
\item Since $T=N\overline{X}$, variance scales quadratically, so $s_T^2=N^2s_{\overline{X}}^2$.
\item If $X_k\in\{0,1\}$ and $\widehat p=\overline{X}$, then $s^2=\frac{1}{n-1}\sum\limits_{k=1}^{n}(X_k-\widehat p)^2=\frac{1}{n-1}(n\widehat p-n\widehat p^2)=\frac{n\widehat p(1-\widehat p)}{n-1}$.\\
Substituting this into (i) gives $s_{\widehat p}^2=\frac{\widehat p(1-\widehat p)c_N}{n-1}=\frac{\widehat p(1-\widehat p)}{n-1}(1-\frac{n}{N})$.
\end{enumerate}
\hlt{Structural Commentary}: Plug-in SEs use $s^2$ and the $1-\frac{n}{N}$ correction.\\
\hlt{Key Insight}: Binary-data variance depends only on $\widehat p$.
\end{proof}
\end{corollary}

\begin{example} \label{ex:srs-finite-population-correction-hospitals} \hlt{Finite Population Correction in a Hospital Survey}\\
For a finite population of $N=393$ hospitals, let $x_i$ denote discharges and suppose the finite-population standard deviation is $\sigma\approx589.7$.
Under simple random sampling without replacement with $n=32$, the sample-mean standard deviation is $\sigma_{\overline{X}}=\frac{\sigma}{\sqrt{n}}\sqrt{1-\frac{n-1}{N-1}}\approx\frac{589.7}{\sqrt{32}}\sqrt{1-\frac{31}{392}}\approx100.0$.
Since $\frac{n}{N}=\frac{32}{393}$ is small, the finite-population correction is near $1$, and the standard error is mainly governed by the $\frac{1}{\sqrt n}$ averaging scale.
\end{example}

\begin{method} \label{meth:reservoir-sampling-simple-random-sample} \hlt{Sequential Simple Random Sampling}\\
Fix target size $n\geq1$, assume terminal stream length is $N\geq n$, $N$ unknown while streaming.\\
Population units arrive sequentially as $1,\ldots,N$. Let $S_t\subseteq\{1,\ldots,t\}$ denote current reservoir after $t$ arrivals.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Initialize}: for $1\leq t\leq n$, set $S_t=\{1,\ldots,t\}$.
\item \hlt{Admit}: for $t>n$, draw $B_t\sim\operatorname{Bernoulli}(\frac{n}{t})$.
\item \hlt{Update}: if $B_t=0$, set $S_t=S_{t-1}$; if $B_t=1$, draw $J_t\mid S_{t-1}\sim\operatorname{Unif}(S_{t-1})$ and set $S_t=(S_{t-1}\setminus\{J_t\})\cup\{t\}$.
\item \hlt{Output}: after the stream ends, return $S_N$.
\end{enumerate}
Then for every $t\geq n$, $S_t\sim\operatorname{Unif}\{s\subseteq\{1,\ldots,t\}:|s|=n\}$.\\
Hence $S_N$ is a simple random sample without replacement.
\end{method}
