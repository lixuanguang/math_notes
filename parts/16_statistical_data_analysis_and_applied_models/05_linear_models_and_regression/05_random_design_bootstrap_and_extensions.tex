\subsection{Random Design, Bootstrap, and Extensions}

The fixed-design model conditions on the observed predictor values.  Many observational studies are more naturally viewed as sampling random predictor-response pairs, with the fixed-design analysis recovered after conditioning on the realised design.

\begin{definition} \label{def:random-design-linear-regression-model} \hlt{Random-Design Linear Regression}\\
Let $[n]\defeq\{1,\ldots,n\}$, $(Y_i,\xi_i)_{i\in[n]}$ be independent copies of $(Y,\xi)$, $\xi\in\R^p$, $\beta\in\R^p$, and $\sigma^2>0$.\\
A \hlt{Random-Design Linear Regression} is an iid sample $(Y_i,\xi_i)_{i\in[n]}$ satisfying $\mathbb{E}(Y\mid \xi=x)=x^t\beta$\\
and $\operatorname{Var}(Y\mid \xi=x)=\sigma^2$ for $x\in\operatorname{supp}(\xi)$.\\
\hlt{Structural Commentary}: Conditioning on $X_{i\cdot}=\xi_i^t$ gives $Y_{1:n}=X\beta+e$ with $e_i=Y_i-\xi_i^t\beta$.
\end{definition}

\begin{proposition} \label{prop:fixed-design-intervals-unconditional-coverage} \hlt{Conditional Coverage Implies Unconditional Coverage}\\
Let $C(X,Y)$ be a confidence set for $\beta_j$, $\alpha\in(0,1)$, and $F\defeq\{\operatorname{rank}(X)=p\}$.\\
If $\mathbb{P}(F)=1$ and $\mathbb{P}(\beta_j\in C(X,Y)\mid X)=1-\alpha$ on $F$, then $\mathbb{P}(\beta_j\in C(X,Y))=1-\alpha$.
\begin{proof}
Set $A\defeq\{\beta_j\in C(X,Y)\}$. Since $\mathbb{P}(F)=1$, $\mathbb{P}(A)=\mathbb{E}[\mathbb{P}(A\mid X)]=\mathbb{E}[(1-\alpha)\mathbf{1}_F+\mathbb{P}(A\mid X)\mathbf{1}_{F^c}]=1-\alpha$.
\hlt{Structural Commentary}: $\mathbb{P}(F)=1$ and $\mathbb{P}(A\mid X)=1-\alpha$ on $F$.\\
\hlt{Key Insight}: $\mathbb{P}(A\mid X)=1-\alpha$ a.s. implies $\mathbb{P}(A)=1-\alpha$.
\end{proof}
\end{proposition}

\begin{remark} \label{rem:random-design-variance-differs} \hlt{Unconditional Variability Can Differ}\\
Under random design, if $\mathbb{E}(\widehat{\beta}\mid X)=\beta$, then $\mathbb{E}\widehat{\beta}=\beta$.\\
Moreover, $\operatorname{Var}(\widehat{\beta})=\mathbb{E}[\operatorname{Var}(\widehat{\beta}\mid X)]+\operatorname{Var}(\mathbb{E}[\widehat{\beta}\mid X])=\sigma^2\mathbb{E}[(X^tX)^{-1}]$, not generally $\sigma^2(X^tX)^{-1}$ for frozen design.
\end{remark}

\begin{method} \label{meth:regression-pairs-bootstrap-cross-reference} \hlt{Pairs Bootstrap for Random-Design Regression}\\
Let $D\defeq\{(Y_i,x_i):i\in[n]\}$, $x_i\in\R^p$, and let $T(D)$ be a fitted coefficient, prediction, or smooth statistic.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Draw $D_b^*\defeq\{(Y_{i_m},x_{i_m}):m\in[n]\}$ with $i_1,\ldots,i_n$ sampled iid from $[n]$.
\item Fit least squares on $D_b^*$ and compute $T_b^*\defeq T(D_b^*)$.
\item Repeat for $b\in[B]$ and estimate $\operatorname{se}(T(D))$ by $\widehat{\operatorname{se}}_*(T)\defeq\left(\frac{1}{B-1}\sum\limits_{b=1}^{B}(T_b^*-\overline{T}^*)^2\right)^{1/2}$, where $\overline{T}^*\defeq B^{-1}\sum\limits_{b=1}^{B}T_b^*$.
\end{enumerate}
\end{method}

\begin{perspective} \label{per:least-squares-extensions-source-chapter} \hlt{Extensions Beyond Ordinary Least Squares}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Robust regression}: Replace $\sum\limits_i\widehat{e}_i^2$ by $\sum\limits_i\Psi(\widehat{e}_i)$; $\Psi(u)=u^2$ gives least squares.
\item \hlt{Variable selection}: For selected predictors $S\subseteq\{1,\ldots,p\}$, $\widehat{\beta}_S$ and $\operatorname{se}(\widehat{\beta}_S)$ are conditional on $S$.
\item \hlt{Influence}: Large leverage $h_{ii}=x_i^t(X^tX)^{-1}x_i$ can make observation $i$ dominate $\widehat{\beta}$.
\item \hlt{Inverse calibration}: Given a fitted response curve $\widehat{m}$ and measured response $y_0$,\\
estimating $x_0$ from $\widehat{m}(x_0)=y_0$ is an inverse problem with uncertainty beyond $\operatorname{se}(\widehat{\beta})$.
\end{enumerate}
\end{perspective}
