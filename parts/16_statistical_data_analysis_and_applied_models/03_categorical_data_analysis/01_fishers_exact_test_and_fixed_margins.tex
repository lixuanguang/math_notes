\subsection{Fisher's Exact Test and Fixed Margins}
\label{subsec:fishers-exact-test-and-fixed-margins}
% Source integration: attached Chapter 13 screenshots, Sections 13.1--13.2.

Categorical data record counts in classes rather than measurements on a continuous scale.  In a two-way table, the rows and columns encode two classifications, and the cells record how many units fall in each cross-classification.  The probability model is therefore built from multinomial and hypergeometric count laws rather than from normal sampling pivots.

\begin{definition} \label{def:two-way-contingency-table-counts} \hlt{Two-Way Contingency Table}\\
Let $I,J\in\N$ and $n_{ij}\in\N_0$ for $1\leq i\leq I$, $1\leq j\leq J$. Set $n_{i\cdot}\defeq\sum\limits_{j=1}^{J}n_{ij}$, $n_{\cdot j}\defeq\sum\limits_{i=1}^{I}n_{ij}$, and $n_{\cdot\cdot}\defeq\sum\limits_{i=1}^{I}\sum\limits_{j=1}^{J}n_{ij}$.\\
For corresponding random counts, write $N_{ij}$ and use capitalized totals.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Two-way contingency table}: $(n_{ij})_{1\leq i\leq I,\ 1\leq j\leq J}$, indexed by row and column classifications.
\item \hlt{Margins}: $\{n_{i\cdot}:1\leq i\leq I\}$, $\{n_{\cdot j}:1\leq j\leq J\}$, and $n_{\cdot\cdot}$.
\end{enumerate}
\hlt{Structural Commentary}: Fixed-margin exact tests condition on the margins of random table $N=(N_{ij})$.
\end{definition}

\begin{method} \label{meth:fishers-exact-test-fixed-margins} \hlt{Fisher's Exact Test for a $2\times2$ Table}\\
Let a $2\times2$ table have fixed margins
\[
\begin{array}{c|cc|c}
&1&2&\text{Total}\\ \hline
1&N_{11}&N_{12}&n_{1\cdot}\\
2&N_{21}&N_{22}&n_{2\cdot}\\ \hline
\text{Total}&n_{\cdot1}&n_{\cdot2}&n_{\cdot\cdot}
\end{array}
\]
Let $R_1,R_2$ be fixed disjoint row sets with $|R_i|=n_{i\cdot}$.\\
Let $C$ be the random column-$1$ set with $|C|=n_{\cdot1}$, and set $\Omega_0\defeq\{A\subseteq R_1\cup R_2:|A|=n_{\cdot1}\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $H_0:\mathbb{P}_0(C=A)=\binom{n_{\cdot\cdot}}{n_{\cdot1}}^{-1}$ $\forall A\in\Omega_0$.
\item $N_{11}=|C\cap R_1|$, $N_{12}=n_{1\cdot}-N_{11}$, $N_{21}=n_{\cdot1}-N_{11}$, and $N_{22}=n_{2\cdot}-n_{\cdot1}+N_{11}$.
\item $\mathcal{S}\defeq\{x\in\Z:0\leq x\leq n_{1\cdot},\ 0\leq n_{\cdot1}-x\leq n_{2\cdot}\}$ and $p_0(x)\defeq\mathbb{P}_{0}(N_{11}=x)=\frac{\binom{n_{1\cdot}}{x}\binom{n_{2\cdot}}{n_{\cdot1}-x}}{\binom{n_{\cdot\cdot}}{n_{\cdot1}}}$, $x\in\mathcal{S}$.
\item $p_+(x_{\mathrm{obs}})\defeq\sum\limits_{x\in\mathcal{S}:x\geq x_{\mathrm{obs}}}p_0(x)$, $p_-(x_{\mathrm{obs}})\defeq\sum\limits_{x\in\mathcal{S}:x\leq x_{\mathrm{obs}}}p_0(x)$, and $p_2(x_{\mathrm{obs}})\defeq\sum\limits_{x\in\mathcal{S}:p_0(x)\leq p_0(x_{\mathrm{obs}})}p_0(x)$.\\
Reject when the selected value is $\leq\alpha$.
\end{enumerate}
\end{method}

\begin{example} \label{ex:promotion-fishers-exact-test} \hlt{Promotion Decisions and Fixed Margins}\\
In randomised label experiment with $24$ male-labeled,$24$ female-labeled files, the observed table is
\[
\begin{array}{c|cc|c}
&\text{Male}&\text{Female}&\text{Total}\\ \hline
\text{Promote}&21&14&35\\
\text{Hold file}&3&10&13\\ \hline
\text{Total}&24&24&48
\end{array}
\]
Under $H_0$ and fixed margins $(35,13)$, $(24,24)$, $N_{11}$ satisfies $\mathbb{P}_{0}(N_{11}=x)=\frac{\binom{35}{x}\binom{13}{24-x}}{\binom{48}{24}}$, $x\in\mathcal{S}\defeq\{11,\ldots,24\}$.\\
At level $0.05$, $R_{0.05}^{(2)}\defeq\{11,12,13,14,21,22,23,24\}$; since $x_{\mathrm{obs}}=21\in R_{0.05}^{(2)}$, reject $H_0$.
\end{example}

\begin{remark} \label{rem:fisher-exact-conditional-design-interpretation} \hlt{What Is Exact Here}\\
For $N=(N_{ij})$ and $\mathcal{M}\defeq(\{N_{i\cdot}\}_i,\{N_{\cdot j}\}_j)$, exactness means evaluating $\mathcal{L}_0(N\mid\mathcal{M})$ without asymptotics. Conditioning is design-justified when $\mathcal{M}$ is ancillary to $H_0$; rejection interpretation is design-dependent.
\end{remark}
