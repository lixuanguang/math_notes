\subsection{Location, Scale, and Robust Summaries}

Numerical summaries compress a batch more aggressively than a plot.  They should therefore be chosen with attention to the feature being summarized and to the influence of extreme observations.

\begin{definition} \label{def:location-summaries-for-data} \hlt{Location Summaries for a Batch}\\
Let $x_1,\ldots,x_n\in\R$ have order statistics $x_{(1)}\leq\cdots\leq x_{(n)}$.\\
When a trim fraction is used, let $0\leq\alpha<\frac{1}{2}$ and $m_\alpha\defeq\floor{n\alpha}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Arithmetic Mean}: $\overline{x}\defeq n^{-1}\sum\limits_{i=1}^n x_i$.
\item \hlt{Median}: $\widetilde{x}\defeq x_{\left(\frac{n+1}{2}\right)}$ if $n$ is odd, and $\widetilde{x}\defeq\frac{x_{\left(\frac{n}{2}\right)}+x_{\left(\frac{n}{2}+1\right)}}{2}$ if $n$ is even.
\item \hlt{$100\alpha\%$ Trimmed Mean}: $\overline{x}_\alpha\defeq(n-2m_\alpha)^{-1}\sum\limits_{i=m_\alpha+1}^{n-m_\alpha}x_{(i)}$.
\end{enumerate}
\hlt{Structural Commentary}: Under $x_i\mapsto x_i+\Delta$, $\overline{x}\mapsto\overline{x}+\frac{\Delta}{n}$; $\widetilde{x}$ and $\overline{x}_\alpha$ depend only on central order statistics.\\
\hlt{Key Insight}: Extreme magnitudes affect $\overline{x}$ directly, but affect $\widetilde{x}$ and $\overline{x}_\alpha$ mainly through rank changes.
\end{definition}

\begin{remark} \label{rem:outliers-and-location-summaries} \hlt{Outliers and Center}\\
Replacing one $x_i$ by $M$ sends $\overline{x}$ to $\overline{x}+\frac{M-x_i}{n}$, hence unbounded as $M\to\pm\infty$.\\
Once an observation is extreme, moving it farther outward leaves $\widetilde{x}$ unchanged and leaves $\overline{x}_\alpha$ unchanged if it remains trimmed. This is descriptive robustness, not an independence or model.
\end{remark}

\begin{definition} \label{def:dispersion-summaries-for-data} \hlt{Dispersion Summaries}\\
Let $x_1,\ldots,x_n\in\R$ have mean $\overline{x}$, median $\widetilde{x}$, and selected sample quartiles $Q_1,Q_3$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Sample Variance}: $s^2\defeq(n-1)^{-1}\sum\limits_{i=1}^n(x_i-\overline{x})^2$.
\item \hlt{Sample Standard Deviation}: $s\defeq\sqrt{s^2}$.
\item \hlt{Interquartile Range}: $\operatorname{IQR}\defeq Q_3-Q_1$.
\item \hlt{Median Absolute Deviation}: $\operatorname{MAD}\defeq\operatorname{median}\{\abs{x_i-\widetilde{x}}:1\leq i\leq n\}$.
\end{enumerate}
\hlt{Structural Commentary}: $s^2$ is residual-based; $\operatorname{IQR}$ and $\operatorname{MAD}$ are order-based.\\
\hlt{Key Insight}: A single extreme value can dominate $s$, but not $\operatorname{IQR}$ or $\operatorname{MAD}$ unless it changes central ranks.
\end{definition}

\begin{remark} \label{rem:normal-scale-calibrations} \hlt{Normal Scale Calibrations}\\
For approximately normal data, $\frac{\operatorname{IQR}}{1.35}$ and $\frac{\operatorname{MAD}}{0.675}$ are common robust scale estimates for the normal standard deviation. The constants calibrate the population normal IQR and median absolute deviation to $\sigma$.
\end{remark}

\begin{remark} \label{rem:compare-multiple-location-summaries} \hlt{Compare More Than One Center}\\
Let $\mathcal{C}\defeq\{\overline{x},\widetilde{x},\overline{x}_\alpha,\widehat{\mu}_\Psi\}$. For symmetric light-tailed data, $\operatorname{range}(\mathcal{C})$ is typically small; a large range signals skewness, heavy tails, outliers, or mixture structure.
\end{remark}
