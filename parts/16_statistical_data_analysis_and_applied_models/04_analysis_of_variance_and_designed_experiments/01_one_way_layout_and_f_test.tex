\subsection{One-Way Layouts and the $F$ Test}
\label{subsec:one-way-layouts-and-f-test}
% Source integration: attached Chapter 12 screenshots, Sections 12.1--12.2.1.

Analysis of variance compares several mean levels by splitting total variation into variation within treatment groups and variation between treatment groups.  The method is a multisample continuation of the equal-variance two-sample procedure in Subsection \ref{subsec:two-sample-normal-inference}, with the design cautions from Principle \ref{prin:randomization-confounding-and-blocking} still in force.

\begin{definition} \label{def:balanced-one-way-layout} \hlt{Balanced One-Way Layout}\\
A \hlt{one-way layout} has one factor with $I$ treatment levels.  In the balanced case, each treatment contributes $J$ independent observations
\[
Y_{ij},\qquad i=1,\ldots,I,\quad j=1,\ldots,J.
\]
The normal fixed-effects model is
\[
Y_{ij}=\mu+\alpha_i+\varepsilon_{ij},
\qquad
\sum_{i=1}^{I}\alpha_i=0,
\]
where the errors are independent $N(0,\sigma^2)$ variables.  The expected response under treatment $i$ is $\mu+\alpha_i$, so the null hypothesis of no treatment effect is
\[
H_0:\alpha_1=\cdots=\alpha_I=0.
\]
\end{definition}

\begin{definition} \label{def:one-way-anova-sums-of-squares} \hlt{One-Way Sums of Squares}\\
Write
\[
\overline{Y}_{i\cdot}\defeq \frac1J\sum_{j=1}^{J}Y_{ij},
\qquad
\overline{Y}_{\cdot\cdot}\defeq \frac{1}{IJ}\sum_{i=1}^{I}\sum_{j=1}^{J}Y_{ij}.
\]
The total, within-treatment, and between-treatment sums of squares are
\[
SS_{\mathrm{tot}}\defeq \sum_{i=1}^{I}\sum_{j=1}^{J}(Y_{ij}-\overline{Y}_{\cdot\cdot})^2,
\]
\[
SS_W\defeq \sum_{i=1}^{I}\sum_{j=1}^{J}(Y_{ij}-\overline{Y}_{i\cdot})^2,
\qquad
SS_B\defeq J\sum_{i=1}^{I}(\overline{Y}_{i\cdot}-\overline{Y}_{\cdot\cdot})^2.
\]
\end{definition}

\begin{proposition} \label{prop:one-way-sums-of-squares-decomposition} \hlt{One-Way Sum-of-Squares Decomposition}\\
In Definition \ref{def:one-way-anova-sums-of-squares},
\[
SS_{\mathrm{tot}}=SS_W+SS_B.
\]
\begin{proof}
Use
\[
Y_{ij}-\overline{Y}_{\cdot\cdot}
=
(Y_{ij}-\overline{Y}_{i\cdot})
+(\overline{Y}_{i\cdot}-\overline{Y}_{\cdot\cdot})
\]
and square.  The cross term vanishes because
\[
\sum_{j=1}^{J}(Y_{ij}-\overline{Y}_{i\cdot})=0
\]
for each treatment $i$.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:mean-centering-expected-square} \hlt{Expected Squared Deviation from a Sample Mean}\\
Let $X_1,\ldots,X_n$ be independent random variables with $\mathbb{E}X_i=\mu_i$ and common variance $\sigma^2$.  If
\[
\overline{X}\defeq \frac1n\sum_{i=1}^{n}X_i,
\qquad
\overline{\mu}\defeq \frac1n\sum_{i=1}^{n}\mu_i,
\]
then
\[
\mathbb{E}(X_i-\overline{X})^2
=
(\mu_i-\overline{\mu})^2+\frac{n-1}{n}\sigma^2.
\]
\begin{proof}
Use $\mathbb{E}U^2=(\mathbb{E}U)^2+\operatorname{Var}(U)$ with $U=X_i-\overline{X}$.  Independence gives
\[
\operatorname{Var}(X_i-\overline{X})
=\sigma^2+\frac{\sigma^2}{n}-2\frac{\sigma^2}{n}
=\frac{n-1}{n}\sigma^2.
\]
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:one-way-anova-expected-mean-squares} \hlt{Expected One-Way Sums of Squares}\\
Under Definition \ref{def:balanced-one-way-layout},
\[
\mathbb{E}(SS_W)=I(J-1)\sigma^2,
\qquad
\mathbb{E}(SS_B)=(I-1)\sigma^2+J\sum_{i=1}^{I}\alpha_i^2.
\]
Thus
\[
S_p^2\defeq \frac{SS_W}{I(J-1)}
\]
is an unbiased estimator of the common error variance $\sigma^2$.
\begin{proof}
For $SS_W$, apply Lemma \ref{lem:mean-centering-expected-square} inside each treatment group, where all observations have the same mean.  For $SS_B$, apply the same lemma to the treatment means $\overline{Y}_{i\cdot}$, whose common error variance is $\sigma^2/J$ and whose expectations are $\mu+\alpha_i$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:one-way-anova-f-test} \hlt{Normal-Theory One-Way $F$ Test}\\
Under Definition \ref{def:balanced-one-way-layout},
\[
\frac{SS_W}{\sigma^2}\sim \chi^2_{I(J-1)}.
\]
Under $H_0:\alpha_1=\cdots=\alpha_I=0$,
\[
\frac{SS_B}{\sigma^2}\sim \chi^2_{I-1},
\qquad
SS_B\perp\!\!\!\perp SS_W.
\]
Therefore the statistic
\[
F=
\frac{SS_B/(I-1)}{SS_W/[I(J-1)]}
\]
has null distribution $F_{I-1,I(J-1)}$.
\begin{proof}
The within-group statement follows by applying Theorem \ref{thm:normal-sample-variance-chi-square} separately in each treatment and summing independent chi-square variables.  Under $H_0$, the centered treatment means form the normal mean-residual part orthogonal to the within-treatment residuals, so the chi-square law and independence follow as in Theorem \ref{thm:normal-mean-residual-independence}.  The final display is the defining ratio in Definition \ref{def:f-distribution}.
\end{proof}
\end{theorem}

\begin{method} \label{meth:one-way-anova-unequal-sample-sizes} \hlt{Unequal Sample Sizes}\\
If treatment $i$ has $J_i$ observations and $N=\sum_i J_i$, use
\[
\overline{Y}_{i\cdot}\defeq \frac1{J_i}\sum_{j=1}^{J_i}Y_{ij},
\qquad
\overline{Y}_{\cdot\cdot}\defeq \frac1N\sum_{i=1}^{I}\sum_{j=1}^{J_i}Y_{ij},
\]
\[
SS_W=\sum_{i=1}^{I}\sum_{j=1}^{J_i}(Y_{ij}-\overline{Y}_{i\cdot})^2,
\qquad
SS_B=\sum_{i=1}^{I}J_i(\overline{Y}_{i\cdot}-\overline{Y}_{\cdot\cdot})^2.
\]
The same algebra gives $SS_{\mathrm{tot}}=SS_W+SS_B$, with degrees of freedom $N-I$ and $I-1$.  Under the usual normality, independence, and common-variance assumptions, the null statistic
\[
F=\frac{SS_B/(I-1)}{SS_W/(N-I)}
\]
is calibrated by $F_{I-1,N-I}$.
\end{method}

\begin{example} \label{ex:tablet-laboratories-one-way-anova} \hlt{Seven Laboratories Measuring Tablets}\\
The source compares $I=7$ laboratories, each with $J=10$ determinations of chlorpheniramine maleate in tablets.  Parallel boxplots, as in Method \ref{meth:boxplot-construction}, suggested differences in both center and spread.  The reported analysis of variance table was
\[
\begin{array}{c|rrrr}
\text{Source} & df & SS & MS & F\\ \hline
\text{Labs} & 6 & .125 & .021 & 5.66\\
\text{Error} & 63 & .231 & .0037 & \\
\text{Total} & 69 & .356 & &
\end{array}
\]
The $F$ statistic has $6$ and $63$ degrees of freedom; the source reports a tail probability below $.01$.  The conclusion is evidence of systematic differences among laboratory mean levels, not merely random within-laboratory variation.  The accompanying normal probability plot of residuals showed some lower-tail deviation but not a gross failure of normality.
\end{example}

\begin{remark} \label{rem:one-way-anova-assumptions} \hlt{Assumptions and Robustness}\\
The one-way $F$ test assumes independent errors, normal error shape, and common variance.  Normality is mainly a small-sample calibration issue; equal sample sizes make the $F$ test less sensitive to moderate variance differences.  Independence is more structural: if measurements are clustered, paired, blocked, or repeated on the same unit, the one-way independent-sample model is not the right owner of the dependence.
\end{remark}
