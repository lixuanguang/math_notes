\subsection{Rank-Based One-Way ANOVA}
\label{subsec:rank-based-one-way-anova}
% Source integration: attached Chapter 12 screenshots, Section 12.2.3 and selected problems.

When normality or outlier sensitivity is a concern, a one-way comparison can be made after replacing observations by pooled ranks.  This continues the rank-test philosophy from Subsection \ref{subsec:rank-based-comparative-tests}.

\begin{definition} \label{def:kruskal-wallis-ranks} \hlt{Kruskal--Wallis Ranks}\\
In an unequal-size one-way layout with $J_i$ observations in treatment $i$ and $N=\sum_iJ_i$, rank all observations together, assigning average ranks to ties.  Let
\[
R_{ij}=\text{rank of }Y_{ij}\text{ in the pooled sample},
\qquad
\overline{R}_{i\cdot}\defeq \frac1{J_i}\sum_{j=1}^{J_i}R_{ij}.
\]
The grand average rank is
\[
\overline{R}_{\cdot\cdot}=\frac{N+1}{2}.
\]
\end{definition}

\begin{method} \label{meth:kruskal-wallis-test} \hlt{Kruskal--Wallis Test}\\
Define the rank between-group sum of squares
\[
SS_B^{(R)}\defeq \sum_{i=1}^{I}J_i(\overline{R}_{i\cdot}-\overline{R}_{\cdot\cdot})^2.
\]
The Kruskal--Wallis statistic is
\[
K=
\frac{12}{N(N+1)}SS_B^{(R)}
=
\frac{12}{N(N+1)}
\left(\sum_{i=1}^{I}J_i\overline{R}_{i\cdot}^{\,2}\right)-3(N+1).
\]
The exact null law is obtained by enumerating rank assignments.  For ordinary moderate sample sizes, the null law is well approximated by $\chi^2_{I-1}$.
\end{method}

\begin{example} \label{ex:tablet-laboratories-kruskal-wallis} \hlt{Tablet Laboratories by Ranks}\\
For the tablet laboratory data of Example \ref{ex:tablet-laboratories-one-way-anova}, the source reports
\[
K=29.51.
\]
Using the $\chi^2_6$ approximation, the tail probability is below $.005$.  Thus the rank analysis, like the normal-theory $F$ test, gives strong evidence that the laboratories do not all have the same response distribution.
\end{example}

\begin{corollary} \label{cor:kruskal-wallis-two-groups-mann-whitney} \hlt{Two Groups Recover Mann--Whitney}\\
When $I=2$, the Kruskal--Wallis statistic is a monotone function of the rank sum for either group.  Therefore the two-group Kruskal--Wallis test is equivalent to the Mann--Whitney rank-sum test of Method \ref{meth:mann-whitney-test}.
\begin{proof}
With two groups, the two rank sums determine each other because their total is $N(N+1)/2$.  The statistic $SS_B^{(R)}$ therefore depends on only one rank sum, and it is largest exactly when that rank sum is farthest from its null center.
\end{proof}
\end{corollary}

\begin{method} \label{meth:rank-based-multiple-comparisons} \hlt{Rank-Based Follow-Up Comparisons}\\
After a Kruskal--Wallis rejection, pairwise Mann--Whitney tests may be used as follow-up comparisons.  To protect the family of decisions, apply the Bonferroni control from Method \ref{meth:bonferroni-multiple-mean-comparisons} to the chosen pairwise rank tests.  This is especially natural when only a small set of scientifically planned comparisons is of interest.
\end{method}

\begin{remark} \label{rem:kruskal-wallis-scope} \hlt{What Rank ANOVA Tests}\\
The Kruskal--Wallis null is equality of the treatment response distributions, not merely equality of means.  It is often interpreted as a location comparison when the group distributions have similar shapes.  Because it uses ranks, it makes no normality assumption and is less driven by extreme numerical values than the one-way $F$ test.
\end{remark}
