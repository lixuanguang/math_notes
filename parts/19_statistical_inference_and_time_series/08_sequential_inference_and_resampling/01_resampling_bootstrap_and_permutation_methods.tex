\subsection{Resampling, Bootstrap, and Permutation Methods}

Resampling methods approximate sampling distributions by recomputing statistics on data-generated pseudo-samples.

\begin{definition} \label{def:nonparametric-bootstrap} \hlt{Nonparametric Bootstrap}\\
For $X_{1:n}\sim P^{\otimes n}$, set $\widehat{P}_n\defeq n^{-1}\sum\limits_{i=1}^{n}\delta_{X_i}$. Conditional on $X_{1:n}$, let $X_{1:n}^\ast\sim\widehat{P}_n^{\otimes n}$ and $\widehat{P}_n^\ast\defeq n^{-1}\sum\limits_{i=1}^{n}\delta_{X_i^\ast}$.\\
For any law functional $T$, $\mathcal{L}(T(\widehat{P}_n^\ast)-T(\widehat{P}_n)\mid X_{1:n})$ approximates $\mathcal{L}(T(\widehat{P}_n)-T(P))$.\\
\hlt{Key Insight}: $P$ is replaced by $\widehat{P}_n$ before pseudo-sampling.
\end{definition}

\begin{method} \label{meth:bootstrap-standard-error} \hlt{Bootstrap Standard Error}\\
Let $\widehat{\psi}=s(X_{1:n})$ be observed statistic, $\mathcal{R}_\ast(X_{1:n})$ be the chosen bootstrap law on pseudo-samples, $B\geq2$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Resample}: $X_{1:n}^{\ast b}\sim\mathcal{R}_\ast(X_{1:n})$ for $b=1,\ldots,B$.
\item \hlt{Recompute}: $\widehat{\psi}^{\ast b}\defeq s(X_{1:n}^{\ast b})$ for $b=1,\ldots,B$.
\item \hlt{Estimate}: $\widehat{\operatorname{se}}_\ast(\widehat{\psi})\defeq\{(B-1)^{-1}\sum\limits_{b=1}^{B}(\widehat{\psi}^{\ast b}-\overline{\psi}^{\ast})^2\}^{\frac{1}{2}}$, where $\overline{\psi}^{\ast}\defeq B^{-1}\sum\limits_{b=1}^{B}\widehat{\psi}^{\ast b}$.
\end{enumerate}
\end{method}

\begin{method} \label{meth:bootstrap-location-standard-errors} \hlt{Bootstrap Variability for Location Summaries}\\
Let $T=T(x_1,\ldots,x_n)$ be a location summary and let $B\geq1$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Resample}: For $1\leq b\leq B$, draw $x_1^{*(b)},\ldots,x_n^{*(b)}$ iid from $\widehat F_n\defeq n^{-1}\sum\limits_{i=1}^n\delta_{x_i}$.
\item \hlt{Recompute}: Set $T_b^\ast\defeq T(x_1^{*(b)},\ldots,x_n^{*(b)})$ and $\overline{T}^{\ast}\defeq B^{-1}\sum\limits_{b=1}^BT_b^\ast$.
\item \hlt{Spread}: Estimate variability by $\widehat{\operatorname{se}}_{\ast}(T)\defeq\left((B-1)^{-1}\sum\limits_{b=1}^B(T_b^\ast-\overline{T}^{\ast})^2\right)^{1/2}$.
\end{enumerate}
\hlt{Key Insight}: This is useful for medians and trimmed means, where analytic standard errors can be tail-sensitive.
\end{method}

\begin{definition} \label{def:parametric-bootstrap} \hlt{Parametric Bootstrap}\\
Let $\{P_\theta:\theta\in\Theta\}$ be parametric model for $X_{1:n}\defeq(X_1,\ldots,X_n)$, let $\widehat{\theta}\defeq\widehat{\theta}(X_{1:n})$, and let $S$ be a statistic.\\
$X_1^\ast,\ldots,X_n^\ast\mid X_{1:n}\sim P_{\widehat{\theta}}^{\otimes n}$, and $\mathcal{L}(S(X_{1:n}^\ast)\mid X_{1:n})$ is the fitted-model resampling law for $S(X_{1:n})$.\\
\hlt{Key Insight}: The model class $\{P_\theta\}$ is retained; only $\theta$ is replaced by $\widehat{\theta}$.
\end{definition}

\begin{method} \label{meth:parametric-bootstrap-standard-errors-and-intervals} \hlt{Parametric Bootstrap Standard Errors and Intervals}\\
Let $\{P_\theta:\theta\in\Theta\}$ be parametric model, $\widehat{\psi}\defeq s(X_{1:n})$ estimate $\psi(\theta)$, fix $B\geq2$, fix confidence level $1-\alpha$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Fit}: $\widehat{\theta}\defeq\widehat{\theta}(X_{1:n})$.
\item \hlt{Simulate}: $X_{1:n}^{\ast b}\sim P_{\widehat{\theta}}^{\otimes n}$ and $\widehat{\psi}^{\ast b}\defeq s(X_{1:n}^{\ast b})$ for $b=1,\ldots,B$.
\item \hlt{Standard error}: $\widehat{\operatorname{se}}_\ast(\widehat{\psi})\defeq\{(B-1)^{-1}\sum\limits_{b=1}^{B}(\widehat{\psi}^{\ast b}-\overline{\psi}^{\ast})^2\}^{\frac{1}{2}}$, where $\overline{\psi}^{\ast}\defeq B^{-1}\sum\limits_{b=1}^{B}\widehat{\psi}^{\ast b}$.
\item \hlt{Pivotal interval}: for empirical quantiles $q_{\frac{\alpha}{2}}^\ast$ and $q_{1-\frac{\alpha}{2}}^\ast$ of $\{\widehat{\psi}^{\ast b}-\widehat{\psi}\}_{b=1}^{B}$, use $[\widehat{\psi}-q_{1-\frac{\alpha}{2}}^\ast,\widehat{\psi}-q_{\frac{\alpha}{2}}^\ast]$.
\end{enumerate}
The percentile interval uses empirical quantiles of $\{\widehat{\psi}^{\ast b}\}_{b=1}^{B}$ directly.\\
The pivotal and percentile intervals coincide when the bootstrap error law is symmetric about $0$.
\end{method}

\begin{remark} \label{rem:parametric-bootstrap-plug-in-principle} \hlt{Plug-In Monte Carlo}\\
The parametric bootstrap estimates model-dependent quantities by replacing the unknown parameter with its estimate before simulation. If $h(X,\theta)$ is a loss, error indicator, or other accuracy summary, then Monte Carlo from $P_{\widehat{\theta}}$ approximates $\mathbb{E}_\theta h(X,\theta)$ by the empirical average of $h(X^{\ast b},\widehat{\theta})$.
\end{remark}

\begin{definition} \label{def:permutation-test} \hlt{Permutation Test}\\
Let a finite group $G$ act on the sample space $\mathcal{X}$, let $P_0$ satisfy $gX\disteq X$ under $P_0$ $\forall g\in G$, and let $T:\mathcal{X}\to\mathbb{R}$ be a test statistic. With $U_G$ uniform on $G$ and independent of $X$, $\mathcal{L}(T(U_GX)\mid X)$, the empirical law on $\{T(gX):g\in G\}$, is the reference law for tail probabilities.\\
\hlt{Key Insight}: Calibration follows from the invariance $gX\disteq X$ under $P_0$.
\end{definition}

\begin{method} \label{meth:permutation-plug-in-fdr} \hlt{Permutation Plug-In FDR Estimate}\\
Let $t_j$ be observed statistics for $j\in[M]$, and let $t_j^{(k)}$ be the statistic after label permutation $k\in[K]$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Observed discoveries}: For cutpoint $C>0$, set $R_{\mathrm{obs}}(C)\defeq\sum\limits_{j=1}^M\mathbf{1}_{\{|t_j|>C\}}$.
\item \hlt{Null discoveries}: Estimate the expected false discoveries by $\widehat{\mathbb{E}}(V;C)\defeq K^{-1}\sum\limits_{k=1}^K\sum\limits_{j=1}^M\mathbf{1}_{\{|t_j^{(k)}|>C\}}$.
\item \hlt{Plug-in FDR}: If $R_{\mathrm{obs}}(C)>0$, set $\widehat{\operatorname{FDR}}(C)\defeq\widehat{\mathbb{E}}(V;C)/R_{\mathrm{obs}}(C)$.
\item \hlt{Permutation $p$-values}: Use $\{t_j^{(k)}\}$ to compare $|t_j|$ with featurewise or pooled permutation null statistics.
\end{enumerate}
\end{method}

\begin{remark} \label{rem:sam-asymmetric-cutpoints} \hlt{Asymmetric Cutpoints and SAM}\\
Let $t_{(1)}\leq\cdots\leq t_{(M)}$ and $\bar{t}^{0}_{(j)}\defeq K^{-1}\sum\limits_{k=1}^Kt^{(k)}_{(j)}$, where $t^{(k)}_{(j)}$ is the $j$th ordered permutation statistic.\\
For $d_j\defeq t_{(j)}-\bar{t}^{0}_{(j)}$, $a>0$, set $c_+(a)\defeq\min\{t_{(j)}:d_j>a\}$, $c_-(a)\defeq\max\{t_{(j)}:d_j<-a\}$ when nonempty.\\
Reject $t_j\in A(c_-,c_+)\defeq(-\infty,c_-]\cup[c_+,\infty)$, estimate FDR by Method~\ref{meth:permutation-plug-in-fdr}, $A(c_-,c_+)$ replace $\{|t|>C\}$.
\end{remark}

\begin{remark} \label{rem:resampling-validity} \hlt{Validity Depends on the Resampling Law}\\
Bootstrap validity is asymptotic; permutation validity is finite-sample exact under invariance; both require the resampling law to match the target sampling variation.
\end{remark}
