\subsection{Diagonal Dominance and Definiteness}

Throughout this subsection, $n\in\mathbb{Z}_{>0}$.

\begin{definition} \label{def:diagonal-dominance} \hlt{Diagonal Dominance}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The matrix $A$ is \hlt{diagonally dominant} iff $(\forall\,k\in J_n)\,\abs{a_{kk}}\geq R_k^{\mathrm{off}}(A)$.
\item The matrix $A$ is \hlt{strictly diagonally dominant} iff $(\forall\,k\in J_n)\,\abs{a_{kk}}>R_k^{\mathrm{off}}(A)$.
\end{enumerate}
\end{definition}

\begin{corollary} \label{cor:strict-diagonal-dominance-invertible} \hlt{Strict Diagonal Dominance Implies Invertibility}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A$ is strictly diagonally dominant, then $A$ is invertible.
\item If $\operatorname{spec}(A)\subseteq\mathbb{R}$, $(\forall\,k\in J_n)\,a_{kk}\in\mathbb{R}_{\geq0}$, and $A$ is diagonally dominant, then $\operatorname{spec}(A)\subseteq[0,\infty)$.
\item If $\operatorname{spec}(A)\subseteq\mathbb{R}$, $(\forall\,k\in J_n)\,a_{kk}\in\mathbb{R}_{\geq0}$, and $A$ is strictly diagonally dominant, then $\operatorname{spec}(A)\subseteq(0,\infty)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $0\notin\operatorname{Ger}(A)$ by Definition \ref{def:gersgorin-disks} and strict diagonal dominance.\\
The relation $0\notin\operatorname{spec}(A)$ holds by Theorem \ref{thm:gersgorin-disk-theorem}, and $A$ is invertible by Corollary \ref{cor:matrix-invertible-iff-zero-not-spectrum}.
\item $(\forall\,k\in J_n)(\forall\,z\in\operatorname{Ger}_k(A))\,\operatorname{Re}(z)\geq a_{kk}-\abs{z-a_{kk}}\geq a_{kk}-R_k^{\mathrm{off}}(A)\geq0$ by Definition \ref{def:gersgorin-disks} and diagonal dominance.\\
The inclusion $\operatorname{spec}(A)\subseteq\operatorname{Ger}(A)\cap\mathbb{R}\subseteq[0,\infty)$ holds by the preceding bound, Theorem \ref{thm:gersgorin-disk-theorem}, and the real-spectrum hypothesis.
\item $(\forall\,k\in J_n)(\forall\,z\in\operatorname{Ger}_k(A))\,\operatorname{Re}(z)\geq a_{kk}-R_k^{\mathrm{off}}(A)>0$ by Definition \ref{def:gersgorin-disks} and strict diagonal dominance.\\
The inclusion $\operatorname{spec}(A)\subseteq\operatorname{Ger}(A)\cap\mathbb{R}\subseteq(0,\infty)$ holds by the preceding bound, Theorem \ref{thm:gersgorin-disk-theorem}, and the real-spectrum hypothesis.
\end{enumerate}
\hlt{Structural Commentary}: Disk exclusion handles (i); closed and open half-plane bounds handle (ii)--(iii).\\
\hlt{Key Insight}: Each Gersgorin radius is controlled by its nonnegative real center.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:weak-diagonal-dominance-invertible-with-one-strict-row} \hlt{Weak Diagonal Dominance with One Strict Row}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$ be diagonally dominant and satisfy $(\forall\,i,j\in J_n)\,[i\neq j\Longrightarrow a_{ij}\neq0]$.\\
If $(\exists\,k\in J_n)\,\abs{a_{kk}}>R_k^{\mathrm{off}}(A)$, then $A$ is invertible.
\begin{proof}
Suppose $\mathbf{0}\neq\mathbf{x}\in\ker(A)$, $M\defeq\norm{\mathbf{x}}_{\infty}$. Choose $p\in J_n$ satisfying $\abs{x_p}=M>0$ by Proposition \ref{prop:coordinate-norms-on-fn}(iii).\\
$\abs{a_{pp}}M=\abs{\sum\limits_{j\in J_n\setminus\{p\}}a_{pj}x_j}\leq\sum\limits_{j\in J_n\setminus\{p\}}\abs{a_{pj}}\abs{x_j}\leq R_p^{\mathrm{off}}(A)M$ by $A\mathbf{x}=\mathbf{0}$, the triangle inequality, and maximality of $\abs{x_p}$.\\
$R_p^{\mathrm{off}}(A)M\leq\abs{a_{pp}}M$ by diagonal dominance; all preceding inequalities are equalities by their common endpoints.\\
The relation $(\forall\,j\in J_n)\,\abs{x_j}=M$ holds by equality in the weighted bound, $\abs{x_p}=M$, and the nonzero off-diagonal hypothesis.\\
Fix $k\in J_n$ satisfying $\abs{a_{kk}}>R_k^{\mathrm{off}}(A)$ by the strict-row hypothesis.\\
$\abs{a_{kk}}M=\abs{\sum\limits_{j\in J_n\setminus\{k\}}a_{kj}x_j}\leq R_k^{\mathrm{off}}(A)M$ by $A\mathbf{x}=\mathbf{0}$ and the preceding modulus identity.\\
$R_k^{\mathrm{off}}(A)M<\abs{a_{kk}}M$ by strict dominance and $M>0$, a contradiction.\\
The equality $\ker(A)=\{\mathbf{0}\}$ holds by contradiction, and $A$ is invertible by Corollary \ref{cor:square-matrix-algebraic-invertibility-dictionary}(vi)$\Rightarrow$(i).\\
\hlt{Structural Commentary}: A maximal kernel coordinate propagates equal moduli before the strict row closes the contradiction.\\
\hlt{Key Insight}: Positive off-diagonal weights propagate one maximal modulus to every coordinate.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:diagonal-dominant-hermitian-positive-semidefinite} \hlt{Diagonal Dominance and Positive Semidefiniteness}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$ be Hermitian.\\
Suppose $(\forall\,k\in J_n)\,a_{kk}\in\mathbb{R}_{\geq0}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A$ is diagonally dominant, then $A$ is positive semidefinite.
\item If $A$ is diagonally dominant and invertible, then $A$ is positive definite.
\item If $A$ is strictly diagonally dominant, then $A$ is positive definite.
\end{enumerate}
\begin{proof}
The induced $T_A\in\LL(\mathbb{C}^n)$ is self-adjoint by Proposition \ref{prop:matrix-induced-map-linear}, Theorems \ref{thm:matrix-of-adjoint-is-conjugate-transpose} and \ref{thm:linear-maps-and-matrices-same-data}(i), and $A=A^*$.\\
The relation $\operatorname{spec}(A)=\operatorname{spec}(T_A)\subseteq\mathbb{R}$ holds by Definitions \ref{def:linear-map-induced-by-matrix}, \ref{def:eigenspace}, and \ref{def:spectrum-and-geometric-multiplicity}, and Theorem \ref{thm:self-adjoint-eigenvalues-real}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{spec}(A)\subseteq[0,\infty)$ by Corollary \ref{cor:strict-diagonal-dominance-invertible}(ii), so $A\succeq0$ by Theorem \ref{thm:positive-semidefinite-matrix-equivalences}(ii).
\item $A\succeq0$ by part (i), so $A\succ0$ by Theorem \ref{thm:positive-definite-matrix-equivalences}(iii) and the invertibility hypothesis.
\item $\operatorname{spec}(A)\subseteq(0,\infty)$ by Corollary \ref{cor:strict-diagonal-dominance-invertible}(iii), so $A\succ0$ by Theorem \ref{thm:positive-definite-matrix-equivalences}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Hermitian reality activates Gersgorin bounds; spectral criteria transfer eigenvalue signs to definiteness.\\
\hlt{Key Insight}: Invertibility removes zero from the nonnegative Hermitian spectrum.
\end{proof}
\end{corollary}
