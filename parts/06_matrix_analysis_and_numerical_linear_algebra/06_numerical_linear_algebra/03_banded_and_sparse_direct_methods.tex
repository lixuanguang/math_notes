\subsection{Banded and Sparse Direct Methods}

Throughout this subsection, $\mathbb{F}$ is a field.

\begin{definition} \label{def:band-and-tridiagonal-matrix} \hlt{Half-Bandwidth and Tridiagonal Matrices}\\
Let $m\in\mathbb{Z}_{>0}$, $A=[a_{ij}]\in\operatorname{Mat}(m,m,\mathbb{F})$, $w\in J_m$. Matrix $A$ has \hlt{half-bandwidth at most $w$} if $(\forall\,i,j\in J_m)[\abs{i-j}\geq w\Longrightarrow a_{ij}=0]$.\\
Its \hlt{half-bandwidth} is the least such $w$, which exists because $w=m$ is admissible.\\
Matrices of half-bandwidth at most $1$ are precisely diagonal matrices. Matrices of half-bandwidth at most $2$ are \hlt{tridiagonal}.
\end{definition}

\begin{computationmethod} \label{meth:banded-elimination-preview} \hlt{Banded Elimination without Row Exchanges}\\
Let $n\in\mathbb{Z}_{>0}$, $w\in J_n$, $A\in\operatorname{Mat}(n,n,\mathbb{F})$ have half-bandwidth at most $w$. Assume every leading principal minor of $A$ is nonzero.\\
Compute the unique exact-arithmetic factors $A=LU$, with $L$ unit lower triangular and $U$ upper triangular, as follows:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Initialize $L\gets I_n$, $U\gets0$, and a current active array $B=[b_{ij}]\gets A$.
\item For $k=1,\ldots,n$ in order, set $u_{kj}\gets b_{kj}$ for $k\leq j\leq\min(n,k+w-1)$.\\
For $i=k+1,\ldots,\min(n,k+w-1)$, set $l_{ik}\gets\frac{b_{ik}}{u_{kk}}$ and $b_{ik}\gets0$.\\
Then set $b_{ij}\gets b_{ij}-l_{ik}u_{kj}$ for $k<j\leq\min(n,k+w-1)$.
\item Return $(L,U)$.
\end{enumerate}
Let $A_k$ be the leading $k\times k$ block and set $\det(A_0)\defeq1$. Prior row additions preserve $\det(A_k)$ and triangularise its first $k-1$ columns.\\
Hence $\det(A_k)=\left(\prod\limits_{s=1}^{k-1}u_{ss}\right)b_{kk}$. The minor hypothesis and induction from $k=1$ imply $u_{kk}=b_{kk}\neq0$, so every division is defined.\\
An update requires $0<i-k<w$ and $0<j-k<w$, so $\abs{i-j}<w$.\\
Thus induction preserves the band, and $L$ and $U$ have half-bandwidth at most $w$.\\
Reversing the stored row operations establishes $A=LU$. The returned pair is the unique factorisation by Theorem \ref{thm:lu-factorization-leading-principal-minors}.\\
At pivot $k$, set $r_k\defeq\min(w-1,n-k)$; the work is $r_k$ divisions and $r_k^2$ multiply-subtract updates.\\
Thus the worst-case scalar-field operation count is $O(nw^2)$ by $r_k\leq w-1$.\\
If $w=1$, the inner ranges are empty and $(L,U)=(I_n,A)$; if $w=2$, the cost is $O(n)$.\\
The finite loops terminate, but without a rounding-error model the method makes no stability claim.\\
Outside the stated hypotheses, preserving the band does not justify suppressing required row exchanges.
\end{computationmethod}

\begin{perspective} \label{per:five-point-stencil-bandwidth} \hlt{Five-Point Stencils Are Sparse with Growing Bandwidth}\\
Let $N\in\mathbb{Z}_{\geq2}$ and place the nodes of an $N\times N$ grid in row-major order.\\
Let $S_N\in\operatorname{Mat}(N^2,N^2,\mathbb{F})$ have nonzero entries precisely at self-couplings and grid-neighbour couplings.\\
Each row has at most five nonzeros, at offsets $0,\pm1,\pm N$; boundary gaps remove horizontal wraparound.\\
Thus $S_N$ has half-bandwidth $N+1$ by Definition \ref{def:band-and-tridiagonal-matrix}. Its row sparsity is bounded independently of $N$.\\
An incomplete LU approximation may control fill by retaining a selected pattern or applying a drop rule.\\
Its quality depends on that rule and the ordering.
\end{perspective}
