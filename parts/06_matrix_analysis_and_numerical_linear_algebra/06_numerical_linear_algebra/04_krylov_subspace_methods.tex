\subsection{Krylov Subspace Methods}

Throughout this subsection, $n\in\mathbb{Z}_{>0}$ and $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$.

\begin{definition} \label{def:krylov-subspace} \hlt{Krylov Subspace}\\
Let $m\in\mathbb{Z}_{>0}$, $A\in\operatorname{Mat}(n,n,\mathbb{F})$, and $\mathbf{b}\in\mathbb{F}^n$. Set $\mathcal{K}_m(A,\mathbf{b})\defeq\operatorname{span}\{\mathbf{b},A\mathbf{b},\ldots,A^{m-1}\mathbf{b}\}$.\\
This is the $m$th \hlt{Krylov subspace} generated by $(A,\mathbf{b})$.
\end{definition}

\begin{computationmethod} \label{meth:arnoldi-lanczos-cg-preview} \hlt{Arnoldi, Lanczos, CG, and GMRES Templates}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ and work in exact arithmetic.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Arnoldi}: Let $\mathbf{b}\in\mathbb{F}^n\setminus\{\mathbf{0}\}$ and $m\in J_n$. Set $\mathbf{q}_1\defeq\mathbf{b}/\norm{\mathbf{b}}_2$. For $j=1,\ldots,m$, set $\mathbf{v}_j\defeq A\mathbf{q}_j$ and $h_{ij}\defeq\mathbf{q}_i^*\mathbf{v}_j$ for $i\in J_j$.\\
Set $\mathbf{w}_j\defeq\mathbf{v}_j-\sum\limits_{i=1}^{j}h_{ij}\mathbf{q}_i$ and $\eta_j\defeq\norm{\mathbf{w}_j}_2$. If $\eta_j=0$, set $s\defeq j$ and stop; otherwise, set $h_{j+1,j}\defeq\eta_j$ and $\mathbf{q}_{j+1}\defeq\mathbf{w}_j/\eta_j$.\\
Write $Q_j\defeq[\mathbf{q}_1\ \cdots\ \mathbf{q}_j]$ and $H_j\defeq[h_{i\ell}]_{i,\ell=1}^{j}$, where $h_{i\ell}\defeq0$ for $i>\ell+1$.\\
The identities $\mathbf{q}_i^*\mathbf{w}_j=0$ for $i\in J_j$ and $\mathbf{q}_{j+1}\in\mathcal{K}_{j+1}(A,\mathbf{b})$ imply inductively that $Q_j^*Q_j=I_j$ and $\operatorname{col}(Q_j)\subseteq\mathcal{K}_j(A,\mathbf{b})$.\\
Before breakdown, $\dim(\operatorname{col}(Q_j))=j$ and $\dim(\mathcal{K}_j(A,\mathbf{b}))\leq j$, so the inclusion is equality.\\
If no breakdown occurs through $m$, then $AQ_m=Q_mH_m+\eta_m\mathbf{q}_{m+1}\mathbf{e}_m^t$, and $H_m$ is upper Hessenberg.\\
At breakdown $s$, the same relation becomes $AQ_s=Q_sH_s$, so $\mathcal{K}_s(A,\mathbf{b})$ is $A$-invariant and the construction terminates.
\item \hlt{Lanczos}: Suppose $A=A^*$ and Arnoldi does not break down through step $m$.\\
Then $H_m=Q_m^*AQ_m=H_m^*$, so its upper-Hessenberg structure is tridiagonal by Definition \ref{def:band-and-tridiagonal-matrix}.\\
Set $\mathbf{q}_0\defeq\mathbf{0}$, $\beta_0\defeq0$, $\alpha_j\defeq h_{jj}\in\mathbb{R}$, and $\beta_j\defeq\eta_j>0$ for $j\in J_m$.\\
The Arnoldi relation becomes $(\forall\,j\in J_m)\,A\mathbf{q}_j=\beta_{j-1}\mathbf{q}_{j-1}+\alpha_j\mathbf{q}_j+\beta_j\mathbf{q}_{j+1}$.\\
At breakdown $s$, set $\beta_s\defeq0$; the recurrence through $j=s$ remains valid.
\item \hlt{CG, conjugate-direction form}: Suppose $A\succ0$ and fix $\mathbf{b},\mathbf{x}_0\in\mathbb{F}^n$. Set $\mathbf{r}_0\defeq\mathbf{b}-A\mathbf{x}_0$. If $\mathbf{r}_0=\mathbf{0}$, return $\mathbf{x}_0$.\\
Otherwise, let $k\in J_n$ satisfy $\dim(\mathcal{K}_k(A,\mathbf{r}_0))=k$. Define $\langle\mathbf{u},\mathbf{v}\rangle_A\defeq\mathbf{v}^*A\mathbf{u}$, an inner product by $A\succ0$.\\
Apply Gram--Schmidt in this inner product to $(\mathbf{r}_0,A\mathbf{r}_0,\ldots,A^{k-1}\mathbf{r}_0)$, obtaining an $A$-orthogonal basis $\mathbf{p}_0,\ldots,\mathbf{p}_{k-1}$ of $\mathcal{K}_k(A,\mathbf{r}_0)$.\\
For $i\in\{0,\ldots,k-1\}$, set $c_i\defeq\frac{\mathbf{p}_i^*\mathbf{r}_0}{\mathbf{p}_i^*A\mathbf{p}_i}$. The denominators are positive by $A\succ0$. Set $\mathbf{x}_k\defeq\mathbf{x}_0+\sum\limits_{i=0}^{k-1}c_i\mathbf{p}_i$ and $\mathbf{r}_k\defeq\mathbf{b}-A\mathbf{x}_k$.\\
For $i\in\{0,\ldots,k-1\}$, $A$-orthogonality reduces $\mathbf{p}_i^*\mathbf{r}_k$ to $\mathbf{p}_i^*\mathbf{r}_0-c_i\mathbf{p}_i^*A\mathbf{p}_i=0$. Hence $\mathbf{r}_k\perp\mathcal{K}_k(A,\mathbf{r}_0)$.\\
This correction is unique: if $\mathbf{z},\mathbf{z}'\in\mathcal{K}_k(A,\mathbf{r}_0)$ both satisfy the residual condition, then $\mathbf{d}\defeq\mathbf{z}-\mathbf{z}'$ lies in the same space.\\
Their residual difference is $-A\mathbf{d}\perp\mathcal{K}_k(A,\mathbf{r}_0)$, so $\mathbf{d}^*A\mathbf{d}=0$ and $\mathbf{d}=\mathbf{0}$ by $A\succ0$.
\item \hlt{GMRES}: Fix $\mathbf{b},\mathbf{x}_0\in\mathbb{F}^n$ and $k\in J_n$. Set $\mathbf{r}_0\defeq\mathbf{b}-A\mathbf{x}_0$. Set $\mathcal{A}_k\defeq\mathbf{x}_0+\mathcal{K}_k(A,\mathbf{r}_0)$.\\
If $\mathbf{r}_0=\mathbf{0}$, return $\mathbf{x}_0$. Otherwise, set $s\defeq\dim(\mathcal{K}_k(A,\mathbf{r}_0))$.\\
Choose an orthonormal basis matrix $Q_s\in\operatorname{Mat}(n,s,\mathbb{F})$ for this subspace and $\mathbf{y}_k\in\operatorname{LS}(AQ_s,\mathbf{r}_0)$ by Theorem \ref{thm:matrix-least-squares-normal-equations}(ii).\\
Set $\mathbf{x}_k\defeq\mathbf{x}_0+Q_s\mathbf{y}_k$. Since $\mathcal{A}_k=\{\mathbf{x}_0+Q_s\mathbf{y}:\mathbf{y}\in\mathbb{F}^s\}$, $\mathbf{x}_k\in\operatorname*{arg\,min}\limits_{\mathbf{x}\in\mathcal{A}_k}\norm{\mathbf{b}-A\mathbf{x}}_2$.
\end{enumerate}
These templates access $A$ through matrix-vector products. With stored products, a basis extension uses one new multiplication by $A$.\\
The remaining work consists of inner products, vector updates, and reduced problems. The prescribed loops are finite because $m,k\leq n$.\\
In finite precision, orthogonality and convergence require a separate rounding-error analysis.
\end{computationmethod}

\begin{perspective} \label{per:krylov-methods-preview} \hlt{Projection Conditions across Krylov Methods}\\
Definition \ref{def:krylov-subspace} supplies the common search spaces in Computation Method \ref{meth:arnoldi-lanczos-cg-preview}.\\
Arnoldi constructs orthonormal coordinates, and Lanczos reduces the recurrence when $A$ is Hermitian.\\
CG uses $A$-orthogonal directions to impose residual orthogonality. GMRES instead minimizes the Euclidean residual norm.\\
These correspondences are exact-arithmetic statements; they do not supply finite-precision convergence guarantees.
\end{perspective}
