\subsection{Hadamard Products}

Throughout this subsection, $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$.

\begin{definition} \label{def:hadamard-product} \hlt{Hadamard Product}\\
Let $m,n\in\mathbb{Z}_{>0}$ and $A=[a_{ij}],B=[b_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$.\\
The \hlt{Hadamard product}, or \hlt{entrywise product}, of $A$ and $B$ is $A\circ B\defeq [a_{ij}b_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$.
\end{definition}

\begin{proposition} \label{prop:basic-hadamard-identities} \hlt{Basic Hadamard Identities}\\
Let $m,n\in\mathbb{Z}_{>0}$, $\gamma\in\mathbb{F}$, and $A=[a_{ij}],B=[b_{ij}],C=[c_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$. Set $E\defeq[1]_{i\in J_m,j\in J_n}\in\operatorname{Mat}(m,n,\mathbb{F})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\circ B=B\circ A$;
\item $A\circ(\gamma B+C)=\gamma(A\circ B)+A\circ C$;
\item $(A\circ B)^*=A^*\circ B^*$;
\item $A\circ E=A$;
\item if $m=n$, $d_1,\ldots,d_n\in\mathbb{F}$, and $D=\operatorname{diag}(d_1,\ldots,d_n)$, then $A\circ D=\operatorname{diag}(a_{11}d_1,\ldots,a_{nn}d_n)$.
\end{enumerate}
\begin{proof}
Fix $(i,j)\in J_m\times J_n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(A\circ B)_{ij}=a_{ij}b_{ij}=b_{ij}a_{ij}=(B\circ A)_{ij}$.
\item $\bigl(A\circ(\gamma B+C)\bigr)_{ij}=a_{ij}(\gamma b_{ij}+c_{ij})=\gamma(A\circ B)_{ij}+(A\circ C)_{ij}$.
\item $\bigl((A\circ B)^*\bigr)_{ji}=\overline{a_{ij}b_{ij}}=\overline{a_{ij}}\,\overline{b_{ij}}=(A^*\circ B^*)_{ji}$.
\item $(A\circ E)_{ij}=a_{ij}=A_{ij}$.
\item If $m=n$, then $(A\circ D)_{ij}=a_{ij}D_{ij}$ equals $a_{ii}d_i$ if $i=j$ and $0$ otherwise.
\end{enumerate}
\hlt{Structural Commentary}: A fixed entry reduces (i)--(iv) to scalar arithmetic; (v) holds by the support of $D$.\\
\hlt{Key Insight}: No summation couples distinct coordinate pairs.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:schur-product-theorem} \hlt{Schur Product Theorem}\\
Let $n\in\mathbb{Z}_{>0}$ and $A=[a_{ij}],B=[b_{ij}]\in\operatorname{Mat}(n,n,\mathbb{F})$ satisfy $A\succeq0$ and $B\succeq0$. Then $A\circ B\succeq0$.
\begin{proof}
Set $R\defeq A^{1/2}$ and $S\defeq B^{1/2}$; then $A=R^2=R^*R$ and $B=S^2=S^*S$ by Corollary \ref{cor:positive-semidefinite-square-root-formula}(i).\\
Write $R=[\mathbf{u}_1\ \cdots\ \mathbf{u}_n]$, $S=[\mathbf{v}_1\ \cdots\ \mathbf{v}_n]$, and set $(\mathbf{w}_j)_{j=1}^{n}\defeq(\mathbf{u}_j\otimes\mathbf{v}_j)_{j=1}^{n}$. $(\forall\,i\in J_n)\,\mathbf{w}_i^*=\mathbf{u}_i^*\otimes\mathbf{v}_i^*$ by Proposition \ref{prop:basic-kronecker-identities}(iv),(vi).\\
$(\forall\,(i,j)\in J_n^2)\,\mathbf{w}_i^*\mathbf{w}_j=(\mathbf{u}_i^*\mathbf{u}_j)(\mathbf{v}_i^*\mathbf{v}_j)=a_{ij}b_{ij}=(A\circ B)_{ij}$ by Theorem \ref{thm:kronecker-mixed-product-property}.\\
$A\circ B=\operatorname{Gram}(\mathbf{w}_1,\ldots,\mathbf{w}_n)\succeq0$ by the preceding identity and Proposition \ref{prop:gram-matrix-positive-semidefinite}(i),(ii).\\
\hlt{Structural Commentary}: Gram factorizations and the mixed-product law identify $A\circ B$ as a Gram matrix.\\
\hlt{Key Insight}: The $j$th tensor column multiplies the two $ij$-Gram entries.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:hadamard-positive-definite-criterion} \hlt{Positive Definite Hadamard Products}\\
Let $n\in\mathbb{Z}_{>0}$ and $A\in\operatorname{Mat}(n,n,\mathbb{F})$, $B=[b_{ij}]\in\operatorname{Mat}(n,n,\mathbb{F})$ satisfy $A\succeq0$ and $B\succeq0$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A\succ0$ and $(\forall\,j\in J_n)\,b_{jj}>0$, then $A\circ B\succ0$.
\item If $A\succ0$ and $B\succ0$, then $A\circ B\succ0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:positive-definite-matrix-equivalences}(ii), set $\lambda_{\min}\defeq\min\operatorname{spec}(A)>0$.\\
Then $A-\lambda_{\min}I_n\succeq0$ by Theorem \ref{thm:positive-semidefinite-matrix-equivalences}(ii) and $\operatorname{spec}(A-\lambda_{\min}I_n)=\{\lambda-\lambda_{\min}:\lambda\in\operatorname{spec}(A)\}\subseteq[0,\infty)$.\\
Set $C\defeq(A-\lambda_{\min}I_n)\circ B$. Then $C\succeq0$ by Theorem \ref{thm:schur-product-theorem}. Fix $\mathbf{x}=(x_1,\ldots,x_n)^t\in\mathbb{F}^n\setminus\{\mathbf{0}\}$.\\
The identity $\mathbf{x}^*(A\circ B)\mathbf{x}=\mathbf{x}^*C\mathbf{x}+\lambda_{\min}\sum\limits_{j=1}^{n}b_{jj}\abs{x_j}^2$ holds by Proposition \ref{prop:basic-hadamard-identities}(i),(ii),(v).\\
Its right side is positive by $C\succeq0$, $\lambda_{\min}>0$, the diagonal hypothesis, and $\mathbf{x}\neq\mathbf{0}$.\\
$A\circ B\succ0$ by Theorem \ref{thm:schur-product-theorem}, the preceding strict inequality, and arbitrariness of $\mathbf{x}$.
\item The hypothesis in (i), $(\forall\,j\in J_n)\,b_{jj}=\mathbf{e}_j^*B\mathbf{e}_j>0$, holds by $B\succ0$. $A\circ B\succ0$ by (i).
\end{enumerate}
\hlt{Structural Commentary}: A spectral-floor split separates a semidefinite Hadamard term from a positive diagonal term.\\
\hlt{Key Insight}: $I_n\circ B=\operatorname{diag}(b_{11},\ldots,b_{nn})$ transfers diagonal positivity.
\end{proof}
\end{corollary}

\begin{proofmethod} \label{meth:schur-product-positivity} \hlt{Hadamard-Product Positivity Certificate}\\
Let $n\in\mathbb{Z}_{>0}$ and $X,Y\in\operatorname{Mat}(n,n,\mathbb{F})$ satisfy $X\succeq0$ and $Y\succeq0$. The following steps construct a Gram certificate for $X\circ Y\succeq0$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item set $R\defeq X^{1/2}$ and $S\defeq Y^{1/2}$; then $X=R^2=R^*R$ and $Y=S^2=S^*S$ by Corollary \ref{cor:positive-semidefinite-square-root-formula}(i);
\item write $R=[\mathbf{u}_1\ \cdots\ \mathbf{u}_n]$, $S=[\mathbf{v}_1\ \cdots\ \mathbf{v}_n]$, and set $(\mathbf{w}_j)_{j=1}^{n}\defeq(\mathbf{u}_j\otimes\mathbf{v}_j)_{j=1}^{n}$;
\item verify $(\forall\,(i,j)\in J_n^2)\,(X\circ Y)_{ij}=\mathbf{w}_i^*\mathbf{w}_j$ by Proposition \ref{prop:basic-kronecker-identities}(iv),(vi) and Theorem \ref{thm:kronecker-mixed-product-property};
\item return $X\circ Y=\operatorname{Gram}(\mathbf{w}_1,\ldots,\mathbf{w}_n)\succeq0$ by Proposition \ref{prop:gram-matrix-positive-semidefinite}(i),(ii).
\end{enumerate}
\end{proofmethod}
