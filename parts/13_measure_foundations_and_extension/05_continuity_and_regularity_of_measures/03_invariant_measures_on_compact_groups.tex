\subsection{Invariant Measures on Compact Groups} \label{subsec:invariant-measures-on-compact-groups}

Compact groups are the meeting point of topological symmetry and Radon measure theory. The compactness supplies fixed points in dual spaces, and Riesz--Markov converts those fixed functionals into invariant Borel measures.

\begin{definition} \label{def:compact-topological-groups-and-haar-measures} \hlt{Compact Groups and Haar Measures}\\
For a group $G$, $A\subseteq G$, and $g\in G$, set $gA\defeq\{ga:a\in A\}$ and $Ag\defeq\{ag:a\in A\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{topological group} is $G$ with Hausdorff topology such that $(g,h)\mapsto gh$ and $g\mapsto g^{-1}$ are continuous.
\item A \hlt{compact group} is a topological group whose underlying topological space is compact.
\item A Borel measure $\mu$ on $\mathcal{B}(G)$ is \hlt{left-invariant} if $\mu(gA)=\mu(A)$ for all $g\in G$ and $A\in\mathcal{B}(G)$.
\item A Borel measure $\mu$ on $\mathcal{B}(G)$ is \hlt{right-invariant} if $\mu(Ag)=\mu(A)$ for all $g\in G$ and $A\in\mathcal{B}(G)$.
\item A \hlt{Haar measure} on a compact group $G$ is a left-invariant Radon probability measure on $\mathcal{B}(G)$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:regular-representation-compact-group} \hlt{Regular Actions on $C(G)$}\\
Let $G$ be compact, $C(G)$ be continuous maps $G\to\mathbb{R}$ with $\norm{f}_\infty\defeq\max_{x\in G}|f(x)|$. For $g,x\in G$ and $f\in C(G)$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{left regular action} is $\pi_{\mathrm{L}}:G\to\mathcal{L}(C(G))$, $[\pi_{\mathrm{L}}(g)f](x)\defeq f(g^{-1}x)$.
\item The \hlt{right regular action} is $\pi_{\mathrm{R}}:G\to\mathcal{L}(C(G))$, $[\pi_{\mathrm{R}}(g)f](x)\defeq f(xg^{-1})$.
\item A \hlt{probability functional} is $\psi\in C(G)^*$ with $\psi(\mathbf{1}_G)=1$ and $f\geq0\Rightarrow\psi(f)\geq0$ for all $f\in C(G)$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:kakutani-fixed-probability-functional-compact-group} \hlt{Kakutani Fixed Probability Functional}\\
Let $G$ be a compact group. Then there are probability functionals $\psi_{\mathrm{L}},\psi_{\mathrm{R}}\in C(G)^*$ such that $\psi_{\mathrm{L}}(\pi_{\mathrm{L}}(g)f)=\psi_{\mathrm{L}}(f)$ and $\psi_{\mathrm{R}}(\pi_{\mathrm{R}}(g)f)=\psi_{\mathrm{R}}(f)$ for every $g\in G$ and $f\in C(G)$.
\begin{proof} Let $\mathcal{P}(G)\defeq\{\psi\in C(G)^*:\psi(\mathbf{1}_G)=1,\ \forall f\in C(G)\ (f\geq0\Rightarrow\psi(f)\geq0)\}$. Then $\delta_e\in\mathcal{P}(G)$, $\mathcal{P}(G)$ is convex and weak-* closed, and $\psi\in\mathcal{P}(G)\Rightarrow\normop{\psi}=1$; hence $\mathcal{P}(G)$ is weak-* compact by Banach--Alaoglu.\\
For $g\in G$, set $T_{\mathrm{L}}(g)\psi\defeq\psi\circ\pi_{\mathrm{L}}(g^{-1})$ and $T_{\mathrm{R}}(g)\psi\defeq\psi\circ\pi_{\mathrm{R}}(g)$. Since $\pi_{\mathrm{L}},\pi_{\mathrm{R}}$ preserve $\mathbf{1}_G$ and $C(G)_+$ and are norm-continuous in $g$, $T_{\mathrm{L}}$ and $T_{\mathrm{R}}$ are continuous affine $G$-actions on $\mathcal{P}(G)$.\\
Kakutani gives $\psi_{\mathrm{L}},\psi_{\mathrm{R}}\in\mathcal{P}(G)$ with $T_{\mathrm{L}}(g)\psi_{\mathrm{L}}=\psi_{\mathrm{L}}$ and $T_{\mathrm{R}}(g)\psi_{\mathrm{R}}=\psi_{\mathrm{R}}$ for all $g\in G$.\\
Thus $\psi_{\mathrm{L}}\circ\pi_{\mathrm{L}}(g)=\psi_{\mathrm{L}}$ and $\psi_{\mathrm{R}}\circ\pi_{\mathrm{R}}(g)=\psi_{\mathrm{R}}$ for all $g\in G$.\\
\hlt{Structural Commentary}: Compactness lives in $\mathcal{P}(G)$; invariance comes from fixed points.\\
\hlt{Key Insight}: A fixed probability functional is invariant.
\end{proof}
\end{theorem}

\begin{method} \label{meth:kakutani-functional-fixed-point-pattern} \hlt{Fixed Functional Pattern}\\
For a compact group $G$, set $\mathcal{P}(G)\defeq\{\psi\in C(G)^*:\psi(\mathbf{1}_G)=1,\ \forall f\in C(G)\ (f\geq0\Rightarrow\psi(f)\geq0)\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Show $\mathcal{P}(G)\neq\varnothing$ via $\delta_x(f)\defeq f(x)$, $x\in G$.
\item Show $\mathcal{P}(G)$ is weak-* compact convex: $\psi\in\mathcal{P}(G)\Rightarrow\normop{\psi}=1$,\\
Banach--Alaoglu, and weak-* closed defining conditions.
\item Define $T_{\mathrm{L}}(g)\psi\defeq\psi\circ\pi_{\mathrm{L}}(g^{-1})$ or $T_{\mathrm{R}}(g)\psi\defeq\psi\circ\pi_{\mathrm{R}}(g)$.\\
For $T=T_{\mathrm{L}},T_{\mathrm{R}}$, check $T(g)\mathcal{P}(G)\subseteq\mathcal{P}(G)$ and weak-* continuity.
\item Apply Kakutani: find $\psi\in\mathcal{P}(G)$ with $T(g)\psi=\psi$ for all $g\in G$.
\item Apply Riesz--Markov, $\psi(f)=\int_G f\,d\mu$, and read $T(g)\psi=\psi$ as invariance of $\mu$.
\end{enumerate}
\hlt{Structural Commentary}: Compact state space plus adjoint action.\\
\hlt{Key Insight}: Haar measure is an invariant state represented by Riesz--Markov.
\end{method}

\begin{proposition} \label{prop:translated-borel-measures-on-compact-groups} \hlt{Translated Measures and Test Functions}\\
Let $G$ be a compact group, let $\mu$ be a Borel measure on $\mathcal{B}(G)$, and fix $g\in G$.\\
Set $L_g(x)\defeq gx$ and $R_g(x)\defeq xg$ for $x\in G$.\\
For $A\in\mathcal{B}(G)$, define $\mu_g^{\mathrm{L}}(A)\defeq\mu(L_g(A))=\mu(gA)$ and $\mu_g^{\mathrm{R}}(A)\defeq\mu(R_g(A))=\mu(Ag)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu_g^{\mathrm{L}}$ and $\mu_g^{\mathrm{R}}$ are Borel measures on $\mathcal{B}(G)$.
\item If $\mu$ is Radon, then $\mu_g^{\mathrm{L}}$ and $\mu_g^{\mathrm{R}}$ are Radon.
\item For every $f\in C(G)$, $\int_G \pi_{\mathrm{L}}(g)f\,d\mu=\int_G f\,d\mu_g^{\mathrm{L}}$ and $\int_G \pi_{\mathrm{R}}(g)f\,d\mu=\int_G f\,d\mu_g^{\mathrm{R}}$.
\end{enumerate}
\begin{proof}
For every $h\in G$, the maps $L_h$ and $R_h$ are homeomorphisms of $G$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $L_g$ and $R_g$ map Borel sets bijectively onto Borel sets, $\mu_g^{\mathrm{L}}$ and $\mu_g^{\mathrm{R}}$ are defined on $\mathcal{B}(G)$.\\
If $(A_n)_{n\geq1}$ are pairwise disjoint Borel sets, then $L_g(\bigcup_n A_n)=\bigcup_n L_g(A_n)$ and $R_g(\bigcup_n A_n)=\bigcup_n R_g(A_n)$, with disjoint unions; countable additivity follows from countable additivity of $\mu$.
\item Since $L_{g^{-1}}^{-1}=L_g$ and $R_{g^{-1}}^{-1}=R_g$, $\mu_g^{\mathrm{L}}=(L_{g^{-1}})_*\mu$ and $\mu_g^{\mathrm{R}}=(R_{g^{-1}})_*\mu$.\\
If $\mu$ is Radon, pushforward by a homeomorphism preserves compact sets, open sets, and Radon regularity.\\
Hence both translated measures are Radon.
\item For $f\in C(G)$, the pushforward identity gives $\int_G f\,d\mu_g^{\mathrm{L}}=\int_G f\circ L_{g^{-1}}\,d\mu=\int_G \pi_{\mathrm{L}}(g)f\,d\mu$\\
and $\int_G f\,d\mu_g^{\mathrm{R}}=\int_G f\circ R_{g^{-1}}\,d\mu=\int_G \pi_{\mathrm{R}}(g)f\,d\mu$.
\end{enumerate}
\hlt{Structural Commentary}: Translated measures are inverse-translation pushforwards.\\
\hlt{Key Insight}: Measure translation is dual to test-function precomposition.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:haar-measure-on-compact-groups} \hlt{Haar Measure on Compact Groups}\\
Let $G$ be a compact group. Let $\mathcal{P}_{\mathrm{L}}(G)$ and $\mathcal{P}_{\mathrm{R}}(G)$ be the left- and right-invariant Radon probability measures on $\mathcal{B}(G)$, respectively.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{P}_{\mathrm{L}}(G)\neq\emptyset$.
\item $\mathcal{P}_{\mathrm{R}}(G)\neq\emptyset$.
\item For every $(\lambda,\rho)\in\mathcal{P}_{\mathrm{L}}(G)\times\mathcal{P}_{\mathrm{R}}(G)$, $\lambda=\rho$.
\end{enumerate}
Consequently, $G$ admits a unique Haar measure, and this measure is right-invariant.
\begin{proof}
For a Radon probability measure $\sigma$ on $\mathcal{B}(G)$, set $\Phi_\sigma(f)\defeq\int_G f\,d\sigma$ for $f\in C(G)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $\psi_{\mathrm{L}}$ from Theorem \ref{thm:kakutani-fixed-probability-functional-compact-group}. Riesz--Markov gives a Radon probability measure $\mu$ with $\psi_{\mathrm{L}}=\Phi_\mu$ on $C(G)=C_c(G)$. For $g\in G$, Proposition \ref{prop:translated-borel-measures-on-compact-groups} gives $\Phi_{\mu_g^{\mathrm{L}}}(f)=\Phi_\mu(\pi_{\mathrm{L}}(g)f)=\Phi_\mu(f)$ for every $f\in C(G)$. Thus $\mu_g^{\mathrm{L}}=\mu$ by Theorem \ref{thm:radon-measures-determined-by-cc-integrals}, so $\mu(gA)=\mu(A)$ for every $A\in\mathcal{B}(G)$ and $\mu\in\mathcal{P}_{\mathrm{L}}(G)$.
\item Choose $\psi_{\mathrm{R}}$ from Theorem \ref{thm:kakutani-fixed-probability-functional-compact-group}. Riesz--Markov gives a Radon probability measure $\nu$ with $\psi_{\mathrm{R}}=\Phi_\nu$ on $C(G)=C_c(G)$. For $g\in G$, Proposition \ref{prop:translated-borel-measures-on-compact-groups} gives $\Phi_{\nu_g^{\mathrm{R}}}(f)=\Phi_\nu(\pi_{\mathrm{R}}(g)f)=\Phi_\nu(f)$ for every $f\in C(G)$. Thus $\nu_g^{\mathrm{R}}=\nu$ by Theorem \ref{thm:radon-measures-determined-by-cc-integrals}, so $\nu(Ag)=\nu(A)$ for every $A\in\mathcal{B}(G)$ and $\nu\in\mathcal{P}_{\mathrm{R}}(G)$.
\item Let $(\lambda,\rho)\in\mathcal{P}_{\mathrm{L}}(G)\times\mathcal{P}_{\mathrm{R}}(G)$, fix $f\in C(G)$, and set $h_f(x,y)\defeq f(xy)$. Fubini applies to $h_f\in C(G\times G)$. Left-invariance gives $\int_G h_f(x,y)\,d\lambda(y)=\Phi_\lambda(f)$ for every $x$, and right-invariance gives $\int_G h_f(x,y)\,d\rho(x)=\Phi_\rho(f)$ for every $y$. Since $\lambda(G)=\rho(G)=1$, Fubini gives $\Phi_\lambda(f)=\Phi_\rho(f)$. Hence $\lambda=\rho$ by Theorem \ref{thm:radon-measures-determined-by-cc-integrals}.
\end{enumerate}
Taking $\mu\in\mathcal{P}_{\mathrm{L}}(G)$ from (i) and $\nu\in\mathcal{P}_{\mathrm{R}}(G)$ from (ii), (iii) gives $\mu=\nu$, so $\mu$ is right-invariant. If $\eta$ is Haar, then $\eta\in\mathcal{P}_{\mathrm{L}}(G)$ and (iii) gives $\eta=\nu$, so Haar measure is unique.\\
\hlt{Structural Commentary}: Fixed states give existence; Fubini gives uniqueness.\\
\hlt{Key Insight}: Invariance makes both iterated integrals constant.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:haar-point-masses-on-compact-groups} \hlt{Point Masses of Haar Measure}\\
Let $G$ be a compact group, and let $\mu$ be Haar measure on $\mathcal{B}(G)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $G$ is finite, then every $A\subseteq G$ is Borel and $\mu(A)=\frac{\abs{A}}{\abs{G}}$.
\item If $G$ is infinite, then $\mu(\{x\})=0$ for every $x\in G$.
\end{enumerate}
\begin{proof}
Set $c\defeq\mu(\{e\})$. For each $x\in G$, left-invariance gives $\mu(\{x\})=\mu(x\{e\})=c$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $G$ is finite, then $G$ is discrete, so every $A\subseteq G$ is Borel. Since $1=\mu(G)=\sum\limits_{x\in G}\mu(\{x\})=\abs{G}c$, one has $c=\frac{1}{\abs{G}}$, and finite additivity gives $\mu(A)=\sum\limits_{a\in A}\mu(\{a\})=\frac{\abs{A}}{\abs{G}}$.
\item If $G$ is infinite and $c>0$, choose $n\in\mathbb{N}$ with $n>\frac{1}{c}$ and distinct $x_1,\ldots,x_n\in G$.\\
Finite additivity gives $1=\mu(G)\geq\sum\limits_{j=1}^{n}\mu(\{x_j\})=nc>1$, a contradiction; hence $c=0$.
\end{enumerate}
\hlt{Structural Commentary}: Translation reduces point masses to one scalar.\\
\hlt{Key Insight}: Normalization fixes or kills that scalar.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:haar-positive-on-nonempty-open-sets} \hlt{Nonempty Open Sets Have Positive Haar Measure}\\
Let $G$ be a compact group, $\mu$ be Haar measure on $\mathcal{B}(G)$, $O\subseteq G$ be nonempty and open. Then $\mu(O)>0$.
\begin{proof} Choose $a\in O$. For each $x\in G$, $x=(xa^{-1})a\in(xa^{-1})O$, so $\{gO:g\in G\}$ is an open cover of $G$.\\
Compactness gives $g_1,\ldots,g_n\in G$ such that $G=\bigcup_{j=1}^{n}g_jO$. If $\mu(O)=0$, then left-invariance and subadditivity give $1=\mu(G)\leq\sum_{j=1}^{n}\mu(g_jO)=\sum_{j=1}^{n}\mu(O)=0$, a contradiction. Hence $\mu(O)>0$.\\
\hlt{Structural Commentary}: Compactness gives a finite translate cover.\\
\hlt{Key Insight}: Zero mass cannot cover total mass.
\end{proof}
\end{proposition}
