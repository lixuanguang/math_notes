\subsection{Caratheodory Measurability and Lebesgue Measure}

\begin{definition} \label{def:outer-measure-and-rudin-measurable-sets} \hlt{Outer Measure and $\mu$-Measurable Sets}\\
Let $\mathcal{E}$ and $\mathcal{E}_{\mathrm{op}}$ denote the elementary and open elementary subsets of $\R^p$, let $\mu:\mathcal{E}\to[0,\infty)$ be additive and regular, and set $A\triangle B\defeq(A\setminus B)\cup(B\setminus A)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $E\subseteq\R^p$, $\mu^\ast(E)\defeq\inf\{\sum_{n=1}^{\infty}\mu(A_n):E\subseteq\bigcup_{n=1}^{\infty}A_n,\ A_n\in\mathcal{E}_{\mathrm{op}}\}$.
\item $d_\mu(A,B)\defeq\mu^\ast(A\triangle B)$.
\item $\mathcal{M}_F(\mu)$ is the class of \hlt{finitely $\mu$-measurable} sets: $E\in\mathcal{M}_F(\mu)$ iff $\exists(E_k)_{k\geq1}\subseteq\mathcal{E}$ with $d_\mu(E_k,E)\to0$.
\item $\mathcal{M}(\mu)$ is the class of \hlt{$\mu$-measurable} sets: $E\in\mathcal{M}(\mu)$ iff $E=\bigcup_{k=1}^{\infty}E_k$ for some $E_k\in\mathcal{M}_F(\mu)$.
\end{enumerate}
\end{definition}

\begin{perspective} \label{per:outer-measure-geometric-box-covers} \hlt{Outer Measure as Outside Approximation}\\
Outer measure covers $A$ from outside; $d_\mu(A,B)=\mu^\ast(A\triangle B)$ measures closeness.\\
\hlt{Key Insight}: Countable subadditivity collects near-optimal covers.
\end{perspective}

\begin{theorem} \label{thm:rudin-measure-extension} \hlt{Elementary Approximation Extension}\\
Let $\mathcal{E}$ be the elementary subsets of $\R^p$, let $\mathcal{E}_{\mathrm{op}}\subseteq\mathcal{E}$ be the open elementary sets, and let $\mu:\mathcal{E}\to[0,\infty)$ be additive and regular. For $E\subseteq\R^p$, set $\mu^\ast(E)\defeq\inf\{\sum_{n=1}^{\infty}\mu(A_n):E\subseteq\bigcup_{n=1}^{\infty}A_n,\ A_n\in\mathcal{E}_{\mathrm{op}}\}$; set $d_\mu(A,B)\defeq\mu^\ast(A\triangle B)$, let $\mathcal{M}_F(\mu)$ be the $d_\mu$-closure of $\mathcal{E}$, and let $\mathcal{M}(\mu)\defeq\{\bigcup_{k=1}^{\infty}E_k:E_k\in\mathcal{M}_F(\mu)\}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu^\ast(A)=\mu(A)$ for every $A\in\mathcal{E}$;
\item $\mu^\ast$ is countably subadditive on $2^{\R^p}$;
\item $\mathcal{M}(\mu)$ is a $\sigma$-ring;
\item $\mu^\ast|_{\mathcal{M}(\mu)}$ is a measure.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $A\in\mathcal{E}$. For $\varepsilon>0$, regularity gives $G\in\mathcal{E}_{\mathrm{op}}$ with $A\subseteq G$ and $\mu(G)\leq\mu(A)+\varepsilon$, so $\mu^\ast(A)\leq\mu(A)$.\\
Conversely, choose closed elementary $F\subseteq A$ with $\mu(F)\geq\mu(A)-\varepsilon$. Every open elementary cover of $A$ covers $F$; compactness gives a finite subcover, whose total $\mu$-mass is at least $\mu(F)$.\\
Taking infima and letting $\varepsilon\downarrow0$ gives $\mu(A)\leq\mu^\ast(A)$.
\item Let $E_j\subseteq\R^p$ for $j\geq1$, and let $\varepsilon>0$. For each $j$, choose an open elementary cover $(A_{j,n})_{n\geq1}$ of $E_j$ with $\sum_{n=1}^{\infty}\mu(A_{j,n})\leq\mu^\ast(E_j)+\varepsilon2^{-j}$. Then $(A_{j,n})_{j,n}$ covers $\bigcup_{j=1}^{\infty}E_j$, so $\mu^\ast(\bigcup_jE_j)\leq\sum_j\mu^\ast(E_j)+\varepsilon$; let $\varepsilon\downarrow0$.
\item If $A_k\to A$ and $B_k\to B$ in $d_\mu$, with $A_k,B_k\in\mathcal{E}$, then $d_\mu(A_k\diamond B_k,A\diamond B)\leq d_\mu(A_k,A)+d_\mu(B_k,B)$ for $\diamond\in\{\cup,\cap,\setminus\}$. Hence $\mathcal{M}_F(\mu)$ is ring. Countable unions give $\sigma$-ring since difference distributes over them.
\item If $A=\bigsqcup_{j=1}^{\infty}A_j$ with $A_j\in\mathcal{M}_F(\mu)$, subadditivity gives $\mu^\ast(A)\leq\sum_j\mu^\ast(A_j)$.\\
Finite additivity on $\mathcal{M}_F(\mu)$ gives $\mu^\ast(A)\geq\sum_{j=1}^{N}\mu^\ast(A_j)$ for every $N$, hence $\mu^\ast(A)\geq\sum_j\mu^\ast(A_j)$.\\
Disjointize countable $\mathcal{M}_F(\mu)$-decompositions.
\end{enumerate}
\hlt{Structural Commentary}: Elementary sets are completed under the distance measured by symmetric difference.\\
\hlt{Key Insight}: Approximate by elementary sets, prove identities there, then pass through $d_\mu(A,B)$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:extended-and-lebesgue-measure} \hlt{Extended and Lebesgue Measure}\\
Let $\mu^\ast$ be the outer measure induced by $\mu$ and let $\mathcal{M}(\mu)$ be its measurable class.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{extended measure} of $\mu$ is $\mu_{\mathrm{ext}}\defeq\mu^\ast|_{\mathcal{M}(\mu)}$, usually denoted again by $\mu$.
\item If $\mu=m$ is elementary volume on $\R^p$, then $m^\ast|_{\mathcal{M}(m)}$ is \hlt{Lebesgue measure} on $\R^p$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:tao-box-cover-outer-measure} \hlt{Box-Cover Outer Measure Model}\\
Fix $n\geq1$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{B}_n\defeq\{\prod_{i=1}^{n}(a_i,b_i):a_i,b_i\in\R,\ a_i\leq b_i\ \forall i\}$ is the class of \hlt{open boxes}.\\
For $B=\prod_{i=1}^{n}(a_i,b_i)$, $\operatorname{vol}_n(B)\defeq\prod_{i=1}^{n}(b_i-a_i)$.
\item For $\Omega\subseteq\R^n$, $m_n^\ast(\Omega)\defeq\inf\{\sum_{j\in J}\operatorname{vol}_n(B_j):\Omega\subseteq\bigcup_{j\in J}B_j,\ B_j\in\mathcal{B}_n,\ \abs{J}\leq\aleph_0\}$.
\end{enumerate}
\end{definition}

\begin{method} \label{meth:lebesgue-measure-construction} \hlt{Lebesgue Measure Construction}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Cover from outside}\\
For $A\subseteq\R^n$, define $m_n^\ast(A)\defeq\inf\left\{\sum_{j\in J}\operatorname{vol}_n(B_j):(B_j)_{j\in J}\text{ covers }A,\ J\text{ at most countable}\right\}$.\\
This is defined for every set, but it is only countably subadditive in general.
\item \hlt{Keep the clean splitters}\\
Call $A$ measurable when every test set $X\subseteq\R^n$ splits additively: $m_n^\ast(X)=m_n^\ast(X\cap A)+m_n^\ast(X\cap A^c)$.\\
These measurable sets form a $\sigma$-algebra.
\item \hlt{Restrict the outer measure}\\
Define Lebesgue measure on measurable sets by $m(A)\defeq m_n^\ast(A)$.\\
On this restricted class, $m$ is countably additive.
\end{enumerate}
\hlt{Key Insight}: Outer measure sizes all sets; clean splitting restores additivity.
\end{method}

\begin{definition} \label{def:lebesgue-sigma-algebra-caratheodory} \hlt{Lebesgue $\sigma$-Algebra via Caratheodory}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{L}^n\defeq\{E\subseteq\R^n:E$ is $m_n^\ast$-measurable$\}$ is the \hlt{Lebesgue $\sigma$-algebra}.
\item $m\defeq m_n^\ast|_{\mathcal{L}^n}$ is \hlt{Lebesgue measure}; for $E\in\mathcal{L}^n$, write $m(E)=m_n^\ast(E)$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:outer-null-sets-lebesgue-measurable} \hlt{Outer-Null Sets Are Lebesgue Measurable}\\
If $Z\subseteq\R^n$ and $m_n^\ast(Z)=0$, then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $Z\in\mathcal{L}^n$;
\item $m(Z)=0$;
\item $\mathcal{P}(Z)\subseteq\mathcal{L}^n$ and $m(W)=0$ for every $W\subseteq Z$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $A\subseteq\R^n$, monotonicity gives $m_n^\ast(A\cap Z)=0$, so $m_n^\ast(A)\leq m_n^\ast(A\cap Z)+m_n^\ast(A\cap Z^c)=m_n^\ast(A\cap Z^c)\leq m_n^\ast(A)$; hence $Z$ satisfies Caratheodory's criterion.
\item Since $Z\in\mathcal{L}^n$, $m(Z)=m_n^\ast(Z)=0$.
\item If $W\subseteq Z$, then $m_n^\ast(W)\leq m_n^\ast(Z)=0$; applying (i)--(ii) to $W$ gives $W\in\mathcal{L}^n$ and $m(W)=0$.
\end{enumerate}
\hlt{Structural Commentary}: Lebesgue measure is complete because outer-null subsets split every test set trivially.\\
\hlt{Key Insight}: A zero piece cannot disturb additivity.
\end{proof}
\end{proposition}

\begin{example} \label{ex:countable-real-sets-outer-null} \hlt{Countable Real Sets Are Outer-Null}\\
Let $C\subseteq\R$ be finite or countable, and choose $(c_k)_{k\geq1}$ with $C\subseteq\{c_k:k\geq1\}$. For $\varepsilon>0$, choose open intervals $I_k\ni c_k$ with $\ell(I_k)<\varepsilon2^{-k}$. Then $C\subseteq\bigcup_{k\geq1}I_k$, so $0\leq m_1^\ast(C)\leq\sum_{k=1}^{\infty}\ell(I_k)<\varepsilon$. Since $\varepsilon>0$ is arbitrary, $m_1^\ast(C)=0$. By Proposition \ref{prop:outer-null-sets-lebesgue-measurable}, $C\in\mathcal{L}^1$ and $m(C)=0$.\\
\hlt{Key Insight}: A countable list can spend a rapidly decreasing measure budget around its points.
\end{example}

\begin{proposition} \label{prop:cells-lebesgue-measurable-volume} \hlt{Cells Are Lebesgue Measurable}\\
Let $I\subseteq\R^n$ be a coordinate cell. Then $I\in\mathcal{L}^n$ and $m(I)=\operatorname{vol}_n(I)$.
\begin{proof}
If $\operatorname{vol}_n(I)=0$, then $m_n^\ast(I)=0$ by Theorem \ref{thm:outer-measure-agrees-with-cell-volume}, so Proposition \ref{prop:outer-null-sets-lebesgue-measurable} gives $I\in\mathcal{L}^n$ and $m(I)=0$.\\
Assume $\operatorname{vol}_n(I)>0$. For small $r>0$, let $I_r$ be the inward $r$-shrink of $I$.\\
Then $\operatorname{dist}(I_r,\R^n\setminus I)>0$ and $\eta_r\defeq m_n^\ast(I\setminus I_r)\to0$ by finite slab covers.\\
Fix $A\subseteq\R^n$ with $m_n^\ast(A)<+\infty$. Positive-distance additivity gives $m_n^\ast(A)\geq m_n^\ast(A\cap I_r)+m_n^\ast(A\setminus I)$.\\
Since $A\cap I\subseteq(A\cap I_r)\cup(I\setminus I_r)$, $m_n^\ast(A\cap I)\leq m_n^\ast(A\cap I_r)+\eta_r$; hence $m_n^\ast(A)\geq m_n^\ast(A\cap I)+m_n^\ast(A\setminus I)-\eta_r$.\\
Letting $r\downarrow0$ gives the finite-measure test inequality, so Lemma \ref{lem:caratheodory-finite-measure-test} gives $I\in\mathcal{L}^n$.\\
Theorem \ref{thm:outer-measure-agrees-with-cell-volume} gives $m(I)=m_n^\ast(I)=\operatorname{vol}_n(I)$.\\
\hlt{Structural Commentary}: Cells are the bridge between coordinate volume and Lebesgue measure.\\
\hlt{Key Insight}: Shrink, split, let the collar vanish.
\end{proof}
\end{proposition}
