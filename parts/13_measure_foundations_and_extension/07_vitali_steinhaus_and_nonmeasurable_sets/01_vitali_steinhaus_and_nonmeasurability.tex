\subsection{Vitali, Steinhaus, and Nonmeasurability}

\begin{definition} \label{def:difference-set-and-vitali-selector} \hlt{Difference Sets and Vitali Selectors}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Difference set}: for $A\subseteq\R^p$, $A-A\defeq\{x-y:x,y\in A\}$; $A\subseteq B\Longrightarrow A-A\subseteq B-B$.
\item \hlt{Rational equivalence}: $x\sim_{\Q^p}y\Longleftrightarrow x-y\in\Q^p$; equivalently, $[x]_{\Q^p}=x+\Q^p$.
\item \hlt{Vitali selector}: choose $\mathcal{V}\subseteq\R^p$ such that $\abs{\mathcal{V}\cap(x+\Q^p)}=1$ for every $x\in\R^p$.
\item \hlt{Rational translate}: for $q\in\Q^p$, $\mathcal{V}_q\defeq q+\mathcal{V}$.
\end{enumerate}
\end{definition}

\begin{lemma} \label{lem:steinhaus-compact-positive-measure} \hlt{Steinhaus Lemma for Compact Sets}\\
If $K\subseteq\R^p$ is compact and $m(K)>0$, then $\exists\delta>0$ such that $B(0,\delta)\subseteq K-K$.
\begin{proof}
Choose open $G\supseteq K$ with $m(G)<2m(K)$. Since $K$ is compact and $G^c$ is closed, set $\delta\defeq\operatorname{dist}(K,G^c)>0$. If $\norm{x}_2<\delta$, then $x+K\subseteq G$. If $(x+K)\cap K=\emptyset$, translation invariance and additivity give\\
$2m(K)=m(x+K)+m(K)=m((x+K)\cup K)\leq m(G)<2m(K)$, a contradiction.\\
Hence $(x+K)\cap K\neq\emptyset$, so $x=k_1-k_2\in K-K$. Thus $B(0,\delta)\subseteq K-K$.\\
\hlt{Structural Commentary}: A positive compact set overlaps all sufficiently small translates of itself.\\
\hlt{Key Insight}: Disjoint small translates cannot fit inside a tight open envelope.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:steinhaus-positive-measure-difference-neighborhood} \hlt{Steinhaus Theorem}\\
If $E\in\mathcal{L}^p$ and $m(E)>0$, then $\exists\delta>0$ such that $B(0,\delta)\subseteq E-E$.
\begin{proof}
Set $E_n\defeq E\cap B(0,n)$. Since $E_n\uparrow E$ and $m(E)>0$, choose $n$ with $0<m(E_n)<+\infty$.\\
By compact inner approximation, choose compact $K\subseteq E_n$ with $m(K)>0$.\\
Lemma \ref{lem:steinhaus-compact-positive-measure} gives $\delta>0$ with $B(0,\delta)\subseteq K-K\subseteq E-E$.\\
\hlt{Structural Commentary}: Positive measurable sets contain positive compact cores.\\
\hlt{Key Insight}: Difference-set thickness follows from compact inner regularity.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:bounded-disjoint-translates-force-null} \hlt{Bounded Disjoint Translates Force Nullity}\\
Let $E\in\mathcal{L}^1$ and $\Lambda\subseteq\R$ be bounded, with $E$ bounded and $\Lambda$ countably infinite.\\
If $(\lambda+E)_{\lambda\in\Lambda}$ are pairwise disjoint, then $m(E)=0$.
\begin{proof} Since $E$ and $\Lambda$ are bounded, $U\defeq\bigcup\limits_{\lambda\in\Lambda}(\lambda+E)$ is bounded, so $m(U)<+\infty$.\\
By disjointness, countable additivity, and translation invariance, $m(U)=\sum\limits_{\lambda\in\Lambda}m(\lambda+E)=\sum\limits_{\lambda\in\Lambda}m(E)$.\\
Since $\Lambda$ is countably infinite, finiteness forces $m(E)=0$.\\
\hlt{Structural Commentary}: Infinitely many disjoint bounded translates cannot all carry positive equal mass.\\
\hlt{Key Insight}: Boundedness gives a finite container; translation invariance makes every summand the same.
\end{proof}
\end{lemma}

\begin{proposition} \label{prop:real-line-vitali-choice-set-positive-outer-measure} \hlt{Real-Line Vitali Choice Sets}\\
Let $E\subseteq\R$ with $m_1^\ast(E)>0$, and write $x\sim y$ iff $x-y\in\Q$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item There exists bounded $E_0\subseteq E$ with $m_1^\ast(E_0)>0$.
\item If $C\subseteq E_0$ contains exactly one point from each $\sim$-class meeting $E_0$, then $C\notin\mathcal{L}^1$.
\end{enumerate}
In particular, $E$ contains a nonmeasurable subset.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $E=\bigcup\limits_{n\geq1}(E\cap[-n,n])$, countable subadditivity gives $n$ with $E_0\defeq E\cap[-n,n]$ satisfying $m_1^\ast(E_0)>0$.
\item Choose $b>0$ with $E_0\subseteq[-b,b]$ and set $\Lambda_0\defeq[-2b,2b]\cap\Q$. Suppose $C\in\mathcal{L}^1$.\\
From $\lambda+c=\lambda'+c'$ with $\lambda,\lambda'\in\Lambda_0$ and $c,c'\in C$, we get $c-c'=\lambda'-\lambda\in\Q$; selector uniqueness gives $c=c'$ and $\lambda=\lambda'$. Thus $(\lambda+C)_{\lambda\in\Lambda_0}$ is pairwise disjoint, and Lemma \ref{lem:bounded-disjoint-translates-force-null} gives $m(C)=0$.\\
For every $x\in E_0$, choose $c\in C$ with $x-c\in\Q$; since $x,c\in[-b,b]$, $x-c\in\Lambda_0$, so $E_0\subseteq\bigcup\limits_{\lambda\in\Lambda_0}(\lambda+C)$.\\
The right side is a countable union of null measurable sets, contradicting $m_1^\ast(E_0)>0$.
\end{enumerate}
\hlt{Structural Commentary}: Bounded rational shifts force the contradiction.\\
\hlt{Key Insight}: Selectors are either null under disjoint shifts or cover too much by null shifts.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:vitali-translates-partition} \hlt{Rational Translates of a Vitali Selector Partition $\R^p$}\\
Let $(q_i)_{i\geq1}$ enumerate $\Q^p$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\R^p=\bigcup_{i=1}^{\infty}\mathcal{V}_{q_i}$.
\item Each $x\in\R^p$ has a unique representation $x=q_i+v$ with $i\in\N$ and $v\in\mathcal{V}$; hence $\R^p=\bigsqcup_{i=1}^{\infty}\mathcal{V}_{q_i}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $x\in\R^p$, choose the unique $v\in\mathcal{V}\cap(x+\Q^p)$. Then $x-v=q_i$ for some $i$, so $x=q_i+v\in\mathcal{V}_{q_i}$.
\item If $q_i+v=q_j+w$ with $v,w\in\mathcal{V}$, then $v-w=q_j-q_i\in\Q^p$.\\
Selector uniqueness gives $v=w$, hence $q_i=q_j$ and $i=j$.
\end{enumerate}
\hlt{Structural Commentary}: Rational translates enumerate the equivalence classes selected by $\mathcal{V}$.\\
\hlt{Key Insight}: One representative per class makes the rational translates disjoint and exhaustive.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:vitali-selector-nonmeasurable} \hlt{Vitali Selectors Are Nonmeasurable}\\
If $\mathcal{V}\subseteq\R^p$ is a Vitali selector, then $\mathcal{V}\notin\mathcal{L}^p$.
\begin{proof}
Assume $\mathcal{V}\in\mathcal{L}^p$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $m(\mathcal{V})>0$, Steinhaus gives $\delta>0$ with $B(0,\delta)\subseteq\mathcal{V}-\mathcal{V}$. Choose $0\neq q\in\Q^p\cap B(0,\delta)$. Then $q=v_1-v_2$ for some $v_1,v_2\in\mathcal{V}$, so $v_1\sim_{\Q^p}v_2$; selector uniqueness gives $v_1=v_2$, hence $q=0$, contradiction.
\item If $m(\mathcal{V})=0$, then each translate $\mathcal{V}_{q_i}$ is null. By Lemma \ref{lem:vitali-translates-partition}, $\R^p=\bigsqcup\limits_i\mathcal{V}_{q_i}$, so every bounded cell is contained in a countable union of null sets and has measure $0$, contradicting positive cell volume.
\end{enumerate}
\hlt{Structural Commentary}: Positive and zero measure are both impossible.\\
\hlt{Key Insight}: Steinhaus forbids positive measure; countable translated partition forbids zero measure.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:positive-outer-measure-contains-nonmeasurable-subset} \hlt{Nonmeasurable Subsets Are Everywhere}\\
If $E\subseteq\R^p$ and $m_p^\ast(E)>0$, then $\exists N\subseteq E$ with $N\notin\mathcal{L}^p$.
\begin{proof} Let $(q_i)$ enumerate $\Q^p$ and set $E_i\defeq E\cap\mathcal{V}_{q_i}$.\\
If $E_i\in\mathcal{L}^p$ and $m(E_i)>0$, then $E_i-E_i$ contains a ball about $0$, so $0\neq q\in\Q^p\cap(E_i-E_i)$.\\
But $E_i\subseteq q_i+\mathcal{V}$, so $q=(q_i+v_1)-(q_i+v_2)=v_1-v_2$ with $v_1,v_2\in\mathcal{V}$, contradicting selector uniqueness.\\
Hence each measurable $E_i$ is null. If every $E_i$ were measurable, then $E=\bigcup_iE_i$ would satisfy $m_p^\ast(E)=0$, contradiction. Thus some $E_i\subseteq E$ is nonmeasurable.\\
\hlt{Structural Commentary}: Any positive outer-measure set meets the Vitali partition in a nonmeasurable piece.\\
\hlt{Key Insight}: Vitali pieces cannot be measurable and positive.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:nonadditive-decompositions-from-nonmeasurable-sets} \hlt{Nonadditive Decompositions}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $E\in\mathcal{L}^p$ and $0<m(E)<+\infty$, then $\exists B,C\notin\mathcal{L}^p$ such that $E=B\sqcup C$ and $m(E)<m_p^\ast(B)+m_p^\ast(C)$.
\item If $B\notin\mathcal{L}^p$, $m_p^\ast(B)<+\infty$, and $H$ is $G_\delta$ with $B\subseteq H$, $m(H)=m_p^\ast(B)$,\\
then $C\defeq H\setminus B\notin\mathcal{L}^p$ and $m(H)=m_p^\ast(B\cup C)<m_p^\ast(B)+m_p^\ast(C)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Pick $B\subseteq E$, $B\notin\mathcal{L}^p$, and put $C\defeq E\setminus B$. Then $C\notin\mathcal{L}^p$, else $B=E\setminus C\in\mathcal{L}^p$.\\
Subadditivity gives $\leq$; equality would force $B\in\mathcal{L}^p$ by the finite-container test in $E$.
\item Since $H\in\mathcal{L}^p$, $C\in\mathcal{L}^p$ would give $B=H\setminus C\in\mathcal{L}^p$, contradiction. Also $m_p^\ast(C)>0$, else $C\in\mathcal{L}^p$.\\
Thus $B\cup C=H$ and $m(H)=m_p^\ast(B)$ give $m_p^\ast(B\cup C)=m(H)<m_p^\ast(B)+m_p^\ast(C)$.
\end{enumerate}
\hlt{Structural Commentary}: Nonmeasurability appears exactly as strict failure of additive splitting.\\
\hlt{Key Insight}: Equality in a finite measurable container is a measurability test.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:ubiquitous-nonmeasurable-set} \hlt{A Nonmeasurable Set with Only Null Measurable Pieces}\\
There exists $\mathcal{U}\subseteq\R^p$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item if $F\in\mathcal{L}^p$ and $F\subseteq\mathcal{U}$, then $m(F)=0$;
\item if $F\in\mathcal{L}^p$ and $F\subseteq\mathcal{U}^c$, then $m(F)=0$;
\item $\mathcal{U}\notin\mathcal{L}^p$.
\end{enumerate}
\begin{proof} Let $e\defeq(1,\ldots,1)$, $G_0\defeq\Q^p+(2\sqrt{2})\Z e$, $G_1\defeq\sqrt{2}e+G_0$, $G\defeq G_0\sqcup G_1=\Q^p+\sqrt{2}\Z e$; $G_0,G_1$ dense.\\
Let $\mathcal{E}$ select one representative from each class modulo $G$, and set $\mathcal{U}\defeq G_0+\mathcal{E}$.\\
Then $\R^p=(G_0+\mathcal{E})\sqcup(G_1+\mathcal{E})$ and $\mathcal{U}^c=G_1+\mathcal{E}=\sqrt{2}e+\mathcal{U}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Suppose $F\in\mathcal{L}^p$, $F\subseteq\mathcal{U}$, and $m(F)>0$. By Steinhaus, $B(0,r)\subseteq F-F$ for some $r>0$; choose $g_1\in G_1\cap B(0,r)$. Since $F\subseteq G_0+\mathcal{E}$, $g_1=(g_0+s_1)-(g_0'+s_2)$ for some $g_0,g_0'\in G_0$ and $s_1,s_2\in\mathcal{E}$. Thus $s_1-s_2=g_1-g_0+g_0'\in G$, so $s_1=s_2$ and $g_1=g_0-g_0'\in G_0$, contradicting $G_0\cap G_1=\emptyset$.
\item If $F\in\mathcal{L}^p$ and $F\subseteq\mathcal{U}^c=\sqrt{2}\,e+\mathcal{U}$, then $F-\sqrt{2}\,e\subseteq\mathcal{U}$ and $m(F)=m(F-\sqrt{2}\,e)=0$ by (i).
\item If $\mathcal{U}\in\mathcal{L}^p$, then (i)--(ii) give $m(\mathcal{U})=m(\mathcal{U}^c)=0$, so $m([0,1]^p)=0$, contradiction.
\end{enumerate}
\hlt{Structural Commentary}: This construction partitions representatives using two dense cosets.\\
\hlt{Key Insight}: Positive measurable pieces force forbidden coset differences.
\end{proof}
\end{theorem}
