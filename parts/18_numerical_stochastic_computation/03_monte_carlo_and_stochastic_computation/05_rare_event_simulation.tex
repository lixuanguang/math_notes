\subsection{Rare-Event Simulation}
% Importance sampling for tails, splitting methods, large deviations preview.

\begin{example} \label{ex:tail-probability-importance-sampling} \hlt{Rare Tail Probability}\\
For $Z\sim N(0,1)$, write $I=\mathbb{P}(Z>3)=\int \mathbf{1}\{x>3\}f(x)\,dx$.\\
If $g$ is centered near the right tail and $X_1,\ldots,X_N\sim g$, estimate $I$ by $\widehat{I}_N=N^{-1}\sum\limits_{i=1}^{N}\frac{\mathbf{1}\{X_i>3\}f(X_i)}{g(X_i)}$.
\end{example}
