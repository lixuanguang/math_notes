\subsection{Continuity and Regularity of Measures}

\begin{corollary} \label{cor:measure-continuity-up-down} \hlt{Measure Continuity}\\
Let $\mu$ be a measure. If $E_k\uparrow E$, then $\mu(E_k)\uparrow\mu(E)$. If $F_k\downarrow F$ and $\mu(F_1)<+\infty$, then $\mu(F_k)\downarrow\mu(F)$.
\begin{proof}
Disjointize $E_k\uparrow E$ by $E'_1=E_1$ and $E'_k=E_k\setminus E_{k-1}$; countable additivity makes $\mu(E_k)$ the $k$th partial sum for $\mu(E)$. For $F_k\downarrow F$, apply this to $F_1\setminus F_k\uparrow F_1\setminus F$ and subtract from finite $\mu(F_1)$.\\
\hlt{Structural Commentary}: Monotone set limits are disjoint series or complements of such series.\\
\hlt{Key Insight}: Downward continuity needs one finite anchor so subtraction is legal.
\end{proof}
\end{corollary}

\begin{remark} \label{rem:borel-null-regular-measurable-sets} \hlt{Borel Sets, Null Sets, and Regular Approximation}\\
Open, closed, and Borel subsets of $\R^p$ are measurable. Each $A\in\mathcal{M}(\mu)$ admits closed--open approximations $F\subseteq A\subseteq G$ with $\mu(G\setminus A)$ and $\mu(A\setminus F)$ arbitrarily small. In Lebesgue measure, one may take $F_\sigma\subseteq A\subseteq G_\delta$ with $\mu(G_\delta\setminus F_\sigma)=0$. Null sets form a $\sigma$-ring, and countable subsets of $\R^p$ are null.\\
\hlt{Structural Commentary}: Measurable sets are Borel sets modified by null sets.\\
\hlt{Key Insight}: Regularity explains why open and closed approximations are enough for integration.
\end{remark}

\begin{corollary} \label{cor:lebesgue-regularity-and-borel-completion-real-line} \hlt{Lebesgue Regularity and Borel Completion on $\R$}\\
For $A\in\mathcal{L}^1$,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $m(A)=\inf\{m(G):A\subseteq G,\ G$ open in $\R\}$.
\item $m(A)=\sup\{m(K):K\subseteq A,\ K$ compact in $\R\}$, finite-measure case applied to $A\cap(-n,n]$; $m(A)=+\infty$.
\item There exists $B\in\mathcal{B}(\R)$ with $A\subseteq B$ and $m(B\setminus A)=0$; equivalently, $\mathcal{L}^1$ is the completion of $\mathcal{B}(\R)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\varepsilon>0$, regular approximation gives open $G\supseteq A$ with $m(G\setminus A)<\varepsilon$, hence $m(G)\leq m(A)+\varepsilon$.\\
Monotonicity gives the reverse inequality.
\item If $m(A)<+\infty$, choose closed $F\subseteq A$ with $m(A\setminus F)<\varepsilon$. Since $F\cap[-n,n]\uparrow F$, continuity from below gives compact $K=F\cap[-n,n]$ with $m(F)-m(K)<\varepsilon$, so $m(A)-m(K)<2\varepsilon$. If $m(A)=+\infty$, apply the finite case to $A_n=A\cap(-n,n]$ and let $n\to\infty$.
\item Choose open $G_k\supseteq A$ with $m(G_k\setminus A)<1/k$, and set $B=\bigcap_kG_k$. Then $B\in\mathcal{B}(\R)$, $A\subseteq B$, and $m(B\setminus A)=0$. Thus every Lebesgue set is Borel plus a null subset.\\
Conversely, Borel sets and null subsets are Lebesgue measurable.
\end{enumerate}
\hlt{Structural Commentary}: Lebesgue sets are Borel sets plus null-set ambiguity.\\
\hlt{Key Insight}: Open outer approximation and compact inner approximation are the usable form of regularity.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:real-line-local-measurability-unit-intervals} \hlt{Local Measurability on Unit Intervals}\\
Let $I_n\defeq(n,n+1]$, $n\in\Z$. For $E\subseteq\R$, $E\in\mathcal{L}^1$ iff $E\cap I_n\in\mathcal{L}^1$ for all $n\in\Z$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Since $I_n\in\mathcal{B}(\R)\subseteq\mathcal{L}^1$, $E\in\mathcal{L}^1$ implies $E\cap I_n\in\mathcal{L}^1$ for all $n$.
\item[$(\Leftarrow)$] If $E\cap I_n\in\mathcal{L}^1$ for all $n$, then $E=\bigcup_{n\in\Z}(E\cap I_n)\in\mathcal{L}^1$.
\end{enumerate}
\hlt{Structural Commentary}: A countable Borel exhaustion localizes measurability.\\
\hlt{Key Insight}: Local measurable pieces patch by countable unions.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:finite-measure-meets-every-unit-interval} \hlt{Finite Measure Meeting Every Unit Interval}\\
With $I_n\defeq(n,n+1]$, there exists $E\in\mathcal{B}(\R)$ such that $0<m(E\cap I_n)$ for all $n\in\Z$ and $m(E)<+\infty$.
\begin{proof}
Set $\varepsilon_n\defeq2^{-\abs{n}-3}$ and $E\defeq\bigcup_{n\in\Z}(n,n+\varepsilon_n]$. Then $(n,n+\varepsilon_n]\subseteq I_n$ are pairwise disjoint Borel intervals, so $E\in\mathcal{B}(\R)$ and $m(E\cap I_n)=\varepsilon_n>0$. Countable additivity gives $m(E)=\sum_{n\in\Z}\varepsilon_n=\frac38<+\infty$.\\
\hlt{Structural Commentary}: Finite measure can be spread over countably many cells.\\
\hlt{Key Insight}: Positive local masses can be summable globally.
\end{proof}
\end{proposition}

\begin{definition} \label{def:symmetric-difference} \hlt{Symmetric Difference}\\
For $A,B\subseteq X$, define $A\triangle B\defeq(A\setminus B)\cup(B\setminus A)=(A\cup B)\setminus(A\cap B)$.\\
Equivalently, $x\in A\triangle B$ iff $x$ belongs to exactly one of $A,B$.
\end{definition}

\begin{proposition} \label{prop:null-sets-sigma-ideal} \hlt{Null Sets Form a Sigma-Ideal}\\
Let $(X,\mathcal{M},\mu)$ be a measure space and $\mathcal{Z}\defeq\{N\in\mathcal{M}:\mu(N)=0\}$.\\
Then $\mathcal{Z}$ is a $\sigma$-ring, and $N\in\mathcal{Z}$, $E\in\mathcal{M}$ imply $N\cap E\in\mathcal{Z}$.
\begin{proof}
If $(N_k)\subseteq\mathcal{Z}$, then $\mu(\bigcup_kN_k)\leq\sum_k\mu(N_k)=0$, so $\bigcup_kN_k\in\mathcal{Z}$.\\
If $A,B\in\mathcal{Z}$, then $A\setminus B\subseteq A$, hence $A\setminus B\in\mathcal{Z}$. Finally, $N\cap E\subseteq N$ gives $N\cap E\in\mathcal{Z}$.\\
\hlt{Structural Commentary}: Null sets are stable under countable unions and measurable cuts.\\
\hlt{Key Insight}: Monotonicity makes every measurable piece of a null set null.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:completed-sigma-algebra} \hlt{Completed Sigma-Algebra}\\
Let $(X,\mathcal{M},\mu)$ be measure space, $\mathcal{Z}\defeq\{N\in\mathcal{M}:\mu(N)=0\}$; $\overline{\mathcal{M}}_\mu\defeq\{A\subseteq X:\exists E\in\mathcal{M},\ \exists N\in\mathcal{Z},\ A\triangle E\subseteq N\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{M}\subseteq\overline{\mathcal{M}}_\mu$, and $\overline{\mathcal{M}}_\mu$ is a $\sigma$-algebra;
\item $A\subseteq N\in\mathcal{Z}\Longrightarrow A\in\overline{\mathcal{M}}_\mu$;
\item $\overline{\mathcal{M}}_\mu=\{E\cup Z:E\in\mathcal{M},\ \exists N\in\mathcal{Z},\ Z\subseteq N\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $E\in\mathcal{M}$, $E\triangle E=\emptyset$, so $\mathcal{M}\subseteq\overline{\mathcal{M}}_\mu$. If $A\triangle E\subseteq N$, then $A^c\triangle E^c=A\triangle E\subseteq N$.\\
If $A_k\triangle E_k\subseteq N_k$, then $(\bigcup_kA_k)\triangle(\bigcup_kE_k)\subseteq\bigcup_kN_k\in\mathcal{Z}$. Hence $\overline{\mathcal{M}}_\mu$ is a $\sigma$-algebra.
\item If $A\subseteq N\in\mathcal{Z}$, then $A\triangle\emptyset\subseteq N$, so $A\in\overline{\mathcal{M}}_\mu$.
\item If $A\triangle E\subseteq N$, then $A=(E\setminus N)\cup(A\cap N)$, with $E\setminus N\in\mathcal{M}$ and $A\cap N\subseteq N$.\\
Conversely, if $A=E\cup Z$ and $Z\subseteq N\in\mathcal{Z}$, then $A\triangle E\subseteq Z\subseteq N$.
\end{enumerate}
\hlt{Structural Commentary}: Completion adds all null modifications of measurable sets.\\
\hlt{Key Insight}: Equality modulo null sets is the membership criterion.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:null-ideal-and-measure-completion} \hlt{Completed Measure}\\
Let $(X,\mathcal{M},\mu)$ be a measure space, $\mathcal{Z}\defeq\{N\in\mathcal{M}:\mu(N)=0\}$,\\
and $\overline{\mathcal{M}}_\mu\defeq\{A\subseteq X:\exists E\in\mathcal{M},\ \exists N\in\mathcal{Z},\ A\triangle E\subseteq N\}$. For $A\in\overline{\mathcal{M}}_\mu$, define $\overline{\mu}(A)\defeq\mu(E)$ whenever $E\in\mathcal{M}$ and $A\triangle E\subseteq N\in\mathcal{Z}$. Then $\overline{\mu}$ is a well-defined complete measure on $\overline{\mathcal{M}}_\mu$, and $\overline{\mu}|_{\mathcal{M}}=\mu$.
\begin{proof}
If $A\triangle E\subseteq N_1$ and $A\triangle F\subseteq N_2$, with $E,F\in\mathcal{M}$ and $N_1,N_2\in\mathcal{Z}$, then $E\triangle F\subseteq N_1\cup N_2$, so $\mu(E)=\mu(F)$. Hence $\overline{\mu}$ is well defined; since $E\triangle E=\emptyset$, $\overline{\mu}|_{\mathcal{M}}=\mu$.\\
For disjoint $(A_k)\subseteq\overline{\mathcal{M}}_\mu$, choose $E_k\in\mathcal{M}$ and $N_k\in\mathcal{Z}$ with $A_k\triangle E_k\subseteq N_k$, and set $N\defeq\bigcup_kN_k$.\\
Then $E_i\cap E_j\subseteq N$ for $i\neq j$. Put $F_1\defeq E_1$ and $F_k\defeq E_k\setminus\bigcup_{j<k}E_j$.\\
Then $(F_k)$ disjoint, $A_k\triangle F_k\subseteq N$, $(\bigcup_kA_k)\triangle(\bigcup_kF_k)\subseteq N$, so $\overline{\mu}\left(\bigcup_kA_k\right)=\mu\left(\bigcup_kF_k\right)=\sum_k\mu(F_k)=\sum_k\overline{\mu}(A_k)$.\\
If $A\subseteq B\in\overline{\mathcal{M}}_\mu$ and $\overline{\mu}(B)=0$, choose $E\in\mathcal{M}$ and $N\in\mathcal{Z}$ with $B\triangle E\subseteq N$.\\
Then $\mu(E)=0$, so $A\subseteq B\subseteq E\cup N\in\mathcal{Z}$; hence $A\in\overline{\mathcal{M}}_\mu$ and $\overline{\mu}(A)=0$.\\
\hlt{Structural Commentary}: The completed measure ignores null errors.\\
\hlt{Key Insight}: Null perturbations do not change countable additivity.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:lebesgue-measurable-non-borel-sets-exist} \hlt{Lebesgue Measurable Non-Borel Sets Exist}\\
There exists $E\subseteq\R$ such that $E\in\mathcal{L}^1$ and $E\notin\mathcal{B}(\R)$; equivalently, $\mathcal{B}(\R)\subsetneq\mathcal{L}^1$.
\begin{proof}
By Lemma \ref{lem:open-sets-countable-unions-rational-boxes}, $\mathcal{B}(\R)$ is countably generated, hence $\abs{\mathcal{B}(\R)}\leq\mathfrak{c}$. Let $C$ be the Cantor set.\\
Since $\abs{C}=\mathfrak{c}$ and $m(C)=0$, Proposition \ref{prop:outer-null-sets-lebesgue-measurable} gives $\mathcal{P}(C)\subseteq\mathcal{L}^1$, so $\abs{\mathcal{L}^1}\geq2^{\mathfrak{c}}$.\\
Also $\mathcal{L}^1\subseteq\mathcal{P}(\R)$, so $\abs{\mathcal{L}^1}\leq2^{\mathfrak{c}}$; hence $\abs{\mathcal{L}^1}=2^{\mathfrak{c}}$. By Cantor's theorem, $\mathfrak{c}<2^{\mathfrak{c}}$, so $\mathcal{B}(\R)\subsetneq\mathcal{L}^1$.\\
\hlt{Structural Commentary}: Countable generation is smaller than completion by null subsets.\\
\hlt{Key Insight}: Completeness creates more measurable sets than topology can generate.
\end{proof}
\end{theorem}

\begin{definition} \label{def:set-liminf-limsup} \hlt{Limit Inferior and Superior of Sets}\\
For $(E_n)\subseteq\mathcal{P}(X)$, define $\limsup_nE_n\defeq\bigcap_{m=1}^{\infty}\bigcup_{n\geq m}E_n$ and $\liminf_nE_n\defeq\bigcup_{m=1}^{\infty}\bigcap_{n\geq m}E_n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Infinitely often}: $x\in\limsup_nE_n$ iff $\{n:x\in E_n\}$ is infinite;
\item \hlt{Eventually always}: $x\in\liminf_nE_n$ iff $\{n:x\notin E_n\}$ is finite;
\item \hlt{Set limit}: if $\limsup_nE_n=\liminf_nE_n$, write $\lim_nE_n$ for the common set.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:set-limits-basic-laws} \hlt{Basic Laws for Set Limits}\\
Let $(E_n)\subseteq\mathcal{P}(X)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\emptyset\subseteq\liminf_nE_n\subseteq\limsup_nE_n\subseteq X$.
\item $E_n\uparrow E\Longrightarrow\liminf_nE_n=\limsup_nE_n=E=\bigcup_nE_n$.
\item $E_n\downarrow E\Longrightarrow\liminf_nE_n=\limsup_nE_n=E=\bigcap_nE_n$.
\item $(E_n)\subseteq\mathcal{M}\Longrightarrow\liminf_nE_n,\limsup_nE_n\in\mathcal{M}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $\bigcap_m\bigcup_{n\geq m}E_n\subseteq\bigcup_{n\geq1}E_n\subseteq X$ and $\bigcap_{n\geq m}E_n\subseteq\bigcup_{n\geq m}E_n$ for all $m$, the inclusions follow.
\item If $E_n\uparrow E$, then $\{n:x\in E_n\}$ is nonempty iff it is cofinite, hence $\liminf_nE_n=\limsup_nE_n=\bigcup_nE_n=E$.
\item If $E_n\downarrow E$, then $\{n:x\in E_n\}$ is infinite iff it is all of $\N$, hence $\liminf_nE_n=\limsup_nE_n=\bigcap_nE_n=E$.
\item If $(E_n)\subseteq\mathcal{M}$, then $\bigcup_{n\geq m}E_n,\bigcap_{n\geq m}E_n\in\mathcal{M}$ $\forall m$; countable closure gives $\limsup_nE_n,\liminf_nE_n\in\mathcal{M}$.
\end{enumerate}
\hlt{Structural Commentary}: Set limsup means infinitely often; set liminf means eventually always.\\
\hlt{Key Insight}: Translate the definitions into pointwise membership statements.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:measure-liminf-limsup-inequalities} \hlt{Measure Bounds for Set Limits}\\
Let $(E_n)$ be measurable in a measure space $(X,\mathcal{M},\mu)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu(\liminf_nE_n)\leq\liminf_n\mu(E_n)$.
\item $\mu(\bigcup_{n=1}^{\infty}E_n)<+\infty\Longrightarrow\limsup_n\mu(E_n)\leq\mu(\limsup_nE_n)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $A_m\defeq\bigcap_{n\geq m}E_n$. Then $A_m\uparrow\liminf_nE_n$ and $A_m\subseteq E_n$ for all $n\geq m$, so $\mu(A_m)\leq\inf_{n\geq m}\mu(E_n)$.\\
By continuity from below, $\mu(\liminf_nE_n)=\lim_m\mu(A_m)\leq\sup_m\inf_{n\geq m}\mu(E_n)=\liminf_n\mu(E_n)$.
\item Set $B_m\defeq\bigcup_{n\geq m}E_n$. $B_m\downarrow\limsup_nE_n$, $\mu(B_1)<+\infty$, $E_n\subseteq B_m$ $\forall n\geq m$, so $\sup_{n\geq m}\mu(E_n)\leq\mu(B_m)$.\\
By continuity from above, $\limsup_n\mu(E_n)=\inf_m\sup_{n\geq m}\mu(E_n)\leq\lim_m\mu(B_m)=\mu(\limsup_nE_n)$.
\end{enumerate}
\hlt{Structural Commentary}: Tail intersections increase to liminf; tail unions decrease to limsup.\\
\hlt{Key Insight}: Compare each tail envelope to the corresponding tail bounds of the measures.
\end{proof}
\end{proposition}

% This subsection expands the regularity summary in Remark \ref{rem:borel-null-regular-measurable-sets}
% after the construction of Lebesgue outer measure in Subsection 5.1.

The Caratheodory definition is intrinsic to outer measure, but in $\R^p$ it has a topological shadow: measurable sets are exactly those that can be approximated by open, closed, compact, and cell-built sets up to arbitrarily small error. That bridge is also the route to the first nonmeasurable and non-Borel examples.

\begin{lemma} \label{lem:outer-approximation-by-open-gdelta} \hlt{Outer Approximation by Open and $G_\delta$ Sets}\\
For $A\subseteq\R^p$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item for every $\varepsilon>0$, there exists open $G\subseteq\R^p$ with $A\subseteq G$ and $m(G)\leq m_p^\ast(A)+\varepsilon$;
\item $m_p^\ast(A)=\inf\{m(G):A\subseteq G,\ G\subseteq\R^p\text{ open}\}$;
\item there exists $G_\delta$ set $H\supseteq A$ with $m(H)=m_p^\ast(A)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $m_p^\ast(A)=+\infty$, take $G=\R^p$.\\
If $m_p^\ast(A)<+\infty$, choose an open-cell cover $A\subseteq\bigcup_k I_k$ with $\sum_k\operatorname{vol}_p(I_k)\leq m_p^\ast(A)+\varepsilon$, and set $G\defeq\bigcup_kI_k$.\\
Then $G$ is open, $A\subseteq G$, and $m(G)\leq\sum_km(I_k)=\sum_k\operatorname{vol}_p(I_k)\leq m_p^\ast(A)+\varepsilon$.
\item By (i), $\inf\{m(G):A\subseteq G,\ G\subseteq\R^p\text{ open}\}\leq m_p^\ast(A)+\varepsilon$ for all $\varepsilon>0$; let $\varepsilon\downarrow0$.\\
Conversely, $A\subseteq G$ gives $m_p^\ast(A)\leq m_p^\ast(G)=m(G)$ for open $G$; take the infimum over $G$.
\item If $m_p^\ast(A)=+\infty$, take $H=\R^p$.\\
If $m_p^\ast(A)<+\infty$, choose open $G_n\supseteq A$ with $m(G_n)\leq m_p^\ast(A)+1/n$ and set $H\defeq\bigcap_nG_n$.\\
Then $H$ is $G_\delta$, $A\subseteq H$, and $m_p^\ast(A)\leq m(H)\leq m(G_n)\leq m_p^\ast(A)+1/n$ for all $n$, so $m(H)=m_p^\ast(A)$.
\end{enumerate}
\hlt{Structural Commentary}: Outer measure can be realised by a Borel envelope from outside.\\
\hlt{Key Insight}: Infima over cell covers can be repackaged as open-set approximation.
\end{proof}
\end{lemma}

\begin{corollary} \label{cor:null-sets-in-borel-null-sets} \hlt{Null Sets Sit Inside Borel Null Sets}\\
If $Z\subseteq\R^p$ and $m_p^\ast(Z)=0$, then there exists $B\in\mathcal{B}(\R^p)$ such that $Z\subseteq B$ and $m(B)=0$.
\begin{proof}
By Lemma \ref{lem:outer-approximation-by-open-gdelta}, choose $G_\delta$ set $B\supseteq Z$ with $m(B)=m_p^\ast(Z)=0$. Since $G_\delta\subseteq\mathcal{B}(\R^p)$, $B$ is Borel.\\
\hlt{Structural Commentary}: Null sets admit Borel envelopes of the same measure.\\
\hlt{Key Insight}: Outer approximation turns arbitrary null sets into Borel null supersets.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:open-gdelta-characterization-lebesgue-measurable} \hlt{Open and $G_\delta$ Characterizations of Measurability}\\
For $E\subseteq\R^p$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E\in\mathcal{L}^p$;
\item $\forall\varepsilon>0$, $\exists$ open $G\subseteq\R^p$ such that $E\subseteq G$ and $m_p^\ast(G\setminus E)<\varepsilon$;
\item $\exists$ $G_\delta$ set $H\subseteq\R^p$ such that $E\subseteq H$ and $m_p^\ast(H\setminus E)=0$;
\item $\exists$ $G_\delta$ set $H\subseteq\R^p$ and $Z\in\mathcal{L}^p$ such that $Z\subseteq H$, $m(Z)=0$, and $E=H\setminus Z$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$. If $m(E)<+\infty$, choose open $G\supseteq E$ with $m(G)\leq m(E)+\varepsilon/2$.\\
Since $E\in\mathcal{L}^p$, $m_p^\ast(G\setminus E)=m(G)-m(E)<\varepsilon$. If $m(E)=+\infty$, set $E_1\defeq E\cap\{x:\norm{x}_2\leq1\}$ and $E_n\defeq E\cap\{x:n-1<\norm{x}_2\leq n\}$ for $n\geq2$. For each $n$, choose open $G_n\supseteq E_n$ with $m_p^\ast(G_n\setminus E_n)<\varepsilon2^{-n}$; then $G\defeq\bigcup_nG_n$ is open, $E\subseteq G$, and $G\setminus E\subseteq\bigcup_n(G_n\setminus E_n)$, so $m_p^\ast(G\setminus E)<\varepsilon$.
\item[$(ii)$]$\Rightarrow (iii)$. Choose open $G_n\supseteq E$ with $m_p^\ast(G_n\setminus E)<1/n$, and set $H\defeq\bigcap_nG_n$.\\
Then $H$ is $G_\delta$, $E\subseteq H$, and $H\setminus E\subseteq G_n\setminus E$ for all $n$; hence $m_p^\ast(H\setminus E)=0$.
\item[$(iii)$]$\Rightarrow (i)$. Since $H$ is $G_\delta$, $H\in\mathcal{B}(\R^p)\subseteq\mathcal{L}^p$. Also $m_p^\ast(H\setminus E)=0$ implies $H\setminus E\in\mathcal{L}^p$ by Proposition \ref{prop:outer-null-sets-lebesgue-measurable}.\\ Thus $E=H\setminus(H\setminus E)\in\mathcal{L}^p$.
\item[$(iii)$]$\Leftrightarrow (iv)$. If (iii) holds, set $Z\defeq H\setminus E$; then $Z\subseteq H$, $m(Z)=0$, and $E=H\setminus Z$.\\
Conversely, if (iv) holds, then $E\subseteq H$ and $H\setminus E=Z$, so $m_p^\ast(H\setminus E)=0$.
\end{enumerate}
\hlt{Structural Commentary}: Lebesgue measurable sets are $G_\delta$ sets with null material removed.\\
\hlt{Key Insight}: Small open excess becomes a null $G_\delta$ excess after countably many approximations.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:open-outer-regularity-lebesgue-measure} \hlt{Open Outer Regularity for Measurable Sets}\\
If $E\in\mathcal{L}^p$, then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\forall\varepsilon>0$, $\exists$ open $G\supseteq E$ such that $m(G)\leq m(E)+\varepsilon$;
\item $m(E)=\inf\{m(G):E\subseteq G,\ G\subseteq\R^p\text{ open}\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:open-gdelta-characterization-lebesgue-measurable}, choose open $G\supseteq E$ with $m_p^\ast(G\setminus E)<\varepsilon$.\\
If $m(E)<+\infty$, then $m(G)=m(E)+m(G\setminus E)\leq m(E)+\varepsilon$; if $m(E)=+\infty$, the inequality is trivial.
\item From (i), $\inf_Gm(G)\leq m(E)$. Conversely, $E\subseteq G$ gives $m(E)\leq m(G)$ by monotonicity; take $\inf_G$.
\end{enumerate}
\hlt{Structural Commentary}: Measurable sets are measured from outside by open supersets.\\
\hlt{Key Insight}: Open excess controls the outer regular infimum.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:closed-fsigma-characterization-lebesgue-measurable} \hlt{Closed and $F_\sigma$ Characterizations of Measurability}\\
For $E\subseteq\R^p$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E\in\mathcal{L}^p$;
\item $\forall\varepsilon>0$, $\exists$ closed $F\subseteq E$ such that $m_p^\ast(E\setminus F)<\varepsilon$;
\item $\exists$ $F_\sigma$ set $K\subseteq E$ such that $m_p^\ast(E\setminus K)=0$;
\item $\exists$ $F_\sigma$ set $K\subseteq E$ and $Z\in\mathcal{L}^p$ such that $Z\subseteq E$, $m(Z)=0$, and $E=K\cup Z$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$. Since $E^c\in\mathcal{L}^p$, Theorem \ref{thm:open-gdelta-characterization-lebesgue-measurable} gives open $G\supseteq E^c$ with $m_p^\ast(G\setminus E^c)<\varepsilon$. Set $F\defeq G^c$.\\
Then $F$ is closed, $F\subseteq E$, and $E\setminus F=G\setminus E^c$, so $m_p^\ast(E\setminus F)<\varepsilon$.
\item[$(ii)$]$\Rightarrow (iii)$. Choose closed $F_n\subseteq E$ with $m_p^\ast(E\setminus F_n)<1/n$, and set $K\defeq\bigcup_nF_n$.\\
Then $K$ is $F_\sigma$, $K\subseteq E$, and $E\setminus K\subseteq E\setminus F_n$ for all $n$; hence $m_p^\ast(E\setminus K)=0$.
\item[$(iii)$]$\Rightarrow (i)$. Since $K$ is $F_\sigma$, $K\in\mathcal{B}(\R^p)\subseteq\mathcal{L}^p$. Also $m_p^\ast(E\setminus K)=0$ implies $E\setminus K\in\mathcal{L}^p$ by Proposition \ref{prop:outer-null-sets-lebesgue-measurable}.\\Thus $E=K\cup(E\setminus K)\in\mathcal{L}^p$.
\item[$(iii)$]$\Leftrightarrow (iv)$. If (iii) holds, set $Z\defeq E\setminus K$; then $Z\subseteq E$, $m(Z)=0$, and $E=K\cup Z$.\\
Conversely, if (iv) holds, then $E\setminus K\subseteq Z$, so $m_p^\ast(E\setminus K)=0$.
\end{enumerate}
\hlt{Structural Commentary}: Inside approximation is the complement of outside approximation.\\
\hlt{Key Insight}: Measurable sets are also $F_\sigma$ sets with null material added.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:closed-inner-regularity-lebesgue-measure} \hlt{Closed Inner Regularity}\\
If $E\in\mathcal{L}^p$, then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\forall\varepsilon>0$, $\exists$ closed $F\subseteq E$ such that $m(E)\leq m(F)+\varepsilon$;
\item $m(E)=\sup\{m(F):F\subseteq E,\ F\subseteq\R^p\text{ closed}\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:closed-fsigma-characterization-lebesgue-measurable}, choose closed $F\subseteq E$ with $m_p^\ast(E\setminus F)<\varepsilon$.\\
Since $F\in\mathcal{L}^p$ and $E=F\sqcup(E\setminus F)$, $m(E)=m(F)+m(E\setminus F)\leq m(F)+\varepsilon$.
\item From (i), $m(E)\leq\sup_Fm(F)$. Conversely, $F\subseteq E$ gives $m(F)\leq m(E)$; take $\sup_F$.
\end{enumerate}
\hlt{Structural Commentary}: Measurable sets are measured from inside by closed subsets.\\
\hlt{Key Insight}: Small discarded measure controls the inner regular supremum.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:compact-characterization-finite-outer-measure} \hlt{Compact Approximation at Finite Outer Measure}\\
Let $E\subseteq\R^p$ and $m_p^\ast(E)<+\infty$. Then $E\in\mathcal{L}^p$ $\Longleftrightarrow$ $\forall\varepsilon>0,\ \exists C\subseteq$ compact with $m_p^\ast(E\setminus C)<\varepsilon$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Set $E_N\defeq E\cap\{x:\norm{x}_2\leq N\}$. Then $E_N\uparrow E$ and $m(E)<+\infty$, so choose $N$ with $m(E\setminus E_N)<\varepsilon/2$. By Theorem \ref{thm:closed-fsigma-characterization-lebesgue-measurable}, choose closed $C\subseteq E_N$ with $m(E_N\setminus C)<\varepsilon/2$.\\
Then $C$ is closed and bounded, hence compact, and $m_p^\ast(E\setminus C)\leq m(E\setminus E_N)+m(E_N\setminus C)<\varepsilon$.
\item[$(\Leftarrow)$] Choose compact $C_n\subseteq E$ with $m_p^\ast(E\setminus C_n)<1/n$, and set $K\defeq\bigcup_nC_n$.\\
Then $K$ is $F_\sigma$, $K\subseteq E$, and $E\setminus K\subseteq E\setminus C_n$ $\forall n$, so $m_p^\ast(E\setminus K)=0$. By Theorem \ref{thm:closed-fsigma-characterization-lebesgue-measurable}, $E\in\mathcal{L}^p$.
\end{enumerate}
\hlt{Structural Commentary}: Finite measure permits bounded truncation before inner closure.\\
\hlt{Key Insight}: Compact approximation is closed approximation plus a finite-measure tail estimate.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:finite-cell-symmetric-difference-approximation} \hlt{Finite Cell Approximation in Symmetric Difference}\\
If $E\in\mathcal{L}^p$, $m(E)<+\infty$, then $\forall \varepsilon>0$ $\exists$ bounded open cells $I_1,\ldots,I_N$ s.t. $K_N\defeq\bigcup_{j=1}^{N}I_j$; $m(E\triangle K_N)<\varepsilon$.
\begin{proof}
Choose bounded open cells $(J_k)$ with $E\subseteq G\defeq\bigcup_kJ_k$ and $\sum_k\operatorname{vol}_p(J_k)<m(E)+\varepsilon/2$.\\
Since $G\in\mathcal{L}^p$ and $E\subseteq G$, $m(G\setminus E)=m(G)-m(E)\leq\sum_k\operatorname{vol}_p(J_k)-m(E)<\varepsilon/2$..\\
By Theorem \ref{thm:compact-characterization-finite-outer-measure}, choose compact $C\subseteq E$ with $m(E\setminus C)<\varepsilon/2$.\\
Since $C\subseteq\bigcup_kJ_k$, compactness gives $I_1,\ldots,I_N\in\{J_k\}$ with $C\subseteq K_N\defeq\bigcup_{j=1}^{N}I_j\subseteq G$.\\
Hence $E\triangle K_N=(E\setminus K_N)\cup(K_N\setminus E) \subseteq (E\setminus C)\cup(G\setminus E)$. Thus $m(E\triangle K_N)\leq m(E\setminus C)+m(G\setminus E)<\varepsilon$.\\
\hlt{Structural Commentary}: Compactness turns a countable open cover into a finite geometric approximant.\\
\hlt{Key Insight}: Approximate from outside by cells and from inside by a compact core.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:cell-approximation-variant-choices} \hlt{Cell Approximation Variant Choices}\\
If $E\in\mathcal{L}^p$, $m(E)<+\infty$, and $\varepsilon>0$, then there exist bounded cells $I_1,\ldots,I_N$ such that $K\defeq\bigcup_{j=1}^{N}I_j$ satisfies $m(E\triangle K)<\varepsilon$, with the cells chosen as any one of:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $I_i\cap I_j=\emptyset$ for $i\neq j$, and each $I_j$ is open;
\item each $I_j$ is closed;
\item each $I_j$ is half-open of a fixed orientation.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose compact $C\subseteq E$ and open $G\supseteq E$ with $m(E\setminus C)+m(G\setminus E)<\varepsilon$.\\
Cover $C$ by bounded open cells $J_1,\ldots,J_M$ with $\overline{J_j}\subseteq G$, and set $U\defeq\bigcup_{j=1}^{M}J_j$.\\
Then $C\subseteq U\subseteq G$, so $m(E\triangle U)\leq m(E\setminus C)+m(G\setminus E)<\varepsilon$.
\item Let $\mathcal{G}$ be finite grid from coordinate endpoints of $J_1,\ldots,J_M$. For $\mathcal{G}'\subseteq\mathcal{G}$ selecting $U$, $\bigsqcup_{Q\in\mathcal{G}'}Q$ gives open or fixed-orientation half-open cells, differs from $U$ only on finitely many faces, hence by a null set.
\item Let $K_c\defeq\bigcup_{j=1}^{M}\overline{J_j}$. Then $K_c$ is a finite union of closed cells and $C\subseteq K_c\subseteq G$.\\
Hence $m(E\triangle K_c)\leq m(E\setminus C)+m(G\setminus E)<\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Cell format changes only boundary-null pieces.\\
\hlt{Key Insight}: A compact core inside an open envelope permits finite cell variants.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:outer-measure-additivity-one-measurable-set} \hlt{Outer Measure Additivity with One Measurable Set}\\
Let $E\in\mathcal{L}^p$ and $F\subseteq\R^p$. Then $m_p^\ast(E\cup F)+m_p^\ast(E\cap F)=m(E)+m_p^\ast(F)$.
\begin{proof} Since $E\in\mathcal{L}^p$, Caratheodory splitting: $m_p^\ast(E\cup F)=m(E)+m_p^\ast(F\setminus E)$; $m_p^\ast(F)=m_p^\ast(E\cap F)+m_p^\ast(F\setminus E)$.\\
Hence $m_p^\ast(E\cup F)+m_p^\ast(E\cap F)=m(E)+m_p^\ast(F)$\\
\hlt{Structural Commentary}: One measurable cut splits arbitrary sets exactly.\\
\hlt{Key Insight}: Split both $E\cup F$ and $F$ along the same measurable cut $E$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:outer-measure-disjoint-union-and-difference} \hlt{Disjoint Additivity and Difference}\\
Let $E\in\mathcal{L}^p$ and $F\subseteq\R^p$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E\cap F=\emptyset\Longrightarrow m_p^\ast(E\cup F)=m(E)+m_p^\ast(F)$.
\item $E\subseteq F,\ m(E)<+\infty\Longrightarrow m_p^\ast(F\setminus E)=m_p^\ast(F)-m(E)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:outer-measure-additivity-one-measurable-set}, $m_p^\ast(E\cup F)+m_p^\ast(E\cap F)=m(E)+m_p^\ast(F)$.\\
Since $E\cap F=\emptyset$ and $m_p^\ast(\emptyset)=0$, $m_p^\ast(E\cup F)=m(E)+m_p^\ast(F)$.
\item Since $E\subseteq F$, Caratheodory splitting gives $m_p^\ast(F)=m(E)+m_p^\ast(F\setminus E)$; subtract $m(E)<+\infty$.
\end{enumerate}
\hlt{Structural Commentary}: Exact splitting yields disjoint sums and finite differences.\\
\hlt{Key Insight}: Disjointness removes the overlap; containment isolates the remainder.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:finite-outer-measure-inner-measurable-subset-test} \hlt{Finite Outer Measure Inner Test}\\
If $m_p^\ast(E)<+\infty$, then $E\in\mathcal{L}^p$ $\Longleftrightarrow$ $\exists B\in\mathcal{L}^p,\ B\subseteq E,\ m(B)=m_p^\ast(E)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Set $B\defeq E$. Then $B\in\mathcal{L}^p$, $B\subseteq E$, and $m(B)=m(E)=m_p^\ast(E)$.
\item[$(\Leftarrow)$] Assume $B\in\mathcal{L}^p$, $B\subseteq E$, and $m(B)=m_p^\ast(E)$.\\
Since $m(B)<+\infty$, the difference formula gives $m_p^\ast(E\setminus B)=m_p^\ast(E)-m(B)=0$.\\
Thus $E\setminus B\in\mathcal{L}^p$, and $E=B\cup(E\setminus B)\in\mathcal{L}^p$.
\end{enumerate}
\hlt{Structural Commentary}: Exact inner mass leaves only a null remainder.\\
\hlt{Key Insight}: Equality of outer mass converts containment into measurability.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:single-container-caratheodory-test} \hlt{Single Finite Container Test}\\
Let $A\in\mathcal{L}^p$, $m(A)<+\infty$, and $E\subseteq A$. Then $E\in\mathcal{L}^p$ $\Longleftrightarrow$ $m(A)=m_p^\ast(E)+m_p^\ast(A\setminus E)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Since $E\in\mathcal{L}^p$ and $E\subseteq A$,\\
Caratheodory splitting of $A$ by $E$ gives $m(A)=m(E)+m(A\setminus E)=m_p^\ast(E)+m_p^\ast(A\setminus E)$.
\item[$(\Leftarrow)$] Choose $G_\delta$ set $H\supseteq A\setminus E$ with $m(H)=m_p^\ast(A\setminus E)$. Since $A\setminus E\subseteq A\cap H\subseteq H$, $m(A\cap H)=m_p^\ast(A\setminus E)$.\\
Set $B\defeq A\setminus H$. $B\in\mathcal{L}^p$, $B\subseteq E$; $m(B)=m(A)-m(A\cap H)=m_p^\ast(E)$. By Theorem \ref{thm:finite-outer-measure-inner-measurable-subset-test}, $E\in\mathcal{L}^p$.
\end{enumerate}
\hlt{Structural Commentary}: Inside a finite measurable container, Caratheodory only needs to be tested once.\\
\hlt{Key Insight}: Outer-approximate the complement; the remainder has full outer mass.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:bounded-inner-measure-test} \hlt{Bounded Inner Measure Test}\\
For bounded cell $Q\subseteq\R^p$ and $E\subseteq Q$, define $m_{*,Q}(E)\defeq m(Q)-m_p^\ast(Q\setminus E)$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E\in\mathcal{L}^p\Longleftrightarrow m_{*,Q}(E)=m_p^\ast(E)$;
\item for arbitrary $E\subseteq\R^p$, $E\in\mathcal{L}^p\Longleftrightarrow E\cap[-n,n]^p\in\mathcal{L}^p$ for all $n\in\N$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$(\Rightarrow)$ If $E\in\mathcal{L}^p$, then $m(Q)=m(E)+m_p^\ast(Q\setminus E)$, hence $m_{*,Q}(E)=m_p^\ast(E)$.
\item[$(i)$]$(\Leftarrow)$ If $m_{*,Q}(E)=m_p^\ast(E)$, then $m(Q)=m_p^\ast(E)+m_p^\ast(Q\setminus E)$, so $E\in\mathcal{L}^p$ by the finite-container test.
\item[$(ii)$]$(\Rightarrow)$ If $E\in\mathcal{L}^p$, then $E\cap[-n,n]^p\in\mathcal{L}^p$ for all $n$, since $[-n,n]^p$ is Borel.
\item[$(ii)$]$(\Leftarrow)$ If $E\cap[-n,n]^p\in\mathcal{L}^p$ for all $n$, then $E=\bigcup_{n=1}^{\infty}(E\cap[-n,n]^p)\in\mathcal{L}^p$.
\end{enumerate}
\hlt{Structural Commentary}: Bounded inner measure tests one finite container.\\
\hlt{Key Insight}: Equality of inner and outer measure is a bounded version of Caratheodory.
\end{proof}
\end{corollary}

\begin{definition} \label{def:difference-set-and-vitali-selector} \hlt{Difference Sets and Vitali Selectors}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Difference set}: for $A\subseteq\R^p$, $A-A\defeq\{x-y:x,y\in A\}$; $A\subseteq B\Longrightarrow A-A\subseteq B-B$.
\item \hlt{Rational equivalence}: $x\sim_{\Q^p}y\Longleftrightarrow x-y\in\Q^p$; equivalently, $[x]_{\Q^p}=x+\Q^p$.
\item \hlt{Vitali selector}: choose $\mathcal{V}\subseteq\R^p$ such that $\abs{\mathcal{V}\cap(x+\Q^p)}=1$ for every $x\in\R^p$.
\item \hlt{Rational translate}: for $q\in\Q^p$, $\mathcal{V}_q\defeq q+\mathcal{V}$.
\end{enumerate}
\end{definition}

\begin{lemma} \label{lem:steinhaus-compact-positive-measure} \hlt{Steinhaus Lemma for Compact Sets}\\
If $K\subseteq\R^p$ is compact and $m(K)>0$, then $\exists\delta>0$ such that $B(0,\delta)\subseteq K-K$.
\begin{proof}
Choose open $G\supseteq K$ with $m(G)<2m(K)$. Since $K$ is compact and $G^c$ is closed, set $\delta\defeq\operatorname{dist}(K,G^c)>0$. If $\norm{x}_2<\delta$, then $x+K\subseteq G$. If $(x+K)\cap K=\emptyset$, translation invariance and additivity give\\
$2m(K)=m(x+K)+m(K)=m((x+K)\cup K)\leq m(G)<2m(K)$, a contradiction.\\
Hence $(x+K)\cap K\neq\emptyset$, so $x=k_1-k_2\in K-K$. Thus $B(0,\delta)\subseteq K-K$.\\
\hlt{Structural Commentary}: A positive compact set overlaps all sufficiently small translates of itself.\\
\hlt{Key Insight}: Disjoint small translates cannot fit inside a tight open envelope.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:steinhaus-positive-measure-difference-neighborhood} \hlt{Steinhaus Theorem}\\
If $E\in\mathcal{L}^p$ and $m(E)>0$, then $\exists\delta>0$ such that $B(0,\delta)\subseteq E-E$.
\begin{proof}
Set $E_n\defeq E\cap B(0,n)$. Since $E_n\uparrow E$ and $m(E)>0$, choose $n$ with $0<m(E_n)<+\infty$.\\
By compact inner approximation, choose compact $K\subseteq E_n$ with $m(K)>0$.\\
Lemma \ref{lem:steinhaus-compact-positive-measure} gives $\delta>0$ with $B(0,\delta)\subseteq K-K\subseteq E-E$.\\
\hlt{Structural Commentary}: Positive measurable sets contain positive compact cores.\\
\hlt{Key Insight}: Difference-set thickness follows from compact inner regularity.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:vitali-translates-partition} \hlt{Rational Translates of a Vitali Selector Partition $\R^p$}\\
Let $(q_i)_{i\geq1}$ enumerate $\Q^p$.\\
Then $\R^p=\bigsqcup_{i=1}^{\infty}\mathcal{V}_{q_i}$; equivalently, each $x\in\R^p$ has a unique representation $x=q_i+v$ with $i\in\N$ and $v\in\mathcal{V}$.
\begin{proof}
For $x\in\R^p$, choose the unique $v\in\mathcal{V}\cap(x+\Q^p)$; then $x-v=q_i$ for some $i$, so $x\in\mathcal{V}_{q_i}$. If $q+v=q'+v'$ with $q,q'\in\Q^p$ and $v,v'\in\mathcal{V}$, then $v-v'=q'-q\in\Q^p$, hence $v=v'$ by selector uniqueness and then $q=q'$.\\
\hlt{Structural Commentary}: Rational translates enumerate the equivalence classes selected by $\mathcal{V}$.\\
\hlt{Key Insight}: One representative per class makes the rational translates disjoint and exhaustive.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:vitali-selector-nonmeasurable} \hlt{Vitali Selectors Are Nonmeasurable}\\
If $\mathcal{V}\subseteq\R^p$ is a Vitali selector, then $\mathcal{V}\notin\mathcal{L}^p$.
\begin{proof}
Assume $\mathcal{V}\in\mathcal{L}^p$.
If $m(\mathcal{V})>0$, Steinhaus gives $\delta>0$ with $B(0,\delta)\subseteq\mathcal{V}-\mathcal{V}$. Choose $0\neq q\in\Q^p\cap B(0,\delta)$; then $q=v_1-v_2$ for $v_1,v_2\in\mathcal{V}$, so $v_1\sim_{\Q^p}v_2$, forcing $v_1=v_2$ and $q=0$, a contradiction.\\
If $m(\mathcal{V})=0$, then each $\mathcal{V}_{q_i}$ is null by translation invariance, and $\R^p=\bigsqcup_i\mathcal{V}_{q_i}$ would imply every bounded cell has measure $0$, contradicting positive cell volume.\\
\hlt{Structural Commentary}: Positive and zero measure are both impossible.\\
\hlt{Key Insight}: Steinhaus forbids positive measure; countable translated partition forbids zero measure.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:positive-outer-measure-contains-nonmeasurable-subset} \hlt{Nonmeasurable Subsets Are Everywhere}\\
If $E\subseteq\R^p$ and $m_p^\ast(E)>0$, then $\exists N\subseteq E$ with $N\notin\mathcal{L}^p$.
\begin{proof}
Let $(q_i)$ enumerate $\Q^p$ and set $E_i\defeq E\cap\mathcal{V}_{q_i}$. If $E_i\in\mathcal{L}^p$ and $m(E_i)>0$, then $E_i-E_i$ contains a ball about $0$, so $0\neq q\in\Q^p\cap(E_i-E_i)$. But $E_i\subseteq q_i+\mathcal{V}$, so $q=(q_i+v_1)-(q_i+v_2)=v_1-v_2$ with $v_1,v_2\in\mathcal{V}$, contradicting selector uniqueness. Hence each measurable $E_i$ is null. If every $E_i$ were measurable, then $E=\bigcup_iE_i$ would satisfy $m_p^\ast(E)=0$, contradiction. Thus some $E_i\subseteq E$ is nonmeasurable.\\
\hlt{Structural Commentary}: Any positive outer-measure set meets the Vitali partition in a nonmeasurable piece.\\
\hlt{Key Insight}: Vitali pieces cannot be measurable and positive.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:nonadditive-decompositions-from-nonmeasurable-sets} \hlt{Nonadditive Decompositions}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $E\in\mathcal{L}^p$ and $0<m(E)<+\infty$, then $\exists B,C\notin\mathcal{L}^p$ such that $E=B\sqcup C$ and $m(E)<m_p^\ast(B)+m_p^\ast(C)$.
\item If $B\notin\mathcal{L}^p$, $m_p^\ast(B)<+\infty$, and $H$ is $G_\delta$ with $B\subseteq H$, $m(H)=m_p^\ast(B)$,\\
then $C\defeq H\setminus B\notin\mathcal{L}^p$ and $m(H)=m_p^\ast(B\cup C)<m_p^\ast(B)+m_p^\ast(C)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Pick $B\subseteq E$, $B\notin\mathcal{L}^p$, and put $C\defeq E\setminus B$. Then $C\notin\mathcal{L}^p$, else $B=E\setminus C\in\mathcal{L}^p$.\\
Subadditivity gives $\leq$; equality would force $B\in\mathcal{L}^p$ by the finite-container test in $E$.
\item Since $H\in\mathcal{L}^p$, $C\in\mathcal{L}^p$ would give $B=H\setminus C\in\mathcal{L}^p$, contradiction. Also $m_p^\ast(C)>0$, else $C\in\mathcal{L}^p$.\\
Thus $B\cup C=H$ and $m(H)=m_p^\ast(B)$ give $m_p^\ast(B\cup C)=m(H)<m_p^\ast(B)+m_p^\ast(C)$.
\end{enumerate}
\hlt{Structural Commentary}: Nonmeasurability appears exactly as strict failure of additive splitting.\\
\hlt{Key Insight}: Equality in a finite measurable container is a measurability test.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:ubiquitous-nonmeasurable-set} \hlt{A Nonmeasurable Set with Only Null Measurable Pieces}\\
There exists $\mathcal{U}\subseteq\R^p$ with $\mathcal{U}\notin\mathcal{L}^p$ such that, for every $F\in\mathcal{L}^p$,
\[
\begin{gathered}
F\subseteq\mathcal{U}\Longrightarrow m(F)=0,\\
F\subseteq\mathcal{U}^c\Longrightarrow m(F)=0.
\end{gathered}
\]
\begin{proof}
Let $e\defeq(1,\ldots,1)\in\R^p$, $G\defeq\Q^p+\sqrt{2}\,\Z e$, $G_0\defeq\Q^p+(2\sqrt{2})\,\Z e$, and $G_1\defeq\sqrt{2}\,e+G_0$.
Then $G=G_0\sqcup G_1$, and $G_0,G_1$ are dense. Choose $\mathcal{E}\subseteq\R^p$ with one representative from each class of $x-y\in G$, and set $\mathcal{U}\defeq G_0+\mathcal{E}$.\\
Unique representatives give $\R^p=(G_0+\mathcal{E})\sqcup(G_1+\mathcal{E})$, hence $\mathcal{U}^c=G_1+\mathcal{E}=\sqrt{2}\,e+\mathcal{U}$.\\
Let $F\in\mathcal{L}^p$, $F\subseteq\mathcal{U}$, and $m(F)>0$. Then $B(0,r)\subseteq F-F$ for some $r>0$; choose $g_1\in G_1\cap B(0,r)$.\\
Since $F\subseteq G_0+\mathcal{E}$, $g_1=(g_0+s_1)-(g_0'+s_2)$ for some $g_0,g_0'\in G_0$ and $s_1,s_2\in\mathcal{E}$.\\
Thus $s_1-s_2=g_1-g_0+g_0'\in G$, so $s_1=s_2$ and $g_1=g_0-g_0'\in G_0$, impossible.\\
Hence every measurable $F\subseteq\mathcal{U}$ is null; translation gives the same for $\mathcal{U}^c$.\\
If $\mathcal{U}\in\mathcal{L}^p$, then $\mathcal{U}$ and $\mathcal{U}^c$ are null, so $m([0,1]^p)=0$, contradiction.\\
\hlt{Structural Commentary}: This construction partitions representatives using two dense cosets.\\
\hlt{Key Insight}: Positive measurable pieces force forbidden coset differences.
\end{proof}
\end{theorem}
