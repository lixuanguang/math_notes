\subsection{Littlewood and Lusin Approximation}

The measurable world is often controlled by ordinary topological objects after discarding a small exceptional set. In the real-line finite-measure setting, this principle has three recurring forms: measurable sets are nearly elementary, measurable functions are nearly continuous, and a.e. convergence is nearly uniform.

\begin{principle} \label{pr:littlewood-three-principles-finite-measure-real-line} \hlt{Littlewood's Three Principles, Finite-Measure Form}\\
For subsets of the real line with finite measure:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item measurable sets are nearly finite unions of intervals; see Corollary \ref{cor:real-line-finite-open-interval-approximation};
\item measurable functions are nearly continuous; this is Lusin's theorem below;
\item a.e. pointwise convergence is nearly uniform; see Egoroff's theorem, Theorem \ref{thm:egoroff-finite-measure}.
\end{enumerate}
\hlt{Key Insight}: The word ``nearly'' means after changing or discarding a set of arbitrarily small measure.
\end{principle}

\begin{definition} \label{def:almost-uniform-convergence} \hlt{Almost Uniform Convergence}\\
Let $(f_n)$ be measurable functions on $X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f_n\to f$ \hlt{almost uniformly} if $\forall\delta>0$ $\exists E_\delta\in\mathcal{M}$ with $\mu(E_\delta)<\delta$ and $f_n\rightrightarrows f$ on $X\setminus E_\delta$.
\item $(f_n)$ is \hlt{almost uniformly Cauchy} if $\forall\delta>0$ $\exists E_\delta\in\mathcal{M}$ with $\mu(E_\delta)<\delta$ s.t. $(f_n)$ uniformly Cauchy on $X\setminus E_\delta$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:egoroff-finite-measure} \hlt{Egoroff's Theorem}\\
If $\mu(X)<+\infty$, $f_n\to f$ a.e., with all functions finite-valued off null set, then $f_n\to f$ almost uniformly.
\begin{proof}
Absorb the null exceptional set into every later $E_\delta$, and assume $\forall x\in X,\ f_n(x)\to f(x)$.\\
For $m,n\in\N$, set $E_n(m)\defeq\bigcup\limits_{k=n}^{\infty}\{\abs{f_k-f}\geq\frac{1}{m}\}$. Then $E_n(m)\downarrow\emptyset$, so $\mu(E_n(m))\to0$ by Corollary \ref{cor:measure-continuity-up-down}.\\
Given $\delta>0$, choose $k_m$ with $\mu(E_{k_m}(m))<\delta2^{-m}$ and set $E_\delta\defeq\bigcup\limits_{m\geq1}E_{k_m}(m)$.\\
Then $\mu(E_\delta)<\delta$, and $x\notin E_\delta$, $k\geq k_m$ imply $\abs{f_k(x)-f(x)}<\frac{1}{m}$; hence $f_n\rightrightarrows f$ on $X\setminus E_\delta$.\\
\hlt{Structural Commentary}: Finite measure turns decreasing bad sets with empty intersection into small bad sets.\\
\hlt{Key Insight}: A.e. convergence is almost uniform on finite measure spaces.
\end{proof}
\end{theorem}

\begin{example} \label{ex:egoroff-needs-finite-valued-limit} \hlt{Egoroff Needs a Finite-Valued Limit}\\
On $([0,1],\lambda)$, set $f_n\defeq n\mathbf{1}_{[0,1]}$. Then $\forall x,\ f_n(x)\to+\infty$, so the pointwise limit is not finite-valued. If $f_n\to f$ almost uniformly for finite-valued measurable $f$, then for some $E$ with $\lambda(E)<\frac{1}{2}$, $f_n\rightrightarrows f$ on $[0,1]\setminus E$; this nonempty set has $f_n(x)=n\to+\infty$, contradicting finite-valued pointwise convergence.
\end{example}

\begin{lemma} \label{lem:real-line-continuous-extension-from-closed-set} \hlt{Continuous Extension from a Closed Subset of $\R$}\\
Let $F\subseteq\R$ be closed and let $h:F\to\R$ be continuous. Then $\exists g\in C(\R)$ such that $g|_F=h$.
\begin{proof} Cases $F=\emptyset$ and $F=\R$ are immediate. Otherwise $\R\setminus F=\bigsqcup\limits_\alpha I_\alpha$, set $g=h$ on $F$; on each component:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $I_\alpha=(a,b)$, set $g(x)\defeq\frac{b-x}{b-a}h(a)+\frac{x-a}{b-a}h(b)$.
\item If $I_\alpha=(-\infty,b)$, set $g(x)\defeq h(b)$.
\item If $I_\alpha=(a,+\infty)$, set $g(x)\defeq h(a)$.
\end{enumerate}
The formulas give continuity off $F$; at $x\in F$, adjacent finite endpoints converge to $x$.\\
Hence continuity of $h$ gives $g(y)\to h(x)$ as $y\to x$. Thus $g\in C(\R)$ and $g|_F=h$.\\
\hlt{Structural Commentary}: Fill complementary intervals.\\
\hlt{Key Insight}: Interpolate between boundary values.
\end{proof}
\end{lemma}

\begin{proposition} \label{prop:simple-functions-continuous-on-large-closed-set} \hlt{Simple Functions Are Nearly Continuous}\\
Let $E\in\mathcal{L}^1$ with $m(E)<+\infty$, and $s:E\to\R$ measurable simple. $\forall \varepsilon>0$, $\exists F\subseteq E$ and $g:\R\to\R$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $F$ is closed.
\item $g\in C(\R)$ and $g=s$ on $F$.
\item $m(E\setminus F)<\varepsilon$.
\end{enumerate}
\begin{proof} If $E=\emptyset$, take $F=\emptyset$ and $g=0$. Otherwise $s=\sum\limits_{i=1}^{N}a_i\mathbf{1}_{E_i}$ with disjoint measurable $E_i$ and $E=\bigsqcup\limits_iE_i$. Choose closed $F_i\subseteq E_i$ with $m(E_i\setminus F_i)<\frac{\varepsilon}{N}$, set $F\defeq\bigcup\limits_iF_i$, and define $h:F\to\R$ by $h|_{F_i}\defeq a_i$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Each $F_i$ is closed and $N<+\infty$, so $F$ is closed; also $F\subseteq E$.
\item For $x\in F_i$, the closed set $\bigcup\limits_{j\neq i}F_j$ misses $x$, so $h$ is locally constant at $x$ in $F$; hence $h\in C(F)$.\\
$\exists g\in C(\R)$ with $g|_F=h=s|_F$ by Lemma \ref{lem:real-line-continuous-extension-from-closed-set}.
\item $m(E\setminus F)\leq\sum\limits_i m(E_i\setminus F_i)<\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Closed cores isolate simple levels.\\
\hlt{Key Insight}: Approximate each level set from inside, then extend.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:lusin-theorem-finite-measure-real-line} \hlt{Lusin's Theorem}\\
Let $E\in\mathcal{L}^1$ with $m(E)<+\infty$, $f:E\to\R$ measurable. $\forall \varepsilon>0$, $\exists F\subseteq E$ and $g:\R\to\R$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $F$ is closed.
\item $g\in C(\R)$ and $g=f$ on $F$.
\item $m(E\setminus F)<\varepsilon$.
\end{enumerate}
\begin{proof} Take simple $s_n:E\to\R$ with $s_n\to f$ pointwise.\\
Pick $A\subseteq E$ with $s_n\to f$ uniformly on $A$ and $m(E\setminus A)<\frac{\varepsilon}{4}$, and closed $F_0\subseteq A$ with $m(A\setminus F_0)<\frac{\varepsilon}{4}$.\\
For $n\geq1$, choose closed $F_n\subseteq E$ with $s_n|_{F_n}\in C(F_n)$ and $m(E\setminus F_n)<\varepsilon2^{-(n+1)}$. Set $F\defeq\bigcap\limits_{n\geq0}F_n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $F$ is an intersection of closed sets and $F\subseteq E$.
\item Since $F\subseteq F_0\subseteq A$, the continuous functions $s_n|_F$ converge uniformly to $f|_F$, so $f|_F\in C(F)$.\\
Since $F$ is closed and $f|_F\in C(F)$, Lemma \ref{lem:real-line-continuous-extension-from-closed-set} gives $\exists g\in C(\R)$ with $g|_F=f|_F$.
\item $m(E\setminus F)\leq m(E\setminus F_0)+\sum\limits_{n\geq1}m(E\setminus F_n)<\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Discard a small set to gain continuity.\\
\hlt{Key Insight}: Simple cores plus Egoroff give uniform continuous limits.
\end{proof}
\end{theorem}
