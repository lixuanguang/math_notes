\subsection{Measure Duality and Riesz Representation}

On a compact Hausdorff space, Riesz--Markov turns positive functionals on $C(K)$ into Radon measures. The remaining step is order-theoretic: every bounded functional is the difference of two positive functionals, exactly as a signed measure is the difference of its Jordan variations.

\begin{proposition} \label{prop:positive-functionals-on-cx-are-continuous} \hlt{Positive Functionals on $C(K)$ Are Continuous}\\
Let $K$ be compact Hausdorff, $C(K)=C(K;\R)$, $\norm{\cdot}_\infty$, $L:C(K)\to\R$ linear with $L(C(K)_+)\subseteq[0,+\infty)$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $L\in C(K)^*$.
\item $\normop{L}=L(\mathbf{1}_K)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $f\in C(K)$, $-\norm{f}_\infty\mathbf{1}_K\leq f\leq\norm{f}_\infty\mathbf{1}_K$;$-L(\mathbf{1}_K)\norm{f}_\infty\leq L(f)\leq L(\mathbf{1}_K)\norm{f}_\infty$; $\abs{L(f)}\leq L(\mathbf{1}_K)\norm{f}_\infty$.
\item By (i), $\normop{L}\leq L(\mathbf{1}_K)$. If $K=\varnothing$, both sides are $0$; if $K\neq\varnothing$, then $\norm{\mathbf{1}_K}_\infty=1$, so $\normop{L}\geq L(\mathbf{1}_K)$.
\end{enumerate}
\hlt{Structural Commentary}: $-\norm{f}_\infty\mathbf{1}_K\leq f\leq\norm{f}_\infty\mathbf{1}_K$ yields $\abs{L(f)}\leq L(\mathbf{1}_K)\norm{f}_\infty$.\\
\hlt{Key Insight}: Positivity plus $\mathbf{1}_K$ controls $\normop{L}$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:bounded-functional-positive-decomposition-cx} \hlt{Positive Decomposition of Functionals on $C(K)$}\\
Let $K$ be compact Hausdorff and let $L\in C(K)^*$. Then $\exists L_+,L_-\in C(K)^*$ positive such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $L=L_+-L_-$.
\item $\normop{L}=L_+(\mathbf{1}_K)+L_-(\mathbf{1}_K)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $f\in C(K)_+$, set $P(f)\defeq\sup\limits_{0\leq g\leq f}L(g)$; boundedness of $L$ implies $P(f)<+\infty$.\\
If $0\leq h\leq f_1+f_2$, then $h=(h\wedge f_1)+(h-h\wedge f_1)$, terms in $[0,f_1]$ and $[0,f_2]$; $P(f_1+f_2)\leq P(f_1)+P(f_2)$.\\
The reverse inequality follows from $0\leq g_i\leq f_i\Rightarrow0\leq g_1+g_2\leq f_1+f_2$; homogeneity is immediate.\\
Define $L_+(u)\defeq P(u^+)-P(u^-)$ and $L_-\defeq L_+-L$. Then $L_+$ is positive linear, $L_-(f)=P(f)-L(f)\geq0$ for $f\in C(K)_+$, and by Proposition \ref{prop:positive-functionals-on-cx-are-continuous}(i), $L_\pm\in C(K)^*$.
\item Set $A\defeq L_+(\mathbf{1}_K)+L_-(\mathbf{1}_K)$. By (i) and Proposition \ref{prop:positive-functionals-on-cx-are-continuous}(ii), $\normop{L}\leq A$.\\
For $\varepsilon>0$, choose $0\leq\varphi\leq\mathbf{1}_K$ with $L(\varphi)>P(\mathbf{1}_K)-\varepsilon=L_+(\mathbf{1}_K)-\varepsilon$.\\
Since $\norm{2\varphi-\mathbf{1}_K}_\infty\leq1$ and $L_-(\mathbf{1}_K)=L_+(\mathbf{1}_K)-L(\mathbf{1}_K)$, $\normop{L}\geq L(2\varphi-\mathbf{1}_K)>A-2\varepsilon$. Let $\varepsilon\downarrow0$.
\end{enumerate}
\hlt{Structural Commentary}: $P(f)=\sup_{0\leq g\leq f}L(g)$ is the positive part on $C(K)_+$; $2\varphi-\mathbf{1}_K$ tests $\normop{L}$.\\
\hlt{Key Insight}: $L=L_+-L_-$ and $\normop{L}=L_+(\mathbf{1}_K)+L_-(\mathbf{1}_K)$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:riesz-representation-compact-hausdorff-cx-dual} \hlt{Riesz Representation for $C(K)^*$}\\
Let $K$ be compact Hausdorff. Let $\mathcal{M}_{\mathrm{Rad}}(K)$ be the finite signed measures $\nu$ on $K$ with $\nu^\pm$ Radon.\\
Let $\norm{\nu}_{\mathrm{var}}\defeq\abs{\nu}(K)$, and define $(T\nu)(f)\defeq\int_K f\,d\nu$ for $f\in C(K)$. Then:
\begin{enumerate}[label=(\roman*)]
    \setlength{\itemsep}{0pt}
\item $T$ is linear and well defined.
\item $\normop{T(\nu)}=\norm{\nu}_{\mathrm{var}}$ for all $\nu\in\mathcal{M}_{\mathrm{Rad}}(K)$.
\item $T$ is bijective.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\nu\in\mathcal{M}_{\mathrm{Rad}}(K)$ and $f\in C(K)$, $\abs{T(\nu)(f)}\leq\int_K\abs{f}\,d\abs{\nu}\leq\norm{f}_\infty\norm{\nu}_{\mathrm{var}}$.\\
Hence $T(\nu)\in C(K)^*$ and $\normop{T(\nu)}\leq\norm{\nu}_{\mathrm{var}}$. Linearity is signed-integral linearity.
\item Set $L\defeq T(\nu)$. By Proposition \ref{prop:bounded-functional-positive-decomposition-cx}(i), Theorem \ref{thm:riesz-markov-positive-functional-cc}, $L=L_+-L_-$, $L_\pm=T(\mu_\pm)$ for finite Radon $\mu_\pm$.\\
Since $T(\nu^++\mu_-)=T(\nu^-+\mu_+)$, by uniqueness in Theorem \ref{thm:riesz-markov-positive-functional-cc}, $\nu^++\mu_-=\nu^-+\mu_+$.\\
By Theorem \ref{thm:jordan-decomposition-charge}(ii),(iii), $\nu^\pm\leq\mu_\pm$. Therefore $\norm{\nu}_{\mathrm{var}}\leq\mu_+(K)+\mu_-(K)=L_+(\mathbf{1}_K)+L_-(\mathbf{1}_K)=\normop{L}$ by Proposition \ref{prop:bounded-functional-positive-decomposition-cx}(ii). Combine with (i).
 \item Let $L\in C(K)^*$. Proposition \ref{prop:bounded-functional-positive-decomposition-cx}(i), Theorem \ref{thm:riesz-markov-positive-functional-cc}: $L=L_+-L_-$, $L_\pm=T(\mu_\pm)$ for finite Radon $\mu_\pm$.\\
 Set $\nu\defeq\mu_+-\mu_-$. By Theorem \ref{thm:jordan-decomposition-charge}(ii),(iii), $\nu^\pm\leq\mu_\pm$.\\
 If $0\leq\theta\leq\mu$ and $\mu$ is finite Radon, then $\theta$ is Radon since the same open and compact approximants work with no larger error. Hence $\nu\in\mathcal{M}_{\mathrm{Rad}}(K)$, $T(\nu)=L$. Thus $T$ is surjective; injectivity follows from (ii).
\end{enumerate}
\hlt{Structural Commentary}: Positive Riesz--Markov: $L_\pm=T(\mu_\pm)$; Jordan minimality: $\normop{T(\nu)}=\norm{\nu}_{\mathrm{var}}$.\\
\hlt{Key Insight}: $C(K)^*\cong\mathcal{M}_{\mathrm{Rad}}(K)$ isometrically via $L(f)=\int_K f\,d\nu$.
\end{proof}
\end{theorem}

\begin{example} \label{ex:point-evaluation-dirac-riesz} \hlt{Point Evaluation Is a Dirac Measure}\\
Let $K\neq\emptyset$ be compact Hausdorff and fix $x_0\in K$. Define $L_{x_0}\in C(K)^*$ by $L_{x_0}(f)\defeq f(x_0)$.\\
Then $L_{x_0}=T(\delta_{x_0})$, since $\forall f\in C(K)$, $T(\delta_{x_0})(f)=\int_K f\,d\delta_{x_0}=f(x_0)$. Hence $\normop{L_{x_0}}=\norm{\delta_{x_0}}_{\mathrm{var}}=1$.
\end{example}

\begin{remark} \label{rem:riesz-interval-stieltjes-form} \hlt{Interval Form and Stieltjes Integrators}\\
Let $a<b$ and let $\nu$ be a finite signed Radon measure on $[a,b]$, extended by $0$ to $\R$. Set $g(x)\defeq\nu((-\infty,x])$.\\
By Proposition \ref{prop:finite-borel-measure-distribution-function} applied to $\nu^\pm$, $g\in BV(\R)$ is right-continuous and $g(-\infty)=0$.\\
For $a\leq c\leq d\leq b$, $\nu((c,d])=g(d)-g(c)$ and $\nu(\{a\})=g(a)-g(a-)$ with $g(a-)=0$.\\
Thus, in the Lebesgue--Stieltjes convention, $\forall f\in C([a,b])$, $T(\nu)(f)=\int_{[a,b]}f\,d\nu=\int_{[a,b]}f\,dg$.
\end{remark}

\begin{corollary} \label{cor:weak-star-compact-radon-measure-sets} \hlt{Weak-* Compactness for Radon Measures}\\
Let $K$ be compact Hausdorff and set $\mathcal M\defeq\mathcal{M}_{\mathrm{Rad}}(K)$, the finite signed measures $\nu$ on $K$ with $\nu^\pm$ Radon.\\
Let $\norm{\nu}_{\mathrm{var}}\defeq\abs{\nu}(K)$ and let $\tau\defeq\sigma(\mathcal M,C(K))$ be induced by $\innerprod{\nu,f}\defeq\int_K f\,d\nu$.\\
If $A\subseteq\mathcal M$ is $\tau$-closed and $\sup_{\nu\in A}\norm{\nu}_{\mathrm{var}}<+\infty$, then $A$ is $\tau$-compact.
\begin{proof}
Let $(T\nu)(f)\defeq\int_K f\,d\nu$. By Theorem \ref{thm:riesz-representation-compact-hausdorff-cx-dual}, $T:\mathcal M\to C(K)^*$ is an isometric bijection identifying $\tau$ with $\sigma(C(K)^*,C(K))$. Choose $R<+\infty$ with $A\subseteq R\overline{B}_{\mathcal M}$. Then $T(A)$ is weak-* closed and $T(A)\subseteq R\overline{B}_{C(K)^*}$.\\
By Theorem \ref{thm:banach-alaoglu}, $R\overline{B}_{C(K)^*}$ is weak-* compact; hence $T(A)$ and $A=T^{-1}(T(A))$ are weak-* compact.\\
\hlt{Structural Commentary}: Riesz isometry: reduce to weak-* closed $T(A)\subseteq R\overline{B}_{C(K)^*}$; Banach--Alaoglu: compact.\\
\hlt{Key Insight}: $T:\mathcal{M}_{\mathrm{Rad}}(K)\cong C(K)^*$ transfers Banach--Alaoglu compactness to Radon measures.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:compact-metric-bounded-measures-weak-star-subsequence} \hlt{Bounded Measures on Compact Metric Spaces Have Weak-* Subsequences}\\
Let $K$ be compact metric and let $\mu_n\geq0$ be Borel measures on $K$ with $M\defeq\sup\limits_n\mu_n(K)<+\infty$.\\
Then $\exists n_k\uparrow\infty$ and a Radon measure $\mu\geq0$ on $K$ such that $\int_K f\,d\mu_{n_k}\to\int_K f\,d\mu$ for all $f\in C(K)$.
\begin{proof} Set $T_n(f)\defeq\int_K f\,d\mu_n$. By Proposition \ref{prop:positive-functionals-on-cx-are-continuous}(ii), $T_n\in C(K)^*$, $T_n\geq0$, and $\normop{T_n}=\mu_n(K)\leq M$.\\
Since $C(K)$ is separable by Theorem \ref{thm:riesz-cx-separable-iff-metrizable}, by Theorem \ref{thm:helly-selection-bounded-dual-sequences}, $\exists n_k\uparrow\infty$ and $T\in C(K)^*$ with $T_{n_k}(f)\to T(f)$ for all $f\in C(K)$. Positivity passes to $T$, so by Theorem \ref{thm:riesz-markov-positive-functional-cc}, $T(f)=\int_K f\,d\mu$ for some Radon $\mu\geq0$.\\
\hlt{Structural Commentary}: Bound $(T_n)$ in $C(K)^*$; use separability and Helly; represent the positive limit.\\
\hlt{Key Insight}: Bounded positive measures on compact metric $K$ are weak-* sequentially compact on $C(K)$.
\end{proof}
\end{corollary}
