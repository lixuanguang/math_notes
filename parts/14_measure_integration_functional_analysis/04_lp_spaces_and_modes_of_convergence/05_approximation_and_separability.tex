\subsection{Approximation and Separability in \texorpdfstring{$L^p$}{Lp}}

Approximation results explain why $L^p$ spaces are large enough to contain rough measurable functions but still small enough to be controlled by countable data when $p<\infty$. The guiding pattern is to localize the $L^p$ mass, approximate measurable atoms by elementary sets, and then code the elementary sets with rational parameters.

\begin{definition} \label{def:normed-density-and-separability-lp} \hlt{Dense Subsets and Separable Normed Spaces}\\
Let $A\subseteq B$ be subsets of a normed space $(V,\norm{\cdot})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ is \hlt{dense in} $B$ if $\forall b\in B\ \forall\varepsilon>0\ \exists a\in A$ with $\norm{a-b}<\varepsilon$.
\item $V$ is \hlt{separable} if it has a countable dense subset.
\end{enumerate}
Equivalently, $A$ is dense in $B$ iff every $b\in B$ is the norm-limit of a sequence from $A$.
\end{definition}

\begin{example} \label{ex:sequence-space-separability} \hlt{Sequence-Space Separability}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\ell^\infty(\F)$ is not separable. The set $\{0,1\}^{\N}$ is uncountable, and $s\ne t$ gives $d_\infty(s,t)=1$; the balls $B_{1/3}(s)$ are pairwise disjoint, so no countable dense set can meet all of them.
\item For $1\leq p<\infty$, $\ell^p(\F)$ is separable. Let $D$ be the countable set of finite-support sequences with rational coordinates, using rational real and imaginary parts if $\F=\C$.\\
For $x\in\ell^p$ and $\varepsilon>0$, choose $N$ with $\sum_{n>N}\abs{x_n}^p<(\varepsilon/2)^p$, then choose $y\in D$ with support in $\{1,\ldots,N\}$ and $\sum_{n=1}^N\abs{x_n-y_n}^p<(\varepsilon/2)^p$. Thus $\norm{x-y}_p<\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Separability is the existence of a countable set meeting every nonempty open ball.\\
\hlt{Key Insight}: Finite rational data are dense in $\ell^p$, while $\ell^\infty$ contains an uncountable $1$-separated set.
\end{example}

\begin{theorem} \label{thm:lp-real-line-separability-and-cc-density} \hlt{Separability and $C_c$ Density in $L^p$}\\
Let $E\subseteq\R$ be Lebesgue measurable and let $1\leq p<\infty$.\\
If $C_c(E)$ denotes restrictions to $E$ of continuous functions on $\R$ with bounded support, then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $L^p(E)$ is separable;
\item $C_c(E)$ is dense in $L^p(E)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\mathbb F_0\defeq\Q$ if real, $\mathbb F_0\defeq\Q+i\Q$ if complex. Let $\mathcal D\defeq\operatorname{span}_{\mathbb F_0}\{\mathbf{1}_{E\cap(r,s)}:r,s\in\Q,\ r<s\}$.\\
Then $\mathcal D$ is countable. Extend $f\in L^p(E)$ by $0$ to $\tilde f\in L^p(\R)$.\\
By Proposition \ref{prop:simple-density-support-and-tails-lp}, it suffices to approximate $s=\sum\limits_{\ell=1}^{M}a_\ell\mathbf{1}_{A_\ell}$ with $m(A_\ell)<+\infty$.\\
For each $\ell$, Corollary \ref{cor:real-line-finite-open-interval-approximation} and rational endpoint perturbation give a finite union $V_\ell$ of rational open intervals with $m(A_\ell\triangle V_\ell)$ small, hence $\norm{\mathbf{1}_{E\cap A_\ell}-\mathbf{1}_{E\cap V_\ell}}_p\leq m(A_\ell\triangle V_\ell)^{1/p}$. Approximate $a_\ell$ by $q_\ell\in\mathbb F_0$; then $\sum\limits_{\ell=1}^{M}q_\ell\mathbf{1}_{E\cap V_\ell}\in\mathcal D$ approximates $f$ arbitrarily well. Hence $L^p(E)$ is separable.
\item For rational $r<s$ and $0<\eta<\frac{s-r}{2}$, choose $\psi_\eta\in C_c(\R)$ with $0\leq\psi_\eta\leq1$, $\psi_\eta=1$ on $[r+\eta,s-\eta]$, and $\psi_\eta=0$ on $\R\setminus(r,s)$. Then $\norm{\psi_\eta|_E-\mathbf{1}_{E\cap(r,s)}}_p^p\leq m((r,r+\eta)\cup(s-\eta,s))\leq2\eta\to0$.\\
Thus each generator of $\mathcal D$ lies in $\overline{C_c(E)}^{\,L^p(E)}$; by finite linearity and (i), $C_c(E)$ is dense in $L^p(E)$.
\end{enumerate}
\hlt{Structural Commentary}: Finite-measure atoms reduce to rational intervals, then to continuous cutoffs.\\
\hlt{Key Insight}: Countable rational steps lie in $\overline{C_c(E)}^{\,L^p(E)}$.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:l1-does-not-exhaust-linfty-dual} \hlt{$L^1$ Does Not Exhaust the Dual of $L^\infty$}\\
Let $I=[a,b]$ with $a<b$, and let $\iota:L^1(I)\to(L^\infty(I))^*$ be $\iota(g)(f)\defeq\int_I fg\,dm$. The map $\iota$ from Proposition \ref{prop:l1-countably-additive-part-of-linfty-dual} is not onto. If $\iota$ were onto, then $(L^\infty(I))^*$ would be separable by Theorem \ref{thm:lp-real-line-separability-and-cc-density}.\\
Let $(\Phi_n)$ be dense in its unit sphere and choose $u_n\in L^\infty(I)$ with $\norm{u_n}_\infty=1$ and $\abs{\Phi_n(u_n)}>\frac{1}{2}$.\\
If $Y\defeq\overline{\operatorname{span}}\{u_n:n\geq1\}\ne L^\infty(I)$, Hahn--Banach gives $\Phi\in(L^\infty(I))^*$ with $\norm{\Phi}=1$ and $\Phi|_Y=0$.\\
Choosing $n$ with $\norm{\Phi-\Phi_n}<\frac{1}{4}$ gives $\abs{\Phi_n(u_n)}<\frac{1}{4}$, a contradiction. Hence $L^\infty(I)$ would be separable.\\
But $v_t\defeq\mathbf{1}_{[a,t]}$ satisfies $\norm{v_t-v_s}_\infty=1$ for $s<t$, an uncountable separated family, so $L^\infty(I)$ is not separable.
\end{remark}

\begin{example} \label{ex:linfty-interval-not-separable} \hlt{$L^\infty$ on an Interval Need Not Be Separable}\\
Let $I=[a,b]$ with $a<b$. For $t\in I$, set $u_t\defeq\mathbf{1}_{[a,t]}$. If $s<t$, then $\norm{u_t-u_s}_\infty=1$, since $u_t-u_s=\mathbf{1}_{(s,t]}$ a.e.\\
Thus $\{B(u_t,\frac{1}{3}):t\in I\}$ is pairwise disjoint. If countable $D\subseteq L^\infty(I)$ were dense, choose $d_t\in D\cap B(u_t,\frac{1}{3})$ for each $t\in I$; disjointness makes $t\mapsto d_t$ injective, impossible. Hence $L^\infty(I)$ is not separable.\\
\hlt{Key Insight}: Uncountably many distance-$1$ indicators preclude countable density.
\end{example}
