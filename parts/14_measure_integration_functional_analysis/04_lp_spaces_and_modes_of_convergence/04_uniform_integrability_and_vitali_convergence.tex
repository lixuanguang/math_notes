\subsection{Uniform Integrability and Vitali Convergence}

Uniform integrability is the $L^1$ condition that rules out concentration of mass on tiny sets. On infinite-measure spaces it must be paired with tightness, which rules out mass escaping to remote finite-measure tails.

\begin{lemma} \label{lem:finite-measure-lebesgue-small-piece-decomposition} \hlt{Finite-Measure Lebesgue Sets Have Small Pieces}\\
Let $E\in\mathcal{L}^1$, $m(E)<+\infty$, and $\delta>0$. Then $\exists E_1,\ldots,E_N\in\mathcal{L}^1$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E_i\cap E_j=\emptyset$ for $i\neq j$.
\item $E=\bigsqcup\limits_{k=1}^NE_k$.
\item $m(E_k)<\delta$ for $1\leq k\leq N$.
\end{enumerate}
\begin{proof}
Choose $n_0$ with $T\defeq E\setminus[-n_0,n_0]$ and $m(T)<\delta$. Partition $[-n_0,n_0]$ into finitely many disjoint intervals $I_j$ with $m(I_j)<\delta$, and let $(E_k)$ be the nonempty sets among $T$ and $E\cap I_j$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $T\cap[-n_0,n_0]=\emptyset$ and the $I_j$ are disjoint, the $E_k$ are pairwise disjoint.
\item $E=T\bigsqcup(E\cap[-n_0,n_0])=T\bigsqcup\bigsqcup_j(E\cap I_j)=\bigsqcup\limits_{k=1}^NE_k$.
\item $m(T)<\delta$ and $m(E\cap I_j)\leq m(I_j)<\delta$.
\end{enumerate}
\hlt{Structural Commentary}: Finite measure gives a small tail $T$; a fine partition makes $E\setminus T$ small-piece.\\
\hlt{Key Insight}: One small tail plus a fine finite grid gives a finite small-piece decomposition.
\end{proof}
\end{lemma}

\begin{proposition} \label{prop:l1-small-set-absolute-continuity-finite-lebesgue-converse} \hlt{$L^1$ Absolute Continuity on Small Sets}\\
Let $E\in\mathcal{L}^1$, let $\mathcal{L}^1_E=\{A\in\mathcal{L}^1:A\subseteq E\}$, let $m_E=m|_{\mathcal{L}^1_E}$, and let $f:E\to\R$ be measurable.\\
Consider the condition $(\ast)$: $\forall \varepsilon>0$ $\exists \delta>0$ such that $A\in\mathcal{L}^1_E$ and $m(A)<\delta$ imply $\int_A\abs f\,dm<\varepsilon$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $f\in L^1(m_E)$, then $(\ast)$ holds.
\item If $m(E)<+\infty$ and $(\ast)$ holds, then $f\in L^1(m_E)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $f\in L^1(m_E)$, Lemma \ref{lem:lp-integral-absolute-continuity} on $(E,\mathcal{L}^1_E,m_E)$ with $p=1$ gives $\forall\varepsilon>0\ \exists\delta>0$ such that $A\in\mathcal{L}^1_E$ and $m_E(A)<\delta$ imply $\int_A\abs f\,dm_E<\varepsilon$. Since $m_E(A)=m(A)$ and $\int_A\abs f\,dm_E=\int_A\abs f\,dm$, $(\ast)$ holds.
\item Choose $\delta>0$ from $(\ast)$ with $\varepsilon=1$. By Lemma \ref{lem:finite-measure-lebesgue-small-piece-decomposition}, write $E=\bigsqcup\limits_{k=1}^{N}E_k$ with $E_k\in\mathcal{L}^1_E$ and $m(E_k)<\delta$. Then $(\ast)$ gives $\int_{E_k}\abs f\,dm<1$ for each $k$, hence $\int_E\abs f\,dm=\sum\limits_{k=1}^{N}\int_{E_k}\abs f\,dm<N<+\infty$.
\end{enumerate}
\hlt{Structural Commentary}: $L^1$ gives small-set control; finite partitions recover $L^1$.\\
\hlt{Key Insight}: A finite small-piece decomposition turns local integral bounds into one global finite integral.
\end{proof}
\end{proposition}

\begin{definition} \label{def:l1-uniform-integrability-and-tightness} \hlt{Uniform Integrability and Tightness}\\
Let $\mathcal{F}$ be a family of measurable functions $h:X\to\R$ on $(X,\mathcal{M},\mu)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{F}$ is \hlt{uniformly integrable} on $X$ if $\forall\varepsilon>0\,\exists\delta>0$ such that $A\in\mathcal{M}$ and $\mu(A)<\delta$ imply $\sup\limits_{h\in\mathcal{F}}\int_A\abs h\,d\mu<\varepsilon$.
\item $\mathcal{F}$ is \hlt{tight} on $X$ if $\forall\varepsilon>0\,\exists X_\varepsilon\in\mathcal{M}$ with $\mu(X_\varepsilon)<+\infty$ and $\sup\limits_{h\in\mathcal{F}}\int_{X\setminus X_\varepsilon}\abs h\,d\mu<\varepsilon$.
\end{enumerate}
On finite measure spaces, tightness is automatic by taking $X_\varepsilon=X$.\\
\hlt{Key Insight}: UI controls mass on small $A$; tightness controls mass on $X\setminus X_\varepsilon$.
\end{definition}

\begin{example} \label{ex:uniform-integrability-tightness-obstructions} \hlt{Concentration versus Escape}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Concentration}: on $(0,1)$, $u_n\defeq n\mathbf{1}_{(0,\frac{1}{n})}$. For every $\delta>0$, choose $n>\delta^{-1}$ and $A_n=(0,\frac{1}{n})$.\\
Then $m(A_n)<\delta$ but $\int_{A_n}u_n\,dm=1$, so $(u_n)$ is not uniformly integrable.
\item \hlt{Escape}: on $\R$, let $I_n\defeq(n,n+1)$ and $v_n\defeq\mathbf{1}_{I_n}$. Since $\int_Av_n\,dm\leq m(A)$, $(v_n)$ is uniformly integrable.\\
If $m(E_0)<+\infty$ and $\sup_n\int_{\R\setminus E_0}v_n\,dm<\frac{1}{2}$, then $m(E_0\cap I_n)>\frac{1}{2}$ for all $n$, contradiction.
\end{enumerate}
\hlt{Structural Commentary}: Concentration occurs even on $m(X)<+\infty$; escape is an $m(X)=+\infty$ obstruction.\\
\hlt{Key Insight}: Uniform integrability controls small-set mass; tightness keeps mass in finite-measure cores.
\end{example}

\begin{proposition} \label{prop:l1-uniform-integrability-basic-sources} \hlt{Basic Sources of Uniform Integrability and Tightness}\\
Let $(X,\mathcal{M},\mu)$ be a measure space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $0\leq g\in L^1(\mu)$ and $\forall h\in\mathcal{F},\ \abs h\leq g$ a.e., then $\mathcal{F}$ is uniformly integrable and tight.
\item If $f_1,\ldots,f_N\in L^1(\mu)$, then $\{f_1,\ldots,f_N\}$ is uniformly integrable and tight.
\end{enumerate}
\begin{proof} Let $\varepsilon>0$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $\delta>0$ with $\mu(A)<\delta\Rightarrow\int_Ag\,d\mu<\varepsilon$ by Lemma \ref{lem:lp-integral-absolute-continuity}; hence $\sup\limits_{h\in\mathcal F}\int_A\abs h\,d\mu<\varepsilon$.\\
Choose finite-measure $X_\varepsilon$ with $\int_{X\setminus X_\varepsilon}g\,d\mu<\varepsilon$ by Proposition \ref{prop:finite-positive-integral-localization}; hence $\sup\limits_{h\in\mathcal F}\int_{X\setminus X_\varepsilon}\abs h\,d\mu<\varepsilon$.
\item For each $j$, apply (i) to $\{f_j\}$ with $\varepsilon$: choose $\delta_j>0$ such that $\mu(A)<\delta_j\Rightarrow\int_A\abs{f_j}\,d\mu<\varepsilon$, and choose $X_{\varepsilon,j}$ with $\mu(X_{\varepsilon,j})<+\infty$ and $\int_{X\setminus X_{\varepsilon,j}}\abs{f_j}\,d\mu<\varepsilon$. Set $\delta\defeq\min\limits_{1\leq j\leq N}\delta_j$ and $X_\varepsilon\defeq\bigcup\limits_{j=1}^NX_{\varepsilon,j}$.\\
Then $\mu(A)<\delta\Rightarrow\sup\limits_j\int_A\abs{f_j}\,d\mu<\varepsilon$, while $\mu(X_\varepsilon)<+\infty$ and $\sup\limits_j\int_{X\setminus X_\varepsilon}\abs{f_j}\,d\mu<\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: An envelope gives common bounds; finite families use $\min\limits_j\delta_j$ and $\bigcup\limits_jX_{\varepsilon,j}$.\\
\hlt{Key Insight}: Domination gives uniform estimates; finite families allow minima and finite unions.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:lp-bounded-families-uniformly-integrable} \hlt{$L^p$-Bounded Families Are Uniformly Integrable}\\
Let $1<p<\infty$, $q$ conjugate to $p$. If $\mathcal{F}\subseteq L^p(\mu)$ and $\forall h\in\mathcal{F},\ \norm{h}_p\leq M$, then $\mathcal{F}$ is uniformly integrable.
\begin{proof}
If $M=0$, every $h\in\mathcal{F}$ vanishes a.e. and the claim is immediate. Else, for measurable $A$ with $\mu(A)<+\infty$,\\
by H\"older's inequality, $\int_A\abs h\,d\mu=\int_X\abs h\,\mathbf 1_A\,d\mu\leq\norm h_p\,\norm{\mathbf 1_A}_q\leq M\mu(A)^{\frac{1}{q}}$.\\
Given $\varepsilon>0$, take $\delta=(\frac{\varepsilon}{M})^q$. Then $\mu(A)<\delta$ implies $\sup\limits_{h\in\mathcal F}\int_A\abs h\,d\mu<\varepsilon$.\\
\hlt{Structural Commentary}: Apply H\"older to $h\mathbf 1_A$ and choose one measure threshold for the whole family.\\
\hlt{Key Insight}: H\"older converts a uniform $L^p$ bound into uniform small-set $L^1$ control.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:uniform-integrability-large-value-tails} \hlt{Uniform Integrability by Large-Value Tails}\\
Let $(X,\mathcal{M},\mu)$ be a finite measure space and let $(f_n)$ be bounded in $L^1(\mu)$. The following are equivalent.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(f_n)$ is uniformly integrable.
\item $\forall\varepsilon>0\ \exists M>0:\ \sup\limits_n\int_{\{\abs{f_n}\geq M\}}\abs{f_n}\,d\mu<\varepsilon$.
\end{enumerate}
\begin{proof}
Let $C=\sup\limits_n\norm{f_n}_1<+\infty$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Given $\varepsilon>0$, choose $\delta>0$ from uniform integrability and $M>C/\delta$.\\
By Chebyshev, $\mu(\{\abs{f_n}\geq M\})\leq M^{-1}\norm{f_n}_1\leq C/M<\delta$, so $\sup\limits_n\int_{\{\abs{f_n}\geq M\}}\abs{f_n}\,d\mu<\varepsilon$.
\item[$(\Leftarrow)$] Given $\varepsilon>0$, choose $M>0$ with $\sup\limits_n\int_{\{\abs{f_n}\geq M\}}\abs{f_n}\,d\mu<\frac{\varepsilon}{2}$ and set $\delta\defeq\frac{\varepsilon}{2M}$.\\
If $\mu(A)<\delta$, then $\int_A\abs{f_n}\,d\mu\leq\int_{\{\abs{f_n}\geq M\}}\abs{f_n}\,d\mu+M\mu(A)<\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Chebyshev shrinks tails; truncation recovers UI.\\
\hlt{Key Insight}: UI iff large tails vanish uniformly.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dunford-pettis-l1-finite-measure} \hlt{Dunford--Pettis Criterion in $L^1$}\\
Let $(X,\mathcal{M},\mu)$ be a finite measure space and let $(f_n)$ be bounded in $L^1(\mu)$. The following are equivalent.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(f_n)$ is uniformly integrable.
\item $\forall(f_{n_k})\subseteq(f_n)\ \exists(f_{n_{k_j}})\subseteq(f_{n_k})$ converging weakly in $L^1(\mu)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Let $(u_j)$ be a subsequence of $(f_n)$, set $C\defeq\sup_j\norm{u_j}_1$, and put $T_m(t)\defeq(-m)\vee(t\wedge m)$.\\
By Proposition \ref{prop:uniform-integrability-large-value-tails}, $\sup_j\norm{u_j-T_m(u_j)}_1\to0$. For fixed $m$ and sign $\pm$, $\norm{T_m(u_j^\pm)}_2^2\leq m\norm{u_j^\pm}_1\leq mC$.\\
By bounded Hilbert weak subsequence extraction and diagonalization over $(m,\pm)$, pass to a subsequence, still $(u_j)$, such that $\int_ET_m(u_j^\pm)\,d\mu$ converges for every $E\in\mathcal M$ and $m\in\N$. Set $\nu_j^\pm(E)\defeq\int_Eu_j^\pm\,d\mu$.\\
For $r,s,m$ and $E\in\mathcal M$, $\abs{\nu_r^\pm(E)-\nu_s^\pm(E)}\leq\abs{\int_E(T_m(u_r^\pm)-T_m(u_s^\pm))\,d\mu}+2\sup_j\norm{u_j^\pm-T_m(u_j^\pm)}_1$.\\
Hence $(\nu_j^\pm(E))_j$ is Cauchy. Put $\nu^\pm(E)\defeq\lim_j\nu_j^\pm(E)$.\\
By Theorem \ref{thm:vitali-hahn-saks-setwise-convergence}, $\nu^\pm$ are finite measures and $\nu^\pm\ll\mu$. Write $h^\pm=\frac{d\nu^\pm}{d\mu}$ and $h\defeq h^+-h^-\in L^1(\mu)$.\\
Then $\int_X\varphi u_j\,d\mu\to\int_X\varphi h\,d\mu$ for simple $\varphi$; by Lemma \ref{lem:bounded-simple-approximation-brackets} and $\sup_j\norm{u_j}_1+\norm{h}_1<+\infty$, the convergence holds for every $\varphi\in L^\infty(\mu)$. Thus $u_j\rightharpoonup h$ in $L^1(\mu)$.
\item[$(\Leftarrow)$] Suppose $(f_n)$ is not uniformly integrable. Since finite subfamilies are uniformly integrable by Proposition \ref{prop:l1-uniform-integrability-basic-sources}(ii), choose a subsequence $(u_j)$ and sets $A_j\in\mathcal M$ with $\mu(A_j)\to0$ and $\int_{A_j}\abs{u_j}\,d\mu\geq\varepsilon_0>0$.\\
By (ii), some $(u_{j_\ell})$ converges weakly in $L^1(\mu)$ to $u$. Set $\eta_\ell(E)\defeq\int_Eu_{j_\ell}\,d\mu$ and $\eta(E)\defeq\int_Eu\,d\mu$.\\
Then $\eta_\ell(E)\to\eta(E)$ for every $E\in\mathcal M$, and $\sup_\ell\abs{\eta_\ell}(X)=\sup_\ell\norm{u_{j_\ell}}_1<+\infty$ by Theorem \ref{thm:integral-charge-variation-density}.\\
The signed form of Theorem \ref{thm:vitali-hahn-saks-setwise-convergence} implies $\eta_\ell\ll_{\mathrm u}\mu$ in variation:\\
$\forall\varepsilon>0\ \exists\delta>0$ such that $\mu(A)<\delta\Rightarrow\sup_\ell\abs{\eta_\ell}(A)<\varepsilon$. Since $\abs{\eta_\ell}(A)=\int_A\abs{u_{j_\ell}}\,d\mu$, the subsequence $(u_{j_\ell})$ is uniformly integrable, contradicting $\int_{A_{j_\ell}}\abs{u_{j_\ell}}\,d\mu\geq\varepsilon_0$ and $\mu(A_{j_\ell})\to0$.
\end{enumerate}
\hlt{Structural Commentary}: Truncate, pass to setwise limits, apply VHS.\\
\hlt{Key Insight}: UI is weak sequential compactness in finite $L^1$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:dominated-l1-weak-subsequence} \hlt{Dominated $L^1$ Families Have Weak Subsequences}\\
Let $(X,\mathcal M,\mu)$ be finite. If $g\in L^1(\mu)$, $\forall n \abs{f_n}\leq g$ a.e., then $(f_n)$ has subsequence that converges weakly in $L^1(\mu)$.
\begin{proof}
Since $\abs{f_n}\leq g$ a.e., $\norm{f_n}_1\leq\norm g_1$ $\forall n$, $(f_n)$ is bounded in $L^1(\mu)$. By Proposition \ref{prop:l1-uniform-integrability-basic-sources}(i), $\{f_n:n\geq1\}$ is uniformly integrable. Theorem \ref{thm:dunford-pettis-l1-finite-measure}, $(i)\Rightarrow(ii)$, yields a weakly convergent subsequence.\\
\hlt{Structural Commentary}: Domination verifies Dunford--Pettis through a common integrable envelope.\\
\hlt{Key Insight}: One $L^1$ envelope prevents concentration throughout the sequence.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:bounded-lp-weak-l1-subsequence} \hlt{Bounded $L^p$ Families Are Weakly Compact in $L^1$}\\
Let $(X,\mathcal M,\mu)$ be finite, let $1<p<+\infty$, and let $(f_n)$ be bounded in $L^p(\mu)$.\\
Then $(f_n)$ has a subsequence that converges weakly in $L^1(\mu)$.
\begin{proof}
Let $q$ be conjugate to $p$ and $M\defeq\sup_n\norm{f_n}_p<+\infty$. By H\"older, $\norm{f_n}_1\leq M\mu(X)^{\frac{1}{q}}$ $\forall n$, so $(f_n)$ is bounded in $L^1(\mu)$. By Corollary \ref{cor:lp-bounded-families-uniformly-integrable}, $\{f_n:n\geq1\}$ is uniformly integrable.\\
Hence Theorem \ref{thm:dunford-pettis-l1-finite-measure}, $(i)\Rightarrow(ii)$, yields a weakly convergent subsequence in $L^1(\mu)$.\\
\hlt{Structural Commentary}: H\"older gives $L^1$ boundedness, uniform integrability before Dunford--Pettis is applied.\\
\hlt{Key Insight}: On finite measure spaces, an $L^p$ bound with $p>1$ rules out $L^1$ concentration.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:vitali-convergence-l1-ui-tight-ae} \hlt{Vitali Convergence for $L^1$ Families}\\
Let $(X,\mathcal{M},\mu)$ be a measure space, let $f_n\in L^1(\mu)$, let $f:X\to\R$ be measurable, and assume $f_n\to f$ a.e.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mu(X)<+\infty$ and $\{f_n:n\geq1\}$ is uniformly integrable, then $f\in L^1(\mu)$ and $\norm{f_n-f}_1\to0$.
\item If $\{f_n:n\geq1\}$ is uniformly integrable and tight, then $f\in L^1(\mu)$ and $\norm{f_n-f}_1\to0$.
\item Under either hypothesis, $\int_Xf_n\,d\mu\to\int_Xf\,d\mu$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $\mu(X)<+\infty$ and $f_n\to f$ a.e., $f_n\to f$ in measure by Corollary \ref{cor:finite-measure-ae-implies-measure}. Uniform integrability is the small-set hypothesis of Theorem \ref{thm:vitali-convergence-lp} with $p=1$; the finite-tail hypothesis holds with $E_\varepsilon=X$, QED.
\item Fix $\alpha,\eta>0$ and set $B_n^\alpha\defeq\{\abs{f_n-f}\geq\alpha\}$.\\
By tightness, choose $X_0\in\mathcal M$ with $\mu(X_0)<+\infty$, $\sup\limits_n\int_{X\setminus X_0}\abs{f_n}\,d\mu<\frac{\alpha\eta}{4}$. By Fatou, $\int_{X\setminus X_0}\abs f\,d\mu\leq\frac{\alpha\eta}{4}$.\\
Then $\alpha\mu(B_n^\alpha\cap(X\setminus X_0))\leq\int_{X\setminus X_0}(\abs{f_n}+\abs f)\,d\mu<\frac{\alpha\eta}{2}$. Since $f_n\to f$ a.e. on finite $X_0$, $\mu(B_n^\alpha\cap X_0)<\frac{\eta}{2}$ eventually by Corollary \ref{cor:finite-measure-ae-implies-measure}. Thus $\mu(B_n^\alpha)<\eta$ eventually, so $f_n\to f$ in measure.\\
By Theorem \ref{thm:vitali-convergence-lp} with $p=1$, using uniform integrability and tightness, $f\in L^1(\mu)$ and $\norm{f_n-f}_1\to0$.
\item In either case, $\abs{\int_Xf_n\,d\mu-\int_Xf\,d\mu}=\abs{\int_X(f_n-f)\,d\mu}\leq\norm{f_n-f}_1\to0$.
\end{enumerate}
\hlt{Structural Commentary}: Tightness gives finite cores; UI upgrades in-measure convergence to $L^1$.\\
\hlt{Key Insight}: Uniform integrability prevents concentration; tightness prevents escape.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:nonnegative-l1-vanishing-vitali-criterion} \hlt{Nonnegative Integral Vanishing Criterion}\\
Let $f_n\in L^1(\mu)$ with $f_n\geq0$ a.e. The following are equivalent.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\int_Xf_n\,d\mu\to0$.
\item $\norm{f_n}_1\to0$.
\item $f_n\to0$ in measure, and $\{f_n:n\geq1\}$ is uniformly integrable and tight.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ Since $f_n\geq0$, $\norm{f_n}_1=\int_Xf_n\,d\mu\to0$.
\item[$(ii)$]$\Rightarrow (iii)$ By Theorem \ref{thm:vitali-convergence-lp} with $p=1$ and limit $0$, $\norm{f_n}_1\to0$ implies $f_n\to0$ in measure,\\
$\forall\varepsilon>0\ \exists E_\varepsilon\in\mathcal M$ with $\mu(E_\varepsilon)<+\infty$ and $F\cap E_\varepsilon=\emptyset\Rightarrow\sup\limits_n\int_Ff_n\,d\mu<\varepsilon$,\\
and $\forall\varepsilon>0\ \exists\delta>0$ with $\mu(A)<\delta\Rightarrow\sup\limits_n\int_Af_n\,d\mu<\varepsilon$; hence tightness and uniform integrability.
\item[$(iii)$]$\Rightarrow (i)$ Condition (iii) is hypotheses (ii)--(iv) of Theorem \ref{thm:vitali-convergence-lp} with $p=1$ and limit $0$, so $\norm{f_n}_1\to0$.\\
Since $f_n\geq0$, $\int_Xf_n\,d\mu=\norm{f_n}_1\to0$.
\end{enumerate}
\hlt{Structural Commentary}: For nonnegative functions, integral convergence to zero is $L^1$ convergence to zero.\\
\hlt{Key Insight}: Vitali's conditions characterize when nonnegative mass vanishes.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:domination-implies-vitali-side-conditions} \hlt{Domination Gives Tail and Small-Set Control}\\
Let $1\leq p<\infty$, $g\in L^p(\mu)$, and $\forall n,\ \abs{f_n}\leq g$ a.e. Then $\forall\varepsilon>0$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\exists E_\varepsilon\in\mathcal M$ with $\mu(E_\varepsilon)<+\infty$ such that $F\cap E_\varepsilon=\emptyset\Rightarrow\sup\limits_n\int_F\abs{f_n}^p\,d\mu<\varepsilon^p$.
\item $\exists\delta(\varepsilon)>0$ such that $\mu(E)<\delta(\varepsilon)\Rightarrow\sup\limits_n\int_E\abs{f_n}^p\,d\mu<\varepsilon^p$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $E_\varepsilon$ with $\mu(E_\varepsilon)<+\infty$ and $F\cap E_\varepsilon=\emptyset\Rightarrow\int_F\abs g^p\,d\mu<\varepsilon^p$ by Proposition \ref{prop:simple-density-support-and-tails-lp}.\\
Since $\abs{f_n}^p\leq\abs g^p$ a.e., $\sup\limits_n\int_F\abs{f_n}^p\,d\mu<\varepsilon^p$.
\item Choose $\delta(\varepsilon)>0$ with $\mu(E)<\delta(\varepsilon)\Rightarrow\int_E\abs g^p\,d\mu<\varepsilon^p$ by Lemma \ref{lem:lp-integral-absolute-continuity}.\\
Since $\abs{f_n}^p\leq\abs g^p$ a.e., $\sup\limits_n\int_E\abs{f_n}^p\,d\mu<\varepsilon^p$.
\end{enumerate}
\hlt{Structural Commentary}: One $L^p$ envelope supplies both tail tightness and small-set control.\\
\hlt{Key Insight}: Domination transfers both estimates from $g$ to every $f_n$.
\end{proof}
\end{corollary}
