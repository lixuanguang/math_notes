\subsection{Weak Convergence in \texorpdfstring{$L^p$}{Lp}}

Weak convergence in $L^p$ keeps only the scalar information detected by conjugate functions. For $1<p<+\infty$, reflexivity turns boundedness into weak subsequential compactness; at $p=1$, bounded mass can still concentrate and destroy weak compactness unless uniform integrability is imposed.

\begin{theorem} \label{thm:lp-weak-convergence-dual-tests} \hlt{$L^p$ Weak Convergence by Dual Tests}\\
Let $(X,\mathcal M,\mu)$ be a measure space, $1<p<+\infty$, $q=\frac{p}{p-1}$, $(f_n)\subseteq L^p(\mu)$, and $f\in L^p(\mu)$, all real-valued.\\
Then $f_n\rightharpoonup f$ in $L^p(\mu)$ iff $\int_X f_ng\,d\mu\to\int_X fg\,d\mu$ for all $g\in L^q(\mu)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] For $g\in L^q(\mu)$, $\Lambda_g(h)\defeq\int_Xhg\,d\mu$ is in $(L^p(\mu))^*$ by Holder.\\
Hence $f_n\rightharpoonup f$ implies $\int_Xf_ng\,d\mu=\Lambda_g(f_n)\to\Lambda_g(f)=\int_Xfg\,d\mu$.
\item[$(\Leftarrow)$] Let $\Lambda\in(L^p(\mu))^*$. By Theorem \ref{thm:riesz-representation-lp}, choose $g\in L^q(\mu)$ with $\Lambda(h)=\int_Xhg\,d\mu$ $\forall h\in L^p(\mu)$.\\
The assumed integral test yields $\Lambda(f_n)\to\Lambda(f)$, hence $f_n\rightharpoonup f$.
\end{enumerate}
\hlt{Structural Commentary}: Dual tests are exactly $L^q$ pairings.\\
\hlt{Key Insight}: Weak convergence in $L^p$ is conjugate-function testing.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:lp-weak-limits-bounded-lsc} \hlt{Weak Limits and Norm Lower Semicontinuity}\\
Let $(X,\mathcal M,\mu)$ be a measure space, $1<p<+\infty$, $(f_n)\subseteq L^p(\mu)$, and $f_n\rightharpoonup f$ in real $L^p(\mu)$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\sup\limits_n\norm{f_n}_p<+\infty$;
\item if $f_n\rightharpoonup h$ in $L^p(\mu)$, then $h=f$ in $L^p(\mu)$;
\item $\norm{f}_p\leq\liminf\limits_{n\to\infty}\norm{f_n}_p$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Proposition \ref{prop:weakly-convergent-sequences-bounded} applied to $u_n=f_n$ in $L^p(\mu)$, $f_n\rightharpoonup f$ implies $\sup\limits_n\norm{f_n}_p<+\infty$.
\item Let $q=\frac{p}{p-1}$, $f_n\rightharpoonup h$. For $g\in L^q(\mu)$, by Theorem \ref{thm:lp-weak-convergence-dual-tests}, $\int_X(f-h)g\,d\mu=\lim\limits_n\int_Xf_ng\,d\mu-\lim\limits_n\int_Xf_ng\,d\mu=0$.\\
By Proposition \ref{prop:holder-dual-norm-attainment}, $\norm{f-h}_p=0$, so $h=f$ in $L^p(\mu)$.
\item By Proposition \ref{prop:weakly-convergent-sequences-bounded}, $u\mapsto\norm{u}_p$ is weakly lower semicontinuous on $L^p(\mu)$;\\
with $u_n=f_n$ and $u=f$, $\norm{f}_p\leq\liminf\limits_{n\to\infty}\norm{f_n}_p$.
\end{enumerate}
\hlt{Structural Commentary}: General weak facts handle bounds; $L^q$ tests force uniqueness.\\
\hlt{Key Insight}: Weak limits are norm-controlled and conjugate-tested.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:lp-reflexive-sigma-finite} \hlt{Reflexivity of $L^p$, $1<p<+\infty$}\\
Let $(X,\mathcal M,\mu)$ be $\sigma$-finite, $1<p<+\infty$, $q=\frac{p}{p-1}$, and $Jf(\Phi)\defeq\Phi(f)$.\\
For $r\in\{p,q\}$ with conjugate $r'$, set $T_ru\in(L^{r'}(\mu))^*$ by $(T_ru)(v)\defeq\int_Xuv\,d\mu$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T_p$ and $T_q$ are isometric isomorphisms;
\item $J:L^p(\mu)\to(L^p(\mu))^{**}$ is onto; hence real $L^p(\mu)$ is reflexive.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:riesz-representation-lp}, $T_p:L^p(\mu)\to(L^q(\mu))^*$ and $T_q:L^q(\mu)\to(L^p(\mu))^*$ are isometric isomorphisms.
\item Let $S\in(L^p(\mu))^{**}$. Since $S\circ T_q\in(L^q(\mu))^*$, by (i), $f\in L^p(\mu)$, $S(T_qg)=(T_pf)(g)=\int_Xfg\,d\mu$ $\forall g\in L^q(\mu)$.\\
For $\Phi\in(L^p(\mu))^*$, by (i) $\Phi=T_qg$; $S(\Phi)=S(T_qg)=\int_Xfg\,d\mu=\Phi(f)=Jf(\Phi)$. Hence $S=Jf$, $J$ is onto.
\end{enumerate}
\hlt{Structural Commentary}: Represent both dual layers by conjugate pairings.\\
\hlt{Key Insight}: Every bidual functional is evaluation at an $L^p$ function.
\end{proof}
\end{theorem}

\begin{definition} \label{def:weak-sequential-compactness} \hlt{Weak Sequential Compactness}\\
Let $\mathcal{X}$ be a normed space and $K\subseteq\mathcal{X}$.\\
Then $K$ is \hlt{weakly sequentially compact} if every $(x_n)\subseteq K$ admits $n_k\uparrow\infty$ and $x\in K$ with $x_{n_k}\rightharpoonup x$ in $\mathcal{X}$.
\end{definition}

\begin{theorem} \label{thm:lp-weak-sequential-compactness-bounded-sequences} \hlt{Riesz Weak Compactness Theorem}\\
Let $(X,\mathcal M,\mu)$ be $\sigma$-finite, $1<p<+\infty$, $q=\frac{p}{p-1}$, and $(f_n)\subseteq L^p(\mu)$ be real-valued with $M\defeq\sup\limits_n\norm{f_n}_p<+\infty$.\\
Then there exist $n_k\uparrow\infty$ and $f\in L^p(\mu)$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f_{n_k}\rightharpoonup f$ in $L^p(\mu)$;
\item $\int_X f_{n_k}g\,d\mu\to\int_Xfg\,d\mu$ for all $g\in L^q(\mu)$.
\end{enumerate}
\begin{proof}
Set $B_M\defeq\{h\in L^p(\mu):\norm{h}_p\leq M\}$. By Theorem \ref{thm:lp-reflexive-sigma-finite}, $L^p(\mu)$ is reflexive; hence $B_M$ is weakly compact. By Eberlein--Smulian, $B_M$ is weakly sequentially compact.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $(f_n)\subseteq B_M$ and $B_M$ is weakly sequentially compact, $\exists n_k\uparrow\infty$ and $f\in B_M$ with $f_{n_k}\rightharpoonup f$ in $L^p(\mu)$.
\item From (i) and Theorem \ref{thm:lp-weak-convergence-dual-tests}, $\forall g\in L^q(\mu)$, $\int_X f_{n_k}g\,d\mu\to\int_Xfg\,d\mu$.
\end{enumerate}
\hlt{Structural Commentary}: Reflexive bounded balls are weakly sequentially compact.\\
\hlt{Key Insight}: Bounded $L^p$ sequences have conjugate-test convergent subsequences.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:lp-closed-unit-ball-weakly-sequentially-compact} \hlt{The $L^p$ Unit Ball Is Weakly Sequentially Compact}\\
Let $(X,\mathcal M,\mu)$ be $\sigma$-finite, $1<p<+\infty$, and $B_p\defeq\{f\in L^p(\mu):\norm{f}_p\leq1\}$.\\
Then $B_p$ is weakly sequentially compact in $L^p(\mu)$.
\begin{proof} Let $(f_n)\subseteq B_p$. By Theorem \ref{thm:lp-weak-sequential-compactness-bounded-sequences}, $\exists n_k\uparrow\infty$ and $f\in L^p(\mu)$ with $f_{n_k}\rightharpoonup f$.\\
By Proposition \ref{prop:lp-weak-limits-bounded-lsc}, $\norm{f}_p\leq\liminf\limits_{k\to\infty}\norm{f_{n_k}}_p\leq1$, so $f\in B_p$.\\
\hlt{Structural Commentary}: Weak limits stay in $B_p$ by lower semicontinuity.\\
\hlt{Key Insight}: Closed norm balls are weakly sequentially closed.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:radon-riesz-lp} \hlt{Radon--Riesz Principle}\\
Let $(X,\mathcal M,\mu)$ be a measure space, $1<p<+\infty$, $(f_n)\subseteq L^p(\mu)$, and $f\in L^p(\mu)$, all real-valued.\\
Then $f_n\rightharpoonup f$ and $\norm{f_n}_p\to\norm{f}_p$ iff $\norm{f_n-f}_p\to0$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\norm{f}_p=0$, then $\norm{f_n-f}_p\leq\norm{f_n}_p+\norm{f}_p\to0$. Let $a\defeq\norm{f}_p>0$.\\
Otherwise, after subsequences, $\norm{f_n-f}_p\geq\varepsilon>0$ and, with $a_n\defeq\norm{f_n}_p$, $u_n\defeq\frac{f_n}{a_n}$, $u\defeq\frac{f}{a}$, one has $u_n\rightharpoonup u$, $\norm{u_n}_p=\norm{u}_p=1$, and $\norm{u_n-u}_p\geq\eta>0$. By uniform convexity, $\norm{\frac{u_n+u}{2}}_p\leq1-\delta$.\\
Since $\frac{u_n+u}{2}\rightharpoonup u$, weak lower semicontinuity implies $1=\norm{u}_p\leq\liminf\limits_n\norm{\frac{u_n+u}{2}}_p\leq1-\delta$, contradiction.
\item[$(\Leftarrow)$] Assume $\norm{f_n-f}_p\to0$. Then $\abs{\norm{f_n}_p-\norm{f}_p}\leq\norm{f_n-f}_p\to0$.\\
For $\Lambda\in(L^p(\mu))^*$, $\abs{\Lambda(f_n)-\Lambda(f)}\leq\norm{\Lambda}\norm{f_n-f}_p\to0$, hence $f_n\rightharpoonup f$.
\end{enumerate}
\hlt{Structural Commentary}: Uniform convexity turns weak lower semicontinuity into strong convergence.\\
\hlt{Key Insight}: Weak direction plus norm size fixes the vector.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:radon-riesz-l2-proof-identity} \hlt{$L^2$ Proof of Radon--Riesz}\\
For $p=2$, $f_n\rightharpoonup f$ in $L^2(\mu)$, $\norm{f_n}_2\to\norm{f}_2$. Then $\int_Xf_nf\,d\mu\to\norm{f}_2^2$,\\
$\norm{f_n-f}_2^2=\norm{f_n}_2^2-2\int_Xf_nf\,d\mu+\norm{f}_2^2\to0$. The converse follows from norm convergence in $L^2(\mu)$.\\
\hlt{Key Insight}: Weak convergence fixes the cross term.
\end{remark}

\begin{corollary} \label{cor:radon-riesz-strong-subsequence-criterion} \hlt{Strong Subsequence Criterion}\\
Let $(X,\mathcal M,\mu)$ be a measure space, $1<p<+\infty$, $(f_n)\subseteq L^p(\mu)$, and $f_n\rightharpoonup f$ in real $L^p(\mu)$.\\
Then $\exists n_k\uparrow\infty$ with $\norm{f_{n_k}-f}_p\to0$ iff $\norm{f}_p=\liminf\limits_{n\to\infty}\norm{f_n}_p$.
\begin{proof}
Set $L\defeq\liminf\limits_{n\to\infty}\norm{f_n}_p$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\norm{f_{n_k}-f}_p\to0$, then $\norm{f_{n_k}}_p\to\norm{f}_p$.\\
Since $L\leq\lim\limits_k\norm{f_{n_k}}_p=\norm{f}_p$ and, by Proposition \ref{prop:lp-weak-limits-bounded-lsc}, $\norm{f}_p\leq L$, one has $L=\norm{f}_p$.
\item[$(\Leftarrow)$] Assume $L=\norm{f}_p$. Choose $n_k\uparrow\infty$ with $\norm{f_{n_k}}_p\to L$.\\
Since $f_{n_k}\rightharpoonup f$ and $\norm{f_{n_k}}_p\to\norm{f}_p$, Theorem \ref{thm:radon-riesz-lp} implies $\norm{f_{n_k}-f}_p\to0$.
\end{enumerate}
\hlt{Structural Commentary}: Match the liminf subsequence to Radon--Riesz.\\
\hlt{Key Insight}: Strong subsequences occur exactly when no norm is lost.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:dunford-pettis-l1-finite-measure} \hlt{Dunford--Pettis Criterion in $L^1$}\\
Let $(X,\mathcal{M},\mu)$ be a finite measure space and let $(f_n)$ be bounded in $L^1(\mu)$. The following are equivalent.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(f_n)$ is uniformly integrable.
\item $\forall(f_{n_k})\subseteq(f_n)\ \exists(f_{n_{k_j}})\subseteq(f_{n_k})$ converging weakly in $L^1(\mu)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Let $(u_j)$ be a subsequence of $(f_n)$, set $C\defeq\sup_j\norm{u_j}_1$, and put $T_m(t)\defeq(-m)\vee(t\wedge m)$.\\
By Proposition \ref{prop:uniform-integrability-large-value-tails}, $\sup_j\norm{u_j-T_m(u_j)}_1\to0$. For fixed $m$ and sign $\pm$, $\norm{T_m(u_j^\pm)}_2^2\leq m\norm{u_j^\pm}_1\leq mC$.\\
By bounded Hilbert weak subsequence extraction and diagonalization over $(m,\pm)$, pass to a subsequence, still $(u_j)$, such that $\int_ET_m(u_j^\pm)\,d\mu$ converges for every $E\in\mathcal M$ and $m\in\N$. Set $\nu_j^\pm(E)\defeq\int_Eu_j^\pm\,d\mu$.\\
For $r,s,m$ and $E\in\mathcal M$, $\abs{\nu_r^\pm(E)-\nu_s^\pm(E)}\leq\abs{\int_E(T_m(u_r^\pm)-T_m(u_s^\pm))\,d\mu}+2\sup_j\norm{u_j^\pm-T_m(u_j^\pm)}_1$.\\
Hence $(\nu_j^\pm(E))_j$ is Cauchy. Put $\nu^\pm(E)\defeq\lim_j\nu_j^\pm(E)$.\\
By Theorem \ref{thm:vitali-hahn-saks-setwise-convergence}, $\nu^\pm$ are finite measures and $\nu^\pm\ll\mu$. Write $h^\pm=\frac{d\nu^\pm}{d\mu}$ and $h\defeq h^+-h^-\in L^1(\mu)$.\\
Then $\int_X\varphi u_j\,d\mu\to\int_X\varphi h\,d\mu$ for simple $\varphi$; by Lemma \ref{lem:bounded-simple-approximation-brackets} and $\sup_j\norm{u_j}_1+\norm{h}_1<+\infty$, the convergence holds for every $\varphi\in L^\infty(\mu)$. Thus $u_j\rightharpoonup h$ in $L^1(\mu)$.
\item[$(\Leftarrow)$] Suppose $(f_n)$ is not uniformly integrable. Since finite subfamilies are uniformly integrable by Proposition \ref{prop:l1-uniform-integrability-basic-sources}(ii), choose a subsequence $(u_j)$ and sets $A_j\in\mathcal M$ with $\mu(A_j)\to0$ and $\int_{A_j}\abs{u_j}\,d\mu\geq\varepsilon_0>0$.\\
By (ii), some $(u_{j_\ell})$ converges weakly in $L^1(\mu)$ to $u$. Set $\eta_\ell(E)\defeq\int_Eu_{j_\ell}\,d\mu$ and $\eta(E)\defeq\int_Eu\,d\mu$.\\
Then $\eta_\ell(E)\to\eta(E)$ for every $E\in\mathcal M$, and $\sup_\ell\abs{\eta_\ell}(X)=\sup_\ell\norm{u_{j_\ell}}_1<+\infty$ by Theorem \ref{thm:integral-charge-variation-density}.\\
The signed form of Theorem \ref{thm:vitali-hahn-saks-setwise-convergence} implies $\eta_\ell\ll_{\mathrm u}\mu$ in variation:\\
$\forall\varepsilon>0\ \exists\delta>0$ such that $\mu(A)<\delta\Rightarrow\sup_\ell\abs{\eta_\ell}(A)<\varepsilon$. Since $\abs{\eta_\ell}(A)=\int_A\abs{u_{j_\ell}}\,d\mu$, the subsequence $(u_{j_\ell})$ is uniformly integrable, contradicting $\int_{A_{j_\ell}}\abs{u_{j_\ell}}\,d\mu\geq\varepsilon_0$ and $\mu(A_{j_\ell})\to0$.
\end{enumerate}
\hlt{Structural Commentary}: Truncate, pass to setwise limits, apply VHS.\\
\hlt{Key Insight}: UI is weak sequential compactness in finite $L^1$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:dominated-l1-weak-subsequence} \hlt{Dominated $L^1$ Families Have Weak Subsequences}\\
Let $(X,\mathcal M,\mu)$ be finite. If $g\in L^1(\mu)$, $\forall n \abs{f_n}\leq g$ a.e., then $(f_n)$ has subsequence that converges weakly in $L^1(\mu)$.
\begin{proof}
Since $\abs{f_n}\leq g$ a.e., $\norm{f_n}_1\leq\norm g_1$ $\forall n$, $(f_n)$ is bounded in $L^1(\mu)$. By Proposition \ref{prop:l1-uniform-integrability-basic-sources}(i), $\{f_n:n\geq1\}$ is uniformly integrable. Theorem \ref{thm:dunford-pettis-l1-finite-measure}, $(i)\Rightarrow(ii)$, yields a weakly convergent subsequence.\\
\hlt{Structural Commentary}: Domination verifies Dunford--Pettis through a common integrable envelope.\\
\hlt{Key Insight}: One $L^1$ envelope prevents concentration throughout the sequence.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:bounded-lp-weak-l1-subsequence} \hlt{Bounded $L^p$ Families Are Weakly Compact in $L^1$}\\
Let $(X,\mathcal M,\mu)$ be finite, let $1<p<+\infty$, and let $(f_n)$ be bounded in $L^p(\mu)$.\\
Then $(f_n)$ has a subsequence that converges weakly in $L^1(\mu)$.
\begin{proof}
Let $q$ be conjugate to $p$ and $M\defeq\sup_n\norm{f_n}_p<+\infty$. By H\"older, $\norm{f_n}_1\leq M\mu(X)^{\frac{1}{q}}$ $\forall n$, so $(f_n)$ is bounded in $L^1(\mu)$. By Corollary \ref{cor:lp-bounded-families-uniformly-integrable}, $\{f_n:n\geq1\}$ is uniformly integrable.\\
Hence Theorem \ref{thm:dunford-pettis-l1-finite-measure}, $(i)\Rightarrow(ii)$, yields a weakly convergent subsequence in $L^1(\mu)$.\\
\hlt{Structural Commentary}: H\"older gives $L^1$ boundedness, uniform integrability before Dunford--Pettis is applied.\\
\hlt{Key Insight}: On finite measure spaces, an $L^p$ bound with $p>1$ rules out $L^1$ concentration.
\end{proof}
\end{corollary}

\begin{example} \label{ex:l1-concentration-no-weak-subsequence} \hlt{A Bounded $L^1$ Sequence with No Weakly Convergent Subsequence}\\
In $L^1[0,1]$, $I_n=[0,\frac{1}{n}]$ $f_n=n\mathbf{1}_{I_n}$. Then $\norm{f_n}_1=1$ $\forall n$, and no subsequence of $(f_n)$ converges weakly in $L^1[0,1]$.
\begin{proof}
Assume $f_{n_k}\rightharpoonup f$ in $L^1[0,1]$. For $0<c<d\leq1$, $\mathbf{1}_{[c,d]}\in L^\infty[0,1]$ and $[c,d]\cap I_{n_k}=\varnothing$ for $k$ large.\\
Hence $\int_c^d f\,dm=\lim\limits_{k\to\infty}\int_c^d f_{n_k}\,dm=0$. Thus $f=0$ a.e. on $(0,1]$.\\
Testing by $\mathbf{1}_{[0,1]}$ yields $0=\int_0^1f\,dm=\lim\limits_{k\to\infty}\int_0^1f_{n_k}\,dm=1$, contradiction.\\
\hlt{Key Insight}: Unit mass concentrates on shrinking supports.
\end{proof}
\end{example}

\begin{remark} \label{rem:l1-concentration-and-uniform-integrability} \hlt{Endpoint Failure and Uniform Integrability}\\
For $f_n=n\mathbf{1}_{[0,\frac{1}{n}]}\in L^1[0,1]$, $\norm{f_n}_1=1$ but $\int_{\{f_n>R\}}f_n\,dm=1$ whenever $n>R$.\\
Hence $(f_n)$ is not uniformly integrable; the $p=1$ failure is concentration, not lack of boundedness.
\end{remark}

\begin{theorem} \label{thm:banach-saks-lp} \hlt{Banach--Saks Theorem}\\
Let $(X,\mathcal M,\mu)$ be a measure space, $1<p<+\infty$, $(f_n)\subseteq L^p(\mu)$, and $f_n\rightharpoonup f$ in real $L^p(\mu)$.\\
Then $\exists n_k\uparrow\infty$ such that, with $A_k\defeq\frac{1}{k}\sum_{j=1}^kf_{n_j}$, $\norm{A_k-f}_p\to0$.
\begin{proof}
Set $h_n\defeq f_n-f$. Then $h_n\rightharpoonup0$ and $\sup\limits_n\norm{h_n}_p<+\infty$ by Proposition \ref{prop:lp-weak-limits-bounded-lsc}.\\
Since real $L^p(\mu)$ is uniformly convex, by the selection lemma there is $n_k\uparrow\infty$ with $\norm{\frac{1}{k}\sum_{j=1}^kh_{n_j}}_p\to0$.\\
Since $\frac{1}{k}\sum_{j=1}^kh_{n_j}=A_k-f$, $\norm{A_k-f}_p\to0$.\\
\hlt{Structural Commentary}: Reduce to weakly null; select Cesaro means by uniform convexity.\\
\hlt{Key Insight}: Convex averaging turns weak nullity into norm decay.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:banach-saks-l2-selection-estimate} \hlt{The $L^2$ Selection Estimate}\\
For $p=2$, replace $f_n$ by $f_n-f$ and assume $f_n\rightharpoonup0$ in $L^2(\mu)$ with $\norm{f_n}_2^2\leq M$. Set $n_1=1$ and $S_k\defeq\sum_{j=1}^kf_{n_j}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Given $S_k$, choose $n_{k+1}>n_k$ so that $\abs{\int_X S_kf_{n_{k+1}}\,d\mu}\leq1$.
\item Then $\norm{S_{k+1}}_2^2=\norm{S_k}_2^2+2\int_X S_kf_{n_{k+1}}\,d\mu+\norm{f_{n_{k+1}}}_2^2\leq\norm{S_k}_2^2+2+M$.
\item Hence $\norm{S_k/k}_2^2\leq\frac{2+M}{k}$ by induction.
\end{enumerate}
Thus $S_k/k\to0$ in $L^2(\mu)$.\\
\hlt{Key Insight}: Weak convergence makes new terms nearly orthogonal to old sums.
\end{remark}

\begin{definition} \label{def:closed-convex-sets-and-convex-functionals} \hlt{Closed Convex Sets and Convex Functionals}\\
Let $C$ be a subset of a normed space $\mathcal{X}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $C$ is \hlt{convex} if $\lambda f+(1-\lambda)g\in C$ whenever $f,g\in C$ and $0\leq\lambda\leq1$.
\item $C$ is \hlt{closed} if every strongly convergent sequence in $C$ has its limit in $C$.
\item A real-valued functional $T:C\to\R$ is \hlt{continuous} if $f_n\to f$ strongly in $C$ implies $T(f_n)\to T(f)$.
\item If $C$ is convex, then $T$ is \hlt{convex} if
$T(\lambda f+(1-\lambda)g)\leq\lambda T(f)+(1-\lambda)T(g)$
$\forall f,g\in C$ and $0\leq\lambda\leq1$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:lp-closed-convex-basic-sets} \hlt{Basic Closed Convex Sets in $L^p$}\\
Let $E\subseteq\R$ be measurable and $1\leq p<+\infty$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $g\in L^p(E)$ and $g\geq0$ a.e., then $C_g\defeq\{f\in L^p(E):\abs{f}\leq g\ \text{a.e.}\}$ is closed and convex.
\item $B\defeq\{f\in L^p(E):\norm{f}_p\leq1\}$ is closed and convex.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $f,h\in C_g$ and $0\leq\lambda\leq1$, then $\abs{\lambda f+(1-\lambda)h}\leq\lambda\abs f+(1-\lambda)\abs h\leq g$ a.e., so $C_g$ is convex.\\
If $f_n\in C_g$ and $f_n\to f$ in $L^p(E)$, choose $f_{n_k}\to f$ a.e. by Theorem \ref{thm:riesz-fischer-lp-banach}; then $\abs f\leq g$ a.e., so $f\in C_g$.
\item If $f,h\in B$ and $0\leq\lambda\leq1$, then by Minkowski, $\norm{\lambda f+(1-\lambda)h}_p\leq\lambda\norm{f}_p+(1-\lambda)\norm{h}_p\leq1$.\\
If $f_n\in B$ and $f_n\to f$ in $L^p(E)$, then $\norm{f}_p\leq\norm{f-f_n}_p+\norm{f_n}_p\leq\norm{f-f_n}_p+1$, hence $\norm{f}_p\leq1$.
\end{enumerate}
\hlt{Key Insight}: Norm limits preserve a.e. bounds and norm bounds.
\end{proof}
\end{example}

\begin{lemma} \label{lem:banach-saks-weak-limit-closed-convex-lsc} \hlt{Weak Limits and Convex Lower Semicontinuity}\\
Let $E\subseteq\R$ be measurable, $1<p<+\infty$, $C\subseteq L^p(E)$ closed, bounded, and convex.\\
Let $T:C\to\R$ be continuous and convex. If $(f_n)\subseteq C$ and $f_n\rightharpoonup f$ in $L^p(E)$, then, with $\alpha\defeq\liminf\limits_{n\to\infty}T(f_n)$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f\in C$;
\item $T(f)\leq\alpha$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:banach-saks-lp}, choose $n_k\uparrow\infty$ with $a_k\defeq\frac{1}{k}\sum_{j=1}^kf_{n_j}\to f$ in $L^p(E)$.\\
Since $a_k\in\operatorname{co}\{f_{n_1},\ldots,f_{n_k}\}\subseteq C$ and $C$ is closed, $f\in C$.
\item Choose $m_k\uparrow\infty$ with $T(f_{m_k})\to\alpha$.\\
By Theorem \ref{thm:banach-saks-lp}, choose $g_j\defeq f_{m_{k_j}}$ such that $b_k\defeq\frac{1}{k}\sum\limits_{j=1}^kg_j\to f$ in $L^p(E)$.\\
Then $b_k\in C$ and $T(g_j)\to\alpha$; by continuity and convexity, $T(f)=\lim\limits_kT(b_k)\leq\lim\limits_k\frac{1}{k}\sum\limits_{j=1}^kT(g_j)=\alpha$.
\end{enumerate}
\hlt{Structural Commentary}: Banach--Saks supplies strong convex averages; convexity transfers the liminf.\\
\hlt{Key Insight}: Weak limits inherit closed convex constraints and convex lsc.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:direct-method-convex-functional-lp} \hlt{Direct Method for Convex Functionals on $L^p$}\\
Let $E\subseteq\R$ be measurable, $1<p<+\infty$, $C\subseteq L^p(E)$ nonempty, closed, bounded, and convex.\\
Let $T:C\to\R$ be continuous and convex. Then $\exists f_0\in C$ such that $T(f_0)=\inf\limits_{h\in C}T(h)$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\inf_{C}T=-\infty$, choose $f_n\in C$ with $T(f_n)\to-\infty$.\\
Since $C$ is bounded, by Theorem \ref{thm:lp-weak-sequential-compactness-bounded-sequences}, $\exists n_k\uparrow\infty$ and $f\in L^p(E)$ with $f_{n_k}\rightharpoonup f$.\\
By Lemma \ref{lem:banach-saks-weak-limit-closed-convex-lsc}, $f\in C$ and $T(f)\leq\liminf\limits_kT(f_{n_k})=-\infty$, contradicting $T:C\to\R$. Thus $\inf_C T>-\infty$.
\item Let $c\defeq\inf_C T$ and choose $f_n\in C$ with $T(f_n)\to c$.\\
Since $C$ is bounded, by Theorem \ref{thm:lp-weak-sequential-compactness-bounded-sequences}, $\exists n_k\uparrow\infty$ and $f_0\in L^p(E)$ with $f_{n_k}\rightharpoonup f_0$.\\
By Lemma \ref{lem:banach-saks-weak-limit-closed-convex-lsc}, $f_0\in C$ and $T(f_0)\leq\liminf\limits_kT(f_{n_k})=c$. Since $c\leq T(f_0)$, $T(f_0)=c$.
\end{enumerate}
\hlt{Structural Commentary}: Weak compactness supplies a limit; convex lsc keeps the infimum.\\
\hlt{Key Insight}: Direct method equals compactness plus lower semicontinuity.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:integral-convex-functional-unit-ball-minimizer} \hlt{Integral Convex Functionals Have Unit-Ball Minimizers}\\
Let $\mu(E)<+\infty$, $1<p<+\infty$, and $\varphi:\R\to\R$ be continuous convex with $\abs{\varphi(s)}\leq c_1+c_2\abs{s}^p$ for some $c_1,c_2\geq0$. Set $B\defeq\{f\in L^p(E):\norm{f}_p\leq1\}$ and $T(f)\defeq\int_E\varphi(f)\,dm$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T:B\to\R$ is continuous and convex;
\item $\exists f_0\in B$ such that $T(f_0)=\min_{f\in B}T(f)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $f\in B$, $\abs{\varphi(f)}\leq c_1+c_2\abs f^p\in L^1(E)$, so $T(f)\in\R$.\\
Convexity of $\varphi$ implies $T(\lambda f+(1-\lambda)h)\leq\lambda T(f)+(1-\lambda)T(h)$ for $f,h\in B$, $0\leq\lambda\leq1$.\\
If $f_n\to f$ in $L^p(E)$, every subsequence admits $f_{n_k}\to f$ a.e. with $\abs{f_{n_k}}\leq g$ for some $g\in L^p(E)$.\\
Then $\abs{\varphi(f_{n_k})}\leq c_1+c_2g^p\in L^1(E)$, so by dominated convergence $T(f_{n_k})\to T(f)$.\\
By the subsequence criterion, $T(f_n)\to T(f)$.
\item The set $B$ is nonempty, bounded, convex by Minkowski, and closed since $f_n\in B$, $f_n\to f$ in $L^p(E)$ imply $\norm f_p\leq\liminf\limits_n\norm{f_n}_p\leq1$. By Theorem \ref{thm:direct-method-convex-functional-lp}, $\exists f_0\in B$ with $T(f_0)=\min_{f\in B}T(f)$.
\end{enumerate}
\hlt{Structural Commentary}: $p$-growth $\Rightarrow$ continuity; direct method $\Rightarrow$ attainment.\\
\hlt{Key Insight}: Growth converts strong $L^p$ convergence to dominated convergence.
\end{proof}
\end{corollary}
