\subsection{Differentiation of Measures}

\begin{definition} \label{def:lebesgue-averages-and-absolute-continuity} \hlt{Averages and Absolute Continuity}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Average}: for $f\in\mathscr{L}^1(E)$ and measurable $A\subseteq E$ with $m(A)>0$, set $[f:A]\defeq m(A)^{-1}\int_Af\,dm$.
\item \hlt{Density ratio}: for measurable $B$ and cube or ball $Q$, $m(Q)>0$, set $[B:Q]\defeq\frac{m(B\cap Q)}{m(Q)}$.
\item \hlt{Indefinite integral}: for $f\in\mathscr{L}^1([a,b])$, $F(x)=F(a)+\int_a^xf(t)\,dt$.
\item \hlt{Absolute continuity}: $F:[a,b]\to\R$ is absolutely continuous iff $\forall\varepsilon>0\ \exists\delta>0$: if $(a_j,b_j)_{j=1}^N\subseteq[a,b]$ is a finite pairwise disjoint family, then $\sum\limits_j(b_j-a_j)<\delta\Rightarrow\sum\limits_j\abs{F(b_j)-F(a_j)}<\varepsilon$.
\end{enumerate}
\end{definition}

\begin{perspective} \label{per:absolute-continuity-interval-bookkeeping} \hlt{Absolute Continuity as Interval Bookkeeping}\\
If $F(x)=\int_a^x f(t)\,dt$, absolute continuity controls total $F$-variation on disjoint intervals of small total length.
\end{perspective}

\begin{perspective} \label{per:vitali-covering-fivefold-control} \hlt{Vitali Covering Control}\\
Select balls disjointly by decreasing radius.\\
If $B$ is rejected, then $B\cap B_j\neq\emptyset$ for some earlier $B_j$ with $r(B)\leq r(B_j)$, hence $B\subseteq5B_j$.\\
\hlt{Key Insight}: Disjoint balls add; fixed dilations cover misses.
\end{perspective}

\begin{definition} \label{def:vitali-coverings-and-density-points} \hlt{Vitali Coverings and Density Points}\\
Let $(M,d)$ be a metric space and let $A\subseteq M$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{V}$ is a \hlt{Vitali covering} of $A$ iff $\forall p\in A\ \forall r>0\ \exists V\in\mathcal{V}$ such that $p\in V$ and $\diam(V)<r$.
\item For measurable $E\subseteq\R^n$, its \hlt{density-point set} is $\operatorname{dp}(E)\defeq\{p\in\R^n:\lim\limits_{Q\downarrow p}[E:Q]=1\}$, with $Q$ cubes, equivalently balls, shrinking to $p$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:vitali-covering-lemma-balls-cubes} \hlt{Vitali Covering Lemma for Balls and Cubes}\\
Let $E\subseteq\R^n$, $m^\ast(E)<+\infty$, and let $\mathcal{V}$ be a Vitali cover of $E$ by closed balls or axis-parallel cubes.\\
Then $\exists(V_j)_{j\geq1}\subseteq\mathcal{V}$ pairwise disjoint with $m^\ast(E\setminus\bigcup\limits_{j\geq1}V_j)=0$.
\begin{proof}
Fix open $O\supseteq E$ with $m(O)<m^\ast(E)+\varepsilon$, and replace $\mathcal{V}$ by $\{V\in\mathcal{V}:V\subseteq O\}$.\\
Greedily choose a maximal disjoint $(V_j)$ by dyadic diameter scales.\\
For a fixed dilation $V\mapsto V^\ast$, if $W\cap V\neq\emptyset$ and $\diam W\leq2\diam V$, then $W\subseteq V^\ast$ and $m(V^\ast)\leq C_nm(V)$.\\
Let $R=E\setminus\bigcup\limits_jV_j$. Fix $N$ and $p\in R$.\\
Since $V_1,\ldots,V_N$ are closed, choose $W\in\mathcal{V}$ with $p\in W$ and $W\cap\bigcup\limits_{j\leq N}V_j=\emptyset$. Maximality gives some $j>N$ with $W\cap V_j\neq\emptyset$ and comparable diameter, hence $p\in W\subseteq V_j^\ast$. Thus $R\subseteq\bigcup\limits_{j>N}V_j^\ast$.\\
Therefore $m^\ast(R)\leq C_n\sum\limits_{j>N}m(V_j)\to0$, because $\sum\limits_jm(V_j)=m(\bigsqcup\limits_jV_j)\leq m(O)<+\infty$. Hence $m^\ast(R)=0$.\\
\hlt{Structural Commentary}: Vitali selection gives a disjoint cover modulo null sets.\\
\hlt{Key Insight}: $R\subseteq\bigcup\limits_{j>N}V_j^\ast$ and disjoint tails vanish.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:lebesgue-density-theorem} \hlt{Lebesgue Density Theorem}\\
For measurable $E\subseteq\R^n$, $\lim\limits_{Q\downarrow p}[E:Q]=\mathbf{1}_E(p)$ for a.e. $p$, with $Q\downarrow p$ through cubes or balls.
\begin{proof}
Assume $m(E)<+\infty$. For $a\in(0,1)$ set $E_a\defeq\{p\in E:\liminf\limits_{Q\downarrow p}[E:Q]<a\}$.\\
Cubes $Q\ni p$ with $[E:Q]<a$ form a Vitali cover of $E_a$. For open $O\supseteq E_a$ with $m(O)<m(E_a)+\varepsilon$, choose disjoint $(Q_j)\subseteq O$ covering $E_a$ modulo null sets by Theorem \ref{thm:vitali-covering-lemma-balls-cubes}.\\
Hence $m(E_a)\leq\sum\limits_jm(E_a\cap Q_j)\leq\sum\limits_jm(E\cap Q_j)<a\sum\limits_jm(Q_j)\leq a\,m(O)<a(m(E_a)+\varepsilon)$.\\
Let $\varepsilon\downarrow0$; since $a<1$, $m(E_a)=0$. Thus $p\in E\setminus\bigcup\limits_{a\in\Q\cap(0,1)}E_a$ has density $1$. Apply same argument to bounded pieces of $E^c$ to get density $0$ a.e. on $E^c$; then exhaust arbitrary $E$ by bounded pieces.\\
\hlt{Structural Commentary}: Measurable sets are locally $0$ or $1$ a.e.\\
\hlt{Key Insight}: Vitali turns bad density into $m(E_a)\le a\,m(E_a)$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:density-points-represent-measurable-set} \hlt{Density Points Represent a Measurable Set}\\
For measurable $E\subseteq\R^n$, $m(E\triangle\operatorname{dp}(E))=0$.
\begin{proof} By Theorem \ref{thm:lebesgue-density-theorem}, $[E:Q]\to1$ at a.e. point of $E$, $[E:Q]\to0$ at a.e. point of $E^c$; hence $E\triangle\operatorname{dp}(E)$ null.\\
\hlt{Structural Commentary}: The pointwise density dichotomy is converted into equality modulo null sets.\\
\hlt{Key Insight}: Density-one membership recovers the measurable set almost everywhere.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:density-points-commute-with-slices} \hlt{Density Points Commute with Slicing Almost Everywhere}\\
Let $E\subseteq\R^n\times\R^k$ be measurable. For a.e. $x\in\R^n$, $\operatorname{dp}(E_x)=\{y:(x,y)\in\operatorname{dp}(E)\}$ up to a null subset of $\R^k$.
\begin{proof}
Set $D=\operatorname{dp}(E)$ and $N=E\triangle D$.\\
By Corollary \ref{cor:density-points-represent-measurable-set}, $m_{n+k}(N)=0$; hence $m_k(N_x)=0$ for a.e. $x$ by Theorem \ref{thm:zero-slices-null-set-criterion}.\\
Intersecting with the Fubini good set, $E_x,D_x$ are measurable and $m_k(E_x\triangle D_x)=m_k((E\triangle D)_x)=0$.\\
$m_k(E_x\triangle\operatorname{dp}(E_x))=0$ by Corollary \ref{cor:density-points-represent-measurable-set} in $\R^k$. Therefore $m_k(\operatorname{dp}(E_x)\triangle D_x)=0$.\\
\hlt{Structural Commentary}: Almost-everywhere local fullness is compatible with almost-everywhere slicing.\\
\hlt{Key Insight}: Null ambient errors have null slices for almost every base point.
\end{proof}
\end{corollary}

\begin{lemma} \label{lem:large-average-set-bound} \hlt{Large Averages Have Small Outer Measure}\\
Let $Q_0\subseteq\R^n$ be a cube, let $0\leq g\in\mathscr{L}^1(Q_0)$, and let $\lambda>0$.\\
Set $A_\lambda\defeq\{p\in Q_0:\forall r>0\ \exists\text{ a cube }Q\subseteq Q_0,\ p\in Q,\ \diam Q<r,\ [g:Q]>\lambda\}$. Then $m^\ast(A_\lambda)\leq\frac{1}{\lambda}\int_{Q_0}g\,dm$.
\begin{proof}
Let $\mathcal{V}_\lambda\defeq\{Q\subseteq Q_0:Q\text{ is a cube and }[g:Q]>\lambda\}$. This is a Vitali cover of $A_\lambda$; choose disjoint $(Q_j)\subseteq\mathcal{V}_\lambda$ covering $A_\lambda$ modulo a null set. Then $\lambda m^\ast(A_\lambda)\leq\lambda\sum\limits_jm(Q_j)<\sum\limits_j\int_{Q_j}g\,dm\leq\int_{Q_0}g\,dm$.\\
\hlt{Structural Commentary}: Weak type for local averages.\\
\hlt{Key Insight}: Disjoint Vitali cubes turn pointwise large averages into one integral bound.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:lebesgue-average-value-theorem} \hlt{Lebesgue Average Value Theorem}\\
For $f\in\mathscr{L}^1_{\mathrm{loc}}(\R^n)$, $\lim\limits_{Q\downarrow p}[f:Q]=f(p)$ for a.e. $p\in\R^n$ with $Q$ cubes, equivalently balls, shrinking to $p$.
\begin{proof} Fix cube $Q_0$; to prove claim for a.e. $p\in Q_0^\circ$. For $\alpha>0$, let $B_\alpha=\left\{p\in Q_0^\circ:\limsup\limits_{Q\downarrow p}|[f:Q]-f(p)|>\alpha\right\}$.\\
Choose $h\in C(Q_0)$ with $\|f-h\|_{L^1(Q_0)}<\eta$, and set $g=|f-h|$. Since $[h:Q]\to h(p)$, $B_\alpha\subseteq \{g>\frac{\alpha}{3}\}\cup A_{\frac{\alpha}{3}}(g)$.\\
By Markov and Lemma \ref{lem:large-average-set-bound}, $m^\ast(B_\alpha)\le \frac{3}{\alpha}\int_{Q_0}g+\frac{3}{\alpha}\int_{Q_0}g <\frac{6\eta}{\alpha}$. Let $\eta\downarrow0$; then $m(B_\alpha)=0$.\\
Hence $B(Q_0)\defeq\{p\in Q_0^\circ:\limsup\limits_{Q\downarrow p}|[f:Q]-f(p)|>0\}=\bigcup\limits_{\alpha\in\Q_{>0}}B_\alpha$ is null.\\
With $Q_N\defeq[-N,N]^n$, $\R^n=\bigcup\limits_NQ_N^\circ$, so $\bigcup\limits_NB(Q_N)$ is null.\\
\hlt{Structural Commentary}: Local averages recover the original integrable function almost everywhere.\\
\hlt{Key Insight}: Continuous approximation handles the good part; Vitali's estimate makes the bad part small.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:lebesgue-antiderivative-theorem} \hlt{Lebesgue's Antiderivative Theorem}\\
For $f\in\mathscr{L}^1([a,b])$, define $F(x)\defeq c+\int_a^x f(t)\,dt$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $F\in AC([a,b])$.
\item $F'=f$ a.e. on $[a,b]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Given $\varepsilon>0$, choose $\delta>0$ such that $m(A)<\delta$ implies $\int_A\abs f\,dm<\varepsilon$.\\
If $(\alpha_j,\beta_j)$ are disjoint and $\sum\limits_j(\beta_j-\alpha_j)<\delta$, then $\sum\limits_j\abs{F(\beta_j)-F(\alpha_j)}\leq\int_{\bigcup_j(\alpha_j,\beta_j)}\abs f\,dm<\varepsilon$.
\item For $x,x+h\in[a,b]$, let $I_h(x)\defeq(\min\{x,x+h\},\max\{x,x+h\})$. Then $\frac{F(x+h)-F(x)}{h}=[f:I_h(x)]$.\\
Extend $f$ by $0$ outside $[a,b]$; by Theorem \ref{thm:lebesgue-average-value-theorem} in one dimension, $[f:I_h(x)]\to f(x)$ for a.e. $x$.
\end{enumerate}
\hlt{Structural Commentary}: The indefinite Lebesgue integral is the correct antiderivative for integrable functions.\\
\hlt{Key Insight}: Differentiating the integral is the same as shrinking the averaging interval.
\end{proof}
\end{theorem}
