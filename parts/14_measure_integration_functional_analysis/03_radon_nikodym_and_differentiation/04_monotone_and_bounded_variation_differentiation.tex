\subsection{Monotone and Bounded-Variation Differentiation}

Monotone functions are the one-dimensional model case for differentiation almost everywhere. Bounded-variation functions enter by Jordan decomposition: finite oscillation is exactly enough to write a function as the difference of two monotone functions.

\begin{definition} \label{def:bounded-variation-one-variable} \hlt{Bounded Variation}\\
Let $F:[a,b]\to\R$. For $a\leq u<v\leq b$, let $\mathcal{P}[u,v]\defeq\{P=(x_0,\ldots,x_N):N\in\N,\ u=x_0<\cdots<x_N=v\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $P=(x_0,\ldots,x_N)\in\mathcal{P}[u,v]$, define $V(F,P)\defeq\sum\limits_{j=1}^N\abs{F(x_j)-F(x_{j-1})}$.
\item Define $V_u^v(F)\defeq\sup\limits_{P\in\mathcal{P}[u,v]}V(F,P)$ for $u<v$, and $V_u^u(F)\defeq0$.
\item $F$ has \hlt{bounded variation} on $[a,b]$ iff $F\in BV([a,b])\defeq\{G:[a,b]\to\R:V_a^b(G)<+\infty\}$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:jordan-decomposition-bounded-variation-functions} \hlt{Variation Function and Jordan Decomposition}\\
Let $F:[a,b]\to\R$. For $F\in BV([a,b])$, set $T_F:[a,b]\to\R$ by $T_F(a)\defeq0$ and $T_F(x)\defeq V_a^x(F)$ for $a<x\leq b$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Variation additivity}: If $F\in BV([a,b])$, then $T_F(v)-T_F(u)=V_u^v(F)$ for $a\leq u\leq v\leq b$.
\item \hlt{Variation monotonicity}: If $F\in BV([a,b])$, then $T_F$ is nondecreasing.
\item \hlt{Compensated monotonicity}: If $F\in BV([a,b])$, then $F+T_F$ is nondecreasing.
\item \hlt{Decomposition}: If $F\in BV([a,b])$, then $F=(F+T_F)-T_F$ is a difference of nondecreasing functions.
\item \hlt{Converse}: If $F=G-H$ for nondecreasing $G,H:[a,b]\to\R$,\\
then $F\in BV([a,b])$ and $V_a^b(F)\leq G(b)-G(a)+H(b)-H(a)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Assume $F\in BV([a,b])$ and fix $a\leq u\leq v\leq b$. The cases $u=a$ and $u=v$ are immediate; for $a<u<v$, concatenation gives $V_a^v(F)\geq V_a^u(F)+V_u^v(F)$, while inserting $u$ into $P\in\mathcal{P}[a,v]$ and using the triangle inequality gives $V(F,P)\leq V_a^u(F)+V_u^v(F)$. Taking suprema yields $V_a^v(F)=V_a^u(F)+V_u^v(F)$, QED.
\item By (i), $T_F(v)-T_F(u)=V_u^v(F)\geq0$.
\item Since $\abs{F(v)-F(u)}\leq T_F(v)-T_F(u)$, $(F+T_F)(v)-(F+T_F)(u)\geq0$.
\item The identity $F=(F+T_F)-T_F$ and (ii)--(iii) give the required monotone difference.
\item Assume $F=G-H$ for nondecreasing $G,H:[a,b]\to\R$.\\
For $P=(x_0,\ldots,x_N)\in\mathcal{P}[a,b]$, by monotonicity and triangle inequality,\\
$V(F,P)\leq\sum\limits_{j=1}^N(G(x_j)-G(x_{j-1}))+\sum\limits_{j=1}^N(H(x_j)-H(x_{j-1}))=G(b)-G(a)+H(b)-H(a)$.\\
Taking the supremum over $P$ gives $V_a^b(F)\leq G(b)-G(a)+H(b)-H(a)<+\infty$.
\end{enumerate}
\hlt{Structural Commentary}: Bounded variation is exactly controlled monotone difference.\\
\hlt{Key Insight}: The variation function supplies the missing monotone part.
\end{proof}
\end{theorem}

\begin{example} \label{ex:bounded-variation-basic-examples} \hlt{Basic Bounded-Variation Tests}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $F:[a,b]\to\R$ is nondecreasing, then $F\in BV([a,b])$ and $V_a^b(F)=F(b)-F(a)$.
\item If $F:[a,b]\to\R$ is $L$-Lipschitz, then $F\in BV([a,b])$ and $V_a^b(F)\leq L(b-a)$.
\item Let $F:[0,1]\to\R$ satisfy $F(0)=0$ and $F(x)=x\cos(\frac{\pi}{2x})$ for $0<x\leq1$. Then $F$ is continuous and $F\notin BV([0,1])$: for $P_n=(0,\frac{1}{2n},\frac{1}{2n-1},\ldots,\frac{1}{2},1)\in\mathcal{P}[0,1]$, $V(F,P_n)=\sum\limits_{k=1}^n\frac{1}{k}\to+\infty$.
\end{enumerate}
\end{example}

\begin{definition} \label{def:upper-lower-derivatives} \hlt{Upper and Lower Derivatives}\\
Let $I\subseteq\R$ be an interval, $F:I\to\R$, and $x\in I^\circ$.\\
For $h>0$, set $Q_h(x)\defeq\{t\in\R:0<\abs{t}\leq h,\ x+t\in I\}$ and $\Delta_tF(x)\defeq\frac{F(x+t)-F(x)}{t}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{upper derivative} at $x$ is $\overline{D}F(x)\defeq\lim\limits_{h\downarrow0}\sup\limits_{t\in Q_h(x)}\Delta_tF(x)$.
\item The \hlt{lower derivative} at $x$ is $\underline{D}F(x)\defeq\lim\limits_{h\downarrow0}\inf\limits_{t\in Q_h(x)}\Delta_tF(x)$.
\end{enumerate}
They exist in $\overline{\R}$, satisfy $\underline{D}F(x)\leq\overline{D}F(x)$, and $F'(x)$ exists iff $\underline{D}F(x)=\overline{D}F(x)\in\R$.
\end{definition}

\begin{lemma} \label{lem:monotone-upper-derivative-weak-bound} \hlt{Weak Bound for Upper Derivatives}\\
Let $F:[a,b]\to\R$ be nondecreasing. For $\alpha>0$, set $E_\alpha\defeq\{x\in(a,b):\overline{D}F(x)\geq\alpha\}$.\\
Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $m^\ast(E_\alpha)\leq\frac{F(b)-F(a)}{\alpha}$ for every $\alpha>0$.
\item $m^\ast\{x\in(a,b):\overline{D}F(x)=+\infty\}=0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $\alpha>0$ and $0<\alpha'<\alpha$. For $x\in E_\alpha$ and $\delta>0$, choose $[c,d]\subset(a,b)$ with $x\in[c,d]$, $d-c<\delta$, and $F(d)-F(c)>\alpha'(d-c)$. These intervals form a Vitali cover of $E_\alpha$.\\
By Theorem \ref{thm:vitali-covering-lemma-balls-cubes}, choose pairwise disjoint $I_k=[c_k,d_k]$ covering $E_\alpha$ modulo a null set.\\
Then $\alpha'm^\ast(E_\alpha)\leq\alpha'\sum\limits_k(d_k-c_k)<\sum\limits_k(F(d_k)-F(c_k))\leq F(b)-F(a)$. Let $\alpha'\uparrow\alpha$.
\item Set $E_\infty\defeq\{x\in(a,b):\overline{D}F(x)=+\infty\}$. Since $E_\infty\subseteq E_n$ $\forall n\in\N$, (i) gives $m^\ast(E_\infty)\leq\frac{F(b)-F(a)}{n}\to0$.
\end{enumerate}
\hlt{Structural Commentary}: Vitali selection converts local slope lower bounds into total rise.\\
\hlt{Key Insight}: Large local slopes are paid for by total monotone rise.
\end{proof}
\end{lemma}

\begin{lemma} \label{lem:dini-gap-sets-null-monotone} \hlt{Dini Gap Sets Are Null}\\
Let $F:[a,b]\to\R$ be nondecreasing and let $\alpha<\beta$.\\
Set $E_{\alpha,\beta}\defeq\{x\in(a,b):\underline{D}F(x)<\alpha<\beta<\overline{D}F(x)\}$. Then $m^\ast(E_{\alpha,\beta})=0$.
\begin{proof} If $\alpha\leq0$, then $E_{\alpha,\beta}=\emptyset$ because every secant quotient of nondecreasing $F$ is nonnegative.\\
Assume $0<\alpha<\beta$, set $E=E_{\alpha,\beta}$, and fix $\varepsilon>0$. Choose open $O$ with $E\subseteq O\subseteq(a,b)$ and $m(O)<m^\ast(E)+\varepsilon$.\\
Since $\underline{D}F(x)<\alpha$ on $E$, intervals $[c,d]\subset O$ with $x\in[c,d]$, $F(d)-F(c)<\alpha(d-c)$ form Vitali cover of $E$.\\
Select pairwise disjoint $I_k=[c_k,d_k]$ covering $E$ modulo a null set.\\
$\forall k$, modulo endpoints, $E\cap I_k\subseteq\{x\in(c_k,d_k):\overline{D}(F|_{I_k})(x)\geq\beta\}$, so $m^\ast(E\cap I_k)\leq\frac{F(d_k)-F(c_k)}{\beta}$ by Lemma \ref{lem:monotone-upper-derivative-weak-bound}.\\
Hence $m^\ast(E)\leq\sum\limits_km^\ast(E\cap I_k)\leq\frac{1}{\beta}\sum\limits_k(F(d_k)-F(c_k))<\frac{\alpha}{\beta}\sum\limits_k(d_k-c_k)\leq\frac{\alpha}{\beta}m(O)$.\\
Let $\varepsilon\downarrow0$; then $m^\ast(E)\leq\frac{\alpha}{\beta}m^\ast(E)$, and since $m^\ast(E)<+\infty$ and $\frac{\alpha}{\beta}<1$, $m^\ast(E)=0$.\\
\hlt{Structural Commentary}: Vitali intervals turn a slope gap into a contraction estimate.\\
\hlt{Key Insight}: Dini gaps force excess rise.
\end{proof}
\end{lemma}

\begin{proposition} \label{prop:divided-difference-average-identity} \hlt{Divided Differences and Moving Averages}\\
Let $f\in\mathscr{L}^1([a,b])$, extend $f$ to $[a,b+1]$ by $f(x)=f(b)$ for $b<x\leq b+1$, and fix $0<h\leq1$.\\
For $x\in[a,b]$, set $\operatorname{Diff}_hf(x)\defeq\frac{f(x+h)-f(x)}{h}$ and $\operatorname{Av}_hf(x)\defeq h^{-1}\int_x^{x+h}f\,dm$.\\
Then $\forall a\leq u<v\leq b$, $\int_u^v\operatorname{Diff}_hf\,dm=\operatorname{Av}_hf(v)-\operatorname{Av}_hf(u)$.
\begin{proof} For $a\leq u<v\leq b$, translation invariance gives $\int_u^vf(\cdot+h)\,dm=\int_{u+h}^{v+h}f\,dm$.\\
Hence $\int_u^v\operatorname{Diff}_hf\,dm=h^{-1}(\int_{u+h}^{v+h}f\,dm-\int_u^vf\,dm)=h^{-1}(\int_v^{v+h}f\,dm-\int_u^{u+h}f\,dm)=\operatorname{Av}_hf(v)-\operatorname{Av}_hf(u)$.\\
\hlt{Structural Commentary}: Difference quotients integrate to endpoint moving averages.\\
\hlt{Key Insight}: The interior contributions cancel after translation.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:lebesgue-last-theorem-and-consequences} \hlt{Lebesgue Differentiability Theorem for Monotone Functions}\\
If $F:[a,b]\to\R$ is monotone, then $F'$ exists a.e. on $[a,b]$.
\begin{proof}
Assume $F$ is nondecreasing. Set $G_{\alpha,\beta}\defeq\{\underline{D}F<\alpha<\beta<\overline{D}F\}$ for $\alpha,\beta\in\Q$, $\alpha<\beta$, and $E_\infty\defeq\{\overline{D}F=+\infty\}$. Lemma \ref{lem:dini-gap-sets-null-monotone} gives $m(G_{\alpha,\beta})=0$, and Lemma \ref{lem:monotone-upper-derivative-weak-bound} gives $m(E_\infty)=0$.\\
Outside $E_\infty\cup\bigcup_{\alpha<\beta}G_{\alpha,\beta}$, $\underline{D}F=\overline{D}F\in\R$, so $F'$ exists. If $F$ is nonincreasing, apply the result to $-F$.\\
\hlt{Structural Commentary}: Null Dini gaps plus finite upper derivative give differentiability.\\
\hlt{Key Insight}: Monotone rise controls exceptional slopes.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:lipschitz-and-bv-differentiability} \hlt{Lipschitz and BV Differentiability}\\
Let $F:[a,b]\to\R$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $F$ is Lipschitz, then $F'$ exists a.e. on $[a,b]$.
\item If $F\in BV([a,b])$, then $F'$ exists a.e. on $[a,b]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $F$ is $L$-Lipschitz, then $G_+(x)\defeq F(x)+Lx$ and $G_-(x)\defeq Lx-F(x)$ are nondecreasing.\\
Their derivatives exist a.e. by Theorem \ref{thm:lebesgue-last-theorem-and-consequences}, so $F'=\frac{G_+'-G_-'}{2}$ a.e.
\item By Theorem \ref{thm:jordan-decomposition-bounded-variation-functions}, $F=G-H$ for nondecreasing $G,H$.\\
Their derivatives exist a.e. by Theorem \ref{thm:lebesgue-last-theorem-and-consequences}, so $F'=G'-H'$ a.e.
\end{enumerate}
\hlt{Structural Commentary}: Lipschitz and BV functions are reduced to finite combinations of monotone functions.\\
\hlt{Key Insight}: Almost-everywhere differentiability is stable under finite linear combinations.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:absolute-continuity-bv-and-differentiability} \hlt{Absolute Continuity Implies BV Differentiability}\\
If $F\in AC([a,b])$, then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $F\in BV([a,b])$.
\item $F'$ exists a.e. on $[a,b]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $\delta>0$ from absolute continuity with $\varepsilon=1$, and choose $a=x_0<\cdots<x_N=b$ with $x_j-x_{j-1}<\delta$. After inserting the $x_j$ into a partition $P$, the increments within each $[x_{j-1},x_j]$ have total length below $\delta$ and total $F$-variation below $1$. Thus $V(F,P)\leq N$ and $F\in BV([a,b])$.
\item By (i), $F\in BV([a,b])$; by Corollary \ref{cor:lipschitz-and-bv-differentiability} $\exists G\subseteq[a,b]$, $m([a,b]\setminus G)=0$ and $F'(x)\in\R$ $\forall x\in G$.
\end{enumerate}
\hlt{Structural Commentary}: Absolute continuity bounds variation; BV decomposition then gives differentiability.\\
\hlt{Key Insight}: One absolute-continuity scale controls every partition after a finite interval decomposition.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:monotone-and-bv-derivatives-integrable} \hlt{Derivative Bounds for Monotone and BV Functions}\\
Let $F:[a,b]\to\R$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $F$ is nondecreasing, then $F'\geq0$ a.e.
\item If $F$ is nondecreasing, then $F'\in\mathscr{L}^1([a,b])$ and $\int_a^bF'\,dm\leq F(b)-F(a)$.
\item If $F\in BV([a,b])$, then $F'$ exists a.e.
\item If $F\in BV([a,b])$, then $F'\in\mathscr{L}^1([a,b])$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Every secant quotient of a nondecreasing function is nonnegative; hence $F'\geq0$ a.e. by Theorem \ref{thm:lebesgue-last-theorem-and-consequences}.
\item Extend $F$ by $F(b)$ on $(b,b+1]$, set $g_n(x)\defeq n(F(x+\frac{1}{n})-F(x))$. Then $g_n\geq0$, $g_n\to F'$ a.e. on $(a,b)$.\\
By Proposition \ref{prop:divided-difference-average-identity}, $\int_a^bg_n\,dm=F(b)-n\int_a^{a+\frac{1}{n}}F\,dm\leq F(b)-F(a)$.\\
Fatou's lemma gives $\int_a^bF'\,dm\leq F(b)-F(a)<+\infty$.
\item Corollary \ref{cor:lipschitz-and-bv-differentiability}(ii) gives $\exists E\subseteq[a,b]$ with $m([a,b]\setminus E)=0$ and $F'(x)\in\R$ $\forall x\in E$.
\item Write $F=G-H$ with $G,H$ nondecreasing by Theorem \ref{thm:jordan-decomposition-bounded-variation-functions}. By (ii), $G',H'\in\mathscr{L}^1([a,b])$; on the full-measure set where $G',H'$ exist, $F'=G'-H'$, so $F'\in\mathscr{L}^1([a,b])$.
\end{enumerate}
\hlt{Structural Commentary}: BV inherits finite derivative mass from monotone parts.\\
\hlt{Key Insight}: Fatou turns right-secant bounds into an integral bound for the derivative.
\end{proof}
\end{corollary}
