\subsection{Noetherian Modules}

\begin{theorem} \label{thm:finitely-generated-modules-over-noetherian-rings} \hlt{Finitely Generated Modules over Noetherian Rings}\\
Let $A$ be a nonzero commutative ring with identity. Then $A$ is Noetherian $\Longleftrightarrow(\forall\,M)\,[M\text{ is an f.g. }A\text{-module}\Rightarrow M\text{ is Noetherian}]$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $A$ is Noetherian.
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Finite Free Modules}: For $n\in\N$, set $Q(n)\defeq(\forall\,S\leq A^n)\,S\text{ is f.g.}$; $Q(1)$ holds by Definition \ref{def:noetherian-module-ring-acc}(ii), Theorem \ref{thm:noetherian-iff-submodules-finitely-generated}(i).\\
Fix $n\in\N$, $Q(n)$, $S\leq A^{n+1}$. Let $\pi\in\operatorname{Hom}_A(S,A)$ be last-coordinate projection. Set $K\defeq\{v\in A^n:(v,0_A)\in S\}\subseteq A^n$.\\
$0_{A^n}\in K$ by $(0_{A^n},0_A)=0_{A^{n+1}}\in S$. $(\forall\,(a,b,v,w)\in A^2\times K^2)\,av+bw\in K$ by $(av+bw,0_A)=a(v,0_A)+b(w,0_A)\in S$.\\
$K\leq A^n$ by Theorem \ref{thm:submodule-test}. $\ker\pi\leq S$ by Proposition \ref{prop:hom-module-kernel-image}(ii). Map $v\mapsto(v,0_A)$ identifies $K$ with $\ker\pi$ by definition of $\pi$.\\
Thus $K$ is f.g. by $Q(n)$. The image $\pi(S)\leq{}_A A$ by Proposition \ref{prop:hom-module-kernel-image}(iii). The module $\pi(S)$ is f.g. by $Q(1)$.\\
Choose $p\in\N$ and $(k_1,\ldots,k_p)\in K^p$ by finite generation of $K$. $K=\langle\!\langle k_1,\ldots,k_p\rangle\!\rangle_A$ by the choice of $(k_1,\ldots,k_p)$.\\
Choose $q\in\N$ and $(t_1,\ldots,t_q)\in\pi(S)^q$ by finite generation of $\pi(S)$. $\pi(S)=\langle\!\langle t_1,\ldots,t_q\rangle\!\rangle_A$ by the choice of $(t_1,\ldots,t_q)$.\\
Choose $(s_1,\ldots,s_q)\in S^q$ satisfying $(\forall\,j\in\{1,\ldots,q\})\,\pi(s_j)=t_j$ by the definition of $\pi(S)$.\\
Fix $s\in S$. Choose $(a_1,\ldots,a_q)\in A^q$ satisfying $\pi(s)=\sum\limits_{j=1}^{q}a_jt_j$ by generation of $\pi(S)$.\\
Set $u\defeq s-\sum\limits_{j=1}^{q}a_js_j\in\ker\pi$ by $A$-linearity of $\pi$ and the preceding equality. Write $u=(k,0_A)$ with $k\in K$.\\
Choose $(b_1,\ldots,b_p)\in A^p$ satisfying $k=\sum\limits_{i=1}^{p}b_ik_i$ by generation of $K$.\\
$s=\sum\limits_{j=1}^{q}a_js_j+\sum\limits_{i=1}^{p}b_i(k_i,0_A)$ by the defining equations for $u,k$, and $(b_i)_{i=1}^{p}$.\\
Thus $S$ is f.g. by arbitrariness of $s$, closure of $S$, and Definition \ref{def:cyclic-and-finitely-generated-modules}(ii).\\
$Q(n+1)$ holds by arbitrariness of $S$; $(\forall\,n\in\N)\,Q(n)$ by Theorem \ref{thm:mathematical-induction}; $(\forall\,n\in\N)\,[A^n\text{ is Noetherian}]$ by Theorem \ref{thm:noetherian-iff-submodules-finitely-generated}(i).
\item \hlt{Finite Free Cover}: Fix an f.g. $A$-module $M$. Choose $n\in\N$ and $(v_1,\ldots,v_n)\in M^n$ by finite generation of $M$.\\
$M=\langle\!\langle v_1,\ldots,v_n\rangle\!\rangle_A$ by choice of $(v_1,\ldots,v_n)$. Define $\varphi:A^n\to M$ by $\varphi(r_1,\ldots,r_n)\defeq\sum\limits_{i=1}^{n}r_iv_i$.\\
The map $\varphi$ is $A$-linear by the module laws and surjective by generation of $M$.
\item \hlt{Submodule Preimages}: Fix $N\leq M$ and set $P\defeq\varphi^{-1}(N)\subseteq A^n$. $0_{A^n}\in P$ by $\varphi(0_{A^n})=0_M\in N$.\\
$(\forall\,(a,b,x,y)\in A^2\times P^2)\,ax+by\in P$ by $A$-linearity of $\varphi$ and closure of $N$. $P\leq A^n$ by Theorem \ref{thm:submodule-test}.\\
Module $P$ is f.g. by Noetherianity of $A^n$, Theorem \ref{thm:noetherian-iff-submodules-finitely-generated}(i). Choose $m\in\N$ and $(p_1,\ldots,p_m)\in P^m$ by finite generation of $P$.\\
$P=\langle\!\langle p_1,\ldots,p_m\rangle\!\rangle_A$ by the choice of $(p_1,\ldots,p_m)$. $N=\varphi(P)$ by surjectivity of $\varphi$ and the definition of $P$.\\
$N=\langle\!\langle\varphi(p_1),\ldots,\varphi(p_m)\rangle\!\rangle_A$ by $A$-linearity of $\varphi$ and generation of $P$. Hence $M$ is Noetherian by Theorem \ref{thm:noetherian-iff-submodules-finitely-generated}(i).\\
The forward implication holds by arbitrariness of $M$.
\end{enumerate}
\item[$(\Leftarrow)$] Assume $(\forall\,M)\,[M\text{ is an f.g. }A\text{-module}\Rightarrow M\text{ is Noetherian}]$. Regular module ${}_A A=\langle\!\langle1_A\rangle\!\rangle_A$ is f.g. by Definition \ref{def:cyclic-and-finitely-generated-modules}(ii).\\
It is Noetherian by the hypothesis. Hence $A$ is Noetherian by Definition \ref{def:noetherian-module-ring-acc}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Induction handles $A^n$; free covers transfer Noetherianity to quotients.\\
\hlt{Key Insight}: Generators of $\varphi^{-1}(N)$ map to generators of $N$.
\end{proof}
\end{theorem}
