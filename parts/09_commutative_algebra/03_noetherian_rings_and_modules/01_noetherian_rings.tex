\subsection{Noetherian Rings}

\begin{theorem} \label{thm:hilbert-basis-theorem} \hlt{Hilbert Basis Theorem}\\
Let $A$ be a nonzero commutative ring with identity. If $A$ is Noetherian, then $A[x]$ is Noetherian.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Reduction}: By Theorem \ref{thm:noetherian-iff-submodules-finitely-generated}(ii), to prove $(\forall\,\mathcal{J}\trianglelefteq A[x])\,\mathcal{J}$ is f.g. Fix $\mathcal{J}\trianglelefteq A[x]$. If $\mathcal{J}=0$, then $\mathcal{J}=\langle0\rangle$; assume $\mathcal{J}\neq0$.\\
For $g=\sum\limits_{j=0}^{n}b_jx^j\in A[x]$ with $n\in\N_0$ and $b_n\neq0$, set $\deg g\defeq n$ and $\operatorname{lc}(g)\defeq b_n$.
\item \hlt{Leading-Coefficient Generators}: Set $L\defeq\langle\operatorname{lc}(g):g\in\mathcal{J}\setminus\{0\}\rangle\trianglelefteq A$.\\
The ideal $L$ is f.g. by Noetherianity of $A$ and Theorem \ref{thm:noetherian-iff-submodules-finitely-generated}(ii). Choose generators $\ell_1,\ldots,\ell_s$ of $L$ by finite generation of $L$.\\
Each $\ell_j$ is a finite $A$-linear combination of displayed leading coefficients by Definition \ref{def:ideal-and-generated-ideal}(ii).\\
Write the collected coefficients as $\operatorname{lc}(f_1),\ldots,\operatorname{lc}(f_m)$ with $(f_1,\ldots,f_m)\in(\mathcal{J}\setminus\{0\})^m$.\\
$L=\langle\operatorname{lc}(f_i):1\leq i\leq m\rangle$ by the generator expansions and the definition of $L$.\\
Set $d\defeq\max\limits_{1\leq i\leq m}\deg f_i$ and replace each $f_i$ with $x^{d-\deg f_i}f_i$.\\
$(\forall\,i\in\{1,\ldots,m\})\,[f_i\in\mathcal{J}\land\deg f_i=d]$ by ideal absorption and the definition of $d$.\\
$L=\langle\operatorname{lc}(f_i):1\leq i\leq m\rangle$ by invariance of leading coefficients under these monomial shifts.
\item \hlt{Low-Degree Generators}: For $0\leq k<d$, set $L_k\defeq\langle\operatorname{lc}(g):g\in\mathcal{J},\ \deg g=k\rangle$. Each $L_k$ is f.g. by Theorem \ref{thm:noetherian-iff-submodules-finitely-generated}(ii).\\
Choose finite generators of $L_k$ by its finite generation.\\
Each generator is a finite $A$-linear combination of those coefficients by Definition \ref{def:ideal-and-generated-ideal}(ii).\\
Let $P_k\subseteq\{p\in\mathcal{J}:\deg p=k\}$ be the corresponding finite polynomial set.\\
Then $L_k=\langle\operatorname{lc}(p):p\in P_k\rangle$ by the generator expansions and the definition of $L_k$.\\
Set $G\defeq\{f_1,\ldots,f_m\}\cup\bigcup\limits_{0\leq k<d}P_k$ and $\mathcal{I}\defeq\langle G\rangle\trianglelefteq A[x]$.\\
The set $G$ is finite by finiteness of the index range and each constituent. $\mathcal{I}\subseteq\mathcal{J}$ by Proposition \ref{prop:generated-ideal-smallest-containing-set}.
\item \hlt{Degree Descent}: Suppose $\mathcal{I}\neq\mathcal{J}$. Set $D\defeq\{\deg u+1:u\in\mathcal{J}\setminus\mathcal{I}\}\subseteq\N$. $D\neq\emptyset$ by $\mathcal{I}\neq\mathcal{J}$.\\
Choose $g\in\mathcal{J}\setminus\mathcal{I}$ with $\deg g+1=\min\limits D$ by Assumption \ref{assump:well-ordering-positive-integers}. Set $r\defeq\deg g$. If $r<d$, then $\operatorname{lc}(g)\in L_r$ by the definition of $L_r$.\\
Choose $(c_p)_{p\in P_r}\in A^{P_r}$ by generation of $L_r$ by the leading coefficients indexed by $P_r$.\\
$\operatorname{lc}(g)=\sum\limits_{p\in P_r}c_p\operatorname{lc}(p)$ by the choice of $(c_p)_{p\in P_r}$. In this case, set $h\defeq\sum\limits_{p\in P_r}c_pp$. If $r\geq d$, then $\operatorname{lc}(g)\in L$ by the definition of $L$.\\
Choose $c_1,\ldots,c_m\in A$ by $L=\langle\operatorname{lc}(f_i):1\leq i\leq m\rangle$. $\operatorname{lc}(g)=\sum\limits_{i=1}^{m}c_i\operatorname{lc}(f_i)$ by the choice of $c_1,\ldots,c_m$.\\
In this case, set $h\defeq\sum\limits_{i=1}^{m}c_ix^{r-d}f_i$. In both cases, $h\in\mathcal{I}$, $\deg h=r$, and $\operatorname{lc}(h)=\operatorname{lc}(g)$ by construction.\\
If $g-h\in\mathcal{I}$, then $g=(g-h)+h\in\mathcal{I}$, a contradiction. Hence $g-h\in\mathcal{J}\setminus\mathcal{I}$ by $g,h\in\mathcal{J}$ and the contradiction above.\\
$\deg(g-h)<r$ by cancellation of the degree-$r$ coefficient.\\
$\deg(g-h)+1\in D$ by $g-h\in\mathcal{J}\setminus\mathcal{I}$, while $\deg(g-h)+1<\min\limits D$ by the preceding inequality, a contradiction.\\
$\mathcal{J}=\mathcal{I}=\langle G\rangle$ by contradiction; hence $\mathcal{J}$ is f.g. $A[x]$ is Noetherian by arbitrariness of $\mathcal{J}$ and Theorem \ref{thm:noetherian-iff-submodules-finitely-generated}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Finite leading-coefficient data span $\mathcal{J}$ by degree descent.\\
\hlt{Key Insight}: Matching the leading coefficient cancels the highest-degree term.
\end{proof}
\end{theorem}
