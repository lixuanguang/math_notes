\subsection{Spectra}

\begin{definition} \label{def:spectrum-vanishing-principal-complement} \hlt{Spectrum, Vanishing Sets, and Principal Complements}\\
Let $R$ be a commutative ring with identity.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{prime spectrum} of $R$ is $\operatorname{Spec}R\defeq\{\mathfrak{p}\trianglelefteq R:\mathfrak{p}\text{ is prime}\}$.
\item For $\mathcal{I}\trianglelefteq R$, set $V(\mathcal{I})\defeq\{\mathfrak{p}\in\operatorname{Spec}R:\mathcal{I}\subseteq\mathfrak{p}\}$.
\item For $f\in R$, set $D(f)\defeq\operatorname{Spec}R\setminus V(\langle f\rangle)$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:vanishing-set-calculus} \hlt{Vanishing-Set Calculus}\\
Let $R$ be a commutative ring with identity, $\mathcal{I},\mathcal{J}\trianglelefteq R$. Let $\{\mathcal{I}_{\alpha}\}_{\alpha\in A}$ be a nonempty family of ideals of $R$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $V(\langle0_R\rangle)=\operatorname{Spec}R$.
\item $V(R)=\emptyset$.
\item $\bigcap\limits_{\alpha\in A}V(\mathcal{I}_{\alpha})=V\left(\left\langle\bigcup\limits_{\alpha\in A}\mathcal{I}_{\alpha}\right\rangle\right)$.
\item $V(\mathcal{I})\cup V(\mathcal{J})=V(\mathcal{I}\cap\mathcal{J})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $\mathfrak{p}\in\operatorname{Spec}R$. The relation $0_R\in\mathfrak{p}$ holds by the ideal property. The inclusion $\langle0_R\rangle\subseteq\mathfrak{p}$ holds by Proposition \ref{prop:generated-ideal-smallest-containing-set}.\\
The equality $V(\langle0_R\rangle)=\operatorname{Spec}R$ holds by Definition \ref{def:spectrum-vanishing-principal-complement}(ii) and set extensionality.
\item Fix $\mathfrak{p}\in\operatorname{Spec}R$. The relation $R\nsubseteq\mathfrak{p}$ holds by Definition \ref{def:prime-ideal}. The equality $V(R)=\emptyset$ holds by Definition \ref{def:spectrum-vanishing-principal-complement}(ii).
\item Set $\mathcal{K}\defeq\left\langle\bigcup\limits_{\alpha\in A}\mathcal{I}_{\alpha}\right\rangle$. Fix $\mathfrak{p}\in\operatorname{Spec}R$. Membership in $\bigcap\limits_{\alpha\in A}V(\mathcal{I}_{\alpha})$ is equivalent to $\bigcup\limits_{\alpha\in A}\mathcal{I}_{\alpha}\subseteq\mathfrak{p}$ by Definition \ref{def:spectrum-vanishing-principal-complement}(ii).\\
The latter inclusion is equivalent to $\mathcal{K}\subseteq\mathfrak{p}$ by Proposition \ref{prop:generated-ideal-smallest-containing-set}.\\
The asserted equality holds by Definition \ref{def:spectrum-vanishing-principal-complement}(ii) and set extensionality.
\item The inclusions $\mathcal{I}\cap\mathcal{J}\subseteq\mathcal{I}$ and $\mathcal{I}\cap\mathcal{J}\subseteq\mathcal{J}$ hold by definition of intersection.\\
The inclusion $V(\mathcal{I})\cup V(\mathcal{J})\subseteq V(\mathcal{I}\cap\mathcal{J})$ holds by Definition \ref{def:spectrum-vanishing-principal-complement}(ii). Fix $\mathfrak{p}\in V(\mathcal{I}\cap\mathcal{J})$ and assume $\mathfrak{p}\notin V(\mathcal{I})\cup V(\mathcal{J})$.\\
Choose $a\in\mathcal{I}\setminus\mathfrak{p}$ and $b\in\mathcal{J}\setminus\mathfrak{p}$ by the failed containments. The relation $ab\in\mathcal{I}\cap\mathcal{J}\subseteq\mathfrak{p}$ holds by ideal absorption.\\
This contradicts primeness by Definition \ref{def:prime-ideal}. The reverse inclusion holds by contradiction.
\end{enumerate}
\hlt{Structural Commentary}: Generated unions control intersections; primality converts ideal intersections into finite unions.\\
\hlt{Key Insight}: A product in $\mathcal{I}\cap\mathcal{J}$ detects which ideal a prime contains.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:principal-complement-calculus} \hlt{Principal-Complement Calculus}\\
Let $R$ be a commutative ring with identity and let $f,g\in R$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $D(0_R)=\emptyset$.
\item $D(1_R)=\operatorname{Spec}R$.
\item $D(f)\cap D(g)=D(fg)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The equality $D(0_R)=\operatorname{Spec}R\setminus V(\langle0_R\rangle)=\emptyset$ holds by Proposition \ref{prop:vanishing-set-calculus}(i).
\item The equality $\langle1_R\rangle=R$ holds by Definition \ref{def:ideal-and-generated-ideal}. The equality $D(1_R)=\operatorname{Spec}R\setminus V(R)=\operatorname{Spec}R$ holds by Proposition \ref{prop:vanishing-set-calculus}(ii).
\item The relation $(\forall\,(\mathfrak{p},h)\in\operatorname{Spec}R\times R)\,[\mathfrak{p}\in D(h)\Longleftrightarrow h\notin\mathfrak{p}]$ holds by Definition \ref{def:spectrum-vanishing-principal-complement}(iii) and Proposition \ref{prop:generated-ideal-smallest-containing-set}.\\
Fix $\mathfrak{p}\in\operatorname{Spec}R$. The equivalence $\mathfrak{p}\in D(f)\cap D(g)\Longleftrightarrow(f\notin\mathfrak{p}\land g\notin\mathfrak{p})$ holds by the preceding test.\\
The equivalence $(f\notin\mathfrak{p}\land g\notin\mathfrak{p})\Longleftrightarrow fg\notin\mathfrak{p}$ holds by ideal absorption and Definition \ref{def:prime-ideal}.\\
The equivalence $fg\notin\mathfrak{p}\Longleftrightarrow\mathfrak{p}\in D(fg)$ holds by the preceding test. The equality $D(f)\cap D(g)=D(fg)$ holds by set extensionality.
\end{enumerate}
\hlt{Structural Commentary}: Complementation transfers boundary values and primality to principal-complement intersections.\\
\hlt{Key Insight}: The product $fg$ avoids a prime exactly when both factors do.
\end{proof}
\end{proposition}
