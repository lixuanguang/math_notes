\subsection{Groups, Subgroups, and Cyclic Groups}

Groups are the algebraic language of reversible operations. They appear here as the additive structure underlying rings and modules, and later as symmetry groups acting on mathematical objects.

\begin{definition} \label{def:group-and-abelian-group} \hlt{Groups}\\
A \hlt{group} is a nonempty set $G$ with binary operation $(a,b)\mapsto ab$, identity element $e_G\in G$, and inversion map $g\mapsto g^{-1}$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,a,b,c\in G)\,(ab)c=a(bc)$;
\item $(\forall\,g\in G)\,e_Gg=ge_G=g$;
\item $(\forall\,g\in G)\,gg^{-1}=g^{-1}g=e_G$.
\end{enumerate}
The group is \hlt{abelian} if $(\forall\,a,b\in G)\,ab=ba$.\\
In an abelian group the operation is often written additively, with identity $0$ and inverse $-a$.
\end{definition}

\begin{example} \label{ex:basic-groups-from-linear-algebra} \hlt{Basic Groups}\\
Let $S$ be a set, let $m,n\in\N$, and let $\mathbb{F}$ be a field.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The bijections $S\to S$ form a group under composition; it is usually nonabelian.
\item $\operatorname{Mat}(m,n,\mathbb{F})$ is an abelian group under matrix addition.
\item $\operatorname{Mat}(n,n,\mathbb{F})^{\times}$, the invertible $n\times n$ matrices, is a group under multiplication.
\item $\Z/n\Z$ is a finite abelian group under addition modulo $n$.
\end{enumerate}
\end{example}

\begin{definition} \label{def:subgroup} \hlt{Subgroup}\\
Let $G$ be a group. A subset $H\subseteq G$ is a \hlt{subgroup}, written $H\leq G$, if $e_G\in H$, $(\forall\,a,b\in H)\,ab\in H$, and $(\forall\,a\in H)\,a^{-1}\in H$.
\end{definition}

\begin{proposition} \label{prop:subgroup-test} \hlt{Subgroup Test}\\
Let $G$ be a group and let $H\subseteq G$ be nonempty. Then $H\leq G$ $\iff$ $(\forall\,a,b\in H)\,ab^{-1}\in H$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $H\leq G$ and $a,b\in H$, then $b^{-1}\in H$ and $ab^{-1}\in H$.
\item[$(\Leftarrow)$] Assume $(\forall\,a,b\in H)\,ab^{-1}\in H$. Choose $h\in H$. Then $e_G=hh^{-1}\in H$. For $b\in H$, $b^{-1}=e_Gb^{-1}\in H$.\\
For $a,c\in H$, since $c^{-1}\in H$, we have $ac=a(c^{-1})^{-1}\in H$. Hence $H\leq G$.
\end{enumerate}
\hlt{Structural Commentary}: Closure under one expression packages identity, inverses, and products.\\
\hlt{Key Insight}: In a group, subtraction-style closure is enough.
\end{proof}
\end{proposition}

\begin{definition} \label{def:cyclic-group-and-generated-subgroup} \hlt{Generated and Cyclic Groups}\\
Let $G$ be a group.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $a\in G$, define $a^0\defeq e_G$, $a^{n+1}\defeq a^na$ for $n\in\N_0$, and $a^{-n}\defeq(a^n)^{-1}$ for $n\in\N$.
\item For $a\in G$, the \hlt{subgroup generated by $a$} is $\langle a\rangle\defeq\{a^k:k\in\Z\}\leq G$.
\item $G$ is \hlt{cyclic} if $(\exists\,a\in G)\,G=\langle a\rangle$. If it exists, $\operatorname{ord}(a)\defeq\min\{n\in\N:a^n=e_G\}$; then $\langle a\rangle=\{e_G,a,\ldots,a^{\operatorname{ord}(a)-1}\}$.
\end{enumerate}
\end{definition}
