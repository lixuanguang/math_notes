\subsection{Group Homomorphisms, Kernels, and Images}

\begin{definition} \label{def:group-homomorphism-kernel-image} \hlt{Group Homomorphisms, Kernels, and Images}\\
Let $G,H$ be groups.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{homomorphism} $\varphi:G\to H$ satisfies $(\forall\,a,b\in G)\,\varphi(ab)=\varphi(a)\varphi(b)$.
\item For a homomorphism $\varphi:G\to H$, $\ker\varphi\defeq\{g\in G:\varphi(g)=e_H\}$; and $\operatorname{im}\varphi\defeq\varphi(G)$. An \hlt{isomorphism} is a bijective homomorphism.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:group-homomorphism-basic-properties} \hlt{Basic Homomorphism Properties}\\
Let $\varphi:G\to H$ be a group homomorphism. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\varphi(e_G)=e_H$;
\item $(\forall\,g\in G)\,\varphi(g^{-1})=\varphi(g)^{-1}$;
\item $\ker\varphi\leq G$;
\item $\operatorname{im}\varphi\leq H$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Put $x\defeq\varphi(e_G)$. Then $x=x^2$, so $e_H=x^{-1}x=x^{-1}x^2=x$.
\item For $g\in G$, the identities $\varphi(g)\varphi(g^{-1})=e_H=\varphi(g^{-1})\varphi(g)$ hold by Definition \ref{def:group-homomorphism-kernel-image}. Thus $\varphi(g^{-1})=\varphi(g^{-1})\varphi(g)\varphi(g)^{-1}=\varphi(g)^{-1}$.
\item The identity $e_G\in\ker\varphi$ holds by (i). If $a,b\in\ker\varphi$, then $\varphi(ab^{-1})=\varphi(a)\varphi(b)^{-1}=e_H$ by Definition \ref{def:group-homomorphism-kernel-image} and (ii).\\
Hence $\ker\varphi\leq G$ by Proposition \ref{prop:subgroup-test}.
\item The identity $e_H=\varphi(e_G)\in\operatorname{im}\varphi$ holds by (i). If $x=\varphi(a)$ and $y=\varphi(b)$ lie in $\operatorname{im}\varphi$, then $xy^{-1}=\varphi(ab^{-1})\in\operatorname{im}\varphi$ by Definition \ref{def:group-homomorphism-kernel-image} and (ii). Hence $\operatorname{im}\varphi\leq H$ by Proposition \ref{prop:subgroup-test}.
\end{enumerate}
\hlt{Structural Commentary}: Homomorphisms preserve all group structure once they preserve multiplication.\\
\hlt{Key Insight}: $\varphi(ab)=\varphi(a)\varphi(b)$ preserves $e$, inverses, kernels, and images.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:group-homomorphism-injective-kernel} \hlt{Kernel Criterion for Injectivity}\\
Let $\varphi:G\to H$ be a group homomorphism. Then $\varphi$ injective $\iff$ $\ker\varphi=\{e_G\}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\varphi$ injective, $g\in\ker\varphi$, then $\varphi(g)=e_H=\varphi(e_G)$ by Proposition \ref{prop:group-homomorphism-basic-properties}, so $g=e_G$. Hence $\ker\varphi=\{e_G\}$.
\item[$(\Leftarrow)$] If $\ker\varphi=\{e_G\}$ and $\varphi(a)=\varphi(b)$, then $\varphi(ab^{-1})=\varphi(a)\varphi(b)^{-1}=e_H$ by Definition \ref{def:group-homomorphism-kernel-image} and Proposition \ref{prop:group-homomorphism-basic-properties}(ii).\\
Thus $ab^{-1}=e_G$ and $a=(ab^{-1})b=b$.
\end{enumerate}
\hlt{Structural Commentary}: Injectivity is detected by the identity fiber $\ker\varphi$.\\
\hlt{Key Insight}: The kernel measures exactly the failure of injectivity.
\end{proof}
\end{proposition}
