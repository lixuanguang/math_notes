\subsection{Units, Divisibility, and Factorization Domains}

\begin{definition} \label{def:unit-associate-prime-irreducible-domain} \hlt{Units, Associates, Prime and Irreducible Elements}\\
Let $R$ be an integral domain.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{units} of $R$ form the set $R^\times\defeq\{u\in R:(\exists\,v\in R)\,uv=1_R\}$.
\item For $a,b\in R$, $a$ \hlt{divides} $b$, written $a\mid b$, if $(\exists\,c\in R)\,b=ac$.
\item Elements $a,b\in R\setminus\{0_R\}$ are \hlt{associates}, written $a\sim b$, if $(\exists\,u\in R^\times)\,a=ub$.
\item An element $p\in R\setminus(R^\times\cup\{0_R\})$ is \hlt{prime} if $(\forall\,a,b\in R)\,[p\mid ab\Rightarrow(p\mid a\lor p\mid b)]$.
\item An element $r\in R\setminus(R^\times\cup\{0_R\})$ is \hlt{irreducible} if $(\forall\,a,b\in R)\,[r=ab\Rightarrow(a\in R^\times\lor b\in R^\times)]$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:principal-ideals-divisibility-associates} \hlt{Divisibility and Principal Ideals}\\
Let $R$ be an integral domain and let $r,s\in R$ be nonzero.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $r\in R^\times \Longleftrightarrow \langle r\rangle=R$.
\item $r\sim s \Longleftrightarrow \langle r\rangle=\langle s\rangle$.
\item $r\mid s \Longleftrightarrow \langle s\rangle\subseteq\langle r\rangle$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) If $r\in R^\times$, choose $v\in R$ with $rv=1_R$. Then $1_R\in\langle r\rangle$, so $\langle r\rangle=R$.
\item[$(i)$] ($\Leftarrow$) If $\langle r\rangle=R$, then $1_R=ar$ for some $a\in R$, hence $r\in R^\times$.
\item[$(ii)$] ($\Rightarrow$) If $r=us$ with $u\in R^\times$, choose $v\in R$ with $uv=1_R$. Then $r\in\langle s\rangle$ and $s=(vu)s=vr\in\langle r\rangle$, so $\langle r\rangle=\langle s\rangle$.
\item[$(ii)$] ($\Leftarrow$) If $\langle r\rangle=\langle s\rangle$, then $r=as$ and $s=br$ for some $a,b\in R$. Thus $r=abr$, so $ab=1_R$ by Proposition \ref{prop:integral-domain-cancellation}; hence $a\in R^\times$ and $r\sim s$.
\item[$(iii)$] ($\Rightarrow$) If $s=ar$, then every element of $\langle s\rangle$ is of the form $xs=xar\in\langle r\rangle$.
\item[$(iii)$] ($\Leftarrow$) If $\langle s\rangle\subseteq\langle r\rangle$, then $s\in\langle r\rangle$, so $s=ar$ for some $a\in R$; hence $r\mid s$.
\end{enumerate}
\hlt{Structural Commentary}: Divisibility reverses inclusion of principal ideals.\\
\hlt{Key Insight}: Principal ideals translate multiplicative divisibility into order by containment.
\end{proof}
\end{proposition}
