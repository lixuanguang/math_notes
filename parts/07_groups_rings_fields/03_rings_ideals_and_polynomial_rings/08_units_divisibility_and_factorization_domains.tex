\subsection{Units, Divisibility, and Factorization Domains}

\begin{definition} \label{def:unit-associate-prime-irreducible-domain} \hlt{Units, Associates, Prime Elements, and Irreducible Elements}\\
Let $R$ be an integral domain.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Units}: $R^\times\defeq\{u\in R:(\exists v\in R)\,uv=1_R\}$.
\item \hlt{Divisibility}: for $a,b\in R$, $a\mid b\Longleftrightarrow(\exists c\in R)\,b=ac$.
\item \hlt{Associates}: for $a,b\in R\setminus\{0_R\}$, $a\sim b\Longleftrightarrow(\exists u\in R^\times)\,a=ub$.
\item \hlt{Prime element}: $p\in R\setminus(R^\times\cup\{0_R\})$ is prime precisely when $(\forall a,b\in R)\,[p\mid ab\Rightarrow(p\mid a\lor p\mid b)]$.
\item \hlt{Irreducible element}: $r\in R\setminus(R^\times\cup\{0_R\})$ is irreducible precisely when $(\forall a,b\in R)\,[r=ab\Rightarrow(a\in R^\times\lor b\in R^\times)]$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:principal-ideals-divisibility-associates} \hlt{Divisibility and Principal Ideals}\\
Let $R$ be an integral domain and let $r,s\in R\setminus\{0_R\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $r\in R^\times \Longleftrightarrow \langle r\rangle=R$.
\item $r\sim s \Longleftrightarrow \langle r\rangle=\langle s\rangle$.
\item $r\mid s \Longleftrightarrow \langle s\rangle\subseteq\langle r\rangle$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) If $r\in R^\times$, choose $v\in R$ with $rv=1_R$. Then $1_R\in\langle r\rangle$, so $\langle r\rangle=R$.
\item[$(i)$] ($\Leftarrow$) If $\langle r\rangle=R$, choose $a\in R$ with $1_R=ar=ra$. Then $r\in R^\times$ by Definition \ref{def:unit-associate-prime-irreducible-domain}(i).
\item[$(ii)$] ($\Rightarrow$) If $r=us$ with $u\in R^\times$, choose $v\in R$ with $uv=1_R$. Then $r=us\in\langle s\rangle$ and $s=(vu)s=v(us)=vr\in\langle r\rangle$, so $\langle r\rangle=\langle s\rangle$.
\item[$(ii)$] ($\Leftarrow$) If $\langle r\rangle=\langle s\rangle$, choose $a,b\in R$ with $r=as$ and $s=br$. The equality $r=(ab)r$ and $r\neq0_R$ imply $ab=1_R$ by Proposition \ref{prop:integral-domain-cancellation}.\\
Thus $a\in R^\times$ and $r\sim s$ by Definition \ref{def:unit-associate-prime-irreducible-domain}(i), (iii).
\item[$(iii)$] ($\Rightarrow$) If $r\mid s$, choose $a\in R$ with $s=ar$ by Definition \ref{def:unit-associate-prime-irreducible-domain}(ii). Then $\langle s\rangle=\{xs:x\in R\}=\{xar:x\in R\}\subseteq\langle r\rangle$.
\item[$(iii)$] ($\Leftarrow$) If $\langle s\rangle\subseteq\langle r\rangle$, then $s\in\langle r\rangle=\{ar:a\in R\}$, so $r\mid s$ by Definition \ref{def:unit-associate-prime-irreducible-domain}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Generators prove containment; cancellation detects units and associates.\\
\hlt{Key Insight}: A principal ideal records its generator up to multiplication by a unit.
\end{proof}
\end{proposition}
