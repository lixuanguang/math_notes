\subsection{Ring Characteristic}

\begin{definition} \label{def:characteristic-of-ring} \hlt{Characteristic}\\
Let $R$ be a ring with identity. For $n\in\N$, write $n1_R\defeq\underbrace{1_R+\cdots+1_R}_{n\text{ summands}}$.\\
Its \hlt{characteristic} is $\operatorname{char}(R)\defeq
\begin{cases}
\min\{n\in\N:n1_R=0_R\}, & \text{if this set is nonempty},\\
0, & \text{otherwise}.
\end{cases}$.
\end{definition}

\begin{proposition} \label{prop:finite-domain-prime-characteristic} \hlt{Finite Domains Have Prime Characteristic}\\
Let $R$ be a finite ring with identity. Then $\operatorname{char}(R)\neq0$. If $R$ is an integral domain, then $\operatorname{char}(R)$ is prime.
\begin{proof}
Choose $m\in\N_0$ and a bijection $e:R\to J_m$ by Definition \ref{def:finite-countable-uncountable}. If the map $f:\N\to R$, $f(n)\defeq n1_R$, were injective, then $e\circ f:\N\to J_m$ would be injective, which is impossible by Proposition \ref{prop:natural-numbers-do-not-inject-into-finite-sets}(i).\\
Thus $i1_R=j1_R$ for some $i,j\in\N$ with $i<j$, so $(j-i)1_R=0_R$ and $\operatorname{char}(R)>0$.\\
Assume $R$ is a domain and set $c=\operatorname{char}(R)$. Since $1_R\neq0_R$, $c>1$. If $c=ab$ with $1<a,b<c$, then $(a1_R)(b1_R)=ab1_R=c1_R=0_R$.\\
By minimality of $c$, $a1_R\neq0_R$ and $b1_R\neq0_R$, contradicting that $R$ has no zero divisors. Hence $c$ is prime.\\
\hlt{Structural Commentary}: Characteristic records the additive order of the unit.\\
\hlt{Key Insight}: In a domain, a composite characteristic would factor zero nontrivially.
\end{proof}
\end{proposition}
