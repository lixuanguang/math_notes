\subsection{Ring Characteristic}

\begin{definition} \label{def:characteristic-of-ring} \hlt{Characteristic}\\
Let $R$ be a ring with identity. For $n\in\N$, define $n1_R\defeq\underbrace{1_R+\cdots+1_R}_{n\text{ summands}}$.\\
The \hlt{characteristic} of $R$ is $\operatorname{char}(R)\defeq0$ if $(\forall\,n\in\N)\,n1_R\neq0_R$; otherwise, $\operatorname{char}(R)\defeq\min\{n\in\N:n1_R=0_R\}$.
\end{definition}

\begin{proposition} \label{prop:finite-domain-prime-characteristic} \hlt{Characteristic of Finite Rings and Domains}\\
Let $R$ be a finite ring with identity.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{char}(R)>0$.
\item If $R$ is an integral domain, then $\operatorname{char}(R)$ is prime.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $m\in\N_0$ and a bijection $e:R\to J_m$ by Definition \ref{def:finite-countable-uncountable}. Define $f:\N\to R$ by $f(n)\defeq n1_R$.\\
The composite $e\circ f:\N\to J_m$ is not injective by Proposition \ref{prop:natural-numbers-do-not-inject-into-finite-sets}(i), so $f$ is not injective.\\
Choose $i,j\in\N$ with $i<j$ and $i1_R=j1_R$. Then $(j-i)1_R=0_R$, and $\operatorname{char}(R)>0$ by Definition \ref{def:characteristic-of-ring}.
\item Suppose $R$ is an integral domain and set $c\defeq\operatorname{char}(R)$. Then $c>1$ by (i), Definition \ref{def:characteristic-of-ring}, and $1_R\neq0_R$.\\
If $c$ is composite, choose $a,b\in\N$ with $c=ab$ and $1<a,b<c$.\\
Then $(a1_R)(b1_R)=ab1_R=c1_R=0_R$, whereas $a1_R\neq0_R$ and $b1_R\neq0_R$ by minimality of $c$.\\
$c$ is prime by this contradiction with the absence of zero divisors in $R$.
\end{enumerate}
\hlt{Structural Commentary}: Finiteness creates an additive period; zero-divisor exclusion forces primality.\\
\hlt{Key Insight}: A composite characteristic factors zero through smaller nonzero multiples.
\end{proof}
\end{proposition}
