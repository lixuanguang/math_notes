\subsection{Rings, Identities, Fields, and Examples}

Rings add multiplication to an abelian group.

\begin{definition} \label{def:ring-commutative-identity-field} \hlt{Rings, Identities, and Fields}\\
Let $R$ be a set with binary operations $+$ and $\cdot$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $R$ is a \hlt{ring} if $(R,+)$ is an abelian group, $(\forall\,a,b,c\in R)\,(ab)c=a(bc)$, and $(\forall\,a,b,c\in R)\,[a(b+c)=ab+ac\land(a+b)c=ac+bc]$. Denote the additive identity by $0_R$ and the additive inverse of $a$ by $-a$.
\item A ring $R$ is \hlt{commutative} if $(\forall\,a,b\in R)\,ab=ba$.
\item A ring $R$ has \hlt{identity} if it is equipped with $1_R\in R$ satisfying $(\forall\,a\in R)\,1_Ra=a1_R=a$.
\item A \hlt{field} is a commutative ring with identity $1_R\neq0_R$ equipped with an inversion map $R\setminus\{0_R\}\to R\setminus\{0_R\}$, $a\mapsto a^{-1}$,\\
satisfying $(\forall\,a\in R\setminus\{0_R\})\,aa^{-1}=a^{-1}a=1_R$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:basic-rings} \hlt{Basic Rings}\\
Let $n\in\N$ and let $\mathbb{F}$ be a field.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\Z,+,\cdot)$ and $(\mathbb{F}[x],+,\cdot)$ are commutative rings with identity $1$.
\item $\operatorname{Mat}(n,n,\mathbb{F})$ is ing with identity under matrix addition and multiplication, with identity $I_n$; noncommutative for $n\geq2$.
\item $2\Z$ is a commutative ring under inherited addition and multiplication, but has no identity.
\item $\Z/n\Z$ is a commutative ring under addition and multiplication modulo $n$, with identity $1+n\Z$.
\end{enumerate}
\end{example}
