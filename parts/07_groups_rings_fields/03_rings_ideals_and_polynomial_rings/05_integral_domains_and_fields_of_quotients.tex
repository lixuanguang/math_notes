\subsection{Integral Domains and Fields of Quotients}

\begin{definition} \label{def:zero-divisor-integral-domain} \hlt{Zero Divisors and Integral Domains}\\
Let $R$ be a ring.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{zero divisor} is an element $a\in R\setminus\{0_R\}$ such that $(\exists\,b\in R\setminus\{0_R\})\,(ab=0_R\lor ba=0_R)$.
\item An \hlt{integral domain} is a commutative ring with identity $1_R\neq0_R$ and no zero divisors.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:integral-domain-cancellation} \hlt{Cancellation in Integral Domains}\\
Let $R$ be a commutative ring with identity $1_R\neq0_R$.\\
Then $R$ is an integral domain $\iff$ $(\forall\,r,x,y\in R)\,[(r\neq0_R\land rx=ry)\Rightarrow x=y]$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $rx=ry$ and $r\neq0_R$, then $r(x-y)=0_R$. Since $R$ has no zero divisors, $x-y=0_R$, so $x=y$.
\item[$(\Leftarrow)$] If $ab=0_R$ and $a\neq0_R$, then $ab=a0_R$, so the stated cancellation condition forces $b=0_R$.\\
Commutativity therefore excludes zero divisors.
\end{enumerate}
\hlt{Structural Commentary}: Domains are exactly cancellative commutative rings.\\
\hlt{Key Insight}: Zero divisors obstruct cancellation.
\end{proof}
\end{proposition}

\begin{definition} \label{def:field-of-quotients} \hlt{Field of Quotients}\\
Let $R$ be an integral domain. For $a,c\in R$ and $b,d\in R\setminus\{0_R\}$, set $(a,b)\sim(c,d)\Longleftrightarrow ad=bc$.\\
The \hlt{field of quotients} of $R$ is $(R\times(R\setminus\{0_R\}))/\sim$; write $[(a,b)]_{\sim}=a/b$. Its operations are $\frac{a}{b}+\frac{c}{d}=\frac{ad+bc}{bd}$ and $\frac{a}{b}\frac{c}{d}=\frac{ac}{bd}$.
\end{definition}

\begin{proposition} \label{prop:field-of-quotients-embeds-domain} \hlt{Domains Embed in Fields}\\
Let $R$ be an integral domain with field of quotients $K$.\\
Then the map $\iota:R\to K$, $\iota(a)=a/1_R$, is an injective ring homomorphism.
\begin{proof}
By Definition \ref{def:field-of-quotients}, for $a,b\in R$, $\iota(a+b)=\frac{a+b}{1_R}=\frac{a}{1_R}+\frac{b}{1_R}$, $\iota(ab)=\frac{ab}{1_R}=\frac{a}{1_R}\frac{b}{1_R}$, and $\iota(1_R)=\frac{1_R}{1_R}$.\\
Thus $\iota$ is a ring homomorphism. If $\iota(a)=\iota(b)$, then $a/1_R=b/1_R$, so $a1_R=b1_R$ by Definition \ref{def:field-of-quotients}; hence $a=b$, and $\iota$ is injective.\\
\hlt{Structural Commentary}: The field of quotients formally adds denominators to a domain.\\
\hlt{Key Insight}: Integral domains are exactly the setting where fraction arithmetic is consistent.
\end{proof}
\end{proposition}
