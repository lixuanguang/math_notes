\subsection{Ideals and Generated Ideals}

Ideals are subsets by which quotient rings can be formed, just as subspaces are the subsets by which quotient vector spaces can be formed.

\begin{definition} \label{def:ideal-and-generated-ideal} \hlt{Ideals and Generated Ideals}\\
Let $R$ be a ring.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A nonempty subset $\mathcal{I}\subseteq R$ is a \hlt{two-sided ideal}, written $\mathcal{I}\trianglelefteq R$, if $(\forall\,a,b\in\mathcal{I})\,a-b\in\mathcal{I}$ and $(\forall\,a\in\mathcal{I})(\forall\,r\in R)\,[ra\in\mathcal{I}\land ar\in\mathcal{I}]$.
\item If $R$ is commutative with identity and $S\subseteq R$, the \hlt{ideal generated by $S$} is $\langle S\rangle\defeq\left\{\sum\limits_{j=1}^{n}r_js_j:n\in\N_0,\ r_j\in R,\ s_j\in S\right\}$,\\
where $n=0$ denotes the empty sum $0_R$.
\item \hlt{Finite generated ideal notation}: if $n\in\N$ and $S=\{s_1,\ldots,s_n\}$, write $\langle s_1,\ldots,s_n\rangle\defeq\langle S\rangle$.
\end{enumerate}
For example, if $\mathbb{F}$ is a field and $p(x)\in\mathbb{F}[x]$, then $\langle p(x)\rangle=\{q(x)p(x):q(x)\in\mathbb{F}[x]\}$ is the \hlt{principal} ideal generated by $p(x)$.
\end{definition}

\begin{proposition} \label{prop:generated-ideal-smallest-containing-set} \hlt{Generated Ideals Are Smallest}\\
Let $R$ be a commutative ring with identity, $S\subseteq R$. Then $\langle S\rangle$ is an ideal, $S\subseteq\langle S\rangle$, and $\forall$ ideal $\mathcal{J}\subseteq R$, $S\subseteq\mathcal{J}\implies \langle S\rangle\subseteq\mathcal{J}$.
\begin{proof} Let $x=\sum\limits_{j=1}^{m}r_js_j$ and $y=\sum\limits_{k=1}^{n}t_ku_k$ lie in $\langle S\rangle$. By empty-sum convention, $0_R\in\langle S\rangle$. Moreover, $x-y=\sum\limits_{j=1}^{m}r_js_j+\sum\limits_{k=1}^{n}(-t_k)u_k\in\langle S\rangle$.\\
For every $a\in R$, $ax=\sum\limits_{j=1}^{m}(ar_j)s_j\in\langle S\rangle$ and $xa=ax$ by commutativity, so $\langle S\rangle$ is an ideal.\\
For every $s\in S$, $s=1_Rs\in\langle S\rangle$, so $S\subseteq\langle S\rangle$. If $\mathcal{J}$ is an ideal with $S\subseteq\mathcal{J}$, then $r_js_j\in\mathcal{J}$ for every summand, while $(\mathcal{J},+)\leq(R,+)$ by Proposition \ref{prop:subgroup-test}. Thus each finite sum lies in $\mathcal{J}$, so $\langle S\rangle\subseteq\mathcal{J}$, QED.\\
\hlt{Structural Commentary}: Generated ideals close a set under exactly the operations ideals must allow.\\
\hlt{Key Insight}: To force $S$ into an ideal, one must include all ring-linear combinations of elements of $S$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:intersections-and-chain-unions-of-ideals} \hlt{Intersections and Chain Unions of Ideals}\\
Let $R$ be a ring.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\{\mathcal{I}_{\alpha}\}_{\alpha\in A}$ is a nonempty family of ideals of $R$, then $\bigcap_{\alpha\in A}\mathcal{I}_{\alpha}$ is an ideal of $R$.
\item If $A\neq\emptyset$ and $\{\mathcal{I}_{\alpha}\}_{\alpha\in A}$ is a chain of ideals of $R$ under inclusion, then $\bigcup_{\alpha\in A}\mathcal{I}_{\alpha}$ is an ideal of $R$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\mathcal{I}=\bigcap_{\alpha\in A}\mathcal{I}_{\alpha}$. For every $\alpha\in A$, $0_R\in\mathcal{I}_{\alpha}$ by Proposition \ref{prop:subgroup-test}, so $0_R\in\mathcal{I}$.\\
If $a,b\in\mathcal{I}$ and $r\in R$, then $a-b,ra,ar\in\mathcal{I}_{\alpha}$ for every $\alpha\in A$, hence $a-b,ra,ar\in\mathcal{I}$.
\item Let $\mathcal{U}=\bigcup_{\alpha\in A}\mathcal{I}_{\alpha}$. Since $A\neq\emptyset$ and $0_R$ belongs to every ideal by Proposition \ref{prop:subgroup-test}, $0_R\in\mathcal{U}$.\\
If $a,b\in\mathcal{U}$, then $a\in\mathcal{I}_{\alpha}$ and $b\in\mathcal{I}_{\beta}$ for some $\alpha,\beta$. Since the family is a chain, one of $\mathcal{I}_{\alpha},\mathcal{I}_{\beta}$ contains both $a,b$; hence $a-b\in\mathcal{U}$.\\
If $a\in\mathcal{U}$ and $r\in R$, then $a\in\mathcal{I}_{\alpha}$ for some $\alpha$, so $ra,ar\in\mathcal{I}_{\alpha}\subseteq\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: Chains avoid the usual failure of unions to preserve closure.\\
\hlt{Key Insight}: Comparability places any finite test inside one ideal.
\end{proof}
\end{proposition}
