\subsection{Ring Homomorphisms and Subrings}

\begin{definition} \label{def:ring-homomorphism-and-subring} \hlt{Ring Homomorphisms and Subrings}\\
Let $R,S$ be rings with identity.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{ring homomorphism} $\varphi:R\to S$ is a map satisfying $(\forall\,a,b\in R)\,[\varphi(a+b)=\varphi(a)+\varphi(b)\land\varphi(ab)=\varphi(a)\varphi(b)]$ and $\varphi(1_R)=1_S$.
\item A \hlt{subring} of $R$ is a subset $T\subseteq R$, written $T\leq R$, that is a ring under the inherited operations and has identity $1_R$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:subring-test} \hlt{Subring Test}\\
Let $R$ be a ring with identity and let $S\subseteq R$. Then $S\leq R \iff [1_R\in S\land(\forall\,a,b\in S)\,(a-b\in S\land ab\in S)]$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $S\leq R$, then $1_R\in S$ and $S$ is closed under subtraction and multiplication.
\item[$(\Leftarrow)$] Assume $1_R\in S$ and $(\forall\,a,b\in S)\,(a-b\in S\land ab\in S)$. Then $S$ is nonempty, $(S,+)\leq(R,+)$ by Proposition \ref{prop:subgroup-test}, and multiplication restricts to $S$. All ring laws and the identity $1_R$ are inherited from $R$, so $S\leq R$.
\end{enumerate}
\hlt{Structural Commentary}: A subring is a subset stable under the operations and containing the ambient unit.\\
\hlt{Key Insight}: Subtraction handles the additive group part in one stroke.
\end{proof}
\end{proposition}
