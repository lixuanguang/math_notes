\subsection{Finite Fields and Characteristic}

\begin{perspective} \label{per:finite-fields-and-characteristic-two} \hlt{Finite Fields and Characteristic}\\
Let $F$ be a finite field. Then $\operatorname{char}(F)$ is prime by Proposition \ref{prop:finite-domain-prime-characteristic}(ii).\\
If $\operatorname{char}(F)=2$, then $(\forall\,a\in F)\,a+a=0_F$, equivalently $(\forall\,a\in F)\,a=-a$.\\
Thus characteristic two identifies addition and subtraction in $F$-linear calculations.
\end{perspective}

\begin{definition} \label{def:linear-code-and-hamming-weight} \hlt{Linear Code and Hamming Weight}\\
Let $n,q\in\N$ and let $\mathbb{F}_q$ be a field with $|\mathbb{F}_q|=q$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $V(n,q)\defeq\mathbb{F}_q^n$, the \hlt{ambient code space}.
\item A \hlt{linear code} of length $n$ over $\mathbb{F}_q$ is a subspace $C\leq V(n,q)$.
\item The \hlt{Hamming weight} of $\mathbf{x}\in V(n,q)$ is $\operatorname{wt}(\mathbf{x})\defeq|\{j\in\{1,\ldots,n\}:x_j\neq0\}|$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:linear-codes-even-weight-subspace} \hlt{Even-Weight Code}\\
Let $n\in\N$ and define $\pi_n:V(n,2)\to\mathbb{F}_2$ by $\pi_n(\mathbf{x})\defeq\sum\limits_{j=1}^{n}x_j$.\\
Then $(\forall\,\mathbf{x},\mathbf{y}\in V(n,2))(\forall\,c\in\mathbb{F}_2)\,\pi_n(c\mathbf{x}+\mathbf{y})=c\pi_n(\mathbf{x})+\pi_n(\mathbf{y})$, so $\pi_n$ is linear. Set $E_n\defeq\ker(\pi_n)=\{\mathbf{x}\in V(n,2):\sum\limits_{j=1}^{n}x_j=0\}$.\\
Then $E_n\leq V(n,2)$ by Proposition \ref{prop:kernel-and-range-are-subspaces}(i), so $E_n$ is a linear code by Definition \ref{def:linear-code-and-hamming-weight}(ii).\\
Since $\mathbb{F}_2=\{0,1\}$ and $1+1=0$, $(\forall\,\mathbf{x}\in V(n,2))\,[\mathbf{x}\in E_n\Longleftrightarrow2\mid\operatorname{wt}(\mathbf{x})]$.\\
\hlt{Structural Commentary}: $E_n=\ker(\pi_n)$ supplies closure; binary addition supplies parity.\\
\hlt{Key Insight}: Over $\mathbb{F}_2$, coordinate summation records Hamming-weight parity.
\end{example}
