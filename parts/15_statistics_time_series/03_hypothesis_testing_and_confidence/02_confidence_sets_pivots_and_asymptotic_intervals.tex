\subsection{Confidence Sets, Pivots, and Asymptotic Intervals}
% Source integration: supplied Chapter 9 screenshots, Section 9.3.

Confidence procedures attach uncertainty to estimators by controlling coverage probability under repeated sampling.

\begin{definition} \label{def:confidence-set-coverage} \label{def:pivot-confidence-construction} \hlt{Confidence Sets and Pivots}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A random set $C(X)\subseteq\Psi$ is a \hlt{$1-\alpha$ confidence set} for $\psi:\Theta\to\Psi$ if $\inf_{\theta\in\Theta}\mathbb{P}_\theta\{\psi(\theta)\in C(X)\}\geq1-\alpha$.\\
The left side is the \hlt{coverage guarantee}.
\item A \hlt{pivot} is a statistic $Q(X,\theta)$ whose distribution under $\mathbb{P}_\theta$ is free of $\theta$.
\item If constants $a_\alpha,b_\alpha$ satisfy $\mathbb{P}_\theta\{a_\alpha\leq Q(X,\theta)\leq b_\alpha\}=1-\alpha$ for all $\theta\in\Theta$,\\
then $C(X)\defeq\{\theta\in\Theta:a_\alpha\leq Q(X,\theta)\leq b_\alpha\}$ is an exact $1-\alpha$ confidence set for $\theta$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:normal-mean-known-variance-confidence-interval} \hlt{Normal Mean with Known Variance}\\
Let $X_1,\ldots,X_n$ be independent $N(\mu,\sigma^2)$ variables with known $\sigma$.\\
The pivot $Z_\mu\defeq\sqrt n(\overline{X}-\mu)/\sigma\sim N(0,1)$ gives $\mathbb{P}_\mu\{\abs{Z_\mu}\leq z_{1-\alpha/2}\}=1-\alpha$.\\
Solving for $\mu$ gives the exact interval $C_\alpha(X)=[\overline{X}-z_{1-\alpha/2}\sigma/\sqrt n,\overline{X}+z_{1-\alpha/2}\sigma/\sqrt n]$.\\
\hlt{Test Connection}: The two-sided level-$\alpha$ test of $H_0:\mu=\mu_0$ rejects iff $\mu_0\notin C_\alpha(X)$.
\end{example}

\begin{example} \label{ex:normal-mean-variance-exact-confidence-intervals} \hlt{Normal Mean and Variance with Unknown Variance}\\
Let $X_1,\ldots,X_n$ be independent $N(\mu,\sigma^2)$ variables, and set $S^2\defeq(n-1)^{-1}\sum_{i=1}^n(X_i-\overline{X})^2$.\\
The exact pivots are $T_\mu\defeq\sqrt n(\overline{X}-\mu)/S\sim t_{n-1}$ and $W_{\sigma^2}\defeq(n-1)S^2/\sigma^2\sim\chi^2_{n-1}$.\\
Writing $t_{\nu,p}$ and $\chi^2_{\nu,p}$ for $p$-quantiles, pivot inversion gives $C_\mu(X)=[\overline{X}-t_{n-1,1-\alpha/2}S/\sqrt n,\overline{X}+t_{n-1,1-\alpha/2}S/\sqrt n]$ and $C_{\sigma^2}(X)=[(n-1)S^2/\chi^2_{n-1,1-\alpha/2},(n-1)S^2/\chi^2_{n-1,\alpha/2}]$.\\
Both intervals have exact marginal coverage $1-\alpha$; $C_{\sigma^2}(X)$ is asymmetric because the $\chi^2$ pivot is not symmetric.
\end{example}

\begin{proposition} \label{prop:test-confidence-duality} \hlt{Test--Confidence Duality}\\
For each $\psi_0\in\Psi$, let $\varphi_{\psi_0}:\mathcal{X}\to\{0,1\}$ satisfy $\sup_{\theta:\psi(\theta)=\psi_0}\mathbb{E}_\theta\varphi_{\psi_0}\leq\alpha$.\\
Then $C(X)\defeq\{\psi_0\in\Psi:\varphi_{\psi_0}(X)=0\}$ is a $1-\alpha$ confidence set for $\psi(\theta)$.
\begin{proof}
Fix $\theta\in\Theta$ and set $\psi_\theta\defeq\psi(\theta)$.\\
By definition, $\{\psi(\theta)\notin C(X)\}=\{\varphi_{\psi_\theta}(X)=1\}$, so $\mathbb{P}_\theta\{\psi(\theta)\notin C(X)\}=\mathbb{E}_\theta\varphi_{\psi_\theta}\leq\sup_{\eta:\psi(\eta)=\psi_\theta}\mathbb{E}_\eta\varphi_{\psi_\theta}\leq\alpha$.\\
Hence $\mathbb{P}_\theta\{\psi(\theta)\in C(X)\}\geq1-\alpha$ for every $\theta$, and $\inf_{\theta\in\Theta}\mathbb{P}_\theta\{\psi(\theta)\in C(X)\}\geq1-\alpha$.\\
\hlt{Structural Commentary}: Inverting all accepted point-null tests turns non-rejection regions into a confidence set.\\
\hlt{Key Insight}: Coverage failure equals rejection of the true null value.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:confidence-test-converse-duality} \hlt{Confidence--Test Converse}\\
Let $C(X)\subseteq\Psi$ satisfy $\inf_{\theta\in\Theta}\mathbb{P}_\theta\{\psi(\theta)\in C(X)\}\geq1-\alpha$.\\
For $\psi_0\in\Psi$, let $\varphi_{\psi_0}(X)\defeq\mathbf{1}_{\{\psi_0\notin C(X)\}}$ for testing $H_0:\psi(\theta)=\psi_0$.\\
Then $\varphi_{\psi_0}$ is level $\alpha$: $\sup_{\theta:\psi(\theta)=\psi_0}\mathbb{E}_\theta\varphi_{\psi_0}\leq\alpha$.
\begin{proof} Fix $\theta$ with $\psi(\theta)=\psi_0$. Then $\mathbb{E}_\theta\varphi_{\psi_0}=\mathbb{P}_\theta\{\psi_0\notin C(X)\}=\mathbb{P}_\theta\{\psi(\theta)\notin C(X)\}=1-\mathbb{P}_\theta\{\psi(\theta)\in C(X)\}\leq\alpha$.\\
Taking $\sup_{\theta:\psi(\theta)=\psi_0}$ gives the level bound.\\
\hlt{Structural Commentary}: A confidence set tests a proposed value by checking membership.\\
\hlt{Key Insight}: Rejection of $\psi_0$ is exactly non-coverage when $\psi_0$ is true.
\end{proof}
\end{proposition}

\begin{remark} \label{rem:duality-acceptance-regions} \hlt{Acceptance Regions and Confidence Regions}\\
For level-$\alpha$ point-null tests with acceptance regions $A_{\psi_0}\subseteq\mathcal{X}$, the inversion $C(X)\defeq\{\psi_0\in\Psi:X\in A_{\psi_0}\}$ is the set of null values not rejected at $X$.\\
Conversely, any confidence set $C(X)$ induces point-null tests $\varphi_{\psi_0}(X)=\mathbf{1}_{\{\psi_0\notin C(X)\}}$.
\end{remark}

\begin{definition} \label{def:wald-score-likelihood-intervals} \hlt{Wald, Score, and Likelihood Intervals}\\
For scalar regular $\theta$ with estimator $\widehat{\theta}$, standard error $\widehat{\operatorname{se}}(\widehat{\theta})$, score $U_n$, and likelihood ratio $\lambda_n(\theta_0)$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
\hlt{Wald}: use $\widehat{\theta}\pm z_{1-\alpha/2}\widehat{\operatorname{se}}(\widehat{\theta})$.
\item
\hlt{Score}: invert $\abs{U_n(\theta_0)}/\sqrt{I_n(\theta_0)}\leq z_{1-\alpha/2}$ over $\theta_0$.
\item
\hlt{Likelihood}: invert $-2\log\lambda_n(\theta_0)\leq\chi^2_{1,1-\alpha}$ over $\theta_0$.
\end{enumerate}
\hlt{Key Insight}: Wald, score, and likelihood intervals are first-order equivalent in regular models.
\end{definition}

\begin{example} \label{ex:poisson-mle-wald-confidence-interval} \hlt{Poisson Mean Wald Interval}\\
Let $X_1,\ldots,X_n$ be independent $\operatorname{Poisson}(\lambda)$ variables.\\
Then $\widehat{\lambda}=\overline{X}$ and $\sqrt n(\overline{X}-\lambda)/\sqrt{\lambda}\Rightarrow N(0,1)$, so the plug-in standard error is $\widehat{\operatorname{se}}(\widehat{\lambda})=\sqrt{\overline{X}/n}$.\\
The Wald interval is $C_\alpha(X)=[\overline{X}-z_{1-\alpha/2}\sqrt{\overline{X}/n},\overline{X}+z_{1-\alpha/2}\sqrt{\overline{X}/n}]$, with asymptotic coverage $1-\alpha$.
\end{example}
