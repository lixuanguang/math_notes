\subsection{Stationarity, Autocovariance, and Linear Processes}

Time-series statistics studies data ordered by time, where dependence is part of the signal rather than a nuisance to be removed.

\begin{definition} \label{def:strict-and-weak-stationarity} \hlt{Strict and Weak Stationarity}\\
A stochastic process $(X_t)_{t\in\Z}$ is \hlt{strictly stationary} if
\[
(X_{t_1},\ldots,X_{t_k})\disteq (X_{t_1+h},\ldots,X_{t_k+h})
\]
for all $k$, all $t_1,\ldots,t_k$, and all shifts $h$.  It is \hlt{weakly stationary} if $\mathbb{E}X_t=\mu$ is constant and
\[
\gamma(h)=\operatorname{Cov}(X_t,X_{t-h})
\]
depends only on the lag $h$.
\end{definition}

\begin{definition} \label{def:acf-pacf-and-spectral-density} \hlt{Autocorrelation and Spectral Density}\\
For a weakly stationary process with $\gamma(0)>0$, the \hlt{autocorrelation function} is
\[
\rho(h)=\frac{\gamma(h)}{\gamma(0)}.
\]
If $\sum_{h\in\Z}\abs{\gamma(h)}<\infty$, the \hlt{spectral density} is
\[
f(\lambda)=\frac{1}{2\pi}\sum_{h\in\Z}\gamma(h)e^{-ih\lambda},
\qquad -\pi\leq\lambda\leq\pi.
\]
\end{definition}

\begin{definition} \label{def:white-noise-linear-process} \hlt{White Noise and Linear Process}\\
A process $(\varepsilon_t)$ is \hlt{white noise} with variance $\sigma^2$ if $\mathbb{E}\varepsilon_t=0$, $\operatorname{Var}(\varepsilon_t)=\sigma^2$, and $\operatorname{Cov}(\varepsilon_t,\varepsilon_s)=0$ for $t\neq s$.  A \hlt{linear process} has representation
\[
X_t=\mu+\sum_{j=0}^{\infty}\psi_j\varepsilon_{t-j},
\]
usually with $\sum_j\abs{\psi_j}<\infty$.
\end{definition}

\begin{example} \label{ex:ar-one-stationarity} \hlt{Autoregression of Order One}\\
For
\[
X_t=\phi X_{t-1}+\varepsilon_t,
\]
with white-noise innovations and $|\phi|<1$, the stationary solution is
\[
X_t=\sum_{j=0}^{\infty}\phi^j\varepsilon_{t-j}.
\]
Its autocovariance is $\gamma(h)=\sigma^2\phi^{|h|}/(1-\phi^2)$ and its autocorrelation is $\rho(h)=\phi^{|h|}$.
\end{example}

\begin{remark} \label{rem:time-series-prerequisite-stochastic-processes} \hlt{Process-Level Prerequisites}\\
The stochastic-process concepts of filtration, conditional expectation, martingales, stationarity, and ergodicity from Part 12 provide the probability backbone for time-series inference.
\end{remark}
