\subsection{ARIMA, State-Space, and Volatility Models}

Classical time-series models combine deterministic filters, stochastic innovations, and likelihood-based estimation.

\begin{definition} \label{def:arma-model} \hlt{ARMA Model}\\
An \hlt{ARMA$(p,q)$} model satisfies
\[
X_t-\phi_1X_{t-1}-\cdots-\phi_pX_{t-p}
=\varepsilon_t+\theta_1\varepsilon_{t-1}+\cdots+\theta_q\varepsilon_{t-q}.
\]
Using the backshift operator $B X_t=X_{t-1}$, this is
\[
\phi(B)X_t=\theta(B)\varepsilon_t.
\]
Causality and invertibility are encoded by roots of $\phi$ and $\theta$ lying outside the unit disk.
\end{definition}

\begin{definition} \label{def:arima-model} \hlt{ARIMA Model}\\
A process is \hlt{ARIMA$(p,d,q)$} if $(1-B)^dX_t$ is ARMA$(p,q)$.  Differencing removes polynomial stochastic trends before stationary ARMA modeling.
\end{definition}

\begin{definition} \label{def:state-space-model-kalman} \hlt{Linear Gaussian State-Space Model}\\
A linear Gaussian state-space model has latent state $\alpha_t$ and observation $Y_t$ satisfying
\[
\alpha_{t+1}=F\alpha_t+G\eta_t,
\qquad
Y_t=H\alpha_t+\varepsilon_t,
\]
where $(\eta_t)$ and $(\varepsilon_t)$ are Gaussian white-noise sequences.  The Kalman filter recursively computes the conditional mean and covariance of $\alpha_t$ given past observations.
\end{definition}

\begin{definition} \label{def:arch-garch-models} \hlt{ARCH and GARCH Volatility}\\
An \hlt{ARCH$(q)$} model writes
\[
X_t=\sigma_t z_t,
\qquad
\sigma_t^2=\omega+\alpha_1X_{t-1}^2+\cdots+\alpha_qX_{t-q}^2,
\]
where $(z_t)$ is standardized white noise.  A \hlt{GARCH$(p,q)$} model adds lagged conditional variances:
\[
\sigma_t^2=\omega+\sum_{i=1}^q\alpha_iX_{t-i}^2+\sum_{j=1}^p\beta_j\sigma_{t-j}^2.
\]
\end{definition}

\begin{remark} \label{rem:time-series-likelihood-filtering} \hlt{Likelihood by Filtering}\\
For ARMA, state-space, and volatility models, likelihood evaluation is often a recursive filtering problem.  This links statistical inference to the numerical recursion and stability ideas of Parts 13 and 14.
\end{remark}
