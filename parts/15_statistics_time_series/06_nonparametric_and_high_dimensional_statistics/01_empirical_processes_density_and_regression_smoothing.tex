\subsection{Empirical Processes, Density, and Regression Smoothing}

Nonparametric statistics replaces finite-dimensional parameter assumptions by smoothness, shape, sparsity, or complexity restrictions.

\begin{definition} \label{def:empirical-measure-and-process} \hlt{Empirical Measure and Empirical Process}\\
For observations $X_1,\ldots,X_n$, the \hlt{empirical measure} is
\[
\widehat{P}_n=\frac1n\sum_{i=1}^n\delta_{X_i}.
\]
For a class of measurable functions $\mathcal{F}$, the \hlt{empirical process} is
\[
\mathbb{G}_n(f)=\sqrt{n}\left(\widehat{P}_nf-Pf\right),
\qquad f\in\mathcal{F}.
\]
\end{definition}

\begin{theorem} \label{thm:glivenko-cantelli} \hlt{Glivenko--Cantelli Theorem}\\
If $X_1,\ldots,X_n$ are independent real-valued observations with distribution function $F$, and
\[
\widehat{F}_n(t)=\frac1n\sum_{i=1}^n\mathbf{1}_{\{X_i\leq t\}},
\]
then
\[
\sup_{t\in\R}\abs{\widehat{F}_n(t)-F(t)}\to0
\]
almost surely.
\end{theorem}

\begin{definition} \label{def:kernel-density-estimator} \hlt{Kernel Density Estimator}\\
Given a kernel $K$ with $\int K(u)\,du=1$ and bandwidth $h>0$, the \hlt{kernel density estimator} is
\[
\widehat{f}_h(x)=\frac{1}{nh}\sum_{i=1}^nK\!\left(\frac{x-X_i}{h}\right).
\]
The bandwidth controls the bias-variance tradeoff: large $h$ oversmooths, while small $h$ leaves high variance.
\end{definition}

\begin{definition} \label{def:nonparametric-regression-smoothing} \hlt{Nonparametric Regression}\\
In the regression model $Y=m(X)+\varepsilon$ with $\mathbb{E}[\varepsilon\mid X]=0$, the target $m(x)=\mathbb{E}[Y\mid X=x]$ may be estimated by local averaging:
\[
\widehat{m}_h(x)=
\frac{\sum_{i=1}^nK((x-X_i)/h)Y_i}
{\sum_{i=1}^nK((x-X_i)/h)}
\]
when the denominator is nonzero.
\end{definition}

\begin{remark} \label{rem:complexity-controls-nonparametric-estimation} \hlt{Complexity Control}\\
Uniform laws of large numbers and concentration inequalities determine whether empirical criteria approximate population criteria over large function classes.  Smoothness and entropy replace finite-dimensional parameter counting.
\end{remark}
