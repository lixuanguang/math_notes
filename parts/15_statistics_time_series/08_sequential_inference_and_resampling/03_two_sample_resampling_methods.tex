\subsection{Two-Sample Resampling Methods}
\label{subsec:two-sample-resampling-methods}
% Source integration: attached Chapter 11 screenshots, Section 11.2.3 and selected problems.

Two-sample resampling procedures must preserve the source of randomness relevant to the question.  Bootstrap procedures usually resample within groups to estimate sampling variability under the fitted empirical distributions; permutation procedures pool labels when the null asserts exchangeability.

\begin{method} \label{meth:two-sample-bootstrap} \hlt{Two-Sample Nonparametric Bootstrap}\\
Let $x_{1:n}$ and $y_{1:m}$ be independent samples from $F$ and $G$, let $\widehat{F}_n\defeq n^{-1}\sum_{i=1}^{n}\delta_{x_i}$ and $\widehat{G}_m\defeq m^{-1}\sum_{j=1}^{m}\delta_{y_j}$.\\
Let $T(F,G)$ be a two-sample functional, and fix $B\geq2$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Resample}: $x_{1:n}^{\ast b}\sim\widehat{F}_n^{\otimes n}$ and $y_{1:m}^{\ast b}\sim\widehat{G}_m^{\otimes m}$ independently for $b=1,\ldots,B$.
\item \hlt{Recompute}: with $\widehat{F}_n^{\ast b}\defeq n^{-1}\sum_{i=1}^{n}\delta_{x_i^{\ast b}}$ and $\widehat{G}_m^{\ast b}\defeq m^{-1}\sum_{j=1}^{m}\delta_{y_j^{\ast b}}$, set $\widehat{T}^{\ast b}\defeq T(\widehat{F}_n^{\ast b},\widehat{G}_m^{\ast b})$.
\item \hlt{Summarize}: $s_T^\ast\defeq\operatorname{sd}\{\widehat{T}^{\ast b}:1\leq b\leq B\}$; $I_{1-\alpha}^\ast\defeq[q_{\alpha/2}^\ast,q_{1-\alpha/2}^\ast]$, where $q_\gamma^\ast$ is the empirical $\gamma$-quantile.
\end{enumerate}
This is the two-sample version of Method \ref{meth:bootstrap-standard-error}.
\end{method}

\begin{example} \label{ex:bootstrap-mann-whitney-effect} \hlt{Bootstrap for $\pi=\mathbb{P}(X<Y)$}\\
Let $x_{1:n}\sim F^{\otimes n}$ and $y_{1:m}\sim G^{\otimes m}$ independently.\\
Set $\widehat{F}_n\defeq n^{-1}\sum_{i=1}^{n}\delta_{x_i}$, $\widehat{G}_m\defeq m^{-1}\sum_{j=1}^{m}\delta_{y_j}$, and $\widehat{\pi}\defeq(mn)^{-1}\sum_{i=1}^{n}\sum_{j=1}^{m}\mathbf{1}_{\{x_i<y_j\}}$.\\
For $b=1,\ldots,B$, draw $x_{1:n}^{\ast b}\sim\widehat{F}_n^{\otimes n}$ and $y_{1:m}^{\ast b}\sim\widehat{G}_m^{\otimes m}$ independently.\\
Set $\widehat{\pi}^{\ast b}\defeq(mn)^{-1}\sum_{i=1}^{n}\sum_{j=1}^{m}\mathbf{1}_{\{x_i^{\ast b}<y_j^{\ast b}\}}$, and use $s_\pi^\ast\defeq\operatorname{sd}\{\widehat{\pi}^{\ast b}:1\leq b\leq B\}$.\\
\hlt{Key Insight}: Separate resampling targets $F\otimes G$ variation; rank-sum calibration imposes $F=G$.
\end{example}

\begin{method} \label{meth:two-sample-permutation-mean-test} \hlt{Permutation Test for a Difference of Means}\\
Let $x_{1:n}$ be control responses, let $y_{1:m}$ be treatment responses, set $\overline{x}\defeq n^{-1}\sum_{i=1}^{n}x_i$ and $\overline{y}\defeq m^{-1}\sum_{j=1}^{m}y_j$.\\
Let $z_{1:N}$ with $N=n+m$ be the pooled responses, and set $T_{\mathrm{obs}}\defeq \overline{y}-\overline{x}$.\\
Let $H_0$ denote no treatment effect, set $\mathcal{S}_{m,N}\defeq\{S\subseteq\{1,\ldots,N\}:\abs{S}=m\}$, and use $S\mid H_0\sim\operatorname{Unif}(\mathcal{S}_{m,N})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Relabel}: $S\in\mathcal{S}_{m,N}$, by enumeration or Monte Carlo draws from $\operatorname{Unif}(\mathcal{S}_{m,N})$.
\item \hlt{Recompute}: $T_S^\ast\defeq m^{-1}\sum_{i\in S}z_i-n^{-1}\sum_{i\notin S}z_i$.
\item \hlt{Calibrate}: $p_{\mathrm{exact}}\defeq\binom{N}{m}^{-1}\#\{S\in\mathcal{S}_{m,N}:\abs{T_S^\ast}\geq\abs{T_{\mathrm{obs}}}\}$, Monte Carlo approximation if $\binom{N}{m}$ large.
\end{enumerate}
\end{method}

\begin{remark} \label{rem:permutation-versus-rank-tests} \hlt{Permutation and Rank Tests}\\
The permutation test for means and the Mann--Whitney test both use the randomization principle that, under the null, labels are exchangeable.  They differ in the statistic: the permutation mean test uses the numerical values and is sensitive to outliers, while the Mann--Whitney test first replaces values by ranks.
\end{remark}

\begin{method} \label{meth:bootstrap-hodges-lehmann-shift} \hlt{Bootstrap for a Hodges--Lehmann Shift}\\
For independent samples $x_{1:n}$ and $y_{1:m}$ and $B\geq2$, set $\widehat{\Delta}_{\mathrm{HL}}\defeq\operatorname{median}\{y_j-x_i:1\leq i\leq n,\ 1\leq j\leq m\}$, resample the two groups separately for $b=1,\ldots,B$, and use the empirical spread or quantiles of the recomputed medians $\widehat{\Delta}_{\mathrm{HL}}^{\ast b}$ for a standard error or interval.
\end{method}
