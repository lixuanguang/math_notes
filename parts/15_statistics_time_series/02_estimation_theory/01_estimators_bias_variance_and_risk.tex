\subsection{Estimators, Bias, Variance, and Risk}

An estimator is a statistic interpreted as an answer to a target question.  Its quality depends on the target, the loss function, and the model law under which performance is measured.

\begin{definition} \label{def:estimator-loss-risk} \hlt{Estimator, Loss, and Risk}\\
Let $\psi:\Theta\to\Psi$ be a target parameter.  An \hlt{estimator} of $\psi(\theta)$ is a statistic $\delta(X)$ with values in an action space $\mathcal{D}$.  A \hlt{loss function} is a map $L:\Theta\times\mathcal{D}\to[0,\infty]$.  The \hlt{risk} of $\delta$ is
\[
R(\theta,\delta)=\mathbb{E}_\theta[L(\theta,\delta(X))].
\]
For squared-error loss $L(\theta,d)=\norm{d-\psi(\theta)}^2$, risk is mean squared error.
\end{definition}

\begin{definition} \label{def:bias-variance-consistency} \hlt{Bias, Variance, and Consistency}\\
For a real-valued estimator $\widehat{\psi}$:
\[
\operatorname{Bias}_\theta(\widehat{\psi})
=\mathbb{E}_\theta[\widehat{\psi}]-\psi(\theta),
\qquad
\operatorname{Var}_\theta(\widehat{\psi})
=\mathbb{E}_\theta[(\widehat{\psi}-\mathbb{E}_\theta\widehat{\psi})^2].
\]
The estimator is \hlt{unbiased} if its bias is zero for every $\theta$.  A sequence $\widehat{\psi}_n$ is \hlt{consistent} at $\theta$ if $\widehat{\psi}_n\to\psi(\theta)$ in $P_\theta$-probability.
\end{definition}

\begin{proposition} \label{prop:mse-bias-variance-decomposition} \hlt{Bias-Variance Decomposition}\\
For squared-error loss,
\[
\mathbb{E}_\theta[(\widehat{\psi}-\psi(\theta))^2]
=\operatorname{Var}_\theta(\widehat{\psi})
\operatorname{Bias}_\theta(\widehat{\psi})^2.
\]
\begin{proof}
Write $\widehat{\psi}-\psi=(\widehat{\psi}-\mathbb{E}_\theta\widehat{\psi})+(\mathbb{E}_\theta\widehat{\psi}-\psi)$ and expand the square.  The cross term has expectation zero.
\end{proof}
\end{proposition}

\begin{example} \label{ex:sample-mean-estimator} \hlt{Sample Mean}\\
If $X_1,\ldots,X_n$ are independent with mean $\mu$ and variance $\sigma^2<\infty$, then $\overline{X}_n=n^{-1}\sum_{i=1}^nX_i$ is unbiased for $\mu$ and
\[
\operatorname{Var}(\overline{X}_n)=\frac{\sigma^2}{n}.
\]
The weak law of large numbers gives consistency.
\end{example}

\begin{definition} \label{def:plug-in-estimator-and-m-estimator} \hlt{Plug-In and M-Estimators}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{plug-in estimator} replaces an unknown distribution or parameter by an empirical or fitted version, for example $T(P)$ by $T(\widehat{P}_n)$.
\item An \hlt{$M$-estimator} is any estimator of the form
\[
\widehat{\theta}_n\in\operatorname*{arg\,min}_{\theta\in\Theta}M_n(\theta),
\]
where $M_n$ is an empirical criterion such as negative log-likelihood or squared residual loss.
\end{enumerate}
\end{definition}
