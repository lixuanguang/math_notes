\subsection{Kernel Methods and RKHS Theory}

Kernel methods keep the algorithm linear in a feature space while computing only inner products.

\begin{definition} \label{def:sl-positive-definite-kernel} \hlt{Positive Definite Kernel}\\
A symmetric function $k:\mathcal{X}\times\mathcal{X}\to\R$ is a \hlt{positive definite kernel} if for every finite set $x_1,\ldots,x_n$ the Gram matrix $K_{ij}=k(x_i,x_j)$ is positive semidefinite.
\end{definition}

\begin{definition} \label{def:sl-rkhs} \hlt{Reproducing Kernel Hilbert Space}\\
A Hilbert space $\mathcal{H}$ of functions on $\mathcal{X}$ is a \hlt{reproducing kernel Hilbert space} if point evaluation is continuous.  Its kernel $k$ satisfies
\[
f(x)=\langle f,k(x,\cdot)\rangle_{\mathcal{H}}
\]
for every $f\in\mathcal{H}$.
\end{definition}

\begin{theorem} \label{thm:sl-representer-theorem-form} \hlt{Representer Theorem Form}\\
For many objectives of the form
\[
\sum_{i=1}^n\ell(f(x_i),y_i)+\lambda\Phi(\norm{f}_{\mathcal{H}}),
\]
where $\Phi$ is increasing, a minimizer in an RKHS has the finite expansion
\[
f(\cdot)=\sum_{i=1}^n\alpha_i k(x_i,\cdot).
\]
\end{theorem}

\begin{example} \label{ex:sl-common-kernels} \hlt{Common Kernels}
Polynomial kernels, Gaussian radial basis kernels, and Laplace kernels encode different assumptions about smoothness, locality, and interaction order.
\end{example}
