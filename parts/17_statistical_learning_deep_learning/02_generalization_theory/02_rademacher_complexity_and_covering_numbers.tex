\subsection{Rademacher Complexity and Covering Numbers}

For real-valued predictors, complexity is often measured by how well the class can fit random noise.

\begin{definition} \label{def:sl-rademacher-complexity} \hlt{Empirical Rademacher Complexity}\\
Given sample points $x_1,\ldots,x_n$ and independent Rademacher signs $\sigma_i\in\{-1,1\}$,
\[
\widehat{\mathfrak{R}}_S(\mathcal{F})
=
\mathbb{E}_{\sigma}
\left[\sup_{f\in\mathcal{F}}\frac1n\sum_{i=1}^n\sigma_if(x_i)\right].
\]
The expectation over the sample gives the population Rademacher complexity.
\end{definition}

\begin{theorem} \label{thm:sl-rademacher-generalization-bound} \hlt{Rademacher Generalization Bound}\\
For a loss-composed class $\ell\circ\mathcal{F}$ bounded in $[0,1]$, with probability at least $1-\delta$,
\[
R(f)\leq \widehat{R}_S(f)
+2\widehat{\mathfrak{R}}_S(\ell\circ\mathcal{F})
+3\sqrt{\frac{\log(2/\delta)}{2n}}
\]
for every $f\in\mathcal{F}$.
\end{theorem}

\begin{definition} \label{def:sl-covering-number} \hlt{Covering Number}\\
For a metric space $(T,d)$, the covering number $N(\varepsilon,T,d)$ is the smallest number of closed $d$-balls of radius $\varepsilon$ needed to cover $T$.  Covering numbers measure metric size rather than parameter count.
\end{definition}

\begin{remark} \label{rem:sl-entropy-integrals} \hlt{Entropy Methods}\\
Bounds on $\log N(\varepsilon,\mathcal{F},d)$ can be integrated over scales to control empirical processes.  This is the route from functional-analytic compactness to statistical-learning rates.
\end{remark}
