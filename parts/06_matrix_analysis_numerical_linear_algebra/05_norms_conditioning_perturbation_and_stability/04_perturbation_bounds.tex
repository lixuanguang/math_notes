\subsection{Perturbation Bounds}

Throughout this subsection, $n\in\mathbb{Z}_{>0}$.

\begin{lemma} \label{lem:small-spectral-norm-perturbation-invertible} \hlt{Small Spectral-Norm Perturbations of the Identity}\\
If $E\in\operatorname{Mat}(n,n,\mathbb{C})$ and $\norm{E}_2<1$, then $I_n+E$ is invertible.
\begin{proof}
$\rho(E)\leq\norm{E}_2<1$ by Theorem \ref{thm:eigenvalues-versus-singular-values}(ii), so $-1\notin\operatorname{spec}(E)$.\\
For $p(z)=1+z$, $\operatorname{spec}(I_n+E)=\{1+\lambda:\lambda\in\operatorname{spec}(E)\}$ by Corollary \ref{cor:polynomial-spectral-mapping-matrix}; $0\notin\operatorname{spec}(I_n+E)$; $I_n+E$ invertible by Corollary \ref{cor:matrix-invertible-iff-zero-not-spectrum}.\\
\hlt{Structural Commentary}: The spectral-radius bound excludes $-1$, and spectral mapping transfers the exclusion to $0$.\\
\hlt{Key Insight}: $\norm{E}_2<1$ keeps $-1$ outside $\operatorname{spec}(E)$.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:bauer-fike-eigenvalue-perturbation} \hlt{Bauer--Fike Eigenvalue Perturbation Bound}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ be diagonalizable. Fix $S\in\operatorname{Mat}(n,n,\mathbb{C})^\times$ and $\Lambda=\operatorname{diag}(\lambda_1,\ldots,\lambda_n)$ satisfying $A=S\Lambda S^{-1}$.\\
If $\Delta A\in\operatorname{Mat}(n,n,\mathbb{C})$ and $\mu\in\operatorname{spec}(A+\Delta A)$, then $\min\limits_{j\in J_n}\abs{\mu-\lambda_j}\leq\kappa_2(S)\norm{\Delta A}_2$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Separation}: If $(\exists\,j\in J_n)\,\mu=\lambda_j$, bound is immediate. Otherwise set $\delta\defeq\min\limits_{j\in J_n}\abs{\mu-\lambda_j}>0$, so $\Lambda-\mu I_n$ is invertible.
\item \hlt{Shift Factorization}: Since $\mu\in\operatorname{spec}(A+\Delta A)$, the matrix $A+\Delta A-\mu I_n$ is singular. Set $F\defeq(\Lambda-\mu I_n)^{-1}S^{-1}\Delta AS$.\\
The identity $S^{-1}(A+\Delta A-\mu I_n)S=(\Lambda-\mu I_n)(I_n+F)$ holds. Thus $I_n+F$ is singular because $S$ and $\Lambda-\mu I_n$ are invertible.
\item \hlt{Norm Obstruction}: The inequality $\norm{F}_2\geq1$ holds by the contrapositive of Lemma \ref{lem:small-spectral-norm-perturbation-invertible}.
\item \hlt{Resolvent Estimate}: Diagonal resolvent has singular values $(\abs{\lambda_j-\mu}^{-1})_{j\in J_n}$. Thus $\norm{(\Lambda-\mu I_n)^{-1}}_2=\delta^{-1}$ by Theorem \ref{thm:spectral-norm-largest-singular-value}(i).\\
By part (iii) and Proposition \ref{prop:spectral-norm-basic-properties}(v), $1\leq\norm{F}_2\leq\delta^{-1}\norm{S^{-1}}_2\norm{\Delta A}_2\norm{S}_2=\delta^{-1}\kappa_2(S)\norm{\Delta A}_2$. Multiply by $\delta$ for bound.
\end{enumerate}
\hlt{Structural Commentary}: Shift factorization converts a perturbed eigenvalue into failure of the small-norm invertibility lemma.\\
\hlt{Key Insight}: Singularity forces $\norm{(\Lambda-\mu I_n)^{-1}S^{-1}\Delta AS}_2\geq1$.
\end{proof}
\end{theorem}

\begin{computationmethod} \label{meth:bauer-fike-perturbation-method} \hlt{Bauer--Fike Perturbation Method}\\
For diagonalizable $A\in\operatorname{Mat}(n,n,\mathbb{C})$ and $\Delta A\in\operatorname{Mat}(n,n,\mathbb{C})$, construct a spectral enclosure:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item choose a diagonalization $A=S\Lambda S^{-1}$ with $S$ invertible and $\Lambda=\operatorname{diag}(\lambda_1,\ldots,\lambda_n)$;
\item compute $\norm{\Delta A}_2$;
\item compute $\kappa_2(S)$ and set $r\defeq\kappa_2(S)\norm{\Delta A}_2$;
\item output $\operatorname{spec}(A+\Delta A)\subseteq\bigcup\limits_{j\in J_n}\{z\in\mathbb{C}:\abs{z-\lambda_j}\leq r\}$ by Theorem \ref{thm:bauer-fike-eigenvalue-perturbation}.
\end{enumerate}
\end{computationmethod}

\begin{definition} \label{def:gersgorin-disks} \hlt{Gersgorin Disks}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{deleted absolute row sums} are defined by $(\forall\,k\in J_n)\,R_k^{\mathrm{off}}(A)\defeq\sum\limits_{j\in J_n\setminus\{k\}}\abs{a_{kj}}$.
\item The \hlt{absolute row sums} are defined by $(\forall\,k\in J_n)\,R_k(A)\defeq\sum\limits_{j\in J_n}\abs{a_{kj}}$.
\item The \hlt{maximum absolute row sum} is $R_{\max}(A)\defeq\max\limits_{k\in J_n}R_k(A)$.
\item The \hlt{Gersgorin disks} are $(\forall\,k\in J_n)\,\operatorname{Ger}_k(A)\defeq\{z\in\mathbb{C}:\abs{z-a_{kk}}\leq R_k^{\mathrm{off}}(A)\}$. Their union is $\operatorname{Ger}(A)\defeq\bigcup\limits_{k\in J_n}\operatorname{Ger}_k(A)$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:gersgorin-disk-theorem} \hlt{Gersgorin Disk Theorem}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $\operatorname{spec}(A)\subseteq\operatorname{Ger}(A)=\bigcup\limits_{k\in J_n}\operatorname{Ger}_k(A)$.
\begin{proof}
Fix $\lambda\in\operatorname{spec}(A)$ and $\mathbf{0}\neq\mathbf{x}=[x_1\ \cdots\ x_n]^t\in E_{\lambda}(A)$. Choose $k\in J_n$ with $\abs{x_k}=\max\limits_{j\in J_n}\abs{x_j}>0$.\\
The $k$th row of $A\mathbf{x}=\lambda\mathbf{x}$ is $(\lambda-a_{kk})x_k=\sum\limits_{j\in J_n\setminus\{k\}}a_{kj}x_j$.\\
Therefore $\abs{\lambda-a_{kk}}\leq\sum\limits_{j\in J_n\setminus\{k\}}\abs{a_{kj}}\abs{\frac{x_j}{x_k}}\leq R_k^{\mathrm{off}}(A)$ by the triangle inequality and maximality of $\abs{x_k}$;  $\lambda\in\operatorname{Ger}_k(A)\subseteq\operatorname{Ger}(A)$.\\
\hlt{Structural Commentary}: The maximal eigenvector component converts one row identity into the deleted-row-sum bound.\\
\hlt{Key Insight}: $(\forall\,j\in J_n)\,\abs{\frac{x_j}{x_k}}\leq1$ closes the disk estimate.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:gersgorin-row-column-refinement} \hlt{Row-Column Gersgorin Refinement}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{spec}(A)\subseteq\operatorname{Ger}(A)\cap\operatorname{Ger}(A^t)$.
\item $\operatorname{spec}(A)\subseteq\{z\in\mathbb{C}:\abs{z}\leq\min(R_{\max}(A),R_{\max}(A^t))\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] The inclusion $\operatorname{spec}(A)\subseteq\operatorname{Ger}(A)$ holds by Theorem \ref{thm:gersgorin-disk-theorem}.\\
Since $\operatorname{spec}(A)=\operatorname{spec}(A^t)$ by Proposition \ref{prop:transpose-preserves-spectrum-and-geometric-multiplicity}(ii), $\operatorname{spec}(A)\subseteq\operatorname{Ger}(A^t)$ by Theorem \ref{thm:gersgorin-disk-theorem}; intersection is (i).
\item[$(ii)$] For $B=[b_{ij}]\in\{A,A^t\}$, $k\in J_n$, and $z\in\operatorname{Ger}_k(B)$, $\abs{z}\leq\abs{b_{kk}}+R_k^{\mathrm{off}}(B)=R_k(B)\leq R_{\max}(B)$ by the triangle inequality.\\
Thus $\operatorname{Ger}(B)\subseteq\{z\in\mathbb{C}:\abs{z}\leq R_{\max}(B)\}$. The enclosure in (ii) follows from the two radial bounds by part (i).
\end{enumerate}
\hlt{Structural Commentary}: Gersgorin applied to $A$ and $A^t$ supplies intersecting row and column enclosures.\\
\hlt{Key Insight}: The triangle inequality places each disk inside its row-sum radius.
\end{proof}
\end{corollary}

\begin{computationmethod} \label{meth:gersgorin-disk-localisation} \hlt{Gersgorin Localisation Method}\\
For $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$, construct a row-column spectral enclosure as follows:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Compute $(\forall\,k\in J_n)\,R_k^{\mathrm{off}}(A)=\sum\limits_{j\in J_n\setminus\{k\}}\abs{a_{kj}}$. Form $\operatorname{Ger}(A)=\bigcup\limits_{k\in J_n}\operatorname{Ger}_k(A)$.
\item Record the certified row enclosure $\operatorname{spec}(A)\subseteq\operatorname{Ger}(A)$ by Theorem \ref{thm:gersgorin-disk-theorem}.
\item Compute the deleted row sums of $A^t$, which are the deleted column sums of $A$, and form $\operatorname{Ger}(A)\cap\operatorname{Ger}(A^t)$.\\
The inclusion $\operatorname{spec}(A)\subseteq\operatorname{Ger}(A)\cap\operatorname{Ger}(A^t)$ holds by Corollary \ref{cor:gersgorin-row-column-refinement}(i).
\item Interpret the output as an intersection of two unions; do not replace it by $\bigcup\limits_{k\in J_n}(\operatorname{Ger}_k(A)\cap\operatorname{Ger}_k(A^t))$.
\end{enumerate}
\end{computationmethod}

\begin{definition} \label{def:closed-spectral-norm-pseudospectrum} \hlt{Closed Spectral-Norm Pseudospectrum}\\
Let $\mathcal{M}\defeq\operatorname{Mat}(n,n,\mathbb{C})$, $A\in\mathcal{M}$, and $\varepsilon\in\mathbb{R}_{\geq0}$.\\
The \hlt{closed spectral-norm $\varepsilon$-pseudospectrum} of $A$ is $\{z\in\mathbb{C}:(\exists\,\Delta A\in\mathcal{M})\,[\norm{\Delta A}_2\leq\varepsilon\land z\in\operatorname{spec}(A+\Delta A)]\}$.
\end{definition}

Let $A,\Delta A\in\operatorname{Mat}(n,n,\mathbb{C})$, let $\varepsilon\in\mathbb{R}_{\geq0}$, and fix a diagonalization $A=S\Lambda S^{-1}$.\\
If $\norm{\Delta A}_2\leq\varepsilon$, then the Bauer--Fike disk radius is at most $\kappa_2(S)\varepsilon$ by Theorem \ref{thm:bauer-fike-eigenvalue-perturbation}. The inclusion $\operatorname{spec}(A+\Delta A)\subseteq\operatorname{Ger}(A+\Delta A)$ holds by Theorem \ref{thm:gersgorin-disk-theorem}.\\
These bounds distinguish spectral location from perturbation sensitivity.
