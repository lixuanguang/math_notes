\subsection{Matrix Norms}

% **Role:** quantitative matrix behavior and computation.
%
% Operator norms, Frobenius norm, singular values, condition numbers, norm inequalities.

Matrix norms measure stretch, energy, approximation quality, and sensitivity. Singular values organize norms, low-rank approximation, pseudoinverses, conditioning, and error bounds.

\begin{definition} \label{def:spectral-norm-and-singular-extremes} \hlt{Spectral Norm and Extreme Singular Values}\\
For $A\in\operatorname{Mat}(m,n,\mathbb{F})$, define the \hlt{spectral norm} $\norm{A}_2\defeq\max\limits_{\norm{\mathbf{x}}_2=1}\norm{A\mathbf{x}}_2$.\\
If the singular values of $A$ are $\sigma_1\geq\cdots\geq\sigma_q\geq0$, $q=\min(m,n)$, set $\sigma_{\max}(A)\defeq\sigma_1$.\\
When $A$ is square, $\sigma_{\min}(A)\defeq\sigma_n$.
\end{definition}

\begin{theorem} \label{thm:spectral-norm-largest-singular-value} \hlt{Spectral Norm is the Largest Singular Value}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ have singular values $\sigma_1\geq\cdots\geq\sigma_q\geq0$. Then $\norm{A}_2=\sigma_{\max}(A)$.\\
Moreover, if $A=V\Sigma W^*$ is an SVD and $\mathbf{w}_1$ is the first column of $W$, then $\norm{A\mathbf{w}_1}_2=\sigma_1$.
\begin{proof}
Let $\mathbf{w}_1,\ldots,\mathbf{w}_n$ be the columns of $W$. For $\mathbf{x}=\sum_{j=1}^{n}c_j\mathbf{w}_j$ with $\norm{\mathbf{x}}_2=1$, unitary invariance of the Euclidean norm gives $\norm{A\mathbf{x}}_2^2 =\norm{\Sigma W^*\mathbf{x}}_2^2 =\sum\limits_{j=1}^{q}\sigma_j^2\abs{c_j}^2 \leq \sigma_1^2\sum\limits_{j=1}^{n}\abs{c_j}^2 =\sigma_1^2$.\\
Thus $\norm{A}_2\leq\sigma_1$. Taking $\mathbf{x}=\mathbf{w}_1$ gives equality.\\
\hlt{Structural Commentary}: The right singular basis diagonalises the stretching computation.\\
\hlt{Key Insight}: The largest stretch occurs in a top right singular direction.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:spectral-norm-basic-properties} \hlt{Basic Properties of the Spectral Norm}\\
For compatible matrices $A,B$ and scalar $c$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\norm{A}_2\geq0$, and $\norm{A}_2=0$ if and only if $A=0$;
\item $\norm{cA}_2=\abs{c}\norm{A}_2$;
\item $\norm{A+B}_2\leq\norm{A}_2+\norm{B}_2$;
\item $\norm{AB}_2\leq\norm{A}_2\norm{B}_2$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $\norm{A\mathbf{x}}_2\geq0$, $\norm{A}_2\geq0$. Also $\norm{A}_2=0$ iff $A\mathbf{x}=0$ for every unit $\mathbf{x}$, iff $A=0$.
\item $\norm{cA}_2=\max_{\norm{\mathbf{x}}_2=1}\norm{cA\mathbf{x}}_2=\abs{c}\norm{A}_2$.
\item For $\norm{\mathbf{x}}_2=1$, $\norm{(A+B)\mathbf{x}}_2\leq\norm{A\mathbf{x}}_2+\norm{B\mathbf{x}}_2\leq\norm{A}_2+\norm{B}_2$. Maximize over unit $\mathbf{x}$.
\item For $\norm{\mathbf{x}}_2=1$, $\norm{AB\mathbf{x}}_2\leq\norm{A}_2\norm{B\mathbf{x}}_2\leq\norm{A}_2\norm{B}_2$. Maximize over unit $\mathbf{x}$.
\end{enumerate}
\hlt{Structural Commentary}: The spectral norm is the operator norm induced by Euclidean geometry.\\
\hlt{Key Insight}: Bound each output vector first, then take the worst case.
\end{proof}
\end{proposition}

\begin{definition} \label{def:operator-norm-rn-rm} \hlt{Operator Norm on $\LL(\R^n,\R^m)$}\\
For $A\in\LL(\R^n,\R^m)$, define $\normop{A}\defeq \sup_{\norm{\h}_2\leq1}\norm{A\h}_2$.\\
Equivalently, after standard bases identify $A$ with its matrix, $\normop{A}$ is the spectral norm from Definition \ref{def:spectral-norm-and-singular-extremes}.\\
Thus $\norm{A\h}_2\leq\normop{A}\norm{\h}_2$, and sums and products obey the norm estimates in Proposition \ref{prop:spectral-norm-basic-properties}.\\
The induced metric on $\LL(\R^n,\R^m)$ is $d(A,B)\defeq\normop{A-B}$; this is the metric used for continuity of derivative maps in Definition \ref{def:c1-map}.
\end{definition}

\begin{proposition} \label{prop:frobenius-and-spectral-norm-comparison} \hlt{Frobenius and Spectral Norm Comparison}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ have singular values $\sigma_1\geq\cdots\geq\sigma_q\geq0$. Then $\norm{A}_F^2=\sum_{j=1}^{q}\sigma_j^2$ and $\norm{A}_2\leq\norm{A}_F$.\\
If $A\neq0$, equality $\norm{A}_2=\norm{A}_F$ holds if and only if $\operatorname{rank}(A)=1$.
\begin{proof}
The Frobenius identity is Proposition \ref{prop:basic-singular-value-identities}. Hence $\norm{A}_2^2=\sigma_1^2\leq\sum_{j=1}^{q}\sigma_j^2=\norm{A}_F^2$. Assume $A\neq0$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\norm{A}_2=\norm{A}_F$, then $\sum_{j=2}^{q}\sigma_j^2=0$, so only $\sigma_1$ is nonzero and $\operatorname{rank}(A)=1$.
\item[$(\Leftarrow)$] If $\operatorname{rank}(A)=1$, then $\sigma_2=\cdots=\sigma_q=0$, so $\norm{A}_2=\norm{A}_F$.
\end{enumerate}
\hlt{Structural Commentary}: Frobenius counts all singular energy; spectral keeps the largest part.\\
\hlt{Key Insight}: Equality means exactly one nonzero singular value.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:frobenius-and-spectral-submultiplicativity} \hlt{Submultiplicativity of Frobenius and Spectral Norms}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $B\in\operatorname{Mat}(n,k,\mathbb{F})$. Then $\norm{AB}_F\leq\norm{A}_2\norm{B}_F\leq\norm{A}_F\norm{B}_F$ and $\norm{AB}_2\leq\norm{A}_2\norm{B}_2$.
\begin{proof}
Write $B=[\mathbf{b}_1\ \cdots\ \mathbf{b}_k]$. Then $\norm{AB}_F^2 =\sum\limits_{j=1}^{k}\norm{A\mathbf{b}_j}_2^2 \leq\norm{A}_2^2\sum\limits_{j=1}^{k}\norm{\mathbf{b}_j}_2^2 =\norm{A}_2^2\norm{B}_F^2$.\\
Proposition \ref{prop:frobenius-and-spectral-norm-comparison} gives $\norm{A}_2\leq\norm{A}_F$. The spectral-norm inequality is Proposition \ref{prop:spectral-norm-basic-properties}.\\
\hlt{Structural Commentary}: Frobenius norm is column energy; each column of $AB$ is $A$ applied to a column of $B$.\\
\hlt{Key Insight}: Control every column by the spectral norm of $A$.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:unitary-invariance-of-singular-values} \hlt{Unitary Invariance of Singular Values}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, and let $U\in\operatorname{Mat}(m,m,\mathbb{F})$ and $V\in\operatorname{Mat}(n,n,\mathbb{F})$ be unitary. Then $A$ and $UAV$ have the same singular values. In particular, $\norm{UAV}_2=\norm{A}_2$ and $\norm{UAV}_F=\norm{A}_F$.
\begin{proof}
Since $(UAV)^*(UAV)=V^*A^*AV$, the Gram matrices of $A$ and $UAV$ are unitarily similar, hence have the same eigenvalues. Singular values are the nonnegative square roots of these eigenvalues. The two norm identities follow from Theorem \ref{thm:spectral-norm-largest-singular-value} and Proposition \ref{prop:frobenius-and-spectral-norm-comparison}.\\
\hlt{Structural Commentary}: Left and right unitary factors change coordinates, not singular-value data.\\
\hlt{Key Insight}: Singular values are spectral data of $A^*A$, and unitary conjugation preserves that spectrum.
\end{proof}
\end{proposition}

\begin{example} \label{ex:similarity-need-not-preserve-singular-values} \hlt{Similarity Need Not Preserve Singular Values}\\
Let $A=\begin{bmatrix}1&1\\0&1\end{bmatrix}$ and $B=\begin{bmatrix}1&2\\0&1\end{bmatrix}$. Then $B=DAD^{-1}$ with $D=\operatorname{diag}(2,1)$, so $A$ and $B$ are similar.\\
However $A^*A=\begin{bmatrix}1&1\\1&2\end{bmatrix}$, $B^*B=\begin{bmatrix}1&2\\2&5\end{bmatrix}$ have different eigenvalues, so $A$ and $B$ have different singular values.\\
Singular values are preserved by unitary left-right equivalence, not by arbitrary similarity.
\end{example}

\begin{theorem} \label{thm:schur-inequality-defect-normality}\label{thm:schur-inequality-normal-defect} \hlt{Schur's Inequality and Normality}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have eigenvalues $\lambda_1,\ldots,\lambda_n$, counted algebraically.\\
Then $\sum\limits_{j=1}^{n}\abs{\lambda_j}^2\leq\norm{A}_F^2$ with equality if and only if $A$ is normal.
\begin{proof}
Let $A=UTU^*$ be a Schur form, where $T=[t_{ij}]$ is upper triangular and $t_{jj}=\lambda_j$.\\
Then $\norm{A}_F^2=\operatorname{tr}(T^*T)=\sum_{j=1}^{n}\abs{\lambda_j}^2+\sum_{i<j}\abs{t_{ij}}^2$.
Inequality holds. Equality holds exactly when $T$ is diagonal.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If equality holds, then $T$ is diagonal, so $A$ is unitarily diagonalizable and hence normal.
\item[$(\Leftarrow)$] If $A$ is normal, then $T$ is normal and upper triangular, hence diagonal by Lemma \ref{lem:upper-triangular-normal-matrix-is-diagonal}.
\end{enumerate}
\hlt{Structural Commentary}: Schur form splits Frobenius norm into eigenvalue mass plus off-diagonal mass.\\
\hlt{Key Insight}: Normality means the Schur off-diagonal part vanishes.
\end{proof}
\end{theorem}

\begin{definition} \label{def:defect-from-normality} \hlt{Defect from Normality}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have eigenvalues $\lambda_1,\ldots,\lambda_n$, counted with algebraic multiplicity.\\
The \hlt{defect from normality} of $A$ is $\Delta(A)\defeq\norm{A}_F^2-\sum\limits_{j=1}^{n}\abs{\lambda_j}^2$.\\
By Theorem \ref{thm:schur-inequality-defect-normality}, $\Delta(A)\geq0$, with equality if and only if $A$ is normal.
\end{definition}

\begin{theorem} \label{thm:normality-transfer-ab-ba} \hlt{Normality Transfers from $AB$ to $BA$}\\
Let $A,B\in\operatorname{Mat}(n,n,\C)$. If $A$, $B$, and $AB$ are normal, then $BA$ is normal.
\begin{proof}
By Theorem \ref{thm:ab-ba-nonzero-eigenvalues}, $AB$ and $BA$ have the same eigenvalues counted with algebraic multiplicity. Also,
\begin{align}
\operatorname{tr}\bigl((BA)^*(BA)\bigr) &=\operatorname{tr}(A^*B^*BA) =\operatorname{tr}(AA^*B^*B) \nonumber\\
&=\operatorname{tr}(A^*AB^*B) =\operatorname{tr}(A^*A BB^*) =\operatorname{tr}(B^*A^*AB) =\operatorname{tr}\bigl((AB)^*(AB)\bigr),
\nonumber
\end{align}
using cyclicity of trace and normality of $A$ and $B$. Thus $\norm{BA}_F^2=\norm{AB}_F^2$, so $\Delta(BA)=\Delta(AB)$.\\
Since $AB$ is normal, $\Delta(AB)=0$; hence $\Delta(BA)=0$, and Theorem \ref{thm:schur-inequality-normal-defect} gives that $BA$ is normal.\\
\hlt{Structural Commentary}: Same spectrum and Frobenius norm give the same Schur defect.\\
\hlt{Key Insight}: Schur defect turns normality of $AB$ into normality of $BA$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:eigenvalues-versus-singular-values} \hlt{Eigenvalues Versus Singular Values}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have eigenvalues $\lambda_1,\ldots,\lambda_n$ ordered so that $\abs{\lambda_1}\geq\cdots\geq\abs{\lambda_n}$,\\
and singular values $\sigma_1\geq\cdots\geq\sigma_n\geq0$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\abs{\det A}=\prod_{j=1}^{n}\abs{\lambda_j}=\prod_{j=1}^{n}\sigma_j$;
\item $\rho(A)\leq\norm{A}_2$;
\item $\sigma_n\leq\abs{\lambda_n}$;
\item $A$ is normal if and only if $\abs{\lambda_j}=\sigma_j$ for every $j$;
\item $A$ is positive semidefinite if and only if $\lambda_j=\sigma_j$ for every $j$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The eigenvalue product gives $\abs{\det A}=\prod_j\abs{\lambda_j}$.\\
If $A=V\Sigma W^*$ is an SVD, then $\abs{\det V}=\abs{\det W}=1$, so $\abs{\det A}=\abs{\det\Sigma}=\prod_j\sigma_j$.
\item If $A\mathbf{x}=\lambda\mathbf{x}$ with $\norm{\mathbf{x}}_2=1$, then $\abs{\lambda}=\norm{A\mathbf{x}}_2\leq\norm{A}_2$. Maximize over eigenvalues.
\item If $A$ is singular, both sides are $0$ or the inequality is immediate.\\
If $A$ is invertible, apply (ii) to $A^{-1}$: $\frac{1}{\abs{\lambda_n}}=\rho(A^{-1})\leq\norm{A^{-1}}_2=\frac{1}{\sigma_n}$.
\item If $A$ is normal, the spectral theorem gives $\sigma_j=\abs{\lambda_j}$. Conversely, equality gives $\sum_j\abs{\lambda_j}^2=\sum_j\sigma_j^2=\norm{A}_F^2$, so Theorem \ref{thm:schur-inequality-defect-normality} gives normality.
\item If $A\succeq0$, then $A$ is normal and $\lambda_j\geq0$, so (iv) gives $\lambda_j=\sigma_j$. Conversely, equality gives normality with nonnegative real eigenvalues, hence $A\succeq0$ by Theorem \ref{thm:positive-semidefinite-matrix-equivalences}.
\end{enumerate}
\hlt{Structural Commentary}: Normality aligns eigenvalue moduli with singular values; positivity removes the moduli.\\
\hlt{Key Insight}: Normality is the bridge that turns spectral data into singular-value data.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:trace-bounded-by-singular-values} \hlt{Trace Bound by Singular Values}\\
If $A\in\operatorname{Mat}(n,n,\mathbb{C})$, then $\abs{\operatorname{tr}A}\leq\sum\limits_{j=1}^{n}\sigma_j(A)$. Consequently, $\sum\limits_{j=1}^{n}\abs{\lambda_j(A)}\leq\sum\limits_{j=1}^{n}\sigma_j(A)$ and $\sum\limits_{j=1}^{n}\abs{a_{jj}}\leq\sum\limits_{j=1}^{n}\sigma_j(A)$.
\begin{proof}
Let $A=V\Sigma W^*$ be an SVD.\\
Writing $\mathbf{v}_k,\mathbf{w}_k$ for the columns of $V,W$, $\operatorname{tr}A =\sum\limits_{k=1}^{n}\sigma_k\mathbf{w}_k^*\mathbf{v}_k$, $\abs{\mathbf{w}_k^*\mathbf{v}_k}\leq1$, so $\abs{\operatorname{tr}A}\leq\sum\limits_k\sigma_k$.\\
For eigenvalues, take a Schur form $T$ of $A$ and a diagonal unitary $D$ with $\operatorname{tr}(DT)=\sum\limits_j\abs{\lambda_j}$.\\
For diagonal entries, take diagonal unitary $D$ with $\operatorname{tr}(DA)=\sum\limits_j\abs{a_{jj}}$.\\
In both cases, unitary invariance of singular values and the trace bound give the result.\\
\hlt{Structural Commentary}: Phase rotations turn trace estimates into eigenvalue and diagonal estimates.\\
\hlt{Key Insight}: Prove $\abs{\operatorname{tr}A}\leq\sum\limits_j\sigma_j$ once, then rotate phases.
\end{proof}
\end{theorem}
