\subsection{Quadratic Forms and Polar Forms}

Throughout this subsection, $\mathcal{V}$ is a vector space over an arbitrary field $\mathbb{F}$.

\begin{definition} \label{def:quadratic-form-and-polar-form} \hlt{Quadratic Forms and Polar Forms}\\
A \hlt{quadratic form} on $\mathcal{V}$ is a function $Q:\mathcal{V}\to\mathbb{F}$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,a\in\mathbb{F})(\forall\,\mathbf{v}\in\mathcal{V})\,Q(a\mathbf{v})=a^2Q(\mathbf{v})$;
\item the \hlt{polar form} $B_Q:\mathcal{V}\times\mathcal{V}\to\mathbb{F}$, defined by $B_Q(\mathbf{u},\mathbf{v})\defeq Q(\mathbf{u}+\mathbf{v})-Q(\mathbf{u})-Q(\mathbf{v})$, is bilinear.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:quadratic-forms-symmetric-bilinear-forms} \hlt{Quadratic Forms and Symmetric Bilinear Forms}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $Q:\mathcal{V}\to\mathbb{F}$ is quadratic, then $\operatorname{Sym}(B_Q)$.
\item Suppose $\operatorname{char}(\mathbb{F})\neq2$ and $B:\mathcal{V}\times\mathcal{V}\to\mathbb{F}$ is symmetric and bilinear. If $Q_B(\mathbf{x})\defeq\frac{1}{2}B(\mathbf{x},\mathbf{x})$, then $Q_B$ is quadratic and $B_{Q_B}=B$.
\item Suppose $\operatorname{char}(\mathbb{F})\neq2$ and $Q:\mathcal{V}\to\mathbb{F}$ is quadratic. If $Q_{B_Q}(\mathbf{x})\defeq\frac{1}{2}B_Q(\mathbf{x},\mathbf{x})$, then $Q_{B_Q}=Q$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] By commutativity of vector addition, $(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})\,B_Q(\mathbf{u},\mathbf{v})=Q(\mathbf{u}+\mathbf{v})-Q(\mathbf{u})-Q(\mathbf{v})=B_Q(\mathbf{v},\mathbf{u})$. Hence $\operatorname{Sym}(B_Q)$.
\item[$(ii)$] By bilinearity, $(\forall\,a\in\mathbb{F})(\forall\,\mathbf{x}\in\mathcal{V})\,Q_B(a\mathbf{x})=a^2Q_B(\mathbf{x})$. Fix $\mathbf{u},\mathbf{v}\in\mathcal{V}$.\\
Then $B_{Q_B}(\mathbf{u},\mathbf{v})=Q_B(\mathbf{u}+\mathbf{v})-Q_B(\mathbf{u})-Q_B(\mathbf{v})=\frac{1}{2}\bigl(B(\mathbf{u},\mathbf{v})+B(\mathbf{v},\mathbf{u})\bigr)=B(\mathbf{u},\mathbf{v})$.\\
Thus $B_{Q_B}=B$ is bilinear, and $Q_B$ is quadratic.
\item[$(iii)$] Fix $\mathbf{x}\in\mathcal{V}$. Then $Q_{B_Q}(\mathbf{x})=\frac{1}{2}B_Q(\mathbf{x},\mathbf{x})=\frac{1}{2}\bigl(Q(2\mathbf{x})-2Q(\mathbf{x})\bigr)$. Since $Q(2\mathbf{x})=4Q(\mathbf{x})$, $Q_{B_Q}(\mathbf{x})=Q(\mathbf{x})$. Hence $Q_{B_Q}=Q$.
\end{enumerate}
\hlt{Structural Commentary}: Items (ii) and (iii) are the two reconstruction identities underlying the odd-characteristic correspondence.\\
\hlt{Key Insight}: When $2$ is invertible, diagonal evaluation and polarization recover each other.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:polar-forms-alternating-in-characteristic-two} \hlt{Polar Forms in Characteristic Two}\\
Let $Q:\mathcal{V}\to\mathbb{F}$ be quadratic. If $\operatorname{char}(\mathbb{F})=2$, then $\operatorname{Alt}(B_Q)$.
\begin{proof} By homogeneity with scalar $0$, $Q(\mathbf{0})=0$. Since $\operatorname{char}(\mathbb{F})=2$, $(\forall\,\mathbf{x}\in\mathcal{V})\,B_Q(\mathbf{x},\mathbf{x})=Q(2\mathbf{x})-2Q(\mathbf{x})=Q(\mathbf{0})=0$. Thus $\operatorname{Alt}(B_Q)$.\\
\hlt{Structural Commentary}: Diagonal evaluation makes every polar form in characteristic $2$ alternating.\\
\hlt{Key Insight}: The identities $2\mathbf{x}=\mathbf{0}$ and $2Q(\mathbf{x})=0$ force the polar diagonal to vanish.
\end{proof}
\end{proposition}

\begin{example} \label{ex:characteristic-two-polar-form-loses-quadratic-data} \hlt{Characteristic Two Loses Diagonal Data}\\
Assume $\operatorname{char}(\mathbb{F})=2$. On $\mathbb{F}^2$, let $Q_0(x,y)=xy$ and $Q_1(x,y)=xy+x^2$.\\
The identity $(\forall\,i\in\{0,1\})(\forall\,a,x,y\in\mathbb{F})\,Q_i(ax,ay)=a^2Q_i(x,y)$ verifies homogeneity. Define $B((x,y),(u,v))=xv+uy$.\\
Then $B_{Q_0}=B_{Q_1}=B$, and $B$ is bilinear. Hence $Q_0$ and $Q_1$ are quadratic with the same polar form, but $Q_0(1,0)=0\neq1=Q_1(1,0)$.\\
More generally, for quadratic $Q$ and linear $L:\mathcal{V}\to\mathbb{F}$, set $L^2(\mathbf{v})\defeq L(\mathbf{v})^2$. Then $(\forall\,a\in\mathbb{F})(\forall\,\mathbf{v}\in\mathcal{V})\,(Q+L^2)(a\mathbf{v})=a^2(Q+L^2)(\mathbf{v})$, while $(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})\,B_{Q+L^2}(\mathbf{u},\mathbf{v})=B_Q(\mathbf{u},\mathbf{v})+2L(\mathbf{u})L(\mathbf{v})=B_Q(\mathbf{u},\mathbf{v})$. Hence $Q+L^2$ is quadratic.\\
\hlt{Structural Commentary}: The pair $Q_0,Q_1$ exemplifies the general square perturbations that polarization cannot detect.\\
\hlt{Key Insight}: In characteristic $2$, a polar form does not determine the diagonal quadratic data.
\end{example}

\begin{example} \label{ex:characteristic-two-alternating-versus-nonalternating-symmetric} \hlt{Alternating versus Nonalternating Symmetric Forms}\\
Assume $\operatorname{char}(\mathbb{F})=2$. The matrix $H=\left[\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\right]$ defines the symmetric bilinear form $B_H((x,y),(u,v))=xv+yu$ on $\mathbb{F}^2$.\\
Since $(\forall\,x,y\in\mathbb{F})\,B_H((x,y),(x,y))=2xy=0$, the form is alternating.\\
Since $H^2=I_2$, the matrix is invertible, so the form is nonsingular by Proposition \ref{prop:square-bilinear-pairing-equivalences}.\\
Suppose a diagonal matrix $D$ is congruent to $H$. By Proposition \ref{prop:metric-form-matrix-recovery-and-congruence}(iii), $D$ represents $B_H$ in some ordered basis.\\
Alternation forces every diagonal entry of $D$ to vanish, so $D=0$. Thus $H$ is not congruent to a nonzero diagonal matrix.\\
By contrast, $[1]$ is symmetric and nonalternating because its basis vector has self-pairing $1$.\\
\hlt{Structural Commentary}: The forms represented by $H$ and $[1]$ separate the alternating and nonalternating symmetric cases.\\
\hlt{Key Insight}: A nonsingular alternating form has zero diagonal entries in every basis.
\end{example}
