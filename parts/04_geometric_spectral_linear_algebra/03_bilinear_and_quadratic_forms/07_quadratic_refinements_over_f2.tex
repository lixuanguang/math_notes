\subsection{Quadratic Refinements over $\mathbb{F}_2$}

Throughout this subsection, vector spaces are finite-dimensional over $\mathbb{F}_2$.

\begin{definition} \label{def:arf-invariant-over-f2} \hlt{Quadratic Isometries and Arf Sums over $\mathbb{F}_2$}\\
Let $Q:\mathcal{V}\to\mathbb{F}_2$ be quadratic with nonsingular polar form $B_Q$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $Q':\mathcal{W}\to\mathbb{F}_2$ is quadratic, a \hlt{quadratic isometry} $Q\to Q'$ is a linear isomorphism $\sigma:\mathcal{V}\to\mathcal{W}$ satisfying $(\forall\,\mathbf{x}\in\mathcal{V})\,Q'(\sigma\mathbf{x})=Q(\mathbf{x})$.\\
Write $Q\cong Q'$ if such a map exists.
\item Given a symplectic basis $\beta=(\mathbf{u}_1,\mathbf{v}_1,\ldots,\mathbf{u}_m,\mathbf{v}_m)$ of $B_Q$, its \hlt{Arf sum} is $\operatorname{Arf}_{\beta}(Q)\defeq\sum\limits_{i=1}^{m}Q(\mathbf{u}_i)Q(\mathbf{v}_i)\in\mathbb{F}_2$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:characteristic-two-quadratic-refinements-finite-field} \hlt{One Polar Form with Two Refinements}\\
On $\mathcal{V}=\mathbb{F}_2^2$, put $H(x,y)=xy$ and $A(x,y)=x^2+xy+y^2$.\\
Both have polar form $B((x,y),(u,v))=xv+yu$, whose radical is zero because $B((x,y),(0,1))=x$ and $B((x,y),(1,0))=y$.\\
However, $H^{-1}(0)\setminus\{(0,0)\}=\{(1,0),(0,1)\}$, whereas $A^{-1}(0)=\{(0,0)\}$.\\
Hence $H\not\cong A$ because a quadratic isometry preserves the zero set.
\end{example}

\begin{theorem} \label{thm:f2-quadratic-refinements-arf-classification} \hlt{Arf Classification over $\mathbb{F}_2$}\\
Let $Q:\mathcal{V}\to\mathbb{F}_2$ be quadratic with nonsingular polar form $B_Q$, and let $\mathscr{B}_Q$ be the set of symplectic bases of $B_Q$.\\
For $m\geq1$, $a\in\mathbb{F}_2$, define $Q_a^{(m)}(x_1,y_1,\ldots,x_m,y_m)\defeq x_1y_1+a(x_1^2+y_1^2)+\sum\limits_{i=2}^{m}x_iy_i$. Let $Q_0^{(0)}$ be zero form on $\{\mathbf{0}\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\exists!\,m\in\mathbb{Z}_{\geq0})\,\dim\mathcal{V}=2m$.
\item For this $m$, $(\forall\,\beta\in\mathscr{B}_Q)\,Q\cong Q_{\operatorname{Arf}_{\beta}(Q)}^{(m)}$.
\item If $m\geq1$, then $(\forall\,a\in\mathbb{F}_2)\,\lvert(Q_a^{(m)})^{-1}(0)\rvert=2^{2m-1}+(-1)^a2^{m-1}$; in particular, $Q_0^{(m)}\not\cong Q_1^{(m)}$.
\item $(\forall\,\beta,\gamma\in\mathscr{B}_Q)\,\operatorname{Arf}_{\beta}(Q)=\operatorname{Arf}_{\gamma}(Q)$; denote this common value by $\operatorname{Arf}(Q)$.
\item $Q\cong Q_{\operatorname{Arf}(Q)}^{(m)}$. If $m\geq1$, then $(\forall\,a\in\mathbb{F}_2)\,\operatorname{Arf}(Q_a^{(m)})=a$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Form $B_Q$ is alternating by Proposition \ref{prop:polar-forms-alternating-in-characteristic-two}. By Theorem \ref{thm:symplectic-geometries-hyperbolic-classification}(i), it has symplectic basis with $2m$ vectors for unique $m\in\mathbb{Z}_{\geq0}$.
\item If $m=0$, then $Q=Q_0^{(0)}$ by $Q(\mathbf{0})=0$. Assume $m\geq1$ and fix $\beta=(\mathbf{u}_1,\mathbf{v}_1,\ldots,\mathbf{u}_m,\mathbf{v}_m)\in\mathscr{B}_Q$.\\
The form $Q$ is the orthogonal sum of its restrictions to the basis planes by their polar orthogonality.\\
On one pair $(\mathbf{u},\mathbf{v})$, put $r=Q(\mathbf{u})$ and $s=Q(\mathbf{v})$; then $(\forall\,x,y\in\mathbb{F}_2)\,Q(x\mathbf{u}+y\mathbf{v})=rx^2+xy+sy^2$.\\
If $(r,s)=(0,0)$, this is $H(x,y)=xy$. If $(r,s)=(0,1)$, replace $\mathbf{v}$ by $\mathbf{u}+\mathbf{v}$; if $(r,s)=(1,0)$, replace $\mathbf{u}$ by $\mathbf{u}+\mathbf{v}$.\\
Each replacement is a hyperbolic pair of $Q$-value zero, so these cases are also $H$.\\
The remaining case $(r,s)=(1,1)$ is $A(x,y)=x^2+xy+y^2$. Thus the parity of the $A$-summands is $\operatorname{Arf}_{\beta}(Q)$.\\
To cancel two $A$-summands with symplectic bases $(\mathbf{e}_1,\mathbf{f}_1)$ and $(\mathbf{e}_2,\mathbf{f}_2)$, put $\mathbf{p}_1=\mathbf{e}_1+\mathbf{e}_2$, $\mathbf{q}_1=\mathbf{f}_1+\mathbf{e}_2$, $\mathbf{p}_2=\mathbf{e}_1+\mathbf{f}_1+\mathbf{f}_2$,\\
and $\mathbf{q}_2=\mathbf{e}_1+\mathbf{f}_1+\mathbf{e}_2+\mathbf{f}_2$. Direct evaluation has $Q(\mathbf{p}_j)=Q(\mathbf{q}_j)=0$, $B_Q(\mathbf{p}_j,\mathbf{q}_j)=1$, and every cross-pairing zero.\\
Hence $(\mathbf{p}_1,\mathbf{q}_1,\mathbf{p}_2,\mathbf{q}_2)$ is a symplectic basis for two $H$-summands, so cancelling pairs of $A$-summands proves the claim.
\item For $P:\mathcal{U}\to\mathbb{F}_2$, put $D(P)\defeq\sum\limits_{\mathbf{x}\in\mathcal{U}}(-1)^{P(\mathbf{x})}$. Then $D(P\odot P')=D(P)D(P')$ by factoring the sum over pairs, while direct evaluation has $D(H)=1+1+1-1=2$ and $D(A)=1-1-1-1=-2$. Hence $D(Q_a^{(m)})=(-1)^a2^m$.\\
Since $\lvert(Q_a^{(m)})^{-1}(0)\rvert+\lvert(Q_a^{(m)})^{-1}(1)\rvert=4^m$, solving the sum and difference identities proves the formula.\\
Its two values are distinct, and zero-set cardinality is invariant under quadratic isometry.
\item Let $\beta,\gamma\in\mathscr{B}_Q$. By (ii), $Q\cong Q_{\operatorname{Arf}_{\beta}(Q)}^{(m)}$ and $Q\cong Q_{\operatorname{Arf}_{\gamma}(Q)}^{(m)}$. If $m=0$, both Arf sums are zero; if $m\geq1$, their values agree by the nonisometry in (iii). Thus $\operatorname{Arf}(Q)$ is well-defined.
\item The classification is (ii) with the basis-independent Arf sum. If $m\geq1$, the first pair in the standard symplectic basis of $Q_a^{(m)}$ contributes $a^2=a$ and every other pair contributes zero; hence $\operatorname{Arf}(Q_a^{(m)})=a$.
\end{enumerate}
\hlt{Structural Commentary}: Reduce the planes, cancel anisotropic pairs, and distinguish parity by zero counts.\\
\hlt{Key Insight}: The Arf invariant is the one-bit quadratic datum invisible to the common polar form.
\end{proof}
\end{theorem}
