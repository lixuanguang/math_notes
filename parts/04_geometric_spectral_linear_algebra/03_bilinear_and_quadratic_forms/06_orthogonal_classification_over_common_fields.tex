\subsection{Orthogonal Classification over Common Fields}

Throughout this subsection, orthogonal geometries are finite-dimensional over a field $\mathbb{F}$; additional hypotheses are stated locally.

\begin{theorem} \label{thm:orthogonal-bases-and-diagonal-congruence} \hlt{Orthogonal Bases and Diagonal Congruence}\\
Let $(\mathcal{V},B)$ be an orthogonal geometry, $q(\mathbf{x})\defeq B(\mathbf{x},\mathbf{x})$, $n=\dim\mathcal{V}$, and $r=\operatorname{rank}B$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item An orthogonal basis $\beta$ with $M_{\beta}=\operatorname{diag}(a_1,\ldots,a_r,0_{n-r})$ and $a_1,\ldots,a_r\in\mathbb{F}^{\times}$ exists if and only if $\operatorname{char}(\mathbb{F})\neq2$, $B=0$, or $q\not\equiv0$.
\item A symmetric $A\in\operatorname{Mat}(n,n,\mathbb{F})$ is congruent to a diagonal matrix if and only if $\operatorname{char}(\mathbb{F})\neq2$, $A=0$, or $(\exists\,\mathbf{x}\in\mathbb{F}^n)\,\mathbf{x}^tA\mathbf{x}\neq0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Suppose that such a basis exists. If $\operatorname{char}(\mathbb{F})=2$ and $q\equiv0$, every diagonal entry $q(\mathbf{v})$ of $M_{\beta}$ vanishes, so $M_{\beta}=0$ and $B=0$.
\item[$(i)$] ($\Leftarrow$) If $B=0$, any basis works. Assume $B\neq0$. Then $q\not\equiv0$ by hypothesis in characteristic $2$, Proposition \ref{prop:form-type-relations-by-characteristic}(ii) otherwise.\\
Successive splittings by Theorem \ref{thm:metric-orthogonal-complement-dimension}(ii) terminate with $\mathcal{V}=\mathbb{F}\mathbf{u}_1\odot\cdots\odot\mathbb{F}\mathbf{u}_s\odot\mathcal{W}$, where $q(\mathbf{u}_i)=a_i\in\mathbb{F}^{\times}$ and $q|_{\mathcal{W}}\equiv0$.\\
If $\operatorname{char}(\mathbb{F})\neq2$, then $B|_{\mathcal{W}}=0$ by Proposition \ref{prop:form-type-relations-by-characteristic}(ii), so append any basis of $\mathcal{W}$. Assume $\operatorname{char}(\mathbb{F})=2$.\\
By Theorem \ref{thm:symplectic-geometries-hyperbolic-classification}(ii), $\mathcal{W}=R\odot\mathcal{H}_1\odot\cdots\odot\mathcal{H}_t$ with zero form on $R$ and hyperbolic planes $\mathcal{H}_j$. Fix a current $\mathbf{u}$ with $q(\mathbf{u})=a\neq0$.\\
In each $\mathcal{H}_j$, rescale a hyperbolic basis $(\mathbf{e},\mathbf{f})$ so $B(\mathbf{e},\mathbf{f})=a$. The vectors $\mathbf{u}+\mathbf{e}$, $\mathbf{u}+\mathbf{f}$, and $\mathbf{u}+\mathbf{e}+\mathbf{f}$ are independent by coefficient comparison, have norm $a$, and are pairwise orthogonal because every cross-pairing is $a+a=0$.\\
Replacing $\mathbb{F}\mathbf{u}\odot\mathcal{H}_j$ by these three lines successively and appending a basis of $R$ completes the construction.
\item[$(ii)$] ($\Rightarrow$) Let $D=P^tAP$ be diagonal with $P$ invertible.\\
If $\operatorname{char}(\mathbb{F})=2$ and $(\forall\,\mathbf{x}\in\mathbb{F}^n)\,\mathbf{x}^tA\mathbf{x}=0$, then every diagonal entry of $D$ vanishes. Thus $D=0$ and $A=0$.
\item[$(ii)$] ($\Leftarrow$) Let $B_A(\mathbf{x},\mathbf{y})=\mathbf{x}^tA\mathbf{y}$ on $\mathbb{F}^n$. The hypothesis for $A$ is the hypothesis in (i) for $B_A$, so $B_A$ has an orthogonal basis by (i).\\
Its change-of-basis matrix diagonalizes $A$ by Proposition \ref{prop:metric-form-matrix-recovery-and-congruence}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Split nonisotropic lines, absorb any alternating residue, and translate bases into matrix congruence.\\
\hlt{Key Insight}: Only a nonzero alternating characteristic-$2$ form resists diagonalization.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:hyperbolic-plane-orthogonal-criteria} \hlt{Recognizing Orthogonal Hyperbolic Planes}\\
Let $\mathcal{H}$ be a $2$-dimensional nonsingular orthogonal geometry with $\operatorname{char}(\mathbb{F})\neq2$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{H}$ is a hyperbolic plane;
\item $(\exists\,\mathbf{u}\in\mathcal{H}\setminus\{\mathbf{0}\})\,\innerprod{\mathbf{u},\mathbf{u}}=0$;
\item $\Delta(\mathcal{H})=-(\mathbb{F}^{\times})^2$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ A hyperbolic pair consists of nonzero isotropic vectors.
\item[$(ii)$]$\Rightarrow (iii)$ By Theorem \ref{thm:isotropic-vector-sits-in-hyperbolic-plane}, fix $\mathbf{v}$ such that $(\mathbf{u},\mathbf{v})$ is hyperbolic. $(\mathbf{u},\mathbf{v})$ is a basis as $\dim\mathcal{H}=2$, and its matrix is $\left(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\right)$. QED.
\item[$(iii)$]$\Rightarrow (i)$ By Theorem \ref{thm:orthogonal-bases-and-diagonal-congruence}(i), write $M=\operatorname{diag}(a,b)$ with $a,b\in\mathbb{F}^{\times}$. Fix $c\in\mathbb{F}^{\times}$ satisfying $ab=-c^2$ by the discriminant hypothesis.\\
Then $a\left(\frac{c}{a}\right)^2+b=0$, so $\left(\frac{c}{a},1\right)$ is isotropic. It lies in a hyperbolic plane by Theorem \ref{thm:isotropic-vector-sits-in-hyperbolic-plane}; that plane equals $\mathcal{H}$ by dimension.
\end{enumerate}
\hlt{Structural Commentary}: Cycle from a hyperbolic basis to its discriminant, then reconstruct an isotropic vector.\\
\hlt{Key Insight}: A nonsingular orthogonal plane is hyperbolic exactly when it represents zero.
\end{proof}
\end{proposition}

\begin{example} \label{ex:characteristic-two-orthogonal-symplectic-plane} \hlt{Characteristic Two Orthogonal-Symplectic Plane}\\
Assume $\operatorname{char}(\mathbb{F})=2$ and let $(\mathcal{H},B)$ have basis $(\mathbf{e},\mathbf{f})$ with Gram matrix $\left(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\right)$.\\
Its matrix is symmetric with determinant $1$, while $(\forall\,a,b\in\mathbb{F})\,B(a\mathbf{e}+b\mathbf{f},a\mathbf{e}+b\mathbf{f})=2ab=0$ makes the form alternating.\\
For $\mathbf{0}\neq\mathbf{x}\in\mathcal{H}$, $\mathbb{F}\mathbf{x}\leq(\mathbb{F}\mathbf{x})^{\perp}$ by alternation, and both spaces are one-dimensional by Theorem \ref{thm:metric-orthogonal-complement-dimension}(i).\\
Hence they are equal, so no orthogonal basis exists, exactly the obstruction in Theorem \ref{thm:orthogonal-bases-and-diagonal-congruence}.
\end{example}

\begin{computationmethod} \label{meth:lagrange-orthogonalization-arbitrary-field} \hlt{Lagrange Orthogonalization}\\
Let $(\mathcal{V},B)$ be an orthogonal geometry with $\operatorname{char}(\mathbb{F})\neq2$ and $q(\mathbf{x})\defeq B(\mathbf{x},\mathbf{x})$. Input: a spanning list $L$. Output: an orthogonal basis.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Initialize $\mathcal{W}_0=\mathcal{V}$ and $L_0=L$. At stage $k$, stop with $R=\mathcal{W}_k$ if $q|_{\mathcal{W}_k}\equiv0$; otherwise fix $\mathbf{u}_{k+1}\in\mathcal{W}_k$ with $q(\mathbf{u}_{k+1})\neq0$.\\
If $(\forall\,\mathbf{x}\in L_k)\,q(\mathbf{x})=0$, fix $i,j$ with $B(\mathbf{x}_i,\mathbf{x}_j)\neq0$ and take $\mathbf{u}_{k+1}=\mathbf{x}_i+\mathbf{x}_j$; then $q(\mathbf{u}_{k+1})=2B(\mathbf{x}_i,\mathbf{x}_j)\neq0$.
\item Put $\mathcal{W}_{k+1}=\mathcal{W}_k\cap(\mathbb{F}\mathbf{u}_{k+1})^{\perp}$. Replace each $\mathbf{x}\in L_k$ by $\mathbf{x}-\frac{B(\mathbf{x},\mathbf{u}_{k+1})}{q(\mathbf{u}_{k+1})}\mathbf{u}_{k+1}$ and discard zero vectors.\\
The resulting list $L_{k+1}$ spans $\mathcal{W}_{k+1}$ by the orthogonal splitting $\mathcal{W}_k=\mathbb{F}\mathbf{u}_{k+1}\odot\mathcal{W}_{k+1}$.
\item The process stops because $\dim\mathcal{W}_{k+1}=\dim\mathcal{W}_k-1$. If $(\mathbf{z}_1,\ldots,\mathbf{z}_m)$ is any basis of $R$, output $\beta=(\mathbf{u}_1,\ldots,\mathbf{u}_s,\mathbf{z}_1,\ldots,\mathbf{z}_m)$.\\
Then $q|_R\equiv0$ and $B|_R=0$ by Proposition \ref{prop:form-type-relations-by-characteristic}(ii), so $\beta$ is orthogonal.
\end{enumerate}
\end{computationmethod}

\begin{definition} \label{def:universal-orthogonal-geometry} \hlt{Universal Orthogonal Geometry}\\
An orthogonal geometry $\mathcal{V}$ over $\mathbb{F}$ is \hlt{universal} if its norm map $q(\mathbf{x})\defeq\innerprod{\mathbf{x},\mathbf{x}}$ satisfies $q(\mathcal{V})=\mathbb{F}$.
\end{definition}

\begin{proposition} \label{prop:finite-field-square-classes-and-universality} \hlt{Finite-Field Square Classes and Universality}\\
Let $\mathbb{F}_q$ be a finite field and let $S=(\mathbb{F}_q^{\times})^2$. For an orthogonal geometry $\mathcal{V}$, write $q_{\mathcal{V}}(\mathbf{x})\defeq\innerprod{\mathbf{x},\mathbf{x}}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\operatorname{char}(\mathbb{F}_q)=2$, then $(\forall\,c\in\mathbb{F}_q)(\exists\,x\in\mathbb{F}_q)\,c=x^2$.
\item If $\operatorname{char}(\mathbb{F}_q)\neq2$, then $|S|=\frac{q-1}{2}$ and $\mathbb{F}_q^{\times}\setminus S=dS$ for any nonsquare $d$.
\item If $\mathcal{V}$ is orthogonal over odd $\mathbb{F}_q$ and contains a nonsingular subspace of dimension at least $2$, then $q_{\mathcal{V}}(\mathcal{V})=\mathbb{F}_q$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item In characteristic $2$, $x^2=y^2\Rightarrow(x-y)^2=0\Rightarrow x=y$. Thus $x\mapsto x^2$ is injective, hence bijective on the finite set $\mathbb{F}_q$.
\item Suppose $\operatorname{char}(\mathbb{F}_q)\neq2$. Then $(\forall\,x,y\in\mathbb{F}_q^{\times})\,x^2=y^2\Longleftrightarrow x=\pm y$, so the $\frac{q-1}{2}$ pairs $\{x,-x\}$ determine the elements of $S$.\\
If $d$ is nonsquare, then $dS\cap S=\varnothing$ and $|dS|=|S|=\frac{q-1}{2}$, so $\mathbb{F}_q^{\times}\setminus S=dS$.
\item Set $q=q_{\mathcal{V}}$. By Theorem \ref{thm:orthogonal-bases-and-diagonal-congruence}(i), the nonsingular subspace contains orthogonal $\mathbf{u},\mathbf{v}$ with $a\defeq q(\mathbf{u})\neq0$ and $b\defeq q(\mathbf{v})\neq0$.\\
Fix $c\in\mathbb{F}_q$. The sets $a\{\alpha^2:\alpha\in\mathbb{F}_q\}$ and $c-b\{\beta^2:\beta\in\mathbb{F}_q\}$ each have $\frac{q+1}{2}$ elements and therefore intersect.\\
Fix $\alpha,\beta\in\mathbb{F}_q$ satisfying $a\alpha^2=c-b\beta^2$; then $c=q(\alpha\mathbf{u}+\beta\mathbf{v})$.
\end{enumerate}
\hlt{Structural Commentary}: Count square fibers, identify the nonsquare coset, and force two quadratic-value sets to intersect.\\
\hlt{Key Insight}: Two subsets of size $\frac{q+1}{2}$ cannot be disjoint.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:orthogonal-canonical-forms-common-fields} \hlt{Canonical Orthogonal Forms over Common Fields}\\
Let $M=M^t\in\operatorname{Mat}(n,n,\mathbb{F})$, let $r=\operatorname{rank}M$, and write $M\sim N$ for congruence. If $\operatorname{char}(\mathbb{F})=2$, assume $M$ is not nonzero alternating.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathbb{F}$ is algebraically closed, then $M\sim I_r\oplus0_{n-r}$, and rank is a complete congruence invariant.
\item If $\mathbb{F}=\mathbb{R}$, then $M\sim I_p\oplus(-I_m)\oplus0_k$ for unique $p,m,k\in\mathbb{Z}_{\geq0}$ satisfying $p+m+k=n$.
\item If $\mathbb{F}=\mathbb{F}_q$ has characteristic $2$, then $M\sim I_r\oplus0_{n-r}$, and rank is a complete congruence invariant among the admitted matrices.
\item If $\mathbb{F}=\mathbb{F}_q$ has odd characteristic and $d\in\mathbb{F}_q^{\times}$ is a fixed nonsquare, then $r=0$ implies $M\sim0_n$; for $r>0$, exactly one of $M\sim I_r\oplus0_{n-r}$ and $M\sim I_{r-1}\oplus[d]\oplus0_{n-r}$ holds.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:orthogonal-bases-and-diagonal-congruence}(ii), write $M\sim\operatorname{diag}(a_1,\ldots,a_r,0_{n-r})$. Fix $s_i\in\mathbb{F}^{\times}$ satisfying $a_i=s_i^2$ by algebraic closure.\\
Rescaling produces $I_r\oplus0_{n-r}$. Rank is congruence-invariant by Proposition \ref{prop:metric-form-matrix-recovery-and-congruence}(iv), hence complete.
\item By Theorem \ref{thm:orthogonal-bases-and-diagonal-congruence}(ii), rescaling produces $D=I_p\oplus(-I_m)\oplus0_k$. The positive coordinate subspace has dimension $p$, while every positive-definite subspace intersects the nonpositive coordinate subspace trivially and therefore has dimension at most $p$.\\
Thus $p$ is its maximal possible dimension and is congruence-invariant. The same argument for $-D$ fixes $m$, and $k=n-p-m$.
\item By Theorem \ref{thm:orthogonal-bases-and-diagonal-congruence}(ii), $M\sim\operatorname{diag}(a_1,\ldots,a_r,0_{n-r})$. Each $a_i$ is a square by Proposition \ref{prop:finite-field-square-classes-and-universality}(i), so rescaling produces $I_r\oplus0_{n-r}$.\\
Rank invariance proves completeness.
\item By Theorem \ref{thm:orthogonal-bases-and-diagonal-congruence}(ii), diagonalize $M$ and let $\mathcal{U}$ be its nonsingular block, with norm map $q$. While $\dim\mathcal{U}\geq2$, fix $\mathbf{u}\in\mathcal{U}$ with $q(\mathbf{u})=1$ by Proposition \ref{prop:finite-field-square-classes-and-universality}(iii), then split its line with nonsingular complement by Theorem \ref{thm:metric-orthogonal-complement-dimension}(ii),(iv).\\
Thus regular block becomes $I_{r-1}\oplus[a]$. By Proposition \ref{prop:finite-field-square-classes-and-universality}(ii), rescale so $a\in\{1,d\}$. Congruence induces an isometry of regular quotients, whose two discriminants are distinct by Proposition \ref{prop:metric-form-matrix-recovery-and-congruence}(v). If $r=0$, then $M=0_n$.
\end{enumerate}
\hlt{Structural Commentary}: Diagonalize first, normalize coefficients field by field, and separate the remaining forms by invariant data.\\
\hlt{Key Insight}: Square classes and signs control the coefficients that scaling cannot remove.
\end{proof}
\end{theorem}

\begin{definition} \label{def:minkowski-metric-vector-space} \hlt{Minkowski Space}\\
\hlt{Minkowski space} $\mathbb{R}^{3,1}$ is $\mathbb{R}^4$ with standard Gram matrix $\operatorname{diag}(1,1,1,-1)$.
\end{definition}

\begin{example} \label{ex:minkowski-null-cone-and-inertia} \hlt{Null Vectors in Minkowski Space}\\
In $\mathbb{R}^{3,1}$, the null cone is $\mathcal{N}=\{\mathbf{x}\in\mathbb{R}^4\setminus\{\mathbf{0}\}:x_1^2+x_2^2+x_3^2=x_4^2\}$, and $\mathbf{e}_1+\mathbf{e}_4\in\mathcal{N}$. The space is not isometric to Euclidean $\mathbb{R}^4$: an isometry matrix $A$ would satisfy $A^tI_4A=\operatorname{diag}(1,1,1,-1)$, whose determinant identity $(\det A)^2=-1$ is impossible over $\mathbb{R}$.
\end{example}

\begin{example} \label{ex:finite-field-orthogonal-rank-two-classes} \hlt{Two Rank-Two Orthogonal Planes over $\mathbb{F}_5$}\\
In $\mathbb{F}_5$, $(\mathbb{F}_5^{\times})^2=\{1,4\}$ and the nonsquare class is $\{2,3\}$. Let $q_1(x,y)=x^2+y^2$ and $q_2(x,y)=x^2+2y^2$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The form $q_1$ has matrix $I_2$ and is hyperbolic by $q_1(1,2)=0$ and Proposition \ref{prop:hyperbolic-plane-orthogonal-criteria}.
\item The form $q_2$ is anisotropic: $q_2(x,y)=0$ and $y=0$ imply $x=0$, while $y\neq0$ would make the nonsquare $3=\left(\frac{x}{y}\right)^2$.
\end{enumerate}
Thus the two forms realize the nonsingular rank-$2$ classes in Theorem \ref{thm:orthogonal-canonical-forms-common-fields}(iv).
\end{example}

\begin{lemma} \label{lem:odd-finite-field-anisotropic-residue-bound} \hlt{Odd Finite-Field Isotropy Bound}\\
Let $\mathcal{V}$ be nonsingular orthogonal over odd $\mathbb{F}_q$, put $q(\mathbf{x})\defeq\innerprod{\mathbf{x},\mathbf{x}}$, and write $\mathcal{H}^{\odot s}$ for $s$ hyperbolic planes.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\dim\mathcal{V}\geq3$, then $(\exists\,\mathbf{x}\in\mathcal{V}\setminus\{\mathbf{0}\})\,q(\mathbf{x})=0$.
\item $\mathcal{V}\cong\mathcal{H}^{\odot s}\odot\mathcal{R}$, where $\mathcal{R}$ is nonsingular anisotropic and $\dim\mathcal{R}\leq2$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix a nonsingular plane $\mathcal{P}\leq\mathcal{V}$ and $\mathbf{0}\neq\mathbf{z}\in\mathcal{P}^{\perp}$ by Theorems \ref{thm:orthogonal-bases-and-diagonal-congruence}(i) and \ref{thm:metric-orthogonal-complement-dimension}(i). If $q(\mathbf{z})=0$, take $\mathbf{x}=\mathbf{z}$.\\
Otherwise fix $\mathbf{p}\in\mathcal{P}$ with $q(\mathbf{p})=-q(\mathbf{z})$ by Proposition \ref{prop:finite-field-square-classes-and-universality}(iii); then $\mathbf{0}\neq\mathbf{p}+\mathbf{z}$ is isotropic by $\mathcal{P}\cap\mathcal{P}^{\perp}=\{\mathbf{0}\}$ and $\mathcal{P}\perp\mathcal{P}^{\perp}$.
\item Induct on $\dim\mathcal{V}$. If $\mathcal{V}$ is anisotropic, take $(s,\mathcal{R})=(0,\mathcal{V})$; then $\dim\mathcal{R}\leq2$ by (i).\\
Otherwise an isotropic vector lies in a hyperbolic plane $\mathcal{H}$ by Theorem \ref{thm:isotropic-vector-sits-in-hyperbolic-plane}. By Theorem \ref{thm:metric-orthogonal-complement-dimension}(ii),(iv), $\mathcal{V}=\mathcal{H}\odot\mathcal{H}^{\perp}$ with $\mathcal{H}^{\perp}$ nonsingular and two dimensions smaller, so induction completes the decomposition.
\end{enumerate}
\hlt{Structural Commentary}: Force an isotropic vector by universality, then split hyperbolic planes inductively.\\
\hlt{Key Insight}: An anisotropic space over an odd finite field has dimension at most two.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:odd-finite-field-witt-decomposition} \hlt{Odd Finite-Field Orthogonal Decomposition}\\
Let $\mathbb{F}_q$ have odd characteristic, fix a nonsquare $d\in\mathbb{F}_q^{\times}$, and let $\mathcal{V}\neq\{\mathbf{0}\}$ be nonsingular orthogonal.\\
Write $\mathcal{H}^{\odot s}$ for an orthogonal sum of $s$ hyperbolic planes, $\langle a\rangle$ for a line of value $a$, and $\mathcal{A}_d\defeq\langle1\rangle\odot\langle-d\rangle$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\dim\mathcal{V}=2m+1$, then $(\exists\,a\in\mathbb{F}_q^{\times})\,\mathcal{V}\cong\mathcal{H}^{\odot m}\odot\langle a\rangle\ \wedge\ a(\mathbb{F}_q^{\times})^2=(-1)^m\Delta(\mathcal{V})$.
\item If $\dim\mathcal{V}=2m$, then $m\geq1$ and exactly one of $\mathcal{V}\cong\mathcal{H}^{\odot m}$ and $\mathcal{V}\cong\mathcal{H}^{\odot(m-1)}\odot\mathcal{A}_d$ holds.\\
The first occurs if and only if $\Delta(\mathcal{V})=(-1)^m(\mathbb{F}_q^{\times})^2$; the second occurs if and only if $\Delta(\mathcal{V})=(-1)^m d(\mathbb{F}_q^{\times})^2$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] By Lemma \ref{lem:odd-finite-field-anisotropic-residue-bound}(ii), $\mathcal{V}\cong\mathcal{H}^{\odot m}\odot\mathcal{R}$ with $\dim\mathcal{R}=1$. Fix $a\in\mathbb{F}_q^{\times}$ satisfying $\mathcal{R}\cong\langle a\rangle$.\\
Since $\Delta(\mathcal{H})=-(\mathbb{F}_q^{\times})^2$, $\Delta(\mathcal{V})=(-1)^ma(\mathbb{F}_q^{\times})^2$, and multiplication by $(-1)^m$ proves the stated square class for $a$.
\item[$(ii)$] ($\Rightarrow$) Since $\mathcal{V}\neq\{\mathbf{0}\}$, $m\geq1$. If $\mathcal{V}\cong\mathcal{H}^{\odot m}$, then $\Delta(\mathcal{V})=(-1)^m(\mathbb{F}_q^{\times})^2$. If $\mathcal{V}\cong\mathcal{H}^{\odot(m-1)}\odot\mathcal{A}_d$, then $\Delta(\mathcal{A}_d)=-d(\mathbb{F}_q^{\times})^2$ and $\Delta(\mathcal{V})=(-1)^m d(\mathbb{F}_q^{\times})^2$. These square classes are distinct, so at most one alternative holds.
\item[$(ii)$] ($\Leftarrow$) By Lemma \ref{lem:odd-finite-field-anisotropic-residue-bound}(ii), $\mathcal{V}\cong\mathcal{H}^{\odot s}\odot\mathcal{R}$ with $\dim\mathcal{R}\in\{0,2\}$. If $\dim\mathcal{R}=0$, then $s=m$ and $\mathcal{V}\cong\mathcal{H}^{\odot m}$.\\
Otherwise $s=m-1$, while $\mathcal{R}$ and $\mathcal{A}_d$ are anisotropic planes because $d$ is nonsquare.\\
They lie in the unique nonhyperbolic rank-$2$ class by Theorem \ref{thm:orthogonal-canonical-forms-common-fields}(iv) and Proposition \ref{prop:hyperbolic-plane-orthogonal-criteria}, so $\mathcal{R}\cong\mathcal{A}_d$.\\
The forward discriminant calculations select the unique alternative.
\end{enumerate}
\hlt{Structural Commentary}: Split to the anisotropic residue, use parity for its dimension, and use discriminant for its class.\\
\hlt{Key Insight}: Over an odd finite field, parity and discriminant determine the Witt residue.
\end{proof}
\end{theorem}
