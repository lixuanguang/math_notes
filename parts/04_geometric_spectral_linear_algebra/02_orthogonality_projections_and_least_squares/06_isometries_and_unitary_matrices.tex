\subsection{Isometries and Unitary Matrices}

Throughout this subsection, $\mathbb{F}$ is $\mathbb{R}$ or $\mathbb{C}$.

\begin{definition} \label{def:isometry-linear-map} \hlt{Isometry and Isometric Isomorphism}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be inner-product spaces and let $T\in\LL(\mathcal{V},\mathcal{W})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is an \hlt{isometry} if $(\forall\,\mathbf{v}\in\mathcal{V})\,\norm{T\mathbf{v}}=\norm{\mathbf{v}}$.
\item $T$ is an \hlt{isometric isomorphism} if it is bijective and isometric.\\
The spaces $\mathcal{V}$ and $\mathcal{W}$ are \hlt{isometrically isomorphic} if an isometric isomorphism from $\mathcal{V}$ to $\mathcal{W}$ exists.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:basic-isometry-properties} \hlt{Basic Properties of Isometries}\\
Let $\mathcal{V}$ be an inner-product space and let $S,T\in\LL(\mathcal{V})$ be isometries.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $ST$ is an isometry;
\item $T$ is injective;
\item if $\mathcal{V}$ is finite-dimensional, then $T$ is invertible;
\item if $\mathcal{V}$ is finite-dimensional, then $T^{-1}$ is an isometry.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,\mathbf{v}\in\mathcal{V})\,\norm{ST\mathbf{v}}=\norm{T\mathbf{v}}=\norm{\mathbf{v}}$.
\item Fix $\mathbf{v}\in\ker(T)$. Then $\norm{\mathbf{v}}=\norm{T\mathbf{v}}=0$, so $\mathbf{v}=\mathbf{0}$ by positive definiteness. Hence $\ker(T)=\{\mathbf{0}\}$ and $T$ is injective.
\item If $\mathcal{V}$ is finite-dimensional, then $T$ is invertible by (ii) and Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences}.
\item Suppose $\mathcal{V}$ is finite-dimensional. By (iii), $T^{-1}$ exists, and $(\forall\,\mathbf{u}\in\mathcal{V})\,\norm{T^{-1}\mathbf{u}}=\norm{T(T^{-1}\mathbf{u})}=\norm{\mathbf{u}}$.
\end{enumerate}
\hlt{Structural Commentary}: Norm identities prove closure and injectivity; finite-dimensional invertibility controls the inverse.\\
\hlt{Key Insight}: Length preservation prevents a nonzero vector from entering the kernel.
\end{proof}
\end{proposition}

\begin{example} \label{ex:l2-shift-isometry-not-surjective} \hlt{An Isometry Need Not Be Onto}\\
Define $T:\ell^2(\mathbb{F})\to\ell^2(\mathbb{F})$ by $T(x_1,x_2,\ldots)=(0,x_1,x_2,\ldots)$. For $\mathbf{x}=(x_j)_{j\geq1}$, $\norm{T\mathbf{x}}_2^2=\sum\limits_{j\geq1}\abs{x_j}^2=\norm{\mathbf{x}}_2^2$, so $T$ is an isometry.\\
Since every vector in $\operatorname{range}(T)$ has first coordinate $0$, $(1,0,0,\ldots)\notin\operatorname{range}(T)$; hence $T$ is not surjective.\\
Thus norm preservation need not imply surjectivity in infinite-dimensional inner-product spaces.
\end{example}

\begin{theorem} \label{thm:isometry-iff-inner-product-preserving} \hlt{Isometries Preserve Inner Products}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be inner-product spaces and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then $[(\forall\,\mathbf{x}\in\mathcal{V})\,\norm{T\mathbf{x}}=\norm{\mathbf{x}}]\Longleftrightarrow[(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})\,\innerprod[\mathcal{W}]{T\mathbf{u},T\mathbf{v}}=\innerprod[\mathcal{V}]{\mathbf{u},\mathbf{v}}]$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $T$ preserves norms and fix $\mathbf{u},\mathbf{v}\in\mathcal{V}$. Each norm in the polarization formula for $\innerprod[\mathcal{W}]{T\mathbf{u},T\mathbf{v}}$ equals its counterpart for $\innerprod[\mathcal{V}]{\mathbf{u},\mathbf{v}}$ by linearity of $T$ and the hypothesis. Hence $\innerprod[\mathcal{W}]{T\mathbf{u},T\mathbf{v}}=\innerprod[\mathcal{V}]{\mathbf{u},\mathbf{v}}$ by Theorem \ref{thm:polarization-identities}.
\item[$(\Leftarrow)$] Suppose $T$ preserves inner products. Then $(\forall\,\mathbf{u}\in\mathcal{V})\,\norm{T\mathbf{u}}^2=\innerprod[\mathcal{W}]{T\mathbf{u},T\mathbf{u}}=\innerprod[\mathcal{V}]{\mathbf{u},\mathbf{u}}=\norm{\mathbf{u}}^2$; $T$ preserves norms by nonnegativity.
\end{enumerate}
\hlt{Structural Commentary}: Polarization converts preservation of norms into preservation of inner products.\\
\hlt{Key Insight}: Universal length preservation also preserves angle information.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:isometry-conventions} \hlt{Isometry Conventions}\\
For linear maps between inner-product spaces, norm preservation and inner-product preservation are equivalent by Theorem \ref{thm:isometry-iff-inner-product-preserving}, so either convention defines the same isometries. For general bilinear forms, however, form preservation need not imply injectivity; an isometry is therefore often required to be bijective and form-preserving.
\end{remark}

\begin{theorem} \label{thm:finite-dimensional-isometry-iff-adjoint-product-identity} \hlt{Isometries and the Adjoint Product}\\
Let $\mathcal{V}$ be a finite-dimensional inner-product space and let $T\in\LL(\mathcal{V})$.\\
Then $[(\forall\,\mathbf{v}\in\mathcal{V})\,\norm{T\mathbf{v}}=\norm{\mathbf{v}}]\Longleftrightarrow T^*T=I_{\mathcal{V}}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $T$ preserves norms. Then $(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})\,\innerprod{\mathbf{u},T^*T\mathbf{v}}=\innerprod{T\mathbf{u},T\mathbf{v}}=\innerprod{\mathbf{u},\mathbf{v}}$ by Theorem \ref{thm:isometry-iff-inner-product-preserving} and the adjoint relation.\\
For $\mathbf{v}\in\mathcal{V}$, set $\mathbf{u}\defeq(T^*T-I_{\mathcal{V}})\mathbf{v}$. Then $\norm{\mathbf{u}}^2=\innerprod{\mathbf{u},(T^*T-I_{\mathcal{V}})\mathbf{v}}=0$, so $\mathbf{u}=\mathbf{0}$ by positive definiteness. Hence $T^*T=I_{\mathcal{V}}$.
\item[$(\Leftarrow)$] Suppose $T^*T=I_{\mathcal{V}}$; $(\forall\,\mathbf{u}\in\mathcal{V})\,\norm{T\mathbf{u}}^2=\innerprod{T\mathbf{u},T\mathbf{u}}=\innerprod{\mathbf{u},T^*T\mathbf{u}}=\norm{\mathbf{u}}^2$ by adjoint relation; $T$ preserves norms by nonnegativity.
\end{enumerate}
\hlt{Structural Commentary}: The operator $T^*T$ records inner products after applying $T$.\\
\hlt{Key Insight}: $T$ is an isometry exactly when this Gram operator is the identity.
\end{proof}
\end{theorem}

\begin{definition} \label{def:unitary-operator-and-matrix} \hlt{Unitary and Orthogonal}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\mathcal{V}$ be a finite-dimensional inner-product space. An operator $U\in\LL(\mathcal{V})$ is \hlt{unitary} if $U^*U=UU^*=I_{\mathcal{V}}$.
\item Let $n\geq1$. A matrix $U\in\operatorname{Mat}(n,n,\mathbb{F})$ is \hlt{unitary} if $U^*U=I_n$. If $\mathbb{F}=\mathbb{R}$, a unitary matrix is \hlt{orthogonal}.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:unitary-matrix-equivalent-conditions} \hlt{Equivalent Conditions for a Unitary Matrix}\\
Let $n\geq1$ and let $U=[\mathbf{u}_1\ \cdots\ \mathbf{u}_n]\in\operatorname{Mat}(n,n,\mathbb{F})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $U^*U=I_n$;
\item the columns $\mathbf{u}_1,\ldots,\mathbf{u}_n$ are orthonormal in $\mathbb{F}^n$;
\item $UU^*=I_n$;
\item $U^*$ is unitary;
\item the rows of $U$ are orthonormal;
\item $U$ is invertible and $U^{-1}=U^*$;
\item $(\forall\,\mathbf{x}\in\mathbb{F}^n)\,\norm{U\mathbf{x}}_2=\norm{\mathbf{x}}_2$;
\item $(\forall\,\mathbf{x},\mathbf{y}\in\mathbb{F}^n)\,\innerprod[\mathbb{F}^n]{U\mathbf{x},U\mathbf{y}}=\innerprod[\mathbb{F}^n]{\mathbf{x},\mathbf{y}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Since $(U^*U)_{ij}=\mathbf{u}_i^*\mathbf{u}_j=\innerprod[\mathbb{F}^n]{\mathbf{u}_j,\mathbf{u}_i}$, the identity $U^*U=I_n$ is equivalent to orthonormality of the columns.
\item[$(ii)$] $\Rightarrow (iii)$ Orthonormality of the columns implies $U^*U=I_n$. Hence $UU^*=I_n$ by Theorem \ref{thm:square-one-sided-inverse-is-two-sided}.
\item[$(iii)$] $\Rightarrow (iv)$ The identity $UU^*=I_n$ is $(U^*)^*U^*=I_n$ by conjugate-transpose involution, so $U^*$ is unitary.
\item[$(iv)$] $\Rightarrow (v)$ Since $U^*$ is unitary, its columns are orthonormal by the calculation in (i).\\
These columns are the conjugates of the rows of $U$, so the rows of $U$ are orthonormal.
\item[$(v)$] $\Rightarrow (vi)$ Orthonormality of the rows implies $UU^*=I_n$. Hence $U^{-1}=U^*$ by Theorem \ref{thm:square-one-sided-inverse-is-two-sided}.
\item[$(vi)$] $\Rightarrow (vii)$ The relation $U^{-1}=U^*$ implies $U^*U=I_n$. Thus $(\forall\,\mathbf{x}\in\mathbb{F}^n)\,\norm{U\mathbf{x}}_2^2=\innerprod[\mathbb{F}^n]{U\mathbf{x},U\mathbf{x}}=\innerprod[\mathbb{F}^n]{\mathbf{x},U^*U\mathbf{x}}=\norm{\mathbf{x}}_2^2$ by the conjugate-transpose adjoint relation, so $U$ preserves norms by nonnegativity.
\item[$(vii)$] $\Rightarrow (viii)$ Condition (viii) holds by Theorem \ref{thm:isometry-iff-inner-product-preserving} for the map $\mathbf{x}\mapsto U\mathbf{x}$.
\item[$(viii)$] $\Rightarrow (i)$ The relation $(\forall\,\mathbf{x},\mathbf{y}\in\mathbb{F}^n)\,\innerprod[\mathbb{F}^n]{\mathbf{x},(U^*U-I_n)\mathbf{y}}=\innerprod[\mathbb{F}^n]{U\mathbf{x},U\mathbf{y}}-\innerprod[\mathbb{F}^n]{\mathbf{x},\mathbf{y}}=0$ holds by the conjugate-transpose adjoint relation and (viii). Taking $\mathbf{x}=(U^*U-I_n)\mathbf{y}$, $(\forall\,\mathbf{y}\in\mathbb{F}^n)\,\norm{(U^*U-I_n)\mathbf{y}}_2^2=0$, so $U^*U=I_n$ by positive definiteness.
\end{enumerate}
\hlt{Structural Commentary}: A cyclic chain links Gram identities, inverse identities, and geometric invariance.\\
\hlt{Key Insight}: Unitarity is orthonormality expressed in coordinates.
\end{proof}
\end{theorem}

\begin{definition} \label{def:unitary-and-permutation-similarity} \hlt{Unitary and Permutation Similarity}\\
Let $n\geq1$ and let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The matrices $A$ and $B$ are \hlt{unitarily similar} if $A=UBU^*$ for a unitary $U\in\operatorname{Mat}(n,n,\mathbb{F})$.
\item They are \hlt{orthogonally similar} if $\mathbb{F}=\mathbb{R}$ and $A=UBU^*$ for a real orthogonal $U\in\operatorname{Mat}(n,n,\mathbb{R})$.
\item Matrix $P=[p_{ij}]\in\operatorname{Mat}(n,n,\mathbb{R})$ is \hlt{permutation matrix} if $(\forall\,i,j\in\{1,\ldots,n\})\,p_{ij}\in\{0,1\}$, $(\forall\,i\in\{1,\ldots,n\})\,\sum\limits_{j=1}^{n}p_{ij}=\sum\limits_{j=1}^{n}p_{ji}=1$.
\item The matrices $A$ and $B$ are \hlt{permutation similar} if $A=PBP^*$ for a permutation matrix $P$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:permutation-matrices-are-orthogonal} \hlt{Permutation Matrices Are Orthogonal}\\
Let $n\geq1$, let $\sigma\in\mathfrak{S}_n$, and set $P\defeq[\mathbf{e}_{\sigma(1)}\ \cdots\ \mathbf{e}_{\sigma(n)}]\in\operatorname{Mat}(n,n,\mathbb{R})$. Then $P^*P=PP^*=I_n$, so $P$ is orthogonal.
\begin{proof}
The columns of $P$ are a rearrangement of the standard orthonormal basis of $\mathbb{R}^n$.\\
Hence $P^*P=PP^*=I_n$ by Theorem \ref{thm:unitary-matrix-equivalent-conditions}(ii)--(iii), and $P$ is orthogonal because it is real.\\
\hlt{Structural Commentary}: A permutation matrix only reorders the standard basis.\\
\hlt{Key Insight}: Reordering orthonormal vectors preserves orthonormality.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:orthogonal-projection-matrix-unitary-coordinate-form} \hlt{Unitary Coordinate Form of an Orthogonal Projection}\\
Let $n\geq1$, let $P\in\operatorname{Mat}(n,n,\mathbb{F})$ be an orthogonal projection matrix, and set $r\defeq\operatorname{rank}(P)$.\\
Then $(\exists\,U\in\operatorname{Mat}(n,n,\mathbb{F}))\,[U^*U=I_n\land U^*PU=I_r\oplus0_{n-r}]$. Equivalently, $P$ is unitarily similar to $I_r\oplus0_{n-r}$.
\begin{proof}
Set $\mathcal{U}\defeq\operatorname{col}(P)$. The operator induced by $P$ is $P_{\mathcal{U}}$ by Theorem \ref{thm:orthogonal-projection-self-adjoint-idempotent}.\\
Hence $(\forall\,\mathbf{u}\in\mathcal{U})\,P\mathbf{u}=\mathbf{u}$ and $(\forall\,\mathbf{w}\in\mathcal{U}^{\perp})\,P\mathbf{w}=\mathbf{0}$ by Proposition \ref{prop:orthogonal-projection-properties}(ii)--(iii).\\
Choose an orthonormal basis $(\mathbf{u}_1,\ldots,\mathbf{u}_r)$ of $\mathcal{U}$ by Corollary \ref{cor:finite-dimensional-inner-product-space-has-orthonormal-basis}, and extend it to an orthonormal basis $(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ of $\mathbb{F}^n$ by Corollary \ref{cor:orthonormal-list-extends-to-basis}. Set $U\defeq[\mathbf{u}_1\ \cdots\ \mathbf{u}_n]$. Then $U^*U=I_n$ by Theorem \ref{thm:unitary-matrix-equivalent-conditions}(i)--(ii).\\
For $j\in\{1,\ldots,r\}$, $U^*PU\mathbf{e}_j=U^*P\mathbf{u}_j=U^*\mathbf{u}_j=\mathbf{e}_j$; for $j\in\{r+1,\ldots,n\}$, $U^*PU\mathbf{e}_j=\mathbf{0}$.\\
Therefore $U^*PU=I_r\oplus0_{n-r}$ by equality on the standard basis.\\
Finally, $P=U(I_r\oplus0_{n-r})U^*$ by multiplication on the left by $U$ and on the right by $U^*$.\\
\hlt{Structural Commentary}: An adapted orthonormal basis converts the range--kernel split into block coordinates.\\
\hlt{Key Insight}: Projection becomes coordinate truncation in an adapted orthonormal basis.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:algebra-of-unitary-matrices} \hlt{Algebra of Unitary Matrices}\\
Let $m,n\geq1$, let $U,V\in\operatorname{Mat}(n,n,\mathbb{F})$, and let $W\in\operatorname{Mat}(m,m,\mathbb{F})$.\\
Let $V\oplus W$ denote the block diagonal matrix with blocks $V$ and $W$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $[U^*U=I_n\land V^*V=I_n]\Longrightarrow(UV)^*(UV)=I_n$;
\item $[(V\oplus W)^*(V\oplus W)=I_{n+m}]\Longleftrightarrow[V^*V=I_n\land W^*W=I_m]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] The identity $(UV)^*(UV)=V^*U^*UV=V^*V=I_n$ holds by the hypotheses.
\item[$(ii)$] ($\Rightarrow$) Suppose $(V\oplus W)^*(V\oplus W)=I_{n+m}$.\\
Then $V^*V\oplus W^*W=I_n\oplus I_m$, so $V^*V=I_n$ and $W^*W=I_m$ by equality of the diagonal blocks.
\item[$(ii)$] ($\Leftarrow$) If $V^*V=I_n$ and $W^*W=I_m$, then $(V\oplus W)^*(V\oplus W)=I_n\oplus I_m=I_{n+m}$.
\end{enumerate}
\hlt{Structural Commentary}: Unitary structure is stable under composition and blockwise orthogonal sums.\\
\hlt{Key Insight}: The equation $U^*U=I$ makes every verification a short adjoint calculation.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:extend-isometric-columns-to-unitary-matrix} \hlt{Extending Isometric Columns to a Unitary Matrix}\\
Let $m,n\geq1$ and let $X\in\operatorname{Mat}(m,n,\mathbb{F})$ satisfy $X^*X=I_n$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $n\leq m$;
\item $(\exists\,X'\in\operatorname{Mat}(m,m-n,\mathbb{F}))\,[X\ X']^*[X\ X']=I_m$.
\end{enumerate}
\begin{proof}
Write $X=[\mathbf{x}_1\ \cdots\ \mathbf{x}_n]$. The list $(\mathbf{x}_1,\ldots,\mathbf{x}_n)$ is orthonormal in $\mathbb{F}^m$ because $(X^*X)_{ij}=\innerprod[\mathbb{F}^m]{\mathbf{x}_j,\mathbf{x}_i}=\delta_{ij}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The list is linearly independent by Corollary \ref{cor:orthonormal-list-linearly-independent}, so $n\leq m$ by Theorem \ref{thm:finitedimlenofvec}.
\item Extend the list to an orthonormal basis $(\mathbf{x}_1,\ldots,\mathbf{x}_m)$ of $\mathbb{F}^m$ by Corollary \ref{cor:orthonormal-list-extends-to-basis}, and set $X'\defeq[\mathbf{x}_{n+1}\ \cdots\ \mathbf{x}_m]$.\\
Then $[X\ X']^*[X\ X']=I_m$ by Theorem \ref{thm:unitary-matrix-equivalent-conditions}(i)--(ii).
\end{enumerate}
\hlt{Structural Commentary}: Column orthonormality proves the dimension bound and supplies an extendable partial basis.\\
\hlt{Key Insight}: Complete an isometric column family by extending its orthonormal list.
\end{proof}
\end{theorem}
