\subsection{Self-Adjoint Operators and Orthogonal Projections}

Throughout this subsection, $\mathbb{F}$ is $\mathbb{R}$ or $\mathbb{C}$.

\begin{definition} \label{def:self-adjoint-hermitian} \hlt{Self-Adjoint and Hermitian}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\mathcal{V}$ be a finite-dimensional inner-product space over $\mathbb{F}$. An operator $T\in\LL(\mathcal{V})$ is \hlt{self-adjoint} if $T^*=T$.
\item Let $n\geq1$. A matrix $A\in\operatorname{Mat}(n,n,\mathbb{F})$ is \hlt{Hermitian} if $A^*=A$.
\end{enumerate}
In orthonormal coordinates, these conditions agree by Theorem \ref{thm:matrix-of-adjoint-is-conjugate-transpose}.
\end{definition}

\begin{lemma} \label{lem:complex-zero-quadratic-form-implies-zero-operator} \hlt{Vanishing Complex Diagonal Values}\\
Let $\mathcal{V}$ be a finite-dimensional complex inner-product space and let $T\in\LL(\mathcal{V})$.\\
If $(\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod{T\mathbf{v},\mathbf{v}}=0$, then $T=0$.
\begin{proof}
Set $Q:\mathcal{V}\to\mathbb{C}$ by $Q(\mathbf{x})\defeq\innerprod{T\mathbf{x},\mathbf{x}}$ for $\mathbf{x}\in\mathcal{V}$.\\
For $(\mathbf{u},\mathbf{w})\in\mathcal{V}\times\mathcal{V}$, the identity $4\innerprod{T\mathbf{u},\mathbf{w}}=Q(\mathbf{u}+\mathbf{w})-Q(\mathbf{u}-\mathbf{w})+iQ(\mathbf{u}+i\mathbf{w})-iQ(\mathbf{u}-i\mathbf{w})$ holds by direct expansion.\\
Thus $(\forall\,(\mathbf{u},\mathbf{w})\in\mathcal{V}\times\mathcal{V})\,\innerprod{T\mathbf{u},\mathbf{w}}=0$ by the hypothesis.\\
Taking $\mathbf{w}=T\mathbf{u}$, $(\forall\,\mathbf{u}\in\mathcal{V})\,\norm{T\mathbf{u}}^2=0$, and hence $T=0$ by positive definiteness.\\
\hlt{Structural Commentary}: Over $\mathbb{C}$, the diagonal values determine all mixed values.\\
\hlt{Key Insight}: Polarization turns the one-vector condition into a two-vector condition.
\end{proof}
\end{lemma}

\begin{proposition} \label{prop:complex-quadratic-form-self-adjoint-characterization} \hlt{Self-Adjointness from Real Diagonal Values}\\
Let $\mathcal{V}$ be a finite-dimensional complex inner-product space and let $T\in\LL(\mathcal{V})$.\\
Then $T=T^*\Longleftrightarrow(\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod{T\mathbf{v},\mathbf{v}}\in\mathbb{R}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $T=T^*$ and fix $\mathbf{v}\in\mathcal{V}$. Then $\innerprod{T\mathbf{v},\mathbf{v}}=\innerprod{\mathbf{v},T\mathbf{v}}=\overline{\innerprod{T\mathbf{v},\mathbf{v}}}$, so $\innerprod{T\mathbf{v},\mathbf{v}}\in\mathbb{R}$.
\item[$(\Leftarrow)$] Suppose $(\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod{T\mathbf{v},\mathbf{v}}\in\mathbb{R}$. Then $(\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod{(T-T^*)\mathbf{v},\mathbf{v}}=\innerprod{T\mathbf{v},\mathbf{v}}-\overline{\innerprod{T\mathbf{v},\mathbf{v}}}=0$. Hence $T-T^*=0$ by Lemma \ref{lem:complex-zero-quadratic-form-implies-zero-operator}.
\end{enumerate}
\hlt{Structural Commentary}: Adjoint symmetry proves reality; polarization eliminates the skew-adjoint difference.\\
\hlt{Key Insight}: Real diagonal values annihilate $T-T^*$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:self-adjoint-zero-quadratic-form-implies-zero} \hlt{Vanishing Diagonal Values for Self-Adjoint Operators}\\
Let $T$ be a self-adjoint operator on a finite-dimensional inner-product space $\mathcal{V}$. If $(\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod{T\mathbf{v},\mathbf{v}}=0$, then $T=0$.
\begin{proof}
If $\mathbb{F}=\mathbb{C}$, then $T=0$ by Lemma \ref{lem:complex-zero-quadratic-form-implies-zero-operator}. Suppose $\mathbb{F}=\mathbb{R}$.\\
For $(\mathbf{u},\mathbf{w})\in\mathcal{V}\times\mathcal{V}$, $\innerprod{T\mathbf{w},\mathbf{u}}=\innerprod{\mathbf{w},T\mathbf{u}}=\innerprod{T\mathbf{u},\mathbf{w}}$ by self-adjointness and symmetry.\\
Hence $4\innerprod{T\mathbf{u},\mathbf{w}}=\innerprod{T(\mathbf{u}+\mathbf{w}),\mathbf{u}+\mathbf{w}}-\innerprod{T(\mathbf{u}-\mathbf{w}),\mathbf{u}-\mathbf{w}}=0$ by direct expansion and the hypothesis.\\
Taking $\mathbf{w}=T\mathbf{u}$, $(\forall\,\mathbf{u}\in\mathcal{V})\,\norm{T\mathbf{u}}^2=0$, and hence $T=0$ by positive definiteness.\\
\hlt{Structural Commentary}: Self-adjointness supplies the missing symmetry needed for real polarization.\\
\hlt{Key Insight}: A self-adjoint operator is determined by its diagonal values.
\end{proof}
\end{proposition}

\begin{definition} \label{def:orthogonal-projection-matrix} \hlt{Orthogonal Projection Matrix}\\
Let $n\geq1$. A matrix $P\in\operatorname{Mat}(n,n,\mathbb{F})$ is an \hlt{orthogonal projection matrix} if $P^2=P$ and $P^*=P$.
\end{definition}

\begin{theorem} \label{thm:orthogonal-projection-self-adjoint-idempotent} \hlt{Orthogonal Projections as Self-Adjoint Idempotents}\\
Let $\mathcal{V}$ be a finite-dimensional inner-product space and let $P\in\LL(\mathcal{V})$.\\
Then $P=P_{\operatorname{range}(P)}\Longleftrightarrow[P^2=P\land P^*=P]$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Set $\mathcal{U}\defeq\operatorname{range}(P)$ and suppose $P=P_{\mathcal{U}}$. Then $P^2=P$ by Proposition \ref{prop:orthogonal-projection-properties}(vii).\\
The identity $P=\rho_{\mathcal{U},\mathcal{U}^{\perp}}$ holds by the defining decompositions of the two projections.\\
Hence $P^*=\rho_{(\mathcal{U}^{\perp})^{\perp},\mathcal{U}^{\perp}}=\rho_{\mathcal{U},\mathcal{U}^{\perp}}=P$ by Proposition \ref{prop:adjoint-product-kernels-ranges-and-projections}(iii) and Corollary \ref{cor:double-orthogonal-complement}.
\item[$(\Leftarrow)$] Suppose $P^2=P$ and $P^*=P$. Then $\mathcal{V}=\operatorname{range}(P)\oplus\ker(P)$ and $P=\rho_{\operatorname{range}(P),\ker(P)}$ by Proposition \ref{prop:algebraic-projection-range-kernel-idempotent}(i).\\
The identity $\ker(P)=\ker(P^*)=(\operatorname{range}(P))^{\perp}$ holds by Proposition \ref{prop:adjoint-kernel-range-orthogonality}(i).\\
Therefore $P=\rho_{\operatorname{range}(P),(\operatorname{range}(P))^{\perp}}=P_{\operatorname{range}(P)}$ by the defining decompositions of the two projections.
\end{enumerate}
\hlt{Structural Commentary}: Algebraic splitting identifies the kernel; adjoint orthogonality makes the decomposition metric.\\
\hlt{Key Insight}: Orthogonality is the kernel identity $\ker(P)=(\operatorname{range}(P))^{\perp}$.
\end{proof}
\end{theorem}
