\subsection{Adjoints and Fundamental Subspaces}


% **Role:** linear algebra with metric and angle structure.
%
% Adjoints, conjugate transpose, invariant subspaces, kernels, and ranges.

Adjoints express dual action in inner-product language. They become conjugate transposes in orthonormal coordinates and exchange kernels, ranges, and orthogonal complements.

Throughout this subsection, $\mathbb{F}$ is $\mathbb{R}$ or $\mathbb{C}$.

\begin{definition} \label{def:adjoint-linear-map} \hlt{Adjoint of a Linear Map}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be inner-product spaces and let $T\in\LL(\mathcal{V},\mathcal{W})$.\\
A map $S:\mathcal{W}\to\mathcal{V}$ is an \hlt{adjoint} of $T$ if $(\forall\,\mathbf{v}\in\mathcal{V})(\forall\,\mathbf{w}\in\mathcal{W})\,\innerprod[\mathcal{W}]{T\mathbf{v},\mathbf{w}}=\innerprod[\mathcal{V}]{\mathbf{v},S\mathbf{w}}$.
\end{definition}

\begin{perspective} \label{per:dual-map-versus-inner-product-adjoint} \hlt{Dual Map and Inner-Product Adjoint}\\
By Definition \ref{def:dual-map}, the \hlt{dual map} of $T\in\LL(\mathcal{V},\mathcal{W})$ is the pullback $T':\mathcal{W}'\to\mathcal{V}'$, $T'\varphi=\varphi\circ T$.\\
This algebraic adjoint, sometimes written $T^\times$, is distinct from the inner-product adjoint $T^*$, whose construction requires inner products.
\end{perspective}

\begin{lemma} \label{lem:adjoint-unique-if-exists} \hlt{Uniqueness of Adjoints}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be inner-product spaces and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then $T$ has at most one adjoint.
\begin{proof}
Suppose $S_1,S_2:\mathcal{W}\to\mathcal{V}$ are adjoints of $T$ and fix $\mathbf{w}\in\mathcal{W}$.\\
Then $(\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod[\mathcal{V}]{\mathbf{v},S_1\mathbf{w}}=\innerprod[\mathcal{V}]{\mathbf{v},S_2\mathbf{w}}$ by the two adjoint identities.\\
Thus $S_1\mathbf{w}=S_2\mathbf{w}$ by Lemma \ref{lem:inner-products-separate-vectors}(i), and $S_1=S_2$ by pointwise equality.\\
\hlt{Structural Commentary}: Pointwise comparison reduces uniqueness to separation by inner products.\\
\hlt{Key Insight}: Distinct vectors cannot agree against every inner-product test.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:finite-dimensional-adjoint-exists} \hlt{Existence of the Adjoint}\\
Let $\mathcal{V}$ be a finite-dimensional inner-product space, let $\mathcal{W}$ be an inner-product space, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ has an adjoint $T^*:\mathcal{W}\to\mathcal{V}$;
\item the adjoint $T^*$ is unique;
\item $T^*\in\LL(\mathcal{W},\mathcal{V})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $\mathbf{w}\in\mathcal{W}$. The map $\varphi_{\mathbf{w}}\defeq\innerprod[\mathcal{W}]{T(\cdot),\mathbf{w}}$ belongs to $\mathcal{V}'$ by linearity of $T$ and the first inner-product slot.\\
By Theorem \ref{thm:riesz-representation-finite-dimensional}, $(\exists!\,\mathbf{u}_{\mathbf{w}}\in\mathcal{V})(\forall\,\mathbf{v}\in\mathcal{V})\,\varphi_{\mathbf{w}}(\mathbf{v})=\innerprod[\mathcal{V}]{\mathbf{v},\mathbf{u}_{\mathbf{w}}}$.\\
Define $T^*\mathbf{w}\defeq\mathbf{u}_{\mathbf{w}}$. Since $\mathbf{w}$ was arbitrary, $T^*:\mathcal{W}\to\mathcal{V}$ is an adjoint of $T$.
\item The adjoint $T^*$ is unique by Lemma \ref{lem:adjoint-unique-if-exists}.
\item Fix $a\in\mathbb{F}$ and $\mathbf{w}_1,\mathbf{w}_2\in\mathcal{W}$, and set $\mathbf{x}\defeq aT^*\mathbf{w}_1+T^*\mathbf{w}_2$.\\
For $\mathbf{v}\in\mathcal{V}$, $\innerprod[\mathcal{V}]{\mathbf{v},T^*(a\mathbf{w}_1+\mathbf{w}_2)-\mathbf{x}}=0$ by the adjoint identity and conjugate linearity in the second slot.\\
Thus $T^*(a\mathbf{w}_1+\mathbf{w}_2)=\mathbf{x}$ by Lemma \ref{lem:inner-products-separate-vectors}(ii), so $T^*\in\LL(\mathcal{W},\mathcal{V})$.
\end{enumerate}
\hlt{Structural Commentary}: Riesz representation constructs codomain-indexed vectors; separation verifies that their assignment is linear.\\
\hlt{Key Insight}: Riesz vectors assemble pointwise into the adjoint.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:riesz-transport-dual-adjoint} \hlt{Riesz Transport of the Dual Map}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional inner-product spaces.\\
For $\mathcal{X}\in\{\mathcal{V},\mathcal{W}\}$, let $R_{\mathcal{X}}:\mathcal{X}'\to\mathcal{X}$ be the Riesz map characterized by $(\forall\,\varphi\in\mathcal{X}')(\forall\,\mathbf{x}\in\mathcal{X})\,\varphi(\mathbf{x})=\innerprod[\mathcal{X}]{\mathbf{x},R_{\mathcal{X}}\varphi}$.\\
Then $(\forall\,T\in\LL(\mathcal{V},\mathcal{W}))\,T'=(R_{\mathcal{V}})^{-1}\circ T^*\circ R_{\mathcal{W}}$, equivalently $R_{\mathcal{V}}T'=T^*R_{\mathcal{W}}$.
\begin{proof}
Fix $T\in\LL(\mathcal{V},\mathcal{W})$ and $\varphi\in\mathcal{W}'$, and set $\mathbf{u}\defeq T^*R_{\mathcal{W}}\varphi$ and $\psi\defeq(R_{\mathcal{V}})^{-1}\mathbf{u}$.\\
For $\mathbf{v}\in\mathcal{V}$, $\psi(\mathbf{v})=\innerprod[\mathcal{V}]{\mathbf{v},\mathbf{u}}=\innerprod[\mathcal{W}]{T\mathbf{v},R_{\mathcal{W}}\varphi}=\varphi(T\mathbf{v})=(T'\varphi)(\mathbf{v})$ by the Riesz-map, adjoint, and dual-map identities.\\
Thus $T'=(R_{\mathcal{V}})^{-1}T^*R_{\mathcal{W}}$ by pointwise equality, and composition with $R_{\mathcal{V}}$ proves the equivalent identity.\\
\hlt{Structural Commentary}: Pointwise evaluation identifies Riesz transport with the dual-map pullback.\\
\hlt{Key Insight}: Two conjugate-linear Riesz maps recover the linear dual map.
\end{proof}
\end{proposition}

\begin{definition} \label{def:conjugate-transpose} \hlt{Conjugate Transpose}\\
For $A=[a_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$, the \hlt{conjugate transpose} of $A$ is $A^*=[\overline{a_{ji}}]\in\operatorname{Mat}(n,m,\mathbb{F})$.\\
Over $\mathbb{R}$, the conjugate transpose is the ordinary transpose.
\end{definition}

\begin{theorem} \label{thm:matrix-of-adjoint-is-conjugate-transpose} \hlt{Matrix of the Adjoint}\\
Let $\beta=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ and $\gamma=(\mathbf{f}_1,\ldots,\mathbf{f}_m)$ be orthonormal bases of the finite-dimensional inner-product spaces $\mathcal{V}$ and $\mathcal{W}$, respectively.\\
Then $(\forall\,T\in\LL(\mathcal{V},\mathcal{W}))\,[T^*]_{\beta\leftarrow\gamma}=([T]_{\gamma\leftarrow\beta})^*$.
\begin{proof}
Set $A\defeq[T]_{\gamma\leftarrow\beta}$ and $B\defeq[T^*]_{\beta\leftarrow\gamma}$.\\
If $n=0$ or $m=0$, then $B$ and $A^*$ are the unique $n\times m$ matrix, so $B=A^*$. Suppose $n,m\geq1$.\\
For $i\in\{1,\ldots,n\}$ and $j\in\{1,\ldots,m\}$, $A_{j,i}=\innerprod[\mathcal{W}]{T\mathbf{e}_i,\mathbf{f}_j}$ by Theorem \ref{thm:matrix-entries-from-orthonormal-codomain}(i).\\
By the same theorem for $T^*$, $B_{i,j}=\innerprod[\mathcal{V}]{T^*\mathbf{f}_j,\mathbf{e}_i}$.\\
The identity $B_{i,j}=\overline{\innerprod[\mathcal{V}]{\mathbf{e}_i,T^*\mathbf{f}_j}}=\overline{A_{j,i}}$ holds by conjugate symmetry and the adjoint identity. Since $i,j$ were arbitrary, $B=A^*$.\\
\hlt{Structural Commentary}: Empty dimensions are immediate; orthonormal coordinates match each remaining entry after conjugation.\\
\hlt{Key Insight}: Moving an operator across an inner product becomes conjugate transposition in orthonormal bases.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:basic-adjoint-identities} \hlt{Basic Adjoint Identities}\\
Let $\mathcal{U}$, $\mathcal{V}$, and $\mathcal{W}$ be finite-dimensional inner-product spaces. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(aS+T)^*=\overline{a}S^*+T^*$ for $S,T\in\LL(\mathcal{V},\mathcal{W})$ and $a\in\mathbb{F}$;
\item $(SR)^*=R^*S^*$ for $R\in\LL(\mathcal{U},\mathcal{V})$ and $S\in\LL(\mathcal{V},\mathcal{W})$;
\item $(T^*)^*=T$ for $T\in\LL(\mathcal{V},\mathcal{W})$;
\item $I_{\mathcal{V}}^*=I_{\mathcal{V}}$.
\end{enumerate}
\begin{proof}
Each claim follows by Lemma \ref{lem:adjoint-unique-if-exists} once the asserted adjoint verifies its defining identity.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{v}\in\mathcal{V}$ and $\mathbf{w}\in\mathcal{W}$, $\innerprod[\mathcal{W}]{(aS+T)\mathbf{v},\mathbf{w}}=\innerprod[\mathcal{V}]{\mathbf{v},(\overline{a}S^*+T^*)\mathbf{w}}$ by linearity and the adjoint identities.
\item For $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{W}$, $\innerprod[\mathcal{W}]{SR\mathbf{u},\mathbf{w}}=\innerprod[\mathcal{U}]{\mathbf{u},R^*S^*\mathbf{w}}$ by the two adjoint identities.
\item The identity $(\forall\,\mathbf{w}\in\mathcal{W})(\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod[\mathcal{V}]{T^*\mathbf{w},\mathbf{v}}=\innerprod[\mathcal{W}]{\mathbf{w},T\mathbf{v}}$ holds by conjugate symmetry and the adjoint identity.\\
Thus $T$ is the adjoint of $T^*$, so $(T^*)^*=T$.
\item For $\mathbf{v},\mathbf{w}\in\mathcal{V}$, $\innerprod{I_{\mathcal{V}}\mathbf{v},\mathbf{w}}=\innerprod{\mathbf{v},I_{\mathcal{V}}\mathbf{w}}$. Thus $I_{\mathcal{V}}^*=I_{\mathcal{V}}$.
\end{enumerate}
\hlt{Structural Commentary}: Each algebraic operation is transported through the two inner-product slots and then settled by uniqueness.\\
\hlt{Key Insight}: Adjoint formation conjugates scalars, reverses products, and is involutive.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:adjoints-inverses-and-polynomials} \hlt{Adjoints of Inverses and Polynomials}\\
Let $T\in\LL(\mathcal{V})$ on a finite-dimensional inner-product space. For $p(z)=\sum\limits_{j=0}^{m}a_jz^j\in\mathbb{F}[z]$, set $p^{\#}(z)\defeq\sum\limits_{j=0}^{m}\overline{a_j}z^j$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $T$ is invertible, then $(T^*)^{-1}=(T^{-1})^*$.
\item $p(T)^*=p^{\#}(T^*)$.
\item $p\in\mathbb{R}[z]\Longrightarrow p(T)^*=p(T^*)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The identities $(T^{-1})^*T^*=(TT^{-1})^*=I_{\mathcal{V}}$ and $T^*(T^{-1})^*=(T^{-1}T)^*=I_{\mathcal{V}}$ hold by Proposition \ref{prop:basic-adjoint-identities}(ii),(iv).\\
Thus $(T^*)^{-1}=(T^{-1})^*$ by uniqueness of two-sided inverses.
\item The base identity $(T^0)^*=I_{\mathcal{V}}=(T^*)^0$ holds by Proposition \ref{prop:basic-adjoint-identities}(iv).\\
If $(T^j)^*=(T^*)^j$, then $(T^{j+1})^*=(T^jT)^*=T^*(T^j)^*=(T^*)^{j+1}$ by Proposition \ref{prop:basic-adjoint-identities}(ii).\\
Thus $(\forall\,j\in\{0,\ldots,m\})\,(T^j)^*=(T^*)^j$ by induction.\\
The identity $p(T)^*=\sum\limits_{j=0}^{m}\overline{a_j}(T^j)^*=\sum\limits_{j=0}^{m}\overline{a_j}(T^*)^j=p^{\#}(T^*)$ holds by Proposition \ref{prop:basic-adjoint-identities}(i) and the power identity.
\item If $p\in\mathbb{R}[z]$, then $p^{\#}=p$, so $p(T)^*=p(T^*)$ by (ii).
\end{enumerate}
\hlt{Structural Commentary}: The product law controls inverses and powers, while conjugate additivity controls polynomial coefficients.\\
\hlt{Key Insight}: Adjoint formation applies coefficientwise conjugation to polynomial expressions in an operator.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:adjoint-kernel-range-orthogonality} \hlt{Adjoint Kernel-Range Orthogonality}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional inner-product spaces and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\ker(T^*)=(\operatorname{range}(T))^{\perp}$;
\item $\ker(T)=(\operatorname{range}(T^*))^{\perp}$;
\item $\operatorname{range}(T^*)=(\ker(T))^{\perp}$;
\item $\operatorname{range}(T)=(\ker(T^*))^{\perp}$.
\end{enumerate}
Equivalently, an adjoint annihilates exactly the vectors orthogonal to the range of the original map.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{w}\in\mathcal{W}$, $\mathbf{w}\in\ker(T^*)\Longleftrightarrow(\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod[\mathcal{V}]{\mathbf{v},T^*\mathbf{w}}=0$ by Lemma \ref{lem:inner-products-separate-vectors}(ii).\\
The latter condition is equivalent to $(\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod[\mathcal{W}]{\mathbf{w},T\mathbf{v}}=0$ by the adjoint identity and conjugate symmetry.\\
Hence $\mathbf{w}\in\ker(T^*)\Longleftrightarrow\mathbf{w}\in(\operatorname{range}(T))^{\perp}$ by the orthogonal-complement definition.
\item The identity $\ker(T)=\ker((T^*)^*)=(\operatorname{range}(T^*))^{\perp}$ holds by (i) for $T^*$ and Proposition \ref{prop:basic-adjoint-identities}(iii).
\item Taking orthogonal complements in (ii), $(\ker(T))^{\perp}=((\operatorname{range}(T^*))^{\perp})^{\perp}=\operatorname{range}(T^*)$ by Corollary \ref{cor:double-orthogonal-complement}.
\item Taking orthogonal complements in (i), $(\ker(T^*))^{\perp}=((\operatorname{range}(T))^{\perp})^{\perp}=\operatorname{range}(T)$ by Corollary \ref{cor:double-orthogonal-complement}.
\end{enumerate}
\hlt{Structural Commentary}: The kernel formula, involution, and double complementation establish all four identities.\\
\hlt{Key Insight}: Kernels and ranges are exchanged by taking an adjoint and an orthogonal complement.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:adjoint-product-kernels-ranges-and-projections} \hlt{Adjoint Products and Algebraic Projections}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional inner-product spaces and let $T\in\LL(\mathcal{V},\mathcal{W})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\ker(T^*T),\ker(TT^*))=(\ker(T),\ker(T^*))$.
\item $(\operatorname{range}(T^*T),\operatorname{range}(TT^*))=(\operatorname{range}(T^*),\operatorname{range}(T))$.
\item If $\mathcal{V}=\mathcal{U}\oplus\mathcal{Z}$, then $\mathcal{V}=\mathcal{Z}^{\perp}\oplus\mathcal{U}^{\perp}$.\\
In this decomposition, $(\rho_{\mathcal{U},\mathcal{Z}})^*=\rho_{\mathcal{Z}^{\perp},\mathcal{U}^{\perp}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{v}\in\mathcal{V}$, $\innerprod[\mathcal{V}]{\mathbf{v},T^*T\mathbf{v}}=\norm{T\mathbf{v}}^2$ by the adjoint identity.\\
Thus $\mathbf{v}\in\ker(T^*T)\Longleftrightarrow T\mathbf{v}=\mathbf{0}\Longleftrightarrow\mathbf{v}\in\ker(T)$ by this identity and definiteness.\\
The equality $\ker(TT^*)=\ker(T^*)$ follows by this result for $T^*$ and Proposition \ref{prop:basic-adjoint-identities}(iii).
\item The identities $(T^*T)^*=T^*T$ and $(TT^*)^*=TT^*$ hold by Proposition \ref{prop:basic-adjoint-identities}(ii),(iii).\\
The identity $\operatorname{range}(T^*T)=(\ker(T^*T))^{\perp}=(\ker(T))^{\perp}=\operatorname{range}(T^*)$ holds by (i) and Proposition \ref{prop:adjoint-kernel-range-orthogonality}(iii).\\
The identity $\operatorname{range}(TT^*)=(\ker(TT^*))^{\perp}=(\ker(T^*))^{\perp}=\operatorname{range}(T)$ holds by (i) and Proposition \ref{prop:adjoint-kernel-range-orthogonality}(iv).
\item Suppose $\mathcal{V}=\mathcal{U}\oplus\mathcal{Z}$ and set $P\defeq\rho_{\mathcal{U},\mathcal{Z}}$.\\
Fix $\mathbf{c}\in\mathcal{Z}^{\perp}\cap\mathcal{U}^{\perp}$. For $\mathbf{x}\in\mathcal{V}$, write $\mathbf{x}=\mathbf{u}+\mathbf{z}$ with $(\mathbf{u},\mathbf{z})\in\mathcal{U}\times\mathcal{Z}$.\\
Then $\innerprod{\mathbf{c},\mathbf{x}}=\innerprod{\mathbf{c},\mathbf{u}}+\innerprod{\mathbf{c},\mathbf{z}}=0$, so $\mathbf{c}\in\mathcal{V}^{\perp}=\{\mathbf{0}\}$ by Proposition \ref{prop:basic-properties-orthogonal-complement}(iii).\\
Hence $\mathcal{Z}^{\perp}\cap\mathcal{U}^{\perp}=\{\mathbf{0}\}$.\\
By Corollaries \ref{cor:dimension-of-orthogonal-complement} and \ref{cor:dimension-direct-sum}, $\dim(\mathcal{Z}^{\perp}\oplus\mathcal{U}^{\perp})=2\dim(\mathcal{V})-\dim(\mathcal{Z})-\dim(\mathcal{U})=\dim(\mathcal{V})$.\\
Thus $\mathcal{V}=\mathcal{Z}^{\perp}\oplus\mathcal{U}^{\perp}$ by Proposition \ref{prop:dimension-bounds-and-basis-criteria}(ii).\\
Set $Q\defeq\rho_{\mathcal{Z}^{\perp},\mathcal{U}^{\perp}}$.\\
For $\mathbf{x},\mathbf{y}\in\mathcal{V}$, write $(\mathbf{x},\mathbf{y})=(\mathbf{u}+\mathbf{z},\mathbf{a}+\mathbf{b})$ with $(\mathbf{u},\mathbf{z},\mathbf{a},\mathbf{b})\in\mathcal{U}\times\mathcal{Z}\times\mathcal{Z}^{\perp}\times\mathcal{U}^{\perp}$.\\
Then $P\mathbf{x}=\mathbf{u}$, $Q\mathbf{y}=\mathbf{a}$, and $\innerprod{\mathbf{u},\mathbf{b}}=\innerprod{\mathbf{z},\mathbf{a}}=0$ by the projection formulas and conjugate symmetry.\\
Consequently, $\innerprod{P\mathbf{x},\mathbf{y}}=\innerprod{\mathbf{u},\mathbf{a}}=\innerprod{\mathbf{x},Q\mathbf{y}}$.\\
Thus $P^*=Q$ by Lemma \ref{lem:adjoint-unique-if-exists}.
\end{enumerate}
\hlt{Structural Commentary}: Positivity fixes kernels, complementation fixes ranges, and decomposition fixes projection adjoints.\\
\hlt{Key Insight}: Adjoints preserve effective directions and orthogonally exchange projection complements.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:matrix-fundamental-subspaces-orthogonal-complements} \hlt{Matrix Fundamental Subspaces as Orthogonal Complements}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ act as $A:\mathbb{F}^n\to\mathbb{F}^m$ with the standard inner products. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\ker(A)=\operatorname{col}(A^*)^{\perp}$;
\item $\operatorname{col}(A^*)=\ker(A)^{\perp}$;
\item $\ker(A^*)=\operatorname{col}(A)^{\perp}$;
\item $\operatorname{col}(A)=\ker(A^*)^{\perp}$.
\item $\mathbb{F}=\mathbb{R}\Longrightarrow\mathbb{R}^n=\ker(A)\odot\operatorname{row}(A)$.
\item $\mathbb{F}=\mathbb{R}\Longrightarrow\mathbb{R}^m=\ker(A^t)\odot\operatorname{col}(A)$.
\end{enumerate}
\begin{proof}
The induced map and its adjoint have ranges $\operatorname{col}(A)$ and $\operatorname{col}(A^*)$, respectively, by Theorem \ref{thm:matrix-of-adjoint-is-conjugate-transpose}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The identity $\ker(A)=\operatorname{col}(A^*)^{\perp}$ holds by Proposition \ref{prop:adjoint-kernel-range-orthogonality}(ii).
\item The identity $\operatorname{col}(A^*)=\ker(A)^{\perp}$ holds by Proposition \ref{prop:adjoint-kernel-range-orthogonality}(iii).
\item The identity $\ker(A^*)=\operatorname{col}(A)^{\perp}$ holds by Proposition \ref{prop:adjoint-kernel-range-orthogonality}(i).
\item The identity $\operatorname{col}(A)=\ker(A^*)^{\perp}$ holds by Proposition \ref{prop:adjoint-kernel-range-orthogonality}(iv).
\item If $\mathbb{F}=\mathbb{R}$, then $A^*=A^t$ and $\operatorname{col}(A^t)=\operatorname{row}(A)$.\\
Hence $\mathbb{R}^n=\ker(A)\odot\operatorname{row}(A)$ by (ii) and Theorem \ref{thm:orthogonal-direct-sum-decomposition}.
\item If $\mathbb{F}=\mathbb{R}$, then $A^*=A^t$.\\
Hence $\mathbb{R}^m=\ker(A^t)\odot\operatorname{col}(A)$ by (iv) and Theorem \ref{thm:orthogonal-direct-sum-decomposition}.
\end{enumerate}
\hlt{Structural Commentary}: Standard coordinates convert operator formulas into matrix fundamental-subspace identities.\\
\hlt{Key Insight}: The row space complements $\ker(A)$, while the column space complements $\ker(A^*)$.
\end{proof}
\end{corollary}
