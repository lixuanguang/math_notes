\subsection{Least Squares and Metric Optimization}


% **Role:** linear algebra with metric and angle structure.
%
% Least squares, normal equations, full-rank formulas, minimum-norm solutions.

Least squares is projection theory as optimization: normal equations, full-rank formulas, and minimum-norm solutions.

Throughout this subsection, $\mathbb{F}$ is $\mathbb{R}$ or $\mathbb{C}$.

\begin{principle} \label{pri:least-squares-projection-principle} \hlt{Least Squares as Orthogonal Projection}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of an inner-product space $\mathcal{V}$, and let $\mathbf{v}\in\mathcal{V}$.\\
The best approximation to $\mathbf{v}$ from $\mathcal{U}$ is $P_{\mathcal{U}}\mathbf{v}$ by Theorem \ref{thm:orthogonal-projection-minimizes-distance}.\\
The residual $\mathbf{v}-P_{\mathcal{U}}\mathbf{v}$ belongs to $\mathcal{U}^{\perp}$ by Proposition \ref{prop:orthogonal-projection-properties}(v).
\end{principle}

\begin{definition} \label{def:matrix-least-squares-solution} \hlt{Matrix Least-Squares Solution}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{y}\in\mathbb{F}^m$.\\
A vector $\mathbf{x}_0\in\mathbb{F}^n$ is a \hlt{least-squares solution} of $A\mathbf{x}=\mathbf{y}$ if $(\forall\,\mathbf{x}\in\mathbb{F}^n)\,\norm{A\mathbf{x}_0-\mathbf{y}}_2\leq\norm{A\mathbf{x}-\mathbf{y}}_2$.\\
The vector $A\mathbf{x}_0$ is the \hlt{fitted vector}, and $\mathbf{y}-A\mathbf{x}_0$ is the \hlt{residual}.
\end{definition}

\begin{theorem} \label{thm:normal-equations-for-projection} \hlt{Normal Equations for Projection}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of an inner-product space $\mathcal{V}$, suppose $\mathcal{U}=\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$, and fix $\mathbf{v}\in\mathcal{V}$.\\
For $\mathbf{c}=(c_1,\ldots,c_n)\in\mathbb{F}^n$, write $\mathsf{N}(\mathbf{c})$ for $(\forall\,i\in\{1,\ldots,n\})\,\sum\limits_{j=1}^{n}c_j\innerprod{\mathbf{u}_j,\mathbf{u}_i}=\innerprod{\mathbf{v},\mathbf{u}_i}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,\mathbf{c}=(c_1,\ldots,c_n)\in\mathbb{F}^n)\,[P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{n}c_j\mathbf{u}_j\Longleftrightarrow\mathsf{N}(\mathbf{c})]$;
\item $(\exists\,\mathbf{c}\in\mathbb{F}^n)\,\mathsf{N}(\mathbf{c})$;
\item if $(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ is linearly independent, then $(\exists!\,\mathbf{c}\in\mathbb{F}^n)\,\mathsf{N}(\mathbf{c})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Fix $\mathbf{c}=(c_1,\ldots,c_n)\in\mathbb{F}^n$ and suppose $P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{n}c_j\mathbf{u}_j$. The residual belongs to $\mathcal{U}^{\perp}$ by Proposition \ref{prop:orthogonal-projection-properties}(v). Hence $(\forall\,i\in\{1,\ldots,n\})\,0=\innerprod{\mathbf{v}-P_{\mathcal{U}}\mathbf{v},\mathbf{u}_i}=\innerprod{\mathbf{v},\mathbf{u}_i}-\sum\limits_{j=1}^{n}c_j\innerprod{\mathbf{u}_j,\mathbf{u}_i}$, so $\mathsf{N}(\mathbf{c})$ holds.
\item[$(i)$] ($\Leftarrow$) Fix $\mathbf{c}=(c_1,\ldots,c_n)\in\mathbb{F}^n$, suppose $\mathsf{N}(\mathbf{c})$, and set $\mathbf{q}\defeq\sum\limits_{j=1}^{n}c_j\mathbf{u}_j\in\mathcal{U}$. For $\mathbf{u}=\sum\limits_{i=1}^{n}a_i\mathbf{u}_i\in\mathcal{U}$, $\innerprod{\mathbf{v}-\mathbf{q},\mathbf{u}}=\sum\limits_{i=1}^{n}\overline{a_i}\innerprod{\mathbf{v}-\mathbf{q},\mathbf{u}_i}=0$ by $\mathsf{N}(\mathbf{c})$ and Proposition \ref{prop:basic-inner-product-identities}(iii)--(iv). Thus $\mathbf{v}-\mathbf{q}\in\mathcal{U}^{\perp}$, and $\mathbf{q}=P_{\mathcal{U}}\mathbf{v}$ by uniqueness in Theorem \ref{thm:orthogonal-direct-sum-decomposition}.
\item[$(ii)$] Choose $\mathbf{c}=(c_1,\ldots,c_n)\in\mathbb{F}^n$ with $P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{n}c_j\mathbf{u}_j$; this choice is possible because $P_{\mathcal{U}}\mathbf{v}\in\mathcal{U}=\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ by Proposition \ref{prop:orthogonal-projection-properties}(iv). Then $\mathsf{N}(\mathbf{c})$ holds by (i).
\item[$(iii)$] Suppose $(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ is linearly independent. Existence holds by (ii). If $\mathsf{N}(\mathbf{c})$ and $\mathsf{N}(\mathbf{d})$, then $\sum\limits_{j=1}^{n}c_j\mathbf{u}_j=P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{n}d_j\mathbf{u}_j$ by (i), so $\mathbf{c}=\mathbf{d}$ by linear independence.
\end{enumerate}
\hlt{Structural Commentary}: Projection membership supplies coefficients; residual orthogonality supplies the equations and converse.\\
\hlt{Key Insight}: Normal equations are coordinates of an orthogonal residual.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:matrix-least-squares-normal-equations} \hlt{Matrix Least Squares and Normal Equations}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{y}\in\mathbb{F}^m$, and set $\operatorname{LS}(A,\mathbf{y})\defeq\operatorname*{arg\,min}\limits_{\mathbf{x}\in\mathbb{F}^n}\norm{A\mathbf{x}-\mathbf{y}}_2$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,\mathbf{x}_0\in\mathbb{F}^n)\,[\mathbf{x}_0\in\operatorname{LS}(A,\mathbf{y})\Longleftrightarrow A^*A\mathbf{x}_0=A^*\mathbf{y}]$;
\item $(\exists\,\mathbf{x}_0\in\mathbb{F}^n)\,A^*A\mathbf{x}_0=A^*\mathbf{y}$;
\item $[(\exists!\,\mathbf{x}_0\in\mathbb{F}^n)\,A^*A\mathbf{x}_0=A^*\mathbf{y}]\Longleftrightarrow\operatorname{rank}(A)=n$.
\end{enumerate}
\begin{proof}
Set $\mathcal{U}\defeq\operatorname{col}(A)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Suppose $\mathbf{x}_0$ is a least-squares solution. Then $A\mathbf{x}_0=P_{\mathcal{U}}\mathbf{y}$ by Theorem \ref{thm:orthogonal-projection-minimizes-distance}, so $\mathbf{y}-A\mathbf{x}_0\in\mathcal{U}^{\perp}=\ker(A^*)$ by Proposition \ref{prop:orthogonal-projection-properties}(v) and Corollary \ref{cor:matrix-fundamental-subspaces-orthogonal-complements}(iii). Hence $A^*A\mathbf{x}_0=A^*\mathbf{y}$.
\item[$(i)$] ($\Leftarrow$) Suppose $A^*A\mathbf{x}_0=A^*\mathbf{y}$. Then $\mathbf{y}-A\mathbf{x}_0\in\ker(A^*)=\mathcal{U}^{\perp}$ by Corollary \ref{cor:matrix-fundamental-subspaces-orthogonal-complements}(iii). Since $A\mathbf{x}_0\in\mathcal{U}$, the identity $A\mathbf{x}_0=P_{\mathcal{U}}\mathbf{y}$ holds by uniqueness in Theorem \ref{thm:orthogonal-direct-sum-decomposition}. Thus $\mathbf{x}_0$ is a least-squares solution by Theorem \ref{thm:orthogonal-projection-minimizes-distance}.
\item[$(ii)$] Since $P_{\mathcal{U}}\mathbf{y}\in\mathcal{U}=\operatorname{col}(A)$, choose $\mathbf{x}_0\in\mathbb{F}^n$ with $A\mathbf{x}_0=P_{\mathcal{U}}\mathbf{y}$. Then $A^*A\mathbf{x}_0=A^*\mathbf{y}$ by (i).
\item[$(iii)$] ($\Rightarrow$) Suppose the normal equations have a unique solution $\mathbf{x}_0$. If $\mathbf{z}\in\ker(A)$, then $\mathbf{x}_0+\mathbf{z}$ also solves them, so $\mathbf{z}=\mathbf{0}$ by uniqueness. Hence $\dim(\ker(A))=0$ and $\operatorname{rank}(A)=n$ by Corollary \ref{cor:matrix-rank-nullity}(i).
\item[$(iii)$] ($\Leftarrow$) Suppose $\operatorname{rank}(A)=n$. Existence holds by (ii), and $\ker(A)=\{\mathbf{0}\}$ by Corollary \ref{cor:matrix-rank-nullity}(i). If $\mathbf{x}_0$ and $\mathbf{x}_1$ solve the normal equations, then $\mathbf{x}_0-\mathbf{x}_1\in\ker(A^*A)=\ker(A)=\{\mathbf{0}\}$ by Proposition \ref{prop:adjoint-product-kernels-ranges-and-projections}(i). Thus $\mathbf{x}_0=\mathbf{x}_1$.
\end{enumerate}
\hlt{Structural Commentary}: Projection characterizes optimality; kernel identities establish consistency and uniqueness.\\
\hlt{Key Insight}: Normal equations annihilate the column-space residual.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:least-squares-affine-set-and-minimum-norm} \hlt{Least-Squares Solution Set and Minimum Norm}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, let $\mathbf{y}\in\mathbb{F}^m$, and set $\operatorname{LS}(A,\mathbf{y})\defeq\operatorname*{arg\,min}\limits_{\mathbf{x}\in\mathbb{F}^n}\norm{A\mathbf{x}-\mathbf{y}}_2$. Fix $\mathbf{x}_0\in\operatorname{LS}(A,\mathbf{y})$ and set $\mathbf{s}\defeq P_{(\ker(A))^{\perp}}\mathbf{x}_0$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{LS}(A,\mathbf{y})=\mathbf{x}_0+\ker(A)$;
\item $\operatorname{LS}(A,\mathbf{y})\cap(\ker(A))^{\perp}=\{\mathbf{s}\}$;
\item $\operatorname*{arg\,min}\limits_{\mathbf{x}\in\operatorname{LS}(A,\mathbf{y})}\norm{\mathbf{x}}_2=\{\mathbf{s}\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{x}\in\mathbb{F}^n$, $\mathbf{x}\in\operatorname{LS}(A,\mathbf{y})\Longleftrightarrow A\mathbf{x}=P_{\operatorname{col}(A)}\mathbf{y}\Longleftrightarrow A\mathbf{x}=A\mathbf{x}_0\Longleftrightarrow\mathbf{x}-\mathbf{x}_0\in\ker(A)$ by Theorem \ref{thm:orthogonal-projection-minimizes-distance}. Hence $\operatorname{LS}(A,\mathbf{y})=\mathbf{x}_0+\ker(A)$.
\item The relations $\mathbf{s}\in(\ker(A))^{\perp}$ and $\mathbf{x}_0-\mathbf{s}\in((\ker(A))^{\perp})^{\perp}=\ker(A)$ hold by Proposition \ref{prop:orthogonal-projection-properties}(iv)--(v) and Corollary \ref{cor:double-orthogonal-complement}. Thus $\mathbf{s}\in\operatorname{LS}(A,\mathbf{y})$ by (i). If $\mathbf{u}\in\operatorname{LS}(A,\mathbf{y})\cap(\ker(A))^{\perp}$, then $\mathbf{u}-\mathbf{s}\in\ker(A)\cap(\ker(A))^{\perp}=\{\mathbf{0}\}$ by (i) and Proposition \ref{prop:basic-properties-orthogonal-complement}(iv), so $\mathbf{u}=\mathbf{s}$.
\item Fix $\mathbf{x}\in\operatorname{LS}(A,\mathbf{y})$ and choose $\mathbf{z}\in\ker(A)$ with $\mathbf{x}=\mathbf{s}+\mathbf{z}$ by (i)--(ii). Then $\mathbf{s}\perp\mathbf{z}$, so $\norm{\mathbf{x}}_2^2=\norm{\mathbf{s}}_2^2+\norm{\mathbf{z}}_2^2\geq\norm{\mathbf{s}}_2^2$ by Theorem \ref{thm:pythagorean-inner-product}, with equality iff $\mathbf{z}=\mathbf{0}$, equivalently $\mathbf{x}=\mathbf{s}$.
\end{enumerate}
\hlt{Structural Commentary}: An affine kernel description and orthogonal decomposition isolate the minimum-norm point.\\
\hlt{Key Insight}: Removing the kernel component uniquely minimizes norm.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:full-column-rank-projection-formula} \hlt{Projection Formula for Full Column Rank}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ satisfy $\operatorname{rank}(A)=n$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A^*A$ is invertible;
\item $P_{\operatorname{col}(A)}=A(A^*A)^{-1}A^*$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The identities $\ker(A^*A)=\ker(A)=\{\mathbf{0}\}$ hold by Proposition \ref{prop:adjoint-product-kernels-ranges-and-projections}(i) and Corollary \ref{cor:matrix-rank-nullity}(i). Hence $A^*A$ is injective and therefore invertible by Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences}.
\item Fix $\mathbf{y}\in\mathbb{F}^m$ and set $\mathbf{x}_0\defeq(A^*A)^{-1}A^*\mathbf{y}$. Then $A^*A\mathbf{x}_0=A^*\mathbf{y}$, so $\mathbf{x}_0$ is a least-squares solution by Theorem \ref{thm:matrix-least-squares-normal-equations}(i). Consequently, $P_{\operatorname{col}(A)}\mathbf{y}=A\mathbf{x}_0=A(A^*A)^{-1}A^*\mathbf{y}$ by Theorem \ref{thm:orthogonal-projection-minimizes-distance}. Since $\mathbf{y}$ was arbitrary, the operator identity follows.
\end{enumerate}
\hlt{Structural Commentary}: Kernel equality proves invertibility; the normal-equation solution determines the projection matrix.\\
\hlt{Key Insight}: Full column rank makes normal-equation coordinates explicit.
\end{proof}
\end{corollary}

\begin{definition} \label{def:minimum-norm-solution} \hlt{Minimum-Norm Solution}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{y}\in\operatorname{col}(A)$.\\
A solution $\mathbf{s}\in\mathbb{F}^n$ of $A\mathbf{x}=\mathbf{y}$ is a \hlt{minimum-norm solution} if $(\forall\,\mathbf{x}\in\mathbb{F}^n)\,[A\mathbf{x}=\mathbf{y}\Longrightarrow\norm{\mathbf{s}}_2\leq\norm{\mathbf{x}}_2]$.
\end{definition}

\begin{theorem} \label{thm:minimum-norm-solution-consistent-system} \hlt{Minimum-Norm Solution of a Consistent Linear System}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, let $\mathbf{y}\in\mathbb{F}^m$, and suppose $A\mathbf{x}=\mathbf{y}$ is consistent. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\exists!\,\mathbf{s}\in\mathbb{F}^n)\,[A\mathbf{s}=\mathbf{y}\land(\forall\,\mathbf{x}\in\mathbb{F}^n)\,(A\mathbf{x}=\mathbf{y}\Longrightarrow\norm{\mathbf{s}}_2\leq\norm{\mathbf{x}}_2)]$;
\item for the vector $\mathbf{s}$ in (i), $\{\mathbf{x}\in\operatorname{col}(A^*):A\mathbf{x}=\mathbf{y}\}=\{\mathbf{s}\}$;
\item $(\exists\,\mathbf{u}\in\mathbb{F}^m)\,AA^*\mathbf{u}=\mathbf{y}$;
\item for the vector $\mathbf{s}$ in (i), $(\forall\,\mathbf{u}_0\in\mathbb{F}^m)\,[AA^*\mathbf{u}_0=\mathbf{y}\Longrightarrow\mathbf{s}=A^*\mathbf{u}_0]$.
\end{enumerate}
\begin{proof}
The identities $\operatorname{col}(A^*)=(\ker(A))^{\perp}$ and $\ker(A)=(\operatorname{col}(A^*))^{\perp}$ hold by Corollary \ref{cor:matrix-fundamental-subspaces-orthogonal-complements}(ii) and (i), respectively. Hence $\mathbb{F}^n=\operatorname{col}(A^*)\odot\ker(A)$ by Theorem \ref{thm:orthogonal-direct-sum-decomposition}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $\mathbf{x}_0\in\mathbb{F}^n$ with $A\mathbf{x}_0=\mathbf{y}$ and write $\mathbf{x}_0=\mathbf{s}+\mathbf{v}$ with $\mathbf{s}\in\operatorname{col}(A^*)$ and $\mathbf{v}\in\ker(A)$. Then $A\mathbf{s}=\mathbf{y}$. Fix $\mathbf{x}\in\mathbb{F}^n$ with $A\mathbf{x}=\mathbf{y}$ and set $\mathbf{w}\defeq\mathbf{x}-\mathbf{s}\in\ker(A)$. Then $\norm{\mathbf{x}}_2^2=\norm{\mathbf{s}}_2^2+\norm{\mathbf{w}}_2^2\geq\norm{\mathbf{s}}_2^2$ by Theorem \ref{thm:pythagorean-inner-product}. Nonnegativity of norms proves minimality, and equality holds iff $\mathbf{w}=\mathbf{0}$, which is equivalent to $\mathbf{x}=\mathbf{s}$ and proves uniqueness.
\item The constructed $\mathbf{s}$ is a solution in $\operatorname{col}(A^*)$. If $\mathbf{s}'$ is another such solution, then $\mathbf{s}'-\mathbf{s}\in\ker(A)\cap\operatorname{col}(A^*)=\{\mathbf{0}\}$, so $\mathbf{s}'=\mathbf{s}$.
\item Since $\mathbf{s}\in\operatorname{col}(A^*)$, choose $\mathbf{u}\in\mathbb{F}^m$ with $\mathbf{s}=A^*\mathbf{u}$. Then $AA^*\mathbf{u}=A\mathbf{s}=\mathbf{y}$.
\item If $AA^*\mathbf{u}_0=\mathbf{y}$, then $A^*\mathbf{u}_0$ is a solution in $\operatorname{col}(A^*)$, so $A^*\mathbf{u}_0=\mathbf{s}$ by (ii).
\end{enumerate}
\hlt{Structural Commentary}: Orthogonal splitting constructs the minimizer; the column-space characterization proves representation.\\
\hlt{Key Insight}: Minimum norm means zero nullspace component.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:least-squares-pseudoinverse-roadmap} \hlt{Roadmap to the Pseudoinverse}\\
Theorem \ref{thm:matrix-least-squares-normal-equations} projects $\mathbf{y}$ onto $\operatorname{col}(A)$; Proposition \ref{prop:least-squares-affine-set-and-minimum-norm} selects the minimum-norm preimage in $(\ker(A))^{\perp}$, identified with $\operatorname{col}(A^*)$ in Theorem \ref{thm:minimum-norm-solution-consistent-system}.\\
Later, Definition \ref{def:moore-penrose-pseudoinverse} and Theorem \ref{thm:pseudoinverse-minimum-norm-least-squares} package both choices as $A^{\dagger}\mathbf{y}$.
\end{perspective}
