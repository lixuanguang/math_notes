\subsection{Least Squares and Metric Optimization}


% **Role:** linear algebra with metric and angle structure.
%
% Least squares, normal equations, full-rank formulas, minimum-norm solutions.

Least squares is projection theory as optimization: normal equations, full-rank formulas, and minimum-norm solutions.

Throughout this subsection, $\mathbb{F}$ is $\mathbb{R}$ or $\mathbb{C}$.

\begin{principle} \label{pri:least-squares-projection-principle} \hlt{Least Squares as Orthogonal Projection}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of an inner-product space $\mathcal{V}$, and let $\mathbf{v}\in\mathcal{V}$.\\
The identity $\operatorname*{arg\,min}\limits_{\mathbf{u}\in\mathcal{U}}\norm{\mathbf{v}-\mathbf{u}}=\{P_{\mathcal{U}}\mathbf{v}\}$ holds by Theorem \ref{thm:orthogonal-projection-minimizes-distance}. The residual $\mathbf{v}-P_{\mathcal{U}}\mathbf{v}$ belongs to $\mathcal{U}^{\perp}$ by Proposition \ref{prop:orthogonal-projection-properties}(v).
\end{principle}

\begin{definition} \label{def:matrix-least-squares-solution} \hlt{Matrix Least-Squares Solution}\\
For $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{y}\in\mathbb{F}^m$, define $\operatorname{LS}(A,\mathbf{y})\defeq\operatorname*{arg\,min}\limits_{\mathbf{x}\in\mathbb{F}^n}\norm{A\mathbf{x}-\mathbf{y}}_2$.\\
An element $\mathbf{x}_0\in\operatorname{LS}(A,\mathbf{y})$ is a \hlt{least-squares solution} of $A\mathbf{x}=\mathbf{y}$. Its \hlt{fitted vector} is $A\mathbf{x}_0$, and its \hlt{residual} is $\mathbf{y}-A\mathbf{x}_0$.
\end{definition}

\begin{theorem} \label{thm:normal-equations-for-projection} \hlt{Normal Equations for Projection}\\
Let $\mathcal{U}=\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ be a finite-dimensional subspace of an inner-product space $\mathcal{V}$, and fix $\mathbf{v}\in\mathcal{V}$.\\
For $\mathbf{c}=(c_1,\ldots,c_n)\in\mathbb{F}^n$, set $\mathsf{N}(\mathbf{c})\defeq(\forall\,i\in\{1,\ldots,n\})\,\sum\limits_{j=1}^{n}c_j\innerprod{\mathbf{u}_j,\mathbf{u}_i}=\innerprod{\mathbf{v},\mathbf{u}_i}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,\mathbf{c}\in\mathbb{F}^n)\,[P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{n}c_j\mathbf{u}_j\Longleftrightarrow\mathsf{N}(\mathbf{c})]$;
\item $(\exists\,\mathbf{c}\in\mathbb{F}^n)\,\mathsf{N}(\mathbf{c})$;
\item if $(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ is linearly independent, then $(\exists!\,\mathbf{c}\in\mathbb{F}^n)\,\mathsf{N}(\mathbf{c})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Fix $\mathbf{c}=(c_1,\ldots,c_n)\in\mathbb{F}^n$ with $P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{n}c_j\mathbf{u}_j$. The residual belongs to $\mathcal{U}^{\perp}$ by Proposition \ref{prop:orthogonal-projection-properties}(v).\\
Thus $(\forall\,i\in\{1,\ldots,n\})\,0=\innerprod{\mathbf{v}-P_{\mathcal{U}}\mathbf{v},\mathbf{u}_i}=\innerprod{\mathbf{v},\mathbf{u}_i}-\sum\limits_{j=1}^{n}c_j\innerprod{\mathbf{u}_j,\mathbf{u}_i}$, so $\mathsf{N}(\mathbf{c})$ holds.
\item[$(i)$] ($\Leftarrow$) Fix $\mathbf{c}=(c_1,\ldots,c_n)\in\mathbb{F}^n$ with $\mathsf{N}(\mathbf{c})$, and set $\mathbf{q}\defeq\sum\limits_{j=1}^{n}c_j\mathbf{u}_j\in\mathcal{U}$.\\
For $\mathbf{u}=\sum\limits_{i=1}^{n}a_i\mathbf{u}_i\in\mathcal{U}$, $\innerprod{\mathbf{v}-\mathbf{q},\mathbf{u}}=\sum\limits_{i=1}^{n}\overline{a_i}\innerprod{\mathbf{v}-\mathbf{q},\mathbf{u}_i}=0$ by $\mathsf{N}(\mathbf{c})$ and Proposition \ref{prop:basic-inner-product-identities}(iii)--(iv).\\
Hence $\mathbf{v}-\mathbf{q}\in\mathcal{U}^{\perp}$, and $\mathbf{q}=P_{\mathcal{U}}\mathbf{v}$ by uniqueness in Theorem \ref{thm:orthogonal-direct-sum-decomposition}.
\item[$(ii)$] Relation $P_{\mathcal{U}}\mathbf{v}\in\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ holds by Proposition \ref{prop:orthogonal-projection-properties}(iv). Choose $\mathbf{c}\in\mathbb{F}^n$ with $P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{n}c_j\mathbf{u}_j$; then $\mathsf{N}(\mathbf{c})$ holds by (i).
\item[$(iii)$] Suppose $(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ is linearly independent. Existence holds by (ii).\\
If $\mathsf{N}(\mathbf{c})$ and $\mathsf{N}(\mathbf{d})$, then $\sum\limits_{j=1}^{n}c_j\mathbf{u}_j=P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{n}d_j\mathbf{u}_j$ by (i), and hence $\mathbf{c}=\mathbf{d}$ by linear independence.
\end{enumerate}
\hlt{Structural Commentary}: Projection coordinates construct solutions; residual orthogonality characterizes every coefficient vector.\\
\hlt{Key Insight}: Normal equations are the spanning-vector tests for an orthogonal residual.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:matrix-least-squares-normal-equations} \hlt{Matrix Least Squares and Normal Equations}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{y}\in\mathbb{F}^m$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,\mathbf{x}_0\in\mathbb{F}^n)\,[\mathbf{x}_0\in\operatorname{LS}(A,\mathbf{y})\Longleftrightarrow A^*A\mathbf{x}_0=A^*\mathbf{y}]$;
\item $(\exists\,\mathbf{x}_0\in\mathbb{F}^n)\,A^*A\mathbf{x}_0=A^*\mathbf{y}$;
\item $[(\exists!\,\mathbf{x}_0\in\mathbb{F}^n)\,A^*A\mathbf{x}_0=A^*\mathbf{y}]\Longleftrightarrow\operatorname{rank}(A)=n$.
\end{enumerate}
\begin{proof}
Set $\mathcal{U}\defeq\operatorname{col}(A)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Fix $\mathbf{x}_0\in\operatorname{LS}(A,\mathbf{y})$. Then $A\mathbf{x}_0=P_{\mathcal{U}}\mathbf{y}$ by Theorem \ref{thm:orthogonal-projection-minimizes-distance}.\\
The relation $\mathbf{y}-A\mathbf{x}_0\in\mathcal{U}^{\perp}=\ker(A^*)$ holds by Proposition \ref{prop:orthogonal-projection-properties}(v) and Corollary \ref{cor:matrix-fundamental-subspaces-orthogonal-complements}(iii). Hence $A^*A\mathbf{x}_0=A^*\mathbf{y}$.
\item[$(i)$] ($\Leftarrow$) Suppose $A^*A\mathbf{x}_0=A^*\mathbf{y}$. Then $\mathbf{y}-A\mathbf{x}_0\in\ker(A^*)=\mathcal{U}^{\perp}$ by Corollary \ref{cor:matrix-fundamental-subspaces-orthogonal-complements}(iii).\\
Since $A\mathbf{x}_0\in\mathcal{U}$, the identity $A\mathbf{x}_0=P_{\mathcal{U}}\mathbf{y}$ holds by uniqueness in Theorem \ref{thm:orthogonal-direct-sum-decomposition}. Thus $\mathbf{x}_0\in\operatorname{LS}(A,\mathbf{y})$ by Theorem \ref{thm:orthogonal-projection-minimizes-distance}.
\item[$(ii)$] The relation $P_{\mathcal{U}}\mathbf{y}\in\mathcal{U}=\operatorname{col}(A)$ holds by Proposition \ref{prop:orthogonal-projection-properties}(iv). Choose $\mathbf{x}_0\in\mathbb{F}^n$ with $A\mathbf{x}_0=P_{\mathcal{U}}\mathbf{y}$; then $A^*A\mathbf{x}_0=A^*\mathbf{y}$ by (i).
\item[$(iii)$] ($\Rightarrow$) Suppose the normal equations have the unique solution $\mathbf{x}_0$.\\
For $\mathbf{z}\in\ker(A)$, $A^*A(\mathbf{x}_0+\mathbf{z})=A^*A\mathbf{x}_0=A^*\mathbf{y}$, so $\mathbf{z}=\mathbf{0}$ by uniqueness. Thus $\dim(\ker(A))=0$ and $\operatorname{rank}(A)=n$ by Corollary \ref{cor:matrix-rank-nullity}(i).
\item[$(iii)$] ($\Leftarrow$) Suppose $\operatorname{rank}(A)=n$. Existence holds by (ii), and $\ker(A)=\{\mathbf{0}\}$ by Corollary \ref{cor:matrix-rank-nullity}(i).\\
If $\mathbf{x}_0$ and $\mathbf{x}_1$ solve the normal equations, then $\mathbf{x}_0-\mathbf{x}_1\in\ker(A^*A)=\ker(A)=\{\mathbf{0}\}$ by Proposition \ref{prop:adjoint-product-kernels-ranges-and-projections}(i).\\
Thus $\mathbf{x}_0=\mathbf{x}_1$.
\end{enumerate}
\hlt{Structural Commentary}: Projection converts optimality to residual orthogonality; kernel identities control existence and uniqueness.\\
\hlt{Key Insight}: The normal equations place the residual in $\ker(A^*)=\operatorname{col}(A)^{\perp}$.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:least-squares-affine-set-and-minimum-norm} \hlt{Least-Squares Solution Set and Minimum Norm}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{y}\in\mathbb{F}^m$. Fix $\mathbf{x}_0\in\operatorname{LS}(A,\mathbf{y})$ and set $\mathbf{s}\defeq P_{(\ker(A))^{\perp}}\mathbf{x}_0$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{LS}(A,\mathbf{y})=\mathbf{x}_0+\ker(A)$;
\item $\operatorname{LS}(A,\mathbf{y})\cap(\ker(A))^{\perp}=\{\mathbf{s}\}$;
\item $\operatorname*{arg\,min}\limits_{\mathbf{x}\in\operatorname{LS}(A,\mathbf{y})}\norm{\mathbf{x}}_2=\{\mathbf{s}\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{x}\in\mathbb{F}^n$, $\mathbf{x}\in\operatorname{LS}(A,\mathbf{y})\Longleftrightarrow A\mathbf{x}=P_{\operatorname{col}(A)}\mathbf{y}$ by Theorem \ref{thm:orthogonal-projection-minimizes-distance}.\\
The equality $A\mathbf{x}_0=P_{\operatorname{col}(A)}\mathbf{y}$ holds by $\mathbf{x}_0\in\operatorname{LS}(A,\mathbf{y})$ and Theorem \ref{thm:orthogonal-projection-minimizes-distance}.\\
Thus $\mathbf{x}\in\operatorname{LS}(A,\mathbf{y})\Longleftrightarrow A\mathbf{x}=A\mathbf{x}_0\Longleftrightarrow\mathbf{x}-\mathbf{x}_0\in\ker(A)$, and hence $\operatorname{LS}(A,\mathbf{y})=\mathbf{x}_0+\ker(A)$.
\item The relation $\mathbf{s}\in(\ker(A))^{\perp}$ holds by Proposition \ref{prop:orthogonal-projection-properties}(iv).\\
The identity $\mathbf{x}_0-\mathbf{s}\in((\ker(A))^{\perp})^{\perp}=\ker(A)$ holds by Proposition \ref{prop:orthogonal-projection-properties}(v) and Corollary \ref{cor:double-orthogonal-complement}. Hence $\mathbf{s}\in\operatorname{LS}(A,\mathbf{y})$ by (i).\\
If $\mathbf{u}\in\operatorname{LS}(A,\mathbf{y})\cap(\ker(A))^{\perp}$, then $\mathbf{u}-\mathbf{s}\in\ker(A)\cap(\ker(A))^{\perp}=\{\mathbf{0}\}$ by (i) and Proposition \ref{prop:basic-properties-orthogonal-complement}(iv), so $\mathbf{u}=\mathbf{s}$.
\item Fix $\mathbf{x}\in\operatorname{LS}(A,\mathbf{y})$ and set $\mathbf{z}\defeq\mathbf{x}-\mathbf{s}\in\ker(A)$ by (i)--(ii).\\
Then $\mathbf{s}\perp\mathbf{z}$, and $\norm{\mathbf{x}}_2^2=\norm{\mathbf{s}}_2^2+\norm{\mathbf{z}}_2^2\geq\norm{\mathbf{s}}_2^2$ by Theorem \ref{thm:pythagorean-inner-product}, with equality iff $\mathbf{z}=\mathbf{0}$, equivalently $\mathbf{x}=\mathbf{s}$.
\end{enumerate}
\hlt{Structural Commentary}: Kernel translation describes every minimizer; orthogonal decomposition isolates the shortest representative.\\
\hlt{Key Insight}: Norm excess is exactly the squared kernel-component norm.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:full-column-rank-projection-formula} \hlt{Projection Formula for Full Column Rank}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ satisfy $\operatorname{rank}(A)=n$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A^*A$ is invertible;
\item $P_{\operatorname{col}(A)}=A(A^*A)^{-1}A^*$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The identities $\ker(A^*A)=\ker(A)=\{\mathbf{0}\}$ hold by Proposition \ref{prop:adjoint-product-kernels-ranges-and-projections}(i) and Corollary \ref{cor:matrix-rank-nullity}(i).\\
Hence $A^*A$ is injective and therefore invertible by Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences}.
\item Fix $\mathbf{y}\in\mathbb{F}^m$ and set $\mathbf{x}_0\defeq(A^*A)^{-1}A^*\mathbf{y}$. Then $A^*A\mathbf{x}_0=A^*\mathbf{y}$.\\
The relation $\mathbf{x}_0\in\operatorname{LS}(A,\mathbf{y})$ holds by Theorem \ref{thm:matrix-least-squares-normal-equations}(i). Thus $P_{\operatorname{col}(A)}\mathbf{y}=A\mathbf{x}_0=A(A^*A)^{-1}A^*\mathbf{y}$ by Theorem \ref{thm:orthogonal-projection-minimizes-distance}.\\
The operator identity follows by arbitrariness of $\mathbf{y}$.
\end{enumerate}
\hlt{Structural Commentary}: Kernel equality makes $A^*A$ invertible; normal-equation coordinates identify the projection operator.\\
\hlt{Key Insight}: Full column rank turns the normal equations into an explicit projector.
\end{proof}
\end{corollary}

\begin{definition} \label{def:minimum-norm-solution} \hlt{Minimum-Norm Solution}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{y}\in\operatorname{col}(A)$. A vector $\mathbf{s}\in\mathbb{F}^n$ is a \hlt{minimum-norm solution} of $A\mathbf{x}=\mathbf{y}$ if:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\mathbf{s}=\mathbf{y}$;
\item $(\forall\,\mathbf{x}\in\mathbb{F}^n)\,[A\mathbf{x}=\mathbf{y}\Longrightarrow\norm{\mathbf{s}}_2\leq\norm{\mathbf{x}}_2]$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:minimum-norm-solution-consistent-system} \hlt{Minimum-Norm Solution of a Consistent Linear System}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, let $\mathbf{y}\in\mathbb{F}^m$, and suppose $A\mathbf{x}=\mathbf{y}$ is consistent. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\exists!\,\mathbf{s}\in\mathbb{F}^n)\,[A\mathbf{s}=\mathbf{y}\land(\forall\,\mathbf{x}\in\mathbb{F}^n)\,(A\mathbf{x}=\mathbf{y}\Longrightarrow\norm{\mathbf{s}}_2\leq\norm{\mathbf{x}}_2)]$;
\item for the vector $\mathbf{s}$ in (i), $\{\mathbf{x}\in\operatorname{col}(A^*):A\mathbf{x}=\mathbf{y}\}=\{\mathbf{s}\}$;
\item $(\exists\,\mathbf{u}\in\mathbb{F}^m)\,AA^*\mathbf{u}=\mathbf{y}$;
\item for the vector $\mathbf{s}$ in (i), $(\forall\,\mathbf{u}_0\in\mathbb{F}^m)\,[AA^*\mathbf{u}_0=\mathbf{y}\Longrightarrow\mathbf{s}=A^*\mathbf{u}_0]$.
\end{enumerate}
\begin{proof}
The identity $(\operatorname{col}(A^*),\ker(A))=((\ker(A))^{\perp},(\operatorname{col}(A^*))^{\perp})$ holds by Corollary \ref{cor:matrix-fundamental-subspaces-orthogonal-complements}(ii),(i).\\
Hence $\mathbb{F}^n=\operatorname{col}(A^*)\odot\ker(A)$ by Theorem \ref{thm:orthogonal-direct-sum-decomposition}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $\mathbf{x}_0\in\mathbb{F}^n$ with $A\mathbf{x}_0=\mathbf{y}$ by consistency.\\
Write $\mathbf{x}_0=\mathbf{s}+\mathbf{v}$ with $\mathbf{s}\in\operatorname{col}(A^*)$ and $\mathbf{v}\in\ker(A)$ by the established orthogonal direct sum. Then $A\mathbf{s}=\mathbf{y}$.\\
For $\mathbf{x}\in\mathbb{F}^n$ with $A\mathbf{x}=\mathbf{y}$, set $\mathbf{w}\defeq\mathbf{x}-\mathbf{s}$. Then $A\mathbf{w}=A\mathbf{x}-A\mathbf{s}=\mathbf{0}$, so $\mathbf{w}\in\ker(A)$.\\
Since $\mathbf{s}\perp\mathbf{w}$, $\norm{\mathbf{x}}_2^2=\norm{\mathbf{s}}_2^2+\norm{\mathbf{w}}_2^2\geq\norm{\mathbf{s}}_2^2$ by Theorem \ref{thm:pythagorean-inner-product}.\\
Equality holds iff $\mathbf{w}=\mathbf{0}$, equivalently $\mathbf{x}=\mathbf{s}$; hence $\mathbf{s}$ is the unique minimum-norm solution.
\item The vector $\mathbf{s}$ from (i) is a solution in $\operatorname{col}(A^*)$.\\
For any solution $\mathbf{s}'\in\operatorname{col}(A^*)$, $\mathbf{s}'-\mathbf{s}\in\ker(A)\cap\operatorname{col}(A^*)=\{\mathbf{0}\}$ by the established direct sum, so $\mathbf{s}'=\mathbf{s}$.
\item Choose $\mathbf{u}\in\mathbb{F}^m$ with $\mathbf{s}=A^*\mathbf{u}$ by $\mathbf{s}\in\operatorname{col}(A^*)$. Then $AA^*\mathbf{u}=A\mathbf{s}=\mathbf{y}$.
\item If $AA^*\mathbf{u}_0=\mathbf{y}$, then $A^*\mathbf{u}_0$ is a solution in $\operatorname{col}(A^*)$, so $A^*\mathbf{u}_0=\mathbf{s}$ by (ii).
\end{enumerate}
\hlt{Structural Commentary}: Orthogonal splitting removes the kernel component; the adjoint range supplies the $AA^*$ representation.\\
\hlt{Key Insight}: $\operatorname{col}(A^*)=(\ker(A))^{\perp}$ contains the unique shortest feasible vector.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:least-squares-pseudoinverse-roadmap} \hlt{Roadmap to the Pseudoinverse}\\
Theorem \ref{thm:matrix-least-squares-normal-equations} identifies the fitted vector with the projection of $\mathbf{y}$ onto $\operatorname{col}(A)$.\\
Proposition \ref{prop:least-squares-affine-set-and-minimum-norm} selects its minimum-norm preimage in $(\ker(A))^{\perp}=\operatorname{col}(A^*)$ by Theorem \ref{thm:minimum-norm-solution-consistent-system}.\\
Later, Definition \ref{def:moore-penrose-pseudoinverse} and Theorem \ref{thm:pseudoinverse-minimum-norm-least-squares} package both choices as $A^{\dagger}\mathbf{y}$.
\end{perspective}
