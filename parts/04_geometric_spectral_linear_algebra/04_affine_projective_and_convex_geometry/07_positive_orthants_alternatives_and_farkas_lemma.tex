\subsection{Positive Orthants, Alternatives, and Farkas's Lemma}

This subsection uses finite-dimensional real inner-product geometry to decide when linear systems have positive solutions. The guiding theme is that a missing positive solution is often detected by a separating hyperplane, hence by a dual vector certificate.

\begin{definition} \label{def:positive-vector-orders-orthants} \hlt{Positive Orthants}\\
Let $n\in\mathbb{Z}_{>0}$ and $\mathbf{x}=(x_1,\ldots,x_n)\in\R^n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The vector $\mathbf{x}$ is \hlt{coordinatewise nonnegative}, written $\mathbf{x}\geq\mathbf{0}$, iff $(\forall\,i\in\{1,\ldots,n\})\,x_i\geq0$. Set $\mathbf{x}\leq\mathbf{0}$ iff $-\mathbf{x}\geq\mathbf{0}$.
\item The vector $\mathbf{x}$ is \hlt{nonzero nonnegative}, written $\mathbf{x}>\mathbf{0}$, iff $\mathbf{x}\geq\mathbf{0}$ and $\mathbf{x}\neq\mathbf{0}$.
\item The vector $\mathbf{x}$ is \hlt{coordinatewise positive}, written $\mathbf{x}\gg\mathbf{0}$, iff $(\forall\,i\in\{1,\ldots,n\})\,x_i>0$.
\item $\R_{\geq0}^n\defeq\{\mathbf{x}\in\R^n:\mathbf{x}\geq\mathbf{0}\}$ is the \hlt{nonnegative orthant}.
\item $\R_+^n\defeq\{\mathbf{x}\in\R^n:\mathbf{x}>\mathbf{0}\}=\R_{\geq0}^n\setminus\{\mathbf{0}\}$ is the \hlt{nonzero nonnegative orthant}.
\item $\R_{++}^n\defeq\{\mathbf{x}\in\R^n:\mathbf{x}\gg\mathbf{0}\}$ is the \hlt{coordinatewise positive orthant}.
\end{enumerate}
\end{definition}

\begin{motivation} \label{mot:positive-solutions-probability-distributions} \hlt{Positive Null Vectors as Probabilities}\\
Let $m,n\in\mathbb{Z}_{>0}$, $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(m,n,\R)$, and $\mathbf{x}=(x_1,\ldots,x_n)>\mathbf{0}$ satisfy $A\mathbf{x}=\mathbf{0}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $s\defeq x_1+\cdots+x_n>0$ and $\boldsymbol{\pi}\defeq s^{-1}\mathbf{x}$.
\item The relations $(\forall\,i\in\{1,\ldots,n\})\,\pi_i\geq0$ and $\sum\limits_{i=1}^{n}\pi_i=1$ make $\boldsymbol{\pi}$ a probability vector.
\item The identity $A\boldsymbol{\pi}=s^{-1}A\mathbf{x}=\mathbf{0}$ is equivalent to $\sum\limits_{i=1}^{n}\pi_i\mathbf{a}_i=\mathbf{0}$.
\item The stronger condition $\mathbf{x}\gg\mathbf{0}$ implies $(\forall\,i\in\{1,\ldots,n\})\,\pi_i>0$.
\end{enumerate}
Thus a positive null vector normalizes to probabilities under which the columns of $A$ have zero expected value.
\end{motivation}

\begin{principle} \label{pri:positive-system-row-space-view} \hlt{Positive Solutions Are Orthant Intersections}\\
Let $m,n\in\mathbb{Z}_{>0}$, $A\in\operatorname{Mat}(m,n,\R)$, and $S\defeq\operatorname{row}(A)$. Then $\ker(A)=S^\perp$ by Corollary \ref{cor:matrix-fundamental-subspaces-orthogonal-complements}(i).
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\{\mathbf{x}\in\R^n:A\mathbf{x}=\mathbf{0},\ \mathbf{x}>\mathbf{0}\}=S^\perp\cap\R_+^n$.
\item $\{\mathbf{x}\in\R^n:A\mathbf{x}=\mathbf{0},\ \mathbf{x}\gg\mathbf{0}\}=S^\perp\cap\R_{++}^n$.
\end{enumerate}
Thus \hlt{positive null vectors} are precisely orthant intersections with $\operatorname{row}(A)^\perp$.
\end{principle}

\begin{theorem} \label{thm:positive-subspace-alternatives} \hlt{Positive Orthant Alternatives for Subspaces}\\
Let $n\in\mathbb{Z}_{>0}$ and $S\leq\R^n$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $S\cap\R_+^n=\emptyset \Longleftrightarrow S^\perp\cap\R_{++}^n\neq\emptyset$.
\item $S\cap\R_{++}^n=\emptyset \Longleftrightarrow S^\perp\cap\R_+^n\neq\emptyset$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Put $\Delta\defeq\operatorname{conv}\{\mathbf{e}_1,\ldots,\mathbf{e}_n\}$. Then $\mathbf{e}_1\in\Delta$ and $\Delta\subseteq\R_+^n$ by Proposition \ref{prop:convex-hull-finite-set-combinations-compact}(i), and $\Delta$ is compact by Proposition \ref{prop:convex-hull-finite-set-combinations-compact}(ii).\\
If $\mathbf{x},\mathbf{y}\in\Delta$ and $t\in[0,1]$, choose $\mathbf{r},\mathbf{s}\in\Delta_{n-1}$ representing $\mathbf{x},\mathbf{y}$ by Proposition \ref{prop:convex-hull-finite-set-combinations-compact}(i).\\
Then $t\mathbf{r}+(1-t)\mathbf{s}\in\Delta_{n-1}$ directly, so $t\mathbf{x}+(1-t)\mathbf{y}\in\Delta$ by the same proposition. Thus $\Delta$ is convex. The hypothesis implies $\Delta\cap S=\emptyset$. Choose $\mathbf{N}\in S^\perp\setminus\{\mathbf{0}\}$ satisfying $(\forall\,i\in\{1,\ldots,n\})\,N_i=\innerprod{\mathbf{N},\mathbf{e}_i}\geq\norm{\mathbf{N}}^2>0$ by Theorem \ref{thm:compact-convex-set-separated-from-subspace}(i). Hence $\mathbf{N}\in S^\perp\cap\R_{++}^n$.
\item[$(i)$] ($\Leftarrow$) Suppose $\mathbf{N}\in S^\perp\cap\R_{++}^n$. If $\mathbf{x}\in S\cap\R_+^n$, then $0=\innerprod{\mathbf{N},\mathbf{x}}=\sum\limits_{i=1}^{n}N_ix_i>0$ as $\mathbf{x}\geq\mathbf{0}$ nonzero, $\mathbf{N}\gg\mathbf{0}$, contradiction. QED.
\item[$(ii)$] ($\Rightarrow$) Suppose $S\cap\R_{++}^n=\emptyset$. If $S^\perp\cap\R_+^n=\emptyset$, then $(S^\perp)^\perp\cap\R_{++}^n\neq\emptyset$ by (i) with $S^\perp$. Contradicts $(S^\perp)^\perp=S$ by Corollary \ref{cor:double-orthogonal-complement}.
\item[$(ii)$] ($\Leftarrow$) Suppose $\mathbf{N}\in S^\perp\cap\R_+^n$. If $\mathbf{x}\in S\cap\R_{++}^n$, then $0=\innerprod{\mathbf{N},\mathbf{x}}=\sum\limits_{i=1}^{n}N_ix_i>0$ as $\mathbf{N}\geq\mathbf{0}$ nonzero, $\mathbf{x}\gg\mathbf{0}$, contradiction. QED.
\end{enumerate}
\hlt{Structural Commentary}: The simplex proves (i); orthogonal-complement involution transfers it to (ii).\\
\hlt{Key Insight}: Positive avoidance and positive certificates are orthogonal duals.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:homogeneous-positive-solution-alternatives} \hlt{Homogeneous Positive Alternatives}\\
Let $m,n\in\mathbb{Z}_{>0}$ and $A\in\operatorname{Mat}(m,n,\R)$. Exactly one alternative holds in each pair:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\ker(A)\cap\R_{++}^n\neq\emptyset$ or $\operatorname{col}(A^t)\cap\R_+^n\neq\emptyset$.
\item $\ker(A)\cap\R_+^n\neq\emptyset$ or $\operatorname{col}(A^t)\cap\R_{++}^n\neq\emptyset$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $S\defeq\operatorname{row}(A)=\operatorname{col}(A^t)\subseteq\R^n$. Then $\ker(A)=S^\perp$ by Corollary \ref{cor:matrix-fundamental-subspaces-orthogonal-complements}(i).\\
Hence the first pair is equivalent to $S^\perp\cap\R_{++}^n\neq\emptyset$ or $S\cap\R_+^n\neq\emptyset$, and exactly one holds by Theorem \ref{thm:positive-subspace-alternatives}(i).
\item With the same $S$, the second pair is equivalent to $S^\perp\cap\R_+^n\neq\emptyset$ or $S\cap\R_{++}^n\neq\emptyset$, and exactly one holds by Theorem \ref{thm:positive-subspace-alternatives}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Row-space orthogonality transports both subspace alternatives to matrix systems.\\
\hlt{Key Insight}: Positive solvability is an orthogonal-complement alternative.
\end{proof}
\end{theorem}

\begin{method} \label{meth:positive-solution-certificates} \hlt{Certifying Positive Homogeneous Systems}\\
Let $m,n\in\mathbb{Z}_{>0}$, $A\in\operatorname{Mat}(m,n,\R)$, and $S\defeq\operatorname{row}(A)=\operatorname{col}(A^t)$. The following certificate tests are exact by Theorem \ref{thm:homogeneous-positive-solution-alternatives}:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\ker(A)\cap\R_{++}^n=\emptyset\Longleftrightarrow(\exists\,\mathbf{v}\in\R^m)\,A^t\mathbf{v}>\mathbf{0}$.
\item $\ker(A)\cap\R_+^n=\emptyset\Longleftrightarrow(\exists\,\mathbf{v}\in\R^m)\,A^t\mathbf{v}\gg\mathbf{0}$.
\item $S\cap\R_+^n=\emptyset\Longleftrightarrow\ker(A)\cap\R_{++}^n\neq\emptyset$, and $S\cap\R_{++}^n=\emptyset\Longleftrightarrow\ker(A)\cap\R_+^n\neq\emptyset$.
\end{enumerate}
\end{method}

\begin{lemma} \label{lem:finitely-generated-nonnegative-cone-closed-convex} \hlt{Nonnegative Matrix Cone}\\
Let $m,n\in\mathbb{Z}_{>0}$, $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(m,n,\R)$, and $C_A\defeq A(\R_{\geq0}^n)=\left\{\sum\limits_{i=1}^{n}t_i\mathbf{a}_i:(t_1,\ldots,t_n)\in\R_{\geq0}^n\right\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $C_A$ is a convex cone in $\R^m$.
\item $C_A$ is closed in $\R^m$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\mathbf{x}_r=A\mathbf{y}_r\in C_A$ with $\mathbf{y}_r\geq\mathbf{0}$ for $r\in\{1,2\}$.\\
For $t\in[0,1]$ and $a\in[0,\infty)$, $t\mathbf{x}_1+(1-t)\mathbf{x}_2=A(t\mathbf{y}_1+(1-t)\mathbf{y}_2)\in C_A$ and $a\mathbf{x}_1=A(a\mathbf{y}_1)\in C_A$.
\item
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Independent supports}: Let $\mathcal{I}$ be family of $I\subseteq\{1,\ldots,n\}$ for which $\{\mathbf{a}_i:i\in I\}$ is linearly independent, including $\emptyset$, and set $K_I\defeq\left\{\sum\limits_{i\in I}t_i\mathbf{a}_i:(\forall\,i\in I)\,t_i\geq0\right\}$. Fix $\mathbf{x}\in C_A\setminus\{\mathbf{0}\}$, choose representation $\mathbf{x}=\sum\limits_{i\in I}t_i\mathbf{a}_i$ with $(\forall\,i\in I)\,t_i>0$, minimal $\abs{I}$.\\
If the active columns are dependent, choose $(d_i)_{i\in I}\in\R^I\setminus\{\mathbf{0}\}$ with $\sum\limits_{i\in I}d_i\mathbf{a}_i=\mathbf{0}$.\\
After changing all signs if necessary, $J\defeq\{i\in I:d_i<0\}\neq\emptyset$. Put $\lambda\defeq\min\limits_{i\in J}\frac{t_i}{-d_i}>0$.\\
Then $(\forall\,i\in I)\,t_i+\lambda d_i\geq0$ and $(\exists\,i\in J)\,t_i+\lambda d_i=0$, whereas $\mathbf{x}=\sum\limits_{i\in I}(t_i+\lambda d_i)\mathbf{a}_i$.\\
This contradicts minimality, so $I\in\mathcal{I}$. Since $\mathbf{0}\in K_{\emptyset}$, $C_A=\bigcup\limits_{I\in\mathcal{I}}K_I$.
\item \hlt{Closed simplicial cones}: Fix $I\in\mathcal{I}$. If $I=\emptyset$, then $K_I=\{\mathbf{0}\}$; and for $\mathbf{x}\neq\mathbf{0}$, $B_{\frac{1}{2}\norm{\mathbf{x}}}(\mathbf{x})\subseteq\R^m\setminus K_I$ by Proposition \ref{prop:reverse-triangle-inequality-inner-product}.\\
Suppose $I\neq\emptyset$ and put $U_I\defeq\operatorname{span}\{\mathbf{a}_i:i\in I\}$. Let $(\varphi_i)_{i\in I}$ be the dual basis of $(\mathbf{a}_i)_{i\in I}$, which exists by Proposition \ref{prop:linear-map-determined-by-basis}.\\
Choose $(\mathbf{d}_i)_{i\in I}\subseteq U_I$ by Theorem \ref{thm:riesz-representation-finite-dimensional}(i) such that $(\forall\,\mathbf{u}\in U_I)(\forall\,i\in I)\,\varphi_i(\mathbf{u})=\innerprod{\mathbf{u},\mathbf{d}_i}$.\\
Thus $K_I=\{\mathbf{u}\in U_I:(\forall\,i\in I)\,\innerprod{\mathbf{u},\mathbf{d}_i}\geq0\}$. The identity $\ker(P_{U_I^\perp})=U_I$ holds by Proposition \ref{prop:orthogonal-projection-properties}(vi) and Corollary \ref{cor:double-orthogonal-complement}.\\
Fix $\mathbf{x}\notin K_I$. If $\mathbf{x}\notin U_I$, put $\mathbf{q}\defeq P_{U_I^\perp}\mathbf{x}\neq\mathbf{0}$. For $\norm{\mathbf{y}-\mathbf{x}}<\frac{1}{2}\norm{\mathbf{q}}$, $\norm{P_{U_I^\perp}\mathbf{y}}\geq\norm{\mathbf{q}}-\norm{P_{U_I^\perp}(\mathbf{y}-\mathbf{x})}\geq\norm{\mathbf{q}}-\norm{\mathbf{y}-\mathbf{x}}>0$ by Proposition \ref{prop:reverse-triangle-inequality-inner-product} and Proposition \ref{prop:orthogonal-projection-properties}(i),(viii), so $\mathbf{y}\notin U_I$. If $\mathbf{x}\in U_I\setminus K_I$, choose $j\in I$ with $\innerprod{\mathbf{x},\mathbf{d}_j}<0$.\\
Since $\innerprod{\mathbf{a}_j,\mathbf{d}_j}=1$, $\mathbf{d}_j\neq\mathbf{0}$. If $\norm{\mathbf{y}-\mathbf{x}}<-\frac{\innerprod{\mathbf{x},\mathbf{d}_j}}{2\norm{\mathbf{d}_j}}$, then $\innerprod{\mathbf{y},\mathbf{d}_j}\leq\innerprod{\mathbf{x},\mathbf{d}_j}+\norm{\mathbf{y}-\mathbf{x}}\norm{\mathbf{d}_j}<\frac{1}{2}\innerprod{\mathbf{x},\mathbf{d}_j}<0$ by Theorem \ref{thm:cauchy-schwarz-inequality}(i), so $\mathbf{y}\notin K_I$. Hence $\R^m\setminus K_I$ is open, and $K_I$ is closed.
\item \hlt{Finite union}: Fix $\mathbf{x}\notin C_A$. Choose $(r_I)_{I\in\mathcal{I}}\subseteq(0,\infty)$ satisfying $(\forall\,I\in\mathcal{I})\,B_{r_I}(\mathbf{x})\subseteq\R^m\setminus K_I$ by (b).\\
Since $\mathcal{I}$ is finite and nonempty, $r\defeq\min\limits_{I\in\mathcal{I}}r_I>0$. Then $B_r(\mathbf{x})\subseteq\R^m\setminus C_A$ by (a). Thus $\R^m\setminus C_A$ is open, so $C_A$ is closed.
\end{enumerate}
\end{enumerate}
\hlt{Structural Commentary}: Minimal supports form a finite simplicial cover; dual coordinates prove each cone closed.\\
\hlt{Key Insight}: Independent generators turn nonnegativity into finitely many closed inequalities.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:farkas-lemma-strict-positive-system} \hlt{Farkas's Lemma}\\
Let $m,n\in\mathbb{Z}_{>0}$, $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(m,n,\R)$, $\mathbf{b}\in\R^m\setminus\{\mathbf{0}\}$, set $C_A\defeq\{A\mathbf{y}:\mathbf{y}\in\R^n,\ \mathbf{y}\geq\mathbf{0}\}$. Exactly one of following holds:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathbf{b}\in C_A$, equivalently $(\exists\,\mathbf{u}\in\R^n)\,[\mathbf{u}>\mathbf{0}\land A\mathbf{u}=\mathbf{b}]$.
\item $(\exists\,\mathbf{v}\in\R^m)\,[A^t\mathbf{v}\leq\mathbf{0}\land\innerprod{\mathbf{v},\mathbf{b}}>0]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The equivalence $\mathbf{b}\in C_A\Longleftrightarrow(\exists\,\mathbf{u}\in\R^n)\,[\mathbf{u}\geq\mathbf{0}\land A\mathbf{u}=\mathbf{b}]\Longleftrightarrow(\exists\,\mathbf{u}\in\R^n)\,[\mathbf{u}>\mathbf{0}\land A\mathbf{u}=\mathbf{b}]$ holds by the definition of $C_A$ and $\mathbf{b}\neq\mathbf{0}$. If (i) and (ii) both hold, choose witnesses $\mathbf{u}$ and $\mathbf{v}$.\\
Then $0<\innerprod{\mathbf{v},\mathbf{b}}=\innerprod{\mathbf{v},A\mathbf{u}}=\innerprod{A^t\mathbf{v},\mathbf{u}}=\sum\limits_{i=1}^{n}(A^t\mathbf{v})_iu_i\leq0$ by the coordinate inequalities, a contradiction.
\item Suppose $\mathbf{b}\notin C_A$. The set $C_A$ is nonempty because $\mathbf{0}\in C_A$, and it is closed and convex by Lemma \ref{lem:finitely-generated-nonnegative-cone-closed-convex}(i),(ii).\\
Choose $\mathbf{N}\neq\mathbf{0}$ satisfying $(\forall\,\mathbf{c}\in C_A)\,\innerprod{\mathbf{N},\mathbf{c}}\geq\norm{\mathbf{N}}^2+\innerprod{\mathbf{N},\mathbf{b}}$ by Theorem \ref{thm:nearest-point-and-point-separation-convex-set}(ii), and set $\mathbf{v}\defeq-\mathbf{N}$.\\
Then $(\forall\,\mathbf{c}\in C_A)\,\innerprod{\mathbf{v},\mathbf{c}}\leq\innerprod{\mathbf{v},\mathbf{b}}-\norm{\mathbf{v}}^2$. Taking $\mathbf{c}=\mathbf{0}$ proves $\innerprod{\mathbf{v},\mathbf{b}}\geq\norm{\mathbf{v}}^2>0$. Fix $i\in\{1,\ldots,n\}$.\\
Since $(\forall\,\lambda\geq0)\,\lambda\mathbf{a}_i\in C_A$, $(\forall\,\lambda\geq0)\,\lambda\innerprod{\mathbf{v},\mathbf{a}_i}\leq\innerprod{\mathbf{v},\mathbf{b}}-\norm{\mathbf{v}}^2$. If $\gamma\defeq\innerprod{\mathbf{v},\mathbf{a}_i}>0$, put $\lambda\defeq\frac{\innerprod{\mathbf{v},\mathbf{b}}-\norm{\mathbf{v}}^2}{\gamma}+1\geq1$.\\
Then $\lambda\gamma>\innerprod{\mathbf{v},\mathbf{b}}-\norm{\mathbf{v}}^2$, a contradiction. Thus $(\forall\,i\in\{1,\ldots,n\})\,\innerprod{\mathbf{v},\mathbf{a}_i}\leq0$, equivalently $A^t\mathbf{v}\leq\mathbf{0}$.\\
Therefore (ii) holds whenever (i) fails; together with (i), exactly one alternative holds.
\end{enumerate}
\hlt{Structural Commentary}: Nearest-point separation constructs a dual vector; cone scaling tests its coordinates.\\
\hlt{Key Insight}: A cone certificate is nonpositive on every generator.
\end{proof}
\end{theorem}

\begin{example} \label{ex:farkas-weak-inequality-necessary} \hlt{Why the Weak Inequality Matters}\\
The conclusion $A^t\mathbf{v}\leq\mathbf{0}$ cannot generally be strengthened to $-A^t\mathbf{v}\gg\mathbf{0}$. Let $A=\left(\begin{smallmatrix}1&0&0\\0&1&-1\end{smallmatrix}\right)$ and $\mathbf{b}=\left(\begin{smallmatrix}-1\\0\end{smallmatrix}\right)$.\\
If $\mathbf{u}\geq\mathbf{0}$, then $(A\mathbf{u})_1=u_1\geq0$, so $A\mathbf{u}\neq\mathbf{b}$. The vector $\mathbf{v}=\left(\begin{smallmatrix}-1\\0\end{smallmatrix}\right)$ satisfies $A^t\mathbf{v}=(-1,0,0)^t\leq\mathbf{0}$ and $\innerprod{\mathbf{v},\mathbf{b}}=1>0$.\\
For $\mathbf{w}=(r,s)^t\in\R^2$, $A^t\mathbf{w}=(r,s,-s)^t$, so $-A^t\mathbf{w}\gg\mathbf{0}$ would require both $s<0$ and $s>0$, impossible.\\
\hlt{Structural Commentary}: Opposite columns force a zero in every certificate.\\
\hlt{Key Insight}: Boundary separation is sometimes the best possible separation.
\end{example}
