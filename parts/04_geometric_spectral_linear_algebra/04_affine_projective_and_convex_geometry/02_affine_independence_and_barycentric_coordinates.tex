\subsection{Affine Independence and Barycentric Coordinates}

\begin{definition} \label{def:affinely-independent-set} \hlt{Affinely Independent Sets}\\
A nonempty set $E\subseteq\mathcal{V}$ is \hlt{affinely independent} if, for every $\mathbf{x}\in E$, the set $(E-\mathbf{x})\setminus\{\mathbf{0}\}$ is linearly independent.
\end{definition}

\begin{theorem} \label{thm:affine-independence-equivalences} \hlt{Affine Independence Criteria}\\
For $\emptyset\neq E\subseteq\mathcal{V}$, conditions (i)--(iv) are equivalent. If $E$ is finite, they are also equivalent to (v).
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E$ is affinely independent.
\item For every $\mathbf{x}\in E$, $\mathbf{x}\notin\operatorname{affhull}(E\setminus\{\mathbf{x}\})$, with $\operatorname{affhull}(\emptyset)=\emptyset$ in the singleton case.
\item $\forall$ distinct $\mathbf{x}_1,\ldots,\mathbf{x}_n\in E$ and $r_1,\ldots,r_n\in\mathbb{F}$, $\sum\limits_{j=1}^n r_j\mathbf{x}_j=\mathbf{0}$ and $\sum\limits_{j=1}^n r_j=0$ imply $r_1=\cdots=r_n=0$.
\item $\forall$ distinct $\mathbf{x}_1,\ldots,\mathbf{x}_n\in E$ and affine coefficient vectors $\mathbf{r},\mathbf{s}\in\mathbb{F}^n$,\\
$\sum\limits_{j=1}^n r_j\mathbf{x}_j=\sum\limits_{j=1}^n s_j\mathbf{x}_j$ and $\sum\limits_{j=1}^n r_j=\sum\limits_{j=1}^n s_j=1$ imply $\mathbf{r}=\mathbf{s}$.
\item If $E=\{\mathbf{x}_1,\ldots,\mathbf{x}_n\}$ is finite, then $\dim(\operatorname{affhull}(E))=n-1$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (iii)$ Let $\sum_{j=1}^n r_j\mathbf{x}_j=\mathbf{0}$ with distinct $\mathbf{x}_j\in E$ and $\sum_j r_j=0$. Then $\sum\limits_{j=1}^{n-1}r_j(\mathbf{x}_j-\mathbf{x}_n)=\mathbf{0}$.\\
By (i), $r_1=\cdots=r_{n-1}=0$, and then $r_n=0$.
\item[$(iii)$]$\Rightarrow (i)$ Fix $\mathbf{x}\in E$. Any relation $\sum\limits_{j=1}^m a_j(\mathbf{y}_j-\mathbf{x})=\mathbf{0}$, $\mathbf{y}_j\in E\setminus\{\mathbf{x}\}$ is the zero-sum relation\\
$\sum\limits_{j=1}^m a_j\mathbf{y}_j-(\sum_j a_j)\mathbf{x}=\mathbf{0}$, so (iii) gives $a_1=\cdots=a_m=0$.
\item[$(iii)$]$\Rightarrow (ii)$ If $\mathbf{x}_k\in\operatorname{affhull}(E\setminus\{\mathbf{x}_k\})$, then $\mathbf{x}_k=\sum_j r_j\mathbf{x}_j$ for $\mathbf{x}_j\neq\mathbf{x}_k$ and $\sum_j r_j=1$,\\
giving a nontrivial zero-sum relation, contradiction.
\item[$(ii)$]$\Rightarrow (iii)$ If a zero-sum relation is nontrivial, choose $r_k\neq0$.\\
Then $\mathbf{x}_k=\sum\limits_{j\neq k}\left(-\frac{r_j}{r_k}\right)\mathbf{x}_j$, $\sum\limits_{j\neq k}-\frac{r_j}{r_k}=1$, so $\mathbf{x}_k\in\operatorname{affhull}(E\setminus\{\mathbf{x}_k\})$, contradicting (ii).
\item[$(iii)$]$\Rightarrow (iv)$ Subtract the two affine representations; the coefficient vector $\mathbf{r}-\mathbf{s}$ has sum $0$, so (iii) gives $\mathbf{r}=\mathbf{s}$.
\item[$(iv)$]$\Rightarrow (iii)$ Let $\sum_j r_j\mathbf{x}_j=\mathbf{0}$ and $\sum_j r_j=0$. Choose $\mathbf{a}\in\mathbb{F}^n$ with $\sum_j a_j=1$.\\
Then $\mathbf{a}$ and $\mathbf{a}+\mathbf{r}$ are affine coefficient vectors for the same point. By (iv), $\mathbf{r}=\mathbf{0}$.
\item[$(i)$]$\Leftrightarrow (v)$ If $E=\{\mathbf{x}_1,\ldots,\mathbf{x}_n\}$ and $\mathbf{x}\in E$, then Theorem \ref{thm:affine-hull-span-formula} gives $\dim(\operatorname{affhull}(E))=\dim(\operatorname{span}(E-\mathbf{x}))$.\\
Since $(E-\mathbf{x})\setminus\{\mathbf{0}\}$ has $n-1$ vectors, this dimension is $n-1$ iff those vectors are linearly independent.
\end{enumerate}
\hlt{Structural Commentary}: Affine independence is linear independence after forgetting one chosen origin.\\
\hlt{Key Insight}: Uniqueness is absence of nontrivial zero-sum affine relations.
\end{proof}
\end{theorem}

\begin{definition} \label{def:affine-basis-barycentric-coordinates} \hlt{Affine Bases and Barycentric Coordinates}
Let $A=\mathbf{x}+\mathcal{S}$ be an $n$-flat.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A subset $\mathcal{B}\subseteq A$ is an \hlt{affine basis} of $A$ if $\mathcal{B}$ is affinely independent and $\operatorname{affspan}(\mathcal{B})=A$; an \hlt{ordered affine basis} is an ordering of such a basis.
\item If $(\mathbf{s}_1,\ldots,\mathbf{s}_n)$ is an ordered basis of $\mathcal{S}$, then $(\mathbf{x}+\mathbf{s}_1,\ldots,\mathbf{x}+\mathbf{s}_n,\mathbf{x})$ is an ordered affine basis of $A$.
\item For an ordered affine basis $(\mathbf{p}_1,\ldots,\mathbf{p}_{n+1})$, every $\mathbf{v}\in A$ has a unique expression $\mathbf{v}=\sum\limits_{j=1}^{n+1}r_j\mathbf{p}_j$, $\sum\limits_{j=1}^{n+1}r_j=1$.\\
The scalars $(r_1,\ldots,r_{n+1})$ are the \hlt{barycentric coordinates} of $\mathbf{v}$ in this basis.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:barycentric-coordinates-on-plane} \hlt{Barycentric Coordinates on a Plane}\\
Let $A=\mathbf{x}+\operatorname{span}(\mathbf{v}_1,\mathbf{v}_2)\subseteq\R^3$, with $(\mathbf{v}_1,\mathbf{v}_2)$ linearly independent.\\
Set $\mathbf{p}_1=\mathbf{x}+\mathbf{v}_1$, $\mathbf{p}_2=\mathbf{x}+\mathbf{v}_2$, and $\mathbf{p}_3=\mathbf{x}$. Then $(\mathbf{p}_1,\mathbf{p}_2,\mathbf{p}_3)$ is an ordered affine basis of $A$, and every $\mathbf{a}\in A$ has a unique barycentric expansion $\mathbf{a}=r_1\mathbf{p}_1+r_2\mathbf{p}_2+r_3\mathbf{p}_3$, $r_1+r_2+r_3=1$, with $r_1,r_2,r_3\in\R$.
\end{example}
