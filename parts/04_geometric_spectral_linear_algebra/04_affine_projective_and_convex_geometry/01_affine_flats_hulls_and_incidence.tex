\subsection{Affine Flats, Hulls, and Incidence}

% **Role:** incidence geometry of linear spaces without a distinguished origin.
%
% Affine flats, affine combinations, affine hulls,
% flat lattices, affine independence, barycentric coordinates, affine maps,
% the projective completion viewpoint.

Affine geometry studies cosets of subspaces as geometric objects in their own right. Throughout this subsection, vector spaces are finite-dimensional over $\mathbb{F}$ unless explicitly stated otherwise.

\begin{definition} \label{def:flat-affine-geometry} \hlt{Flats}
Let $\mathcal{S}\leq\mathcal{V}$ and $\mathbf{x}\in\mathcal{V}$, and set $\mathbf{x}+\mathcal{S}=\{\mathbf{x}+\mathbf{s}:\mathbf{s}\in\mathcal{S}\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The affine subset $\mathbf{x}+\mathcal{S}$ from Definition \ref{def:affine-subset-parallel-subspace} is a \hlt{flat} in $\mathcal{V}$; $\mathcal{S}$ is its \hlt{base}, and $\mathbf{x}$ is a \hlt{flat representative}.
\item Let $\mathcal{A}(\mathcal{V})$ denote the set of all flats in $\mathcal{V}$, with $\dim(\mathcal{A}(\mathcal{V}))\defeq\dim(\mathcal{V})$.
\item The \hlt{dimension} of $\mathbf{x}+\mathcal{S}$ is $\dim(\mathcal{S})$.\\
A $0$-flat is a \hlt{point}, a $1$-flat is a \hlt{line}, a $2$-flat is a \hlt{plane}, and a codimension-one flat is a \hlt{hyperplane}.
\item Two flats $X=\mathbf{x}+\mathcal{S}$ and $Y=\mathbf{y}+\mathcal{T}$ are \hlt{parallel}, written $X\parallel Y$, if $\mathcal{S}\subseteq\mathcal{T}$ or $\mathcal{T}\subseteq\mathcal{S}$.
\end{enumerate}
\hlt{Structural Commentary}: A flat has many representatives but one base: $\mathbf{x}+\mathcal{S}=\mathbf{y}+\mathcal{T}\Rightarrow\mathcal{S}=\mathcal{T}$.\\
\end{definition}

\begin{proposition} \label{prop:basic-flat-incidence} \hlt{Basic Incidence of Flats}\\
Let $X=\mathbf{x}+\mathcal{S}$ and $Y=\mathbf{y}+\mathcal{T}$ be flats in $\mathcal{V}$, and write $\mathbf{w}+X\defeq\{\mathbf{w}+\mathbf{u}:\mathbf{u}\in X\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\exists\,\mathbf{w}\in\mathcal{V})(\mathbf{w}+X\subseteq Y)\iff\mathcal{S}\subseteq\mathcal{T}$.
\item $(\exists\,\mathbf{w}\in\mathcal{V})(\mathbf{w}+X=Y)\iff\mathcal{S}=\mathcal{T}$.
\item If $X\cap Y\neq\emptyset$, then $X\subseteq Y\iff\mathcal{S}\subseteq\mathcal{T}$.
\item If $X\cap Y\neq\emptyset$, then $X=Y\iff\mathcal{S}=\mathcal{T}$.
\item If $X\parallel Y$, then $X\subseteq Y$, or $Y\subseteq X$, or $X\cap Y=\emptyset$.
\item $X\parallel Y\iff[(\exists\,\mathbf{w}\in\mathcal{V})(\mathbf{w}+X\subseteq Y)]\lor[(\exists\,\mathbf{w}\in\mathcal{V})(\mathbf{w}+Y\subseteq X)]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $\mathbf{w}+X=(\mathbf{w}+\mathbf{x})+\mathcal{S}$, every translate of $X$ has base $\mathcal{S}$. If $\mathbf{w}+X\subseteq Y$, choose $\mathbf{z}\in\mathbf{w}+X$; then $\mathbf{w}+X=\mathbf{z}+\mathcal{S}$ and $Y=\mathbf{z}+\mathcal{T}$, hence $\mathcal{S}\subseteq\mathcal{T}$. Conversely, if $\mathcal{S}\subseteq\mathcal{T}$, then $(\mathbf{y}-\mathbf{x})+X=\mathbf{y}+\mathcal{S}\subseteq Y$.
\item If $\mathbf{w}+X=Y$, then the bases are equal. Conversely, if $\mathcal{S}=\mathcal{T}$, then $(\mathbf{y}-\mathbf{x})+X=Y$.
\item Choose $\mathbf{z}\in X\cap Y$. Then $X=\mathbf{z}+\mathcal{S}$ and $Y=\mathbf{z}+\mathcal{T}$, so translating by $-\mathbf{z}$ gives $X\subseteq Y\iff\mathcal{S}\subseteq\mathcal{T}$.
\item With the same $\mathbf{z}$, translate by $-\mathbf{z}$ to get $X=Y\iff\mathcal{S}=\mathcal{T}$.
\item If $X\parallel Y$, then $\mathcal{S}\subseteq\mathcal{T}$ or $\mathcal{T}\subseteq\mathcal{S}$. If $X\cap Y\neq\emptyset$, part (iii) gives $X\subseteq Y$ or $Y\subseteq X$; otherwise $X\cap Y=\emptyset$.
\item If $X\parallel Y$, then either $\mathcal{S}\subseteq\mathcal{T}$ or $\mathcal{T}\subseteq\mathcal{S}$, so part (i) gives a translate $X'\subseteq Y$ or a translate $Y'\subseteq X$.\\
Conversely, either containment gives $\mathcal{S}\subseteq\mathcal{T}$ or $\mathcal{T}\subseteq\mathcal{S}$ by part (i), hence $X\parallel Y$.
\end{enumerate}
\hlt{Structural Commentary}: Intersecting flats reduce affine incidence to subspace incidence.\\
\hlt{Key Insight}: The representative may move, but the base controls containment, equality, and parallelism.
\end{proof}
\end{proposition}

\begin{definition} \label{def:affine-combination-and-closed-set} \hlt{Affine Combinations}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{x}_1,\ldots,\mathbf{x}_n\in\mathcal{V}$ and $r_1,\ldots,r_n\in\mathbb{F}$, the sum $\sum_{j=1}^n r_j\mathbf{x}_j$ is an \hlt{affine combination} if $\sum_{j=1}^n r_j=1$.
\item A nonempty $E\subseteq\mathcal{V}$ is \hlt{affine closed} if $\sum_{j=1}^n r_j\mathbf{x}_j\in E$ whenever $\mathbf{x}_j\in E$, $r_j\in\mathbb{F}$, and $\sum_{j=1}^n r_j=1$.
\item A nonempty $E\subseteq\mathcal{V}$ is \hlt{two-affine closed} if $r\mathbf{x}+(1-r)\mathbf{y}\in E$ for all $\mathbf{x},\mathbf{y}\in E$ and $r\in\mathbb{F}$.
\item For $\mathbf{x}\neq\mathbf{y}$, the affine line through them is $\overline{\mathbf{x}\mathbf{y}}\defeq\{r\mathbf{x}+(1-r)\mathbf{y}:r\in\mathbb{F}\} =\mathbf{y}+\operatorname{span}(\mathbf{x}-\mathbf{y})$.\\
Equivalently, a nonempty $E\subseteq\mathcal{V}$ is two-affine closed iff $\overline{\mathbf{x}\mathbf{y}}\subseteq E$ for all distinct $\mathbf{x},\mathbf{y}\in E$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:affine-closed-two-affine-closed} \hlt{Two-Point Affine Closure}\\
Assume $\operatorname{char}(\mathbb{F})\neq2$. For every $\emptyset\neq E\subseteq\mathcal{V}$, $E$ is affine closed $\iff$ $E$ is two-affine closed.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] A two-point affine combination is an affine combination, so affine closure implies two-affine closure.
\item[$(\Leftarrow)$] Assume $E$ is two-affine closed.\\
Prove by induction on $n$ that every affine combination $\sum\limits_{j=1}^n r_j\mathbf{x}_j$, with $\mathbf{x}_j\in E$ and $\sum\limits_{j=1}^n r_j=1$, lies in $E$. The cases $n=1,2$ are immediate. Let $n\geq3$ and set $\mathbf{z}\defeq\sum\limits_{j=1}^n r_j\mathbf{x}_j$, $\sum\limits_{j=1}^n r_j=1$.\\
If some $r_j\neq1$, relabel so $r_1\neq1$ and put $\mathbf{u}\defeq\sum\limits_{j=2}^n\frac{r_j}{1-r_1}\mathbf{x}_j$. Then $\sum\limits_{j=2}^n \frac{r_j}{1-r_1}=1$, so $\mathbf{u}\in E$ by induction, and $\mathbf{z}=r_1\mathbf{x}_1+(1-r_1)\mathbf{u}\in E$ by two-affine closure.\\
Otherwise all $r_j=1$; since $2$ is invertible, $\mathbf{z} =2\left(\frac12\mathbf{x}_1+\frac12\mathbf{x}_2\right)+r_3\mathbf{x}_3+\cdots+r_n\mathbf{x}_n$, where $\frac12\mathbf{x}_1+\frac12\mathbf{x}_2\in E$ by two-affine closure; this is an $(n-1)$-term affine combination, hence lies in $E$ by induction.
\end{enumerate}
\hlt{Structural Commentary}: For $\operatorname{char}(\mathbb{F})\neq2$, two-point closure generates finite affine closure.\\
\hlt{Key Insight}: The only exceptional induction case is handled by averaging two coefficient-$1$ terms.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:characteristic-two-affine-closure-warning} \hlt{Characteristic Two Warning}\\
Condition $\operatorname{char}(\mathbb{F})\neq2$ is necessary: in $\mathbb{F}_2^2$, $E\defeq\{(0,0),(1,0),(0,1)\}$ is two-affine closed, but $\operatorname{aff}(E)=\mathbb{F}_2^2\neq E$.
\end{remark}

\begin{theorem} \label{thm:flats-are-affine-closed} \hlt{Flats Are the Affine-Closed Sets}\\
For every $\emptyset\neq E\subseteq\mathcal{V}$, $E\in\mathcal{A}(\mathcal{V})$ $\iff$ $E$ is affine closed.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Suppose $E=\mathbf{x}+\mathcal{S}$ with $\mathcal{S}\leq\mathcal{V}$. If $\mathbf{x}_j=\mathbf{x}+\mathbf{s}_j\in E$ and $\sum_j r_j=1$,\\
then $\sum_j r_j\mathbf{x}_j =\sum_j r_j(\mathbf{x}+\mathbf{s}_j) =\mathbf{x}+\sum_j r_j\mathbf{s}_j\in\mathbf{x}+\mathcal{S}$. Thus $E$ is affine closed.
\item Suppose $E$ is affine closed. Choose $\mathbf{x}\in E$ and set $\mathcal{S}\defeq E-\mathbf{x}=\{\mathbf{e}-\mathbf{x}:\mathbf{e}\in E\}$.\\
For $\mathbf{s}_1,\mathbf{s}_2\in\mathcal{S}$, write $\mathbf{s}_j=\mathbf{x}_j-\mathbf{x}$ with $\mathbf{x}_j\in E$.\\
For $r_1,r_2\in\mathbb{F}$, $r_1\mathbf{s}_1+r_2\mathbf{s}_2+\mathbf{x} =r_1\mathbf{x}_1+r_2\mathbf{x}_2+(1-r_1-r_2)\mathbf{x}\in E$, so $r_1\mathbf{s}_1+r_2\mathbf{s}_2\in\mathcal{S}$.\\
Hence $\mathcal{S}\leq\mathcal{V}$ by Theorem \ref{thm:subspacetest}, and $E=\mathbf{x}+\mathcal{S}$ is a flat.
\end{enumerate}
\hlt{Structural Commentary}: An affine-closed set becomes a subspace after choosing one point as an origin.\\
\hlt{Key Insight}: Flats are precisely translated subspaces.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:flats-are-two-affine-closed} \hlt{Flats Are the Two-Affine-Closed Sets}\\
Assume $\operatorname{char}(\mathbb{F})\neq2$. For every $\emptyset\neq E\subseteq\mathcal{V}$, $E\in\mathcal{A}(\mathcal{V})$ $\iff$ $E$ is two-affine closed.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $E\in\mathcal{A}(\mathcal{V})$, then $E$ is affine closed by Theorem \ref{thm:flats-are-affine-closed}, hence two-affine closed by Theorem \ref{thm:affine-closed-two-affine-closed}.
\item If $E$ is two-affine closed, then $E$ is affine closed by Theorem \ref{thm:affine-closed-two-affine-closed}, hence $E\in\mathcal{A}(\mathcal{V})$ by Theorem \ref{thm:flats-are-affine-closed}.
\end{enumerate}
\hlt{Structural Commentary}: In characteristic not $2$, two-point closure detects flats.\\
\hlt{Key Insight}: Two-affine closure equals finite affine closure, hence flatness.
\end{proof}
\end{theorem}

\begin{definition} \label{def:affine-hull-and-affine-span} \hlt{Affine Hull and Affine Span}\\
Let $\emptyset\neq E\subseteq\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{affine hull} $\operatorname{affhull}(E)$ is the least flat $X\in\mathcal{A}(\mathcal{V})$ with $E\subseteq X$.
\item The \hlt{affine span} $\operatorname{affspan}(E)$ is the set of all finite affine combinations of points of $E$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:affine-hull-span-formula} \hlt{Affine Hull Formula}\\
Let $\emptyset\neq E\subseteq\mathcal{V}$, $\mathbf{x}\in E$, and $\mathcal{S}_{\mathbf{x}}\defeq\operatorname{span}(E-\mathbf{x})$.\\
Then $\operatorname{affhull}(E)=\operatorname{affspan}(E) =\mathbf{x}+\mathcal{S}_{\mathbf{x}}$, and hence $\dim(\operatorname{affhull}(E)) =\dim(\mathcal{S}_{\mathbf{x}})$.
\begin{proof}
If $\mathbf{z}=\sum\limits_{j=1}^n r_j\mathbf{e}_j$ with $\mathbf{e}_j\in E$ and $\sum\limits_j r_j=1$, then $\mathbf{z} =\mathbf{x}+\sum\limits_{j=1}^n r_j(\mathbf{e}_j-\mathbf{x})\in\mathbf{x}+\mathcal{S}_{\mathbf{x}}$, so $\operatorname{affspan}(E)\subseteq\mathbf{x}+\mathcal{S}_{\mathbf{x}}$.\\
Conversely, for $\sum\limits_{j=1}^n a_j(\mathbf{e}_j-\mathbf{x})\in\mathcal{S}_{\mathbf{x}}$, $\mathbf{x}+\sum\limits_{j=1}^n a_j(\mathbf{e}_j-\mathbf{x}) =\left(1-\sum\limits_{j=1}^n a_j\right)\mathbf{x}+\sum_{j=1}^n a_j\mathbf{e}_j\in\operatorname{affspan}(E)$, hence $\operatorname{affspan}(E)=\mathbf{x}+\mathcal{S}_{\mathbf{x}}$. This set is a flat containing $E$; any flat containing $E$ is affine closed by Theorem \ref{thm:flats-are-affine-closed}, hence contains $\operatorname{affspan}(E)$. Therefore it is $\operatorname{affhull}(E)$. The dimension formula follows from Definition \ref{def:flat-affine-geometry}.\\
\hlt{Structural Commentary}: Affine span is ordinary span after temporarily choosing one point as the origin.\\
\hlt{Key Insight}: Differences from a base point record all directions in the affine hull.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:intersection-of-flats} \hlt{Intersections of Flats}\\
Let $X_i=\mathbf{x}_i+\mathcal{S}_i$ be flats indexed by $i\in I$. If $\bigcap\limits_{i\in I}X_i\neq\emptyset$ and $\mathbf{z}\in\bigcap\limits_{i\in I}X_i$, then $\bigcap\limits_{i\in I}X_i=\mathbf{z}+\bigcap\limits_{i\in I}\mathcal{S}_i$.
\begin{proof}
Since $\mathbf{z}\in X_i$, each flat has the form $X_i=\mathbf{z}+\mathcal{S}_i$. Hence $\mathbf{u}\in\bigcap\limits_{i\in I}X_i$ $\iff$ $\mathbf{u}-\mathbf{z}\in\bigcap\limits_{i\in I}\mathcal{S}_i$, QED.\\
\hlt{Structural Commentary}: A common point turns affine intersection into subspace intersection.\\
\hlt{Key Insight}: Translate to the common point; only the bases remain.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:lattice-of-flats} \hlt{Lattice of Flats}\\
Let $\mathcal{A}_0(\mathcal{V})\defeq\mathcal{A}(\mathcal{V})\cup\{\emptyset\}$, ordered by inclusion. Then $\mathcal{A}_0(\mathcal{V})$ is a complete lattice.\\
For every family $\mathcal{C}\subseteq\mathcal{A}_0(\mathcal{V})$, $\bigwedge\mathcal{C}=\bigcap\limits_{X\in\mathcal{C}}X$, $\bigvee\mathcal{C}=
\begin{cases}
\operatorname{affhull}\left(\bigcup\limits_{X\in\mathcal{C}}X\right), & \bigcup\limits_{X\in\mathcal{C}}X\neq\emptyset,\\
\emptyset, & \bigcup\limits_{X\in\mathcal{C}}X=\emptyset.
\end{cases}$
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Arbitrary intersections of elements of $\mathcal{A}_0(\mathcal{V})$ lie in $\mathcal{A}_0(\mathcal{V})$: they are either empty, equal to $\mathcal{V}$ for the empty family, or flats by Lemma \ref{lem:intersection-of-flats}.
\item If $\bigcup\limits_{X\in\mathcal{C}}X=\emptyset$, then $\emptyset$ is the least upper bound of $\mathcal{C}$.\\
Otherwise, $\operatorname{affhull}(\bigcup\limits_{X\in\mathcal{C}}X)$ is the least flat containing every $X\in\mathcal{C}$, so it is the join.
\item Since arbitrary meets and joins exist, $\mathcal{A}_0(\mathcal{V})$ is complete.
\end{enumerate}
\hlt{Structural Commentary}: Adding the empty set repairs the failure of affine flats to be closed under intersection.\\
\hlt{Key Insight}: Meet is common incidence; join is the affine hull forced by union.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:empty-affine-hull-lattice-convention} \hlt{Empty Affine Hull Convention}\\
Within $\mathcal{A}_0(\mathcal{V})$, one may set $\operatorname{affhull}(\emptyset)\defeq\emptyset$ to write the join formula uniformly.
\end{remark}

\begin{corollary} \label{cor:join-of-two-flats-formula} \hlt{Join of Two Flats}\\
Let $X=\mathbf{x}+\mathcal{S}$ and $Y=\mathbf{y}+\mathcal{T}$ be flats. Then $X\vee Y=\mathbf{x}+\operatorname{span}\bigl(\{\mathbf{x}-\mathbf{y}\}\cup\mathcal{S}\cup\mathcal{T}\bigr)$.\\
If $X\cap Y\neq\emptyset$, then $X\vee Y=\mathbf{x}+(\mathcal{S}+\mathcal{T})$.
\begin{proof}
By Theorem \ref{thm:lattice-of-flats}, $X\vee Y=\operatorname{affhull}(X\cup Y)$. Apply Theorem \ref{thm:affine-hull-span-formula} to $X\cup Y$ with base point $\mathbf{x}$,\\
then $(X\cup Y)-\mathbf{x}=\mathcal{S}\cup\bigl((\mathbf{y}-\mathbf{x})+\mathcal{T}\bigr)$, whose span is $\operatorname{span}(\{\mathbf{x}-\mathbf{y}\}\cup\mathcal{S}\cup\mathcal{T})$.\\
If $X\cap Y\neq\emptyset$, then $\mathbf{x}-\mathbf{y}\in\mathcal{S}+\mathcal{T}$, so this span is $\mathcal{S}+\mathcal{T}$.\\
\hlt{Structural Commentary}: The join adds both bases and the displacement between representatives.\\
\hlt{Key Insight}: The displacement direction disappears exactly when the flats meet.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:dimension-of-join-of-flats} \hlt{Dimension of the Join of Two Flats}\\
Let $X=\mathbf{x}+\mathcal{S}$ and $Y=\mathbf{y}+\mathcal{T}$ be flats.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $X\cap Y\neq\emptyset$, then $\dim(X\vee Y)=\dim(X)+\dim(Y)-\dim(X\cap Y)$.
\item If $X\cap Y=\emptyset$, then $\dim(X\vee Y)=\dim(\mathcal{S}+\mathcal{T})+1$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $X\cap Y\neq\emptyset$, choose $\mathbf{z}\in X\cap Y$.\\
By Corollary \ref{cor:join-of-two-flats-formula} and Lemma \ref{lem:intersection-of-flats}, $X\vee Y=\mathbf{x}+(\mathcal{S}+\mathcal{T})$; $X\cap Y=\mathbf{z}+(\mathcal{S}\cap\mathcal{T})$. Thus the dimension formula for subspace sums gives $\dim(X\vee Y)=\dim(\mathcal{S}+\mathcal{T}) =\dim(\mathcal{S})+\dim(\mathcal{T})-\dim(\mathcal{S}\cap\mathcal{T})$.\\
Since $\dim X=\dim\mathcal{S}$, $\dim Y=\dim\mathcal{T}$, and $\dim(X\cap Y)=\dim(\mathcal{S}\cap\mathcal{T})$, this is (i).
\item If $X\cap Y=\emptyset$, Corollary \ref{cor:join-of-two-flats-formula} gives base $\mathcal{U}\defeq\operatorname{span}(\{\mathbf{x}-\mathbf{y}\}\cup\mathcal{S}\cup\mathcal{T})$.\\
Since $\mathbf{x}-\mathbf{y}\notin\mathcal{S}+\mathcal{T}$, we have $\dim\mathcal{U}=\dim(\mathcal{S}+\mathcal{T})+1$, hence $\dim(X\vee Y)=\dim(\mathcal{S}+\mathcal{T})+1$.
\end{enumerate}
\hlt{Structural Commentary}: Intersections subtract overlap; disjoint joins add one connecting direction.\\
\hlt{Key Insight}: The vector between representatives is new exactly when the flats are disjoint.
\end{proof}
\end{corollary}

\begin{definition} \label{def:affinely-independent-set} \hlt{Affinely Independent Sets}\\
A nonempty set $E\subseteq\mathcal{V}$ is \hlt{affinely independent} if, for every $\mathbf{x}\in E$, the set $(E-\mathbf{x})\setminus\{\mathbf{0}\}$ is linearly independent.
\end{definition}

\begin{theorem} \label{thm:affine-independence-equivalences} \hlt{Affine Independence Criteria}\\
For $\emptyset\neq E\subseteq\mathcal{V}$, conditions (i)--(iv) are equivalent. If $E$ is finite, they are also equivalent to (v).
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $E$ is affinely independent.
\item For every $\mathbf{x}\in E$, $\mathbf{x}\notin\operatorname{affhull}(E\setminus\{\mathbf{x}\})$, with $\operatorname{affhull}(\emptyset)=\emptyset$ in the singleton case.
\item $\forall$ distinct $\mathbf{x}_1,\ldots,\mathbf{x}_n\in E$ and $r_1,\ldots,r_n\in\mathbb{F}$, $\sum\limits_{j=1}^n r_j\mathbf{x}_j=\mathbf{0}$ and $\sum\limits_{j=1}^n r_j=0$ imply $r_1=\cdots=r_n=0$.
\item $\forall$ distinct $\mathbf{x}_1,\ldots,\mathbf{x}_n\in E$ and affine coefficient vectors $\mathbf{r},\mathbf{s}\in\mathbb{F}^n$,\\
$\sum\limits_{j=1}^n r_j\mathbf{x}_j=\sum\limits_{j=1}^n s_j\mathbf{x}_j$ and $\sum\limits_{j=1}^n r_j=\sum\limits_{j=1}^n s_j=1$ imply $\mathbf{r}=\mathbf{s}$.
\item If $E=\{\mathbf{x}_1,\ldots,\mathbf{x}_n\}$ is finite, then $\dim(\operatorname{affhull}(E))=n-1$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (iii)$ Let $\sum_{j=1}^n r_j\mathbf{x}_j=\mathbf{0}$ with distinct $\mathbf{x}_j\in E$ and $\sum_j r_j=0$. Then $\sum\limits_{j=1}^{n-1}r_j(\mathbf{x}_j-\mathbf{x}_n)=\mathbf{0}$.\\
By (i), $r_1=\cdots=r_{n-1}=0$, and then $r_n=0$.
\item[$(iii)$]$\Rightarrow (i)$ Fix $\mathbf{x}\in E$. Any relation $\sum\limits_{j=1}^m a_j(\mathbf{y}_j-\mathbf{x})=\mathbf{0}$, $\mathbf{y}_j\in E\setminus\{\mathbf{x}\}$ is the zero-sum relation\\
$\sum\limits_{j=1}^m a_j\mathbf{y}_j-(\sum_j a_j)\mathbf{x}=\mathbf{0}$, so (iii) gives $a_1=\cdots=a_m=0$.
\item[$(iii)$]$\Rightarrow (ii)$ If $\mathbf{x}_k\in\operatorname{affhull}(E\setminus\{\mathbf{x}_k\})$, then $\mathbf{x}_k=\sum_j r_j\mathbf{x}_j$ for $\mathbf{x}_j\neq\mathbf{x}_k$ and $\sum_j r_j=1$,\\
giving a nontrivial zero-sum relation, contradiction.
\item[$(ii)$]$\Rightarrow (iii)$ If a zero-sum relation is nontrivial, choose $r_k\neq0$.\\
Then $\mathbf{x}_k=\sum\limits_{j\neq k}\left(-\frac{r_j}{r_k}\right)\mathbf{x}_j$, $\sum\limits_{j\neq k}-\frac{r_j}{r_k}=1$, so $\mathbf{x}_k\in\operatorname{affhull}(E\setminus\{\mathbf{x}_k\})$, contradicting (ii).
\item[$(iii)$]$\Rightarrow (iv)$ Subtract the two affine representations; the coefficient vector $\mathbf{r}-\mathbf{s}$ has sum $0$, so (iii) gives $\mathbf{r}=\mathbf{s}$.
\item[$(iv)$]$\Rightarrow (iii)$ Let $\sum_j r_j\mathbf{x}_j=\mathbf{0}$ and $\sum_j r_j=0$. Choose $\mathbf{a}\in\mathbb{F}^n$ with $\sum_j a_j=1$.\\
Then $\mathbf{a}$ and $\mathbf{a}+\mathbf{r}$ are affine coefficient vectors for the same point. By (iv), $\mathbf{r}=\mathbf{0}$.
\item[$(i)$]$\Leftrightarrow (v)$ If $E=\{\mathbf{x}_1,\ldots,\mathbf{x}_n\}$ and $\mathbf{x}\in E$, then Theorem \ref{thm:affine-hull-span-formula} gives $\dim(\operatorname{affhull}(E))=\dim(\operatorname{span}(E-\mathbf{x}))$.\\
Since $(E-\mathbf{x})\setminus\{\mathbf{0}\}$ has $n-1$ vectors, this dimension is $n-1$ iff those vectors are linearly independent.
\end{enumerate}
\hlt{Structural Commentary}: Affine independence is linear independence after forgetting one chosen origin.\\
\hlt{Key Insight}: Uniqueness is absence of nontrivial zero-sum affine relations.
\end{proof}
\end{theorem}

\begin{definition} \label{def:affine-basis-barycentric-coordinates} \hlt{Affine Bases and Barycentric Coordinates}
Let $A=\mathbf{x}+\mathcal{S}$ be an $n$-flat.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A subset $\mathcal{B}\subseteq A$ is an \hlt{affine basis} of $A$ if $\mathcal{B}$ is affinely independent and $\operatorname{affspan}(\mathcal{B})=A$; an \hlt{ordered affine basis} is an ordering of such a basis.
\item If $(\mathbf{s}_1,\ldots,\mathbf{s}_n)$ is an ordered basis of $\mathcal{S}$, then $(\mathbf{x}+\mathbf{s}_1,\ldots,\mathbf{x}+\mathbf{s}_n,\mathbf{x})$ is an ordered affine basis of $A$.
\item For an ordered affine basis $(\mathbf{p}_1,\ldots,\mathbf{p}_{n+1})$, every $\mathbf{v}\in A$ has a unique expression $\mathbf{v}=\sum\limits_{j=1}^{n+1}r_j\mathbf{p}_j$, $\sum\limits_{j=1}^{n+1}r_j=1$.\\
The scalars $(r_1,\ldots,r_{n+1})$ are the \hlt{barycentric coordinates} of $\mathbf{v}$ in this basis.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:barycentric-coordinates-on-plane} \hlt{Barycentric Coordinates on a Plane}\\
Let $A=\mathbf{x}+\operatorname{span}(\mathbf{v}_1,\mathbf{v}_2)\subseteq\R^3$, with $(\mathbf{v}_1,\mathbf{v}_2)$ linearly independent.\\
Set $\mathbf{p}_1=\mathbf{x}+\mathbf{v}_1$, $\mathbf{p}_2=\mathbf{x}+\mathbf{v}_2$, and $\mathbf{p}_3=\mathbf{x}$. Then $(\mathbf{p}_1,\mathbf{p}_2,\mathbf{p}_3)$ is an ordered affine basis of $A$, and every $\mathbf{a}\in A$ has a unique barycentric expansion $\mathbf{a}=r_1\mathbf{p}_1+r_2\mathbf{p}_2+r_3\mathbf{p}_3$, $r_1+r_2+r_3=1$, with $r_1,r_2,r_3\in\R$.
\end{example}

\begin{definition} \label{def:affine-transformation} \hlt{Affine Transformation}\\
A map $f:\mathcal{V}\to\mathcal{V}$ is an \hlt{affine transformation} if it preserves finite affine combinations:\\
for all $\mathbf{x}_j\in\mathcal{V}$ and $r_j\in\mathbb{F}$ with $\sum\limits_j r_j=1$, $f\left(\sum\limits_j r_j\mathbf{x}_j\right)=\sum\limits_j r_j f(\mathbf{x}_j)$.
\end{definition}

\begin{theorem} \label{thm:affine-map-two-point-test} \hlt{Two-Point Test for Affine Maps}\\
Assume $\operatorname{char}(\mathbb{F})\neq2$.\\
A map $f:\mathcal{V}\to\mathcal{V}$ is affine if and only if $f(r\mathbf{x}+(1-r)\mathbf{y})=r f(\mathbf{x})+(1-r)f(\mathbf{y})$ for all $\mathbf{x},\mathbf{y}\in\mathcal{V}$ and $r\in\mathbb{F}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Since $r+(1-r)=1$, affinity gives $f(r\mathbf{x}+(1-r)\mathbf{y})=r f(\mathbf{x})+(1-r)f(\mathbf{y})$.
\item[$(\Leftarrow)$] Assume displayed identity. Prove by induction on $n$ that $f\left(\sum\limits_{j=1}^n r_j\mathbf{x}_j\right)=\sum\limits_{j=1}^n r_j f(\mathbf{x}_j)$; $\sum\limits_{j=1}^n r_j=1$.\\
The cases $n=1,2$ are immediate. For $n\geq3$, set $\mathbf{z}\defeq\sum\limits_{j=1}^n r_j\mathbf{x}_j$. If some $r_j\neq1$, relabel so $r_1\neq1$ and put $\mathbf{u}\defeq\sum\limits_{j=2}^n\frac{r_j}{1-r_1}\mathbf{x}_j$. Then $f(\mathbf{z}) =f(r_1\mathbf{x}_1+(1-r_1)\mathbf{u}) =r_1f(\mathbf{x}_1)+(1-r_1)f(\mathbf{u}) =\sum\limits_{j=1}^n r_j f(\mathbf{x}_j)$, by the two-point identity and induction. Otherwise all $r_j=1$; since $2$ is invertible, $\mathbf{z} =2\left(\frac12\mathbf{x}_1+\frac12\mathbf{x}_2\right)+r_3\mathbf{x}_3+\cdots+r_n\mathbf{x}_n$, an $(n-1)$-term affine combination, so induction plus two-point identity gives same formula for $f(\mathbf{z})$.
\end{enumerate}
\hlt{Structural Commentary}: In characteristic not $2$, line preservation gives affine preservation.\\
\hlt{Key Insight}: The same two-point assembly that builds affine-closed sets also tests affine maps.
\end{proof}
\end{theorem}

\begin{definition} \label{def:translation-map} \hlt{Translations}\\
For $\mathbf{a}\in\mathcal{V}$, the \hlt{translation by $\mathbf{a}$} is the map $T_{\mathbf{a}}:\mathcal{V}\to\mathcal{V}$ defined by $T_{\mathbf{a}}(\mathbf{x})=\mathbf{x}+\mathbf{a}$.
\end{definition}

\begin{theorem} \label{thm:affine-map-linear-part-translation} \hlt{Affine Maps Are Linear Maps Followed by Translations}\\
A map $f:\mathcal{V}\to\mathcal{V}$ is affine iff $f=T_{\mathbf{a}}\circ\tau$ for unique $\mathbf{a}\in\mathcal{V}$ and $\tau\in\LL(\mathcal{V})$, namely $\mathbf{a}=f(\mathbf{0})$ and $\tau=T_{-\mathbf{a}}\circ f$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Set $\mathbf{a}\defeq f(\mathbf{0})$ and $\tau(\mathbf{x})\defeq f(\mathbf{x})-\mathbf{a}$. For $r,s\in\mathbb{F}$ and $\mathbf{x},\mathbf{y}\in\mathcal{V}$,\\
$\tau(r\mathbf{x}+s\mathbf{y}) =f(r\mathbf{x}+s\mathbf{y}+(1-r-s)\mathbf{0})-\mathbf{a} =r\tau(\mathbf{x})+s\tau(\mathbf{y})$, so $\tau\in\LL(\mathcal{V})$ and $f=T_{\mathbf{a}}\circ\tau$.\\
Uniqueness follows from $\mathbf{a}=f(\mathbf{0})$ and $\tau(\mathbf{x})=f(\mathbf{x})-\mathbf{a}$.
\item[$(\Leftarrow)$] If $f=T_{\mathbf{a}}\circ\tau$ with $\tau$ linear,\\
then for $\sum\limits_j r_j=1$, $f\left(\sum\limits_j r_j\mathbf{x}_j\right) =\mathbf{a}+\tau\left(\sum\limits_j r_j\mathbf{x}_j\right) =\sum\limits_j r_j(\mathbf{a}+\tau\mathbf{x}_j) =\sum\limits_j r_j f(\mathbf{x}_j)$, hence $f$ is affine.
\end{enumerate}
\hlt{Structural Commentary}: An affine map is linear after moving its image of the origin back to the origin.\\
\hlt{Key Insight}: The failure of an affine map to be linear is exactly a translation.
\end{proof}
\end{theorem}

\begin{definition} \label{def:affine-group-translations-linear-part} \hlt{Affine Group Data}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{general linear group} is $\operatorname{GL}(\mathcal{V})\defeq\{\tau\in\LL(\mathcal{V}):\tau \text{ is invertible}\}$.
\item The \hlt{affine group} is $\operatorname{Aff}(\mathcal{V})\defeq\{T_{\mathbf{a}}\circ\tau:\mathbf{a}\in\mathcal{V},\,\tau\in\operatorname{GL}(\mathcal{V})\}$.
\item The \hlt{translation group} is $\operatorname{Trans}(\mathcal{V})\defeq\{T_{\mathbf{a}}:\mathbf{a}\in\mathcal{V}\}$.
\item The \hlt{linear-part map} is $\Lambda:\operatorname{Aff}(\mathcal{V})\to\operatorname{GL}(\mathcal{V})$; $\Lambda(T_{\mathbf{a}}\circ\tau)=\tau$, by Theorem \ref{thm:affine-map-linear-part-translation}.
\end{enumerate}
\end{definition}

\begin{corollary} \label{cor:affine-map-bijective-linear-part} \hlt{Bijective Affine Maps}\\
For $\mathbf{a}\in\mathcal{V}$ and $\tau\in\LL(\mathcal{V})$, $T_{\mathbf{a}}\circ\tau$ is bijective $\iff$ $\tau\in\operatorname{GL}(\mathcal{V})$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $T_{\mathbf{a}}\circ\tau$ is bijective, then $\tau=T_{-\mathbf{a}}\circ(T_{\mathbf{a}}\circ\tau)$ is bijective, hence $\tau\in\operatorname{GL}(\mathcal{V})$.
\item[$(\Leftarrow)$] If $\tau\in\operatorname{GL}(\mathcal{V})$, then $T_{\mathbf{a}}\circ\tau$ is bijective with inverse $\tau^{-1}\circ T_{-\mathbf{a}}$.
\end{enumerate}
\hlt{Structural Commentary}: Translation never affects invertibility; only the linear part does.\\
\hlt{Key Insight}: Bijective affine maps are exactly translations of linear isomorphisms.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:affine-group-linear-part} \hlt{Affine Group and Translations}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{a},\mathbf{b}\in\mathcal{V}$ and $\tau,\sigma\in\LL(\mathcal{V})$, $(T_{\mathbf{a}}\circ\tau)\circ(T_{\mathbf{b}}\circ\sigma) =T_{\mathbf{a}+\tau\mathbf{b}}\circ(\tau\sigma)$.
\item $\operatorname{Aff}(\mathcal{V})$ is a group, $\operatorname{Trans}(\mathcal{V})\trianglelefteq\operatorname{Aff}(\mathcal{V})$, and $\Lambda$ is a surjective group homomorphism with $\ker\Lambda=\operatorname{Trans}(\mathcal{V})$.
\item Consequently, $\operatorname{Aff}(\mathcal{V})/\operatorname{Trans}(\mathcal{V})\cong\operatorname{GL}(\mathcal{V})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item For $\mathbf{x}\in\mathcal{V}$, $((T_{\mathbf{a}}\circ\tau)\circ(T_{\mathbf{b}}\circ\sigma))(\mathbf{x}) =\mathbf{a}+\tau\mathbf{b}+(\tau\sigma)\mathbf{x}$.
\item For $\tau,\sigma\in\operatorname{GL}(\mathcal{V})$, (i) gives closure in $\operatorname{Aff}(\mathcal{V})$; the identity is $T_{\mathbf{0}}\circ I_{\mathcal{V}}$, and $(T_{\mathbf{a}}\circ\tau)^{-1}=T_{-\tau^{-1}\mathbf{a}}\circ\tau^{-1}$.\\
Thus $\operatorname{Aff}(\mathcal{V})$ is a group. Also (i) gives $\Lambda(gh)=\Lambda(g)\Lambda(h)$, $\ker\Lambda=\{T_{\mathbf{a}}\circ I_{\mathcal{V}}:\mathbf{a}\in\mathcal{V}\}=\operatorname{Trans}(\mathcal{V})$, and $\Lambda$ is surjective since $\tau=T_{\mathbf{0}}\circ\tau$ for all $\tau\in\operatorname{GL}(\mathcal{V})$. Hence $\operatorname{Trans}(\mathcal{V})\trianglelefteq\operatorname{Aff}(\mathcal{V})$.
\item Since $\Lambda$ is surjective and $\ker\Lambda=\operatorname{Trans}(\mathcal{V})$, the first isomorphism theorem gives the induced isomorphism\\
$\overline{\Lambda}:\operatorname{Aff}(\mathcal{V})/\operatorname{Trans}(\mathcal{V})\to\operatorname{GL}(\mathcal{V})$; $\overline{\Lambda}(g\,\operatorname{Trans}(\mathcal{V}))=\Lambda(g)$.
\end{enumerate}
\hlt{Structural Commentary}: The affine group is the translation part together with the invertible linear part.\\
\hlt{Key Insight}: Forgetting the translation records a homomorphism onto the linear automorphism group.
\end{proof}
\end{corollary}
