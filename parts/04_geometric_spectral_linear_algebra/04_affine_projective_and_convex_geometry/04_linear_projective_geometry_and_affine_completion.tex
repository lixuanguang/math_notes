\subsection{Linear Projective Geometry and Affine Completion}

\begin{definition} \label{def:projective-geometry-vector-space} \hlt{Projective Geometry of a Vector Space}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\operatorname{Sub}(\mathcal{V})$ be the set of subspaces of $\mathcal{V}$.\\
The \hlt{projective geometry} of $\mathcal{V}$ is $\operatorname{Sub}(\mathcal{V})$ with $\operatorname{pdim}(\mathcal{U})\defeq\dim(\mathcal{U})-1$.
\item A \hlt{projective point} is a $1$-dimensional subspace; a \hlt{projective line} is a $2$-dimensional subspace.
\item If $H=\mathbf{x}+\mathcal{K}$ is an affine hyperplane with $\mathbf{0}\notin H$ and $X\subseteq H$ is an affine flat, then $X$ is represented projectively by $\operatorname{span}(X)$, with $\operatorname{pdim}(\operatorname{span}(X))=\dim(X)$.
\end{enumerate}
\end{definition}

\begin{perspective} \label{per:projective-completion-from-affine-hyperplane} \hlt{Projective Completion of an Affine Hyperplane}\\
Let $H=\mathbf{x}+\mathcal{K}$ be a hyperplane with $\mathbf{0}\notin H$. The map $X\subseteq H \mapsto \operatorname{span}(X)\leq\mathcal{V}$ adjoins the missing directions in $\mathcal{K}$. Thus, when $\dim H=2$, parallel affine lines in $H$ acquire a common projective point at infinity.
\end{perspective}

\begin{theorem} \label{thm:linear-span-map-affine-to-projective} \hlt{Linear Span Map from Affine to Projective Geometry}\\
Let $H=\mathbf{x}+\mathcal{K}$ be an affine hyperplane with $\mathbf{0}\notin H$, and set $\mathcal{A}(H)\defeq\{X\in\mathcal{A}(\mathcal{V}):X\subseteq H\}$.\\
The map $P:\mathcal{A}(H)\to\operatorname{Sub}(\mathcal{V})$, $P(X)=\operatorname{span}(X)$ satisfies:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $P$ is injective and $P^{-1}(U)=U\cap H$ for $U\in\operatorname{im}P$.
\item $\operatorname{im}P=\{\mathcal{U}\leq\mathcal{V}:\mathcal{U}\nsubseteq\mathcal{K}\}$.
\item $X\subseteq Y$ if and only if $P(X)\subseteq P(Y)$.
\item If $\bigcap\limits_{i\in I}X_i\neq\emptyset$, then $P\left(\bigcap\limits_{i\in I}X_i\right)=\bigcap\limits_{i\in I}P(X_i)$.
\item For any family $\{X_i:i\in I\}\subseteq\mathcal{A}(H)$, $P\left(\bigvee\limits_{i\in I}X_i\right)=\bigvee\limits_{i\in I}P(X_i)$, where $\bigvee\limits_{i\in I}P(X_i)$ is computed in $\operatorname{Sub}(\mathcal{V})$, i.e. $\bigvee\limits_{i\in I}P(X_i)=\sum\limits_{i\in I}P(X_i)$.
\item $\operatorname{pdim}(P(X))=\dim(X)$.
\item $X\parallel Y$ if and only if one of $P(X)\cap\mathcal{K}$ and $P(Y)\cap\mathcal{K}$ is contained in the other.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $X=\mathbf{u}+\mathcal{S}\subseteq H$. Then $\mathcal{S}\leq\mathcal{K}$, $\mathbf{u}\notin\mathcal{K}$, and $P(X)=\mathbb{F}\mathbf{u}\oplus\mathcal{S}$.\\
If $\mathbf{z}\in P(X)\cap H$, then $\mathbf{z}=r\mathbf{u}+\mathbf{s}=\mathbf{u}+\mathbf{k}$ with $\mathbf{s}\in\mathcal{S}$ and $\mathbf{k}\in\mathcal{K}$. Hence $(1-r)\mathbf{u}\in\mathcal{K}$, so $r=1$, and $P(X)\cap H=X$. Thus $P$ is injective and $P^{-1}(U)=U\cap H$ on $\operatorname{im}P$.
\item If $\mathcal{U}\nsubseteq\mathcal{K}$, choose $\mathbf{u}\in\mathcal{U}\setminus\mathcal{K}$. Since $\mathcal{V}/\mathcal{K}$ is $1$-dimensional and $H=\mathbf{x}+\mathcal{K}$, there is $r\neq0$ with $\mathbf{h}\defeq r\mathbf{u}\in H$.\\
Then $\mathcal{U}\cap H=\mathbf{h}+(\mathcal{U}\cap\mathcal{K})$; $P(\mathcal{U}\cap H)=\mathbb{F}\mathbf{h}+(\mathcal{U}\cap\mathcal{K})=\mathcal{U}$. Thus $\operatorname{im}P=\{\mathcal{U}\leq\mathcal{V}:\mathcal{U}\nsubseteq\mathcal{K}\}$.
\item If $X\subseteq Y$, then $P(X)\subseteq P(Y)$. Conversely, if $P(X)\subseteq P(Y)$, then $X=P(X)\cap H\subseteq P(Y)\cap H=Y$.
\item If $\mathbf{z}\in\bigcap_i X_i$ and $X_i=\mathbf{z}+\mathcal{S}_i$, then Lemma \ref{lem:intersection-of-flats} gives $\bigcap_i X_i=\mathbf{z}+\bigcap_i\mathcal{S}_i$.\\
Since $P(X_i)=\mathbb{F}\mathbf{z}\oplus\mathcal{S}_i$, $\bigcap_i P(X_i)=\mathbb{F}\mathbf{z}\oplus\bigcap_i\mathcal{S}_i =P\left(\bigcap_iX_i\right)$.
\item By Theorem \ref{thm:lattice-of-flats}, $P\left(\bigvee_i X_i\right) =\operatorname{span}\left(\operatorname{affhull}\left(\bigcup_i X_i\right)\right) =\operatorname{span}\left(\bigcup_i X_i\right) =\bigvee_i P(X_i)$.
\item For $X=\mathbf{u}+\mathcal{S}\subseteq H$, $\operatorname{pdim}(P(X))=\dim(\mathbb{F}\mathbf{u}\oplus\mathcal{S})-1=\dim\mathcal{S}=\dim X$.
\item If $X=\mathbf{u}+\mathcal{S}$ and $Y=\mathbf{v}+\mathcal{T}$, then $P(X)\cap\mathcal{K}=\mathcal{S}$ and $P(Y)\cap\mathcal{K}=\mathcal{T}$.\\
Hence $X\parallel Y$ $\iff$ $\mathcal{S}\subseteq\mathcal{T}$ or $\mathcal{T}\subseteq\mathcal{S}$ $\iff$ $P(X)\cap\mathcal{K}\subseteq P(Y)\cap\mathcal{K}$ or $P(Y)\cap\mathcal{K}\subseteq P(X)\cap\mathcal{K}$.
\end{enumerate}
\hlt{Structural Commentary}: $H$ selects affine representatives for projective subspaces outside $\mathcal{K}$.\\
\hlt{Key Insight}: Intersecting with $H$ undoes the linear span exactly because $H$ does not pass through the origin.
\end{proof}
\end{theorem}
