\subsection{Convex Combinations, Cones, and Hulls}

\begin{definition} \label{def:convex-combinations-cones-hulls} \hlt{Convex Combinations, Cones, and Hulls}\\
Let $n,k\in\mathbb{Z}_{>0}$ and $\mathbf{x}_1,\ldots,\mathbf{x}_k\in\R^n$. Put $\Delta_{k-1}\defeq\{\mathbf{t}=(t_1,\ldots,t_k)\in[0,\infty)^k:\sum\limits_{i=1}^{k}t_i=1\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{convex combinations} of $\mathbf{x}_1,\ldots,\mathbf{x}_k$ form $\{\sum\limits_{i=1}^{k}t_i\mathbf{x}_i:\mathbf{t}\in\Delta_{k-1}\}$.
\item A set $C\subseteq\R^n$ is \hlt{convex} iff $(\forall\,\mathbf{x},\mathbf{y}\in C)(\forall\,t\in[0,1])\,t\mathbf{x}+(1-t)\mathbf{y}\in C$.
\item A set $K\subseteq\R^n$ is a \hlt{cone} iff $(\forall\,\mathbf{x}\in K)(\forall\,a\in[0,\infty))\,a\mathbf{x}\in K$.
\item A \hlt{convex cone} is a subset of $\R^n$ that is both convex and a cone.
\item For $S\defeq\{\mathbf{x}_1,\ldots,\mathbf{x}_k\}$, let $\mathcal{F}_S$ be the family of convex subsets of $\R^n$ containing $S$.\\
The \hlt{convex hull} of $S$ is $\operatorname{conv}(S)\defeq\bigcap\mathcal{F}_S$; the family $\mathcal{F}_S$ contains $\R^n$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:euclidean-topological-scope-convex-geometry} \hlt{Euclidean Topological Scope}\\
Let $d\in\mathbb{Z}_{>0}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{a}\in\R^d$ and $r>0$, put $B_r(\mathbf{a})\defeq\{\mathbf{x}\in\R^d:\norm{\mathbf{x}-\mathbf{a}}<r\}$.
\item A set $G\subseteq\R^d$ is \hlt{open} iff $(\forall\,\mathbf{x}\in G)(\exists\,r>0)\,B_r(\mathbf{x})\subseteq G$, and \hlt{closed} iff $\R^d\setminus G$ is open.
\item A set $K\subseteq\R^d$ is \hlt{compact} iff every cover of $K$ by open subsets of $\R^d$ has a finite subcover.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:convex-hull-finite-set-combinations-compact} \hlt{Finite Convex Hulls}\\
Let $n,k\in\mathbb{Z}_{>0}$, $\mathbf{x}_1,\ldots,\mathbf{x}_k\in\R^n$, and $S\defeq\{\mathbf{x}_1,\ldots,\mathbf{x}_k\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{conv}(S)=\left\{\sum\limits_{i=1}^{k}t_i\mathbf{x}_i:\mathbf{t}\in\Delta_{k-1}\right\}$.
\item $\operatorname{conv}(S)$ is compact.
\end{enumerate}
\begin{proof}
Define $T:\R^k\to\R^n$ by $T(\mathbf{t})\defeq\sum\limits_{i=1}^{k}t_i\mathbf{x}_i$, and put $C\defeq T(\Delta_{k-1})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,i\in\{1,\ldots,k\})\,[\mathbf{e}_i\in\Delta_{k-1}\land T(\mathbf{e}_i)=\mathbf{x}_i]$, so $S\subseteq C$. Let $K\subseteq\R^n$ be convex with $S\subseteq K$.\\
For $1\leq q\leq k$, put $C_q\defeq\{\sum\limits_{i=1}^{q}t_i\mathbf{x}_i:\mathbf{t}\in\Delta_{q-1}\}$. We prove $C_q\subseteq K$ by induction on $q$. The base case is $C_1=\{\mathbf{x}_1\}\subseteq K$.\\
Fix $2\leq q\leq k$ and assume $C_{q-1}\subseteq K$. Fix $\mathbf{t}\in\Delta_{q-1}$ and put $\mathbf{z}\defeq\sum\limits_{i=1}^{q}t_i\mathbf{x}_i\in C_q$.\\
If $t_q=1$, then $(\forall\,i<q)\,t_i=0$ by nonnegativity, and $\mathbf{z}=\mathbf{x}_q\in K$. If $t_q<1$, put $s\defeq1-t_q>0$ and $\mathbf{y}\defeq\sum\limits_{i=1}^{q-1}\frac{t_i}{s}\mathbf{x}_i$.\\
Then $(\frac{t_1}{s},\ldots,\frac{t_{q-1}}{s})\in\Delta_{q-2}$ by $\mathbf{t}\in\Delta_{q-1}$ and $s=1-t_q$, so $\mathbf{y}\in C_{q-1}$.\\
Then $\mathbf{z}=s\mathbf{y}+t_q\mathbf{x}_q\in K$ by the induction hypothesis and convexity. Hence $C=C_k\subseteq K$. Thus $C\subseteq\operatorname{conv}(S)$ by Definition \ref{def:convex-combinations-cones-hulls}(v).\\
Conversely, $t\mathbf{r}+(1-t)\mathbf{s}\in\Delta_{k-1}$ for $\mathbf{r},\mathbf{s}\in\Delta_{k-1}$ and $t\in[0,1]$ by nonnegativity and coefficient sums.\\
For $\mathbf{r},\mathbf{s}\in\Delta_{k-1}$ and $t\in[0,1]$, $T(t\mathbf{r}+(1-t)\mathbf{s})=tT(\mathbf{r})+(1-t)T(\mathbf{s})\in C$; hence $C$ is convex.\\
Since $S\subseteq C$, $\operatorname{conv}(S)\subseteq C$ by Definition \ref{def:convex-combinations-cones-hulls}(v), and therefore $\operatorname{conv}(S)=C$.
\item
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Coefficient compactness}: Let $\mathcal{U}$ be an open cover of $\Delta_{k-1}$ with no finite subcover, and put $Q_0\defeq[0,1]^k$.\\
Recursively, after $Q_m$ is chosen, partition it into $2^k$ equal closed subcubes.\\
Choose $Q_{m+1}$ among them so that $\Delta_{k-1}\cap Q_{m+1}$ has no finite $\mathcal{U}$-subcover.\\
At least one such subcube exists; otherwise finite subcovers of the $2^k$ intersections combine into a finite subcover of $\Delta_{k-1}\cap Q_m$.\\
Thus $(Q_m)_{m\geq0}$ is nested, $Q_m$ has side length $2^{-m}$, and $\Delta_{k-1}\cap Q_m\neq\emptyset$.\\
Write $Q_m=\prod\limits_{i=1}^{k}[a_{m,i},b_{m,i}]$ and put $t_i\defeq\sup\{a_{m,i}:m\geq0\}$.\\
Nestedness makes $(a_{m,i})_{m\geq0}$ nondecreasing, and $b_{m,i}$ is an upper bound of $\{a_{\ell,i}:\ell\geq0\}$.\\
Hence $a_{m,i}\leq t_i\leq b_{m,i}$, so $\mathbf{t}\defeq(t_1,\ldots,t_k)\in\bigcap\limits_{m\geq0}Q_m$. For $m\geq0$, choose $\mathbf{t}_m\in\Delta_{k-1}\cap Q_m$.\\
Then $\norm{\mathbf{t}_m-\mathbf{t}}\leq\sqrt{k}\,2^{-m}$, so $\mathbf{t}_m\to\mathbf{t}$. Since $\mathbf{t}\in Q_0$, its coordinates are nonnegative.\\
The equality $\sum\limits_{i=1}^{k}t_i=1$ holds by $\left|\sum\limits_{i=1}^{k}t_i-1\right|\leq k\,2^{-m}\to0$, so $\mathbf{t}\in\Delta_{k-1}$. Choose $U\in\mathcal{U}$ and $r>0$ with $\mathbf{t}\in B_r(\mathbf{t})\subseteq U$.\\
Choose $m$ with $\sqrt{k}\,2^{-m}<r$; then $Q_m\subseteq B_r(\mathbf{t})\subseteq U$, contrary to the choice of $Q_m$. Therefore $\Delta_{k-1}$ is compact.
\item \hlt{Open inverse images}: Put $L\defeq1+\sum\limits_{i=1}^{k}\norm{\mathbf{x}_i}>0$. For $\mathbf{r},\mathbf{s}\in\R^k$, $\norm{T(\mathbf{r})-T(\mathbf{s})}\leq\sum\limits_{i=1}^{k}|r_i-s_i|\norm{\mathbf{x}_i}\leq L\norm{\mathbf{r}-\mathbf{s}}$.\\
If $V\subseteq\R^n$ is open and $\mathbf{t}\in T^{-1}(V)$, choose $r>0$ with $B_r(T(\mathbf{t}))\subseteq V$. Then $B_{\frac{r}{L}}(\mathbf{t})\subseteq T^{-1}(V)$, so $T^{-1}(V)$ is open.
\item \hlt{Finite image cover}: Let $\mathcal{V}$ be an open cover of $C$. By (b), $\{T^{-1}(V):V\in\mathcal{V}\}$ consists of open subsets of $\R^k$.\\
The family covers $\Delta_{k-1}$ by $C=T(\Delta_{k-1})$. By (a), choose $m\in\mathbb{Z}_{>0}$ and $V_1,\ldots,V_m\in\mathcal{V}$ with $\Delta_{k-1}\subseteq\bigcup\limits_{j=1}^{m}T^{-1}(V_j)$.\\
Then $C\subseteq\bigcup\limits_{j=1}^{m}V_j$, so $C=\operatorname{conv}(S)$ is compact by (i).
\end{enumerate}
\end{enumerate}
\hlt{Structural Commentary}: Induction identifies the hull; cube bisection and the coefficient map establish compactness.\\
\hlt{Key Insight}: A finite convex hull is the image of a compact coefficient simplex.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:minkowski-sum-convex-compact-closed} \hlt{Sums of Convex and Closed Sets}\\
Let $n\in\mathbb{Z}_{>0}$ and $X,Y\subseteq\R^n$. Put $X+Y\defeq\{\mathbf{x}+\mathbf{y}:\mathbf{x}\in X,\ \mathbf{y}\in Y\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $X$ and $Y$ are convex, then $X+Y$ is convex.
\item If $X$ is compact and $Y$ is closed, then $X+Y$ is closed.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Assume $X$ and $Y$ are convex. Fix $\mathbf{z}_0,\mathbf{z}_1\in X+Y$ and $t\in[0,1]$. For $r\in\{0,1\}$, choose $(\mathbf{x}_r,\mathbf{y}_r)\in X\times Y$ with $\mathbf{z}_r=\mathbf{x}_r+\mathbf{y}_r$.\\
By convexity, $t\mathbf{x}_0+(1-t)\mathbf{x}_1\in X$ and $t\mathbf{y}_0+(1-t)\mathbf{y}_1\in Y$. Hence $t\mathbf{z}_0+(1-t)\mathbf{z}_1=\bigl(t\mathbf{x}_0+(1-t)\mathbf{x}_1\bigr)+\bigl(t\mathbf{y}_0+(1-t)\mathbf{y}_1\bigr)\in X+Y$.
\item Assume $X$ compact, $Y$ closed. If $X=\emptyset$, then $X+Y=\emptyset$. Its complement $\R^n$ is open by $(\forall\,\mathbf{u}\in\R^n)\,B_1(\mathbf{u})\subseteq\R^n$, so $\emptyset$ is closed.\\
Suppose $X\neq\emptyset$ and fix $\mathbf{z}\notin X+Y$. For $\mathbf{x}\in X$, choose $r_{\mathbf{x}}>0$ with $B_{2r_{\mathbf{x}}}(\mathbf{z}-\mathbf{x})\subseteq\R^n\setminus Y$ by openness of $\R^n\setminus Y$.\\
Fix $\mathbf{x}\in X$ and $\mathbf{u}\in B_{r_{\mathbf{x}}}(\mathbf{x})$, and put $\varepsilon\defeq r_{\mathbf{x}}-\norm{\mathbf{u}-\mathbf{x}}>0$. Then $B_\varepsilon(\mathbf{u})\subseteq B_{r_{\mathbf{x}}}(\mathbf{x})$ by Theorem \ref{thm:triangle-inequality-inner-product}(i).\\
Thus the balls $B_{r_{\mathbf{x}}}(\mathbf{x})$ form an open cover of $X$. Choose $m\in\mathbb{Z}_{>0}$ and $\mathbf{x}_1,\ldots,\mathbf{x}_m\in X$ with $X\subseteq\bigcup\limits_{j=1}^{m}B_{r_{\mathbf{x}_j}}(\mathbf{x}_j)$ by compactness.\\
Put $\delta\defeq\min\limits_{1\leq j\leq m}r_{\mathbf{x}_j}>0$. Suppose $\mathbf{w}=\mathbf{x}+\mathbf{y}\in B_\delta(\mathbf{z})\cap(X+Y)$ with $(\mathbf{x},\mathbf{y})\in X\times Y$. Choose $j$ with $\mathbf{x}\in B_{r_{\mathbf{x}_j}}(\mathbf{x}_j)$.\\
By Theorem \ref{thm:triangle-inequality-inner-product}(i), $\norm{\mathbf{y}-(\mathbf{z}-\mathbf{x}_j)}\leq\norm{\mathbf{w}-\mathbf{z}}+\norm{\mathbf{x}-\mathbf{x}_j}<\delta+r_{\mathbf{x}_j}\leq2r_{\mathbf{x}_j}$. This contradicts $\mathbf{y}\in Y$, so $B_\delta(\mathbf{z})\subseteq\R^n\setminus(X+Y)$.\\
Since $\mathbf{z}\notin X+Y$ was arbitrary, $\R^n\setminus(X+Y)$ is open and $X+Y$ is closed.
\end{enumerate}
\hlt{Structural Commentary}: Part (i) separates summands; compactness makes the exclusion radii uniform in (ii).\\
\hlt{Key Insight}: A finite subcover converts local separation into one neighborhood radius.
\end{proof}
\end{proposition}
