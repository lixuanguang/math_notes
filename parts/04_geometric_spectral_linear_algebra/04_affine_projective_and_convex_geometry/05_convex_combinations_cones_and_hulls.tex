\subsection{Convex Combinations, Cones, and Hulls}

\begin{definition} \label{def:convex-combinations-cones-hulls} \hlt{Convex Combinations, Cones, and Hulls}\\
Let $n,k\in\mathbb{Z}_{>0}$ and $\mathbf{x}_1,\ldots,\mathbf{x}_k\in\R^n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The set of \hlt{convex combinations} of $\mathbf{x}_1,\ldots,\mathbf{x}_k$ is $\left\{\sum\limits_{i=1}^{k}t_i\mathbf{x}_i:(t_1,\ldots,t_k)\in\R^k,\ (\forall\,i\in\{1,\ldots,k\})\,t_i\geq0,\ \sum\limits_{i=1}^{k}t_i=1\right\}$.
\item A set $C\subseteq\R^n$ is \hlt{convex} iff $(\forall\,\mathbf{x},\mathbf{y}\in C)(\forall\,t\in[0,1])\,t\mathbf{x}+(1-t)\mathbf{y}\in C$.
\item A set $K\subseteq\R^n$ is a \hlt{cone} iff $(\forall\,\mathbf{x}\in K)(\forall\,a\in[0,\infty))\,a\mathbf{x}\in K$.
\item A \hlt{convex cone} is a subset of $\R^n$ that is both convex and a cone.
\item For the nonempty finite set $S=\{\mathbf{x}_1,\ldots,\mathbf{x}_k\}$, its \hlt{convex hull} is $\operatorname{conv}(S)\defeq\bigcap\{C\subseteq\R^n:C\text{ is convex and }S\subseteq C\}$.\\
The defining family is nonempty because $\R^n$ is convex and contains $S$.
\end{enumerate}
\end{definition}

\hlt{Scope Note}: For $d\in\mathbb{Z}_{>0}$, $\mathbf{a}\in\R^d$, and $r>0$, write $B_r(\mathbf{a})\defeq\{\mathbf{x}\in\R^d:\norm{\mathbf{x}-\mathbf{a}}<r\}$.\\
A set $G\subseteq\R^d$ is \hlt{open} iff $(\forall\,\mathbf{x}\in G)(\exists\,r>0)\,B_r(\mathbf{x})\subseteq G$, and it is \hlt{closed} iff $\R^d\setminus G$ is open.\\
A set $K\subseteq\R^d$ is \hlt{compact} iff every cover of $K$ by open subsets of $\R^d$ has a finite subcover.

\begin{proposition} \label{prop:convex-hull-finite-set-combinations-compact} \hlt{Finite Convex Hulls}\\
Let $n,k\in\mathbb{Z}_{>0}$, $S=\{\mathbf{x}_1,\ldots,\mathbf{x}_k\}\subseteq\R^n$, and $\Delta_{k-1}\defeq\{(t_1,\ldots,t_k)\in\R^k:(\forall\,i\in\{1,\ldots,k\})\,t_i\geq0,\ \sum\limits_{i=1}^{k}t_i=1\}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{conv}(S)=\left\{\sum\limits_{i=1}^{k}t_i\mathbf{x}_i:(t_1,\ldots,t_k)\in\Delta_{k-1}\right\}$.
\item $\operatorname{conv}(S)$ is compact.
\end{enumerate}
\begin{proof}
Define $\widetilde{T}:\R^k\to\R^n$ by $\widetilde{T}(t_1,\ldots,t_k)\defeq\sum\limits_{i=1}^{k}t_i\mathbf{x}_i$, and put $C\defeq\widetilde{T}(\Delta_{k-1})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For the $i$th standard basis vector $\mathbf{e}_i\in\Delta_{k-1}$, $\widetilde{T}(\mathbf{e}_i)=\mathbf{x}_i$, so $S\subseteq C$. Let $K\subseteq\R^n$ be convex with $S\subseteq K$.\\
For $q\in\{1,\ldots,k\}$, every convex combination of $\mathbf{x}_1,\ldots,\mathbf{x}_q$ lies in $K$ by induction on $q$. The case $q=1$ is immediate.\\
Fix $q\geq2$, assume the assertion for $q-1$, and let $(t_1,\ldots,t_q)\in\R^q$ satisfy $(\forall\,i\in\{1,\ldots,q\})\,t_i\geq0$ and $\sum\limits_{i=1}^{q}t_i=1$.\\
For $t_q=1$, $(\forall\,i<q)\,t_i=0$ by nonnegativity, so $\sum\limits_{i=1}^{q}t_i\mathbf{x}_i=\mathbf{x}_q\in K$. For $t_q<1$, put $s\defeq1-t_q>0$ and $\mathbf{y}\defeq\sum\limits_{i=1}^{q-1}(t_i/s)\mathbf{x}_i$.\\
Coefficients $t_i/s$ are nonnegative, sum to $1$, so $\mathbf{y}\in K$ by induction hypothesis and $\sum\limits_{i=1}^{q}t_i\mathbf{x}_i=s\mathbf{y}+t_q\mathbf{x}_q\in K$ by convexity.\\
Thus $C\subseteq K$, and arbitrariness of $K$ proves $C\subseteq\operatorname{conv}(S)$.\\
Conversely, $(\forall\,\mathbf{r},\mathbf{s}\in\Delta_{k-1})(\forall\,t\in[0,1])\,t\mathbf{r}+(1-t)\mathbf{s}\in\Delta_{k-1}$ and $\widetilde{T}(t\mathbf{r}+(1-t)\mathbf{s})=t\widetilde{T}(\mathbf{r})+(1-t)\widetilde{T}(\mathbf{s})$, so $C$ is convex.\\
Since $S\subseteq C$, $\operatorname{conv}(S)\subseteq C$, and therefore $\operatorname{conv}(S)=C$.
\item
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Coefficient compactness}: Let $\mathcal{U}$ be an open cover of $\Delta_{k-1}$ and suppose it has no finite subcover. Put $Q_0\defeq[0,1]^k$.\\
Recursively choose closed subcube $Q_{m+1}$ from $2^k$ equal subcubes of $Q_m$ such that $\Delta_{k-1}\cap Q_{m+1}$ has no finite $\mathcal{U}$-subcover.\\
At least one such subcube exists; otherwise finite subcovers of the $2^k$ intersections would combine to cover $\Delta_{k-1}\cap Q_m$.\\
Thus $(Q_m)_{m\geq0}$ is nested, $Q_m$ has side length $2^{-m}$, and $\Delta_{k-1}\cap Q_m\neq\emptyset$\\
Write $Q_m=\prod\limits_{i=1}^{k}[a_{m,i},b_{m,i}]$ and put $t_i\defeq\sup\{a_{m,i}:m\geq0\}$.\\
The sequence $(a_{m,i})_{m\geq0}$ is nondecreasing by nestedness, and every $b_{m,i}$ is an upper bound for $\{a_{\ell,i}:\ell\geq0\}$.\\
Hence $a_{m,i}\leq t_i\leq b_{m,i}$. Thus $\mathbf{t}\defeq(t_1,\ldots,t_k)\in\bigcap\limits_{m\geq0}Q_m$. Choose $\mathbf{t}_m=(t_{m,1},\ldots,t_{m,k})\in\Delta_{k-1}\cap Q_m$.\\
Then $\norm{\mathbf{t}_m-\mathbf{t}}\leq\sqrt{k}\,2^{-m}$, so $\mathbf{t}_m\to\mathbf{t}$. Since $\mathbf{t}\in Q_0$, each $t_i\geq0$, and $\left|\sum\limits_{i=1}^{k}t_i-1\right|\leq\sum\limits_{i=1}^{k}|t_i-t_{m,i}|\to0$. Hence $\mathbf{t}\in\Delta_{k-1}$.\\
Choose $U\in\mathcal{U}$ with $\mathbf{t}\in U$, $r>0$ with $B_r(\mathbf{t})\subseteq U$. For $m$ with $\sqrt{k}\,2^{-m}<r$, $Q_m\subseteq B_r(\mathbf{t})\subseteq U$, contradicting choice of $Q_m$.\\
Therefore $\Delta_{k-1}$ is compact.
\item \hlt{Open inverse images}: Put $M\defeq\sum\limits_{i=1}^{k}\norm{\mathbf{x}_i}$. For $\mathbf{r},\mathbf{s}\in\R^k$, $\norm{\widetilde{T}(\mathbf{r})-\widetilde{T}(\mathbf{s})}\leq\sum\limits_{i=1}^{k}|r_i-s_i|\norm{\mathbf{x}_i}\leq M\norm{\mathbf{r}-\mathbf{s}}$.\\
If $M=0$, every inverse image $\widetilde{T}^{-1}(V)$ is either $\R^k$ or $\emptyset$.\\
If $M>0$, $V\subseteq\R^n$ is open, and $\mathbf{t}\in\widetilde{T}^{-1}(V)$, choose $r>0$ with $B_r(\widetilde{T}(\mathbf{t}))\subseteq V$; then $B_{r/M}(\mathbf{t})\subseteq\widetilde{T}^{-1}(V)$.\\
Thus inverse images of open sets under $\widetilde{T}$ are open.
\item \hlt{Finite image cover}: Let $\mathcal{V}$ be an open cover of $C$. The family $\{\widetilde{T}^{-1}(V):V\in\mathcal{V}\}$ is an open cover of $\Delta_{k-1}$ by (b).\\
By (a), fix $m\in\mathbb{Z}_{>0}$, $V_1,\ldots,V_m\in\mathcal{V}$ satisfying $\Delta_{k-1}\subseteq\bigcup\limits_{j=1}^{m}\widetilde{T}^{-1}(V_j)$. Hence $C\subseteq\bigcup\limits_{j=1}^{m}V_j$, so $C=\operatorname{conv}(S)$ is compact by (i).
\end{enumerate}
\end{enumerate}
\hlt{Structural Commentary}: Cube bisection proves simplex compactness; a Lipschitz estimate transports finite subcovers.\\
\hlt{Key Insight}: Compactness passes from coefficients to their convex combinations.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:minkowski-sum-convex-compact-closed} \hlt{Sums of Convex and Closed Sets}\\
For $X,Y\subseteq\R^n$, set $X+Y\defeq\{\mathbf{x}+\mathbf{y}:\mathbf{x}\in X,\ \mathbf{y}\in Y\}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $X,Y\text{ convex}\Longrightarrow X+Y\text{ convex}$.
\item $X\text{ compact and }Y\text{ closed}\Longrightarrow X+Y\text{ closed}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Assume $X,Y$ are convex. For $\mathbf{z}_r=\mathbf{x}_r+\mathbf{y}_r\in X+Y$, where $r\in\{0,1\}$, and\\
$t\in[0,1]$, $t\mathbf{z}_0+(1-t)\mathbf{z}_1=\bigl(t\mathbf{x}_0+(1-t)\mathbf{x}_1\bigr)+\bigl(t\mathbf{y}_0+(1-t)\mathbf{y}_1\bigr)\in X+Y$. Thus $X+Y$ is convex.
\item Assume $X$ is compact and $Y$ is closed. The case $X=\emptyset$ is immediate, so fix $\mathbf{z}\notin X+Y$ with $X\neq\emptyset$.\\
For $\mathbf{x}\in X$, choose $r_{\mathbf{x}}>0$ with $B_{2r_{\mathbf{x}}}(\mathbf{z}-\mathbf{x})\subseteq\R^n\setminus Y$ by openness of $\R^n\setminus Y$.\\
The balls $B_{r_{\mathbf{x}}}(\mathbf{x})$ are open by the triangle inequality and cover $X$.\\
Choose $m\in\mathbb{Z}_{>0}$ and $\mathbf{x}_1,\ldots,\mathbf{x}_m\in X$ with $X\subseteq\bigcup\limits_{j=1}^{m}B_{r_{\mathbf{x}_j}}(\mathbf{x}_j)$ by compactness, and put $\delta\defeq\min\limits_{1\leq j\leq m}r_{\mathbf{x}_j}>0$.\\
If $\mathbf{w}=\mathbf{x}+\mathbf{y}\in B_\delta(\mathbf{z})\cap(X+Y)$ with $(\mathbf{x},\mathbf{y})\in X\times Y$, choose $j$ with $\mathbf{x}\in B_{r_{\mathbf{x}_j}}(\mathbf{x}_j)$.\\
Then $\norm{\mathbf{y}-(\mathbf{z}-\mathbf{x}_j)}\leq\norm{\mathbf{w}-\mathbf{z}}+\norm{\mathbf{x}-\mathbf{x}_j}<\delta+r_{\mathbf{x}_j}\leq2r_{\mathbf{x}_j}$, contradicting $\mathbf{y}\in Y$.\\
Therefore $B_\delta(\mathbf{z})\subseteq\R^n\setminus(X+Y)$. The complement of $X+Y$ is open, so $X+Y$ is closed.
\end{enumerate}
\hlt{Structural Commentary}: A finite compact cover makes the pointwise separation from $Y$ uniform.\\
\hlt{Key Insight}: Compactness converts varying exclusion radii into one positive radius.
\end{proof}
\end{proposition}
