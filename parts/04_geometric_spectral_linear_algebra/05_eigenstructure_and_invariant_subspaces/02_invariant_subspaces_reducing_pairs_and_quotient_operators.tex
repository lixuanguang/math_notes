\subsection{Invariant Subspaces, Reducing Pairs, and Quotient Operators}

Throughout this subsection, $\mathcal{V}$ is a vector space over $\mathbb{F}$. In statements involving inner products, adjoints, or unitary matrices, $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$.

\begin{definition} \label{def:invariant-subspace} \hlt{Invariant Subspace}\\
Let $T\in\LL(\mathcal{V})$. A subspace $\mathcal{U}\leq\mathcal{V}$ is \hlt{invariant under $T$} iff $(\forall\,\mathbf{u}\in\mathcal{U})\,T\mathbf{u}\in\mathcal{U}$, equivalently, iff $T\mathcal{U}\subseteq\mathcal{U}$.\\
The set of all $T$-invariant subspaces is $\operatorname{Inv}(T)\defeq\{\mathcal{U}\leq\mathcal{V}:T\mathcal{U}\subseteq\mathcal{U}\}$.
\end{definition}

\begin{definition} \label{def:restriction-and-quotient-operators} \hlt{Restriction and Quotient Operators}\\
Let $T\in\LL(\mathcal{V})$ and $\mathcal{U}\in\operatorname{Inv}(T)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{restriction operator} $T|_{\mathcal{U}}:\mathcal{U}\to\mathcal{U}$ is prescribed by $(\forall\,\mathbf{u}\in\mathcal{U})\,T|_{\mathcal{U}}(\mathbf{u})\defeq T\mathbf{u}$.
\item The \hlt{quotient operator} $T/\mathcal{U}:\mathcal{V}/\mathcal{U}\to\mathcal{V}/\mathcal{U}$ is prescribed by $(\forall\,\mathbf{v}\in\mathcal{V})\,(T/\mathcal{U})(\mathbf{v}+\mathcal{U})\defeq T\mathbf{v}+\mathcal{U}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:restriction-and-quotient-operators-well-defined} \hlt{Restriction and Quotient Operators Are Well-Defined}\\
Let $T\in\LL(\mathcal{V})$ and $\mathcal{U}\in\operatorname{Inv}(T)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T|_{\mathcal{U}}\in\LL(\mathcal{U})$.
\item $T/\mathcal{U}\in\LL(\mathcal{V}/\mathcal{U})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $\mathcal{U}\in\operatorname{Inv}(T)$, $(\forall\,\mathbf{u}\in\mathcal{U})\,T\mathbf{u}\in\mathcal{U}$, so $T|_{\mathcal{U}}$ has the asserted codomain.\\
For $a,b\in\mathbb{F}$ and $\mathbf{u},\mathbf{v}\in\mathcal{U}$, $T|_{\mathcal{U}}(a\mathbf{u}+b\mathbf{v})=aT|_{\mathcal{U}}\mathbf{u}+bT|_{\mathcal{U}}\mathbf{v}$ by the linearity of $T$. Hence $T|_{\mathcal{U}}\in\LL(\mathcal{U})$.
\item Suppose $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$. Then $\mathbf{v}-\mathbf{w}\in\mathcal{U}$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}, so $T\mathbf{v}-T\mathbf{w}=T(\mathbf{v}-\mathbf{w})\in\mathcal{U}$ by invariance.\\
Thus $T\mathbf{v}+\mathcal{U}=T\mathbf{w}+\mathcal{U}$ by Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint}, and the quotient rule is independent of the representative.\\
For $a,b\in\mathbb{F}$ and $\mathbf{v},\mathbf{w}\in\mathcal{V}$, $(T/\mathcal{U})(a(\mathbf{v}+\mathcal{U})+b(\mathbf{w}+\mathcal{U}))=T(a\mathbf{v}+b\mathbf{w})+\mathcal{U}=a(T\mathbf{v}+\mathcal{U})+b(T\mathbf{w}+\mathcal{U})$ by Proposition \ref{prop:quotient-space-structure}(ii) and the linearity of $T$. Hence $T/\mathcal{U}\in\LL(\mathcal{V}/\mathcal{U})$.
\end{enumerate}
\hlt{Structural Commentary}: Restriction controls the codomain; quotient formation first controls representatives.\\
\hlt{Key Insight}: Invariance is exactly the condition that lets $T$ act both inside $\mathcal{U}$ and on cosets modulo $\mathcal{U}$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:one-dimensional-invariant-subspaces-and-eigenvectors} \hlt{One-Dimensional Invariant Subspaces and Eigenvectors}\\
Let $T\in\LL(\mathcal{V})$ and $\mathbf{v}\in\mathcal{V}\setminus\{\mathbf{0}\}$. Then $\operatorname{span}(\mathbf{v})\in\operatorname{Inv}(T)$ iff $\mathbf{v}$ is an eigenvector of $T$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $\operatorname{span}(\mathbf{v})\in\operatorname{Inv}(T)$. Then $T\mathbf{v}\in\operatorname{span}(\mathbf{v})$, so $(\exists\,\lambda\in\mathbb{F})\,T\mathbf{v}=\lambda\mathbf{v}$. Since $\mathbf{v}\neq\mathbf{0}$, it is an eigenvector of $T$ by Definition \ref{def:eigenvalue-eigenvector}.
\item[$(\Leftarrow)$] Suppose $\mathbf{v}$ is an eigenvector of $T$. Choose $\lambda\in\mathbb{F}$ such that $T\mathbf{v}=\lambda\mathbf{v}$ by Definition \ref{def:eigenvalue-eigenvector}.\\
Then $(\forall\,a\in\mathbb{F})\,T(a\mathbf{v})=a\lambda\mathbf{v}\in\operatorname{span}(\mathbf{v})$, so $\operatorname{span}(\mathbf{v})\in\operatorname{Inv}(T)$.
\end{enumerate}
\hlt{Structural Commentary}: Each direction tests the action of $T$ on a generator of the one-dimensional subspace.\\
\hlt{Key Insight}: A nonzero vector spans an invariant line precisely when its image is a scalar multiple of it.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:invariant-subspaces-and-block-form} \hlt{Invariant Subspaces and Block Form}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and suppose $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathcal{U}\in\operatorname{Inv}(T)$, then the matrix of $T$ in every basis adapted to $\mathcal{U}\oplus\mathcal{W}$ has the form $\left[\begin{smallmatrix}A&B\\0&D\end{smallmatrix}\right]$.
\item If $\mathcal{U},\mathcal{W}\in\operatorname{Inv}(T)$, then the matrix of $T$ in every basis adapted to $\mathcal{U}\oplus\mathcal{W}$ has the form $\left[\begin{smallmatrix}A&0\\0&D\end{smallmatrix}\right]$.
\end{enumerate}
\begin{proof}
Choose bases $(\mathbf{u}_1,\ldots,\mathbf{u}_p)$ of $\mathcal{U}$ and $(\mathbf{w}_1,\ldots,\mathbf{w}_q)$ of $\mathcal{W}$ by Corollary \ref{cor:finite-dimensional-spaces-and-subspaces-have-bases}. Concatenation is a basis of $\mathcal{V}$ by Theorem \ref{thm:direct-sum-iff-dimensions-add}(ii).
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Suppose $\mathcal{U}\in\operatorname{Inv}(T)$. For $j\in\{1,\ldots,p\}$, $T\mathbf{u}_j\in\mathcal{U}$, so the coordinates of $T\mathbf{u}_j$ along $\mathbf{w}_1,\ldots,\mathbf{w}_q$ vanish.\\
These entries constitute the lower-left block, which is therefore zero.
\item Suppose also that $\mathcal{W}\in\operatorname{Inv}(T)$. For $j\in\{1,\ldots,q\}$, $T\mathbf{w}_j\in\mathcal{W}$, so the coordinates of $T\mathbf{w}_j$ along $\mathbf{u}_1,\ldots,\mathbf{u}_p$ vanish.\\
Thus the upper-right block is also zero, and the matrix is block diagonal.
\end{enumerate}
\hlt{Structural Commentary}: Columns belonging to each adapted basis block reveal which components of their images must vanish.\\
\hlt{Key Insight}: One invariant summand forces block triangularity; two invariant summands force block diagonality.
\end{proof}
\end{proposition}

\begin{definition} \label{def:reducing-pair-and-direct-sum-operator} \hlt{Reducing Pair}\\
Let $T\in\LL(\mathcal{V})$. A pair $(\mathcal{U},\mathcal{W})$ of subspaces \hlt{reduces $T$} iff $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$ and $\mathcal{U},\mathcal{W}\in\operatorname{Inv}(T)$.\\
Under this condition, $T=T|_{\mathcal{U}}\oplus T|_{\mathcal{W}}$, meaning that $(\forall\,(\mathbf{u},\mathbf{w})\in\mathcal{U}\times\mathcal{W})\,T(\mathbf{u}+\mathbf{w})=T|_{\mathcal{U}}\mathbf{u}+T|_{\mathcal{W}}\mathbf{w}$.
\end{definition}

\begin{theorem} \label{thm:adjoint-invariant-subspace-orthogonal-complement} \hlt{Adjoints and Invariant Orthogonal Complements}\\
Let $T\in\LL(\mathcal{V})$ on a finite-dimensional inner-product space and let $\mathcal{U}\leq\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{U}\in\operatorname{Inv}(T)\Longleftrightarrow\mathcal{U}^{\perp}\in\operatorname{Inv}(T^*)$.
\item $(\mathcal{U},\mathcal{U}^{\perp})$ reduces $T\Longleftrightarrow\mathcal{U}\in\operatorname{Inv}(T)\cap\operatorname{Inv}(T^*)$.
\item If $(\mathcal{U},\mathcal{U}^{\perp})$ reduces $T$, then $(T|_{\mathcal{U}})^*=(T^*)|_{\mathcal{U}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Suppose $\mathcal{U}\in\operatorname{Inv}(T)$. If $\mathbf{z}\in\mathcal{U}^{\perp}$ and $\mathbf{u}\in\mathcal{U}$, then $\innerprod{T^*\mathbf{z},\mathbf{u}}=\overline{\innerprod{\mathbf{u},T^*\mathbf{z}}}=\overline{\innerprod{T\mathbf{u},\mathbf{z}}}=\innerprod{\mathbf{z},T\mathbf{u}}=0$ by conjugate symmetry, Definition \ref{def:adjoint-linear-map}, and $T\mathbf{u}\in\mathcal{U}$. Hence $T^*\mathbf{z}\in\mathcal{U}^{\perp}$, so $\mathcal{U}^{\perp}\in\operatorname{Inv}(T^*)$.
\item[$(i)$] ($\Leftarrow$) Suppose $\mathcal{U}^{\perp}\in\operatorname{Inv}(T^*)$. If $\mathbf{u}\in\mathcal{U}$ and $\mathbf{z}\in\mathcal{U}^{\perp}$, then $\innerprod{T\mathbf{u},\mathbf{z}}=\innerprod{\mathbf{u},T^*\mathbf{z}}=0$ by Definition \ref{def:adjoint-linear-map}.\\
Thus $T\mathbf{u}\in(\mathcal{U}^{\perp})^{\perp}=\mathcal{U}$ by Corollary \ref{cor:double-orthogonal-complement}, and $\mathcal{U}\in\operatorname{Inv}(T)$.
\item[$(ii)$] ($\Rightarrow$) Suppose $(\mathcal{U},\mathcal{U}^{\perp})$ reduces $T$. Then $\mathcal{U}\in\operatorname{Inv}(T)$ and $\mathcal{U}^{\perp}\in\operatorname{Inv}(T)$ by Definition \ref{def:reducing-pair-and-direct-sum-operator}.\\
Thus $\mathcal{U}\in\operatorname{Inv}(T^*)$ by (i) for $T^*$ and $(T^*)^*=T$ from Proposition \ref{prop:basic-adjoint-identities}(iii).
\item[$(ii)$] ($\Leftarrow$) Suppose $\mathcal{U}\in\operatorname{Inv}(T)\cap\operatorname{Inv}(T^*)$. Then $\mathcal{U}^{\perp}\in\operatorname{Inv}(T)$ by (i) for $T^*$ and Proposition \ref{prop:basic-adjoint-identities}(iii).\\
The identity $\mathcal{V}=\mathcal{U}\oplus\mathcal{U}^{\perp}$ holds by Theorem \ref{thm:orthogonal-direct-sum-decomposition}. Hence $(\mathcal{U},\mathcal{U}^{\perp})$ reduces $T$ by Definition \ref{def:reducing-pair-and-direct-sum-operator}.
\item[$(iii)$] Suppose $(\mathcal{U},\mathcal{U}^{\perp})$ reduces $T$. Then $\mathcal{U}\in\operatorname{Inv}(T)\cap\operatorname{Inv}(T^*)$ by (ii), and $T|_{\mathcal{U}},(T^*)|_{\mathcal{U}}\in\LL(\mathcal{U})$ by Proposition \ref{prop:restriction-and-quotient-operators-well-defined}(i).\\
For $\mathbf{u},\mathbf{v}\in\mathcal{U}$, $\innerprod[\mathcal{U}]{T|_{\mathcal{U}}\mathbf{u},\mathbf{v}}=\innerprod[\mathcal{V}]{T\mathbf{u},\mathbf{v}}=\innerprod[\mathcal{V}]{\mathbf{u},T^*\mathbf{v}}=\innerprod[\mathcal{U}]{\mathbf{u},(T^*)|_{\mathcal{U}}\mathbf{v}}$ by Definition \ref{def:adjoint-linear-map}.\\
Therefore $(T^*)|_{\mathcal{U}}$ is the adjoint of $T|_{\mathcal{U}}$, and $(T|_{\mathcal{U}})^*=(T^*)|_{\mathcal{U}}$ by Lemma \ref{lem:adjoint-unique-if-exists}.
\end{enumerate}
\hlt{Structural Commentary}: Part (i) transfers invariance across the adjoint; the remaining claims use orthogonal decomposition.\\
\hlt{Key Insight}: A subspace reduces $T$ exactly when both $T$ and $T^*$ preserve it.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:reducing-pairs-and-algebraic-projections} \hlt{Reducing Pairs and Projections}\\
Let $T\in\LL(\mathcal{V})$, let $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, and set $\rho\defeq\rho_{\mathcal{U},\mathcal{W}}$. Then $(\mathcal{U},\mathcal{W})$ reduces $T$ iff $T\rho=\rho T$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $(\mathcal{U},\mathcal{W})$ reduces $T$. For $\mathbf{v}=\mathbf{u}+\mathbf{w}$ with $(\mathbf{u},\mathbf{w})\in\mathcal{U}\times\mathcal{W}$, $T\rho\mathbf{v}=T\mathbf{u}$.\\
Also $\rho T\mathbf{v}=\rho(T\mathbf{u}+T\mathbf{w})=T\mathbf{u}$ because $T\mathbf{u}\in\mathcal{U}$ and $T\mathbf{w}\in\mathcal{W}$. Hence $T\rho=\rho T$.
\item[$(\Leftarrow)$] Suppose $T\rho=\rho T$. If $\mathbf{u}\in\mathcal{U}$, then $\rho T\mathbf{u}=T\rho\mathbf{u}=T\mathbf{u}$, so $T\mathbf{u}\in\operatorname{range}(\rho)=\mathcal{U}$ by Proposition \ref{prop:algebraic-projection-range-kernel-idempotent}(iii).\\
If $\mathbf{w}\in\mathcal{W}$, then $\rho T\mathbf{w}=T\rho\mathbf{w}=\mathbf{0}$, so $T\mathbf{w}\in\ker(\rho)=\mathcal{W}$ by the same proposition. Thus $(\mathcal{U},\mathcal{W})$ reduces $T$ by Definition \ref{def:reducing-pair-and-direct-sum-operator}.
\end{enumerate}
\hlt{Structural Commentary}: Evaluating the commutator on the range and kernel of $\rho$ isolates the two summands.\\
\hlt{Key Insight}: A direct-sum projection commutes with $T$ precisely when both of its components are $T$-invariant.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:invariant-subspace-projection-criterion} \hlt{Invariant Subspaces and Algebraic Projections}\\
Let $T\in\LL(\mathcal{V})$, let $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, and set $\rho\defeq\rho_{\mathcal{U},\mathcal{W}}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{U}\in\operatorname{Inv}(T)\Longleftrightarrow\rho T\rho=T\rho$.
\item If $\mathcal{U}\in\operatorname{Inv}(T)$, then $(\forall\,\mathcal{Z}\leq\mathcal{V})\,[\mathcal{V}=\mathcal{U}\oplus\mathcal{Z}\Longrightarrow\rho_{\mathcal{U},\mathcal{Z}}T\rho_{\mathcal{U},\mathcal{Z}}=T\rho_{\mathcal{U},\mathcal{Z}}]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Suppose $\mathcal{U}\in\operatorname{Inv}(T)$, and write $\mathbf{v}=\mathbf{u}+\mathbf{w}$ with $(\mathbf{u},\mathbf{w})\in\mathcal{U}\times\mathcal{W}$.\\
Then $T\mathbf{u}\in\mathcal{U}$, so $(\rho T\rho)\mathbf{v}=\rho T\mathbf{u}=T\mathbf{u}=(T\rho)\mathbf{v}$ by Definition \ref{def:projection-onto-subspace-along-complement}. Hence $\rho T\rho=T\rho$.
\item[$(i)$] ($\Leftarrow$) Suppose $\rho T\rho=T\rho$. If $\mathbf{u}\in\mathcal{U}$, then $\rho T\mathbf{u}=\rho T\rho\mathbf{u}=T\rho\mathbf{u}=T\mathbf{u}$.\\
Thus $T\mathbf{u}\in\operatorname{range}(\rho)=\mathcal{U}$ by Proposition \ref{prop:algebraic-projection-range-kernel-idempotent}(iii), so $\mathcal{U}\in\operatorname{Inv}(T)$.
\item[$(ii)$] Suppose $\mathcal{U}\in\operatorname{Inv}(T)$ and $\mathcal{V}=\mathcal{U}\oplus\mathcal{Z}$. Then $\rho_{\mathcal{U},\mathcal{Z}}T\rho_{\mathcal{U},\mathcal{Z}}=T\rho_{\mathcal{U},\mathcal{Z}}$ by forward implication in (i) for decomposition $\mathcal{V}=\mathcal{U}\oplus\mathcal{Z}$.\\
Since $\mathcal{Z}$ was arbitrary, the universal claim follows.
\end{enumerate}
\hlt{Structural Commentary}: Part (i) detects escape from $\mathcal{U}$; part (ii) records complement independence.\\
\hlt{Key Insight}: Invariance is characterized by the identity $\rho T\rho=T\rho$ for any projection onto the subspace.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:matrix-invariant-subspace-projection-block-criteria} \hlt{Invariant Subspaces, Blocks, and Projections}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$, let $\mathcal{U}\leq\mathbb{F}^n$ satisfy $\dim(\mathcal{U})=r$ and $1\leq r\leq n-1$, and let $Q=[Q_1\ Q_2]$ be unitary, where the columns of $Q_1$ and $Q_2$ are orthonormal bases of $\mathcal{U}$ and $\mathcal{U}^{\perp}$, respectively. Set $P\defeq Q_1Q_1^*$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\mathcal{U}\subseteq\mathcal{U}$.
\item $(\exists\,B\in\operatorname{Mat}(r,r,\mathbb{F}))(\exists\,X\in\operatorname{Mat}(r,n-r,\mathbb{F}))(\exists\,C\in\operatorname{Mat}(n-r,n-r,\mathbb{F}))\,Q^*AQ=\left[\begin{smallmatrix}B&X\\0&C\end{smallmatrix}\right]$.
\item $PAP=AP$.
\end{enumerate}
\begin{proof}
Since $Q$ is unitary, $Q_1^*Q_1=I_r$ and $Q_2^*Q_1=0$. Thus $P^*=P$, $P^2=Q_1(Q_1^*Q_1)Q_1^*=P$. For $\mathbf{x}\in\mathbb{F}^n$, $P\mathbf{x}=Q_1(Q_1^*\mathbf{x})\in\operatorname{col}(Q_1)$, while $P(Q_1\mathbf{c})=Q_1\mathbf{c}$ for $\mathbf{c}\in\mathbb{F}^r$. Hence $P$ is an orthogonal projection matrix by Definition \ref{def:orthogonal-projection-matrix}, and $\operatorname{range}(P)=\operatorname{col}(Q_1)=\mathcal{U}$.\\
Set $M\defeq Q^*AQ=\left[\begin{smallmatrix}B&X\\Y&C\end{smallmatrix}\right]$ and $E\defeq Q^*PQ=I_r\oplus0_{n-r}$. Then $Y=Q_2^*AQ_1$.\\
For $\mathbf{x}\in\mathbb{F}^n$, $Q_2^*\mathbf{x}=0\Longleftrightarrow\mathbf{x}\in(\operatorname{col}(Q_2))^{\perp}=(\mathcal{U}^{\perp})^{\perp}=\mathcal{U}$ by Corollary \ref{cor:double-orthogonal-complement}; hence $\ker(Q_2^*)=\mathcal{U}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Suppose $A\mathcal{U}\subseteq\mathcal{U}$. Each column of $AQ_1$ then belongs to $\mathcal{U}=\ker(Q_2^*)$, so $Y=Q_2^*AQ_1=0$.
\item[$(ii)$] $\Rightarrow (iii)$ Suppose $Y=0$. Then $EME=\left[\begin{smallmatrix}B&0\\0&0\end{smallmatrix}\right]=ME$.\\
Consequently, $Q^*(PAP)Q=EME=ME=Q^*(AP)Q$, and $PAP=AP$ by multiplication by $Q$ and $Q^*$.
\item[$(iii)$] $\Rightarrow (i)$ Suppose $PAP=AP$. Then $EME=ME$. Since $EME=\left[\begin{smallmatrix}B&0\\0&0\end{smallmatrix}\right]$ and $ME=\left[\begin{smallmatrix}B&0\\Y&0\end{smallmatrix}\right]$, one has $Y=0$.\\
If $\mathbf{u}\in\mathcal{U}$, choose $\mathbf{c}\in\mathbb{F}^r$ such that $\mathbf{u}=Q_1\mathbf{c}$. Then $Q_2^*A\mathbf{u}=Y\mathbf{c}=\mathbf{0}$, so $A\mathbf{u}\in\ker(Q_2^*)=\mathcal{U}$. Hence $A\mathcal{U}\subseteq\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: Unitary coordinates reduce all three conditions to vanishing of one block.\\
\hlt{Key Insight}: Invariance, a zero lower-left block, and $PAP=AP$ are three forms of the same condition.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:reducing-subspace-projection-commutation} \hlt{Reducing Subspaces and Commuting Projections}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$, let $\mathcal{U}\leq\mathbb{F}^n$ satisfy $\dim(\mathcal{U})=r$ and $1\leq r\leq n-1$. Let $Q=[Q_1\ Q_2]$ be unitary with $\operatorname{col}(Q_1)=\mathcal{U}$, $\operatorname{col}(Q_2)=\mathcal{U}^{\perp}$.\\
Set $P\defeq Q_1Q_1^*$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\mathcal{U}\subseteq\mathcal{U}$ and $A^*\mathcal{U}\subseteq\mathcal{U}$.
\item $(\exists\,B\in\operatorname{Mat}(r,r,\mathbb{F}))(\exists\,C\in\operatorname{Mat}(n-r,n-r,\mathbb{F}))\,Q^*AQ=B\oplus C$.
\item $PA=AP$.
\end{enumerate}
\begin{proof}
Set $M\defeq Q^*AQ=\left[\begin{smallmatrix}B&X\\Y&C\end{smallmatrix}\right]$ and $E\defeq Q^*PQ=I_r\oplus0_{n-r}$. Then $Q^*A^*Q=M^*$, $Q^*(PA)Q=EM$, and $Q^*(AP)Q=ME$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Suppose $A\mathcal{U}\subseteq\mathcal{U}$ and $A^*\mathcal{U}\subseteq\mathcal{U}$. Then $Y=0$ by Theorem \ref{thm:matrix-invariant-subspace-projection-block-criteria}, (i)$\Rightarrow$(ii), for $A$.\\
Similarly, $X^*=0$ by the same implication for $A^*$, because $X^*$ is the lower-left block of $M^*$. Thus $X=Y=0$ and $M=B\oplus C$.
\item[$(ii)$] $\Rightarrow (iii)$ Suppose $M=B\oplus C$. Then $EM=ME$ by block multiplication, so $PA=AP$ by conjugation with $Q$.
\item[$(iii)$] $\Rightarrow (i)$ Suppose $PA=AP$. Then $EM=ME$. Since $EM=\left[\begin{smallmatrix}B&X\\0&0\end{smallmatrix}\right]$ and $ME=\left[\begin{smallmatrix}B&0\\Y&0\end{smallmatrix}\right]$, one has $X=Y=0$. Then $A\mathcal{U}\subseteq\mathcal{U}$ by Theorem \ref{thm:matrix-invariant-subspace-projection-block-criteria}, (ii)$\Rightarrow$(i). Since $X^*=0$ is the lower-left block of $M^*$, one likewise has $A^*\mathcal{U}\subseteq\mathcal{U}$ by the same implication for $A^*$.
\end{enumerate}
\hlt{Structural Commentary}: The invariant-subspace criterion for $A$ and $A^*$ controls the two opposite off-diagonal blocks.\\
\hlt{Key Insight}: A subspace reduces $A$ exactly when its orthogonal projection commutes with $A$.
\end{proof}
\end{corollary}

Proposition \ref{prop:invariant-subspaces-and-block-form}, Theorem \ref{thm:matrix-invariant-subspace-projection-block-criteria}, and Corollary \ref{cor:reducing-subspace-projection-commutation} own these claims.

\begin{perspective} \label{per:restriction-quotient-perspective} \hlt{Two Smaller Views of an Operator}\\
For $\mathcal{U}\in\operatorname{Inv}(T)$, Definition \ref{def:restriction-and-quotient-operators} separates the action inside $\mathcal{U}$ from the action after $\mathcal{U}$ is collapsed, and Proposition \ref{prop:restriction-and-quotient-operators-well-defined} validates both induced operators. In the adapted block form of Proposition \ref{prop:invariant-subspaces-and-block-form}(i), the restriction and quotient operators record the diagonal blocks but not the upper-right block. Thus they simplify the study of $T$ without necessarily determining $T$.
\end{perspective}
