\subsection{Inner Products and Norms}

% **Role:** linear algebra with metric and angle structure.
%
% Inner products, norms, Cauchy--Schwarz, orthogonality, orthonormal bases, Gram--Schmidt, Gram matrices.

Throughout, $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$. This subsection develops the norm and metric geometry determined by an inner product.

\begin{definition} \label{def:inner-product-preliminary-terminology} \hlt{Preliminary Terminology}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{x},\mathbf{y}\in\mathbb{R}^n$, the \hlt{dot product} is $\mathbf{x}\cdot\mathbf{y}\defeq x_1y_1+\cdots+x_ny_n$.
\item For $z=a+bi\in\mathbb{C}$ with $a,b\in\mathbb{R}$, its \hlt{complex conjugate} and \hlt{modulus} are $\overline{z}\defeq a-bi$ and $\abs{z}\defeq\sqrt{a^2+b^2}$.
\item A map $f:\mathcal{V}\to\mathbb{F}$ is \hlt{conjugate linear} if $(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})(\forall\,\lambda\in\mathbb{F})\,\bigl(f(\mathbf{u}+\mathbf{v})=f(\mathbf{u})+f(\mathbf{v})\land f(\lambda\mathbf{v})=\overline{\lambda}f(\mathbf{v})\bigr)$.
\item A map $B:\mathcal{V}\times\mathcal{V}\to\mathbb{F}$ is \hlt{sesquilinear} if it is linear in its first slot and conjugate linear in its second slot.
\end{enumerate}
\end{definition}

\begin{motivation} \label{mot:inner-product-from-dot-product} \hlt{From Dot Products to Inner Products}\\
For $\mathbf{x}\in\mathbb{R}^n$, Euclidean squared length is $\mathbf{x}\cdot\mathbf{x}=\sum\limits_{j=1}^{n}x_j^2$.\\
For $z\in\mathbb{C}$, the scalar $z^2$ need not be real, whereas $z\overline{z}=\abs{z}^2\geq0$.\\
Conjugating one factor therefore preserves nonnegative squared length and motivates the following axioms.
\end{motivation}

\begin{definition} \label{def:inner-product} \hlt{Inner Product}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. An \hlt{inner product} on $\mathcal{V}$ is a map $\innerprod{\cdot,\cdot}:\mathcal{V}\times\mathcal{V}\to\mathbb{F}$ satisfying, for $(\mathbf{u},\mathbf{v},\mathbf{w},\lambda)\in\mathcal{V}^3\times\mathbb{F}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{positivity}: $\innerprod{\mathbf{v},\mathbf{v}}\in[0,\infty)$;
\item \hlt{definiteness}: $\innerprod{\mathbf{v},\mathbf{v}}=0\Longleftrightarrow\mathbf{v}=\mathbf{0}$;
\item \hlt{additivity in the first slot}: $\innerprod{\mathbf{u}+\mathbf{v},\mathbf{w}}=\innerprod{\mathbf{u},\mathbf{w}}+\innerprod{\mathbf{v},\mathbf{w}}$;
\item \hlt{homogeneity in the first slot}: $\innerprod{\lambda\mathbf{u},\mathbf{v}}=\lambda\innerprod{\mathbf{u},\mathbf{v}}$;
\item \hlt{conjugate symmetry}: $\innerprod{\mathbf{u},\mathbf{v}}=\overline{\innerprod{\mathbf{v},\mathbf{u}}}$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:inner-product-space} \hlt{Inner-Product Space}\\
An \hlt{inner-product space} is a vector space $\mathcal{V}$ together with an inner product on $\mathcal{V}$.
\end{definition}

Henceforth, unless stated otherwise, $\mathcal{V}$ denotes an inner-product space over $\mathbb{F}$.

\begin{definition} \label{def:norm-induced-by-inner-product} \hlt{Norm Induced by an Inner Product}\\
For $\mathbf{v}\in\mathcal{V}$, the \hlt{norm} of $\mathbf{v}$ is $\norm{\mathbf{v}}\defeq\sqrt{\innerprod{\mathbf{v},\mathbf{v}}}$.
\end{definition}

\begin{example} \label{ex:standard-inner-product-on-fn} \hlt{Standard Inner Product on $\mathbb{F}^n$}\\
For $\mathbf{x},\mathbf{y}\in\mathbb{F}^n$, set $\innerprod{\mathbf{x},\mathbf{y}}\defeq\sum\limits_{j=1}^{n}x_j\overline{y_j}$. Sesquilinearity and conjugate symmetry hold componentwise, while $\innerprod{\mathbf{x},\mathbf{x}}=\sum\limits_{j=1}^{n}\abs{x_j}^2$ verifies positivity and definiteness. For $\mathbb{F}=\mathbb{R}$, this is the dot product.
\end{example}

\begin{example} \label{ex:function-and-matrix-inner-products} \hlt{Function and Matrix Inner Products}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $a<b$, then $\innerprod{f,g}\defeq\int_a^b f(t)\overline{g(t)}\,dt$ is an inner product on $C_{\mathbb{F}}[a,b]$.\\
Positivity and definiteness follow from $\innerprod{f,f}=\int_a^b\abs{f(t)}^2\,dt\geq0$, with equality only for $f=0$ by continuity.
\item On $\operatorname{Mat}(m,n,\mathbb{F})$, the \hlt{Frobenius inner product} is $\innerprod[F]{A,B}\defeq\sum\limits_{i=1}^{m}\sum\limits_{j=1}^{n}a_{ij}\overline{b_{ij}}$.\\
Sesquilinearity holds entrywise, and $\innerprod[F]{A,A}=\sum\limits_{i=1}^{m}\sum\limits_{j=1}^{n}\abs{a_{ij}}^2$ is nonnegative and vanishes exactly when $A=0$.
\end{enumerate}
\end{example}

\begin{example} \label{ex:square-summable-sequence-inner-product} \hlt{Square-Summable Sequences}\\
Set $\ell^2(\mathbb{F})\defeq\left\{(x_j)_{j\geq1}:\sum\limits_{j=1}^{\infty}\abs{x_j}^2<\infty\right\}$ and $\innerprod{\mathbf{x},\mathbf{y}}\defeq\sum\limits_{j=1}^{\infty}x_j\overline{y_j}$. The bound $\abs{x_j\overline{y_j}}\leq\frac{1}{2}(\abs{x_j}^2+\abs{y_j}^2)$ ensures absolute convergence.\\
Termwise algebra verifies sesquilinearity and conjugate symmetry, while $\innerprod{\mathbf{x},\mathbf{x}}=\sum\limits_{j=1}^{\infty}\abs{x_j}^2$ verifies positivity and definiteness.\\
Thus $\ell^2(\mathbb{F})$ is an inner-product space containing $\mathbb{F}^{(\mathbb{N})}$.
\end{example}

\begin{proposition} \label{prop:basic-inner-product-identities} \hlt{Basic Inner-Product Identities}\\
For $(\mathbf{u},\mathbf{v},\mathbf{w},\lambda)\in\mathcal{V}^3\times\mathbb{F}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item for fixed $\mathbf{u}$, the map $\mathbf{v}\mapsto \innerprod{\mathbf{v},\mathbf{u}}$ is linear;
\item $\innerprod{\mathbf{0},\mathbf{u}}=0$ and $\innerprod{\mathbf{u},\mathbf{0}}=0$;
\item $\innerprod{\mathbf{u},\mathbf{v}+\mathbf{w}}=\innerprod{\mathbf{u},\mathbf{v}}+\innerprod{\mathbf{u},\mathbf{w}}$;
\item $\innerprod{\mathbf{u},\lambda\mathbf{v}}=\overline{\lambda}\innerprod{\mathbf{u},\mathbf{v}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The map $\mathbf{v}\mapsto\innerprod{\mathbf{v},\mathbf{u}}$ is linear by first-slot additivity and homogeneity.
\item $\innerprod{\mathbf{0},\mathbf{u}}=0$ by (i), and $\innerprod{\mathbf{u},\mathbf{0}}=\overline{\innerprod{\mathbf{0},\mathbf{u}}}=0$ by conjugate symmetry.
\item $\innerprod{\mathbf{u},\mathbf{v}+\mathbf{w}}=\overline{\innerprod{\mathbf{v}+\mathbf{w},\mathbf{u}}}=\overline{\innerprod{\mathbf{v},\mathbf{u}}}+\overline{\innerprod{\mathbf{w},\mathbf{u}}}=\innerprod{\mathbf{u},\mathbf{v}}+\innerprod{\mathbf{u},\mathbf{w}}$ by conjugate symmetry and additivity.
\item $\innerprod{\mathbf{u},\lambda\mathbf{v}}=\overline{\innerprod{\lambda\mathbf{v},\mathbf{u}}}=\overline{\lambda\innerprod{\mathbf{v},\mathbf{u}}}=\overline{\lambda}\innerprod{\mathbf{u},\mathbf{v}}$ by conjugate symmetry and homogeneity.
\end{enumerate}
\hlt{Structural Commentary}: Conjugate symmetry transfers the first-slot axioms to the second slot.\\
\hlt{Key Insight}: One linear slot determines conjugate linearity in the other.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:inner-products-separate-vectors} \hlt{Inner Products Separate Vectors}\\
For $\mathbf{u},\mathbf{v},\mathbf{w}\in\mathcal{V}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\bigl((\forall\,\mathbf{x}\in\mathcal{V})\,\innerprod{\mathbf{x},\mathbf{u}}=\innerprod{\mathbf{x},\mathbf{v}}\bigr)\Longrightarrow\mathbf{u}=\mathbf{v}$;
\item $\bigl((\forall\,\mathbf{x}\in\mathcal{V})\,\innerprod{\mathbf{x},\mathbf{w}}=0\bigr)\Longleftrightarrow\mathbf{w}=\mathbf{0}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] With $\mathbf{x}=\mathbf{u}-\mathbf{v}$, $\norm{\mathbf{u}-\mathbf{v}}^2=\innerprod{\mathbf{u}-\mathbf{v},\mathbf{u}}-\innerprod{\mathbf{u}-\mathbf{v},\mathbf{v}}=0$ by Proposition \ref{prop:basic-inner-product-identities}. The equality $\mathbf{u}=\mathbf{v}$ holds by definiteness.
\item[$(ii)$] ($\Rightarrow$) $\norm{\mathbf{w}}^2=\innerprod{\mathbf{w},\mathbf{w}}=0$ by the hypothesis with $\mathbf{x}=\mathbf{w}$, and $\mathbf{w}=\mathbf{0}$ by definiteness.
\item[$(ii)$] ($\Leftarrow$) If $\mathbf{w}=\mathbf{0}$, then $(\forall\,\mathbf{x}\in\mathcal{V})\,\innerprod{\mathbf{x},\mathbf{w}}=0$ by Proposition \ref{prop:basic-inner-product-identities}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Each claim tests a difference vector against a strategically chosen argument.\\
\hlt{Key Insight}: Testing a vector against itself detects whether it vanishes.
\end{proof}
\end{lemma}

\begin{example} \label{ex:norms-induced-by-inner-products} \hlt{Norms from Standard Inner Products}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item On $\mathbb{F}^n$, the standard inner product induces the \hlt{Euclidean norm} $\norm{(x_1,\ldots,x_n)}_2=\sqrt{\sum\limits_{j=1}^{n}\abs{x_j}^2}$.
\item On $C_{\mathbb{F}}[a,b]$ with $a<b$, the integral inner product induces the \hlt{$L^2$ norm} $\norm{f}_2=\sqrt{\int_a^b\abs{f(t)}^2\,dt}$.
\item On $\operatorname{Mat}(m,n,\mathbb{F})$, the Frobenius inner product induces $\norm{A}_F=\sqrt{\innerprod[F]{A,A}}=\sqrt{\sum\limits_{i=1}^{m}\sum\limits_{j=1}^{n}\abs{a_{ij}}^2}$ for $A=[a_{ij}]$.\\
This is the \hlt{Frobenius}, \hlt{Schur}, or \hlt{Hilbert--Schmidt norm}.
\end{enumerate}
\end{example}

\begin{proposition} \label{prop:derived-norm-basic-properties} \hlt{Basic Properties of the Derived Norm}\\
For $(\mathbf{v},\lambda)\in\mathcal{V}\times\mathbb{F}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\norm{\mathbf{v}}$ is real and nonnegative;
\item $\norm{\mathbf{v}}=0\Longleftrightarrow\mathbf{v}=\mathbf{0}$;
\item $\norm{\lambda\mathbf{v}}=\abs{\lambda}\norm{\mathbf{v}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] The scalar $\norm{\mathbf{v}}=\sqrt{\innerprod{\mathbf{v},\mathbf{v}}}$ is real and nonnegative by positivity.
\item[$(ii)$] ($\Rightarrow$) If $\norm{\mathbf{v}}=0$, then $\innerprod{\mathbf{v},\mathbf{v}}=0$, and $\mathbf{v}=\mathbf{0}$ by definiteness.
\item[$(ii)$] ($\Leftarrow$) If $\mathbf{v}=\mathbf{0}$, then $\innerprod{\mathbf{v},\mathbf{v}}=0$ by definiteness and $\norm{\mathbf{v}}=0$ by the norm definition.
\item[$(iii)$] $\norm{\lambda\mathbf{v}}^2=\innerprod{\lambda\mathbf{v},\lambda\mathbf{v}}=\lambda\overline{\lambda}\innerprod{\mathbf{v},\mathbf{v}}=\abs{\lambda}^2\norm{\mathbf{v}}^2$ by sesquilinearity. The equality $\norm{\lambda\mathbf{v}}=\abs{\lambda}\norm{\mathbf{v}}$ holds by nonnegative square roots.
\end{enumerate}
\hlt{Structural Commentary}: Positivity controls the square root, while sesquilinearity controls scalar rescaling.\\
\hlt{Key Insight}: Squared length is easier to manipulate than length itself.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:cauchy-schwarz-inequality} \hlt{Cauchy--Schwarz Inequality}\\
For $\mathbf{u},\mathbf{v}\in\mathcal{V}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\abs{\innerprod{\mathbf{u},\mathbf{v}}}\leq\norm{\mathbf{u}}\norm{\mathbf{v}}$;
\item equality in (i) holds if and only if $\mathbf{u},\mathbf{v}$ are linearly dependent.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] If $\mathbf{v}=\mathbf{0}$, the inequality is an equality. Suppose $\mathbf{v}\neq\mathbf{0}$ and set $\lambda\defeq\frac{\innerprod{\mathbf{u},\mathbf{v}}}{\norm{\mathbf{v}}^2}$.\\
Then $0\leq\norm{\mathbf{u}-\lambda\mathbf{v}}^2=\norm{\mathbf{u}}^2-\frac{\abs{\innerprod{\mathbf{u},\mathbf{v}}}^2}{\norm{\mathbf{v}}^2}$ by positivity, sesquilinearity, and conjugate symmetry.\\
The inequality $\abs{\innerprod{\mathbf{u},\mathbf{v}}}\leq\norm{\mathbf{u}}\norm{\mathbf{v}}$ follows by multiplication by $\norm{\mathbf{v}}^2$ and nonnegative square roots.
\item[$(ii)$] ($\Rightarrow$) If equality holds and $\mathbf{v}=\mathbf{0}$, then $\mathbf{u},\mathbf{v}$ are linearly dependent.\\
If $\mathbf{v}\neq\mathbf{0}$, then $\norm{\mathbf{u}-\lambda\mathbf{v}}=0$ by equality in (i) and the preceding identity, and $\mathbf{u}=\lambda\mathbf{v}$ by Proposition \ref{prop:derived-norm-basic-properties}(ii).
\item[$(ii)$] ($\Leftarrow$) If $\mathbf{v}=\mathbf{0}$, equality holds. Else linear dependence has form $\mathbf{u}=\lambda\mathbf{v}$; $\abs{\innerprod{\mathbf{u},\mathbf{v}}}=\abs{\lambda}\norm{\mathbf{v}}^2=\norm{\mathbf{u}}\norm{\mathbf{v}}$ by Proposition \ref{prop:derived-norm-basic-properties}(iii).
\end{enumerate}
\hlt{Structural Commentary}: A least-residual scalar supplies the bound and characterizes equality.\\
\hlt{Key Insight}: Positivity controls the mixed term through optimal scalar comparison.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:triangle-inequality-inner-product} \hlt{Triangle Inequality}\\
For $\mathbf{u},\mathbf{v}\in\mathcal{V}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\norm{\mathbf{u}+\mathbf{v}}\leq\norm{\mathbf{u}}+\norm{\mathbf{v}}$;
\item equality in (i) holds if and only if one of $\mathbf{u},\mathbf{v}$ is a nonnegative real multiple of the other.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\norm{\mathbf{u}+\mathbf{v}}^2=\norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2+2\operatorname{Re}\innerprod{\mathbf{u},\mathbf{v}}$ by sesquilinearity and conjugate symmetry.\\
By Theorem \ref{thm:cauchy-schwarz-inequality}(i), this is at most $\bigl(\norm{\mathbf{u}}+\norm{\mathbf{v}}\bigr)^2$. The inequality $\norm{\mathbf{u}+\mathbf{v}}\leq\norm{\mathbf{u}}+\norm{\mathbf{v}}$ follows by nonnegative square roots.
\item[$(ii)$] ($\Rightarrow$) If either vector is $\mathbf{0}$, it is a nonnegative multiple of the other.\\
Otherwise equality requires $\operatorname{Re}\innerprod{\mathbf{u},\mathbf{v}}=\abs{\innerprod{\mathbf{u},\mathbf{v}}}$ and equality in Cauchy--Schwarz.\\
Choose $a\in\mathbb{F}$ with $\mathbf{u}=a\mathbf{v}$ by Theorem \ref{thm:cauchy-schwarz-inequality}(ii). Then $a\in[0,\infty)$ by $a\norm{\mathbf{v}}^2=\innerprod{\mathbf{u},\mathbf{v}}\in[0,\infty)$.
\item[$(ii)$] ($\Leftarrow$) If $\mathbf{u}=a\mathbf{v}$ for $a\in[0,\infty)$, then $\norm{\mathbf{u}+\mathbf{v}}=(a+1)\norm{\mathbf{v}}=\norm{\mathbf{u}}+\norm{\mathbf{v}}$. Interchanging $\mathbf{u},\mathbf{v}$ in this calculation covers $\mathbf{v}=a\mathbf{u}$.
\end{enumerate}
\hlt{Structural Commentary}: Cauchy--Schwarz bounds the cross term, and its equality case fixes the direction.\\
\hlt{Key Insight}: Equality requires aligned vectors with nonnegative relative scale.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:reverse-triangle-inequality-inner-product} \hlt{Reverse Triangle Inequality}\\
$(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})\,\abs{\norm{\mathbf{u}}-\norm{\mathbf{v}}}\leq\norm{\mathbf{u}-\mathbf{v}}$.
\begin{proof}
$\norm{\mathbf{u}}-\norm{\mathbf{v}}\leq\norm{\mathbf{u}-\mathbf{v}}$ by Theorem \ref{thm:triangle-inequality-inner-product}(i) applied to $\mathbf{u}=(\mathbf{u}-\mathbf{v})+\mathbf{v}$.\\
Likewise, $\norm{\mathbf{v}}-\norm{\mathbf{u}}\leq\norm{\mathbf{u}-\mathbf{v}}$ by the same result with $\mathbf{u},\mathbf{v}$ interchanged.\\
The two bounds are equivalent to the asserted absolute-value inequality.\\
\hlt{Structural Commentary}: Two triangle-inequality bounds control both possible signs of the norm difference.\\
\hlt{Key Insight}: Distance bounds the change in length in either direction.
\end{proof}
\end{proposition}

\begin{definition} \label{def:norm-and-normed-vector-space} \hlt{Norm and Normed Vector Space}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. A \hlt{norm} on $\mathcal{V}$ is a map $\norm{\cdot}:\mathcal{V}\to[0,\infty)$ satisfying, for $(\mathbf{u},\mathbf{v},\lambda)\in\mathcal{V}^2\times\mathbb{F}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\norm{\mathbf{u}}\geq0$;
\item $\norm{\mathbf{u}}=0\Longleftrightarrow\mathbf{u}=\mathbf{0}$;
\item $\norm{\lambda\mathbf{u}}=\abs{\lambda}\norm{\mathbf{u}}$;
\item $\norm{\mathbf{u}+\mathbf{v}}\leq\norm{\mathbf{u}}+\norm{\mathbf{v}}$.
\end{enumerate}
A \hlt{normed vector space} is a vector space equipped with a norm. Its \hlt{unit ball} is $\{\mathbf{v}\in\mathcal{V}:\norm{\mathbf{v}}\leq1\}$.
\end{definition}

\begin{definition} \label{def:inner-product-distance} \hlt{Distance Induced by an Inner Product}\\
The \hlt{distance induced by the inner product} is $d:\mathcal{V}\times\mathcal{V}\to[0,\infty)$ defined by $d(\mathbf{u},\mathbf{v})\defeq\norm{\mathbf{u}-\mathbf{v}}$.
\end{definition}

\begin{proposition} \label{prop:inner-product-distance-is-metric} \hlt{Inner-Product Distance Is a Metric}\\
For $d(\mathbf{u},\mathbf{v})\defeq\norm{\mathbf{u}-\mathbf{v}}$ and $\mathbf{u},\mathbf{v},\mathbf{w}\in\mathcal{V}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $d(\mathbf{u},\mathbf{v})\geq0$;
\item $d(\mathbf{u},\mathbf{v})=0\Longleftrightarrow\mathbf{u}=\mathbf{v}$;
\item $d(\mathbf{u},\mathbf{v})=d(\mathbf{v},\mathbf{u})$;
\item $d(\mathbf{u},\mathbf{v})\leq d(\mathbf{u},\mathbf{w})+d(\mathbf{w},\mathbf{v})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $d(\mathbf{u},\mathbf{v})=\norm{\mathbf{u}-\mathbf{v}}\geq0$ by Proposition \ref{prop:derived-norm-basic-properties}(i).
\item[$(ii)$] ($\Rightarrow$) If $d(\mathbf{u},\mathbf{v})=0$, then $\mathbf{u}-\mathbf{v}=\mathbf{0}$ by Proposition \ref{prop:derived-norm-basic-properties}(ii); equivalently, $\mathbf{u}=\mathbf{v}$.
\item[$(ii)$] ($\Leftarrow$) If $\mathbf{u}=\mathbf{v}$, then $d(\mathbf{u},\mathbf{v})=\norm{\mathbf{0}}=0$ by Proposition \ref{prop:derived-norm-basic-properties}(ii).
\item[$(iii)$] $d(\mathbf{u},\mathbf{v})=\norm{-(\mathbf{v}-\mathbf{u})}=\norm{\mathbf{v}-\mathbf{u}}=d(\mathbf{v},\mathbf{u})$ by Proposition \ref{prop:derived-norm-basic-properties}(iii) with $\lambda=-1$.
\item[$(iv)$] Since $\mathbf{u}-\mathbf{v}=(\mathbf{u}-\mathbf{w})+(\mathbf{w}-\mathbf{v})$, $d(\mathbf{u},\mathbf{v})\leq d(\mathbf{u},\mathbf{w})+d(\mathbf{w},\mathbf{v})$ by Theorem \ref{thm:triangle-inequality-inner-product}(i).
\end{enumerate}
\hlt{Structural Commentary}: Each metric axiom reduces to the corresponding norm property on a difference vector.\\
\hlt{Key Insight}: Distances are norms of displacements.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:inner-product-metric-topology-finite-dimensional} \hlt{Metric Viewpoint}\\
Because $d(\mathbf{v}_n,\mathbf{v})=\norm{\mathbf{v}_n-\mathbf{v}}$, metric convergence is exactly norm convergence.\\
Closedness, continuity, and completeness require topological results developed later; no such conclusion is used as authority here.
\end{perspective}

\begin{theorem} \label{thm:parallelogram-identity-inner-product} \hlt{Parallelogram Identity}\\
$(\forall\,\mathbf{u},\mathbf{v}\in\mathcal{V})\,\norm{\mathbf{u}+\mathbf{v}}^2+\norm{\mathbf{u}-\mathbf{v}}^2=2\bigl(\norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2\bigr)$.
\begin{proof}
For $\varepsilon\in\{1,-1\}$, $\norm{\mathbf{u}+\varepsilon\mathbf{v}}^2=\norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2+2\varepsilon\operatorname{Re}\innerprod{\mathbf{u},\mathbf{v}}$ by sesquilinearity and conjugate symmetry.\\
The asserted identity is the sum of the cases $\varepsilon=1,-1$.\\
\hlt{Structural Commentary}: Paired expansions cancel the cross terms while preserving both squared norms.\\
\hlt{Key Insight}: Opposite diagonals erase the mixed inner-product contribution.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:coordinate-norms-on-fn} \hlt{Coordinate Norms on $\mathbb{F}^n$}\\
For $\mathbf{x}=(x_1,\ldots,x_n)\in\mathbb{F}^n$, each of the following defines a norm on $\mathbb{F}^n$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\norm{\mathbf{x}}_1\defeq\sum\limits_{j=1}^{n}\abs{x_j}$;
\item $\norm{\mathbf{x}}_2\defeq\left(\sum\limits_{j=1}^{n}\abs{x_j}^2\right)^{1/2}$;
\item $\norm{\mathbf{x}}_{\infty}\defeq\max\limits_{1\leq j\leq n}\abs{x_j}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Nonnegativity, definiteness, and absolute homogeneity of $\norm{\cdot}_1$ hold by the corresponding scalar properties.\\
Its triangle inequality is $\norm{\mathbf{x}+\mathbf{y}}_1=\sum\limits_{j=1}^{n}\abs{x_j+y_j}\leq\sum\limits_{j=1}^{n}(\abs{x_j}+\abs{y_j})=\norm{\mathbf{x}}_1+\norm{\mathbf{y}}_1$ by the scalar triangle inequality.
\item Standard inner product induces $\norm{\cdot}_2$, first $3$ norm axioms hold  (Proposition \ref{prop:derived-norm-basic-properties}), triangle inequality hold  (Theorem \ref{thm:triangle-inequality-inner-product}(i)).
\item Nonnegativity and absolute homogeneity of $\norm{\cdot}_{\infty}$ hold by modulus, and $\norm{\mathbf{x}}_{\infty}=0\Longleftrightarrow(\forall\,j\in\{1,\ldots,n\})\,x_j=0\Longleftrightarrow\mathbf{x}=\mathbf{0}$.\\
Choose $k\in\{1,\ldots,n\}$ with $\norm{\mathbf{x}+\mathbf{y}}_{\infty}=\abs{x_k+y_k}$.\\
Then $\norm{\mathbf{x}+\mathbf{y}}_{\infty}\leq\abs{x_k}+\abs{y_k}\leq\norm{\mathbf{x}}_{\infty}+\norm{\mathbf{y}}_{\infty}$ by the scalar triangle inequality and the definition of the maximum.
\end{enumerate}
\hlt{Structural Commentary}: Scalar modulus handles the $\ell_1$ and $\ell_\infty$ cases; inner-product geometry handles the $\ell_2$ case.\\
\hlt{Key Insight}: Distinct coordinate aggregations satisfy the same norm axioms.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:l1-linfty-not-inner-product-norms} \hlt{$\ell_1$ and $\ell_{\infty}$ Are Not Inner-Product Norms}\\
If $n\geq2$, then the norms $\norm{\cdot}_1$ and $\norm{\cdot}_{\infty}$ on $\mathbb{F}^n$ are not induced by any inner product.
\begin{proof}
For the first two coordinate vectors $\mathbf{e}_1,\mathbf{e}_2$, direct evaluation makes the two sides of the parallelogram identity $8$ and $4$ under $\norm{\cdot}_1$, and $2$ and $4$ under $\norm{\cdot}_{\infty}$. Neither norm is induced by an inner product by Theorem \ref{thm:parallelogram-identity-inner-product}.\\
\hlt{Structural Commentary}: Two coordinate tests contradict the necessary parallelogram identity.\\
\hlt{Key Insight}: A failed parallelogram identity excludes every inducing inner product.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:polarization-identities} \hlt{Polarization Identities}\\
For $\mathbf{u},\mathbf{v}\in\mathcal{V}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathbb{F}=\mathbb{R}$, then $\innerprod{\mathbf{u},\mathbf{v}}=\frac{1}{4}\bigl(\norm{\mathbf{u}+\mathbf{v}}^2-\norm{\mathbf{u}-\mathbf{v}}^2\bigr)$.
\item If $\mathbb{F}=\mathbb{C}$, then $\innerprod{\mathbf{u},\mathbf{v}}=\frac{1}{4}\bigl(\norm{\mathbf{u}+\mathbf{v}}^2-\norm{\mathbf{u}-\mathbf{v}}^2+i\norm{\mathbf{u}+i\mathbf{v}}^2-i\norm{\mathbf{u}-i\mathbf{v}}^2\bigr)$.
\end{enumerate}
\begin{proof}
Let $z\defeq\innerprod{\mathbf{u},\mathbf{v}}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathbb{F}=\mathbb{R}$, then $\norm{\mathbf{u}+\mathbf{v}}^2-\norm{\mathbf{u}-\mathbf{v}}^2=4\operatorname{Re}z=4z$ by sesquilinearity, which is equivalent to (i).
\item If $\mathbb{F}=\mathbb{C}$, then $\norm{\mathbf{u}+\mathbf{v}}^2-\norm{\mathbf{u}-\mathbf{v}}^2=4\operatorname{Re}z$ and $\norm{\mathbf{u}+i\mathbf{v}}^2-\norm{\mathbf{u}-i\mathbf{v}}^2=4\operatorname{Im}z$ by sesquilinearity.\\
Substitution into $z=\operatorname{Re}z+i\operatorname{Im}z$ is exactly (ii).
\end{enumerate}
\hlt{Structural Commentary}: Norm differences recover the real part directly and the imaginary part after rotation.\\
\hlt{Key Insight}: The norm determines both components of every complex inner product.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:operator-zero-tests-inner-products} \hlt{Operator Zero Tests from Inner Products}\\
Let $T\in\LL(\mathcal{V})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\bigl((\forall\,\mathbf{v},\mathbf{w}\in\mathcal{V})\,\innerprod{T\mathbf{v},\mathbf{w}}=0\bigr)\Longrightarrow T=0$.
\item If $\mathbb{F}=\mathbb{C}$, then $\bigl((\forall\,\mathbf{v}\in\mathcal{V})\,\innerprod{T\mathbf{v},\mathbf{v}}=0\bigr)\Longrightarrow T=0$.
\item If $\mathbb{F}=\mathbb{R}$, the implication in (ii) can fail.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,\mathbf{v}\in\mathcal{V})\,\norm{T\mathbf{v}}^2=\innerprod{T\mathbf{v},T\mathbf{v}}=0$ by the hypothesis with $\mathbf{w}=T\mathbf{v}$; $T=0$ by definiteness.
\item Set $Q(\mathbf{x})\defeq\innerprod{T\mathbf{x},\mathbf{x}}$. For fixed $\mathbf{x},\mathbf{y}\in\mathcal{V}$, set $a\defeq\innerprod{T\mathbf{x},\mathbf{y}}$ and $b\defeq\innerprod{T\mathbf{y},\mathbf{x}}$.\\
For $r\in\mathbb{C}$, $0=Q(r\mathbf{x}+\mathbf{y})=ra+\overline{r}b$ by the hypothesis and sesquilinearity.\\
The choices $r=1,i$ establish $a+b=a-b=0$, and $a=b=0$ by these equations.\\
The identity $(\forall\,\mathbf{x},\mathbf{y}\in\mathcal{V})\,\innerprod{T\mathbf{x},\mathbf{y}}=0$ holds by the arbitrary choice of $\mathbf{x},\mathbf{y}$, and $T=0$ by (i).
\item On standard $\mathbb{R}^2$, define $T(x_1,x_2)\defeq(-x_2,x_1)$. Then $(\forall\,\mathbf{v}\in\mathbb{R}^2)\,\innerprod{T\mathbf{v},\mathbf{v}}=-v_2v_1+v_1v_2=0$, while $T(1,0)=(0,1)\neq\mathbf{0}$.
\end{enumerate}
\hlt{Structural Commentary}: Self-testing handles (i), complex scalar probes handle (ii), and rotation refutes the real analogue.\\
\hlt{Key Insight}: Complex phases recover mixed terms that real diagonal tests miss.
\end{proof}
\end{proposition}

\begin{principle} \label{pri:inner-product-geometry} \hlt{Geometry from Algebra}\\
For inner-product norms, expanding $\norm{\mathbf{u}\pm\mathbf{v}}^2$ exposes the cross terms that control angles, inequalities, and polarization. This heuristic does not apply to arbitrary norms, as the $\ell_1$ and $\ell_\infty$ countertests demonstrate.
\end{principle}
