\subsection{Gram Matrices}

Throughout this subsection, $\mathbb{F}$ is $\mathbb{R}$ or $\mathbb{C}$ and $\mathcal{V}$ is an inner-product space over $\mathbb{F}$.

\begin{definition} \label{def:gram-matrix} \hlt{Gram Matrix}\\
For $n\geq1$ and $\mathbf{u}_1,\ldots,\mathbf{u}_n\in\mathcal{V}$, the \hlt{Gram matrix} is $\operatorname{Gram}(\mathbf{u}_1,\ldots,\mathbf{u}_n)\defeq[\innerprod{\mathbf{u}_j,\mathbf{u}_i}]_{i,j=1}^{n}\in\operatorname{Mat}(n,n,\mathbb{F})$.
\end{definition}

\begin{proposition} \label{prop:gram-matrix-positive-semidefinite} \hlt{Gram Matrix Quadratic Form}\\
Let $n\geq1$, let $\mathbf{u}_1,\ldots,\mathbf{u}_n\in\mathcal{V}$, and set $G=[g_{ij}]\defeq\operatorname{Gram}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,i,j\in\{1,\ldots,n\})\,g_{ij}=\overline{g_{ji}}$, so $G$ is Hermitian;
\item $(\forall\,\mathbf{c}=(c_1,\ldots,c_n)^t\in\mathbb{F}^n)\,\mathbf{c}^*G\mathbf{c}=\norm{\sum\limits_{j=1}^{n}c_j\mathbf{u}_j}^2\geq0$;
\item if $\mathbf{u}_1,\ldots,\mathbf{u}_n$ are linearly independent, then $(\forall\,\mathbf{c}\in\mathbb{F}^n\setminus\{\mathbf{0}\})\,\mathbf{c}^*G\mathbf{c}>0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $i,j\in\{1,\ldots,n\}$, $g_{ij}=\innerprod{\mathbf{u}_j,\mathbf{u}_i}=\overline{\innerprod{\mathbf{u}_i,\mathbf{u}_j}}=\overline{g_{ji}}$ by conjugate symmetry.
\item Fix $\mathbf{c}=(c_1,\ldots,c_n)^t\in\mathbb{F}^n$ and set $\mathbf{x}\defeq\sum\limits_{j=1}^{n}c_j\mathbf{u}_j$.\\
By sesquilinearity and definiteness, $\mathbf{c}^*G\mathbf{c}=\sum\limits_{i=1}^{n}\sum\limits_{j=1}^{n}\overline{c_i}\innerprod{\mathbf{u}_j,\mathbf{u}_i}c_j=\innerprod{\mathbf{x},\mathbf{x}}=\norm{\mathbf{x}}^2\geq0$.
\item Suppose $\mathbf{u}_1,\ldots,\mathbf{u}_n$ are linearly independent and fix $\mathbf{c}=(c_1,\ldots,c_n)^t\neq\mathbf{0}$.\\
Then $\sum\limits_{j=1}^{n}c_j\mathbf{u}_j\neq\mathbf{0}$ by linear independence, so $\mathbf{c}^*G\mathbf{c}>0$ by (ii) and definiteness.
\end{enumerate}
\hlt{Structural Commentary}: Entry symmetry and a norm-square expansion establish Hermitian positivity; independence makes it strict.\\
\hlt{Key Insight}: Gram quadratic forms measure norms of coefficient-weighted vector sums.
\end{proof}
\end{proposition}
