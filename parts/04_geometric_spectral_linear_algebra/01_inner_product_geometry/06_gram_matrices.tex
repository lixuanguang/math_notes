\subsection{Gram Matrices}

\begin{definition} \label{def:gram-matrix} \hlt{Gram Matrix}\\
For vectors $\mathbf{u}_1,\ldots,\mathbf{u}_n$ in an inner-product space, the \hlt{Gram matrix} is
\begin{align}
\operatorname{Gram}(\mathbf{u}_1,\ldots,\mathbf{u}_n)
\defeq
\begin{bmatrix}
\innerprod{\mathbf{u}_1,\mathbf{u}_1} & \innerprod{\mathbf{u}_2,\mathbf{u}_1} & \cdots & \innerprod{\mathbf{u}_n,\mathbf{u}_1}\\
\innerprod{\mathbf{u}_1,\mathbf{u}_2} & \innerprod{\mathbf{u}_2,\mathbf{u}_2} & \cdots & \innerprod{\mathbf{u}_n,\mathbf{u}_2}\\
\vdots & \vdots & \ddots & \vdots\\
\innerprod{\mathbf{u}_1,\mathbf{u}_n} & \innerprod{\mathbf{u}_2,\mathbf{u}_n} & \cdots & \innerprod{\mathbf{u}_n,\mathbf{u}_n}
\end{bmatrix}.
\nonumber
\end{align}
\end{definition}

\begin{proposition} \label{prop:gram-matrix-positive-semidefinite} \hlt{Gram Matrix Quadratic Form}\\
For any vectors $\mathbf{u}_1,\ldots,\mathbf{u}_n$, Gram matrix $G=\operatorname{Gram}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ is Hermitian and satisfies $\mathbf{c}^*G\mathbf{c} =\norm{\sum\limits_{j=1}^{n}c_j\mathbf{u}_j}^2$ for every $\mathbf{c}=[c_1\ \cdots\ c_n]^t\in\mathbb{F}^n$. Hence the Gram quadratic form is nonnegative. If $\mathbf{u}_1,\ldots,\mathbf{u}_n$ are linearly independent, then it is positive for every nonzero $\mathbf{c}$.
\begin{proof}
Hermitian symmetry follows from conjugate symmetry of the inner product. Expanding the squared norm gives $\norm{\sum\limits_{j=1}^{n}c_j\mathbf{u}_j}^2 =\sum\limits_{i,j}\overline{c_i}\innerprod{\mathbf{u}_j,\mathbf{u}_i}c_j =\mathbf{c}^*G\mathbf{c}$.
The right-hand side is a squared norm, so it is nonnegative. If $\mathbf{u}_1,\ldots,\mathbf{u}_n$ are linearly independent, the squared norm is $0$ only when all $c_j=0$.\\
\hlt{Structural Commentary}: A Gram matrix records every pairwise inner product in a list of vectors.\\
\hlt{Key Insight}: The quadratic form of a Gram matrix is exactly the squared norm of a linear combination.
\end{proof}
\end{proposition}
