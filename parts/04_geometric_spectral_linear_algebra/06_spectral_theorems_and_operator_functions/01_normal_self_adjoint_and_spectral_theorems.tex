\subsection{Normal, Self-Adjoint, and Spectral Theorems}

% **Role:** structure revealed by invariant subspaces and eigenbehavior.
%
% Normal and self-adjoint operators; Schur and spectral theorems; spectral resolutions.

Structured spectral theory studies operators whose geometry and spectrum align.\\
This subsection derives orthogonal eigenbasis and projection resolutions from normality and self-adjointness.\\
Throughout this subsection, inner-product spaces are finite-dimensional over $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$ unless the field is specified explicitly.

\begin{definition} \label{def:normal-operator} \hlt{Normal Operator}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item An operator $T\in\LL(\mathcal{V})$ is \hlt{normal} iff $T^*T=TT^*$.
\item For $n\geq1$, a matrix $A\in\operatorname{Mat}(n,n,\mathbb{F})$ is \hlt{normal} iff $A^*A=AA^*$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:normal-iff-equal-adjoint-norms} \hlt{Normal Operators and Equal Adjoint Norms}\\
Let $T\in\LL(\mathcal{V})$. Then $T$ is normal iff $(\forall\mathbf{v}\in\mathcal{V})\,\norm{T\mathbf{v}}=\norm{T^*\mathbf{v}}$.
\begin{proof}
Set $C\defeq T^*T-TT^*$. Then $C^*=C$ by Proposition \ref{prop:basic-adjoint-identities}(i)--(iii), and $(\forall\mathbf{v}\in\mathcal{V})\,\innerprod{C\mathbf{v},\mathbf{v}}=\norm{T\mathbf{v}}^2-\norm{T^*\mathbf{v}}^2$ by adjoint identity.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $T$ is normal, then $C=0$ by Definition \ref{def:normal-operator}(i), so the norm identity holds on $\mathcal{V}$ by the preceding equality.
\item[$(\Leftarrow)$] If the norm identity holds on $\mathcal{V}$, then $(\forall\mathbf{v}\in\mathcal{V})\,\innerprod{C\mathbf{v},\mathbf{v}}=0$. Thus $C=0$ by Proposition \ref{prop:self-adjoint-zero-quadratic-form-implies-zero}, $T$ is normal by Definition \ref{def:normal-operator}(i).
\end{enumerate}
\hlt{Structural Commentary}: The proof converts the commutator to a quadratic form and recovers the operator from it.\\
\hlt{Key Insight}: Equality of the two norm functions detects the vanishing of $T^*T-TT^*$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:normal-operator-adjoint-eigenvectors} \hlt{Normal Operators Share Eigenvectors with Their Adjoints}\\
Let $T\in\LL(\mathcal{V})$ be normal, let $\lambda\in\mathbb{F}$, and let $\mathbf{v}\in\mathcal{V}\setminus\{\mathbf{0}\}$. If $T\mathbf{v}=\lambda\mathbf{v}$, then $T^*\mathbf{v}=\overline{\lambda}\mathbf{v}$.
\begin{proof}
Set $S\defeq T-\lambda I_{\mathcal{V}}$. Then $S^*=T^*-\overline{\lambda}I_{\mathcal{V}}$ by Proposition \ref{prop:basic-adjoint-identities}(i),(iv), and $S^*S=SS^*$ by direct expansion with $T^*T=TT^*$.\\
Thus $S$ is normal. Since $S\mathbf{v}=\mathbf{0}$, $\norm{S^*\mathbf{v}}=\norm{S\mathbf{v}}=0$ by Theorem \ref{thm:normal-iff-equal-adjoint-norms}.\\
Hence $S^*\mathbf{v}=\mathbf{0}$ by positive definiteness, which is equivalent to $T^*\mathbf{v}=\overline{\lambda}\mathbf{v}$.\\
\hlt{Structural Commentary}: Scalar shifting preserves normality, so the equal-norm criterion transfers the kernel of $T-\lambda I$ to its adjoint.\\
\hlt{Key Insight}: Normality transfers the shifted kernel to the adjoint shift.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:normal-eigenvectors-distinct-eigenvalues-orthogonal} \hlt{Orthogonal Eigenvectors for Normal Operators}\\
Let $T\in\LL(\mathcal{V})$ be normal, let $\alpha,\beta\in\operatorname{spec}(T)$ satisfy $\alpha\neq\beta$, and let $\mathbf{u}\in E_{\alpha}(T)\setminus\{\mathbf{0}\}$ and $\mathbf{v}\in E_{\beta}(T)\setminus\{\mathbf{0}\}$. Then $\innerprod{\mathbf{u},\mathbf{v}}=0$.
\begin{proof}
The identities $T\mathbf{u}=\alpha\mathbf{u}$ and $T\mathbf{v}=\beta\mathbf{v}$ hold by eigenspace membership, while $T^*\mathbf{v}=\overline{\beta}\mathbf{v}$ by Theorem \ref{thm:normal-operator-adjoint-eigenvectors}.\\
Thus $\alpha\innerprod{\mathbf{u},\mathbf{v}}=\innerprod{T\mathbf{u},\mathbf{v}}=\innerprod{\mathbf{u},T^*\mathbf{v}}=\beta\innerprod{\mathbf{u},\mathbf{v}}$. Since $\alpha\neq\beta$, $\innerprod{\mathbf{u},\mathbf{v}}=0$.\\
\hlt{Structural Commentary}: The adjoint-eigenvector theorem moves $T$ across the inner product while preserving the relevant eigenline.\\
\hlt{Key Insight}: The factor $\alpha-\beta$ can annihilate the inner product only through orthogonality.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:self-adjoint-eigenvalues-real} \hlt{Eigenvalues of Self-Adjoint Operators Are Real}\\
Let $T\in\LL(\mathcal{V})$ be self-adjoint. Then $\operatorname{spec}(T)\subseteq\mathbb{R}$.
\begin{proof}
Fix $\lambda\in\operatorname{spec}(T)$ and $\mathbf{v}\in E_{\lambda}(T)\setminus\{\mathbf{0}\}$. Then $\lambda\norm{\mathbf{v}}^2=\innerprod{T\mathbf{v},\mathbf{v}}=\innerprod{\mathbf{v},T\mathbf{v}}=\overline{\lambda}\norm{\mathbf{v}}^2$ by the eigenvector relation and $T=T^*$.\\
Since $\norm{\mathbf{v}}^2>0$, $\lambda=\overline{\lambda}$ and hence $\lambda\in\mathbb{R}$.\\
\hlt{Structural Commentary}: The proof evaluates self-adjointness on an eigenvector and compares the eigenvalue with its conjugate.\\
\hlt{Key Insight}: Positivity of $\norm{\mathbf{v}}^2$ turns the inner-product identity into $\lambda=\overline{\lambda}$.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:invertible-quadratic-self-adjoint} \hlt{Invertible Quadratic Expressions}\\
Let $T\in\LL(\mathcal{V})$ be self-adjoint. If $b,c\in\mathbb{R}$ and $b^2<4c$, then $T^2+bT+cI_{\mathcal{V}}$ is invertible.
\begin{proof}
Set $S\defeq T^2+bT+cI_{\mathcal{V}}$, fix $\mathbf{v}\in\mathcal{V}\setminus\{\mathbf{0}\}$, and write $x\defeq\norm{T\mathbf{v}}$, $y\defeq\norm{\mathbf{v}}$, and $a\defeq\innerprod{T\mathbf{v},\mathbf{v}}$.\\
Relations $a\in\mathbb{R}$, $\innerprod{T^2\mathbf{v},\mathbf{v}}=x^2$ hold by self-adjointness. Bound $\abs{a}\leq xy$ holds by Theorem \ref{thm:cauchy-schwarz-inequality}(i), so $ba\geq-\abs{b}\abs{a}\geq-\abs{b}xy$.\\
Consequently, $\innerprod{S\mathbf{v},\mathbf{v}}\geq x^2-\abs{b}xy+cy^2=\left(x-\frac{\abs{b}}{2}y\right)^2+\left(c-\frac{b^2}{4}\right)y^2>0$ because $c-\frac{b^2}{4}>0$ and $y>0$.\\
Thus $S\mathbf{v}\neq\mathbf{0}$. Since $\mathbf{v}$ was arbitrary, $S$ is injective and therefore invertible by Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences}(ii),(i).\\
\hlt{Structural Commentary}: The proof reduces the operator polynomial to a scalar quadratic in the two norms $\norm{T\mathbf{v}}$ and $\norm{\mathbf{v}}$.\\
\hlt{Key Insight}: The negative discriminant keeps the completed square positive on nonzero vectors.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:self-adjoint-operators-have-eigenvalues} \hlt{Self-Adjoint Operators Have Eigenvalues}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be a finite-dimensional real inner-product space. Every self-adjoint $T\in\LL(\mathcal{V})$ has an eigenvalue.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Hermitian Extension}: For $\mathbf{z}=\mathbf{u}+i\mathbf{v}$ and $\mathbf{w}=\mathbf{x}+i\mathbf{y}$ in $\mathcal{V}_{\mathbb{C}}$, define $\innerprod[\mathcal{V}_{\mathbb{C}}]{\mathbf{z},\mathbf{w}}\defeq\innerprod[\mathcal{V}]{\mathbf{u},\mathbf{x}}+\innerprod[\mathcal{V}]{\mathbf{v},\mathbf{y}}+i\bigl(\innerprod[\mathcal{V}]{\mathbf{v},\mathbf{x}}-\innerprod[\mathcal{V}]{\mathbf{u},\mathbf{y}}\bigr)$.\\
Additivity and real homogeneity in the first slot hold componentwise.\\
The identity $\innerprod[\mathcal{V}_{\mathbb{C}}]{i\mathbf{z},\mathbf{w}}=i\innerprod[\mathcal{V}_{\mathbb{C}}]{\mathbf{z},\mathbf{w}}$ holds by substituting $i\mathbf{z}=-\mathbf{v}+i\mathbf{u}$, so the first slot is complex-linear.\\
The identity $\innerprod[\mathcal{V}_{\mathbb{C}}]{\mathbf{w},\mathbf{z}}=\overline{\innerprod[\mathcal{V}_{\mathbb{C}}]{\mathbf{z},\mathbf{w}}}$ holds by symmetry of the real inner product.\\
Also $\innerprod[\mathcal{V}_{\mathbb{C}}]{\mathbf{z},\mathbf{z}}=\norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2$ is nonnegative and vanishes iff $\mathbf{z}=\mathbf{0}$.\\
Thus the formula defines a complex inner product. Choose a real basis of $\mathcal{V}$. Then $\mathcal{V}_{\mathbb{C}}$ is finite-dimensional over $\mathbb{C}$ by Proposition \ref{prop:complexification-vector-space-basis}(i)--(iii), and it is nonzero by the hypothesis and injectivity of $\operatorname{cpx}$ in Proposition \ref{prop:complexification-real-dimension-and-embedding}(iii).
\item \hlt{Adjoint Preservation}: The operator $T_{\mathbb{C}}$ is complex-linear by Proposition \ref{prop:complexification-operator-basic-properties}(i). Fix $\mathbf{z}=\mathbf{u}+i\mathbf{v}$ and $\mathbf{w}=\mathbf{x}+i\mathbf{y}$ in $\mathcal{V}_{\mathbb{C}}$.\\
The identities $\innerprod{T\mathbf{u},\mathbf{x}}=\innerprod{\mathbf{u},T\mathbf{x}}$, $\innerprod{T\mathbf{v},\mathbf{y}}=\innerprod{\mathbf{v},T\mathbf{y}}$, $\innerprod{T\mathbf{v},\mathbf{x}}=\innerprod{\mathbf{v},T\mathbf{x}}$, and $\innerprod{T\mathbf{u},\mathbf{y}}=\innerprod{\mathbf{u},T\mathbf{y}}$ hold by self-adjointness of $T$.\\
The identity $\innerprod[\mathcal{V}_{\mathbb{C}}]{T_{\mathbb{C}}\mathbf{z},\mathbf{w}}=\innerprod[\mathcal{V}_{\mathbb{C}}]{\mathbf{z},T_{\mathbb{C}}\mathbf{w}}$ holds by substitution into the defining formula, so $T_{\mathbb{C}}$ is self-adjoint.
\item \emph{Spectral descent.} Choose $\lambda\in\operatorname{spec}(T_{\mathbb{C}})$ by Theorem \ref{thm:complex-operator-has-eigenvalue}. Then $\lambda\in\mathbb{R}$ by Theorem \ref{thm:self-adjoint-eigenvalues-real}, and $\lambda\in\operatorname{spec}(T)$ by Proposition \ref{prop:real-eigenvalues-complexification}.
\end{enumerate}
\hlt{Structural Commentary}: The proof extends the inner product and preserves self-adjointness before spectral descent.\\
\hlt{Key Insight}: Complexification preserves self-adjointness, forcing its eigenvalue into the real spectrum.
\end{proof}
\end{lemma}

\begin{proposition} \label{prop:self-adjoint-invariant-subspaces} \hlt{Self-Adjoint Operators and Invariant Subspaces}\\
Let $T\in\LL(\mathcal{V})$ be self-adjoint, and let $\mathcal{U}\in\operatorname{Inv}(T)$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{U}^{\perp}$ is invariant under $T$;
\item $T|_{\mathcal{U}}\in\LL(\mathcal{U})$ is self-adjoint;
\item $T|_{\mathcal{U}^{\perp}}\in\LL(\mathcal{U}^{\perp})$ is self-adjoint.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $\mathcal{U}$ is invariant under $T=T^*$, the subspace $\mathcal{U}^{\perp}$ is invariant under $T^*=T$ by Theorem \ref{thm:adjoint-invariant-subspace-orthogonal-complement}(i).
\item The pair $(\mathcal{U},\mathcal{U}^{\perp})$ reduces $T$ by part (i), the hypothesis, Theorem \ref{thm:orthogonal-direct-sum-decomposition}, and Definition \ref{def:reducing-pair-and-direct-sum-operator}.\\
Therefore $(T|_{\mathcal{U}})^*=(T^*)|_{\mathcal{U}}=T|_{\mathcal{U}}$ by Theorem \ref{thm:adjoint-invariant-subspace-orthogonal-complement}(iii).
\item The pair $(\mathcal{U}^{\perp},\mathcal{U})$ reduces $T$ by part (i), the hypothesis, Theorem \ref{thm:orthogonal-direct-sum-decomposition}, and $(\mathcal{U}^{\perp})^{\perp}=\mathcal{U}$ by Corollary \ref{cor:double-orthogonal-complement}.\\
Hence $(T|_{\mathcal{U}^{\perp}})^*=(T^*)|_{\mathcal{U}^{\perp}}=T|_{\mathcal{U}^{\perp}}$ by Theorem \ref{thm:adjoint-invariant-subspace-orthogonal-complement}(iii) for $\mathcal{U}^{\perp}$.
\end{enumerate}
\hlt{Structural Commentary}: The adjoint-complement theorem underlies complement invariance and both restriction-adjoint identities.\\
\hlt{Key Insight}: Self-adjointness reduces $T$ on both orthogonal summands.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:schur-triangularization} \hlt{Schur Triangularization}\\
Let $\mathcal{V}$ be a finite-dimensional complex inner-product space and let $T\in\LL(\mathcal{V})$.\\
The space $\mathcal{V}$ has an orthonormal basis $\gamma$ for which $[T]_{\gamma\leftarrow\gamma}$ is upper triangular.
\begin{proof}
If $\mathcal{V}=\{\mathbf{0}\}$, then the empty orthonormal basis has the required property. Assume $n\defeq\dim(\mathcal{V})\geq1$.\\
Choose an orthonormal basis $\beta$ of $\mathcal{V}$ by Corollary \ref{cor:finite-dimensional-inner-product-space-has-orthonormal-basis}, and set $A\defeq[T]_{\beta\leftarrow\beta}$.\\
For the nonempty commuting family $\{A\}$, fix a unitary $U$ such that $U^*AU$ is upper triangular by Theorem \ref{thm:simultaneous-schur-triangularization-commuting-family}.\\
Let $\gamma$ be the ordered list whose $\beta$-coordinate columns are the columns of $U$.\\
These columns form an orthonormal basis of $\mathbb{C}^n$ by Theorems \ref{thm:unitary-matrix-equivalent-conditions}(i),(ii),(vi) and \ref{thm:invertible-matrix-columns-basis}.\\
The coordinate map is an isomorphism by Proposition \ref{prop:coordinate-map-isomorphism}(iii), so $\gamma$ is a basis by Proposition \ref{prop:isomorphisms-preserve-bases}.\\
It is orthonormal by Theorem \ref{thm:parseval-identity-finite-dimensional}(ii). The identity $C_{\beta\leftarrow\gamma}=U$ holds by Definition \ref{def:change-of-basis-matrix}, while $C_{\gamma\leftarrow\beta}=U^{-1}=U^*$ by Theorems \ref{thm:change-of-basis-matrices}(i) and \ref{thm:unitary-matrix-equivalent-conditions}(vi). Hence $[T]_{\gamma\leftarrow\gamma}=U^*AU$ by Theorem \ref{thm:two-sided-change-of-basis-for-linear-maps}, so this matrix is upper triangular.\\
\hlt{Structural Commentary}: The proof transports singleton Schur triangularization from unitary coordinates to an orthonormal basis.\\
\hlt{Key Insight}: Unitary columns form the required orthonormal basis.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:upper-triangular-normal-matrix-is-diagonal} \hlt{Upper-Triangular Normal Matrices Are Diagonal}\\
Let $n\geq1$. If $A\in\operatorname{Mat}(n,n,\mathbb{C})$ is upper triangular and normal, then $A$ is diagonal.
\begin{proof}
Induct on $n$. If $n=1$, then $A$ is diagonal. Suppose $n\geq2$ and the claim holds in dimension $n-1$.\\
The $(1,1)$ entries of $A^*A=AA^*$ are $\abs{a_{11}}^2$ and $\abs{a_{11}}^2+\sum\limits_{j=2}^{n}\abs{a_{1j}}^2$, respectively.\\
Hence $(\forall j\in\{2,\ldots,n\})\,a_{1j}=0$, so $A=a_{11}\oplus B$ with $B$ upper triangular. The identity $B^*B=BB^*$ holds by comparison of the lower-right blocks, so $B$ is normal and therefore diagonal by the induction hypothesis. Thus $A$ is diagonal.\\
\hlt{Structural Commentary}: The $(1,1)$ identity removes the first-row tail and exposes a smaller normal block.\\
\hlt{Key Insight}: Once $a_{12},\ldots,a_{1n}$ vanish, block normality closes the induction.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:complex-spectral-theorem} \hlt{Complex Spectral Theorem}\\
Let $\mathcal{V}$ be a finite-dimensional complex inner-product space and let $T\in\LL(\mathcal{V})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is normal;
\item $\mathcal{V}$ has an orthonormal basis consisting of eigenvectors of $T$;
\item $T$ has a diagonal matrix in an orthonormal basis of $\mathcal{V}$.
\end{enumerate}
\begin{proof}
If $\mathcal{V}=\{\mathbf{0}\}$, all three conditions hold for the empty orthonormal basis. Assume $n\defeq\dim(\mathcal{V})\geq1$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Fix an orthonormal basis $\gamma$ such that $A\defeq[T]_{\gamma\leftarrow\gamma}$ is upper triangular by Theorem \ref{thm:schur-triangularization}.\\
The identities $[T^*]_{\gamma\leftarrow\gamma}=A^*$, $[T^*T]_{\gamma\leftarrow\gamma}=A^*A$, and $[TT^*]_{\gamma\leftarrow\gamma}=AA^*$ hold by Theorem \ref{thm:matrix-of-adjoint-is-conjugate-transpose} and Proposition \ref{prop:composition-and-matrix-multiplication}.\\
Thus $A^*A=AA^*$ by normality of $T$, so $A$ is normal by Definition \ref{def:normal-operator}(ii).\\
The matrix $A$ is diagonal by Lemma \ref{lem:upper-triangular-normal-matrix-is-diagonal}; hence $\gamma$ consists of eigenvectors of $T$.
\item[$(ii)$] $\Rightarrow (iii)$ The matrix of $T$ in an orthonormal eigenbasis is diagonal by Definition \ref{def:matrix-representation-of-linear-map}.
\item[$(iii)$] $\Rightarrow (i)$ Fix an orthonormal basis $\gamma$ in which $A\defeq[T]_{\gamma\leftarrow\gamma}$ is diagonal. Then $[T^*]_{\gamma\leftarrow\gamma}=A^*$ by Theorem \ref{thm:matrix-of-adjoint-is-conjugate-transpose}.\\
Since $AA^*=A^*A$, $[TT^*]_{\gamma\leftarrow\gamma}=[T^*T]_{\gamma\leftarrow\gamma}$ by Proposition \ref{prop:composition-and-matrix-multiplication}.\\
Therefore $TT^*=T^*T$ by Theorem \ref{thm:linear-maps-and-matrices-same-data}(i), and $T$ is normal by Definition \ref{def:normal-operator}(i).
\end{enumerate}
\hlt{Structural Commentary}: Schur form handles the forward direction; diagonal adjoints handle the reverse.\\
\hlt{Key Insight}: A triangular normal matrix is diagonal.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:real-spectral-theorem} \hlt{Real Spectral Theorem}\\
Let $\mathcal{V}$ be a finite-dimensional real inner-product space and let $T\in\LL(\mathcal{V})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is self-adjoint;
\item $\mathcal{V}$ has an orthonormal basis consisting of eigenvectors of $T$;
\item $T$ has a diagonal matrix in an orthonormal basis of $\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Assume $T=T^*$ and induct on $n\defeq\dim(\mathcal{V})$. For $n=0$, the empty basis suffices. For $n=1$, choose $\mathbf{v}\in\mathcal{V}\setminus\{\mathbf{0}\}$.\\
Then $\operatorname{span}(\mathbf{v})=\mathcal{V}\in\operatorname{Inv}(T)$, so $\mathbf{v}$ is an eigenvector by Proposition \ref{prop:one-dimensional-invariant-subspaces-and-eigenvectors}, and $\frac{\mathbf{v}}{\norm{\mathbf{v}}}$ is an orthonormal eigenbasis.\\
Suppose $n\geq2$, implication holds in dimension $n-1$. An eigenvalue $\lambda\in\mathbb{R}$, corresponding unit eigenvector $\mathbf{u}$ exist by Lemma \ref{lem:self-adjoint-operators-have-eigenvalues} and normalization. Set $\mathcal{U}\defeq\operatorname{span}(\mathbf{u})$.\\
Then $\mathcal{U}\in\operatorname{Inv}(T)$ by Proposition \ref{prop:one-dimensional-invariant-subspaces-and-eigenvectors}, while $\mathcal{U}^{\perp}\in\operatorname{Inv}(T)$ and $T|_{\mathcal{U}^{\perp}}$ is self-adjoint by Proposition \ref{prop:self-adjoint-invariant-subspaces}(i),(iii).\\
The identity $\dim(\mathcal{U}^{\perp})=n-1$ holds by Corollary \ref{cor:dimension-of-orthogonal-complement}, so $\mathcal{U}^{\perp}$ has an orthonormal eigenbasis by the induction hypothesis.\\
Adjoining $\mathbf{u}$ produces an orthonormal eigenbasis of $\mathcal{V}=\mathcal{U}\oplus\mathcal{U}^{\perp}$ by Theorem \ref{thm:orthogonal-direct-sum-decomposition}.
\item[$(ii)$] $\Rightarrow (iii)$ The matrix of $T$ in an orthonormal eigenbasis is diagonal by Definition \ref{def:matrix-representation-of-linear-map}.
\item[$(iii)$] $\Rightarrow (i)$ Fix an orthonormal basis $\gamma$ in which $A\defeq[T]_{\gamma\leftarrow\gamma}$ is real diagonal. Then $[T^*]_{\gamma\leftarrow\gamma}=A^t=A$ by Theorem \ref{thm:matrix-of-adjoint-is-conjugate-transpose}.\\
If $\mathcal{V}\neq\{\mathbf{0}\}$, then $T^*=T$ by Theorem \ref{thm:linear-maps-and-matrices-same-data}(i).\\
If $\mathcal{V}=\{\mathbf{0}\}$, then $T$ and $T^*$ are both the unique operator on the zero space, so again $T^*=T$.
\end{enumerate}
\hlt{Structural Commentary}: The forward implication inducts on a self-adjoint orthogonal complement.\\
\hlt{Key Insight}: Orthogonal-complement invariance turns real eigenvalue existence into an orthonormal eigenbasis.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:spectral-resolution-normal-operator} \hlt{Spectral Resolution of a Normal Operator}\\
Let $\mathcal{V}$ be a finite-dimensional complex inner-product space, $d\geq1$, $\mu_1,\ldots,\mu_d\in\mathbb{C}$ be distinct. For $T\in\LL(\mathcal{V})$, the following hold:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is normal with $\operatorname{spec}(T)\subseteq\{\mu_1,\ldots,\mu_d\}$ iff $(\exists(P_j)_{j=1}^{d}\in\LL(\mathcal{V})^d)$ such that\\
$(\forall j\in\{1,\ldots,d\})\,P_j^*=P_j=P_j^2$, $(\forall i,j\in\{1,\ldots,d\})\,[i\neq j\Longrightarrow P_iP_j=0]$, $\sum\limits_{j=1}^{d}P_j=I_{\mathcal{V}}$, and $T=\sum\limits_{j=1}^{d}\mu_jP_j$.
\item Every family $(P_1,\ldots,P_d)$ satisfying the right side of (i) obeys $(\forall j\in\{1,\ldots,d\})\,\operatorname{range}(P_j)=E_{\mu_j}(T)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Fix an orthonormal eigenbasis of $T$ by Theorem \ref{thm:complex-spectral-theorem}(i),(ii).\\
The decomposition $\mathcal{V}=\bigoplus\limits_{j=1}^{d}E_{\mu_j}(T)$ holds by grouping its vectors by eigenvalue, where an absent eigenvalue contributes the zero subspace; the summands are pairwise orthogonal by Theorem \ref{thm:normal-eigenvectors-distinct-eigenvalues-orthogonal}. Set $P_j\defeq P_{E_{\mu_j}(T)}$ by Definition \ref{def:orthogonal-projection-subspace}.\\
Then $(\forall j\in\{1,\ldots,d\})\,\operatorname{range}(P_j)=E_{\mu_j}(T)$ by Proposition \ref{prop:orthogonal-projection-properties}(iv), so $(\forall j\in\{1,\ldots,d\})\,P_j^*=P_j=P_j^2$ by Theorem \ref{thm:orthogonal-projection-self-adjoint-idempotent}.\\
Fix $\mathbf{x}\in\mathcal{V}$ and write $\mathbf{x}=\sum\limits_{j=1}^{d}\mathbf{x}_j$ with $\mathbf{x}_j\in E_{\mu_j}(T)$. Then $(\forall j\in\{1,\ldots,d\})\,P_j\mathbf{x}=\mathbf{x}_j$, so $(\forall i,j\in\{1,\ldots,d\})\,[i\neq j\Longrightarrow P_iP_j\mathbf{x}=\mathbf{0}]$, $\sum\limits_{j=1}^{d}P_j\mathbf{x}=\mathbf{x}$, and $T\mathbf{x}=\sum\limits_{j=1}^{d}\mu_j\mathbf{x}_j=\sum\limits_{j=1}^{d}\mu_jP_j\mathbf{x}$. Since $\mathbf{x}$ was arbitrary, the required operator identities hold.
\item[$(i)$] ($\Leftarrow$) Fix a family satisfying the right side of (i).\\
By Proposition \ref{prop:basic-adjoint-identities}(i), $T^*=\sum\limits_{j=1}^{d}\overline{\mu_j}P_j$, and the product relations imply $T^*T=TT^*=\sum\limits_{j=1}^{d}\abs{\mu_j}^2P_j$.\\
Thus $T$ is normal by Definition \ref{def:normal-operator}(i). Fix $\lambda\in\operatorname{spec}(T)$ and $\mathbf{x}\in E_{\lambda}(T)\setminus\{\mathbf{0}\}$.\\
Then $\mathbf{0}=(T-\lambda I_{\mathcal{V}})\mathbf{x}=\sum\limits_{j=1}^{d}(\mu_j-\lambda)P_j\mathbf{x}$. If $i\neq j$, then $\innerprod{P_i\mathbf{x},P_j\mathbf{x}}=\innerprod{\mathbf{x},P_iP_j\mathbf{x}}=0$ by self-adjointness and pairwise annihilation.\\
Hence $0=\sum\limits_{j=1}^{d}\abs{\mu_j-\lambda}^2\norm{P_j\mathbf{x}}^2$. The identity $\sum\limits_{j=1}^{d}P_j\mathbf{x}=\mathbf{x}\neq\mathbf{0}$ implies $(\exists k\in\{1,\ldots,d\})\,P_k\mathbf{x}\neq\mathbf{0}$, so the nonnegative sum forces $\lambda=\mu_k$. Therefore $\operatorname{spec}(T)\subseteq\{\mu_1,\ldots,\mu_d\}$.
\item[$(ii)$] Fix a family satisfying the right side of (i), $j\in\{1,\ldots,d\}$, and $\mathbf{x}\in\mathcal{V}$. Then $\mathbf{x}=\sum\limits_{i=1}^{d}P_i\mathbf{x}$ and $(T-\mu_jI_{\mathcal{V}})\mathbf{x}=\sum\limits_{i\neq j}(\mu_i-\mu_j)P_i\mathbf{x}$.\\
The vectors $P_i\mathbf{x}$ are pairwise orthogonal by self-adjointness and pairwise annihilation, so $\norm{(T-\mu_jI_{\mathcal{V}})\mathbf{x}}^2=\sum\limits_{i\neq j}\abs{\mu_i-\mu_j}^2\norm{P_i\mathbf{x}}^2$.\\
Distinctness and idempotence therefore imply $\mathbf{x}\in E_{\mu_j}(T)\Longleftrightarrow(\forall i\neq j)\,P_i\mathbf{x}=\mathbf{0}\Longleftrightarrow\mathbf{x}=P_j\mathbf{x}\Longleftrightarrow\mathbf{x}\in\operatorname{range}(P_j)$. QED.
\end{enumerate}
\hlt{Structural Commentary}: The spectral theorem constructs the projections; their algebra recovers normality, spectrum, and ranges.\\
\hlt{Key Insight}: Orthogonal idempotents are coordinate selectors for eigenspace decomposition.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:spectral-resolution-as-orthogonal-resolution} \hlt{Spectral Resolutions Are Orthogonal Resolutions}\\
Part (ii) of Theorem \ref{thm:spectral-resolution-normal-operator} identifies each $P_j$ as the orthogonal projection onto $E_{\mu_j}(T)$.\\
Thus a spectral resolution records the orthogonal eigenspace decomposition without choosing vectors inside its summands.\\
This basis-free viewpoint is confined here to finite-dimensional normal operators.
\end{perspective}

\begin{corollary} \label{cor:normal-spectral-resolution-functional-calculus} \hlt{Functional Calculus from Spectral Resolution}\\
Let $\mathcal{V}$ be finite-dimensional complex inner-product space, $d\geq1$, $\mu_1,\ldots,\mu_d\in\mathbb{C}$ be distinct, $P_1,\ldots,P_d\in\LL(\mathcal{V})$ be orthogonal projections satisfying $(\forall i,j\in\{1,\ldots,d\})\,[i\neq j\Longrightarrow P_iP_j=0]$ and $\sum\limits_{j=1}^{d}P_j=I_{\mathcal{V}}$. Set $T\defeq\sum\limits_{j=1}^{d}\mu_jP_j$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall f\in\mathbb{C}[z])\,f(T)=\sum\limits_{j=1}^{d}f(\mu_j)P_j$;
\item $(\forall j\in\{1,\ldots,d\})(\exists p_j\in\mathbb{C}[z])\,p_j(T)=P_j$.
\end{enumerate}
\begin{proof}
Each $P_j$ is idempotent by Theorem \ref{thm:orthogonal-projection-self-adjoint-idempotent}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $m\in\mathbb{Z}_{\geq0}$, $T^m=\sum\limits_{j=1}^{d}\mu_j^mP_j$. The case $m=0$ is $I_{\mathcal{V}}=\sum\limits_{j=1}^{d}P_j$. If the identity holds for $m$, then $T^{m+1}=\sum\limits_{j=1}^{d}\mu_j^{m+1}P_j$ by multiplication with $T$ and the product relations; hence the identity holds by induction.\\
For $f(z)=\sum\limits_{m=0}^{r}a_mz^m$, $f(T)=\sum\limits_{m=0}^{r}a_m\sum\limits_{j=1}^{d}\mu_j^mP_j=\sum\limits_{j=1}^{d}f(\mu_j)P_j$ by the power identity.
\item Fix $j\in\{1,\ldots,d\}$ and define $p_j(z)\defeq\prod\limits_{1\leq i\leq d,\ i\neq j}\frac{z-\mu_i}{\mu_j-\mu_i}$, with the empty product interpreted as $1$.\\
Then $p_j(\mu_i)=\delta_{ij}$ by distinctness of the $\mu_i$. Hence $p_j(T)=\sum\limits_{i=1}^{d}\delta_{ij}P_i=P_j$ by part (i).
\end{enumerate}
\hlt{Structural Commentary}: Projector powers give functional calculus; interpolation isolates each projection.\\
\hlt{Key Insight}: Lagrange interpolation constructs a polynomial that selects exactly one spectral projection.
\end{proof}
\end{corollary}
