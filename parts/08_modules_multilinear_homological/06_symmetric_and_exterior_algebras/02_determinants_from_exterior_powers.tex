\subsection{Determinants from Exterior Powers}

\begin{corollary} \label{cor:determinant-from-exterior-universal-property} \hlt{Determinant from the Exterior Universal Property}\\
Let $\dim\mathcal{V}=n\in\mathbb{Z}_{>0}$, $E=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ be ordered basis. Set $\Omega_E\defeq\mathbf{e}_1\wedge\cdots\wedge\mathbf{e}_n$.\\
Define $A_E:\mathcal{V}^{\times n}\to\operatorname{Mat}(n,n,\mathbb{F})$ by $A_E(\mathbf{v})\defeq([\mathbf{v}_j]_i)_{i,j=1}^{n}$ for $\mathbf{v}=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\exists!\,\ell_E\in(\bigwedge^n\mathcal{V})')\,\ell_E(\Omega_E)=1$.
\item Define $d_E:\mathcal{V}^{\times n}\to\mathbb{F}$ by $d_E(\mathbf{v})\defeq\ell_E(\mathbf{v}_1\wedge\cdots\wedge\mathbf{v}_n)$. The map $d_E$ is the unique alternating form satisfying $d_E(E)=1$.
\item $(\forall\,\mathbf{v}\in\mathcal{V}^{\times n})\,d_E(\mathbf{v})=\sum\limits_{\sigma\in\mathfrak{S}_n}\sign(\sigma)\prod\limits_{j=1}^{n}[\mathbf{v}_j]_{\sigma(j)}=\det A_E(\mathbf{v})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Singleton $\{\Omega_E\}$ is basis of $\bigwedge^n\mathcal{V}$ by Theorem \ref{thm:symmetric-exterior-dimensions}(ii)(a). A unique $\ell_E\in(\bigwedge^n\mathcal{V})'$ satisfies $\ell_E(\Omega_E)=1$ by Proposition \ref{prop:linear-map-determined-by-basis}.
\item The form $d_E$ is alternating by Theorem \ref{thm:symmetric-exterior-universal-properties}(ii), $d_E(E)=\ell_E(\Omega_E)=1$. Let $g:\mathcal{V}^{\times n}\to\mathbb{F}$ be alternating with $g(E)=1$.\\
A unique $\ell\in(\bigwedge^n\mathcal{V})'$ satisfies $g(\mathbf{v})=\ell(\mathbf{v}_1\wedge\cdots\wedge\mathbf{v}_n)$ by Theorem \ref{thm:symmetric-exterior-universal-properties}(ii).\\
$\ell(\Omega_E)=g(E)=1$, so $\ell=\ell_E$ by (i) and $g=d_E$ by the two factorizations.
\item Define $g:\mathcal{V}^{\times n}\to\mathbb{F}$ by $g(\mathbf{v})\defeq\det A_E(\mathbf{v})$. Map $g$ is multilinear and alternating by coordinate linearity and Theorem \ref{thm:determinant-basic-laws}(ii),(iv).\\
The identity $g(E)=\det I_n=1$ holds by Theorem \ref{thm:determinant-basic-laws}(i). Hence $g=d_E$ by (ii).\\
The permutation identity in (iii) holds by Definition \ref{def:matrix-trace-and-determinant}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Top-degree normalization fixes $\ell_E$; exterior universality transfers uniqueness.\\
\hlt{Key Insight}: One basis value determines a normalized top-degree alternating form.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:exterior-determinant-basic-properties} \hlt{Classical Laws for the Exterior Determinant}\\
Let $n\in\mathbb{Z}_{>0}$ and $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$. For the standard basis $E$ of $\mathbb{F}^n$, define $d:\operatorname{Mat}(n,n,\mathbb{F})\to\mathbb{F}$ by $d(C)\defeq d_E(C_{\cdot,1},\ldots,C_{\cdot,n})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $d(A^t)=d(A)$.
\item $d(AB)=d(A)d(B)$.
\item $A\in\operatorname{GL}(n,\mathbb{F})\Longleftrightarrow d(A)\neq0$.
\end{enumerate}
\begin{proof}
The identity $(\forall\,C\in\operatorname{Mat}(n,n,\mathbb{F}))\,d(C)=\det C$ holds by Corollary \ref{cor:determinant-from-exterior-universal-property}(iii).
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $d(A^t)=\det(A^t)=\det A=d(A)$ by Theorem \ref{thm:determinant-basic-laws}(ix).
\item[$(ii)$] $d(AB)=\det(AB)=\det(A)\det(B)=d(A)d(B)$ by Theorem \ref{thm:determinant-basic-laws}(vi).
\item[$(iii)$] ($\Rightarrow$) $A\in\operatorname{GL}(n,\mathbb{F})\Rightarrow\det A\neq0\Rightarrow d(A)\neq0$ by Theorem \ref{thm:determinant-basic-laws}(vii) and $d(A)=\det A$.
\item[$(iii)$] ($\Leftarrow$) $d(A)\neq0\Rightarrow\det A\neq0\Rightarrow A\in\operatorname{GL}(n,\mathbb{F})$ by $d(A)=\det A$ and Theorem \ref{thm:determinant-basic-laws}(vii).
\end{enumerate}
\hlt{Structural Commentary}: The identity $d=\det$ transfers transpose, product, and invertibility laws.\\
\hlt{Key Insight}: Exterior normalization and the permutation determinant define the same scalar.
\end{proof}
\end{corollary}
