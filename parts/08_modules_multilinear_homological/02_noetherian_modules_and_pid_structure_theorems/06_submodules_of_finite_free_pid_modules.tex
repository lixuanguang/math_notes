\subsection{Submodules of Finite Free PID Modules}

\begin{definition} \label{def:minimum-number-module-generators} \hlt{Minimum Number of Module Generators}\\
For an $R$-module $M$, define $\mu_R(M)\defeq\min\{\abs{X}:X\subseteq M\land\langle\!\langle X\rangle\!\rangle_R=M\}$.
\end{definition}

\begin{theorem} \label{thm:submodule-of-finite-free-pid-is-free} \hlt{Submodules of Finite Free PID Modules}\\
Let $n\in\N_0$, let $F$ be a free $R$-module with $\operatorname{rk}_R(F)=n$, and let $N\leq F$. Then $(\exists\,m\in\N_0)\,[m\leq n\land N\cong R^m]$.
\begin{proof}
Identify $F$ with $R^n$ by Proposition \ref{prop:free-module-coordinate-model-and-isomorphism}(ii).
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Coordinate Ideals}: Set $J\defeq\{1,\ldots,n\}$. For $k\in J$, set $E_k\defeq R^k\times\{0_R\}^{n-k}$ and $N_k\defeq N\cap E_k$.\\
The relation $E_k\leq R^n$ holds by coordinatewise closure, and $N_k\leq R^n$ by Proposition \ref{prop:submodule-intersection-sum}(i).\\
Let $\pi_k\in\operatorname{Hom}_R(N_k,R)$ be $k$th-coordinate map, set $I_k\defeq\operatorname{im}(\pi_k)$. Then $I_k\trianglelefteq R$ by Propositions \ref{prop:hom-module-kernel-image}(iii) and \ref{prop:submodules-of-regular-module-are-ideals}.\\
Choose $(r_k)_{k\in J}\in R^J$ satisfying $(\forall k\in J)\,I_k=\langle r_k\rangle$ by Definition \ref{def:principal-ideal-domain-and-ufd}(i).
\item \hlt{Triangular Independence}: Set $K\defeq\{k\in J:r_k\neq0_R\}$.\\
For $k\in K$, choose $u_k\in N_k$ with $(u_k)_k=r_k$ by $r_k\in I_k=\operatorname{im}(\pi_k)$, and set $U\defeq\{u_k:k\in K\}$.\\
Let $(c_k)_{k\in K}\in R^K$ satisfy $\sum\limits_{k\in K}c_ku_k=0_{R^n}$, and set $C\defeq\{k\in K:c_k\neq0_R\}$. If $C\neq\emptyset$, set $j\defeq\max C$.\\
The equality $c_jr_j=0_R$ holds in coordinate $j$ by $c_k=0_R$ for $k>j$ and $(u_k)_j=0_R$ for $k<j$.\\
Then $c_j=0_R$ by Proposition \ref{prop:integral-domain-cancellation}, $r_j\neq0_R$, contrary to $j\in C$; $C=\emptyset$, so $U$ is $R$-linearly independent by Definition \ref{def:module-linear-independence}.
\item \hlt{Height Descent}: For $x\in N$, set $\ell(x)\defeq\max(\operatorname{supp}(x)\cup\{0\})\in\N_0$.\\
The implication $\ell(x)=0\Rightarrow x=0_{R^n}\in\langle\!\langle U\rangle\!\rangle_R$ holds by Definition \ref{def:algebraic-support-of-function}.\\
For the inductive step, fix $x\in N$ with $\ell(x)=j>0$ and assume $(\forall y\in N)\,[\ell(y)<j\Rightarrow y\in\langle\!\langle U\rangle\!\rangle_R]$.\\
Then $x\in N_j$ and $x_j\neq0_R$ by the definition of $\ell$. The membership $j\in K$ holds by $x_j\in I_j=\langle r_j\rangle$ and $x_j\neq0_R$.\\
Choose $c\in R$ with $x_j=cr_j$ by $x_j\in I_j$. Then $x-cu_j\in N$ and $\ell(x-cu_j)<j$ by $u_j\in N_j$ and $(u_j)_j=r_j$.\\
The membership $x-cu_j\in\langle\!\langle U\rangle\!\rangle_R$ holds by the induction hypothesis. Then $x=(x-cu_j)+cu_j\in\langle\!\langle U\rangle\!\rangle_R$.\\
Therefore $N=\langle\!\langle U\rangle\!\rangle_R$ by strong induction and $U\subseteq N$.
\item \hlt{Rank Conclusion}: Set $m\defeq\abs K\in\N_0$ and enumerate $K=\{k_1<\cdots<k_m\}$. Define $\Phi:R^m\to N$ by $\Phi((a_i)_{i=1}^m)\defeq\sum\limits_{i=1}^ma_iu_{k_i}$.\\
The map $\Phi$ is an isomorphism by $R$-linearity, independence, and spanning of $U$.\\
The inequalities $m\leq\abs J=n$ hold by $K\subseteq J$, so $m\leq n$ and $N\cong R^m$.
\end{enumerate}
\hlt{Structural Commentary}: Coordinate ideals construct triangular pivots; height descent proves they span $N$.\\
\hlt{Key Insight}: The largest nonzero coordinate isolates each pivot coefficient.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:submodule-generator-bound-pid} \hlt{Submodules Need No More Generators over a PID}\\
Let $n\in\N_0$, let $M$ be an $R$-module, and let $S\leq M$. If $\mu_R(M)\leq n$, then $\mu_R(S)\leq n$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Free Cover}: Set $d\defeq\mu_R(M)\in\N_0$; the inequality $d\leq n$ holds by hypothesis.\\
Choose $(x_i)_{i=1}^d\in M^d$ with $M=\langle\!\langle x_1,\ldots,x_d\rangle\!\rangle_R$ by Definition \ref{def:minimum-number-module-generators}.\\
Extend this tuple to $(x_i)_{i=1}^n$ by $(\forall i\in\{1,\ldots,n\})\,[i>d\Rightarrow x_i=0_M]$.\\
Then $M=\langle\!\langle x_1,\ldots,x_n\rangle\!\rangle_R$ by the preceding spanning relation. Define $\pi:R^n\to M$ by $\pi((a_i)_{i=1}^n)\defeq\sum\limits_{i=1}^na_ix_i$.\\
The map $\pi$ is an epimorphism by the module laws and the spanning relation.
\item \hlt{Pullback}: Set $P\defeq\pi^{-1}(S)$. Then $0_{R^n}\in P$ and $(\forall(a,b,u,v)\in R^2\times P^2)\,\pi(au+bv)=a\pi(u)+b\pi(v)\in S$.\\
Thus $P\leq R^n$ by Theorem \ref{thm:submodule-test}. Choose $m\in\N_0$ and $\theta:R^m\xrightarrow{\sim}P$ with $m\leq n$ by Theorem \ref{thm:submodule-of-finite-free-pid-is-free}.
\item \hlt{Generator Transfer}: Let $(e_i)_{i=1}^m$ be the standard basis of $R^m$ and set $v_i\defeq\theta(e_i)$.\\
The equality $P=\langle\!\langle v_1,\ldots,v_m\rangle\!\rangle_R$ holds by surjectivity of $\theta$. The equality $\pi(P)=S$ holds by surjectivity of $\pi$ and $P=\pi^{-1}(S)$.\\
Thus $S=\langle\!\langle\pi(v_1),\ldots,\pi(v_m)\rangle\!\rangle_R$ by the preceding equalities. The bound $\mu_R(S)\leq m\leq n$ holds by Definition \ref{def:minimum-number-module-generators}.
\end{enumerate}
\hlt{Structural Commentary}: Pullback to $R^n$ transfers finite-free rank control through a surjection.\\
\hlt{Key Insight}: Images of a pullback basis generate $S$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:finitely-generated-pid-modules-noetherian} \hlt{Finitely Generated PID Modules Are Noetherian}\\
Let $M$ be an $R$-module. If $\mu_R(M)\in\N_0$, then $M$ is Noetherian.
\begin{proof}
Set $n\defeq\mu_R(M)$. Then $(\forall\,N\leq M)\,\mu_R(N)\leq n$ by Corollary \ref{cor:submodule-generator-bound-pid}.\\
Every submodule of $M$ is finitely generated by Definition \ref{def:minimum-number-module-generators} and the bound. Thus $M$ is Noetherian by Theorem \ref{thm:noetherian-iff-submodules-finitely-generated}(i).\\
\hlt{Structural Commentary}: The generator bound verifies that every submodule is finitely generated.\\
\hlt{Key Insight}: One finite bound controls every submodule of $M$.
\end{proof}
\end{corollary}

\begin{perspective} \label{per:pid-noetherian-specialization} \hlt{Noetherian Modules over a PID}\\
Theorem \ref{thm:submodule-of-finite-free-pid-is-free} restores finite-rank control for submodules of finite free modules over a PID.\\
Corollary \ref{cor:submodule-generator-bound-pid} transfers that control through free covers.\\
Corollary \ref{cor:finitely-generated-pid-modules-noetherian} deduces Noetherianity; neither result asserts a submodule complement.
\end{perspective}

\begin{proposition} \label{prop:spanning-set-size-rank-basis-pid} \hlt{A Full-Size Spanning Set Is a Basis}\\
Let $n\in\N_0$ and let $F$ be a free $R$-module with $\operatorname{rk}_R(F)=n$. If $F=\langle\!\langle s_1,\ldots,s_n\rangle\!\rangle_R$, then $(s_1,\ldots,s_n)$ is an $R$-basis of $F$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Surjective Endomorphism}: Choose an $R$-basis $(b_i)_{i=1}^n$ of $F$.\\
The assignment $b_i\mapsto s_i$ extends to $\tau\in\operatorname{End}_R(F)$ by Theorem \ref{thm:free-module-map-extension}. The map $\tau$ is surjective by $F=\langle\!\langle s_1,\ldots,s_n\rangle\!\rangle_R$.
\item \hlt{Splitting}: Choose $(a_i)_{i=1}^n\in F^n$ satisfying $(\forall i\in\{1,\ldots,n\})\,\tau(a_i)=b_i$ by surjectivity of $\tau$.\\
The assignment $b_i\mapsto a_i$ extends to $\rho\in\operatorname{End}_R(F)$ by Theorem \ref{thm:free-module-map-extension}.\\
The equality $\tau\rho=\operatorname{id}_F$ holds by basis agreement and uniqueness in Theorem \ref{thm:free-module-map-extension}.\\
Then $F=\ker\tau\oplus\operatorname{im}\rho$ by Theorem \ref{thm:module-one-sided-inverses-and-complements}(ii). The map $\rho$ is injective by $\tau\rho=\operatorname{id}_F$.\\
Its corestriction $F\to\operatorname{im}\rho$ is an isomorphism by $R$-linearity, injectivity, and the definition of image.
\item \hlt{Rank Cancellation}: Choose $r\in\N_0$ with $r\leq n$ and $\ker\tau\cong R^r$ by Theorem \ref{thm:submodule-of-finite-free-pid-is-free}.\\
The chain $R^n\cong F\cong\ker\tau\oplus\operatorname{im}\rho\cong R^r\oplus R^n$ holds by the preceding isomorphisms.\\
Thus $R^n\cong R^{r+n}$ by coordinate concatenation. Thus $n=r+n$ by Proposition \ref{prop:free-module-coordinate-model-and-isomorphism}(ii), so $r=0$.
\item \hlt{Basis Transport}: The equality $\ker\tau=\{0_F\}$ holds by $r=0$ and $\ker\tau\cong R^r$.\\
The map $\tau$ is injective by Proposition \ref{prop:hom-module-kernel-image}(iv). It is an isomorphism by Definition \ref{def:module-homomorphism}(iv) and surjectivity.\\
Therefore $(s_i)_{i=1}^n=(\tau(b_i))_{i=1}^n$ is an $R$-basis by bijectivity and linearity of $\tau$.
\end{enumerate}
\hlt{Structural Commentary}: A splitting isolates the kernel; finite-free rank comparison forces it to vanish.\\
\hlt{Key Insight}: The equality $n=r+n$ leaves only $r=0$.
\end{proof}
\end{proposition}
