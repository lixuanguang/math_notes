\subsection{Finitely Generated PID Modules}

\begin{theorem} \label{thm:finitely-generated-torsion-free-pid-free} \hlt{Finitely Generated Torsion-Free Modules Are Free}\\
Let $M$ be a f.g. $R$-module. Then $M$ is free $\iff M$ is torsion-free.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Let $\mathcal{B}$ be an $R$-basis of $M$, $a\in R\setminus\{0_R\}$, $am=0_M$. Choose finite $H\subseteq\mathcal{B}$, $(r_b)_{b\in H}\in R^H$ such that $m=\sum\limits_{b\in H}r_bb$.\\
Then $\sum\limits_{b\in H}(ar_b)b=0_M$, so $(\forall b\in H)\,[ar_b=0_R]$ by the independence of $\mathcal{B}$.\\
Since $R$ is an integral domain and $a\neq0_R$, $(\forall b\in H)\,[r_b=0_R]$. Thus $m=0_M$, and $M$ is torsion-free by Definition \ref{def:module-torsion}(ii).
\item[$(\Leftarrow)$] Assume $M$ is torsion-free.
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Independent Core}: Choose a finite generating set $G=\{u_1,\ldots,u_n\}$ of $M$.\\
Choose an inclusion-maximal $R$-linearly independent subset $S=\{v_1,\ldots,v_k\}\subseteq G$.\\
Set $L\defeq\langle\!\langle S\rangle\!\rangle_R$. Then $S$ is an $R$-basis of $L$, so $L$ is finite free of rank $k$.
\item \hlt{Common Scalar}: Set $W\defeq G\setminus S$. Maximality makes $S\cup\{w\}$ dependent for $w\in W$.\\
The independence of $S$ forces the coefficient of $w$ in each nontrivial relation to be nonzero.\\
Choose coefficients $a_w\in R\setminus\{0_R\}$ and $b_{w,i}\in R$, indexed by $w\in W$, such that $(\forall w\in W)\,[a_ww=\sum\limits_{i=1}^k b_{w,i}v_i]$.\\
Set $a\defeq\prod\limits_{w\in W}a_w$, with the empty product equal to $1_R$. Then $a\neq0_R$ because $R$ is an integral domain.\\
Set $aM\defeq\{am:m\in M\}$. For $w\in W$, write $a=c_wa_w$; then $aw=\sum\limits_{i=1}^k c_wb_{w,i}v_i\in L$.\\
Together with $av_i\in L$ for $1\leq i\leq k$, these relations imply $(\forall u\in G)\,[au\in L]$. Since $G$ generates $M$, one has $aM\subseteq L$.
\item \hlt{Free Image}: Define $\mu_a:M\to M$ by $\mu_a(m)\defeq am$. Then $\mu_a\in\operatorname{Hom}_R(M,M)$ and $aM=\operatorname{im}(\mu_a)$.\\
Hence $aM\leq L$ by Proposition \ref{prop:hom-module-kernel-image}(iii) and the containment $aM\subseteq L$.\\
The module $aM$ is free by Theorem \ref{thm:submodule-of-finite-free-pid-is-free}, since $L$ is finite free.
\item \hlt{Transport}: The corestriction $\mu_a:M\to aM$ is surjective by the definition of $aM$.\\
It is injective because $a\neq0_R$ and $M$ is torsion-free. Thus $M\cong aM$, so $M$ is free.
\end{enumerate}
\end{enumerate}
\noindent
\hlt{Structural Commentary}: A maximal independent core produces a nonzero scalar sending $M$ into a finite free span.\\
\hlt{Key Insight}: Multiplication by that nonzero product identifies $M$ with the free submodule $aM$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:finitely-generated-pid-free-plus-torsion} \hlt{Free Plus Torsion Decomposition}\\
Let $M$ be a f.g. $R$-module and set $T\defeq M_{\operatorname{tor}}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item There is a f.g. free submodule $F\leq M$ such that $M=T\oplus F$ and $F\cong M/T$.
\item If $M=T'\oplus G$, where $T'\leq M$ is torsion and $G\leq M$ is free, then $T'=T$ and $G\cong M/T$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $Q\defeq M/T$. Then $T\leq M$ and $Q$ is torsion-free by Proposition \ref{prop:torsion-submodule-over-domain}(i)--(ii). Choose generators $u_1,\ldots,u_n$ of $M$.\\
Then $Q=\langle\!\langle u_1+T,\ldots,u_n+T\rangle\!\rangle_R$, so $Q$ is f.g. Hence $Q$ is free by Theorem \ref{thm:finitely-generated-torsion-free-pid-free}. Let $\pi:M\twoheadrightarrow Q$ be the quotient map.\\
Choose a section $\sigma:Q\to M$ of $\pi$ by Theorem \ref{thm:epimorphism-onto-free-module-splits}(i). Set $F\defeq\operatorname{im}(\sigma)$. The identity $\ker\pi=T$ holds by Proposition \ref{prop:quotient-module-well-defined}(iii).\\
Hence $M=T\oplus F$ and $F\cong Q=M/T$ by Theorem \ref{thm:epimorphism-onto-free-module-splits}(i). Since $Q$ is free and $F\cong Q$, the module $F$ is free.\\
The images under $\sigma$ of the generators $u_i+T$ generate $F$, so $F$ is f.g.
\item Assume $M=T'\oplus G$, where $T'$ is torsion, $G$ is free. Then $T'\subseteq T$ by Definition \ref{def:module-torsion}(iv). Fix $x\in T$, $x=t'+g$ with $(t',g)\in T'\times G$.\\
Since $T\leq M$ and $x,t'\in T$, one has $g=x-t'\in T\cap G$. Let $\rho:M\twoheadrightarrow G$, $t'+g\mapsto g$, be the coordinate projection.\\
The image under $\rho$ of a finite generating set of $M$ generates $G$, so $G$ is f.g. Hence $G$ is torsion-free by Theorem \ref{thm:finitely-generated-torsion-free-pid-free}.\\
Since $g\in T\cap G$, one has $g=0_M$. Thus $x=t'\in T'$, and hence $T=T'$. Finally, $G\cong M/T$ by Theorem \ref{thm:complements-and-quotients-modules}(i).
\end{enumerate}
\noindent
\hlt{Structural Commentary}: The torsion-free quotient is free and splits; a free summand contains no nonzero torsion.\\
\hlt{Key Insight}: Every free complement is isomorphic to the common quotient model $M/T$.
\end{proof}
\end{theorem}
