\subsection{Free Modules, Bases, and Universal Property}

Throughout this subsection, $R$ denotes a nonzero commutative ring with identity $1_R$.\\
The symbol $M$ denotes an $R$-module with additive identity $0_M$.

\begin{definition} \label{def:free-module-basis} \hlt{Free Module and Basis}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A subset $\mathcal{B}\subseteq M$ is a \hlt{basis} of $M$ if $\mathcal{B}$ is $R$-linearly independent and $\langle\!\langle\mathcal{B}\rangle\!\rangle_R=M$.
\item An $R$-module $M$ is \hlt{free} if $M$ has a basis.
\item If $\mathcal{B}$ is a basis of $M$, then $M$ is \hlt{free on $\mathcal{B}$}.
\end{enumerate}
The empty set is independent and generates $\{0_M\}$, so the zero module is free on $\emptyset$.
\end{definition}

\begin{theorem} \label{thm:module-basis-unique-coordinates} \hlt{Basis Means Unique Finite Coordinates}\\
For $\mathcal{B}\subseteq M$, define $\Sigma_{\mathcal{B}}:\bigoplus\limits_{b\in\mathcal{B}}R\to M$ by $\Sigma_{\mathcal{B}}((r_b)_{b\in\mathcal{B}})\defeq\sum\limits_{b\in\mathcal{B}}r_bb$. Then $\mathcal{B}$ is a basis of $M$ if and only if $\Sigma_{\mathcal{B}}$ is bijective.
\begin{proof}
The sum defining $\Sigma_{\mathcal{B}}$ is finite by membership in $\bigoplus\limits_{b\in\mathcal{B}}R$, so $\Sigma_{\mathcal{B}}$ is well-defined.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $\mathcal{B}$ is a basis. Then $\Sigma_{\mathcal{B}}$ is surjective by $\langle\!\langle\mathcal{B}\rangle\!\rangle_R=M$.\\
Fix $r=(r_b)_{b\in\mathcal{B}}$, $s=(s_b)_{b\in\mathcal{B}}$ with $\Sigma_{\mathcal{B}}(r)=\Sigma_{\mathcal{B}}(s)$, set $H\defeq\operatorname{supp}(r)\cup\operatorname{supp}(s)$. Then $\abs{H}<\infty$ and $\sum\limits_{b\in H}(r_b-s_b)b=0_M$.\\
Thus $(\forall\,b\in H)\,r_b=s_b$ by the $R$-linear independence of $\mathcal{B}$. Both families vanish on $\mathcal{B}\setminus H$, so $r=s$ and $\Sigma_{\mathcal{B}}$ is injective.
\item[$(\Leftarrow)$] Assume $\Sigma_{\mathcal{B}}$ is bijective. Then $M=\operatorname{im}\Sigma_{\mathcal{B}}=\langle\!\langle\mathcal{B}\rangle\!\rangle_R$ by surjectivity and Definition \ref{def:generated-submodule}(i), so $\mathcal{B}$ spans $M$.\\
Fix finite $F\subseteq\mathcal{B}$ and $(r_b)_{b\in F}\in R^F$ with $\sum\limits_{b\in F}r_bb=0_M$, and extend this family by $0_R$ to $\bar r\in\bigoplus\limits_{b\in\mathcal{B}}R$.\\
Then $\Sigma_{\mathcal{B}}(\bar r)=0_M=\Sigma_{\mathcal{B}}(0)$, so $\bar r=0$ by injectivity. Thus $(\forall\,b\in F)\,r_b=0_R$, so $\mathcal{B}$ is $R$-linearly independent by Definition \ref{def:module-linear-independence}(i).\\
Therefore $\mathcal{B}$ is a basis by Definition \ref{def:free-module-basis}(i).
\end{enumerate}
\hlt{Structural Commentary}: Surjectivity encodes spanning, while injectivity encodes rigidity of finite relations.\\
\hlt{Key Insight}: Coordinate families are the fibers of the summation map.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:module-basis-minimal-maximal} \hlt{Bases Are Minimal Spanning and Maximal Independent}\\
If $\mathcal{B}$ is a basis of $M$, then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall b\in\mathcal{B})\ \langle\!\langle\mathcal{B}\setminus\{b\}\rangle\!\rangle_R\neq M$;
\item $(\forall v\in M\setminus\mathcal{B})\ \mathcal{B}\cup\{v\}$ is $R$-linearly dependent.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $b\in\mathcal{B}$ and suppose $b\in\langle\!\langle\mathcal{B}\setminus\{b\}\rangle\!\rangle_R$.\\
Then $b$ has a coordinate family vanishing at $b$, whereas the singleton family with coefficient $1_R$ at $b$ represents $b$.\\
These distinct families both map to $b$ under $\Sigma_{\mathcal{B}}$, contrary to injectivity by Theorem \ref{thm:module-basis-unique-coordinates}. Hence $b\notin\langle\!\langle\mathcal{B}\setminus\{b\}\rangle\!\rangle_R$.
\item Fix $v\in M\setminus\mathcal{B}$. There is $r=(r_b)_{b\in\mathcal{B}}\in\bigoplus\limits_{b\in\mathcal{B}}R$ with $\Sigma_{\mathcal{B}}(r)=v$ by Theorem \ref{thm:module-basis-unique-coordinates}.\\
Thus $1_Rv-\sum\limits_{b\in\mathcal{B}}r_bb=0_M$ is a finite relation in $\mathcal{B}\cup\{v\}$ with nonzero coefficient $1_R$ at $v$.\\
Hence $\mathcal{B}\cup\{v\}$ is $R$-linearly dependent by Definition \ref{def:module-linear-independence}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Coordinate uniqueness obstructs deletion; coordinate existence creates dependence after enlargement.\\
\hlt{Key Insight}: Removing or adjoining one element destroys one basis condition.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:free-module-map-extension} \hlt{Maps Out of a Free Module}\\
Let $M$ be free on $\mathcal{B}$, let $N$ be an $R$-module, and let $f:\mathcal{B}\to N$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\exists\,\widetilde{f}\in\operatorname{Hom}_R(M,N))\,\widetilde{f}|_{\mathcal{B}}=f$.
\item If $\phi,\psi\in\operatorname{Hom}_R(M,N)$ satisfy $\phi|_{\mathcal{B}}=\psi|_{\mathcal{B}}=f$, then $\phi=\psi$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $r=(r_b)_{b\in\mathcal{B}}\in\bigoplus\limits_{b\in\mathcal{B}}R$, define $\widetilde{f}(\Sigma_{\mathcal{B}}(r))\defeq\sum\limits_{b\in\mathcal{B}}r_bf(b)$.\\
This defines $\widetilde{f}:M\to N$ by the bijectivity in Theorem \ref{thm:module-basis-unique-coordinates} and the finite support of $r$.\\
For $r,s\in\bigoplus\limits_{b\in\mathcal{B}}R$ and $a,c\in R$, $\Sigma_{\mathcal{B}}(ar+cs)=a\Sigma_{\mathcal{B}}(r)+c\Sigma_{\mathcal{B}}(s)$ by the module laws.\\
Thus $\widetilde{f}(a\Sigma_{\mathcal{B}}(r)+c\Sigma_{\mathcal{B}}(s))=a\widetilde{f}(\Sigma_{\mathcal{B}}(r))+c\widetilde{f}(\Sigma_{\mathcal{B}}(s))$, so $\widetilde{f}\in\operatorname{Hom}_R(M,N)$.\\
The singleton coordinate families satisfy $(\forall\,b\in\mathcal{B})\,\widetilde{f}(b)=f(b)$, so $\widetilde{f}|_{\mathcal{B}}=f$.
\item Let $\phi,\psi\in\operatorname{Hom}_R(M,N)$ satisfy $\phi|_{\mathcal{B}}=\psi|_{\mathcal{B}}=f$. For $r=(r_b)_{b\in\mathcal{B}}$, $\phi(\Sigma_{\mathcal{B}}(r))=\sum\limits_{b\in\mathcal{B}}r_bf(b)=\psi(\Sigma_{\mathcal{B}}(r))$ by $R$-linearity.\\
Thus $\phi=\psi$ by the surjectivity of $\Sigma_{\mathcal{B}}$ and Assumption \ref{assump:function-extensionality}.
\end{enumerate}
\hlt{Structural Commentary}: Coordinate evaluation constructs the extension; summation-map surjectivity globalizes basis agreement.\\
\hlt{Key Insight}: Agreement on a basis propagates through finite coordinate sums.
\end{proof}
\end{theorem}

\begin{counterexample} \label{cex:free-module-with-nonfree-submodule} \hlt{A Free Module with a Nonfree Submodule}\\
A submodule of a free module need not be free. Let $R=\mathbb{Z}\times\mathbb{Z}$ with componentwise operations, so $1_R=(1,1)$ and $0_R=(0,0)$.\\
Regular module $R$ is free on $\{(1,1)\}$ as $(\forall\,x\in R)(\exists!\,r\in R)\,x=r(1,1)$. Set $N\defeq\mathbb{Z}\times\{0\}$.\\
Then $N\trianglelefteq R$, so $N\leq{}_R R$ by Proposition \ref{prop:submodules-of-regular-module-are-ideals}. The nonzero scalar $(0,1)$ satisfies $(\forall\,v\in N)\,(0,1)v=0_R$.\\
For $\emptyset\neq S\subseteq N$, fix $v\in S$; then $(0,1)v=0_R$ is a relation with nonzero coefficient.\\
Thus $S$ is dependent by Definition \ref{def:module-linear-independence}(ii). The set $\emptyset$ does not span $N$ because $(1,0)\in N\setminus\{0_R\}$.\\
Hence $N$ has no basis and is not free. Therefore $\emptyset$ is maximal independent but not spanning.
\end{counterexample}

\begin{counterexample} \label{cex:quotient-free-module-need-not-free} \hlt{A Quotient of a Free Module Need Not Be Free}\\
A quotient of a free module by a free submodule need not be free. Fix $n\in\N$ with $n>1$.\\
As a $\mathbb{Z}$-module, $\mathbb{Z}$ is free on $\{1\}$, and $n\mathbb{Z}$ is free on $\{n\}$.\\
Set $Q\defeq\mathbb{Z}/n\mathbb{Z}$ and $\overline{1}\defeq1+n\mathbb{Z}$. Then $Q=\langle\!\langle\overline{1}\rangle\!\rangle_{\mathbb{Z}}\neq\{0_Q\}$, so $\{\overline{1}\}$ minimally spans $Q$.\\
However, $(\forall\,q\in Q)\,nq=0_Q$ with $n\neq0_{\mathbb{Z}}$. Thus each $\emptyset\neq S\subseteq Q$ is linearly dependent by Definition \ref{def:module-linear-independence}(ii).\\
Hence $Q$ has no basis and is not free; the minimally spanning set $\{\overline{1}\}$ is therefore not a basis.
\end{counterexample}

\begin{principle} \label{princ:modules-not-vector-spaces} \hlt{Modules Are Not Just Vector Spaces with New Letters}\\
For a general ring $R$, the following vector-space implications fail:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $N\leq M\nRightarrow(\exists\,H\leq M)\ M=N\oplus H$.
\item $M$ finitely generated and $N\leq M\nRightarrow N$ finitely generated.
\item $M\neq\{0_M\}\nRightarrow(\exists\,\emptyset\neq S\subseteq M)\ S$ is linearly independent.
\item $S$ minimally spans $M\nRightarrow S$ is a basis.
\item $S$ is maximal linearly independent in $M\nRightarrow S$ is a basis.
\item $M$ free and $N\leq M\nRightarrow N$ free.
\item $F$ free and $S\leq F\nRightarrow F/S$ free.
\end{enumerate}
Counterexamples \ref{cex:z-submodules-mostly-not-complemented} and \ref{cex:finitely-generated-module-with-non-fg-submodule} witness (i) and (ii). Counterexample \ref{cex:free-module-with-nonfree-submodule} witnesses (iii), (v), and (vi).\\
Counterexample \ref{cex:quotient-free-module-need-not-free} witnesses (iv) and (vii). Over a field, these obstructions vanish; algebraically, $R\setminus\{0_R\}\subseteq R^\times$.
\end{principle}
