\subsection{Rank and Coordinate Models of Free Modules}

Throughout this subsection, assume Principles \ref{princ:zorns-lemma} and \ref{princ:axiom-of-choice-analysis}.

\begin{perspective} \label{per:dimension-as-field-case-of-rank} \hlt{Dimension as the Field Case of Rank}\\
An $\mathbb{F}$-vector space is a free $\mathbb{F}$-module by Theorem \ref{thm:hamel-basis-extension}(ii) and Definition \ref{def:free-module-basis}.\\
Finite basis cardinality is invariant by Theorem \ref{thm:basis-length-well-defined}. Arbitrary Hamel-basis cardinality is invariant by Theorem \ref{thm:arbitrary-bases-same-cardinality}.\\
Over general scalar rings, rank invariance requires additional hypotheses.
\end{perspective}

\begin{theorem} \label{thm:free-module-rank-well-defined-commutative} \hlt{Rank Is Well Defined for Free Modules}\\
Let $A$ be a nonzero commutative ring with identity $1_A$, and let $M$ be a free $A$-module.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $B,B'$ are $A$-bases of $M$, then $\abs B=\abs{B'}$.
\item If $B$ is an $A$-basis of $M$ and $S\subseteq M$ satisfies $\langle\!\langle S\rangle\!\rangle_A=M$, then $\abs B\leq\abs S$.
\end{enumerate}
\begin{proof}
Choose a maximal ideal $\mathfrak{m}\trianglelefteq A$ by Theorem \ref{thm:maximal-ideals-exist}. Set $\mathfrak{m}M\defeq\langle\!\langle\{ax:(a,x)\in\mathfrak{m}\times M\}\rangle\!\rangle_A$, $K\defeq A/\mathfrak{m}$, and $V\defeq M/\mathfrak{m}M$.\\
The ring $K$ is a field by Theorem \ref{thm:quotient-by-maximal-ideal-is-field}. If $a-a'\in\mathfrak{m}$ and $x-x'\in\mathfrak{m}M$, then $ax-a'x'=(a-a')x+a'(x-x')\in\mathfrak{m}M$.\\
Hence $(a+\mathfrak{m})(x+\mathfrak{m}M)\defeq ax+\mathfrak{m}M$ is well-defined, and the inherited module identities make $V$ a $K$-vector space.\\
Fix an $A$-basis $B$ of $M$, set $X_B\defeq\{\sum\limits_{b\in B}i_bb:(i_b)_{b\in B}\in\bigoplus\limits_{b\in B}\mathfrak{m}\}$. Iinclusion $X_B\leq_A M$ holds by coefficientwise ideal closure.\\
If $i\in\mathfrak{m}$ and $x=\sum\limits_{b\in B}r_bb$, then $ix=\sum\limits_{b\in B}(ir_b)b\in X_B$. Thus $\mathfrak{m}M\subseteq X_B$ by Proposition \ref{prop:generated-submodule-smallest}(iii).\\
Conversely, $X_B\subseteq\mathfrak{m}M$ by Definition \ref{def:generated-submodule}(i), so $\mathfrak{m}M=X_B$. Write $\overline{x}\defeq x+\mathfrak{m}M$ for $x\in M$ and set $\overline B\defeq\{\overline b:b\in B\}$.\\
The set $\overline B$ spans $V$ by spanning of $B$. Let $F\subseteq B$ be finite and $(a_b)_{b\in F}\in A^F$ satisfy $\sum\limits_{b\in F}(a_b+\mathfrak{m})\overline b=0_V$.\\
Then $\sum\limits_{b\in F}a_bb\in\mathfrak{m}M=X_B$. Consequently, $(\forall b\in F)\ a_b\in\mathfrak{m}$ by coordinate uniqueness in Theorem \ref{thm:module-basis-unique-coordinates}.\\
Thus $\overline B$ is $K$-linearly independent by Definition \ref{def:module-linear-independence}. If $\overline b=\overline{b'}$ for distinct $b,b'\in B$, then $b-b'\in X_B$.\\
The same coordinate uniqueness forces $1_A\in\mathfrak{m}$, contrary to $\mathfrak{m}\neq A$.\\
The map $b\mapsto\overline b$ is a bijection $B\to\overline B$ by the preceding injectivity and the definition of $\overline B$.\\
Hence $\overline B$ is a $K$-basis of $V$ by spanning and independence.\\
Because $B$ was arbitrary, the construction applies to every $A$-basis of $M$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $B,B'$ are $A$-bases of $M$, then $\overline B,\overline{B'}$ are $K$-bases of $V$. Thus $\abs{\overline B}=\abs{\overline{B'}}$ by Theorem \ref{thm:arbitrary-bases-same-cardinality}.\\
$\abs B=\abs{\overline B}=\abs{\overline{B'}}=\abs{B'}$ by bijections above.
\item Fix an $A$-basis $B$ of $M$ and $S\subseteq M$ with $\langle\!\langle S\rangle\!\rangle_A=M$. Then $\overline S\defeq\{\overline s:s\in S\}$ spans $V$ by spanning of $S$.\\
Choose a $K$-basis $C\subseteq\overline S$ by Theorem \ref{thm:hamel-basis-extension}(iv). Since $\overline B,C$ are $K$-bases, $\abs{\overline B}=\abs C$ by Theorem \ref{thm:arbitrary-bases-same-cardinality}; $\abs B=\abs{\overline B}=\abs C\leq\abs{\overline S}\leq\abs S$.
\end{enumerate}
\hlt{Structural Commentary}: Reduction modulo $\mathfrak{m}$ transfers each $A$-basis to an equipotent $K$-basis for vector-space comparison.\\
\hlt{Key Insight}: Basis coordinates detect exactly which cosets vanish modulo $\mathfrak{m}$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:rank-of-free-module} \hlt{Rank of a Free Module}\\
Let $A$ be a nonzero commutative ring with identity, let $M$ be a free $A$-module, and fix an $A$-basis $B$ of $M$.\\
Define the \hlt{rank} of $M$ by $\operatorname{rk}_A(M)\defeq\abs B$. This value is independent of $B$ by Theorem \ref{thm:free-module-rank-well-defined-commutative}(i).\\
The empty basis of the zero module satisfies $\operatorname{rk}_A(\{0_M\})=0$.
\end{definition}

\begin{counterexample} \label{cex:noncommutative-rank-failure-endomorphism-ring} \hlt{Why Commutativity Matters for Rank}\\
Commutativity cannot be omitted from Theorem \ref{thm:free-module-rank-well-defined-commutative}. Here \emph{basis} and \emph{free} mean the left-module analogues of Definition \ref{def:free-module-basis}.\\
Let $\mathbb{F}$ be a field, let $V=\bigoplus\limits_{k\in\N_0}\mathbb{F}e_k$, and set $E\defeq\LL(V)$ with multiplication given by composition.\\
Define $p,u\in E$ by $p(e_0)=e_0$, $(\forall k\in\N)\ p(e_k)=0_V$, and $(\forall k\in\N_0)\ u(e_k)=e_{k+1}$. These operators exist uniquely by Theorem \ref{thm:free-module-map-extension}.\\
Ring $E$ has identity $\operatorname{id}_V$, $\operatorname{id}_V(e_0)=e_0\neq0_V$, so $\operatorname{id}_V\neq0_E$. Since $(pu)(e_0)=0_V\neq e_1=(up)(e_0)$, ring $E$ is noncommutative.\\
The left regular $E$-module $E$ is free on $\{\operatorname{id}_V\}$ because $(\forall h\in E)(\exists!\alpha\in E)\ h=\alpha\operatorname{id}_V$.\\
Fix $n\in\N$ and define $\beta_0,\ldots,\beta_{n-1}\in E$ on the basis by $(\forall(k,s)\in\N_0\times\{0,\ldots,n-1\})\ \beta_s(e_{kn+s})=e_k$.\\
On the complementary residue classes, $(\forall(k,s,t)\in\N_0\times\{0,\ldots,n-1\}^2)\,[s\neq t\Rightarrow\beta_s(e_{kn+t})=0_V]$.\\
These operators exist uniquely by Theorem \ref{thm:free-module-map-extension}, since Euclidean division indexes each basis vector exactly once.\\
If $(\alpha_s)_{s=0}^{n-1}\in E^n$ and $\sum\limits_{s=0}^{n-1}\alpha_s\beta_s=0_E$, then $(\forall(k,t)\in\N_0\times\{0,\ldots,n-1\})\ \alpha_t(e_k)=0_V$.\\
Hence $(\forall t\in\{0,\ldots,n-1\})\ \alpha_t=0_E$ by uniqueness in Theorem \ref{thm:free-module-map-extension}. Thus $\{\beta_0,\ldots,\beta_{n-1}\}$ is $E$-linearly independent.\\
Fix $h\in E$ and define $\alpha_0,\ldots,\alpha_{n-1}\in E$ by $(\forall(k,s)\in\N_0\times\{0,\ldots,n-1\})\ \alpha_s(e_k)\defeq h(e_{kn+s})$.\\
These operators exist uniquely by Theorem \ref{thm:free-module-map-extension}. Then $(\forall(k,t)\in\N_0\times\{0,\ldots,n-1\})\ (\sum\limits_{s=0}^{n-1}\alpha_s\beta_s)(e_{kn+t})=h(e_{kn+t})$.\\
Hence $h=\sum\limits_{s=0}^{n-1}\alpha_s\beta_s$ by the same uniqueness. Thus $\{\beta_0,\ldots,\beta_{n-1}\}$ is an $E$-basis of $E$.\\
If $s,t\in\{0,\ldots,n-1\}$ and $s\neq t$, then $\beta_s(e_s)=e_0\neq0_V=\beta_t(e_s)$. Therefore $\abs{\{\beta_0,\ldots,\beta_{n-1}\}}=n$.\\
Since $n\in\N$ was arbitrary, ${}_E E$ has bases of every positive finite cardinality. Thus rank invariance fails without commutativity.
\end{counterexample}

\begin{theorem} \label{thm:finite-support-functions-free-module} \hlt{Finite-Support Coordinate Modules}\\
Let $A$ be a nonzero commutative ring with identity, let $B$ be a set, and equip $A^B$ with pointwise operations.\\
Using Definition \ref{def:algebraic-support-of-function}, set $A^{(B)}\defeq\{f\in A^B:\abs{\operatorname{supp}(f)}<\infty\}$. For $b\in B$, define $\delta_b\in A^{(B)}$ by $\delta_b(b)=1_A$, $(\forall x\in B\setminus\{b\})\ \delta_b(x)=0_A$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A^{(B)}\leq_A A^B$.
\item The set $\{\delta_b:b\in B\}$ is an $A$-basis of $A^{(B)}$; equivalently, $A^{(B)}=\bigoplus\limits_{b\in B}A\delta_b$.
\item $\operatorname{rk}_A(A^{(B)})=\abs B$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The module identities in $A^B$ hold pointwise, and equality is by Assumption \ref{assump:function-extensionality}. The zero function has empty support.\\
$(\forall(a,c,f,g)\in A^2\times(A^{(B)})^2)\ \operatorname{supp}(af+cg)\subseteq\operatorname{supp}(f)\cup\operatorname{supp}(g)$. Thus $A^{(B)}\leq_A A^B$ by Theorem \ref{thm:submodule-test}.
\item $(\forall b\in B)\ \operatorname{supp}(\delta_b)=\{b\}$, so $\{\delta_b:b\in B\}\subseteq A^{(B)}$. The identity $f=\sum\limits_{b\in\operatorname{supp}(f)}f(b)\delta_b$ holds for $f\in A^{(B)}$ by Assumption \ref{assump:function-extensionality}.\\
Hence the $\delta_b$ span $A^{(B)}$. Let $F\subseteq B$ be finite and $(a_b)_{b\in F}\in A^F$ satisfy $\sum\limits_{b\in F}a_b\delta_b=0_{A^{(B)}}$.\\
Then $(\forall x\in F)\ a_x=0_A$ by evaluation at $x$. Hence $\{\delta_b:b\in B\}$ is independent by Definition \ref{def:module-linear-independence}.\\
If $x_b\in A\delta_b$ for $b\in F$ and $\sum\limits_{b\in F}x_b=0_{A^{(B)}}$, write $x_b=a_b\delta_b$. Then $(\forall b\in F)\ x_b=0_{A^{(B)}}$ by independence of $\{\delta_b:b\in B\}$.\\
Therefore $\sum\limits_{b\in B}A\delta_b$ is direct by Proposition \ref{prop:module-direct-sum-characterizations}(ii). $A^{(B)}=\bigoplus\limits_{b\in B}A\delta_b$ by the spanning established above.
\item The map $b\mapsto\delta_b$ is surjective onto $\{\delta_b:b\in B\}$ by definition. If $b\neq b'$, then $\delta_b(b)=1_A\neq0_A=\delta_{b'}(b)$, so the map is injective.\\
Thus $\operatorname{rk}_A(A^{(B)})=\abs B$ by Definition \ref{def:rank-of-free-module}.
\end{enumerate}
\hlt{Structural Commentary}: Support closure defines the submodule; delta evaluation proves independence, directness, and cardinality.\\
\hlt{Key Insight}: A delta coefficient equals the function value at its index.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:free-module-coordinate-model-and-isomorphism} \hlt{Free Modules Are Coordinate Modules}\\
Let $A$ be a nonzero commutative ring with identity.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $M$ is a free $A$-module with $A$-basis $B$, then $b\mapsto\delta_b$ extends to an isomorphism $\Phi_B:M\xrightarrow{\sim}A^{(B)}$.
\item If $M,N$ are free $A$-modules, then $M\cong N\Longleftrightarrow\operatorname{rk}_A(M)=\operatorname{rk}_A(N)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] Assignment $b\mapsto\delta_b$ extends uniquely to $\Phi_B\in\operatorname{Hom}_A(M,A^{(B)})$ by Theorem \ref{thm:free-module-map-extension}. Set $\{\delta_b:b\in B\}$ is basis by Theorem \ref{thm:finite-support-functions-free-module}(ii).\\
Thus $\delta_b\mapsto b$ extends uniquely to $\Psi_B\in\operatorname{Hom}_A(A^{(B)},M)$ by Theorem \ref{thm:free-module-map-extension}.\\
The composites $\Psi_B\Phi_B$ and $\Phi_B\Psi_B$ restrict to the identity on their respective bases.\\
Therefore they equal identity maps by uniqueness in Theorem \ref{thm:free-module-map-extension}. Hence $\Phi_B$ is isomorphism with inverse $\Psi_B$.
\item[$(ii)$] ($\Rightarrow$) Let $\psi:M\xrightarrow{\sim}N$ and fix an $A$-basis $C$ of $M$. The set $\psi(C)$ spans $N$ by surjectivity.\\
If $F\subseteq C$ is finite and $(a_c)_{c\in F}\in A^F$ satisfies $\sum\limits_{c\in F}a_c\psi(c)=0_N$, then $\sum\limits_{c\in F}a_cc=0_M$ by injectivity.\\
Thus $(\forall c\in F)\ a_c=0_A$ by independence of $C$. Set $\psi(C)$ is $A$-basis of $N$ by preceding spanning, independence.\\
Since $\psi|_C:C\to\psi(C)$ is a bijection, $\operatorname{rk}_A(M)=\abs C=\abs{\psi(C)}=\operatorname{rk}_A(N)$.
\item[$(ii)$] ($\Leftarrow$) Assume $\operatorname{rk}_A(M)=\operatorname{rk}_A(N)$ and fix $A$-bases $C,D$. Then $\abs C=\abs D$ by Definition \ref{def:rank-of-free-module}; choose bijection $u:C\to D$.\\
The maps $u,u^{-1}$ extend to homomorphisms $\Phi:M\to N$ and $\Psi:N\to M$ by Theorem \ref{thm:free-module-map-extension}.\\
Their composites restrict to the identity on $C,D$, respectively.\\
Hence $\Psi\Phi=\operatorname{id}_M$ and $\Phi\Psi=\operatorname{id}_N$ by uniqueness in Theorem \ref{thm:free-module-map-extension}, so $\Phi$ is an isomorphism.
\end{enumerate}
\hlt{Structural Commentary}: Basis-map extension constructs coordinate isomorphisms, while basis transport proves rank invariance.\\
\hlt{Key Insight}: A basis bijection extends to mutually inverse module maps.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:independent-sets-bounded-by-free-rank-domain} \hlt{Independent Sets Are Bounded by Rank over a Domain}\\
Let $A$ be integral domain with field of quotients $Q$, $M$ be free $A$-module. If $S\subseteq M$ is $A$-linearly independent, then $\abs S\leq\operatorname{rk}_A(M)$.
\begin{proof}
Fix an $A$-basis $B$ of $M$ and identify $M\cong A^{(B)}$ by Proposition \ref{prop:free-module-coordinate-model-and-isomorphism}(i).\\
The coordinate isomorphism preserves finite relations, so $S$ remains $A$-linearly independent.\\
Identify $A$ with its image in $Q$ by Proposition \ref{prop:field-of-quotients-embeds-domain}; then $A^{(B)}\subseteq Q^{(B)}$. Fix $n\in\N$ and distinct $s_1,\ldots,s_n\in S$.\\
Let $(q_1,\ldots,q_n)\in Q^n$ satisfy $\sum\limits_{i=1}^nq_is_i=0_{Q^{(B)}}$. For $i\in\{1,\ldots,n\}$, choose $(c_i,e_i)\in A\times(A\setminus\{0_A\})$ with $q_i=\frac{c_i}{e_i}$ by Definition \ref{def:field-of-quotients}.\\
Set $d\defeq\prod\limits_{i=1}^ne_i$. Then $d\neq0_A$ by integral-domain hypothesis, $(\forall i\in\{1,\ldots,n\})\ dq_i\in A$.\\
Relation $\sum\limits_{i=1}^n(dq_i)s_i=0_{Q^{(B)}}$ holds by multiplication by $d$. Both sides lie in $A^{(B)}$, so the same equality holds in $A^{(B)}$.\\
Thus $(\forall i\in\{1,\ldots,n\})\ dq_i=0_A$ by $A$-linear independence. Since $d\neq0_Q$, $(\forall i\in\{1,\ldots,n\})\ q_i=0_Q$.\\
Hence $S$ is $Q$-linearly independent in $Q^{(B)}$ by Definition \ref{def:module-linear-independence}. Choose $Q$-basis $C$ of $Q^{(B)}$ with $S\subseteq C$ by Theorem \ref{thm:hamel-basis-extension}(iii).\\
Set $\{\delta_b:b\in B\}$ is $Q$-basis by Theorem \ref{thm:finite-support-functions-free-module}(ii); $\abs C=\abs B$ by Theorem \ref{thm:arbitrary-bases-same-cardinality}; $\abs S\leq\abs C=\abs B=\operatorname{rk}_A(M)$ by $S\subseteq C$, Definition \ref{def:rank-of-free-module}.\\
\hlt{Structural Commentary}: Coordinate embedding and denominator clearing preserve independence before vector-space basis comparison.\\
\hlt{Key Insight}: One common denominator converts each finite $Q$-relation into an $A$-relation.
\end{proof}
\end{theorem}
