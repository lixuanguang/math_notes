\subsection{Smith Normal Form and Module Presentations}

\begin{definition} \label{def:finite-free-presentation-and-cokernel} \hlt{Finite Free Presentations and Cokernels}\\
Let $m,n\in\N_0$ and let $A\in\operatorname{Mat}(m,n,R)$ act by $A:R^n\to R^m$. The \hlt{cokernel} of $A$ is $\operatorname{coker}(A)\defeq R^m/\operatorname{im}(A)$.\\
The matrix $A$ is \hlt{finite presentation matrix} for $R$-module $M$ if $M\cong\operatorname{coker}(A)$. Notation $R^n\xrightarrow{A}R^m\to M\to0$ records this presentation.
\end{definition}

\begin{proposition} \label{prop:finitely-generated-pid-module-finite-presentation} \hlt{Finitely Generated PID Modules Are Finitely Presented}\\
Let $M$ be a f.g. $R$-module. Then there are $m,n\in\N_0$ and $A\in\operatorname{Mat}(m,n,R)$ such that $M\cong\operatorname{coker}(A)$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Free Cover}: Choose generators $u_1,\ldots,u_m$ of $M$ and define $\varepsilon:R^m\to M$ by $\varepsilon((a_i)_{i=1}^m)=\sum_{i=1}^m a_iu_i$.\\
Its image contains every $u_i$, so it equals $M$ by Proposition \ref{prop:generated-submodule-smallest}(iii). Thus $\varepsilon$ is an epimorphism by Definition \ref{def:module-homomorphism}(iv).
\item \hlt{Relation Module}: Set $K\defeq\ker\varepsilon\leq R^m$ by Proposition \ref{prop:hom-module-kernel-image}(ii).\\
By Theorem \ref{thm:submodule-of-finite-free-pid-is-free}, choose $n\in\N_0$ with $n\leq m$ and an isomorphism $\theta:R^n\xrightarrow{\sim}K$.
\item \hlt{Presentation Matrix}: Let $A\in\operatorname{Mat}(m,n,R)$ be the matrix of the composite $R^n\xrightarrow{\theta}K\hookrightarrow R^m$ in the standard bases.\\
Then $\operatorname{im}(A)=K$, and $M\cong R^m/K=\operatorname{coker}(A)$ by Theorem \ref{thm:module-isomorphism-theorems}(i).
\end{enumerate}
\hlt{Structural Commentary}: A finite generating map has a finite free kernel over a PID.\\
\hlt{Key Insight}: The inclusion of the relation module is a finite presentation matrix.
\end{proof}
\end{proposition}

\begin{definition} \label{def:pid-matrix-equivalence-and-determinantal-ideals} \hlt{Matrix Equivalence and Determinantal Ideals over a PID}\\
Let $m,n\in\N_0$, set $q\defeq\min(m,n)$, and let $A,B\in\operatorname{Mat}(m,n,R)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The matrices $A$ and $B$ are \hlt{equivalent}, written $A\sim_{\mathrm{eq}}B$.\\
This relation is defined by $(\exists P\in\operatorname{Mat}(m,m,R)^\times)(\exists Q\in\operatorname{Mat}(n,n,R)^\times)\,[B=PAQ]$.
\item Set $\Delta_0(A)\defeq R$. For $1\leq k\leq q$, the \hlt{$k$th determinantal ideal} $\Delta_k(A)\trianglelefteq R$ is generated by all $k\times k$ minors of $A$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:smith-normal-form-over-pid} \hlt{Smith Normal Form over a PID}\\
Let $m,n\in\N_0$, $A\in\operatorname{Mat}(m,n,R)$, $q\defeq\min(m,n)$. For $0\leq r\leq q$ and $\mathbf{d}=(d_i)_{i=1}^r$, set $D_{\mathbf{d}}\defeq\operatorname{diag}(d_1,\ldots,d_r,0,\ldots,0)\in\operatorname{Mat}(m,n,R)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\exists$ $0\leq r\leq q$ and $d_1,\ldots,d_r\in R\setminus\{0_R\}$ such that $d_1\mid\cdots\mid d_r$ and $A\sim_{\mathrm{eq}}D_{\mathbf{d}}$. For these data, $r=\operatorname{rk}_R(\operatorname{im}(A))$.
\item Suppose $A\sim_{\mathrm{eq}}D_{\mathbf{d}}$ and $A\sim_{\mathrm{eq}}D_{\mathbf{e}}$ for nonzero divisibility chains $\mathbf{d}=(d_i)_{i=1}^r$ and $\mathbf{e}=(e_i)_{i=1}^s$.\\
Then $r=s$ and $(\forall i\in\{1,\ldots,r\})\,[\langle d_i\rangle=\langle e_i\rangle]$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Bézout Pivot}: The existence assertion is proved by induction on $q$.\\
If $q=0$ or $A=0$, take $r=0$ and $D_{\mathbf{d}}=0$. Then $A\sim_{\mathrm{eq}}D_{\mathbf{d}}$ and $\operatorname{rk}_R(\operatorname{im}(A))=0$.\\
Otherwise, invertible permutation matrices move a nonzero entry to position $(1,1)$; denote the resulting pivot by $p$.\\
Suppose an entry $a$ in the first column is not divisible by $p$.\\
Choose $g\in R$ with $\langle p,a\rangle=\langle g\rangle$ by Definition \ref{def:principal-ideal-domain-and-ufd}(i), and write $p=gu$ and $a=gv$.\\
Since $g\in\langle p,a\rangle$, choose $x,y\in R$ with $g=xp+ya=g(xu+yv)$. Then $xu+yv=1_R$ by $g\neq0_R$ and Proposition \ref{prop:integral-domain-cancellation}.\\
Set $U\defeq\left(\begin{smallmatrix}x&y\\-v&u\end{smallmatrix}\right)$. The matrix $U^{-1}=\left(\begin{smallmatrix}u&-y\\v&x\end{smallmatrix}\right)$ is its inverse, and $U\left(\begin{smallmatrix}p\\a\end{smallmatrix}\right)=\left(\begin{smallmatrix}g\\0_R\end{smallmatrix}\right)$.\\
Embedding $U$ into the corresponding two rows replaces the pivot by $g$.\\
Right multiplication by the embedded transpose $U^{\mathsf{T}}$ is the analogous first-row reduction. Finally, $\langle p\rangle\subsetneq\langle p,a\rangle=\langle g\rangle$ as $p\nmid a$.
\item \hlt{Termination}: For a fixed pivot, entries divisible by it are cleared from its row and column by invertible row and column additions. If an interior entry is not divisible by the pivot after this clearing, adding its row to the first row exposes it to the Bézout pivot operation. Every such pivot improvement strictly enlarges the principal pivot ideal.\\
Only finitely many improvements occur by Proposition \ref{prop:pid-ascending-chain-condition}. At termination, the pivot $p$ divides every entry, so the first row and column can be cleared to obtain a matrix equivalent to $\operatorname{diag}(p,B)$. Every entry of $B$ remains divisible by $p$.
\item \hlt{Block Induction}: The complementary matrix $B$ has $\min(m-1,n-1)=q-1$. By the induction hypothesis, invertible operations internal to this block put it in Smith form with nonzero diagonal entries $d_2\mid\cdots\mid d_r$.\\
Those operations take $R$-linear combinations of entries of $B$, so $(\forall i\in\{2,\ldots,r\})\,[p\mid d_i]$.\\
Setting $d_1\defeq p$ establishes $d_1\mid\cdots\mid d_r$ and $A\sim_{\mathrm{eq}}D_{\mathbf{d}}$.
\item \hlt{Rank}: If $D_{\mathbf{d}}=PAQ$, then $\operatorname{im}(D_{\mathbf{d}})=P(\operatorname{im}(A))$ because $Q$ is surjective. The vectors $d_1e_1,\ldots,d_re_r$ span $\operatorname{im}(D_{\mathbf{d}})$.\\
They are linearly independent because $R$ is an integral domain and every $d_i$ is nonzero.\\
Hence their inverse images under $P$ form a basis of $\operatorname{im}(A)$, so $\operatorname{rk}_R(\operatorname{im}(A))=r$.
\end{enumerate}
\item Let $B=PAQ$ with $P$ and $Q$ invertible. Two applications of the Cauchy--Binet expansion express each $k\times k$ minor of $B$ as an $R$-linear combination of the $k\times k$ minors of $A$. Thus $\Delta_k(B)\subseteq\Delta_k(A)$.\\
Same argument applied to $A=P^{-1}BQ^{-1}$ proves reverse inclusion, so matrix equivalence preserves every determinantal ideal.\\
For $1\leq k\leq r$, each nonzero $k\times k$ minor of $D_{\mathbf{d}}$ is a product $d_{i_1}\cdots d_{i_k}$ with $i_j\geq j$.\\
It is therefore divisible by $d_1\cdots d_k$, while the leading $k\times k$ minor equals this latter product.\\
Consequently, $\Delta_k(D_{\mathbf{d}})=\langle d_1\cdots d_k\rangle$ for $k\leq r$, while $\Delta_k(D_{\mathbf{d}})=\{0_R\}$ for $r<k\leq q$. The same formulas hold for $D_{\mathbf{e}}$.\\
Since $R$ is an integral domain, the nonzero determinantal ideals occur precisely for $k\leq r$ and for $k\leq s$, respectively; hence $r=s$.\\
If $r=0$, the uniqueness assertion follows. Assume $r\geq1$. Thus $\langle d_1\rangle=\langle e_1\rangle$ by equality of the first determinantal ideals.\\
Assume inductively that these principal ideals agree through index $k-1$.\\
By Proposition \ref{prop:principal-ideals-divisibility-associates}(ii), the corresponding generators are associates.\\
After absorbing their unit factors, equality of the $k$th determinantal ideals becomes $\langle cd_k\rangle=\langle ce_k\rangle$, where $c=d_1\cdots d_{k-1}\neq0_R$.\\
Thus $\langle d_k\rangle=\langle e_k\rangle$ by mutual divisibility and Proposition \ref{prop:integral-domain-cancellation}, completing the induction.
\end{enumerate}
\noindent
\hlt{Structural Commentary}: Bézout pivots establish existence; invertible row and column operations preserve determinantal ideals.\\
\hlt{Key Insight}: The ideals $\Delta_k(A)$ recover the successive invariant factors up to associates.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:adapted-basis-submodule-free-pid} \hlt{Adapted Bases for Submodules of Free Modules}\\
Let $n\in\N_0$, $F$ be free $R$-module with $\operatorname{rk}_R(F)=n$, $N\leq F$. There are $k\in\N_0$ with $k\leq n$ and an $R$-basis $(v_i)_{i=1}^n$ of $F$.\\
For these choices, there are $r_1,\ldots,r_k\in R\setminus\{0_R\}$ such that $r_1\mid\cdots\mid r_k$ and $(r_iv_i)_{i=1}^k$ is an $R$-basis of $N$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Free Setup}: By Theorem \ref{thm:submodule-of-finite-free-pid-is-free}, choose $k\in\N_0$ with $k\leq n$ and $N\cong R^k$.\\
Choose an $R$-basis $(w_j)_{j=1}^k$ of $N$ and an $R$-basis $(f_i)_{i=1}^n$ of $F$.\\
Let $A\in\operatorname{Mat}(n,k,R)$ be the matrix of the inclusion $\iota:N\hookrightarrow F$ in these bases.
\item \hlt{Zero Case}: If $k=0$, then $N=\{0_F\}$, chosen basis of $F$ together with empty basis of $N$ satisfies conclusion. Assume $k\geq1$.
\item \hlt{Smith Reduction}: The Smith diagonal of $A$ has $k$ nonzero entries by Theorem \ref{thm:smith-normal-form-over-pid}(i), since $\operatorname{im}(\iota)=N$ has rank $k$.\\
Denote these entries by $r_1\mid\cdots\mid r_k$ and set $D\defeq\operatorname{diag}(r_1,\ldots,r_k,0,\ldots,0)$.\\
Choose $P\in\operatorname{Mat}(n,n,R)^\times$ and $Q\in\operatorname{Mat}(k,k,R)^\times$ with $PAQ=D$ by Definition \ref{def:pid-matrix-equivalence-and-determinantal-ideals}(i).
\item \hlt{Basis Transport}: Define $w'_j\defeq\sum\limits_{\ell=1}^kQ_{\ell j}w_\ell$ for $1\leq j\leq k$. Define $v_i\defeq\sum\limits_{\ell=1}^n(P^{-1})_{\ell i}f_\ell$ for $1\leq i\leq n$.\\
Since $Q$ and $P^{-1}$ are invertible, $(w'_j)_{j=1}^k$ and $(v_i)_{i=1}^n$ are bases of $N$ and $F$, respectively.\\
Identity $AQ=P^{-1}D$ implies $(\forall j\in\{1,\ldots,k\})\,[\iota(w'_j)=r_jv_j]$. Because $\iota$ is inclusion, $(r_jv_j)_{j=1}^k=(w'_j)_{j=1}^k$ is $R$-basis of $N$.
\end{enumerate}
\hlt{Structural Commentary}: Smith reduction changes the domain and codomain bases of the inclusion.\\
\hlt{Key Insight}: In the transported bases, the inclusion sends the $j$th basis vector to $r_jv_j$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:smith-normal-form-presents-pid-modules} \hlt{Smith Form Presents Finitely Generated PID Modules}\\
Let $A\in\operatorname{Mat}(m,n,R)$ be a finite presentation matrix for $M$. Suppose $A\sim_{\mathrm{eq}}D$, where $D\defeq\operatorname{diag}(d_1,\ldots,d_r,0,\ldots,0)$.\\
Assume $0\leq r\leq\min(m,n)$, $d_1\mid\cdots\mid d_r$ are nonzero. Set $I\defeq\{i\in\{1,\ldots,r\}:d_i\notin R^\times\}$. Then $M\cong R^{m-r}\oplus\bigoplus_{i\in I}R/\langle d_i\rangle$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Equivalent Cokernels}: Choose invertible $P,Q$ with $D=PAQ$ by Definition \ref{def:pid-matrix-equivalence-and-determinantal-ideals}(i). As $Q(R^n)=R^n$, thus $\operatorname{im}(D)=P(\operatorname{im}(A))$.\\
Consequently, $x+\operatorname{im}(A)\mapsto Px+\operatorname{im}(D)$ is an isomorphism $\operatorname{coker}(A)\xrightarrow{\sim}\operatorname{coker}(D)$; its inverse is induced by $P^{-1}$.
\item \hlt{Diagonal Cokernel}: With $(e_i)_{i=1}^m$ the standard basis of $R^m$, $\operatorname{im}(D)=\bigoplus_{i=1}^r\langle d_i\rangle e_i$.\\
Thus, coordinatewise quotienting produces $\operatorname{coker}(D)\cong\left(\bigoplus_{i=1}^rR/\langle d_i\rangle\right)\oplus R^{m-r}$.
\item \hlt{Unit Factors}: If $d_i\in R^\times$, then $R/\langle d_i\rangle=\{0_R\}$.\\
Removing these zero summands and using $M\cong\operatorname{coker}(A)$ proves the asserted decomposition.
\end{enumerate}
\hlt{Structural Commentary}: Smith form turns presentations into direct sums.\\
\hlt{Key Insight}: Diagonal relations create cyclic torsion summands, and zero rows create free summands.
\end{proof}
\end{corollary}

\begin{perspective} \label{per:smith-normal-form-and-module-presentations} \hlt{Smith Form as Diagonalized Relations}\\
Proposition \ref{prop:finitely-generated-pid-module-finite-presentation} encodes generators and relations as a finite matrix cokernel.\\
Theorem \ref{thm:smith-normal-form-over-pid} changes bases to isolate those relations.\\
Corollary \ref{cor:smith-normal-form-presents-pid-modules} reads off the resulting cyclic and free summands.\\
This mechanism concerns finite presentations; it does not classify arbitrary non-f.g. modules.
\end{perspective}

\begin{example} \label{ex:smith-normal-form-abelian-group-decomposition} \hlt{Smith Form for a Finite Abelian Group}\\
Let $M=\mathbb{Z}^2/\langle\!\langle(6,0),(0,10),(2,4)\rangle\!\rangle_{\mathbb{Z}}$. Its three displayed relations are the columns of the finite presentation matrix $A=\left(\begin{smallmatrix}6&0&2\\0&10&4\end{smallmatrix}\right)$.\\
The entries generate $\Delta_1(A)=\langle6,10,2,4\rangle=\langle2\rangle$.\\
The $2\times2$ minors from the three column pairs are $60$, $24$, and $-20$, so $\Delta_2(A)=\langle60,24,-20\rangle=\langle4\rangle$.\\
The Smith form has two nonzero invariant factors by the determinantal-ideal formulas in the proof of Theorem \ref{thm:smith-normal-form-over-pid}.\\
Those factors satisfy $d_1\sim2$ and $d_1d_2\sim4$ by same formulas. Hence $d_2\sim2$ by cancellation; $M\cong\mathbb{Z}/2\mathbb{Z}\oplus\mathbb{Z}/2\mathbb{Z}$ by Corollary \ref{cor:smith-normal-form-presents-pid-modules}.\\
\hlt{Key Insight}: The first two determinantal ideals recover the two cyclic summands.
\end{example}
