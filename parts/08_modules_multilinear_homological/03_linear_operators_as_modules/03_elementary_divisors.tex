\subsection{Elementary Divisors}

\begin{theorem} \label{thm:operator-primary-cyclic-decomposition} \hlt{Primary Cyclic Decomposition for an Operator}\\
Let $0<\dim\mathcal{V}<\infty$, $T\in\LL(\mathcal{V})$, $s\in\N$. Set $I\defeq\{1,\ldots,s\}$. Let $(p_i)_{i\in I}\in\mathbb{F}[z]^I$ be distinct monic irreducibles and $(e_i)_{i\in I}\in\N^I$.\\
Suppose $\mu_T=\prod\limits_{i\in I}p_i^{e_i}$ and set $\mathcal{V}_{p_i}\defeq\ker p_i(T)^{e_i}$ for $i\in I$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{V}_T=\bigoplus\limits_{i\in I}\mathcal{V}_{p_i}$.
\item For $i\in I$, there are $k_i\in\N$ and exponents $e_i=e_{i,1}\geq\cdots\geq e_{i,k_i}\geq1$. There are vectors $(\mathbf{v}_{i,j})_{j=1}^{k_i}\in(\mathcal{V}_{p_i}\setminus\{\mathbf{0}\})^{k_i}$.\\
These data satisfy $\mathcal{V}_{p_i}=\bigoplus\limits_{j=1}^{k_i}\mathcal{C}_T(\mathbf{v}_{i,j})$. They satisfy $m_{\mathbf{v}_{i,j}}=p_i^{e_{i,j}}$ for $j\in\{1,\ldots,k_i\}$.
\item For $i\in I$ and $j\in\{1,\ldots,k_i\}$, set $d_{i,j}\defeq\deg(p_i^{e_{i,j}})$. Set $\mathcal{B}_{i,j}\defeq(\mathbf{v}_{i,j},T\mathbf{v}_{i,j},\ldots,T^{d_{i,j}-1}\mathbf{v}_{i,j})$.\\
Then $\operatorname{ElemDiv}(T)=\{p_i^{e_{i,j}}:i\in I,\ 1\leq j\leq k_i\}$ as multisets.\\
The lexicographically ordered concatenation of the $\mathcal{B}_{i,j}$ is an elementary divisor basis.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $\dim\mathcal{V}>0$, $\mathcal{V}_T\neq\{\mathbf{0}\}$. It is finitely generated and torsion by Proposition \ref{prop:operator-modules-are-fg-torsion}(i)--(ii).\\
Its order satisfies $o(\mathcal{V}_T)\sim\mu_T$ by Proposition \ref{prop:minimal-polynomial-module-order}(i). For $i\in I$, set $M_i\defeq\{\mathbf{v}\in\mathcal{V}:p_i^{e_i}\cdot\mathbf{v}=\mathbf{0}\}$.\\
Then $M_i$ is the $p_i$-primary component and $\mathcal{V}_T=\bigoplus\limits_{i\in I}M_i$ by Theorem \ref{thm:primary-decomposition-pid-modules}(i).\\
The identity $p_i^{e_i}\cdot\mathbf{v}=p_i(T)^{e_i}\mathbf{v}$ holds by Definition \ref{def:operator-module-and-matrix-module}. Hence $M_i=\ker p_i(T)^{e_i}=\mathcal{V}_{p_i}$.
\item For $i\in I$, $\mathcal{V}_{p_i}$ is finitely generated and $p_i$-primary by Theorem \ref{thm:primary-decomposition-pid-modules}(ii). Its order satisfies $o(\mathcal{V}_{p_i})\sim p_i^{e_i}$ by the same theorem.\\
Choose $k_i$, $(\mathbf{v}_{i,j})_{j=1}^{k_i}$, and $(e_{i,j})_{j=1}^{k_i}$ by Theorem \ref{thm:cyclic-decomposition-primary-pid}(i).\\
They satisfy the exponent chain and $\mathcal{V}_{p_i}=\bigoplus\limits_{j=1}^{k_i}\mathbb{F}[z]\mathbf{v}_{i,j}$ by the same theorem.\\
Their annihilators are $\langle p_i^{e_{i,j}}\rangle$ by the same theorem. Each $\mathbf{v}_{i,j}$ is nonzero because $\langle p_i^{e_{i,j}}\rangle\neq\mathbb{F}[z]$.\\
$\mathbb{F}[z]\mathbf{v}_{i,j}=\mathcal{C}_T(\mathbf{v}_{i,j})$ by Proposition \ref{prop:cyclic-subspace-smallest-invariant-subspace}(i). Thus $m_{\mathbf{v}_{i,j}}=p_i^{e_{i,j}}$ by Definition \ref{def:vector-order-under-operator} and monicity.
\item $\operatorname{ElemDiv}(\mathcal{V}_T)=\{p_i^{e_{i,j}}:i\in I,\ 1\leq j\leq k_i\}$ by part (ii) and Definition \ref{def:elementary-divisors-pid-module}.\\
The multiset identity in (iii) follows by Definition \ref{def:operator-invariant-factors-elementary-divisors}(iii) and monicity. Each $\mathcal{B}_{i,j}$ is a basis of $\mathcal{C}_T(\mathbf{v}_{i,j})$ by Theorem \ref{thm:cyclic-subspace-quotient-and-companion}(ii).\\
Their lexicographically ordered concatenation is basis of $\mathcal{V}$ by parts (i)--(ii). It is elementary divisor basis by Definition \ref{def:operator-invariant-factors-elementary-divisors}(iv).
\end{enumerate}
\hlt{Structural Commentary}: Primary decomposition identifies kernels; cyclic decomposition refines each into annihilator blocks.\\
\hlt{Key Insight}: Each elementary divisor is one cyclic summand's monic annihilator.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:reducing-decompositions-combine-elementary-divisors} \hlt{Reducing Decompositions Combine the Data}\\
Let $0<\dim\mathcal{V}<\infty$, let $T\in\LL(\mathcal{V})$, and suppose $\mathcal{V}=\mathcal{A}\oplus\mathcal{B}$.\\
Assume $\mathcal{A}$ and $\mathcal{B}$ are nonzero $T$-invariant subspaces. Set $T_{\mathcal{A}}\defeq T|_{\mathcal{A}}$ and $T_{\mathcal{B}}\defeq T|_{\mathcal{B}}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu_T=\operatorname{lcm}(\mu_{T_{\mathcal{A}}},\mu_{T_{\mathcal{B}}})$.
\item $\operatorname{ElemDiv}(T)=\operatorname{ElemDiv}(T_{\mathcal{A}})\uplus\operatorname{ElemDiv}(T_{\mathcal{B}})$. Here $\uplus$ denotes multiset union.
\item The multiset in (ii) is grouped and zero-padded as in Definition \ref{def:invariant-factors-pid-module}. Its column products form $\operatorname{InvFact}(T)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The subspaces $\mathcal{A}$ and $\mathcal{B}$ are $\mathbb{F}[z]$-submodules by Theorem \ref{thm:operator-submodules-invariant-subspaces}. Hence $\mathcal{V}_T=\mathcal{A}_{T_{\mathcal{A}}}\oplus\mathcal{B}_{T_{\mathcal{B}}}$ by the assumed direct sum.\\
The order of this module sum is the least common multiple of the summand orders by Proposition \ref{prop:orders-direct-sum-lcm}.\\
They are $\mu_T$, $\mu_{T_{\mathcal{A}}}$, and $\mu_{T_{\mathcal{B}}}$ by Proposition \ref{prop:minimal-polynomial-module-order}(i), proving (i).
\item Set $M_{\mathcal{A}}\defeq\mathcal{A}_{T_{\mathcal{A}}}$ and $M_{\mathcal{B}}\defeq\mathcal{B}_{T_{\mathcal{B}}}$. $\mathcal{V}_T=M_{\mathcal{A}}\oplus M_{\mathcal{B}}$ by the assumed direct sum and Theorem \ref{thm:operator-submodules-invariant-subspaces}.\\
Hence $\operatorname{ElemDiv}(\mathcal{V}_T)=\operatorname{ElemDiv}(M_{\mathcal{A}}\oplus M_{\mathcal{B}})$. $\operatorname{ElemDiv}(M_{\mathcal{A}}\oplus M_{\mathcal{B}})=\operatorname{ElemDiv}(M_{\mathcal{A}})\uplus\operatorname{ElemDiv}(M_{\mathcal{B}})$ by Proposition \ref{prop:elementary-divisors-direct-sums}.\\
The identity in (ii) follows by Definition \ref{def:operator-invariant-factors-elementary-divisors}(iii) after taking monic representatives.
\item Grouping, padding, and column multiplication produce $\operatorname{Inv}(\mathcal{V}_T)$ from (ii) by Definition \ref{def:invariant-factors-pid-module}.\\
Taking its unique monic representatives produces $\operatorname{InvFact}(T)$ by Definition \ref{def:operator-invariant-factors-elementary-divisors}(ii).
\end{enumerate}
\hlt{Structural Commentary}: The reducing sum becomes a module sum; order and elementary-divisor laws separate its data.\\
\hlt{Key Insight}: Direct sums take least common multiples and multiset unions.
\end{proof}
\end{theorem}

\begin{definition} \label{def:characteristic-polynomial-from-elementary-divisors} \hlt{Characteristic Polynomial from Elementary Divisors}\\
Let $0<\dim\mathcal{V}<\infty$ and $T\in\LL(\mathcal{V})$. Define $\chi_T(z)\defeq\prod\limits_{q\in\operatorname{ElemDiv}(T)}q(z)$, retaining multiset multiplicities.\\
For $n\geq1$ and $A\in\operatorname{Mat}(n,n,\mathbb{F})$, set $\chi_A\defeq\chi_{T_A}$.
\end{definition}

\begin{theorem} \label{thm:minimal-characteristic-elementary-divisor-relations} \hlt{Minimal and Characteristic Polynomials from Elementary Divisors}\\
Let $0<\dim\mathcal{V}<\infty$, $T\in\LL(\mathcal{V})$, $s\in\N$. $I\defeq\{1,\ldots,s\}$. Let $(k_i)_{i\in I}\in\N^I$, $(p_i)_{i\in I}\in\mathbb{F}[z]^I$ be distinct monic irreducibles.\\
Let $e_{i,j}\in\N$ for $i\in I$, $j\in\{1,\ldots,k_i\}$. Let $\operatorname{ElemDiv}(T)=\{p_i^{e_{i,j}}:i\in I,\ 1\leq j\leq k_i\}$ be multisets. For $i\in I$, set $m_i\defeq\max\limits_{1\leq j\leq k_i}e_{i,j}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu_T=\prod\limits_{i\in I}p_i^{m_i}$.
\item $q_T=\chi_T=\prod\limits_{i\in I}\prod\limits_{j=1}^{k_i}p_i^{e_{i,j}}$.
\item $\deg\chi_T=\deg q_T=\dim\mathcal{V}$.
\item $\mu_T\mid q_T$, and $\mu_T$, $q_T$, and $\chi_T$ have the same monic irreducible factors.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The module $\mathcal{V}_T$ is finitely generated and torsion by Proposition \ref{prop:operator-modules-are-fg-torsion}(i)--(ii).\\
Choose a primary cyclic decomposition of $\mathcal{V}_T$ by Theorem \ref{thm:primary-cyclic-decomposition-pid}(i).\\
Its cyclic orders form $\operatorname{ElemDiv}(\mathcal{V}_T)$ by Definition \ref{def:elementary-divisors-pid-module}. Their monic representatives form $\operatorname{ElemDiv}(T)$ by Definition \ref{def:operator-invariant-factors-elementary-divisors}(iii).\\
Set $\lambda\defeq\operatorname{lcm}\{p_i^{e_{i,j}}:i\in I,\ 1\leq j\leq k_i\}$. Then $o(\mathcal{V}_T)\sim\lambda$ by repeated application of Proposition \ref{prop:orders-direct-sum-lcm}.\\
The monic representative of this order is $\mu_T$ by Proposition \ref{prop:minimal-polynomial-module-order}(i).\\
Since the $p_i$ are distinct and monic, this least common multiple is $\prod\limits_{i\in I}p_i^{m_i}$.
\item Write $\operatorname{InvFact}(T)=(f_1,\ldots,f_r)$. Then $q_T=\prod\limits_{\ell=1}^rf_\ell$ by Theorem \ref{thm:operator-invariant-factor-skeleton}(iv).\\
The column construction uses every elementary divisor once by Definition \ref{def:invariant-factors-pid-module}.\\
Its monic column products are the $f_\ell$ by Definition \ref{def:operator-invariant-factors-elementary-divisors}(ii)--(iii).\\
Hence $\prod\limits_{\ell=1}^rf_\ell=\prod\limits_{i\in I}\prod\limits_{j=1}^{k_i}p_i^{e_{i,j}}$. The latter product is $\chi_T$ by Definition \ref{def:characteristic-polynomial-from-elementary-divisors}.
\item $\deg q_T=\dim\mathcal{V}$ by Proposition \ref{prop:operator-characteristic-polynomial-degree-roots}(i). The equality $\deg\chi_T=\deg q_T$ holds by part (ii).
\item For $i\in I$, set $b_i\defeq\sum\limits_{j=1}^{k_i}e_{i,j}$. Then $m_i$ is exponent of $p_i$ in $\mu_T$ by part (i), $b_i$ is its exponent in $q_T=\chi_T$ by part (ii).\\
Since $e_{i,j}\geq1$, $1\leq m_i\leq b_i$. Thus $\mu_T\mid q_T$, and all three polynomials have the monic irreducible factors $(p_i)_{i\in I}$.
\end{enumerate}
\hlt{Structural Commentary}: Module orders take primewise maxima; invariant-factor products retain every primary exponent.\\
\hlt{Key Insight}: Minimal exponents take maxima; characteristic exponents take sums.
\end{proof}
\end{theorem}

\begin{definition} \label{def:nonderogatory-operator-module} \hlt{Nonderogatory Operators}\\
Let $0<\dim\mathcal{V}<\infty$ and $T\in\LL(\mathcal{V})$. The operator $T$ is \hlt{nonderogatory} if $\mu_T=q_T$.\\
The equivalence $\mu_T=q_T\Longleftrightarrow\mu_T=\chi_T$ holds by Theorem \ref{thm:minimal-characteristic-elementary-divisor-relations}(ii).\\
The equivalence $\mu_T=q_T\Longleftrightarrow\deg\mu_T=\dim\mathcal{V}$ holds by Theorem \ref{thm:minimal-characteristic-elementary-divisor-relations}(iii)--(iv).\\
For $n\geq1$ and $A\in\operatorname{Mat}(n,n,\mathbb{F})$, $A$ is nonderogatory if $T_A$ is.
\end{definition}

\begin{theorem} \label{thm:nonderogatory-cyclic-equivalences} \hlt{Cyclic and Nonderogatory Are the Same}\\
Let $0<\dim\mathcal{V}<\infty$, $T\in\LL(\mathcal{V})$, $s\in\N$, $I\defeq\{1,\ldots,s\}$. Let $(p_i)_{i\in I}\in\mathbb{F}[z]^I$ be distinct monic irreducibles and $(e_i)_{i\in I}\in\N^I$.\\
Suppose $\mu_T=\prod\limits_{i\in I}p_i^{e_i}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{V}_T$ is cyclic.
\item $T$ has a cyclic vector.
\item $\operatorname{ElemDiv}(T)=\{p_1^{e_1},\ldots,p_s^{e_s}\}$ as multisets.
\item $T$ is nonderogatory.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ Choose $\mathbf{v}\in\mathcal{V}$ with $\mathcal{V}_T=\mathbb{F}[z]\mathbf{v}$ by Definition \ref{def:cyclic-and-finitely-generated-modules}(iii). Then $\mathcal{V}=\mathcal{C}_T(\mathbf{v})$ by Proposition \ref{prop:cyclic-subspace-smallest-invariant-subspace}(i).\\
Thus $\mathbf{v}$ is cyclic for $T$ by Definition \ref{def:cyclic-subspace-and-vector-annihilator}(ii).
\item[$(ii)$]$\Rightarrow (iii)$ Let $\mathbf{v}$ be cyclic for $T$. Then $\mathcal{V}=\mathcal{C}_T(\mathbf{v})$ by Definition \ref{def:cyclic-subspace-and-vector-annihilator}(ii). Thus $\mathcal{V}_T=\mathbb{F}[z]\mathbf{v}$ by Proposition \ref{prop:cyclic-subspace-smallest-invariant-subspace}(i).\\
Since $\dim\mathcal{V}>0$, $\mathcal{V}_T\neq\{\mathbf{0}\}$. It is finitely generated and torsion by Proposition \ref{prop:operator-modules-are-fg-torsion}(i)--(ii).\\
Its order satisfies $o(\mathcal{V}_T)\sim\mu_T$ by Proposition \ref{prop:minimal-polynomial-module-order}(i). Hence $\operatorname{ElemDiv}(\mathcal{V}_T)=\{p_i^{e_i}:i\in I\}$ by Theorem \ref{thm:cyclic-torsion-module-characterization}(i),(iii).\\
$\operatorname{ElemDiv}(T)=\{p_i^{e_i}:i\in I\}$ by Definition \ref{def:operator-invariant-factors-elementary-divisors}(iii) and monicity.
\item[$(iii)$]$\Rightarrow (iv)$ The identity $\mu_T=q_T=\prod\limits_{i\in I}p_i^{e_i}$ holds by Theorem \ref{thm:minimal-characteristic-elementary-divisor-relations}(i)--(ii). Thus $T$ is nonderogatory by Definition \ref{def:nonderogatory-operator-module}.
\item[$(iv)$]$\Rightarrow (i)$ Choose $k_i\in\N$ and $e_i=e_{i,1}\geq\cdots\geq e_{i,k_i}\geq1$ by Theorem \ref{thm:operator-primary-cyclic-decomposition}(ii).\\
Then $\operatorname{ElemDiv}(T)=\{p_i^{e_{i,j}}:i\in I,\ 1\leq j\leq k_i\}$ by Theorem \ref{thm:operator-primary-cyclic-decomposition}(iii). Identity $\mu_T=q_T$ holds by condition (iv), Definition \ref{def:nonderogatory-operator-module}.\\
For $i\in I$, $e_i=\sum\limits_{j=1}^{k_i}e_{i,j}$ by Theorem \ref{thm:minimal-characteristic-elementary-divisor-relations}(i)--(ii). The inequalities $e_{i,j}\geq1$ and $e_i=e_{i,1}$ force $k_i=1$.\\
Hence $\operatorname{ElemDiv}(T)=\{p_i^{e_i}:i\in I\}$. $\operatorname{ElemDiv}(\mathcal{V}_T)=\{p_i^{e_i}:i\in I\}$ by Definition \ref{def:operator-invariant-factors-elementary-divisors}(iii); $\mathcal{V}_T$ is cyclic by Theorem \ref{thm:cyclic-torsion-module-characterization}(i),(iii).
\end{enumerate}
\hlt{Structural Commentary}: Module generators become cyclic vectors; exponent sums detect multiple primary blocks.\\
\hlt{Key Insight}: Cyclicity fails exactly when one prime contributes multiple blocks.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:elementary-divisors-and-jordan-blocks} \hlt{Elementary Divisors and Jordan Blocks}\\
Let $0<\dim_{\mathbb{C}}\mathcal{V}<\infty$ and $T\in\LL(\mathcal{V})$. Choose a Jordan form of $T$ by Theorem \ref{thm:complex-jordan-form}.\\
Multiplication by $z$ on $\mathbb{C}[z]/\langle(z-\lambda)^k\rangle$ has matrix $J_k(\lambda)$ in the reversed shifted-power basis.\\
Hence each Jordan block is cyclic with order $(z-\lambda)^k$. The Jordan direct sum is a primary cyclic decomposition.\\
Its elementary-divisor multiset is multiset of $(z-\lambda)^k$ by Definition \ref{def:elementary-divisors-pid-module}; $\operatorname{ElemDiv}(T)$ records pairs $(\lambda,k)$ by Definition \ref{def:operator-invariant-factors-elementary-divisors}(iii).\\
The prime-power polynomials and Jordan blocks encode corresponding data; they are not the same objects.
\end{perspective}
