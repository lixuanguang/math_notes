\subsection{Division Algebras}

\begin{definition} \label{def:division-algebra} \hlt{Division Algebras}\\
An $\mathbb{F}$-algebra $\mathcal{D}$ is a \hlt{division algebra} iff $1_{\mathcal{D}}\neq0$ and $\mathcal{D}^{\times}=\mathcal{D}\setminus\{0\}$.
\end{definition}

\begin{example} \label{ex:hamilton-quaternions} \hlt{Hamilton's Quaternions}\\
Hamilton's quaternion algebra is associative real algebra $\mathbb{H}\defeq\mathbb{R}1\oplus\mathbb{R}i\oplus\mathbb{R}j\oplus\mathbb{R}k$. Multiplication is $\mathbb{R}$-bilinear, and $1$ is the identity.\\
The basis-unit products satisfy $i^2=j^2=k^2=-1$, $ij=k=-ji$, $jk=i=-kj$, and $ki=j=-ik$.\\
The algebra is noncommutative by $ij=k=-ji$ and $k\neq0$. For $q=r_0+r_1i+r_2j+r_3k$, set $\overline{q}\defeq r_0-r_1i-r_2j-r_3k$.\\
Then $q\overline{q}=\overline{q}q=r_0^2+r_1^2+r_2^2+r_3^2\defeq N(q)$. If $q\neq0$, then $N(q)>0$ and $q^{-1}=N(q)^{-1}\overline{q}$; hence $\mathbb{H}$ is a real division algebra.
\end{example}

\begin{theorem} \label{thm:division-algebra-algebraically-closed-field} \hlt{Division Algebras over Algebraically Closed Fields}\\
Let $\mathcal{D}$ be a division $\mathbb{F}$-algebra with $\dim_{\mathbb{F}}\mathcal{D}<\infty$. Suppose $(\forall\,f\in\mathbb{F}[x])\,[\deg f\geq1\Rightarrow f\text{ splits over }\mathbb{F}]$. Then $\mathcal{D}=\mathbb{F}1_{\mathcal{D}}$.
\begin{proof}
Fix $a\in\mathcal{D}$. The element $a$ is algebraic by Theorem \ref{thm:finite-dimensional-algebras-algebraic-elements}; let $m_a$ be its minimal polynomial.\\
The polynomial $m_a$ is nonconstant because $1_{\mathcal{D}}\neq0$. Suppose $m_a=pq$ with $0<\deg p,\deg q<\deg m_a$.\\
Then $p(a),q(a)\neq0$ by minimality, whereas $q(a)=p(a)^{-1}m_a(a)=0$, a contradiction.\\
Thus $m_a$ is irreducible, and the splitting hypothesis forces $\deg m_a=1$. Monicity fixes $m_a=x-r$ with $r\in\mathbb{F}$; hence $a=r1_{\mathcal{D}}$.\\
Thus $\mathcal{D}\subseteq\mathbb{F}1_{\mathcal{D}}$, while the reverse inclusion holds by the scalar action.\\
\hlt{Structural Commentary}: Algebraicity supplies a minimal polynomial; division makes it irreducible; splitting makes it linear.\\
\hlt{Key Insight}: A nontrivial factorization would create nonzero factors with zero product.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:wedderburn-finite-division-algebra} \hlt{Wedderburn's Theorem}\\
Every division algebra with finite underlying set is a field.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Centralizer Data}: Suppose $\mathcal{D}$ is noncommutative and set $F\defeq Z(\mathcal{D})$.\\
The set $F$ is a finite field: every nonzero central element has a central inverse. Algebra $\mathcal{D}$ is $F$-algebra by Proposition \ref{prop:field-embeds-in-center-of-algebra}(iii).\\
Set $z\defeq|F|\geq2$, $n\defeq\dim_F\mathcal{D}>1$, and $G\defeq\mathcal{D}^{\times}$. Let $B$ represent the noncentral conjugacy classes of $G$.\\
For $\beta\in B$, set $C_\beta\defeq\{d\in\mathcal{D}:d\beta=\beta d\}$ and $k_\beta\defeq\dim_F C_\beta$. Noncentrality implies $F\subsetneq C_\beta\subsetneq\mathcal{D}$.\\
The set $C_\beta$ is a division subalgebra. Choose a right $C_\beta$-basis of $\mathcal{D}$ with $r_\beta$ members.\\
Then $n=r_\beta k_\beta$ by expansion against $F$-basis of $C_\beta$. Thus $1<k_\beta<n$, $k_\beta\mid n$. Identities $Z(G)=F^\times$, $C_G(\beta)=C_\beta^\times$ hold directly.\\
Finite vector-space counting therefore implies $|G|=z^n-1$, $|Z(G)|=z-1$, and $|C_G(\beta)|=z^{k_\beta}-1$.
\item \hlt{Class Equation}: By Theorem \ref{thm:finite-group-class-equation}, $z^n-1=z-1+\sum\limits_{\beta\in B}\frac{z^n-1}{z^{k_\beta}-1}$.
\item \hlt{Cyclotomic Divisor}: Fix $\beta\in B$. Since $k_\beta\mid n$ and $k_\beta<n$, each divisor of $k_\beta$ divides $n$ but $n\nmid k_\beta$.\\
Thus $\Phi_n(z)\mid\frac{z^n-1}{z^{k_\beta}-1}$ by Propositions \ref{prop:roots-of-unity-cyclic-and-cyclotomic-factorization} and \ref{prop:cyclotomic-polynomials-integer-coefficients}. The same propositions imply $\Phi_n(z)\mid z^n-1$.
\item \hlt{Contradiction}: The class equation and (iii) imply $\Phi_n(z)\mid z-1$. Yet $|\Phi_n(z)|>z-1$ by Lemma \ref{lem:cyclotomic-size-estimate-for-wedderburn}, because $n>1$ and $z\geq2$.\\
This contradicts divisibility in $\mathbb{Z}$. Hence $\mathcal{D}$ is commutative and therefore is a finite field.
\end{enumerate}
\hlt{Structural Commentary}: Centralizer dimensions convert the class equation into divisibility; the cyclotomic estimate excludes it.\\
\hlt{Key Insight}: The common cyclotomic divisor cannot divide the smaller integer $z-1$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:frobenius-real-division-algebras} \hlt{Frobenius Classification over $\mathbb{R}$}\\
If $\mathcal{D}$ is a finite-dimensional real division algebra, then $\mathcal{D}\cong\mathbb{R}$, $\mathcal{D}\cong\mathbb{C}$, or $\mathcal{D}\cong\mathbb{H}$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{A Complex Unit}: The relation $\mathbb{R}1_{\mathcal{D}}\cong\mathbb{R}$ holds by Proposition \ref{prop:field-embeds-in-center-of-algebra}(ii). If $\mathcal{D}=\mathbb{R}1_{\mathcal{D}}$, then $\mathcal{D}\cong\mathbb{R}$.\\
Otherwise choose $a\in\mathcal{D}\setminus\mathbb{R}1_{\mathcal{D}}$. The element $a$ is algebraic by Theorem \ref{thm:finite-dimensional-algebras-algebraic-elements}; let $m_a$ be its minimal polynomial.\\
Suppose $m_a=pq$, $0<\deg p,\deg q<\deg m_a$. Then $p(a),q(a)\neq0$ by minimality, whereas $q(a)=p(a)^{-1}m_a(a)=0$, contradiction.\\
Thus $m_a$ is irreducible. Set $d\defeq\deg m_a$, regard $C_{m_a}$ as operator on $\mathbb{C}^d$. Characteristic polynomial is $m_a$ by Theorem \ref{thm:companion-matrix-has-prescribed-minimal-polynomial}(ii).\\
Choose eigenvalue $\lambda\in\mathbb{C}$ by Theorem \ref{thm:complex-operator-has-eigenvalue}; $m_a(\lambda)=0$ by Proposition \ref{prop:operator-characteristic-polynomial-degree-roots}(ii). If $\lambda\in\mathbb{R}$, then $x-\lambda\mid m_a$ by polynomial division.\\
Irreducibility and monicity would make $m_a=x-\lambda$ and $a=\lambda1_{\mathcal{D}}$, contradiction. Thus $\lambda\notin\mathbb{R}$.\\
Set $h(x)\defeq(x-\lambda)(x-\overline{\lambda})\in\mathbb{R}[x]$. Then $h\mid m_a$ by polynomial division at the roots $\lambda,\overline{\lambda}$.\\
Hence $m_a=h$ by irreducibility and monicity. Set $r\defeq\operatorname{Re}\lambda$ and $s\defeq\operatorname{Im}\lambda$.\\
Then $s\neq0$ and $m_a(x)=(x-r)^2+s^2$. Set $i\defeq s^{-1}(a-r1_{\mathcal{D}})$; then $i^2=-1_{\mathcal{D}}$.
\item \hlt{Commuting and Anticommuting Parts}: Set $K\defeq\mathbb{R}1_{\mathcal{D}}\oplus\mathbb{R}i$; then $K\cong\mathbb{C}$. Define the $\mathbb{R}$-linear map $\tau:\mathcal{D}\to\mathcal{D}$ by $\tau(x)\defeq-ixi$.\\
The identity $\tau^2=I_{\mathcal{D}}$ holds by $i^2=-1_{\mathcal{D}}$. Set $\mathcal{D}_+\defeq\ker(\tau-I_{\mathcal{D}})$ and $\mathcal{D}_-\defeq\ker(\tau+I_{\mathcal{D}})$. For $x\in\mathcal{D}$, $x=\frac12(x+\tau(x))+\frac12(x-\tau(x))$.\\
The two summands lie in $\mathcal{D}_+$ and $\mathcal{D}_-$, respectively. Since $\mathcal{D}_+\cap\mathcal{D}_-=\{0\}$, $\mathcal{D}=\mathcal{D}_+\oplus\mathcal{D}_-$.\\
Equivalently, $\mathcal{D}_+=\{x\in\mathcal{D}:xi=ix\}$ and $\mathcal{D}_-=\{x\in\mathcal{D}:xi=-ix\}$. The set $\mathcal{D}_+$ is a division subalgebra, and $K\subseteq Z(\mathcal{D}_+)$.\\
For $y\in\mathcal{D}_+$, the map $L_y$ is complex-linear on the finite-dimensional complex space $\mathcal{D}_+$.\\
Choose $\mu\in K$ and $0\neq x\in\mathcal{D}_+$ with $yx=\mu x$ by Theorem \ref{thm:complex-operator-has-eigenvalue}. Since $x^{-1}\in\mathcal{D}_+$, $y=\mu\in K$. Thus $\mathcal{D}_+=K$.
\item \hlt{The Anticommuting Line}: If $\mathcal{D}_-=\{0\}$, then $\mathcal{D}=K\cong\mathbb{C}$. Otherwise choose $0\neq j_0\in\mathcal{D}_-$.\\
For $x\in\mathcal{D}_-$, $(xj_0^{-1})i=x(-ij_0^{-1})=ixj_0^{-1}$; hence $xj_0^{-1}\in\mathcal{D}_+=K$.\\
Conversely, $Kj_0\subseteq\mathcal{D}_-$ by $j_0i=-ij_0$, so $\mathcal{D}_-=Kj_0$. The relation $j_0^2i=ij_0^2$ places $j_0^2$ in $\mathcal{D}_+=K$.\\
Write $j_0^2=u1_{\mathcal{D}}+vi$, $u,v\in\mathbb{R}$. Relations $j_0i=-ij_0$ and $j_0j_0^2=j_0^2j_0$ imply $0=-2vij_0$, so $v=0$; $j_0^2=u1_{\mathcal{D}}$ with $u\neq0$.\\
Suppose $u>0$. Then $(j_0-\sqrt{u}1_{\mathcal{D}})(j_0+\sqrt{u}1_{\mathcal{D}})=0$. One factor vanishes by the absence of zero divisors, so $j_0$ is scalar.\\
This contradicts $\mathcal{D}_+\cap\mathcal{D}_-=\{0\}$. Hence $u<0$. Set $j\defeq(-u)^{-1/2}j_0$; then $j^2=-1_{\mathcal{D}}$ and $\mathcal{D}_-=Kj$.
\item \hlt{Quaternion Identification}: Now $\mathcal{D}=K\oplus Kj$ has real basis $(1_{\mathcal{D}},i,j,ij)$. Set $k\defeq ij$.\\
The relations $i^2=j^2=-1_{\mathcal{D}}$ and $ji=-ij$ imply $k^2=-1_{\mathcal{D}}$.\\
They also imply $jk=i$ and $ki=j$; the reversed products are negated. Let $\varphi:\mathbb{H}\to\mathcal{D}$ send $(1,i,j,k)$ to $(1_{\mathcal{D}},i,j,k)$ $\mathbb{R}$-linearly.\\
The map $\varphi$ is bijective by the basis and multiplicative by the relations; hence $\mathcal{D}\cong\mathbb{H}$.
\end{enumerate}
\hlt{Structural Commentary}: Quadratic minimal polynomial constructs $K$; conjugation splits $\mathcal{D}$ into commuting, anticommuting parts.\\
\hlt{Key Insight}: The anticommuting eigenspace is at most one-dimensional over $K$.
\end{proof}
\end{theorem}
