\subsection{Associative Algebra Structures}

% Role: foundational algebra operations, regular representations, and
% endomorphism-algebra examples.

The module viewpoint in Definition \ref{def:operator-module-and-matrix-module} uses one operator at a time. An algebra keeps track of a whole vector space of operators or products at once. The guiding example is $\LL(\mathcal{V})$, where addition and scalar multiplication are pointwise and multiplication is composition.

\begin{definition} \label{def:unital-associative-algebra} \hlt{Unital Associative Algebra}\\
Let $\mathbb{F}$ be a field.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item An \hlt{associative $\mathbb{F}$-algebra} is an $\mathbb{F}$-vector space $\mathcal{A}$ and unital ring $(\mathcal{A},+,\cdot,1_{\mathcal{A}})$ such that $r(ab)=(ra)b=a(rb)$ for all $r\in\mathbb{F}$ and $a,b\in\mathcal{A}$.
\item $\mathcal{A}$ is \hlt{finite-dimensional} iff $\dim_{\mathbb{F}}\mathcal{A}<\infty$, and \hlt{commutative} iff $ab=ba$ for all $a,b\in\mathcal{A}$.
\item $a\in\mathcal{A}$ is \hlt{invertible} iff $\exists b\in\mathcal{A}$ such that $ab=ba=1_{\mathcal{A}}$.
\item In this subsection, \hlt{algebra} means unital associative $\mathbb{F}$-algebra unless stated otherwise.
\end{enumerate}
\end{definition}

\begin{remark} \label{rem:algebra-convention-relative-to-ring-definition} \hlt{Convention}\\
Relative to Definition \ref{def:algebra-over-field}, assume multiplication is associative and unital.\\
Thus, \hlt{algebra} means unital associative $\mathbb{F}$-algebra.
\end{remark}

\begin{definition} \label{def:center-and-central-algebra} \hlt{Center and Central Algebras}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{center} of an $\mathbb{F}$-algebra $\mathcal{A}$ is $Z(\mathcal{A})\defeq\{a\in\mathcal{A}: ax=xa\ \forall x\in\mathcal{A}\}$.
\item $\mathcal{A}$ is \hlt{central over $\mathbb{F}$} iff $Z(\mathcal{A})=\mathbb{F}1_{\mathcal{A}}\defeq\{r1_{\mathcal{A}}:r\in\mathbb{F}\}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:field-embeds-in-center-of-algebra} \hlt{The Scalar Field Sits in the Center}\\
If $\mathcal{A}$ is an $\mathbb{F}$-algebra with $1_{\mathcal{A}}\neq0$, then $\iota_{\mathcal{A}}:\mathbb{F}\hookrightarrow Z(\mathcal{A}),\quad r\mapsto r1_{\mathcal{A}}$, is an injective unital ring homomorphism.\\
Conversely, if $R$ is a unital ring and $\mathbb{F}\leq Z(R)$ is a subfield, then $R$ is an $\mathbb{F}$-algebra via $r\cdot x\defeq rx$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Write $\iota=\iota_{\mathcal{A}}$. For $r,s\in\mathbb{F}$, $\iota(r+s)=(r+s)1_{\mathcal{A}}=\iota(r)+\iota(s)$, $\iota(rs)=(rs)1_{\mathcal{A}}=(r1_{\mathcal{A}})(s1_{\mathcal{A}})$, $\iota(1)=1_{\mathcal{A}}$.\\
For $r\in\mathbb{F}$ and $a\in\mathcal{A}$, $\iota(r)a=(r1_{\mathcal{A}})a=r(1_{\mathcal{A}}a)=ra=r(a1_{\mathcal{A}})=a(r1_{\mathcal{A}})=a\iota(r)$, so $\operatorname{im}\iota\subseteq Z(\mathcal{A})$.\\
If $\iota(r)=0$ and $r\neq0$, then $1_{\mathcal{A}}=\iota(r^{-1})\iota(r)=0$, contradiction; hence $\ker\iota=\{0\}$.
\item[$(\Leftarrow)$] Define $r\cdot x=rx$. The field laws in $\mathbb{F}\subseteq R$ give the vector-space axioms, and $1_{\mathbb{F}}\cdot x=x$.\\
For $r\in\mathbb{F}$ and $x,y\in R$, centrality gives $r\cdot(xy)=rxy=(rx)y=x(ry)=x(r\cdot y)$, so the scalar action is compatible with the ring product.
\end{enumerate}
\hlt{Structural Commentary}: An algebra structure is a choice of central scalar copy $\mathbb{F}1_{\mathcal{A}}\subseteq Z(\mathcal{A})$.\\
\hlt{Key Insight}: Scalars are precisely central multipliers of the unit.
\end{proof}
\end{proposition}

\begin{definition} \label{def:subalgebra-generated-subalgebra} \label{def:algebra-ideal-simple-quotient} \label{def:algebra-homomorphism} \label{def:left-regular-representation} \hlt{Subalgebras, Ideals, Quotients, Homomorphisms, and Regular Representation}\\
Let $\mathcal{A}$ be an $\mathbb{F}$-algebra.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{subalgebra} is a subset $\mathcal{B}\subseteq\mathcal{A}$ such that $1_{\mathcal{A}}\in\mathcal{B}$, $\mathcal{B}\leq_{\mathbb{F}}\mathcal{A}$, and $b,b'\in\mathcal{B}\Rightarrow bb'\in\mathcal{B}$.
\item For $X\subseteq\mathcal{A}$, the \hlt{subalgebra generated by $X$} is $\langle X\rangle_{\operatorname{subalg}}\defeq\bigcap\{\mathcal{B}:\mathcal{B}\text{ is a subalgebra of }\mathcal{A},\ X\subseteq\mathcal{B}\}$.\\
Equivalently, $\langle X\rangle_{\operatorname{subalg}}=\operatorname{span}_{\mathbb{F}}\{1_{\mathcal{A}},\ x_1\cdots x_m:m\geq1,\ x_i\in X\}$.
\item A \hlt{two-sided ideal} is a subset $\mathcal{I}\subseteq\mathcal{A}$ such that $\mathcal{I}\neq\emptyset$, $u,v\in\mathcal{I}\Rightarrow u-v\in\mathcal{I}$, and $a\in\mathcal{A},\ u\in\mathcal{I}\Rightarrow au,ua\in\mathcal{I}$.
\item For nonempty $X\subseteq\mathcal{A}$, the \hlt{ideal generated by $X$} is $\langle X\rangle_{\operatorname{ideal}}\defeq\{\sum_{j=1}^{n}a_jx_jb_j:n\geq1,\ x_j\in X,\ a_j,b_j\in\mathcal{A}\}$.
\item $\mathcal{A}$ is \hlt{simple} iff $\mathcal{A}^2\neq0$ and its only two-sided ideals are $0$ and $\mathcal{A}$.
\item If $\mathcal{I}\triangleleft\mathcal{A}$, then $\mathcal{A}/\mathcal{I}$ is an $\mathbb{F}$-algebra by $(a+\mathcal{I})(b+\mathcal{I})\defeq ab+\mathcal{I}$ and $r(a+\mathcal{I})\defeq ra+\mathcal{I}$.
\item For $\mathbb{F}$-algebras $\mathcal{A},\mathcal{B}$, an \hlt{algebra homomorphism} is a map $\sigma:\mathcal{A}\to\mathcal{B}$ such that, for all $a,a'\in\mathcal{A}$ and $r\in\mathbb{F}$,\\
$\sigma(a+a')=\sigma(a)+\sigma(a')$, $\sigma(aa')=\sigma(a)\sigma(a')$, $\sigma(1_{\mathcal{A}})=1_{\mathcal{B}}$, and $\sigma(ra)=r\sigma(a)$.
\item For $a\in\mathcal{A}$, the \hlt{left multiplication operator} is $L_a:\mathcal{A}\to\mathcal{A}$, $x\mapsto ax$.\\
The map $L_{\mathcal{A}}:\mathcal{A}\to\LL(\mathcal{A})$, $a\mapsto L_a$, is the \hlt{left regular representation} of $\mathcal{A}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:algebra-kernel-first-isomorphism} \hlt{Kernel and First Isomorphism Theorem for Algebras}\\
If $\sigma:\mathcal{A}\to\mathcal{B}$ is an $\mathbb{F}$-algebra homomorphism,\\
then $\ker\sigma\triangleleft\mathcal{A}$ and $\overline{\sigma}:\mathcal{A}/\ker\sigma\to\operatorname{im}\sigma$, $a+\ker\sigma\mapsto\sigma(a)$, is an $\mathbb{F}$-algebra isomorphism.
\begin{proof} Set $K\defeq\ker\sigma$. Linearity gives $K\leq_{\mathbb{F}}\mathcal{A}$.\\
If $u\in K$ and $a\in\mathcal{A}$, then $\sigma(au)=\sigma(a)\sigma(u)=0=\sigma(u)\sigma(a)=\sigma(ua)$, hence $au,ua\in K$ and $K\triangleleft\mathcal{A}$.\\
Define $\overline{\sigma}(a+K)\defeq\sigma(a)$. Since $a+K=a'+K\Longleftrightarrow a-a'\in K\Longleftrightarrow\sigma(a)=\sigma(a')$, $\overline{\sigma}$ is well defined and injective; it is surjective onto $\operatorname{im}\sigma$.\\
For $a,a'\in\mathcal{A}$ and $r\in\mathbb{F}$, $\overline{\sigma}((a+K)(a'+K))=\sigma(aa')=\sigma(a)\sigma(a')$, $\overline{\sigma}(r(a+K))=r\sigma(a)$, and $\overline{\sigma}(1_{\mathcal{A}}+K)=1_{\mathcal{B}}$.\\
Thus $\overline{\sigma}$ is an $\mathbb{F}$-algebra isomorphism.\\
\hlt{Structural Commentary}: Quotienting by the kernel removes exactly the relations invisible to $\sigma$.\\
\hlt{Key Insight}: The image is the faithful algebra obtained after collapsing $\ker\sigma$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:left-regular-representation-algebra} \hlt{Left Regular Representation}\\
For every $\mathbb{F}$-algebra $\mathcal{A}$, $L_{\mathcal{A}}:\mathcal{A}\hookrightarrow\LL(\mathcal{A})$, $a\mapsto L_a$, is an $\mathbb{F}$-algebra monomorphism.
\begin{proof}
For $a,b\in\mathcal{A}$, $r\in\mathbb{F}$, and $x\in\mathcal{A}$, $L_{a+b}(x)=(a+b)x=L_a(x)+L_b(x)$ and $L_{ra}(x)=(ra)x=rL_a(x)$, so $L_{\mathcal{A}}$ is $\mathbb{F}$-linear. Also $L_{ab}(x)=(ab)x=a(bx)=(L_aL_b)(x)$ and $L_{1_{\mathcal{A}}}=I_{\mathcal{A}}$, so products and units are preserved.\\
If $L_a=0$, then $a=L_a(1_{\mathcal{A}})=0$, so $L_{\mathcal{A}}$ is injective.\\
\hlt{Structural Commentary}: Left multiplication realises $\mathcal{A}$ inside its own operator algebra.\\
\hlt{Key Insight}: The unit detects elements, since $L_a(1_{\mathcal{A}})=a$.
\end{proof}
\end{theorem}

\begin{method} \label{meth:structure-constants-algebra} \hlt{Building Multiplication from Structure Constants}\\
Let $V=\bigoplus_{i\in I}\mathbb{F}b_i$. Choose $c_{ij}^k\in\mathbb{F}$ with $\{k:c_{ij}^k\neq0\}$ finite for each $(i,j)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Define $b_i b_j\defeq\sum_{k\in I}c_{ij}^k b_k$ and extend bilinearly: $(\sum_i r_i b_i)(\sum_j s_j b_j)\defeq\sum_{i,j,k}r_i s_j c_{ij}^k b_k$.
\item The product is associative iff $\sum_{\ell\in I}c_{ij}^{\ell}c_{\ell k}^{m}=\sum_{\ell\in I}c_{jk}^{\ell}c_{i\ell}^{m}$ for all $i,j,k,m\in I$.
\item The product is commutative iff $c_{ij}^k=c_{ji}^k$ for all $i,j,k\in I$.
\item A basis vector $b_e$ is an identity iff $c_{ej}^m=\delta_j^m=c_{je}^m$ for all $j,m\in I$.
\end{enumerate}
\end{method}

\begin{example} \label{ex:basic-associative-algebras} \hlt{Basic Algebras}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathbb{F}\subseteq\mathbb{E}$ are fields, then $\mathbb{E}$ is an $\mathbb{F}$-algebra by $r\cdot x=rx$ and multiplication in $\mathbb{E}$.
\item $\mathbb{F}[x]$ is a commutative $\mathbb{F}$-algebra with unit $1$ and scalar embedding $r\mapsto r$.
\item $\operatorname{Mat}(n,n,\mathbb{F})$ is an $\mathbb{F}$-algebra with unit $I_n$ and product $AB$.
\item For an $\mathbb{F}$-vector space $\mathcal{V}$, $\LL(\mathcal{V})=\operatorname{End}_{\mathbb{F}}(\mathcal{V})$ is an $\mathbb{F}$-algebra with unit $I_{\mathcal{V}}$ and product $S\circ T$.
\end{enumerate}
\end{example}

\begin{definition} \label{def:group-algebra} \hlt{Group Algebra}\\
For a group $G$ and field $\mathbb{F}$, the \hlt{group algebra} is $\mathbb{F}[G]\defeq\{\sum_{g\in G}r_g g:r_g\in\mathbb{F},\ |\{g:r_g\neq0\}|<\infty\}$.\\
It has coefficientwise vector-space operations and product $(\sum_g r_g g)(\sum_h s_h h)\defeq\sum_{g,h}r_gs_h(gh)$.
\end{definition}

\begin{example} \label{ex:group-algebra} \hlt{Group Algebras}\\
In $\mathbb{F}[G]$, $G$ is an $\mathbb{F}$-basis, $e_G$ is the algebra identity, and associativity follows from $(gh)k=g(hk)$ by bilinearity.\\
If $G$ is finite, then $\dim_{\mathbb{F}}\mathbb{F}[G]=|G|$. Moreover, $\mathbb{F}[G]$ is commutative iff $G$ is abelian.
\end{example}

\begin{example} \label{ex:cyclic-group-algebra-regular-matrices} \hlt{Cyclic Group Algebras and Circulant Shifts}\\
Let $G=\langle\alpha\rangle$, $|G|=n$, and let $\mathcal{B}=(1,\alpha,\ldots,\alpha^{n-1})$ be the ordered $\mathbb{F}$-basis of $\mathbb{F}[G]$.\\
For $0\leq j,k<n$, $L_{\alpha^k}(\alpha^j)=\alpha^{j+k}$ with exponents read modulo $n$, so $L_{\alpha^k}$ is the $k$-fold cyclic shift of $\mathcal{B}$.\\
Let $P\defeq[L_{\alpha}]_{\mathcal{B}}$, one has $[L_{\alpha^k}]_{\mathcal{B}}=P^k$, where
\begin{align*}
P&=
\begin{bmatrix}
0&0&\cdots&0&1\\
1&0&\cdots&0&0\\
0&1&\ddots&0&0\\
\vdots&\vdots&\ddots&\vdots&\vdots\\
0&0&\cdots&1&0
\end{bmatrix}.
\end{align*}
Hence $[L_{\sum_{k=0}^{n-1}c_k\alpha^k}]_{\mathcal{B}}=\sum_{k=0}^{n-1}c_kP^k$, precisely the circulant matrices.
\end{example}

\begin{proposition} \label{prop:endomorphism-algebra-matrix-algebra-isomorphism} \hlt{Endomorphisms and Matrices Are the Same Algebra after Choosing a Basis}\\
Let $\dim\mathcal{V}=n$ and fix an ordered basis $\mathcal{B}$ of
$\mathcal{V}$.\\
Then $\mu_{\mathcal{B}}:\LL(\mathcal{V})\to\operatorname{Mat}(n,n,\mathbb{F})$, $T\mapsto[T]_{\mathcal{B}}$, is an $\mathbb{F}$-algebra isomorphism.
\begin{proof}
By Theorem \ref{thm:linear-maps-and-matrices-same-data}, $\mu_{\mathcal{B}}$ is an $\mathbb{F}$-linear bijection.\\
For $S,T\in\LL(\mathcal{V})$ and $x\in\mathcal{V}$, $[STx]_{\mathcal{B}}=[S]_{\mathcal{B}}[Tx]_{\mathcal{B}}=[S]_{\mathcal{B}}[T]_{\mathcal{B}}[x]_{\mathcal{B}}$, hence $\mu_{\mathcal{B}}(ST)=\mu_{\mathcal{B}}(S)\mu_{\mathcal{B}}(T)$.\\
Also $\mu_{\mathcal{B}}(I_{\mathcal{V}})=I_n$; therefore $\mu_{\mathcal{B}}$ is an $\mathbb{F}$-algebra isomorphism.\\
\hlt{Structural Commentary}: A basis is a coordinate bridge from abstract endomorphisms to concrete matrices.\\
\hlt{Key Insight}: Composition becomes matrix multiplication after taking coordinate columns.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:endomorphism-algebra-center-ideals-simple} \hlt{Center and Ideals of an Endomorphism Algebra}\\
Let $0\neq\mathcal{V}$ be an $\mathbb{F}$-vector space, and set $\mathcal{F}(\mathcal{V})\defeq\{T\in\LL(\mathcal{V}):\dim\operatorname{im}T<\infty\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $Z(\LL(\mathcal{V}))=\mathbb{F}I_{\mathcal{V}}$, so $\LL(\mathcal{V})$ is central.
\item $\mathcal{F}(\mathcal{V})\triangleleft\LL(\mathcal{V})$, and $0\neq\mathcal{J}\triangleleft\LL(\mathcal{V})\Rightarrow\mathcal{F}(\mathcal{V})\subseteq\mathcal{J}$.
\item $\LL(\mathcal{V})$ is simple iff $\dim\mathcal{V}<\infty$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $T\in Z(\LL(\mathcal{V}))$. For $0\neq v\in\mathcal{V}$ and $\varphi\in\mathcal{V}^*$ with $\varphi(v)=1$, commutation with $v\otimes\varphi:x\mapsto\varphi(x)v$ gives $Tv=T(v\otimes\varphi)(v)=(v\otimes\varphi)T(v)=\varphi(Tv)v$, so every line is $T$-stable. Thus $Tv=\lambda_vv$ for $v\neq0$, and applying this to independent $u,v$ and to $u+v$ gives $\lambda_u=\lambda_v$; hence $T=rI_{\mathcal{V}}$.\\
Scalars commute with all endomorphisms.
\item If $R,S\in\LL(\mathcal{V})$ and $T\in\mathcal{F}(\mathcal{V})$, then $\operatorname{im}(RTS)\subseteq R(\operatorname{im}T)$, so $RTS\in\mathcal{F}(\mathcal{V})$ and $\mathcal{F}(\mathcal{V})\triangleleft\LL(\mathcal{V})$. If $0\neq\mathcal{J}\triangleleft\LL(\mathcal{V})$, choose $0\neq S\in\mathcal{J}$, $e$ with $Se\neq0$, and $\eta\in\mathcal{V}^*$ with $\eta(Se)=1$. For any $v\in\mathcal{V}$ and $\varphi\in\mathcal{V}^*$, $v\otimes\varphi=(v\otimes\eta)S(e\otimes\varphi)\in\mathcal{J}$, so $\mathcal{J}$ contains all rank-one maps, hence all finite-rank maps.
\item If $\dim\mathcal{V}<\infty$, then $\mathcal{F}(\mathcal{V})=\LL(\mathcal{V})$, so (ii) makes every nonzero two-sided ideal equal to $\LL(\mathcal{V})$.\\
If $\dim\mathcal{V}=\infty$, then $0\neq\mathcal{F}(\mathcal{V})\neq\LL(\mathcal{V})$, so $\LL(\mathcal{V})$ is not simple.
\end{enumerate}
\hlt{Structural Commentary}: Rank-one operators are the atoms controlling both centrality and ideals in $\LL(\mathcal{V})$.\\
\hlt{Key Insight}: Sandwich one nonzero map to obtain every rank-one map.
\end{proof}
\end{theorem}
