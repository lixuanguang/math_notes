\subsection{Zero Divisors and Spectra of Algebraic Elements}

\begin{definition} \label{def:algebra-zero-divisors} \hlt{Zero Divisors}\\
Let $\mathcal{A}$ be an $\mathbb{F}$-algebra and let $0\neq a\in\mathcal{A}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $a$ is a \hlt{left zero divisor} iff $(\exists\,b\in\mathcal{A}\setminus\{0\})\,ab=0$.
\item $a$ is a \hlt{right zero divisor} iff $(\exists\,c\in\mathcal{A}\setminus\{0\})\,ca=0$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:algebraic-invertible-iff-not-zero-divisor} \hlt{Invertibility and Zero Divisors}\\
Let $\mathcal{A}$ be an $\mathbb{F}$-algebra and let $0\neq a\in\mathcal{A}$ be algebraic. The following conditions are equivalent.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $a\in\mathcal{A}^{\times}$.
\item $a$ is not a left zero divisor.
\item $a$ is not a right zero divisor.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Suppose $a\in\mathcal{A}^{\times}$. If $ab=0$, then $b=a^{-1}ab=0$, so $a$ is not a left zero divisor.
\item[$(ii)$] $\Rightarrow (iii)$ Suppose $a$ is not a left zero divisor. If $a\notin\mathcal{A}^{\times}$, then $m_a(0)=0$ by Theorem \ref{thm:algebraic-element-annihilator-and-invertibility}.\\
Write $m_a=xp$ with $0\neq p\in\mathbb{F}[x]$, $\deg p<\deg m_a$. Then $p(a)\neq0$ by minimality, while $0=m_a(a)=ap(a)=p(a)a$, contradiction.\\
Hence $a\in\mathcal{A}^{\times}$. If $ca=0$, then $c=caa^{-1}=0$, so $a$ is not a right zero divisor.
\item[$(iii)$] $\Rightarrow (i)$ Suppose $a$ is not a right zero divisor. If $a\notin\mathcal{A}^{\times}$, then $m_a(0)=0$ by Theorem \ref{thm:algebraic-element-annihilator-and-invertibility}.\\
Write $m_a=xp$ with $0\neq p\in\mathbb{F}[x]$ and $\deg p<\deg m_a$.\\
Then $p(a)\neq0$ by minimality, while $0=m_a(a)=ap(a)=p(a)a$, a contradiction. Hence $a\in\mathcal{A}^{\times}$.
\end{enumerate}
\hlt{Structural Commentary}: The implication cycle combines unit cancellation with minimal-polynomial factorization.\\
\hlt{Key Insight}: A zero constant term creates one nonzero polynomial witness annihilated by $a$ on both sides.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:algebraic-one-sided-inverse-two-sided} \hlt{One-Sided Inverses Become Two-Sided}\\
Let $\mathcal{A}$ be an $\mathbb{F}$-algebra and let $a\in\mathcal{A}$ be algebraic. If $(\exists\,b\in\mathcal{A})\,[ba=1_{\mathcal{A}}\lor ab=1_{\mathcal{A}}]$, then $a\in\mathcal{A}^{\times}$.
\begin{proof}
If $1_{\mathcal{A}}=0$, then $\mathcal{A}=\{0\}$ and $a=1_{\mathcal{A}}\in\mathcal{A}^{\times}$. Assume $1_{\mathcal{A}}\neq0$ and choose $b\in\mathcal{A}$ with $ba=1_{\mathcal{A}}$ or $ab=1_{\mathcal{A}}$.\\
If $ba=1_{\mathcal{A}}$, then $a\neq0$, and $ac=0$ implies $c=bac=0$. Thus $a$ is not a left zero divisor, so $a\in\mathcal{A}^{\times}$ by Theorem \ref{thm:algebraic-invertible-iff-not-zero-divisor}.\\
If $ab=1_{\mathcal{A}}$, then $a\neq0$, and $ca=0$ implies $c=cab=0$. Thus $a$ is not a right zero divisor, so $a\in\mathcal{A}^{\times}$ by Theorem \ref{thm:algebraic-invertible-iff-not-zero-divisor}.\\
\hlt{Structural Commentary}: The two cases convert a left or right inverse into the matching cancellation law.\\
\hlt{Key Insight}: Algebraicity upgrades either one-sided cancellation law to two-sided invertibility.
\end{proof}
\end{corollary}

\begin{definition} \label{def:algebra-spectrum} \hlt{Spectrum}\\
Let $\mathcal{A}$ be an $\mathbb{F}$-algebra and let $a\in\mathcal{A}$ be algebraic. The \hlt{spectrum of $a$} is $\operatorname{spec}_{\mathcal{A}}(a)\defeq\{r\in\mathbb{F}:m_a(r)=0\}$.\\
An element of $\operatorname{spec}_{\mathcal{A}}(a)$ is an \hlt{algebraic eigenvalue of $a$}.
\end{definition}

\begin{theorem} \label{thm:minimal-polynomial-roots-by-algebra-invertibility} \hlt{Spectrum Detected by Invertibility}\\
Let $\mathcal{A}$ be an $\mathbb{F}$-algebra, let $a\in\mathcal{A}$ be algebraic, and let $r\in\mathbb{F}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $r\in\operatorname{spec}_{\mathcal{A}}(a)\Longleftrightarrow a-r1_{\mathcal{A}}\notin\mathcal{A}^{\times}$.
\item $a\in\mathcal{A}^{\times}\Longleftrightarrow0\notin\operatorname{spec}_{\mathcal{A}}(a)$.
\end{enumerate}
\begin{proof}
Choose $q\in\mathbb{F}[x]$ and $c\in\mathbb{F}$ by Theorem \ref{thm:polynomial-division-algorithm}(i) with $m_a=(x-r)q+c$. Then $c=m_a(r)$ by evaluation at $r$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] ($\Rightarrow$) Suppose $r\in\operatorname{spec}_{\mathcal{A}}(a)$. Then $c=m_a(r)=0$ by Definition \ref{def:algebra-spectrum}; $m_a=(x-r)q$, where $q\neq0$ and $\deg q<\deg m_a$.\\
The relation $0=m_a(a)=(a-r1_{\mathcal{A}})q(a)=q(a)(a-r1_{\mathcal{A}})$ holds, and $q(a)\neq0$ by minimality. Hence $a-r1_{\mathcal{A}}\notin\mathcal{A}^{\times}$.
\item[$(i)$] ($\Leftarrow$) Suppose $a-r1_{\mathcal{A}}\notin\mathcal{A}^{\times}$ and $r\notin\operatorname{spec}_{\mathcal{A}}(a)$. Then $c=m_a(r)\neq0$ by Definition \ref{def:algebra-spectrum}. Set $v\defeq-c^{-1}q(a)$.\\
Since $m_a(a)=0$ and $q(a)$ commutes with $a-r1_{\mathcal{A}}$, $v(a-r1_{\mathcal{A}})=1_{\mathcal{A}}=(a-r1_{\mathcal{A}})v$. This contradicts $a-r1_{\mathcal{A}}\notin\mathcal{A}^{\times}$, so $r\in\operatorname{spec}_{\mathcal{A}}(a)$.
\item[$(ii)$] ($\Rightarrow$) Suppose $a\in\mathcal{A}^{\times}$. Then $0\notin\operatorname{spec}_{\mathcal{A}}(a)$ by (i) with $r=0$.
\item[$(ii)$] ($\Leftarrow$) Suppose $0\notin\operatorname{spec}_{\mathcal{A}}(a)$. Then $a\in\mathcal{A}^{\times}$ by (i) with $r=0$.
\end{enumerate}
\hlt{Structural Commentary}: Division isolates $m_a(r)$; zero supplies an annihilator, while nonzero constructs an inverse.\\
\hlt{Key Insight}: The scalar remainder is exactly the obstruction to invertibility of $a-r1_{\mathcal{A}}$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:algebra-polynomial-spectral-mapping} \hlt{Polynomial Spectral Mapping for Algebraic Elements}\\
Let $\mathcal{A}$ be an $\mathbb{F}$-algebra and suppose $(\forall\,f\in\mathbb{F}[x])\,[\deg f\geq1\Rightarrow f\text{ splits over }\mathbb{F}]$. Let $a\in\mathcal{A}$ be algebraic and let $p\in\mathbb{F}[x]$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $p(a)$ is algebraic over $\mathbb{F}$.
\item $\operatorname{spec}_{\mathcal{A}}(p(a))=p(\operatorname{spec}_{\mathcal{A}}(a))\defeq\{p(r):r\in\operatorname{spec}_{\mathcal{A}}(a)\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] Set $d\defeq\deg m_a$. Then $p(a)\in\mathbb{F}[a]$ and $\dim_{\mathbb{F}}\mathbb{F}[a]\leq d$ by Proposition \ref{prop:generated-algebra-by-one-algebraic-element}(i)--(ii).\\
Hence $p(a)$ is algebraic by Theorem \ref{thm:finite-dimensional-algebras-algebraic-elements} for the $\mathbb{F}$-algebra $\mathbb{F}[a]$.
\item[$(ii)$]
\begin{enumerate}[label=(\alph*)]
\setlength{\itemsep}{0pt}
\item \hlt{Degenerate Cases}: If $1_{\mathcal{A}}=0$, then $\mathcal{A}=\{0\}$ and $\ker\operatorname{ev}_a=\mathbb{F}[x]=\ker\operatorname{ev}_{p(a)}$. Thus $m_a=m_{p(a)}=1$ by Definition \ref{def:algebraic-element-minimal-polynomial-algebra}(i).\\
Thus both spectra are empty, and the equality holds. Assume $1_{\mathcal{A}}\neq0$.\\
If $p=c\in\mathbb{F}$, then $m_a$ is nonconstant as $m_a(a)=0$, $1_{\mathcal{A}}\neq0$. It has a root in $\mathbb{F}$ by splitting hypothesis, so $\operatorname{spec}_{\mathcal{A}}(a)\neq\emptyset$.\\
If $\lambda\neq c$, then $(c-\lambda)^{-1}1_{\mathcal{A}}$ is the inverse of $(c-\lambda)1_{\mathcal{A}}$; if $\lambda=c$, then $0$ is noninvertible because $1_{\mathcal{A}}\neq0$.\\
Thus $\operatorname{spec}_{\mathcal{A}}(p(a))=\{c\}=p(\operatorname{spec}_{\mathcal{A}}(a))$ by Theorem \ref{thm:minimal-polynomial-roots-by-algebra-invertibility}(i).
\item \hlt{Forward Inclusion}: Suppose $p$ is nonconstant, $\lambda\in\operatorname{spec}_{\mathcal{A}}(p(a))$. Then $p(a)-\lambda1_{\mathcal{A}}$ noninvertible by Theorem \ref{thm:minimal-polynomial-roots-by-algebra-invertibility}(i).\\
Choose $N\in\N$, $c\in\mathbb{F}^{\times}$, and $r_1,\ldots,r_N\in\mathbb{F}$ by the splitting hypothesis with $p(x)-\lambda=c\prod\limits_{j=1}^{N}(x-r_j)$.\\
If $(\forall\,j\in\{1,\ldots,N\})\,a-r_j1_{\mathcal{A}}\in\mathcal{A}^{\times}$, then $p(a)-\lambda1_{\mathcal{A}}$ is invertible. Hence $(\exists\,j\in\{1,\ldots,N\})\,a-r_j1_{\mathcal{A}}\notin\mathcal{A}^{\times}$.\\
For such $j$, $r_j\in\operatorname{spec}_{\mathcal{A}}(a)$ by Theorem \ref{thm:minimal-polynomial-roots-by-algebra-invertibility}(i), and $p(r_j)=\lambda$. Therefore $\operatorname{spec}_{\mathcal{A}}(p(a))\subseteq p(\operatorname{spec}_{\mathcal{A}}(a))$.
\item \hlt{Reverse Inclusion}: Suppose $p$ is nonconstant and let $r\in\operatorname{spec}_{\mathcal{A}}(a)$.\\
Choose $q\in\mathbb{F}[x]$ and $s\in\mathbb{F}$ by Theorem \ref{thm:polynomial-division-algorithm}(i) with $p(x)-p(r)=(x-r)q(x)+s$.\\
$s=0$ by evaluation at $r$, so $p(x)-p(r)=(x-r)q(x)$. The element $a-r1_{\mathcal{A}}$ is algebraic by (i) with the polynomial $x-r$.\\
If $p(a)-p(r)1_{\mathcal{A}}=(a-r1_{\mathcal{A}})q(a)$ were invertible with inverse $w$, then $q(a)w$ would be a right inverse of $a-r1_{\mathcal{A}}$.\\
Then $a-r1_{\mathcal{A}}$ would be invertible by Corollary \ref{cor:algebraic-one-sided-inverse-two-sided}. Yet $a-r1_{\mathcal{A}}\notin\mathcal{A}^{\times}$ by Theorem \ref{thm:minimal-polynomial-roots-by-algebra-invertibility}(i), a contradiction.\\
Thus $p(a)-p(r)1_{\mathcal{A}}$ is noninvertible; $p(r)\in\operatorname{spec}_{\mathcal{A}}(p(a))$ by Theorem \ref{thm:minimal-polynomial-roots-by-algebra-invertibility}(i). Hence $p(\operatorname{spec}_{\mathcal{A}}(a))\subseteq\operatorname{spec}_{\mathcal{A}}(p(a))$.
\end{enumerate}
\end{enumerate}
\hlt{Structural Commentary}: Finite-dimensional reduction handles algebraicity; splitting and one-sided rigidity handle opposite inclusions.\\
\hlt{Key Insight}: Linear factors recover spectral preimages, while $x-r$ carries spectral membership forward.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:algebra-spectra-of-ab-and-ba} \hlt{Spectra of $ab$ and $ba$}\\
Let $\mathcal{A}$ be an $\mathbb{F}$-algebra with $\dim_{\mathbb{F}}\mathcal{A}<\infty$, $a,b\in\mathcal{A}$. Then $\operatorname{spec}_{\mathcal{A}}(ab)=\operatorname{spec}_{\mathcal{A}}(ba)$.
\begin{proof}
Every element of $\mathcal{A}$ is algebraic by Theorem \ref{thm:finite-dimensional-algebras-algebraic-elements}. Let $r\in\mathbb{F}^{\times}$, $ba-r1_{\mathcal{A}}\in\mathcal{A}^{\times}$. Set $u\defeq a(ba-r1_{\mathcal{A}})^{-1}b-1_{\mathcal{A}}$.\\
Then $(ab-r1_{\mathcal{A}})u=a(ba-r1_{\mathcal{A}})(ba-r1_{\mathcal{A}})^{-1}b-ab+r1_{\mathcal{A}}=r1_{\mathcal{A}}$. Thus $r^{-1}u$ is a right inverse of $ab-r1_{\mathcal{A}}$.\\
The element $ab-r1_{\mathcal{A}}$ is invertible by Corollary \ref{cor:algebraic-one-sided-inverse-two-sided}. Hence $r\notin\operatorname{spec}_{\mathcal{A}}(ba)\Rightarrow r\notin\operatorname{spec}_{\mathcal{A}}(ab)$ by Theorem \ref{thm:minimal-polynomial-roots-by-algebra-invertibility}(i).\\
$\operatorname{spec}_{\mathcal{A}}(ab)\setminus\{0\}=\operatorname{spec}_{\mathcal{A}}(ba)\setminus\{0\}$ by symmetry. Suppose $ba\in\mathcal{A}^{\times}$ with inverse $y$. Then $b(ay)=1_{\mathcal{A}}$ and $(yb)a=1_{\mathcal{A}}$.\\
Thus $a,b\in\mathcal{A}^{\times}$ by Corollary \ref{cor:algebraic-one-sided-inverse-two-sided}, so $ab\in\mathcal{A}^{\times}$. The converse holds by symmetry, so $ab\in\mathcal{A}^{\times}\Longleftrightarrow ba\in\mathcal{A}^{\times}$.\\
Thus $0\in\operatorname{spec}_{\mathcal{A}}(ab)\Longleftrightarrow0\in\operatorname{spec}_{\mathcal{A}}(ba)$ by Theorem \ref{thm:minimal-polynomial-roots-by-algebra-invertibility}(ii). Therefore $\operatorname{spec}_{\mathcal{A}}(ab)=\operatorname{spec}_{\mathcal{A}}(ba)$.\\
\hlt{Structural Commentary}: The resolvent calculation handles nonzero scalars; one-sided inverse rigidity handles the zero shift.\\
\hlt{Key Insight}: The resolvent correction $a(ba-r1_{\mathcal{A}})^{-1}b-1_{\mathcal{A}}$ transfers invertibility across reversed products.
\end{proof}
\end{proposition}
