\subsection{Iterated Tensor Products and Tensor Spaces}

% This subsection extends the tensor-product universal property to multilinear
% maps and develops tensor spaces and contractions.

The tensor-product universal property extends from bilinear maps to maps with more variables, and then to tensor spaces and contractions.

\begin{definition} \label{def:multilinear-map}\label{def:iterated-tensor-product-universal-property} \hlt{Multilinear Maps and Iterated Tensor Products}\\
Let $\mathcal{V}_1,\ldots,\mathcal{V}_n,\mathcal{W}$ be $\mathbb{F}$-vector spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A map $f:\mathcal{V}_1\times\cdots\times\mathcal{V}_n\to\mathcal{W}$ is \hlt{multilinear}, or \hlt{$n$-linear}, if for each $k$, the map $\mathbf{v}_k\mapsto f(\mathbf{v}_1,\ldots,\mathbf{v}_k,\ldots,\mathbf{v}_n)$ is linear with the other entries fixed.
\item \hlt{$\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathcal{W})$}$\defeq\{f:\mathcal{V}_1\times\cdots\times\mathcal{V}_n\to\mathcal{W}:f\text{ multilinear}\}$.
\item If $\mathcal{W}=\mathbb{F}$, elements of $\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathbb{F})$ are \hlt{multilinear forms}.
\item A pair $(\mathcal{T},t)$ is an \hlt{iterated tensor product} of $\mathcal{V}_1,\ldots,\mathcal{V}_n$ if $t\in\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathcal{T})$ and, for every $\mathbb{F}$-vector space $\mathcal{Y}$ and $f\in\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathcal{Y})$, $\exists!\widetilde{f}\in\LL(\mathcal{T},\mathcal{Y})$ such that $f=\widetilde{f}\circ t$.
\item Write $\mathcal{T}=\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$ and $\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n\defeq t(\mathbf{v}_1,\ldots,\mathbf{v}_n)$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:basic-multilinear-maps} \hlt{Basic Multilinear Maps}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A$ is an associative $\mathbb{F}$-algebra, then $m_n:A^n\to A$, $m_n(a_1,\ldots,a_n)\defeq a_1\cdots a_n$, is $n$-linear.
\item If $\dim\mathcal{V}=n$, then $\det:\mathcal{V}^n\to\mathbb{F}$, applied to coordinate columns in any basis, is alternating $n$-linear form.
\end{enumerate}
\end{example}

\begin{theorem} \label{thm:iterated-tensor-products-basic-laws} \hlt{Basic Laws for Iterated Tensor Products}\\
Let $\mathcal{V}_1,\ldots,\mathcal{V}_n,\mathcal{V},\mathcal{W}$ be $\mathbb{F}$-vector spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathcal{W})\cong\LL(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n,\mathcal{W})$ naturally.
\item For $1\leq r<n$, the canonical $\alpha_r$ is $\alpha_r:(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_r)\otimes(\mathcal{V}_{r+1}\otimes\cdots\otimes\mathcal{V}_n)\cong\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$,\\
where $\alpha_r((\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_r)\otimes(\mathbf{v}_{r+1}\otimes\cdots\otimes\mathbf{v}_n))=\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n$.
\item For $\pi\in S_n$, the canonical $\tau_\pi$ is $\tau_\pi:\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n\cong\mathcal{V}_{\pi(1)}\otimes\cdots\otimes\mathcal{V}_{\pi(n)}$,\\
where $\tau_\pi(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n)=\mathbf{v}_{\pi(1)}\otimes\cdots\otimes\mathbf{v}_{\pi(n)}$.
\item $\lambda:\mathbb{F}\otimes\mathcal{V}\cong\mathcal{V}$, $\lambda(r\otimes\mathbf{v})=r\mathbf{v}$, and $\rho:\mathcal{V}\otimes\mathbb{F}\cong\mathcal{V}$, $\rho(\mathbf{v}\otimes r)=r\mathbf{v}$.
\item If $\dim\mathcal{V}_i<\infty$ for all $i$, then $\dim(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n)=\prod_{i=1}^{n}\dim(\mathcal{V}_i)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $t:\mathcal{V}_1\times\cdots\times\mathcal{V}_n\to\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$ be canonical. Define $\Phi(f)=\widetilde{f}$, where $\widetilde{f}$ is the unique linear map satisfying $\widetilde{f}(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n)=f(\mathbf{v}_1,\ldots,\mathbf{v}_n)$; conversely define $\Psi(T)=T\circ t$. Then $\Psi\circ\Phi$ and $\Phi\circ\Psi$ are identity maps, so $\Phi$ is a linear bijection.
\item Set $A_r=(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_r)\otimes(\mathcal{V}_{r+1}\otimes\cdots\otimes\mathcal{V}_n)$ and $B=\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$. The rules defining $\alpha_r:A_r\to B$ and $\beta_r:B\to A_r$, $\beta_r(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n)=(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_r)\otimes(\mathbf{v}_{r+1}\otimes\cdots\otimes\mathbf{v}_n)$, are induced by multilinearity; $\alpha_r\circ\beta_r$ and $\beta_r\circ\alpha_r$ fix the spanning decomposable tensors, hence are identities.
\item Set $C_\pi=\mathcal{V}_{\pi(1)}\otimes\cdots\otimes\mathcal{V}_{\pi(n)}$. The map $p_\pi:\mathcal{V}_1\times\cdots\times\mathcal{V}_n\to C_\pi$, $p_\pi(\mathbf{v}_1,\ldots,\mathbf{v}_n)=\mathbf{v}_{\pi(1)}\otimes\cdots\otimes\mathbf{v}_{\pi(n)}$, is multilinear, so the universal property induces the unique $\tau_\pi$ with $\tau_\pi\circ t=p_\pi$.\\
The map $\sigma_\pi:C_\pi\to\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$, $\sigma_\pi(\mathbf{v}_{\pi(1)}\otimes\cdots\otimes\mathbf{v}_{\pi(n)})=\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n$, is induced similarly; $\sigma_\pi\tau_\pi$ and $\tau_\pi\sigma_\pi$ fix decomposable spanning tensors, hence are identities.
\item The maps $m_L:\mathbb{F}\times\mathcal{V}\to\mathcal{V}$, $m_L(r,\mathbf{v})=r\mathbf{v}$, and $m_R:\mathcal{V}\times\mathbb{F}\to\mathcal{V}$, $m_R(\mathbf{v},r)=r\mathbf{v}$, are bilinear, hence induce $\lambda$ and $\rho$. Since $\lambda(1\otimes\mathbf{v})=\mathbf{v}$ and $1\otimes r\mathbf{v}=r\otimes\mathbf{v}$, $\lambda^{-1}(\mathbf{v})=1\otimes\mathbf{v}$.\\
Since $\rho(\mathbf{v}\otimes1)=\mathbf{v}$ and $r\mathbf{v}\otimes1=\mathbf{v}\otimes r$, $\rho^{-1}(\mathbf{v})=\mathbf{v}\otimes1$.
\item Choose finite bases $B_i$ of $\mathcal{V}_i$ and set $\mathcal{B}=\{\mathbf{b}_1\otimes\cdots\otimes\mathbf{b}_n:\mathbf{b}_i\in B_i\}$. By repeated application of Theorem \ref{thm:tensor-bases-dimension-and-rank}, $\mathcal{B}$ is a basis; hence $\dim(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n)=\#\mathcal{B}=\prod_{i=1}^{n}\#B_i=\prod_{i=1}^{n}\dim\mathcal{V}_i$.
\end{enumerate}
\hlt{Structural Commentary}: Tensor parenthesising and ordering are canonical bookkeeping.\\
\hlt{Key Insight}: Universal properties extend pure-tensor rules uniquely.
\end{proof}
\end{theorem}

\begin{definition} \label{def:tensor-spaces-type-pq} \hlt{Tensor Spaces of Type $(p,q)$}\\
Let $\dim\mathcal{V}<\infty$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $(p,q)\in\mathbb{Z}_{\geq0}^{2}\setminus\{(0,0)\}$, with empty tensor powers equal to $\mathbb{F}$, $\mathcal{T}^{p}_{q}(\mathcal{V})\defeq\mathcal{V}^{\otimes p}\otimes(\mathcal{V}')^{\otimes q}$.
\item $\mathcal{T}^{p}_{q}(\mathcal{V})$ is the \hlt{tensor space of type $(p,q)$}; $p$ is the \hlt{contravariant type}, $q$ the \hlt{covariant type}.
\item $\mathcal{T}^{0}_{0}(\mathcal{V})\defeq\mathbb{F}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:tensor-spaces-as-multilinear-functionals} \hlt{Tensor Spaces as Multilinear Functionals}\\
Let $\dim\mathcal{V}=n<\infty$ and $p,q\in\mathbb{Z}_{\geq0}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\dim\mathcal{T}^{p}_{q}(\mathcal{V})=n^{p+q}$.
\item Let $\iota:\mathcal{V}\to\mathcal{V}''$, $\iota(\mathbf{v})(\eta)=\eta(\mathbf{v})$. Via $\iota$, natural isomorphism is $\Theta_{p,q}:\mathcal{T}^{p}_{q}(\mathcal{V})\xrightarrow{\sim}\operatorname{Mult}_{\mathbb{F}}((\mathcal{V}')^{\times p}\times\mathcal{V}^{\times q};\mathbb{F})$;\\
where $\Theta_{p,q}(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_p\otimes\varphi_1\otimes\cdots\otimes\varphi_q)(\eta_1,\ldots,\eta_p,\mathbf{x}_1,\ldots,\mathbf{x}_q)=\prod_{i=1}^{p}\eta_i(\mathbf{v}_i)\prod_{j=1}^{q}\varphi_j(\mathbf{x}_j)$,\\
with empty products equal to $1$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:iterated-tensor-products-basic-laws}, $\dim\mathcal{V}'=n$, $\dim\mathcal{T}^{p}_{q}(\mathcal{V})=\dim(\mathcal{V}^{\otimes p})\dim((\mathcal{V}')^{\otimes q})=n^pn^q=n^{p+q}$, with $n^0=1$.
\item Set $X=(\mathcal{V}')^{\otimes p}\otimes\mathcal{V}^{\otimes q}$. Since $\iota:\mathcal{V}\cong\mathcal{V}''$ and all factors are finite-dimensional,\\
$\mathcal{T}^{p}_{q}(\mathcal{V})=\mathcal{V}^{\otimes p}\otimes(\mathcal{V}')^{\otimes q} \cong((\mathcal{V}')')^{\otimes p}\otimes\mathcal{V}'^{\otimes q} \cong X' \cong\operatorname{Mult}_{\mathbb{F}}((\mathcal{V}')^{\times p}\times\mathcal{V}^{\times q};\mathbb{F})$.\\
On decomposables, the composite is $\Theta_{p,q}$; spanning gives uniqueness.
\end{enumerate}
\hlt{Structural Commentary}: Tensor elements are multilinear functionals after finite-dimensional duality.\\
\hlt{Key Insight}: Choosing which factors are active turns tensor factors into input slots.
\end{proof}
\end{proposition}

\begin{definition} \label{def:contraction-map} \hlt{Contraction}\\
Let $p,q\geq1$. The \hlt{contraction} is the unique linear map $C:\mathcal{T}^{p}_{q}(\mathcal{V})\to\mathcal{T}^{p-1}_{q-1}(\mathcal{V})$, where\\
$C(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_p\otimes\varphi_1\otimes\cdots\otimes\varphi_q)=\varphi_1(\mathbf{v}_1)\mathbf{v}_2\otimes\cdots\otimes\mathbf{v}_p\otimes\varphi_2\otimes\cdots\otimes\varphi_q$.\\
For $(i,j)$, $C_{ij}$ is $C$ conjugated by the factor permutation placing $\mathbf{v}_i,\varphi_j$ first.
\end{definition}

\begin{example} \label{ex:rank-one-operator-contraction-trace} \hlt{Rank-One Operators and Trace}\\
Under $\mathcal{V}\otimes\mathcal{V}'\cong\LL(\mathcal{V})$, $\mathbf{v}\otimes\varphi\mapsto T_{\mathbf{v},\varphi}$, where $T_{\mathbf{v},\varphi}(\mathbf{x})=\varphi(\mathbf{x})\mathbf{v}$. Hence $C(\mathbf{v}\otimes\varphi)=\varphi(\mathbf{v})=\operatorname{tr}(T_{\mathbf{v},\varphi})$.
\end{example}
