\subsection{Retracts, One-Sided Inverses, and Split Maps}

Throughout this subsection, $R$ denotes a nonzero commutative ring with identity $1_R$.\\
All modules are $R$-modules.

\begin{theorem} \label{thm:direct-summands-extend-epimorphisms} \hlt{Direct Summands Extend Epimorphisms}\\
Let $M,M_1$ be $R$-modules, let $N\leq M$, and let $\sigma\in\operatorname{Hom}_R(N,M_1)$ be an epimorphism.\\
Set $\mathcal{C}_N(M)\defeq\{H\leq M:M=N\oplus H\}$. Set $\mathcal{E}_{\sigma}\defeq\{\tau\in\operatorname{Hom}_R(M,M_1):\tau|_N=\sigma,\ \operatorname{im}\tau=M_1\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(\forall\,H\in\mathcal{C}_N(M))(\exists!\,\overline{\sigma}_H\in\operatorname{Hom}_R(M,M_1))(\forall\,(n,h)\in N\times H)\,\overline{\sigma}_H(n+h)=\sigma(n)$.
\item $(\forall\,H\in\mathcal{C}_N(M))\,\overline{\sigma}_H\in\mathcal{E}_{\sigma}$.
\item $(\forall\,H\in\mathcal{C}_N(M))\,\ker\overline{\sigma}_H=\ker\sigma\oplus H$.
\item If $\sigma$ is an isomorphism, then $H\mapsto\overline{\sigma}_H$ is a bijection $\mathcal{C}_N(M)\to\mathcal{E}_{\sigma}$ with inverse $\tau\mapsto\ker\tau$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $H\in\mathcal{C}_N(M)$. If $n+h=n'+h'$ for $(n,n',h,h')\in N^2\times H^2$, then $(n-n')+(h-h')=0_M$.\\
Hence $n=n'$ and $h=h'$ by Proposition \ref{prop:module-direct-sum-characterizations}(ii), so $\overline{\sigma}_H(n+h)\defeq\sigma(n)$ is well-defined.\\
Fix $(r,s,n,n',h,h')\in R^2\times N^2\times H^2$. Then $\overline{\sigma}_H(r(n+h)+s(n'+h'))=\sigma(rn+sn')=r\sigma(n)+s\sigma(n')$.\\
Thus $\overline{\sigma}_H\in\operatorname{Hom}_R(M,M_1)$ by Definition \ref{def:module-homomorphism}(i). The extension is unique by the defining formula and Proposition \ref{prop:module-direct-sum-characterizations}(iii).
\item Fix $H\in\mathcal{C}_N(M)$. Then $\overline{\sigma}_H|_N=\sigma$ by the defining formula with $h=0_M$.\\
The identity $\operatorname{im}\overline{\sigma}_H=\operatorname{im}\sigma=M_1$ holds by the same formula, so $\overline{\sigma}_H\in\mathcal{E}_{\sigma}$.
\item Fix $H\in\mathcal{C}_N(M)$. Then $\ker\overline{\sigma}_H=\{n+h:(n,h)\in N\times H,\ \sigma(n)=0_{M_1}\}=\ker\sigma+H$.\\
Since $\ker\sigma\leq N$ and $N\cap H=\{0_M\}$, $(\ker\sigma)\cap H=\{0_M\}$. Therefore $\ker\overline{\sigma}_H=\ker\sigma\oplus H$ by Definition \ref{def:module-direct-sums}(ii).
\item Suppose $\sigma$ is an isomorphism. For $H\in\mathcal{C}_N(M)$, $\ker\overline{\sigma}_H=H$ by (iii) and $\ker\sigma=\{0_M\}$.\\
Conversely, fix $\tau\in\mathcal{E}_{\sigma}$. If $x\in N\cap\ker\tau$, then $0_{M_1}=\tau(x)=\sigma(x)$ and $x=0_M$ by injectivity of $\sigma$.\\
For $m\in M$, set $n\defeq\sigma^{-1}(\tau(m))\in N$. Then $\tau(m-n)=0_{M_1}$, so $m=n+(m-n)\in N+\ker\tau$.\\
Thus $\ker\tau\in\mathcal{C}_N(M)$ by Definition \ref{def:module-direct-sums}(ii). Starting with $H$ recovers $H$ by the first sentence.\\
Starting with $\tau$, write $m=n+k$ with $(n,k)\in N\times\ker\tau$. Then $\tau(m)=\sigma(n)=\overline{\sigma}_{\ker\tau}(m)$, so $\overline{\sigma}_{\ker\tau}=\tau$.
\end{enumerate}
\hlt{Structural Commentary}: Coordinates extend maps; kernels classify extensions when $\sigma$ is invertible.\\
\hlt{Key Insight}: An isomorphic restriction makes the extension kernel recover its complement.
\end{proof}
\end{theorem}

\begin{definition} \label{def:module-retract} \hlt{Retract}\\
Let $N\leq M$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{retraction} onto $N$ is a map $\rho\in\operatorname{Hom}_R(M,N)$ such that $\rho|_N=\operatorname{id}_N$.
\item The submodule $N$ is a \hlt{retract} of $M$ if $(\exists\,\rho\in\operatorname{Hom}_R(M,N))\,\rho|_N=\operatorname{id}_N$.
\end{enumerate}
\end{definition}

\begin{corollary} \label{cor:retract-iff-direct-summand-module} \hlt{Retracts Are Direct Summands}\\
For $N\leq M$, $N$ is a retract of $M\Longleftrightarrow(\exists\,H\leq M)\,M=N\oplus H$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Choose $\rho\in\operatorname{Hom}_R(M,N)$ with $\rho|_N=\operatorname{id}_N$ by Definition \ref{def:module-retract}(ii).\\
For $m\in M$, $\rho(m-\rho(m))=\rho(m)-\rho(\rho(m))=0_N$ by $\rho(m)\in N$, $\rho|_N=\operatorname{id}_N$; $m=\rho(m)+(m-\rho(m))\in N+\ker\rho$, so $M=N+\ker\rho$.\\
If $x\in N\cap\ker\rho$, then $x=\rho(x)=0_N$; hence $N\cap\ker\rho=\{0_M\}$. Therefore $M=N\oplus\ker\rho$ by Definition \ref{def:module-direct-sums}(ii).
\item[$(\Leftarrow)$] Fix $H\leq M$ with $M=N\oplus H$. Define $\rho:M\to N$ by $\rho(n+h)\defeq n$. The map is well-defined by Proposition \ref{prop:module-direct-sum-characterizations}(ii).\\
For $(r,s,n,n',h,h')\in R^2\times N^2\times H^2$, $\rho(r(n+h)+s(n'+h'))=rn+sn'=r\rho(n+h)+s\rho(n'+h')$.\\
Thus $\rho\in\operatorname{Hom}_R(M,N)$ by Definition \ref{def:module-homomorphism}(i), and $\rho|_N=\operatorname{id}_N$. Hence $N$ is a retract of $M$ by Definition \ref{def:module-retract}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Retraction kernels are complements; projections reverse the construction.\\
\hlt{Key Insight}: A retraction separates every element into image and kernel components.
\end{proof}
\end{corollary}

\begin{definition} \label{def:one-sided-inverses-module-maps} \hlt{One-Sided Inverses}\\
Let $\tau\in\operatorname{Hom}_R(A,B)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{left inverse} of $\tau$ is $\lambda\in\operatorname{Hom}_R(B,A)$ such that $\lambda\tau=\operatorname{id}_A$.
\item A \hlt{right inverse} of $\tau$ is $\rho\in\operatorname{Hom}_R(B,A)$ such that $\tau\rho=\operatorname{id}_B$.
\item A \hlt{two-sided inverse} of $\tau$ is $\upsilon\in\operatorname{Hom}_R(B,A)$ such that $\upsilon\tau=\operatorname{id}_A$ and $\tau\upsilon=\operatorname{id}_B$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:module-one-sided-inverses-and-complements} \hlt{One-Sided Inverses and Complements}\\
Let $\sigma\in\operatorname{Hom}_R(M,M_1)$. For an $R$-module $P$ and $S\leq P$, set $\mathcal{C}_S(P)\defeq\{H\leq P:P=S\oplus H\}$.\\
Set $\mathcal{L}_\sigma\defeq\{\lambda\in\operatorname{Hom}_R(M_1,M):\lambda\sigma=\operatorname{id}_M\}$. Set $\mathcal{R}_\sigma\defeq\{\rho\in\operatorname{Hom}_R(M_1,M):\sigma\rho=\operatorname{id}_{M_1}\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\sigma$ is injective, then $\lambda\mapsto\ker\lambda$ is a bijection $\mathcal{L}_\sigma\to\mathcal{C}_{\operatorname{im}\sigma}(M_1)$.
\item If $\sigma$ is surjective, then $\rho\mapsto\operatorname{im}\rho$ is a bijection $\mathcal{R}_\sigma\to\mathcal{C}_{\ker\sigma}(M)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Assume $\sigma$ is injective. Fix $H\in\mathcal{C}_{\operatorname{im}\sigma}(M_1)$. For $y\in M_1$, write $y=u+h$ with $(u,h)\in\operatorname{im}\sigma\times H$.\\
This pair is unique by Proposition \ref{prop:module-direct-sum-characterizations}(iii). Choose unique $m\in M$ with $u=\sigma(m)$ by $u\in\operatorname{im}\sigma$, injectivity of $\sigma$.\\
Define $\lambda_H(y)\defeq m$. For $y=\sigma(m)+h$, $z=\sigma(m')+h'$, and $(r,s)\in R^2$, $\lambda_H(ry+sz)=rm+sm'=r\lambda_H(y)+s\lambda_H(z)$.\\
Thus $\lambda_H\in\operatorname{Hom}_R(M_1,M)$ by Definition \ref{def:module-homomorphism}(i). Identities $\lambda_H\sigma=\operatorname{id}_M$ and $\ker\lambda_H=H$ hold by defining formula, so $\lambda_H\in\mathcal{L}_\sigma$.\\
Conversely, fix $\lambda\in\mathcal{L}_\sigma$. If $y\in\operatorname{im}\sigma\cap\ker\lambda$, write $y=\sigma(m)$. Then $m=\lambda\sigma(m)=0_M$, so $y=0_{M_1}$.\\
For $y\in M_1$, $y=\sigma\lambda(y)+(y-\sigma\lambda(y))$ and $y-\sigma\lambda(y)\in\ker\lambda$. Hence $\ker\lambda\in\mathcal{C}_{\operatorname{im}\sigma}(M_1)$ by Definition \ref{def:module-direct-sums}(ii).\\
Start with $H$, recover $H$ by $\ker\lambda_H=H$. Start with $\lambda$, recover $\lambda$ by decomposition $y=\sigma\lambda(y)+(y-\sigma\lambda(y))$, injectivity of $\sigma$.
\item Assume $\sigma$ is surjective. Fix $H\in\mathcal{C}_{\ker\sigma}(M)$ and set $\sigma_H\defeq\sigma|_H:H\to M_1$.\\
The map $\sigma_H$ is an $R$-homomorphism with $\ker\sigma_H=H\cap\ker\sigma=\{0_M\}$.\\
Fix $y\in M_1$. Choose $m\in M$ with $\sigma(m)=y$, and write $m=k+h$ with $(k,h)\in\ker\sigma\times H$.\\
Then $y=\sigma_H(h)$, so $\sigma_H$ is an isomorphism by Proposition \ref{prop:hom-module-kernel-image}(iv) and Definition \ref{def:module-homomorphism}(iv).\\
For $y\in M_1$, define $\rho_H(y)\defeq\sigma_H^{-1}(y)\in H\subseteq M$. For $(r,s,y,z)\in R^2\times M_1^2$, $\sigma_H(r\rho_H(y)+s\rho_H(z))=ry+sz=\sigma_H(\rho_H(ry+sz))$.\\
Hence $\rho_H(ry+sz)=r\rho_H(y)+s\rho_H(z)$ by injectivity of $\sigma_H$. Thus $\rho_H\in\operatorname{Hom}_R(M_1,M)$ by Definition \ref{def:module-homomorphism}(i).\\
The identities $\sigma\rho_H=\operatorname{id}_{M_1}$ and $\operatorname{im}\rho_H=H$ hold by construction, so $\rho_H\in\mathcal{R}_\sigma$.\\
Conversely, fix $\rho\in\mathcal{R}_\sigma$. The map $\rho$ is injective by $\rho(y)=0_M\Rightarrow y=\sigma\rho(y)=0_{M_1}$.\\
If $x\in\operatorname{im}\rho\cap\ker\sigma$, write $x=\rho(y)$. Then $y=\sigma\rho(y)=\sigma(x)=0_{M_1}$, so $x=0_M$.\\
For $m\in M$, $m=(m-\rho\sigma(m))+\rho\sigma(m)$ with $m-\rho\sigma(m)\in\ker\sigma$ and $\rho\sigma(m)\in\operatorname{im}\rho$. Thus $\operatorname{im}\rho\in\mathcal{C}_{\ker\sigma}(M)$ by Definition \ref{def:module-direct-sums}(ii).\\
Starting with $H$ recovers $H$ by $\operatorname{im}\rho_H=H$. Starting with $\rho$ recovers $\rho$ by $(\sigma|_{\operatorname{im}\rho})^{-1}(y)=\rho(y)$.
\end{enumerate}
\hlt{Structural Commentary}: Coordinates construct inverses; decompositions recover their complements.\\
\hlt{Key Insight}: Left inverses split images; right inverses split kernels.
\end{proof}
\end{theorem}
