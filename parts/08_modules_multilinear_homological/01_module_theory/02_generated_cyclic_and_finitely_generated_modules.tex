\subsection{Generated, Cyclic, and Finitely Generated Modules}

Throughout this subsection, $R$ is a nonzero commutative ring with identity $1_R$. Fix an $R$-module $M$.

\begin{definition} \label{def:generated-submodule} \hlt{Generated Submodule}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $S\subseteq M$, set $\langle\!\langle S\rangle\!\rangle_R\defeq\left\{\sum\limits_{j=1}^{n}r_js_j:n\in\N_0,\ (r_1,\ldots,r_n)\in R^n,\ (s_1,\ldots,s_n)\in S^n\right\}$.\\
Call this the \hlt{submodule generated by $S$}; the term $n=0$ denotes the empty sum $0_M$. Hence $\langle\!\langle\emptyset\rangle\!\rangle_R=\{0_M\}$.
\item A subset $S\subseteq M$ \hlt{spans} or \hlt{generates} $M$ if $\langle\!\langle S\rangle\!\rangle_R=M$.
\item For $n\in\N$ and $(s_1,\ldots,s_n)\in M^n$, write $\langle\!\langle s_1,\ldots,s_n\rangle\!\rangle_R\defeq\langle\!\langle\{s_1,\ldots,s_n\}\rangle\!\rangle_R$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:generated-submodule-smallest} \hlt{Generated Submodule Is Smallest}\\
Let $S\subseteq M$, $G\defeq\langle\!\langle S\rangle\!\rangle_R$, and $\mathcal{S}_S\defeq\{N\leq M:S\subseteq N\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $G\leq M$.
\item $S\subseteq G$.
\item $G=\bigcap\limits_{N\in\mathcal{S}_S}N$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $0_M\in G$ by the empty-sum convention.\\
Fix $x,y\in G$ and $a,b\in R$. Choose representations $x=\sum\limits_{j=1}^{m}r_js_j$ and $y=\sum\limits_{k=1}^{n}t_ku_k$ by Definition \ref{def:generated-submodule}(i).\\
Then $ax+by=\sum\limits_{j=1}^{m}(ar_j)s_j+\sum\limits_{k=1}^{n}(bt_k)u_k\in G$ by distributivity and Definition \ref{def:generated-submodule}(i). Hence $G\leq M$ by Theorem \ref{thm:submodule-test}.
\item $S\subseteq G$ by $(\forall\,s\in S)\,s=1_Rs\in G$.
\item $G\in\mathcal{S}_S$ by (i) and (ii). Fix $N\in\mathcal{S}_S$. Then $G\subseteq N$ by $S\subseteq N$ and scalar and additive closure in $N$.\\
Hence $G\subseteq\bigcap\limits_{N\in\mathcal{S}_S}N$ by the arbitrariness of $N$. Conversely, $\bigcap\limits_{N\in\mathcal{S}_S}N\subseteq G$ by $G\in\mathcal{S}_S$.
\end{enumerate}
\hlt{Structural Commentary}: Closure is by the submodule test; minimality is by unit coefficients and containment.\\
\hlt{Key Insight}: Every containing submodule absorbs all finite $R$-linear combinations.
\end{proof}
\end{proposition}

\begin{definition} \label{def:cyclic-and-finitely-generated-modules} \hlt{Cyclic and Finitely Generated Modules}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $v\in M$, the \hlt{cyclic submodule generated by $v$} is $\langle\!\langle v\rangle\!\rangle_R=Rv=\{rv:r\in R\}$.
\item $M$ is \hlt{finitely generated} if $(\exists\,n\in\N)(\exists\,(v_1,\ldots,v_n)\in M^n)\,M=\langle\!\langle v_1,\ldots,v_n\rangle\!\rangle_R$.
\item $M$ is \hlt{cyclic} if $(\exists\,v\in M)\,M=Rv$.
\end{enumerate}
\end{definition}

\begin{counterexample} \label{cex:finitely-generated-module-with-non-fg-submodule} \hlt{A Cyclic Module with a Non-Finitely Generated Submodule}\\
A submodule of a finitely generated module need not be finitely generated. Let $\mathbb{F}$ be a field and $R\defeq\mathbb{F}[x_1,x_2,\ldots]$.\\
This is the polynomial ring in countably many commuting indeterminates.\\
Every $f\in R$ involves only finitely many indeterminates, and ${}_R R=\langle\!\langle1_R\rangle\!\rangle_R$ is cyclic.\\
Set $I\defeq\langle\{x_j:j\in\N\}\rangle$ and $\mathcal{Z}\defeq\{f\in R:f(0,0,\ldots)=0\}$. $I\trianglelefteq R$ by Proposition \ref{prop:generated-ideal-smallest-containing-set}.\\
Hence $I\leq{}_R R$ by Proposition \ref{prop:submodules-of-regular-module-are-ideals}. $I\subseteq\mathcal{Z}$ by the zero constant terms of the generators.\\
Every $f\in\mathcal{Z}$ is a finite sum of nonconstant monomials.\\
Each such monomial lies in $I$ because it is divisible by some $x_j$; hence $\mathcal{Z}\subseteq I$ and $I=\mathcal{Z}$. Assume $I$ is finitely generated.\\
Choose $n\in\N$ and $(p_1,\ldots,p_n)\in I^n$ satisfying $I=\langle\!\langle p_1,\ldots,p_n\rangle\!\rangle_R$ by Definition \ref{def:cyclic-and-finitely-generated-modules}(ii).\\
Only finitely many indeterminates occur among $p_1,\ldots,p_n$. Choose $k\in\N$ such that $x_k$ occurs in none of them.\\
Define $\varphi_k:R\to\mathbb{F}[x_k]$ by $\varphi_k|_{\mathbb{F}}=\operatorname{id}_{\mathbb{F}}$, $\varphi_k(x_k)=x_k$, and $(\forall\,j\in\N\setminus\{k\})\,\varphi_k(x_j)=0$.\\
The map $\varphi_k$ preserves sums, products, and identity by substitution. Hence $\varphi_k$ is a ring homomorphism by Definition \ref{def:ring-homomorphism-and-subring}(i).\\
Each $p_i$ has zero constant term by $p_i\in I=\mathcal{Z}$, and no $p_i$ contains $x_k$. Hence $(\forall\,i\in\{1,\ldots,n\})\,\varphi_k(p_i)=0$.\\
$(\forall\,(r_1,\ldots,r_n)\in R^n)\,\varphi_k\left(\sum\limits_{i=1}^{n}r_ip_i\right)=\sum\limits_{i=1}^{n}\varphi_k(r_i)\varphi_k(p_i)=0$. $\varphi_k(I)=\{0\}$ by $I=\langle\!\langle p_1,\ldots,p_n\rangle\!\rangle_R$ and the preceding identity.\\
However, $x_k\in I$ and $\varphi_k(x_k)=x_k\neq0$, a contradiction. Therefore $I$ is a non-finitely generated submodule of the cyclic module ${}_R R$.
\end{counterexample}
