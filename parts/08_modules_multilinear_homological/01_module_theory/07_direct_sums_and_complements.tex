\subsection{Direct Sums and Complements}

Throughout this subsection, $R$ denotes a nonzero commutative ring with identity $1_R$.\\
The symbol $M$ denotes an $R$-module with additive identity $0_M$.

\begin{definition} \label{def:module-direct-sums} \hlt{Direct Sums and Direct Summands}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $m=(m_i)_{i\in I}\in\prod\limits_{i\in I}M_i$, set $\operatorname{supp}(m)\defeq\{i\in I:m_i\neq0_{M_i}\}$.\\
The \hlt{external direct sum} is $\bigoplus\limits_{i\in I}M_i\defeq\{m\in\prod\limits_{i\in I}M_i:\abs{\operatorname{supp}(m)}<\infty\}$, with componentwise operations.
\item For a family $(S_i)_{i\in I}$ of submodules of $M$, define $\sum\limits_{i\in I}S_i\defeq\left\langle\!\left\langle\bigcup\limits_{i\in I}S_i\right\rangle\!\right\rangle_R$.\\
Equivalently, $\sum\limits_{i\in I}S_i=\{\sum\limits_{i\in F}s_i:F\subseteq I,\ \abs{F}<\infty,\ (s_i)_{i\in F}\in\prod\limits_{i\in F}S_i\}$.\\
Here the empty sum is $0_M$. The equivalence follows by absorbing coefficients into each submodule and collecting like indices.\\
Sum satisfies $\sum\limits_{i\in I}S_i\leq M$ by Proposition \ref{prop:generated-submodule-smallest}(i). It is \hlt{direct} if $(\forall\,k\in I)\,S_k\cap\sum\limits_{i\in I\setminus\{k\}}S_i=\{0_M\}$.\\
If it is direct and equals $M$, write $M=\bigoplus\limits_{i\in I}S_i$.
\item A submodule $S\leq M$ is a \hlt{direct summand} if $(\exists\,T\leq M)\,M=S\oplus T$. Such a $T$ is a \hlt{complement} of $S$ in $M$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:module-direct-sum-characterizations} \hlt{Direct Sum Characterizations}\\
For a family $(S_i)_{i\in I}$ of submodules of $M$, the following conditions are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\sum\limits_{i\in I}S_i$ is direct.
\item $(\forall\,F\subseteq I)\left[\abs{F}<\infty\Rightarrow(\forall\,(x_i)_{i\in F}\in\prod\limits_{i\in F}S_i)\left[\sum\limits_{i\in F}x_i=0_M\Rightarrow(\forall\,i\in F)\,x_i=0_M\right]\right]$.
\item $(\forall\,x\in\sum\limits_{i\in I}S_i)(\exists!\,(x_i)_{i\in I}\in\bigoplus\limits_{i\in I}S_i)\,x=\sum\limits_{i\in I}x_i$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Assume (i). Fix $F\subseteq I$ with $\abs{F}<\infty$ and $(x_i)_{i\in F}\in\prod\limits_{i\in F}S_i$ with $\sum\limits_{i\in F}x_i=0_M$.\\
For $k\in F$, $x_k=-\sum\limits_{i\in F\setminus\{k\}}x_i\in S_k\cap\sum\limits_{i\in I\setminus\{k\}}S_i$. This intersection is $\{0_M\}$ by Definition \ref{def:module-direct-sums}(ii); hence $(\forall\,k\in F)\,x_k=0_M$.
\item[$(ii)$] $\Rightarrow (iii)$ Fix $x\in\sum\limits_{i\in I}S_i$. Choose $F\subseteq I$ with $\abs{F}<\infty$ and $(x_i)_{i\in F}\in\prod\limits_{i\in F}S_i$ with $x=\sum\limits_{i\in F}x_i$ by Definition \ref{def:module-direct-sums}(ii).\\
Extend this family by $0_M$. The resulting $(x_i)_{i\in I}$ lies in $\bigoplus\limits_{i\in I}S_i$ by Definition \ref{def:module-direct-sums}(i), so a decomposition exists.\\
Suppose $(x_i)_{i\in I},(y_i)_{i\in I}\in\bigoplus\limits_{i\in I}S_i$ both sum to $x$, and set $H\defeq\{i\in I:x_i\neq0_M\}\cup\{i\in I:y_i\neq0_M\}$.\\
Then $\abs{H}<\infty$ by Definition \ref{def:module-direct-sums}(i), and $\sum\limits_{i\in H}(x_i-y_i)=0_M$.\\
Thus $(\forall\,i\in H)\,x_i=y_i$ by (ii), while $(\forall\,i\in I\setminus H)\,x_i=y_i=0_M$ by the definition of $H$. Hence $(x_i)_{i\in I}=(y_i)_{i\in I}$.
\item[$(iii)$] $\Rightarrow (i)$ Fix $k\in I$ and $x\in S_k\cap\sum\limits_{i\in I\setminus\{k\}}S_i$. Suppose $x\neq0_M$.\\
Choose $F\subseteq I\setminus\{k\}$ with $\abs{F}<\infty$ and $(y_i)_{i\in F}\in\prod\limits_{i\in F}S_i$ with $x=\sum\limits_{i\in F}y_i$ by Definition \ref{def:module-direct-sums}(ii).\\
Extend $(y_i)_{i\in F}$ by $0_M$, and define $(z_i)_{i\in I}$ by $z_k=x$ and $(\forall\,i\in I\setminus\{k\})\,z_i=0_M$.\\
These are distinct elements of $\bigoplus\limits_{i\in I}S_i$ whose coordinates both sum to $x$, contrary to (iii).\\
Hence $x=0_M$, so $(\forall\,k\in I)\,S_k\cap\sum\limits_{i\in I\setminus\{k\}}S_i=\{0_M\}$.
\end{enumerate}
\hlt{Structural Commentary}: Zero-sum rigidity turns the intersection criterion into coordinate uniqueness.\\
\hlt{Key Insight}: Decompositions are unique exactly when cross-summand cancellation is impossible.
\end{proof}
\end{proposition}

\begin{counterexample} \label{cex:z-submodules-mostly-not-complemented} \hlt{Not Every Submodule Is Complemented}\\
The $\Z$-module $\Z$ refutes the assertion that every submodule of a module is complemented.\\
Each $S\leq{}_{\Z}\Z$ is an ideal by Proposition \ref{prop:submodules-of-regular-module-are-ideals}. The zero submodule is $0\Z$.\\
Each nonzero $S$ has a unique positive generator by Theorem \ref{thm:integers-and-polynomial-ring-are-pids}(i),(ii). Hence $(\forall\,S\leq{}_{\Z}\Z)(\exists!\,n\in\N_0)\,S=n\Z$.\\
For $(m,n)\in\N^2$, $0\neq mn\in m\Z\cap n\Z$, so no two nonzero submodules form a direct sum.\\
Thus, for $k\in\N_0$, $k\Z$ is complemented only if $k\in\{0,1\}$; conversely, $\Z=\{0\}\oplus\Z=\Z\oplus\{0\}$.\\
Hence the complemented submodules of $\Z$ are exactly $\{0\}$ and $\Z$.
\end{counterexample}

\begin{theorem} \label{thm:complements-and-quotients-modules} \hlt{Complements and Quotients}\\
Let $S\leq M$ and let $\pi:M\twoheadrightarrow M/S$ be the quotient map.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $T\leq M$ and $M=S\oplus T$, then $\pi|_T:T\xrightarrow{\sim}M/S$, $t\mapsto t+S$.
\item If $T_1,T_2\leq M$ satisfy $M=S\oplus T_1=S\oplus T_2$, then $T_1\cong T_2$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Fix $T\leq M$ and assume $M=S\oplus T$. Then $\pi|_T\in\operatorname{Hom}_R(T,M/S)$ by Proposition \ref{prop:quotient-module-well-defined}(iii).\\
The identity $\ker(\pi|_T)=T\cap S=\{0_M\}$ holds by Proposition \ref{prop:quotient-module-well-defined}(i) and Definition \ref{def:module-direct-sums}(ii).\\
Hence $\pi|_T$ is a monomorphism by Proposition \ref{prop:hom-module-kernel-image}(iv). Fix $m+S\in M/S$.\\
Choose $(s,t)\in S\times T$ with $m=s+t$ by $M=S+T$. Then $m+S=t+S=\pi|_T(t)$, so $\pi|_T$ is an epimorphism by Definition \ref{def:module-homomorphism}(iv).\\
Therefore $\pi|_T$ is an isomorphism by Definition \ref{def:module-homomorphism}(iv).
\item Assume $M=S\oplus T_1=S\oplus T_2$. Then $T_1\cong M/S\cong T_2$ by (i), and hence $T_1\cong T_2$.
\end{enumerate}
\hlt{Structural Commentary}: Kernel and image tests transfer each complement to the common quotient model.\\
\hlt{Key Insight}: Its kernel on a complement is precisely the intersection with $S$.
\end{proof}
\end{theorem}
