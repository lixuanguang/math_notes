\subsection{Modules and Submodules}

Throughout this subsection, $R$ denotes a nonzero commutative ring with identity $1$.\\
Modules are vector spaces with the scalar field replaced by a ring; the price is that nonzero scalars may fail to be invertible and may even kill nonzero vectors.

\begin{perspective} \label{per:operator-as-module-motivation} \hlt{Polynomial Scalars from One Operator}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$ and let $T\in\LL(\mathcal{V})$. Every polynomial $p(z)\in\mathbb{F}[z]$ acts on $\mathcal{V}$ by $p\cdot\mathbf{v}\defeq p(T)\mathbf{v}$.\\
Thus one operator turns $\mathcal{V}$ into an object with scalar ring $\mathbb{F}[z]$.
\end{perspective}

\begin{definition} \label{def:module-over-ring} \hlt{$R$-Module}\\
An \hlt{$R$-module} is an abelian group $(M,+)$ equipped with a scalar action
$R\times M\to M$, $(r,u)\mapsto ru$, such that for all $r,s\in R$ and $u,v\in M$,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $r(u+v)=ru+rv$.
\item $(r+s)u=ru+su$.
\item $(rs)u=r(su)$.
\item $1u=u$.
\end{enumerate}
The ring $R$ is the \hlt{base ring} of $M$.
\end{definition}

\begin{example} \label{ex:basic-module-models} \hlt{Basic Module Models}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $R^n$ is an $R$-module under componentwise addition and scalar multiplication.
\item $\operatorname{Mat}(m,n,R)$ is an $R$-module under entrywise operations.
\item $R$ is an $R$-module over itself by ring multiplication.
\item Every vector space over a field $\mathbb{F}$ is an $\mathbb{F}$-module.
\end{enumerate}
\hlt{Structural Commentary}: Modules are abelian groups with compatible scalar actions.\\
\hlt{Key Insight}: The new behavior comes from the scalar ring, not from changing addition.
\end{example}

\begin{perspective} \label{per:base-ring-controls-modules} \hlt{The Base Ring Controls the Geometry}\\
The structure of an $R$-module is governed by the arithmetic of $R$. When $R$ is a field, every nonzero scalar is invertible, giving the usual theory of bases, dimension, and complements. For a general ring, nonunits and zero divisors obstruct division and cancellation, so these notions need not retain their vector-space properties.
\end{perspective}

\begin{definition} \label{def:submodule} \hlt{Submodule}\\
A \hlt{submodule} of an $R$-module $M$ is a nonempty subset $N\subseteq M$ that is an $R$-module under the operations inherited from $M$, written $N\leq M$.
\end{definition}

\begin{theorem} \label{thm:submodule-test} \hlt{Submodule Test}\\
Let $M$ be an $R$-module and let $\emptyset\neq N\subseteq M$. Then $N\leq M$ $\Longleftrightarrow$ $\forall r,s\in R$, $\forall u,v\in N$, $ru+sv\in N$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $N\leq M$, then $ru,sv\in N$, hence $ru+sv\in N$.
\item[$(\Leftarrow)$] Assume the displayed condition. Choose $n\in N$. Then $0=0n+0n\in N$.\\
For $x,y\in N$ and $r\in R$, $x+y=1x+1y\in N$, $-x=(-1)x+0x\in N$, $rx=rx+0x\in N$.\\
Thus $N$ is an additive subgroup of $M$ closed under $R$-scalar multiplication, so the remaining module axioms are inherited from $M$ and $N\leq M$.
\end{enumerate}
\hlt{Structural Commentary}: One closure test packages addition, additive inverses, and scalar multiplication.\\
\hlt{Key Insight}: Submodules are exactly the subsets closed under all two-term $R$-linear combinations.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:submodule-intersection-sum} \hlt{Intersections and Sums of Submodules}\\
Let $M$ be an $R$-module.\\
If $A,B\leq M$, then $A\cap B\leq M$; $A+B\defeq\{a+b:a\in A,\ b\in B\}\leq M$, and $A+B=\bigcap\limits_{\substack{N\leq M\\ A\cup B\subseteq N}}N$.
\begin{proof}
Since $0\in A\cap B$ and $rx+sy\in A\cap B$ for all $x,y\in A\cap B$ and $r,s\in R$, the submodule test gives $A\cap B\leq M$. If $x=a+b$ and $y=a'+b'$ with $a,a'\in A$ and $b,b'\in B$, then $rx+sy=(ra+sa')+(rb+sb')\in A+B$.\\
Thus $A+B\leq M$. Moreover, $A\cup B\subseteq A+B$, and if $N\leq M$ contains $A\cup B$, then $a+b\in N$ for all $a\in A$, $b\in B$; hence $A+B\subseteq N$. This proves the displayed intersection formula.\\
\hlt{Structural Commentary}: $A\cap B$ is the meet; $A+B$ is the join.\\
\hlt{Key Insight}: Sums are forced by closure under addition.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:submodules-of-regular-module-are-ideals} \hlt{Ideals Are Submodules of the Regular Module}\\
View $R$ as the regular $R$-module ${}_R R$. For every $N\subseteq R$,
\[
N\leq {}_R R \quad\Longleftrightarrow\quad N\trianglelefteq R.
\]
\begin{proof}
For $N\subseteq R$,
\[
N\leq {}_R R
\quad\Longleftrightarrow\quad
N\leq (R,+)\ \text{and}\ RN\subseteq N
\quad\Longleftrightarrow\quad
N\trianglelefteq R,
\]
by Definition \ref{def:ideal-and-generated-ideal}.\\
\hlt{Structural Commentary}: Ideals are the module-theoretic subobjects of the ring acting on itself.\\
\hlt{Key Insight}: Ring multiplication becomes scalar multiplication.
\end{proof}
\end{proposition}
