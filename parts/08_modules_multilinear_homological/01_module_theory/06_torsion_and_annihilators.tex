\subsection{Torsion and Annihilators}

Throughout this subsection, $R$ is a nonzero commutative ring with identity $1_R$ and additive identity $0_R$.\\
Fix an $R$-module $M$ with additive identity $0_M$.

\begin{definition} \label{def:module-torsion} \hlt{Torsion}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item An element $v\in M\setminus\{0_M\}$ is a \hlt{torsion element} if $(\exists\,r\in R\setminus\{0_R\})\,rv=0_M$.
\item The module $M$ is \hlt{torsion-free} if $(\forall\,(r,v)\in(R\setminus\{0_R\})\times M)\,[rv=0_M\Rightarrow v=0_M]$.
\item The module $M$ is a \hlt{torsion module} if $(\forall\,v\in M\setminus\{0_M\})(\exists\,r\in R\setminus\{0_R\})\,rv=0_M$.
\item The \hlt{torsion set} of $M$ is $M_{\operatorname{tor}}\defeq\{v\in M:(\exists\,r\in R\setminus\{0_R\})\,rv=0_M\}$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:zn-is-torsion-module} \hlt{$\mathbb{Z}/n\mathbb{Z}$ as a Torsion Module}\\
Let $n\in\N$ with $n>1$. Quotient $\mathbb{Z}/n\mathbb{Z}$ is $\mathbb{Z}$-module with $k\overline{a}\defeq\overline{ka}$. $(\forall\,\overline{a}\in\mathbb{Z}/n\mathbb{Z})\,n\overline{a}=\overline{0}$; $\mathbb{Z}/n\mathbb{Z}$ is torsion module by Definition \ref{def:module-torsion}(iii).\\
If $\overline{a}\neq\overline{0}$, then $n\overline{a}=\overline{0}$ with $n\neq0_{\mathbb{Z}}$. Thus $\{\overline{a}\}$ is linearly dependent by Definition \ref{def:module-linear-independence}(i)--(ii).
\end{example}

\begin{proposition} \label{prop:torsion-submodule-over-domain} \hlt{Torsion Submodule over an Integral Domain}\\
Assume $R$ is an integral domain.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $M_{\operatorname{tor}}\leq M$.
\item $(M/M_{\operatorname{tor}})_{\operatorname{tor}}=\{0_{M/M_{\operatorname{tor}}}\}=\{M_{\operatorname{tor}}\}$; equivalently, $M/M_{\operatorname{tor}}$ is torsion-free.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $0_M\in M_{\operatorname{tor}}$ by $1_R0_M=0_M$ and Definition \ref{def:module-torsion}(iv). Fix $(r,s,x,y)\in R^2\times M_{\operatorname{tor}}^2$.\\
Choose $(a,b)\in(R\setminus\{0_R\})^2$ with $ax=by=0_M$ by Definition \ref{def:module-torsion}(iv).\\
By Definition \ref{def:zero-divisor-integral-domain}(ii), $ab\neq0_R$. $ab(rx+sy)=rb(ax)+sa(by)=0_M$.\\
Thus $rx+sy\in M_{\operatorname{tor}}$ by Definition \ref{def:module-torsion}(iv). Hence $M_{\operatorname{tor}}\leq M$ by Theorem \ref{thm:submodule-test}.
\item Set $T\defeq M_{\operatorname{tor}}$ and fix $x\in(M/T)_{\operatorname{tor}}$. Choose $m\in M$ with $x=m+T$ by Definition \ref{def:quotient-module}.\\
Choose $r\in R\setminus\{0_R\}$ with $rx=T$ by Definition \ref{def:module-torsion}(iv). Then $rm\in T$ by Proposition \ref{prop:quotient-module-well-defined}(i).\\
Choose $s\in R\setminus\{0_R\}$ with $srm=0_M$ by Definition \ref{def:module-torsion}(iv). By Definition \ref{def:zero-divisor-integral-domain}(ii), $sr\neq0_R$.\\
Hence $m\in T$ by Definition \ref{def:module-torsion}(iv), so $x=T$ by Proposition \ref{prop:quotient-module-well-defined}(i).\\
Therefore $(M/T)_{\operatorname{tor}}\subseteq\{0_{M/T}\}$. The reverse inclusion holds by Definition \ref{def:module-torsion}(iv).\\
Thus $(M/T)_{\operatorname{tor}}=\{0_{M/T}\}=\{T\}$. Equivalently, $M/T$ is torsion-free by Definition \ref{def:module-torsion}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Products of annihilators prove closure and eliminate quotient torsion.\\
\hlt{Key Insight}: Domain multiplication preserves nonzero annihilators.
\end{proof}
\end{proposition}

\begin{definition} \label{def:module-annihilator} \hlt{Annihilators}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $v\in M$, the \hlt{annihilator} of $v$ is $\operatorname{Ann}_R(v)\defeq\{r\in R:rv=0_M\}$.
\item For $N\leq M$, the \hlt{annihilator} of $N$ is $\operatorname{Ann}_R(N)\defeq\{r\in R:(\forall\,n\in N)\,rn=0_M\}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:annihilator-basic-properties} \hlt{Annihilator Properties}\\
Let $v\in M$ and $N\leq M$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{Ann}_R(v)\trianglelefteq R$ and $\operatorname{Ann}_R(N)\trianglelefteq R$.
\item $v$ is torsion $\Longleftrightarrow v\neq0_M$ and $\operatorname{Ann}_R(v)\neq\{0_R\}$.
\item If $A\leq B\leq M$, then $\operatorname{Ann}_R(B)\subseteq\operatorname{Ann}_R(A)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $0_R\in\operatorname{Ann}_R(v)$ by $0_Rv=0_M$ and Definition \ref{def:module-annihilator}(i). Fix $(r,s,a,b)\in R^2\times\operatorname{Ann}_R(v)^2$.\\
Then $(ra+sb)v=r(av)+s(bv)=0_M$ by Definition \ref{def:module-annihilator}(i). Thus $\operatorname{Ann}_R(v)\leq{}_R R$ by Theorem \ref{thm:submodule-test}.\\
Hence $\operatorname{Ann}_R(v)\trianglelefteq R$ by Proposition \ref{prop:submodules-of-regular-module-are-ideals}. $(\forall\,n\in N)\,0_Rn=0_M$, so $0_R\in\operatorname{Ann}_R(N)$ by Definition \ref{def:module-annihilator}(ii).\\
Fix $(r,s,a,b)\in R^2\times\operatorname{Ann}_R(N)^2$. $(\forall\,n\in N)\,[(ra+sb)n=r(an)+s(bn)=0_M]$ by Definition \ref{def:module-annihilator}(ii).\\
Thus $\operatorname{Ann}_R(N)\leq{}_R R$ by Theorem \ref{thm:submodule-test}. Hence $\operatorname{Ann}_R(N)\trianglelefteq R$ by Proposition \ref{prop:submodules-of-regular-module-are-ideals}.
\item[$(ii)$] ($\Rightarrow$) Suppose $v$ is torsion. Then $v\neq0_M$, $(\exists\,r\in R\setminus\{0_R\})\,rv=0_M$ by Definition \ref{def:module-torsion}(i); $\operatorname{Ann}_R(v)\neq\{0_R\}$ by Definition \ref{def:module-annihilator}(i).
\item[$(ii)$] ($\Leftarrow$) Suppose $v\neq0_M$ and $\operatorname{Ann}_R(v)\neq\{0_R\}$. Choose $r\in\operatorname{Ann}_R(v)\setminus\{0_R\}$.\\
Then $rv=0_M$ by Definition \ref{def:module-annihilator}(i). Hence $v$ is torsion by Definition \ref{def:module-torsion}(i).
\item[$(iii)$] Fix $A,B\leq M$ with $A\leq B$, and let $r\in\operatorname{Ann}_R(B)$. Then $(\forall\,a\in A)\,ra=0_M$ by $A\subseteq B$ and Definition \ref{def:module-annihilator}(ii).\\
Hence $r\in\operatorname{Ann}_R(A)$ by Definition \ref{def:module-annihilator}(ii).
\end{enumerate}
\hlt{Structural Commentary}: Closure establishes ideals; annihilators detect torsion and reverse containment.\\
\hlt{Key Insight}: Annihilation conditions strengthen as the tested subset grows.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:finitely-generated-torsion-module-annihilator} \hlt{A Nonzero Annihilator from Torsion Generators over a Domain}\\
Assume $R$ is an integral domain. Let $n\in\N$ and $M=\langle\!\langle u_1,\ldots,u_n\rangle\!\rangle_R$. If $(u_1,\ldots,u_n)\in M_{\operatorname{tor}}^n$, then $\operatorname{Ann}_R(M)\neq\{0_R\}$.
\begin{proof}
Choose $(a_1,\ldots,a_n)\in(R\setminus\{0_R\})^n$ with $(\forall\,i\in\{1,\ldots,n\})\,a_iu_i=0_M$ by Definition \ref{def:module-torsion}(iv).\\
Set $a\defeq\prod\limits_{i=1}^{n}a_i$. By repeated application of Definition \ref{def:zero-divisor-integral-domain}(ii), $a\neq0_R$.\\
Fix $i\in\{1,\ldots,n\}$ and choose $b_i\in R$ with $a=b_ia_i$ by commutativity of $R$. Then $au_i=b_i(a_iu_i)=0_M$.\\
Thus $(\forall\,i\in\{1,\ldots,n\})\,au_i=0_M$. Fix $m\in M$. Choose $(r_1,\ldots,r_n)\in R^n$ with $m=\sum\limits_{i=1}^{n}r_iu_i$ by Definition \ref{def:generated-submodule}(i),(iii).\\
By commutativity of $R$, $am=\sum\limits_{i=1}^{n}r_i(au_i)=0_M$. Thus $a\in\operatorname{Ann}_R(M)\setminus\{0_R\}$ by Definition \ref{def:module-annihilator}(ii).\\
\hlt{Structural Commentary}: Generator annihilators multiply to a scalar killing all finite combinations.\\
\hlt{Key Insight}: Finite products stay nonzero because $R$ is a domain.
\end{proof}
\end{proposition}
