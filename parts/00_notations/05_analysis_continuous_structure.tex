\subsection*{Analysis and Continuous Structure Notation}

\begingroup
\small

\notationgroup{Sets and Functions in Analysis}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\mathcal{P}(X)}{power set $\{A:A\subseteq X\}$; projection symbols use $P_{\mathcal{U}}$ or $P_j$}
\notationentry{f(A),\ f^{-1}(B)}{image $\{f(a):a\in A\}$ / preimage $\{x:f(x)\in B\}$ under $f$}
\notationentry{\operatorname{range}(f),\ f|_A}{range $f(\operatorname{dom}f)$ / restriction of $f$ to $A$}
\notationentry{g\circ f,\ \operatorname{id}_X}{composition of functions with compatible domain and codomain / identity function on $X$}
\notationentry{J_n,\ \#E,\ A\sim_{\#}B,\ [x]}{finite initial segment $J_0=\emptyset$, $J_n=\{1,\ldots,n\}$ / cardinality of a finite set / existence of a bijection $A\to B$ / equivalence class of $x$ for the specified relation}
\end{tabularx}

\notationgroup{Order and Completeness}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\sup E,\ \inf E}{least upper bound / greatest lower bound in the ambient ordered set, with values in $\overline{\R}$ when extended values are allowed}
\notationentry{\abs{x},\ \abs{a-b}}{real absolute value or complex modulus / usual real-line distance $d(a,b)=\abs{a-b}$}
\notationentry{\overline{\R}}{extended real line $\R\cup\{-\infty,+\infty\}$ ordered by $-\infty<x<+\infty$ for $x\in\R$; the bar is conventional, not topological closure}
\notationentry{x<y,\ x\leq y}{strict order / non-strict order in the ambient ordered set}
\end{tabularx}

\notationgroup{Sequences and Series}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{(a_n),\ a_n\to a}{sequence indexed by $n\in\N$ / convergence of the sequence to $a$}
\notationentry{(a_{n_k}),\ n_1<n_2<\cdots}{subsequence indexed by a strictly increasing sequence $(n_k)$ of positive integers}
\notationentry{s_n=\sum_{k=1}^{n}a_k}{partial-sum sequence associated to $\sum a_k$}
\notationentry{\limsup a_n=s^*,\ \liminf a_n=s_*}{upper and lower limits in $\overline{\R}$, equivalently $\inf_N\sup_{n\geq N}a_n$ and $\sup_N\inf_{n\geq N}a_n$ for real sequences}
\notationentry{\sum_{n=1}^{\infty} a_n,\ \prod_{n=1}^{\infty} a_n}{limit of partial sums / limit of finite partial products when defined}
\end{tabularx}

\notationgroup{Metric and Topological Notation}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{d(x,y),\ B_r(x)}{metric distance / open ball $\{y:d(x,y)<r\}$}
\notationentry{B_\varepsilon(a)=V_\varepsilon(a)}{real-line open ball $(a-\varepsilon,a+\varepsilon)$; $V_\varepsilon(a)$ is Abbott's notation}
\notationentry{\overline{B}_r(x),\ \{G_\alpha\}_{\alpha\in A}}{closed ball $\{y:d(x,y)\leq r\}$ / indexed open cover}
\notationentry{d_E,\ d_{\max},\ d_{\mathrm{sum}}}{Euclidean, maximum, and sum product metrics on a product space}
\notationentry{E^\circ,\ \overline{E},\ E^{\operatorname{acc}}}{interior / topological closure / set of limit points of $E$}
\notationentry{\operatorname{bd}(E),\ E^c}{boundary $\overline{E}\setminus E^\circ$ / complement in the ambient space}
\notationentry{\diam E}{diameter $\sup\{d(p,q):p,q\in E\}$ of a nonempty metric set}
\notationentry{X\simeq Y}{homeomorphic metric spaces, i.e. related by a bijection with continuous inverse}
\notationentry{A\cap\overline{B}=\emptyset,\ \overline{A}\cap B=\emptyset}{definition of separated subsets $A$ and $B$}
\notationentry{F_\sigma,\ G_\delta,\ \text{meager/thin/first category}}{countable union of closed sets / countable intersection of open sets / subset of a countable union of nowhere-dense sets}
\notationentry{\text{residual/thick},\ \text{generic}}{set containing a countable intersection of dense open sets / property whose truth set is residual}
\end{tabularx}

\notationgroup{Functions, Differentiation, and Integration}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\lim_{x\to p}f(x),\ f(x+),\ f(x-)}{limit of $f$ at $p$ / right and left limits at $x$}
\notationentry{C(X),\ C([a,b]),\ C_b(X),\ \norm{f}_\infty}{continuous functions on $X$ and $[a,b]$ / bounded continuous functions / supremum norm}
\notationentry{f_n\to f,\ f_n\rightrightarrows f}{pointwise convergence / uniform convergence on the ambient domain}
\notationentry{f'(x),\ Df(a),\ D_j f_i,\ \partial_i f,\ J_f(x)}{one-variable derivative, total derivative at $a$, component partial $D_j f_i$, scalar partial, and Jacobian determinant; $J_f$ is distinct from $J_n$ and $J_k(\lambda)$}
\notationentry{\nabla f,\ D_{\mathbf{u}}f,\ C^r,\ C^\infty,\ C^\omega}{gradient, directional derivative, and classes $C^r$, $C^\infty$, $C^\omega$}
\notationentry{\Pi=\{x_0,\ldots,x_n\},\ c_i,\ \Delta x_i,\ \Delta\alpha_i}{partition of $[a,b]$, tags $c_i\in[x_{i-1},x_i]$, and increments $\Delta x_i=x_i-x_{i-1}$, $\Delta\alpha_i=\alpha(x_i)-\alpha(x_{i-1})$}
\notationentry{R(f,\Pi,\{c_i\}),\ \delta,\ \delta(x)}{$R(f,\Pi,\{c_i\})=\sum_{i=1}^{n}f(c_i)\Delta x_i$ / constant mesh bound / gauge; uppercase $R(\cdots)$ is local to Riemann sums}
\notationentry{U(\Pi,f,\alpha),\ L(\Pi,f,\alpha)}{Riemann--Stieltjes upper and lower sums with integrator $\alpha$}
\notationentry{f\in\mathcal{R},\ f\in\mathcal{R}(\alpha)}{ordinary Riemann integrability / Riemann--Stieltjes integrability with respect to $\alpha$}
\notationentry{\int_a^b f,\ \int_a^b f\,d\alpha,\ \int_E f\,d\mu}{ordinary Riemann integral / Riemann--Stieltjes integral / measure integral}
\notationentry{\int_a^\infty f,\ \int_{-\infty}^{\infty} f}{one-sided and two-sided improper integrals as limits over finite truncations}
\notationentry{\ell(\Pi,\gamma),\ \ell(\gamma)}{polygonal length of $\gamma$ along $\Pi$ / curve length}
\notationentry{I^k,\ Q^k,\ \operatorname{supp}(f)}{$k$-cell, standard $k$-simplex, and closed support $\overline{\{x:f(x)\neq0\}}$; $I^k$ is not an identity operator}
\notationentry{dx_I,\ dx_I\wedge dx_J}{basic form indexed by $I$ / wedge product of basic forms}
\notationentry{\omega,\ d\omega,\ T^\ast\omega,\ \partial_{\mathrm{ch}}\Psi}{differential form, exterior derivative, pullback by $T$, and chain boundary}
\notationentry{\int_\Psi\omega,\ \partial_{\mathrm{ch}}[\mathbf{p}_0,\ldots,\mathbf{p}_k]}{integral of $\omega$ over the chain $\Psi$ / oriented boundary $\sum_{j=0}^{k}(-1)^j[\mathbf{p}_0,\ldots,\widehat{\mathbf{p}_j},\ldots,\mathbf{p}_k]$}
\notationentry{dV,\ dA,\ ds}{volume, area, and arc-length elements}
\end{tabularx}

\notationgroup{Power Series, Fourier, and Special Functions}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\sum_{n\geq0} c_n(x-a)^n,\ R}{power series centered at $a$ / radius of convergence in this context, distinct from $R(\cdots)$ and the $R$ in $QR$}
\notationentry{I_R,\ K_\rho}{centered interval $I_R=(-R,R)$ / compact subinterval $K_\rho=[-\rho,\rho]\subset I_R$ for $0<\rho<R$}
\notationentry{\widehat{f}(m),\ s_N(f;x),\ s_N(f)}{$\widehat{f}(m)=(2\pi)^{-1}\int_{-\pi}^{\pi}f(x)e^{-imx}\,dx$ / $s_N(f;x)=\sum_{\abs{m}\leq N}\widehat{f}(m)e^{imx}$, viewed pointwise or as a function}
\notationentry{f\sim_{\mathrm{coef}}\sum \widehat{f}(n)e^{inx},\ f\sim_{\mathrm{coef}}\sum c_n\phi_n}{formal Fourier or orthonormal coefficient expansion; no convergence assertion is implied}
\notationentry{D_N(t)}{Dirichlet kernel $D_N(t)=\sum_{m=-N}^{N}e^{imt}$; uppercase $D_N$ is not derivative notation}
\notationentry{\mathsf{E},\ \mathsf{L},\ \mathsf{C},\ \mathsf{S}}{power-series exponential, inverse logarithm, cosine, and sine}
\notationentry{\Gamma(x),\ \mathrm{B}(x,y)}{gamma function / beta function in roman type, distinct from balls $B_r(x)$ and variables $B$}
\end{tabularx}

\notationgroup{Measure and Function Spaces}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{E\in\mathcal{M},\ E\in\mathcal{M}(\mu)}{$E$ measurable in the ambient $\sigma$-algebra / Rudin measurable set generated by $\mu$}
\notationentry{\mathcal{B}(\R^n),\ \mathcal{L}^n}{Borel $\sigma$-algebra on $\R^n$ / Lebesgue measurable subsets of $\R^n$; not a basis $\mathcal{B}$, a linear-map space $\LL$, or a least-squares set $\operatorname{LS}$}
\notationentry{\mathcal{E},\ \mathcal{M}_F(\mu),\ A\triangle B,\ d_\mu(A,B)}{elementary sets, finitely $\mu$-measurable sets, symmetric difference, and $d_\mu(A,B)=\mu^\ast(A\triangle B)$; $\mathcal{M}$ is measure-theoretic here}
\notationentry{\mu^\ast(E),\ m^\ast(E),\ \mathbf{1}_E}{$\mu$-outer measure / Lebesgue outer measure / indicator $\mathbf{1}_E(x)=1$ for $x\in E$ and $0$ otherwise}
\notationentry{\operatorname{vol}_n(B),\ m_n^\ast(\Omega)}{$n$-dimensional box volume / box-cover outer measure $\inf\{\sum_{j\in J}\operatorname{vol}_n(B_j):(B_j)_{j\in J}\text{ covers }\Omega,\ J\text{ at most countable}\}$}
\notationentry{m(E),\ \mu(E)}{Lebesgue or elementary volume of $E$ / measure of $E$; not algebraic multiplicity $m_A(\lambda)$ or minimal polynomial $\mu_A$}
\notationentry{\lambda^+,\ \lambda^-,\ \abs{\lambda},\ \lambda\ll\mu,\ \lambda\perp\mu,\ \frac{d\lambda}{d\mu}}{positive variation, negative variation, total variation, absolute continuity, mutual singularity, and Radon--Nikodym derivative}
\notationentry{\bowtie,\ \{f\bowtie a\}}{relation in $\{>,\geq,<,\leq\}$ / level set $\{x:f(x)\bowtie a\}$}
\notationentry{f^+,\ f^-,\ \mathcal{S}^+(f),\ \mathcal{I}_E(s)}{$f^+=\max(f,0)$, $f^-=-\min(f,0)$ / simple minorants of $f$ / simple-function integral on $E$}
\notationentry{f\sim_{\mathrm{a.e.}}g\ \text{on }E,\ \text{a.e.}}{$\mu(\{x\in E:f(x)\neq g(x)\})=0$ / outside a null set}
\notationentry{E_x,\ E^y,\ f_x,\ f^y}{product slices $E_x=\{y:(x,y)\in E\}$, $E^y=\{x:(x,y)\in E\}$, and $f_x(y)=f^y(x)=f(x,y)$}
\notationentry{\mathcal{U}f,\ F_a}{undergraph $\{(x,t):0<t<f(x)\}$ for $f\geq0$ / superlevel set $F_a=\{x:f(x)>a\}$; $\mathcal{U}f$ is not a vector subspace}
\notationentry{[f:A],\ [B:Q],\ \operatorname{dp}(E)}{$[f:A]=m(A)^{-1}\int_A f$ / $[B:Q]=m(B\cap Q)/m(Q)$ / density-point set of $E$}
\notationentry{\mathscr{L}^1(\mu),\ \mathscr{L}^1_{\mathrm{loc}}(\R^n)}{measurable $f$ with $\int\abs{f}\,d\mu<+\infty$ / locally integrable functions; script $\mathscr{L}$ is distinct from $\mathcal{L}^n$ and $\LL$}
\notationentry{L^p(\mu),\ L^p(E),\ \norm{f}_p}{Lebesgue space modulo a.e. equality with respect to $\mu$ or the ambient measure on $E$ / $\norm{f}_p=(\int\abs{f}^p\,d\mu)^{1/p}$}
\end{tabularx}

\endgroup
