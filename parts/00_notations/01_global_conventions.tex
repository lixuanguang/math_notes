\subsection*{Global Conventions and Basic Notation}

\begingroup
\small

\noindent\emph{Convention.} Notation is scoped by subject, argument type, font, decoration, and any local definition in force. Unadorned letters denote local objects by default; persistent meanings are carried by subscripts, superscripts, operators, fonts, or declared argument patterns. When a glyph is reused, the distinguishing feature is recorded below.

\notationgroup{Set, Function, and Logical Notation}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{x\in E,\ x\notin E}{membership / non-membership}
\notationentry{E\subseteq F,\ \emptyset}{subset inclusion, not necessarily proper / empty set}
\notationentry{X\times Y,\ f:X\to Y}{Cartesian product / function with domain $X$ and codomain $Y$}
\notationentry{f(A),\ f^{-1}(B),\ f^{-1}}{image of $A$ / preimage of $B$ / inverse function when $f$ is bijective; argument type and local context distinguish preimages from inverse-function values}
\notationentry{\operatorname{range}(f),\ f|_A}{range of $f$ / restriction of $f$ to $A$}
\notationentry{g\circ f,\ \operatorname{id}_X}{composition of functions with compatible domain and codomain / identity function on $X$}
\notationentry{\{A_i\}_{i\in I},\ \bigcup_{i\in I}A_i,\ \bigcap_{i\in I}A_i}{indexed family of sets / set of points lying in some $A_i$ / set of points lying in every $A_i$}
\notationentry{\neg,\ \land,\ \lor}{logical negation / conjunction / inclusive disjunction}
\notationentry{\Rightarrow,\ \Longrightarrow,\ \Longleftrightarrow}{logical implication (either arrow length) / logical equivalence}
\notationentry{\forall,\ \exists,\ \exists!}{universal / existential / unique-existence quantifier}
\notationentry{\defeq}{definition by equality}
\end{tabularx}

\notationgroup{Number, Size, and Scalar Notation}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\N,\ \N_0,\ \Z,\ \Q,\ \R,\ \C}{positive integers / nonnegative integers / integers / rationals / reals / complexes}
\notationentry{\mathbb{F},\ \mathbb{F}_0}{ambient scalar field unless fixed as $\R$ or $\C$ / prime subfield of $\mathbb{F}$; probabilities use explicit $\operatorname{Pr}$ or locally declared probability spaces}
\notationentry{\abs{x},\ \abs{E},\ \norm{x}}{absolute value or complex modulus for scalars; interval length or cardinality according to argument type; norm of a vector, function, or operator}
\notationentry{a,b,c}{locally declared scalars or interval endpoints}
\end{tabularx}

\endgroup
