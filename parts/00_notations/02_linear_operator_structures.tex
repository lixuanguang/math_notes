\subsection*{Linear and Operator Structures Notation}

\begingroup
\small

\notationgroup{Vector Spaces and Lists}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\mathbb{F},\ \mathbb{F}_0}{ambient scalar field, typically $\mathbb{R}$ or $\mathbb{C}$ / prime subfield of $\mathbb{F}$}
\notationentry{V(n,q),\ \mathbb{F}_q^n}{$n$-dimensional coordinate space over the finite field with $q$ elements}
\notationentry{\mathbb{F}^{(\mathbb{N})}}{vector space of finitely supported sequences over $\mathbb{F}$}
\notationentry{\mathbf{u},\mathbf{v},\mathbf{w},\mathbf{0}}{vectors / zero vector}
\notationentry{\mathcal{U},\mathcal{V},\mathcal{W}}{vector spaces or subspaces; other calligraphic letters such as $\mathcal{A},\mathcal{B},\mathcal{F},\mathcal{M}$ are local unless listed below; umbral algebras use script $\mathscr{P},\mathscr{F}$}
\notationentry{I_{\mathcal{V}},\ I_n,\ 0}{identity operator on $\mathcal{V}$ / identity matrix / zero map or zero matrix; distinct from the cell $I^k$ and interval $I_R$}
\notationentry{\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)}{ordered vector list; a basis iff it is linearly independent and spans}
\notationentry{(),\ \beta_{\widehat{j}}}{empty list / list $\beta$ with the $j$th entry omitted}
\notationentry{(\beta,\mathbf{v}),\ \beta_{j\leftarrow \mathbf{v}}}{list with $\mathbf{v}$ appended / list with the $j$th entry replaced by $\mathbf{v}$}
\notationentry{\mathcal{B}}{locally declared basis set, such as a Hamel, Hilbert, module, or affine basis; $\mathcal{B}(\R^n)$ denotes the Borel $\sigma$-algebra in analysis}
\notationentry{\operatorname{span}(E),\ \dim(\mathcal{V})}{linear span of $E$ / dimension of $\mathcal{V}$}
\notationentry{\operatorname{codim}_{\mathcal{V}}(\mathcal{U})}{codimension of $\mathcal{U}$ in $\mathcal{V}$; equal to $\dim(\mathcal{V}/\mathcal{U})$ in finite dimension}
\notationentry{\bigwedge_{i\in I}\mathcal{U}_i,\ \bigvee_{i\in I}\mathcal{U}_i}{meet / join of a family of subspaces}
\notationentry{\sum_{i\in I}\mathcal{U}_i}{sum of an indexed family of subspaces; equivalently the join}
\notationentry{\mathcal{U}+\mathcal{W},\ \mathcal{U}\oplus\mathcal{W},\ \mathcal{V}/\mathcal{U}}{sum, direct sum, quotient space}
\notationentry{\mathcal{V}_1\times\cdots\times\mathcal{V}_m}{product vector space with componentwise operations}
\notationentry{\prod_{i\in I}\mathcal{V}_i,\ \bigoplus_{i\in I}^{\mathrm{ext}}\mathcal{V}_i}{external direct product / external direct sum of finite-support families}
\notationentry{\mathcal{V}\cong\mathcal{W},\ \mathbf{e}_{j,k}}{isomorphic vector spaces / basis vector supported in one product component}
\notationentry{\mathbf{v}+\mathcal{U},\ \pi(\mathbf{v})}{coset modulo $\mathcal{U}$ / locally declared quotient map $\mathcal{V}\to\mathcal{V}/\mathcal{U}$}
\end{tabularx}

\notationgroup{Linear Maps and Duality}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{T:\mathcal{V}\to\mathcal{W},\ \LL(\mathcal{V},\mathcal{W})}{linear map / vector space of linear maps; distinct from measure-theoretic $\mathcal{L}^n$, least-squares $\operatorname{LS}(A,\mathbf{y})$, and local loss notation $\mathcal{L}(\theta)$}
\notationentry{ST,\ T|_{\mathcal{U}},\ T/\mathcal{U}}{composition $S\circ T$ / restriction to $\mathcal{U}$ / induced quotient operator}
\notationentry{T^{-1},\ T^m,\ p(T)}{inverse of an invertible map or operator / operator power / polynomial functional calculus}
\notationentry{\ker(T),\ \operatorname{range}(T)}{kernel and range of $T$}
\notationentry{\operatorname{rank}(T),\ \operatorname{nullity}(T)}{dimensions of the range and kernel of a finite-domain linear map $T$}
\notationentry{\mathcal{V}_{\mathbb{C}},\ \operatorname{cpx},\ T_{\mathbb{C}}}{complexification of a real vector space / canonical embedding $\mathbf{v}\mapsto\mathbf{v}+i\mathbf{0}$ / complexification of a real operator}
\notationentry{\operatorname{graph}(T),\ [S,T]}{graph of a map / commutator $ST-TS$}
\notationentry{\operatorname{Comm}(\mathcal{F}),\ \operatorname{Comm}(A)}{commutant of a family / commutant of a single matrix or operator}
\notationentry{A\sim_{\mathrm{row}}B,\ S\sim T}{row-equivalent matrices / operator similarity when declared; distinct from cardinal, a.e., and coefficient equivalence}
\notationentry{\mathcal{V}',\ \beta',\ T'}{dual space, dual basis, and dual map; prime denotes duality in this context}
\notationentry{\varphi,\psi,\ \mathcal{U}^0}{linear functionals / annihilator}
\notationentry{\mathcal{V}'',\ \Lambda_{\mathcal{V}}}{double dual / natural evaluation map $\mathcal{V}\to\mathcal{V}''$}
\notationentry{[\mathbf{v}]_{\beta},\ [T]_{\gamma\leftarrow\beta}}{coordinate vector in basis $\beta$ / matrix of $T$ from $\beta$-coordinates to $\gamma$-coordinates}
\notationentry{C_{\gamma\leftarrow\beta}}{change-of-basis matrix sending $\beta$-coordinate columns to $\gamma$-coordinate columns}
\end{tabularx}

\notationgroup{Matrices and Entrywise Structure}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{A,B,C,\ \operatorname{Mat}(m,n,\mathbb{F})}{matrix variables / space of $m\times n$ matrices over $\mathbb{F}$; decorated forms such as $C_f$ and $C_{ij}$ have separate meanings}
\notationentry{A_{ij},\ A_{j,\cdot},\ A_{\cdot,k}}{entry in row $i$ and column $j$ / $j$th row / $k$th column}
\notationentry{[a_{ij}],\ E_{j,k},\ A\mathbf{x}}{matrix with entries $a_{ij}$ / matrix unit / matrix-vector product}
\notationentry{A^{-1},\ A^t,\ A^T,\ A^*}{inverse matrix when it exists / transpose / alternate transpose notation / conjugate transpose or adjoint}
\notationentry{\operatorname{diag}(\lambda_1,\ldots,\lambda_n)}{diagonal matrix with diagonal entries $\lambda_1,\ldots,\lambda_n$}
\notationentry{\overline{z},\ \operatorname{Re}z,\ \operatorname{Im}z}{complex conjugate / real part / imaginary part; for sets, overline denotes closure}
\notationentry{\operatorname{tr}(A),\det(A)}{trace and determinant}
\notationentry{\operatorname{adj}(A)}{adjugate matrix}
\notationentry{M_{ij},\ C_{ij},\ A_j(\mathbf{b})}{minor matrix / cofactor / matrix obtained from $A$ by replacing column $j$ with $\mathbf{b}$}
\notationentry{\operatorname{col}(A),\ \operatorname{row}(A)}{column space and row space of $A$}
\notationentry{\operatorname{rank}(A),\ \operatorname{nullity}(A)}{rank and nullity of $A$}
\notationentry{A\oplus B,\ A\otimes B}{matrix direct sum / Kronecker product; vector-space direct sums use subspace notation}
\notationentry{A\otimes I_q+I_n\otimes B}{Kronecker sum of square matrices $A$ and $B$ in compatible dimensions}
\notationentry{A\circ B,\ \operatorname{vec}(A)}{Hadamard product / column-stacking vectorisation}
\notationentry{M/A,\ M/D}{Schur complement of the block $A$ or $D$ in a compatible block matrix $M$; distinct from quotient-space notation}
\notationentry{B,\ L=B^tB}{directed incidence matrix / associated incidence Laplacian in a graph context}
\end{tabularx}

\notationgroup{Inner Product Geometry}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\norm{\mathbf{v}},\ \normop{T}}{vector norm / operator norm}
\notationentry{\norm{\cdot}_1,\ \norm{\cdot}_2,\ \norm{\cdot}_{\infty}}{coordinate one-norm / Euclidean or spectral two-norm according to argument type / sup norm}
\notationentry{\norm{\cdot}_F}{Frobenius norm}
\notationentry{\innerprod{\mathbf{u},\mathbf{v}},\ \innerprod[F]{A,B}}{inner product / Frobenius inner product}
\notationentry{\mathbf{u}\perp\mathbf{v},\ \mathcal{U}^{\perp},\ \delta_{ij}}{orthogonal vectors, orthogonal complement, Kronecker delta}
\notationentry{\operatorname{proj}_{\mathbf{v}}(\mathbf{u}),\ P_{\mathcal{U}}}{orthogonal projection of $\mathbf{u}$ onto $\operatorname{span}(\mathbf{v})$ / orthogonal projection onto $\mathcal{U}$}
\notationentry{\operatorname{LS}(A,\mathbf{y})}{least-squares solution set $\operatorname*{arg\,min}_{\mathbf{x}\in\mathbb{F}^n}\norm{A\mathbf{x}-\mathbf{y}}_2$ for $A\in\operatorname{Mat}(m,n,\mathbb{F})$; roman notation avoids collision with $\LL$ and $\mathcal{L}^n$}
\notationentry{d(\mathbf{v},\mathcal{U})}{distance $\inf_{\mathbf{u}\in\mathcal{U}}\norm{\mathbf{v}-\mathbf{u}}$ from $\mathbf{v}$ to $\mathcal{U}$}
\notationentry{\operatorname{Gram}(\mathbf{u}_1,\ldots,\mathbf{u}_n)}{Gram matrix of $\mathbf{u}_1,\ldots,\mathbf{u}_n$}
\notationentry{\operatorname{GramDet}(\mathbf{u}_1,\ldots,\mathbf{u}_n)}{Gram determinant}
\notationentry{P_{\mathbf{u}},\ U_{\mathbf{w}}}{rank-one projection matrix / Householder matrix, with defining vector indicated by the subscript}
\notationentry{\gamma=(\mathbf{e}_1,\ldots,\mathbf{e}_n)}{ordered basis; orthonormal when declared in an inner-product context}
\notationentry{\mathbf{e}=(1,\ldots,1)^t}{all-ones vector}
\notationentry{\operatorname{Ray}_A(\mathbf{x}),\ \mathbf{x}^*A\mathbf{x}}{Rayleigh quotient notation; for unit $\mathbf{x}$, $\operatorname{Ray}_A(\mathbf{x})=\mathbf{x}^*A\mathbf{x}$ / Hermitian quadratic form}
\notationentry{\operatorname{Sym}(B),\ \operatorname{Skew}(B),\ \operatorname{Alt}(B)}{predicates that the bilinear form $B$ is symmetric, skew-symmetric, or alternating}
\notationentry{M_{\beta},\ \Delta(\mathcal{V}),\ \operatorname{rad}(\mathcal{V})}{matrix of a metric bilinear form in basis $\beta$ / discriminant set or square class / radical of a metric vector space}
\notationentry{\mathcal{V}=\mathcal{S}\odot\mathcal{T},\ \mathcal{H},\ w(\mathcal{V})}{orthogonal direct sum / hyperbolic space or plane in metric-vector-space contexts / Witt index; affine hyperplanes use $\mathcal{H}(\mathbf{N},b)$}
\notationentry{\mathcal{O}(\mathcal{V}),\ \operatorname{Sp}(\mathcal{V}),\ \tau_{\mathbf{v},a},\ \sigma_{\mathbf{u}}}{orthogonal group / symplectic group / symplectic transvection / orthogonal symmetry}
\end{tabularx}

\notationgroup{Affine, Projective, and Convex Geometry}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\mathcal{A}(\mathcal{V}),\ \mathcal{A}_0(\mathcal{V}),\ \operatorname{Sub}(\mathcal{V})}{flats in $\mathcal{V}$ / $\mathcal{A}(\mathcal{V})\cup\{\emptyset\}$ / subspaces of $\mathcal{V}$; $\mathcal{A}$ without arguments remains local}
\notationentry{\operatorname{affspan}(E),\ \operatorname{affhull}(E)}{affine span / least flat containing $E$}
\notationentry{X\parallel Y,\ X\vee Y}{parallel flats / join of flats or projective subspaces; meet is intersection in the relevant lattice}
\notationentry{\R_+^n,\ \R_{++}^n,\ \mathbf{x}\geq\mathbf{0},\ \mathbf{x}>\mathbf{0},\ \mathbf{x}\gg\mathbf{0}}{nonzero nonnegative orthant / coordinatewise positive orthant / coordinatewise nonnegative, nonzero nonnegative, and coordinatewise positive inequalities}
\notationentry{\operatorname{conv}(S),\ \Delta_{k-1},\ C_A}{convex hull / coefficient simplex $\{(t_i):t_i\geq0,\ \sum_i t_i=1\}$ / nonnegative cone generated by the columns of $A$; distinct from cofactor and companion notation}
\notationentry{\mathcal{H}(\mathbf{N},b),\ \mathcal{H}_{\pm}(\mathbf{N},b),\ \innerprod{\mathbf{N},X}<b}{affine hyperplane / closed half-spaces / setwise inequality, meaning $\innerprod{\mathbf{N},\mathbf{x}}<b$ for all $\mathbf{x}\in X$}
\end{tabularx}

\notationgroup{Spectral and Polynomial Structure}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{E_{\lambda}(T),\ \operatorname{spec}(T)}{eigenspace $\ker(T-\lambda I_{\mathcal{V}})$ / spectrum}
\notationentry{\operatorname{Inv}(T)}{lattice or set of $T$-invariant subspaces}
\notationentry{\lambda_i(A),\ m_A(\lambda)}{eigenvalues in the locally declared order / algebraic multiplicity of $\lambda$ as a root of $p_A$; distinct from measure notation $m(E)$}
\notationentry{\mathcal{G}_{\lambda}(T),\ J_k(\lambda)}{generalized eigenspace for $\lambda$ / Jordan block of size $k$ and eigenvalue $\lambda$}
\notationentry{p_A,\ q_T,\ \chi_T}{matrix characteristic polynomial / operator characteristic polynomial / characteristic polynomial determined by elementary divisors}
\notationentry{\mu_A,\ \mu_T,\ w_p(\lambda,A)}{minimal polynomial of a matrix or operator / Weyr count of $\lambda$-Jordan blocks of size at least $p$; distinct from measure notation $\mu(E)$}
\notationentry{P_j,\ T=\sum_j\mu_jP_j}{spectral projections / spectral decomposition with distinct eigenvalues $\mu_j$}
\notationentry{\sqrt{T},\ A^{1/2},\ \rho(A)}{positive square root of a positive operator or matrix / spectral radius of $A$; distinct from the interval parameter in $K_\rho$}
\notationentry{\mathbb{F}[z]}{polynomials in $z$ over $\mathbb{F}$}
\notationentry{\mathscr{P}=\C[x],\ \mathscr{P}',\ \mathscr{F}=\C[[t]]}{umbral polynomial algebra / algebraic dual / formal power-series algebra; script notation distinguishes these from $\mathcal{P}(X)$ and local families $\mathcal{F}$}
\notationentry{\mathcal{V}_T,\ \mathcal{C}_T(\mathbf{v}),\ \operatorname{ann}_T(\mathbf{v})}{$\mathbb{F}[z]$-module induced by $T$ / $T$-cyclic subspace generated by $\mathbf{v}$ / annihilator ideal of $\mathbf{v}$}
\notationentry{m_{\mathbf{v}},\ C_f,\ \operatorname{InvFact}(T),\ \operatorname{ElemDiv}(T)}{order polynomial of $\mathbf{v}$ under $T$ / companion matrix of $f$ / invariant factors / elementary divisors}
\notationentry{e^{tA}}{matrix exponential $\sum_{k\geq0}(tA)^k/k!$}
\end{tabularx}

\notationgroup{Matrix Analysis and Numerical Structure}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}

\notationentry{A=QR,\ A=LL^*,\ A=LDL^t}{QR factorization / Cholesky factorization / symmetric or Hermitian $LDL^t$ factorization}
\notationentry{A=V\Sigma W^*,\ A=X\Sigma_rY^*}{singular value decomposition / compact SVD with rank $r$}

\notationentry{\kappa_2(A),\ \Delta A,\ \Delta\mathbf{x}}{spectral condition number of an invertible matrix / matrix and solution perturbations}
\notationentry{\Delta(A)}{defect from normality; distinct from perturbation notation $\Delta A$}
\notationentry{R_k(A),\ R_k^{\mathrm{off}}(A),\ R_{\max}(A)}{absolute row sum $\sum_j\abs{a_{kj}}$ / deleted row sum $\sum_{j\neq k}\abs{a_{kj}}$ / maximum absolute row sum}

\notationentry{\operatorname{tridiag}(a,b,c)}{tridiagonal matrix with subdiagonal $a$, diagonal $b$, and superdiagonal $c$}

\notationentry{\mathcal{K}_m(A,\mathbf{b})}{Krylov subspace $\operatorname{span}(\mathbf{b},A\mathbf{b},\ldots,A^{m-1}\mathbf{b})$}

\notationentry{A=S-T,\ B=S^{-1}T,\ \mathbf{e}_k}{matrix splitting / stationary iteration matrix / $k$th error vector}
\notationentry{A=D+L+U,\ B_J,\ B_{\operatorname{GS}}}{diagonal, lower, and upper splitting / Jacobi and Gauss--Seidel iteration matrices}
\notationentry{\omega,\ S_\omega,\ T_\omega,\ B_\omega}{SOR parameter / SOR splitting matrices / SOR iteration matrix; distinct from the differential form $\omega$}
\notationentry{M,\ M^{-1}A,\ AM^{-1}}{preconditioner / left and right preconditioned matrices}
\end{tabularx}
