\subsection*{Algebraic Structures Notation}

\begingroup
\small

\notationgroup{Algebraic Language and Categories}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\mathcal{C},\ A\xrightarrow{f}B,\ \operatorname{id}_A}{ambient category / morphism $f:A\to B$ / identity morphism of $A$; $\mathcal{C}$ remains local unless declared}
\notationentry{\operatorname{Hom}_{\mathcal{C}}(A,B),\ g\circ f}{set of morphisms $A\to B$ in $\mathcal{C}$ / composition for compatible morphisms}
\notationentry{A\simeq B,\ F:\mathcal{C}\to\mathcal{D}}{isomorphic objects / functor with domain $\mathcal{C}$ and codomain $\mathcal{D}$}
\end{tabularx}

\notationgroup{Groups, Actions, Rings, and Ideals}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{G,H,K,\ e_G,\ \langle S\rangle}{groups or subgroups / identity element of $G$ / substructure generated by $S$ in the ambient algebraic category}
\notationentry{H\leq G,\ H\trianglelefteq G,\ G/H,\ [G:H]}{subgroup / normal subgroup / quotient group / subgroup index}
\notationentry{\operatorname{ord}(g),\ \operatorname{Aut}(G)}{order of an element / automorphism group}
\notationentry{R,S,\ 0_R,\ 1_R,\ R^\times}{rings / zero and identity elements of $R$ / unit group of $R$}
\notationentry{Z(R),\ \operatorname{char}(R)}{center of $R$ / characteristic of $R$}
\notationentry{I,J,\ \mathfrak{a},\mathfrak{p},\ R/I}{ideals; fraktur letters denote distinguished ideals, such as prime ideals, when declared / quotient ring}
\notationentry{R[x],\ \mathbb{F}[z],\ \deg f}{polynomial rings in the displayed indeterminate / degree of a polynomial}
\notationentry{U_n,\ \Omega_n,\ \Phi_n(x),\ \varphi(n)}{complex $n$th roots of unity / primitive $n$th roots / $n$th cyclotomic polynomial / Euler totient; $\Omega_n$ is algebraic root notation, distinct from domain notation}
\end{tabularx}

\notationgroup{Modules and Associative Algebras}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{M,N,\ S\leq M,\ M/N}{modules / submodule $S$ of $M$ / quotient module}
\notationentry{\operatorname{Hom}_R(M,N),\ \operatorname{End}_R(M)}{set of $R$-module homomorphisms $M\to N$ / endomorphism ring of $M$}
\notationentry{\ker\tau,\ \operatorname{im}\tau,\ \overline{\tau}}{kernel / image / quotient-induced map $M/\ker\tau\to\operatorname{im}\tau$ for a homomorphism $\tau:M\to N$}
\notationentry{\bigoplus_i M_i,\ M=N\oplus H}{external direct sum of modules / internal direct-sum decomposition}
\notationentry{\mathcal{C}_S(N),\ \mathcal{L}_\sigma,\ \mathcal{R}_\sigma}{set $\{H\leq N:N=S\oplus H\}$ of complements / left inverses of $\sigma$ / right inverses of $\sigma$; subscript distinguishes these from $\LL$ and $\mathcal{L}^n$}
\notationentry{\mathcal{F}(\mathcal{V})}{finite-rank operators $\{T\in\LL(\mathcal{V}):\dim\operatorname{im}T<\infty\}$ in associative-algebra contexts; distinct from local function-family notation}
\end{tabularx}

\endgroup
