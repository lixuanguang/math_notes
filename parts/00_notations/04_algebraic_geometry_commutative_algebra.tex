\subsection*{Algebraic Geometry and Commutative Algebra Notation}

\begingroup
\small

\noindent\emph{Status.} This part of the notes is still mostly scaffolded. The entries below reserve the notation promised by the section headings and keep it distinct from already-active algebra and analysis notation.

\notationgroup{Commutative Algebra}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\operatorname{Spec}R,\ \mathfrak{p}\in\operatorname{Spec}R}{set of prime ideals of a commutative ring $R$ / prime ideal $\mathfrak{p}\subseteq R$}
\notationentry{D(f),\ V(I)}{standard open set $\{\mathfrak{p}\in\operatorname{Spec}R:f\notin\mathfrak{p}\}$ / Zariski closed set $\{\mathfrak{p}:I\subseteq\mathfrak{p}\}$}
\notationentry{S^{-1}R,\ R_{\mathfrak{p}},\ k(\mathfrak{p})}{localization at a multiplicative set $S$ / localization $(R\setminus\mathfrak{p})^{-1}R$ / residue field $R_{\mathfrak{p}}/\mathfrak{p}R_{\mathfrak{p}}$}
\notationentry{\operatorname{rad}(I),\ \sqrt{I}}{radical ideal $\{r\in R:r^n\in I\text{ for some }n\geq1\}$; use $\operatorname{rad}(\mathcal{V})$ for metric-vector-space radicals}
\end{tabularx}

\notationgroup{Affine and Projective Geometry}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\mathbb{A}^n_k,\ \mathbb{P}^n_k}{affine $n$-space over a field $k$ / projective $n$-space over $k$; geometric notation distinct from probability notation}
\notationentry{I(X),\ Z(I),\ k[X]}{vanishing ideal $\{f\in k[x_1,\ldots,x_n]:f|_X=0\}$ / zero locus of an ideal / coordinate ring, typically $k[x_1,\ldots,x_n]/I(X)$}
\notationentry{f^\#,\ \varphi^\ast}{coordinate-ring pullback, e.g. $f^\#:k[Y]\to k[X]$ for $f:X\to Y$; distinct from adjoints and differential-form pullbacks by context}
\end{tabularx}

\endgroup
