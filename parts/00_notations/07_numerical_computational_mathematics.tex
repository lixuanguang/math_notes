\subsection*{Numerical and Computational Mathematics Notation}

\begingroup
\small

\noindent\emph{Status.} Apart from numerical linear algebra, already covered under Linear and Operator Structures, the numerical and computational chapters remain placeholders. The entries below reserve notation only; they do not introduce PDE or machine-learning conventions.

\notationgroup{Floating Point, Approximation, and Error}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\operatorname{fl}(x),\ u,\ \varepsilon_{\mathrm{mach}}}{floating-point rounded value, unit roundoff, and machine epsilon}
\notationentry{x_h,\ e_h,\ \mathcal{O}(h^p)}{approximation at discretization scale $h$, corresponding error, and Landau order as $h\to0$}
\notationentry{\widehat{x},\ \Delta x,\ r=b-A\widehat{x}}{computed approximation, perturbation or error in $x$, and residual for $Ax=b$}
\end{tabularx}

\notationgroup{Monte Carlo and Stochastic Computation}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{X_1,\ldots,X_N,\ \widehat{\mu}_N}{samples / sample-mean estimator $\widehat{\mu}_N=N^{-1}\sum_{i=1}^{N}X_i$ when estimating $\mu=\mathbb{E}X$}
\notationentry{\operatorname{Var}(\widehat{\mu}_N),\ \operatorname{SE}(\widehat{\mu}_N)}{estimator variance / standard error $\operatorname{SE}(\widehat{\mu}_N)=\sqrt{\operatorname{Var}(\widehat{\mu}_N)}$}
\notationentry{w_i,\ \widehat{I}_N}{importance weights / Monte Carlo estimate of an integral or expectation, when locally introduced}
\end{tabularx}

\endgroup
