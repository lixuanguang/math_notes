\subsection{Shift Operators and Kernel-Range Examples}
Fix a field $\mathbb{F}$, let $\mathcal{E}\defeq\mathbb{F}^{(\mathbb{N})}$ be its finitely supported sequence space, and set $\mathbf{e}_1\defeq(1,0,0,\ldots)$ and $\mathcal{E}_0\defeq\{\mathbf{x}\in\mathcal{E}:x_1=0\}$.

\begin{proposition} \label{prop:shift-operators-finitely-nonzero} \hlt{Shift Operators on Finitely Supported Sequences}\\
Define $R,S:\mathcal{E}\to\mathcal{E}$ by $R(x_1,x_2,\ldots)\defeq(0,x_1,x_2,\ldots)$ and $S(x_1,x_2,\ldots)\defeq(x_2,x_3,\ldots)$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $R,S\in\LL(\mathcal{E})$.
\item $\ker(R)=\{\mathbf{0}_{\mathcal{E}}\}$.
\item $\operatorname{range}(R)=\mathcal{E}_0$.
\item $\ker(S)=\operatorname{span}(\mathbf{e}_1)$.
\item $\operatorname{range}(S)=\mathcal{E}$.
\item $R$ is injective but not surjective, whereas $S$ is surjective but not injective.
\item $\mathcal{E}$ is infinite-dimensional.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{x}\in\mathcal{E}$, $\abs{\operatorname{supp}(R\mathbf{x})}=\abs{\operatorname{supp}(\mathbf{x})}<\infty$ and $\abs{\operatorname{supp}(S\mathbf{x})}\leq\abs{\operatorname{supp}(\mathbf{x})}$, so $R$ and $S$ are well-defined.\\
Coordinatewise operations establish $(\forall T\in\{R,S\})(\forall a,b\in\mathbb{F})(\forall\mathbf{x},\mathbf{y}\in\mathcal{E})\,T(a\mathbf{x}+b\mathbf{y})=aT\mathbf{x}+bT\mathbf{y}$; hence $R,S\in\LL(\mathcal{E})$.
\item For $\mathbf{x}=(x_1,x_2,\ldots)\in\mathcal{E}$, $R\mathbf{x}=\mathbf{0}_{\mathcal{E}}\Longleftrightarrow(\forall j\in\mathbb{N})\,x_j=0\Longleftrightarrow\mathbf{x}=\mathbf{0}_{\mathcal{E}}$. Thus $\ker(R)=\{\mathbf{0}_{\mathcal{E}}\}$.
\item By definition, $\operatorname{range}(R)\subseteq\mathcal{E}_0$. Conversely, $\mathbf{y}=(0,y_2,y_3,\ldots)\in\mathcal{E}_0$ has $\mathbf{x}\defeq(y_2,y_3,\ldots)\in\mathcal{E}$ with $R\mathbf{x}=\mathbf{y}$; hence $\operatorname{range}(R)=\mathcal{E}_0$.
\item For $\mathbf{x}=(x_1,x_2,\ldots)\in\mathcal{E}$, $S\mathbf{x}=\mathbf{0}_{\mathcal{E}}\Longleftrightarrow(\forall j\in\mathbb{N}\setminus\{1\})\,x_j=0\Longleftrightarrow\mathbf{x}=x_1\mathbf{e}_1$. Thus $\ker(S)=\operatorname{span}(\mathbf{e}_1)$.
\item The identity $(\forall\mathbf{y}\in\mathcal{E})\,S(R\mathbf{y})=\mathbf{y}$ holds by the definitions, with $R\mathbf{y}\in\mathcal{E}$ by (i); hence $\operatorname{range}(S)=\mathcal{E}$.
\item The map $R$ is injective by Proposition \ref{prop:injective-iff-kernel-zero} with (ii), and $R$ is not surjective by (iii), since $\mathbf{e}_1\notin\mathcal{E}_0$.\\
The map $S$ is surjective by (v) and not injective by Proposition \ref{prop:injective-iff-kernel-zero} with (iv), since $\mathbf{e}_1\neq\mathbf{0}_{\mathcal{E}}$.
\item The assumption $\dim(\mathcal{E})<\infty$ would make $R$ surjective by Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences} with (i) and injectivity from (vi), contradicting (vi).
\end{enumerate}
\hlt{Structural Commentary}: Matched claim items verify linearity, fibers, images, map properties, and the dimension obstruction.\\
\hlt{Key Insight}: The shifts exchange a missing coordinate with a nontrivial kernel.
\end{proof}
\end{proposition}

\begin{example} \label{ex:shift-operators-show-operator-algebra-noncommutative} \hlt{Shift Operators Show Noncommutativity}\\
For the shifts above, $SR,RS\in\LL(\mathcal{E})$ by Proposition \ref{prop:linear-map-product-algebra}(i). For $\mathbf{x}=(x_1,x_2,\ldots)\in\mathcal{E}$, $SR\mathbf{x}=\mathbf{x}$ and $RS\mathbf{x}=(0,x_2,x_3,\ldots)$.\\
Hence $SR=I_{\mathcal{E}}$ by Assumption \ref{assump:function-extensionality}. Since $RS\mathbf{e}_1=\mathbf{0}_{\mathcal{E}}\neq\mathbf{e}_1=SR\mathbf{e}_1$, $RS\neq SR$ and $\LL(\mathcal{E})$ is noncommutative.
\end{example}
