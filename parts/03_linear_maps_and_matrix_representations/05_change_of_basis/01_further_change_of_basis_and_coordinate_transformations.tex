\subsection{Further Change of Basis and Coordinate Transformations}

\begin{theorem} \label{thm:change-of-basis-formula} \hlt{Change of Basis Formula}\\
Let $n\geq1$, let $\dim(\mathcal{V})=n$, let $T\in\LL(\mathcal{V})$, and fix a basis $\beta$ of $\mathcal{V}$. Set $A\defeq[T]_{\beta\leftarrow\beta}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\gamma$ is a basis of $\mathcal{V}$ and $S=C_{\gamma\leftarrow\beta}$, then $[T]_{\gamma\leftarrow\gamma}=SAS^{-1}$.
\item If $S\in\operatorname{Mat}(n,n,\mathbb{F})^{\times}$, then some basis $\gamma$ of $\mathcal{V}$ satisfies $[T]_{\gamma\leftarrow\gamma}=SAS^{-1}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Suppose $\gamma$ is a basis and $S=C_{\gamma\leftarrow\beta}$. Then $S^{-1}=C_{\beta\leftarrow\gamma}$ by Theorem \ref{thm:change-of-basis-matrices}(i).\\
Hence $[T]_{\gamma\leftarrow\gamma}=C_{\gamma\leftarrow\beta}AC_{\beta\leftarrow\gamma}=SAS^{-1}$ by Theorem \ref{thm:two-sided-change-of-basis-for-linear-maps}, with $\beta$ as both original bases and $\gamma$ as both new bases.
\item Let $S\in\operatorname{Mat}(n,n,\mathbb{F})^{\times}$. Choose a basis $\gamma$ satisfying $S=C_{\gamma\leftarrow\beta}$ by Theorem \ref{thm:change-of-basis-matrices}(ii). Then $[T]_{\gamma\leftarrow\gamma}=SAS^{-1}$ by part (i).
\end{enumerate}
\hlt{Structural Commentary}: The proof specializes two-sided transport, then realizes an arbitrary invertible coordinate map.\\
\hlt{Key Insight}: Conjugation records one operator after a simultaneous coordinate change.
\end{proof}
\end{theorem}
