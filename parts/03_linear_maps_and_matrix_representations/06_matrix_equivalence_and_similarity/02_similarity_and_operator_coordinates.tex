\subsection{Similarity and Operator Coordinates}
Throughout this subsection, $m,n\geq1$, and all vector spaces and matrices are over the same fixed field $\mathbb{F}$.

\begin{definition} \label{def:similar-matrices} \hlt{Similarity}\\
Matrices $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$ are similar iff $(\exists\,P\in\operatorname{Mat}(n,n,\mathbb{F})^{\times})\,B=PAP^{-1}$.
\end{definition}

\begin{proposition} \label{prop:similarity-equivalence-relation} \hlt{Similarity Is an Equivalence Relation}\\
Similarity is an equivalence relation on $\operatorname{Mat}(n,n,\mathbb{F})$.
\begin{proof}
Let $A,B,C\in\operatorname{Mat}(n,n,\mathbb{F})$. Similarity is reflexive by Definition \ref{def:similar-matrices}, with $P=I_n$.\\
If $B=PAP^{-1}$ for invertible $P$, then $A=P^{-1}BP$, so symmetry holds by Definition \ref{def:similar-matrices}. If $B=PAP^{-1}$ and $C=QBQ^{-1}$ for invertible $P,Q$, then $C=(QP)A(QP)^{-1}$ and $QP$ is invertible by Corollary \ref{cor:matrix-inverse-product-and-transpose}(i), so transitivity holds by Definition \ref{def:similar-matrices}.\\
\hlt{Structural Commentary}: Identity, inverse, and product witnesses verify reflexivity, symmetry, and transitivity.\\
\hlt{Key Insight}: Conjugation is an orbit relation under invertible matrices.
\end{proof}
\end{proposition}

\begin{remark} \label{rem:similarity-convention} \hlt{Similarity Convention}\\
The forms $PAP^{-1}$ and $M^{-1}AM$ are the same convention with $P=M^{-1}$.\
In Theorem \ref{thm:change-of-basis-formula}, this means choosing opposite coordinate-change directions.
\end{remark}

\begin{corollary} \label{cor:similar-iff-same-linear-operator} \hlt{Similarity as One Operator in Two Bases}\\
For $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ and $B$ are similar.
\item Some $n$-dimensional vector space $\mathcal{V}$, bases $\beta,\gamma$ of $\mathcal{V}$, and $T\in\LL(\mathcal{V})$ satisfy $A=[T]_{\beta\leftarrow\beta}$ and $B=[T]_{\gamma\leftarrow\gamma}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ Suppose $A$ and $B$ are similar. Choose $P\in\operatorname{Mat}(n,n,\mathbb{F})^{\times}$ satisfying $B=PAP^{-1}$ by Definition \ref{def:similar-matrices}.\\
Let $\mathcal{V}\defeq\mathbb{F}^n$, let $\beta$ be its standard basis, and set $T_A(\mathbf{x})\defeq A\mathbf{x}$, which is linear by Proposition \ref{prop:matrix-induced-map-linear}.\\
Then $[T_A]_{\beta\leftarrow\beta}=A$ by Definition \ref{def:linear-map-induced-by-matrix} and Definition \ref{def:matrix-representation-of-linear-map}. Choose a basis $\gamma$ satisfying $P=C_{\gamma\leftarrow\beta}$ by Theorem \ref{thm:change-of-basis-matrices}(ii).\\
Hence $[T_A]_{\gamma\leftarrow\gamma}=PAP^{-1}=B$ by Theorem \ref{thm:change-of-basis-formula}(i).
\item[$(ii)$] $\Rightarrow (i)$ Suppose $A=[T]_{\beta\leftarrow\beta}$ and $B=[T]_{\gamma\leftarrow\gamma}$. Set $P\defeq C_{\gamma\leftarrow\beta}$. Then $P$ is invertible with $P^{-1}=C_{\beta\leftarrow\gamma}$ by Theorem \ref{thm:change-of-basis-matrices}(i), and $B=PAP^{-1}$ by Theorem \ref{thm:change-of-basis-formula}(i). Thus $A$ and $B$ are similar by Definition \ref{def:similar-matrices}.
\end{enumerate}
\hlt{Structural Commentary}: One direction realizes conjugation as basis change; the converse reads the transport formula.\\
\hlt{Key Insight}: Similarity is one operator in two coordinate systems.
\end{proof}
\end{corollary}

\begin{perspective} \hlt{Similarity as Same Operator, New Coordinates}\\
For matrices, similarity means that they represent the same operator in two bases, not merely that their entries are related.\\
This is the structural content behind change of basis.
\end{perspective}

\begin{definition} \label{def:similar-linear-operators} \hlt{Similarity of Operators}\\
For $T\in\LL(\mathcal{V})$ and $S\in\LL(\mathcal{W})$, $T\sim S$ iff $(\exists\,\Phi:\mathcal{V}\xrightarrow{\sim}\mathcal{W})\,S=\Phi T\Phi^{-1}$.\\
When $\mathcal{V}=\mathcal{W}$, the witnesses lie in $\operatorname{Aut}(\mathcal{V})$, so similarity is conjugacy in $\LL(\mathcal{V})$.
\end{definition}

\begin{proposition} \label{prop:operator-similarity-and-common-matrices} \hlt{Operator Similarity and Common Matrix Representatives}\\
Let $\dim\mathcal{V}=\dim\mathcal{W}=n$, $T\in\LL(\mathcal{V})$, and $S\in\LL(\mathcal{W})$.\\
Then $T\sim S$ iff some bases $\beta$ of $\mathcal{V}$ and $\gamma$ of $\mathcal{W}$ satisfy $[T]_{\beta\leftarrow\beta}=[S]_{\gamma\leftarrow\gamma}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $T\sim S$. Choose an isomorphism $\Phi:\mathcal{V}\to\mathcal{W}$ satisfying $S=\Phi T\Phi^{-1}$ by Definition \ref{def:similar-linear-operators}.\\
Choose a basis $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ of $\mathcal{V}$ by Corollary \ref{cor:finite-dimensional-spaces-and-subspaces-have-bases}. Set $\gamma\defeq(\Phi\mathbf{v}_1,\ldots,\Phi\mathbf{v}_n)$, which is a basis of $\mathcal{W}$ by Proposition \ref{prop:isomorphisms-preserve-bases}.\\
The identities $(\forall\,\mathbf{v}\in\mathcal{V})\,[\Phi\mathbf{v}]_{\gamma}=[\mathbf{v}]_{\beta}$ and $S\Phi=\Phi T$ hold by Definition \ref{def:coordinate-vector-and-coordinate-map} and Proposition \ref{prop:linear-map-product-algebra}(ii),(iii), respectively.\\
Consequently, each $j\in\{1,\ldots,n\}$ satisfies $[S(\Phi\mathbf{v}_j)]_{\gamma}=[T\mathbf{v}_j]_{\beta}$. These are the respective $j$th columns of $[S]_{\gamma\leftarrow\gamma}$ and $[T]_{\beta\leftarrow\beta}$ by Definition \ref{def:matrix-representation-of-linear-map}; hence the matrices are equal by Definition \ref{def:matrices-and-matrix-operations}(i).
\item[$(\Leftarrow)$] Suppose bases $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ satisfy $[T]_{\beta\leftarrow\beta}=[S]_{\gamma\leftarrow\gamma}$.\\
Choose $\Phi\in\LL(\mathcal{V},\mathcal{W})$ satisfying $(\forall\,j\in\{1,\ldots,n\})\,\Phi\mathbf{v}_j=\mathbf{w}_j$ by Proposition \ref{prop:linear-map-determined-by-basis}. Its range is $\operatorname{span}(\gamma)=\mathcal{W}$ by Theorem \ref{thm:range-spanned-by-images-of-basis}(i), so $\Phi$ is surjective and hence injective by Corollary \ref{cor:equal-dimension-kernel-range}(ii). Therefore $\Phi$ is an isomorphism by Proposition \ref{prop:linear-map-invertible-iff-bijective}.\\
The identity $(\forall\,\mathbf{v}\in\mathcal{V})\,[\Phi\mathbf{v}]_{\gamma}=[\mathbf{v}]_{\beta}$ holds by Definition \ref{def:coordinate-vector-and-coordinate-map}.\\
Thus $(\forall\,j\in\{1,\ldots,n\})\,[\Phi T\mathbf{v}_j]_{\gamma}=[S\Phi\mathbf{v}_j]_{\gamma}$ by Definition \ref{def:matrix-representation-of-linear-map}, and $(\forall\,j\in\{1,\ldots,n\})\,\Phi T\mathbf{v}_j=S\Phi\mathbf{v}_j$ by Proposition \ref{prop:coordinate-map-isomorphism}(ii).\\
The compositions are linear by Proposition \ref{prop:linear-map-product-algebra}(i), so $\Phi T=S\Phi$ by Proposition \ref{prop:linear-map-determined-by-basis}.\\
Thus $S=\Phi T\Phi^{-1}$ by Proposition \ref{prop:linear-map-product-algebra}(ii),(iii), so $T\sim S$ by Definition \ref{def:similar-linear-operators}.
\end{enumerate}
\hlt{Structural Commentary}: A conjugating isomorphism transports a basis; a shared matrix reconstructs the isomorphism.\\
\hlt{Key Insight}: Matching basis actions force operator conjugacy.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:similarity-preserves-scalar-shifts} \hlt{Similarity and Scalar Shifts}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A$ and $B$ are similar, then $(\forall\,\lambda\in\mathbb{F})\,[A-\lambda I_n\text{ is similar to }B-\lambda I_n]$.
\item If $(\exists\,\lambda\in\mathbb{F})\,[A-\lambda I_n\text{ is similar to }B-\lambda I_n]$, then $A$ and $B$ are similar.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $P\in\operatorname{Mat}(n,n,\mathbb{F})^{\times}$ satisfying $B=PAP^{-1}$ by the hypothesis and Definition \ref{def:similar-matrices}.\\
For $\lambda\in\mathbb{F}$, $B-\lambda I_n=PAP^{-1}-\lambda PP^{-1}=P(A-\lambda I_n)P^{-1}$, so the shifted matrices are similar by Definition \ref{def:similar-matrices}.
\item Choose $\lambda\in\mathbb{F}$ and $P\in\operatorname{Mat}(n,n,\mathbb{F})^{\times}$ satisfying $B-\lambda I_n=P(A-\lambda I_n)P^{-1}$ by the hypothesis and Definition \ref{def:similar-matrices}.\\
After adding $\lambda I_n=\lambda PP^{-1}$, the equality becomes $B=PAP^{-1}$, so $A$ and $B$ are similar by Definition \ref{def:similar-matrices}.
\end{enumerate}
\hlt{Structural Commentary}: Both directions add or remove the central term $\lambda I_n$ inside a conjugation.\\
\hlt{Key Insight}: Scalar identity shifts commute with every basis change.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:upper-block-triangular-similarity-shift} \hlt{Upper Block-Triangular Similarity Shift}\\
Let $B\in\operatorname{Mat}(m,m,\mathbb{F})$, $D\in\operatorname{Mat}(n,n,\mathbb{F})$,  $C,X\in\operatorname{Mat}(m,n,\mathbb{F})$. Set $M\defeq\left(\begin{smallmatrix}B&C\\0&D\end{smallmatrix}\right)$, $N\defeq\left(\begin{smallmatrix}B&C+XD-BX\\0&D\end{smallmatrix}\right)$. Then $M$ and $N$ are similar.
\begin{proof}
Set $U\defeq\left(\begin{smallmatrix}I_m&X\\0&I_n\end{smallmatrix}\right)$. By block multiplication, $U^{-1}=\left(\begin{smallmatrix}I_m&-X\\0&I_n\end{smallmatrix}\right)$ and $UMU^{-1}=\left(\begin{smallmatrix}B&-BX+C+XD\\0&D\end{smallmatrix}\right)=N$.\\
Thus $M$ and $N$ are similar by Definition \ref{def:similar-matrices}.\\
\hlt{Structural Commentary}: Block multiplication verifies the inverse and computes the conjugated coupling block.\\
\hlt{Key Insight}: Unit upper conjugation preserves diagonal blocks and shears the coupling.
\end{proof}
\end{proposition}
