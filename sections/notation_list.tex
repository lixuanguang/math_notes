\notationgroup{Collision and Scope Rules}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\text{bare }A,B,\ldots;\ \mathcal{A},\mathcal{B},\ldots}{bare Latin and calligraphic letters are local unless a decorated form is listed below; local definitions inside a theorem, proof, or subsection take precedence}
\notationentry{B,\ B_r(x),\ \overline{B}_r(x),\ \mathrm{B}(x,y),\ \mathcal{B}(\R^n)}{matrix or set variable / open ball / closed ball / beta function / Borel $\sigma$-algebra; font and argument determine the meaning}
\notationentry{I_{\mathcal{V}},\ I_n,\ I^k,\ I_R}{identity operator, identity matrix, $k$-cell, and power-series interval}
\notationentry{P,\ \mathcal{P}(X),\ \mathscr{P},\ P_{\mathcal{U}},\ P_j}{locally declared projection or map / power set / umbral polynomial algebra / orthogonal projection / spectral projection}
\notationentry{\mathcal{F},\ \mathscr{F},\ \mathcal{F}(K),\ \mathcal{F}(\mathcal{V})}{local family of functions or operators / umbral formal-series algebra / finite subsets of $K$ / finite-rank operators}
\notationentry{\LL(\mathcal{V},\mathcal{W}),\ \mathcal{L}^n,\ \mathscr{L}^1,\ \operatorname{LS}(A,\mathbf{y}),\ \mathcal{L}(\theta,\alpha)}{linear maps / Lebesgue measurable sets / integrable functions / least-squares solution set / local Lagrangian or loss notation}
\notationentry{R(f,\Pi,\{c_i\}),\ R,\ R_k(A),\ A=QR}{Riemann sum, locally declared radius or scalar, row-sum quantity, and QR factorization; the argument pattern disambiguates}
\notationentry{\Delta x_i,\ \Delta A,\ \Delta(A),\ \Delta(\mathcal{V}),\ \Delta_{k-1}}{increment, matrix perturbation, defect from normality, metric discriminant, and coefficient simplex}
\notationentry{\Omega,\ \Omega_T,\ \Omega_n,\ \Omega(\theta),\ \operatorname{vol}(\Omega)}{local domain or set / finite spectrum of $T$ / primitive roots of unity or DFT diagonal by context / learning penalty / volume domain}
\notationentry{Q^k,\ Q_n,\ \Phi_n}{standard $k$-simplex / normalized Fourier matrix / cyclotomic polynomial; the old cyclotomic $Q_n$ notation is avoided}
\notationentry{\overline{z},\ \overline{E},\ \overline{\mathcal{A}},\ A^*,\ T^\ast\omega,\ f',\ \mathcal{V}'}{conjugation, closure, uniform closure, adjoint or conjugate transpose, pullback, derivative, and dual space; bars, stars, and primes are object-sensitive}
\end{tabularx}
