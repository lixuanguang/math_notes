\begin{principle} \label{princ:analysis-language-as-structure} \hlt{Language Carries Structure}\\
Objects, hypotheses, conclusions, and quantifier order matter. For instance,
\begin{align}
\forall x\in X\ \exists y\in Y\ P(x,y), \qquad
\exists y\in Y\ \forall x\in X\ P(x,y) \nonumber
\end{align}
are different: in the first $y$ may depend on $x$; in the second one $y$ works for all $x$.
\end{principle}

\begin{principle} \label{princ:well-formed-statements-and-bound-variables} \hlt{Well-Formed Statements and Bound Variables}\\
A mathematical statement is well formed only after every symbol has a declared role and every expression has a defined meaning. A variable introduced by a quantifier, summation sign, integral, set-builder condition, or similar construction is \hlt{bound} inside that construction; changing its name does not change the statement as long as its scope is unchanged and no free variable is captured.
\end{principle}

\begin{principle} \label{princ:mathematical-induction} \hlt{Mathematical Induction}\\
Let $P_1,P_2,\ldots$ be statements indexed by positive integers. To prove $P_n$ for every $n\in\N$, it is enough to prove:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{basis}: $P_1$ is true;
\item \hlt{induction step}: for every $n\in\N$, $P_n$ implies $P_{n+1}$.
\end{enumerate}
The induction step is an implication; it does not prove $P_{n+1}$ in isolation.\\
The basis starts the chain, and the implication carries truth from one statement to the next.
\end{principle}

\begin{method} \label{meth:proof-grammar-and-negation} \hlt{Proof Grammar and Negation}\\
In theorem statements, put the hypotheses before the conclusion and keep quantifiers in the order meant.\\
In particular, switching $\forall$ and $\exists$ usually changes the statement; see Principle \ref{princ:analysis-language-as-structure}.\\
Before applying a rule, also check that the displayed expressions are well formed; an undefined expression makes the sentence ill formed, not merely false.\\
The word ``or'' is inclusive in mathematics, and the basic negation rules are
\begin{align}
\neg(a\land b)\Longleftrightarrow \neg a\lor\neg b, \qquad \neg(a\lor b)\Longleftrightarrow \neg a\land\neg b, \qquad \neg(a\Rightarrow b)&\Longleftrightarrow a\land\neg b \nonumber
\end{align}
Thus a proof by contradiction of $a\Rightarrow b$ assumes both $a$ and $\neg b$ and derives an impossibility.\\
\hlt{Key Insight}: Before proving or negating a statement, parse its quantifiers and logical connectives.
\end{method}

\begin{definition} \label{def:finite-sums-indexed-by-integers} \hlt{Finite Sums Indexed by Integers}\\
Let $a_i\in\C$ be assigned for integers $m\leq i\leq n$. Define $\sum\limits_{i=m}^{n}a_i\defeq a_m+a_{m+1}+\cdots+a_n$.\\
If $m>n$, the sum over the empty index set is $0$.
\end{definition}

\begin{proposition} \label{prop:finite-sum-basic-laws} \hlt{Basic Laws for Finite Sums}\\
Let $a_i,b_i\in\C$ and $c\in\C$ be assigned for $m\leq i\leq n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $m\leq r<n$, then $\sum\limits_{i=m}^{n}a_i=\sum\limits_{i=m}^{r}a_i+\sum\limits_{i=r+1}^{n}a_i$.
\item $\sum\limits_{i=m}^{n}(a_i+b_i)=\sum\limits_{i=m}^{n}a_i+\sum\limits_{i=m}^{n}b_i$ and $\sum\limits_{i=m}^{n}ca_i=c\sum\limits_{i=m}^{n}a_i$.
\item $\abs{\sum\limits_{i=m}^{n}a_i}\leq\sum\limits_{i=m}^{n}\abs{a_i}$.
\item If $a_i,b_i\in\R$ and $a_i\leq b_i$ for every $i$, then $\sum\limits_{i=m}^{n}a_i\leq\sum\limits_{i=m}^{n}b_i$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Definition \ref{def:finite-sums-indexed-by-integers} and associativity of addition in $\C$,
$\sum\limits_{i=m}^{n}a_i=\sum\limits_{i=m}^{r}a_i+\sum\limits_{i=r+1}^{n}a_i$.
\item This follows from associativity, commutativity, and distributivity in the field $\C$.
\item Induct on $n-m$. The case $n=m$ is immediate.\\
If the claim holds up to $n$, then the triangle inequality gives
\[
\abs{\sum\limits_{i=m}^{n+1}a_i}
\leq \abs{\sum\limits_{i=m}^{n}a_i}+\abs{a_{n+1}}
\leq \sum\limits_{i=m}^{n}\abs{a_i}+\abs{a_{n+1}}
=\sum\limits_{i=m}^{n+1}\abs{a_i}.
\]
\item Induct on $n-m$. The case $n=m$ is the hypothesis $a_m\leq b_m$.\\
If the claim holds up to $n$, then compatibility of the order on $\R$ with addition gives
\[
\sum\limits_{i=m}^{n+1}a_i
=\sum\limits_{i=m}^{n}a_i+a_{n+1}
\leq \sum\limits_{i=m}^{n}b_i+b_{n+1}
=\sum\limits_{i=m}^{n+1}b_i.
\]
\end{enumerate}
\hlt{Structural Commentary}: Finite sums are iterated addition, so their laws are inherited from the scalar field.\\
\hlt{Key Insight}: No convergence hypothesis is needed for a finite sum.
\end{proof}
\end{proposition}

\begin{definition} \label{def:analysis-sets-and-operations} \hlt{Sets and Operations}\\
A \hlt{set} is a collection. Write $x\in E$, $x\notin E$, $\emptyset$, and $A\subseteq B$; $A=B$ iff $A\subseteq B$ and $B\subseteq A$. For $A,B\subseteq X$,
\begin{align}
A\cup B &\defeq \{x\in X:x\in A\text{ or }x\in B\}, \nonumber\\
A\cap B &\defeq \{x\in X:x\in A\text{ and }x\in B\}, \nonumber\\
A\setminus B &\defeq \{x\in X:x\in A\text{ and }x\notin B\}, \nonumber\\
A^c &\defeq X\setminus A. \nonumber
\end{align}
The \hlt{power set} is $\mathcal{P}(X)\defeq\{A:A\subseteq X\}$.
\end{definition}

\begin{definition} \label{def:indexed-families-unions-intersections} \hlt{Indexed Families of Sets}\\
An \hlt{indexed family} $\{A_i\}_{i\in I}$ is one set $A_i\subseteq X$ for each index $i\in I$. Define
\begin{align}
\bigcup_{i\in I}A_i&\defeq\{x\in X:x\in A_i\text{ for some }i\in I\}, \nonumber\\
\bigcap_{i\in I}A_i&\defeq\{x\in X:x\in A_i\text{ for every }i\in I\}. \nonumber
\end{align}
\hlt{Key Insight}: Union means ``some index''; intersection means ``every index''.
\end{definition}

\begin{definition} \label{def:cartesian-product} \hlt{Cartesian Product}\\
For sets $X$ and $Y$, the \hlt{Cartesian product} is $X\times Y\defeq\{(x,y):x\in X,\ y\in Y\}$.\\
The order matters: the first coordinate comes from $X$, and the second comes from $Y$.
\end{definition}

\begin{definition} \label{def:relations-and-equivalence} \hlt{Relations and Equivalence}\\
A \hlt{relation} from $X$ to $Y$ is a subset $R\subseteq X\times Y$. If $(x,y)\in R$, one often writes $xRy$.\\
A relation $\sim$ on $X$ is an \hlt{equivalence relation} if, for all $x,y,z\in X$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $x\sim x$;
\item $x\sim y$ implies $y\sim x$;
\item $x\sim y$ and $y\sim z$ imply $x\sim z$.
\end{enumerate}
The equivalence class of $x$ is $[x]\defeq\{y\in X:y\sim x\}$.
\end{definition}

\begin{principle} \label{princ:axiom-of-choice-analysis} \hlt{Choice Principle}\\
For nonempty sets $\{X_i\}_{i\in I}$, the axiom of choice asserts $\exists c:I\to\bigcup\limits_{i\in I}X_i$ such that $c(i)\in X_i\ \forall i\in I$.\\
Use direct construction for finite or explicit choices; use choice only for arbitrary simultaneous choices.
\end{principle}

\begin{definition} \label{def:ordered-set-vocabulary} \hlt{Ordered Set Vocabulary}\\
Let $S$ be a set, let $E,C\subseteq S$, and let $\leq_S$ be a relation on $S$. Write $x<_Sy$ iff $x\leq_Sy$ and $x\neq y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\leq_S$ is a \hlt{partial order} if it is reflexive, antisymmetric, and transitive; then $(S,\leq_S)$ is a \hlt{poset}.
\item $\leq_S$ is an \hlt{order} if it is a partial order and any $x,y\in S$ are comparable; then $(S,\leq_S)$ is an \hlt{ordered set}.
\item $C$ is a \hlt{chain} if $\leq_S|_C$ is an order.
\item $S$ is \hlt{well-ordered} if every nonempty $Y\subseteq S$ has $\ell\in Y$ with $\ell\leq_Sy$ for all $y\in Y$.
\item $\beta\in S$ is an \hlt{upper bound} of $E$ if $x\leq_S\beta$ for all $x\in E$; lower bounds reverse the inequality.
\item $E$ is \hlt{bounded above} if it has an upper bound; bounded below is dual.
\item $\sup E$ is the least upper bound of $E$, and $\inf E$ is the greatest lower bound of $E$, when these exist.
\item $m\in E$ is \hlt{maximal} if no $x\in E$ satisfies $m<_Sx$; minimal is dual.
\item $S$ has the \hlt{least-upper-bound property} if every nonempty bounded-above $E\subseteq S$ has $\sup E\in S$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:lattice-vocabulary} \hlt{Lattice Vocabulary}\\
Let $(L,\leq)$ be a partially ordered set.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{meet} of $x,y\in L$ is $\inf\{x,y\}$ when it exists, often written $x\wedge y$.
\item A \hlt{join} of $x,y\in L$ is $\sup\{x,y\}$ when it exists, often written $x\vee y$.
\item A \hlt{lattice} is a partially ordered set in which every two elements have a meet and a join.
\item A lattice is \hlt{complete} if every subset $E\subseteq L$ has an infimum and a supremum.
\item A \hlt{bottom} element is an element $\bot\in L$ with $\bot\leq x$ for every $x\in L$; a \hlt{top} element is an element $\top\in L$ with $x\leq\top$ for every $x\in L$.
\item A lattice is \hlt{modular} if $x\leq z$ implies $z\wedge(y\vee x)=(z\wedge y)\vee x$ for all $x,y,z\in L$.
\end{enumerate}
\end{definition}

\begin{principle} \label{princ:zorns-lemma} \hlt{Zorn's Lemma}\\
If a nonempty partially ordered set $X$ has the property that every chain in $X$ has an upper bound in $X$, then $X$ contains a maximal element. This is equivalent in strength to the axiom of choice, Principle \ref{princ:axiom-of-choice-analysis}.
\end{principle}

\begin{definition} \label{def:functions-images-preimages} \hlt{Functions, Images, and Preimages}\\
Let $f:X\to Y$ be a \hlt{function}, assigning each $x\in X$ a unique $f(x)\in Y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $X=\operatorname{dom}(f)$ is the \hlt{domain} and $Y$ is the \hlt{codomain};
\item $\operatorname{range}(f)\defeq f(X)\defeq\{f(x):x\in X\}$ is the \hlt{range} of $f$;
\item for $A\subseteq X$, $f(A)\defeq\{f(x):x\in A\}$ is the \hlt{image} of $A$;
\item for $B\subseteq Y$, $f^{-1}(B)\defeq\{x\in X:f(x)\in B\}$ is the \hlt{preimage} of $B$; no inverse function is required;
\item for $A\subseteq X$, $f|_A:A\to Y$ is the \hlt{restriction} of $f$ to $A$;
\item if $X\subseteq U$, an \hlt{extension} of $f$ to $U$ is $\widetilde f:U\to Y$ with $\widetilde f|_X=f$;
\item if $Y$ has distinguished $0$, $\operatorname{supp}(f)\defeq\{x\in X:f(x)\neq0\}$;
\item if $g:Y\to Z$, then $g\circ f:X\to Z$ is given by $(g\circ f)(x)=g(f(x))$;
\item $\operatorname{id}_X:X\to X$ denotes the \hlt{identity function}.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:injective-surjective-bijective} \hlt{Injective, Surjective, and Bijective Maps}\\
Let $f:X\to Y$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $f$ is \hlt{injective} if $f(x_1)=f(x_2)$ implies $x_1=x_2$.
\item $f$ is \hlt{surjective} if $\operatorname{range}(f)=Y$.
\item $f$ is \hlt{bijective} if it is both injective and surjective.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:bijective-iff-invertible-function} \hlt{Bijective Means Invertible}\\
A function $f:X\to Y$ is bijective iff there exists a function $g:Y\to X$ s.t. $g\circ f=\operatorname{id}_X$, $f\circ g=\operatorname{id}_Y$.\\
When such a function exists, it is unique and is denoted $f^{-1}:Y\to X$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $f$ is bijective, define $g(y)$ as the unique $x$ with $f(x)=y$. Then $g\circ f=\operatorname{id}_X$ and $f\circ g=\operatorname{id}_Y$.
\item[$(\Leftarrow)$] If such a $g$ exists, applying $g$ to $f(x_1)=f(x_2)$ gives $x_1=x_2$, and $y=f(g(y))$ gives surjectivity.
\end{enumerate}
If $g$ and $h$ are two-sided inverses, then $g(y)=g(f(h(y)))=h(y)$, so the inverse is unique.\\
\hlt{Structural Commentary}: A true inverse undoes $f$ on both sides.\\
\hlt{Key Insight}: Bijective means exactly one preimage per output.
\end{proof}
\end{theorem}

\begin{definition} \label{def:cardinality-equivalence} \hlt{Same Cardinality}\\
Two sets $A$ and $B$ have \hlt{same cardinality}, or are \hlt{equivalent}, if $\exists$ a bijection $f:A\to B$. Then $A\sim_{\#}B$.\\
The relation $\sim_{\#}$ is an equivalence relation: $A\sim_{\#}A$ by identity map, symmetry follows from Theorem \ref{thm:bijective-iff-invertible-function}, and transitivity follows by composing bijections.
\end{definition}

\begin{definition} \label{def:finite-countable-uncountable} \hlt{Finite, Countable, and Uncountable}\\
Set $J_0\defeq\emptyset$, and for $n\geq1$ set $J_n\defeq\{1,2,\ldots,n\}$. A set $A$ is
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{finite} if $A\sim_{\#} J_n$ for some $n\in\N$;
\item \hlt{infinite} if it is not finite;
\item \hlt{countable} if $A\sim_{\#} \N$;
\item \hlt{uncountable} if it is neither finite nor countable;
\item \hlt{at most countable} if it is finite or countable.
\end{enumerate}
Countable sets are also called enumerable or denumerable.
\end{definition}

\begin{principle} \label{princ:well-ordering-natural-numbers} \hlt{Well-Ordering of $\N$}\\
Every nonempty subset of $\N$ has a least element, justifying first unused indices, first repeated values, and increasing listings of infinite subsets of $\N$.
\end{principle}

\begin{theorem} \label{thm:cantor-power-set-theorem} \hlt{Cantor's Theorem}\\
For every set $X$, no function $f:X\to\mathcal{P}(X)$ is surjective. Hence $X$ and $\mathcal{P}(X)$ cannot have the same cardinality.
\begin{proof}
Suppose $f:X\to\mathcal{P}(X)$ were surjective, and set $D\defeq\{x\in X:x\notin f(x)\}$.\\
Since $D\subseteq X$, we have $D\in\mathcal{P}(X)$, so surjectivity gives $D=f(a)$ for some $a\in X$.\\
But then $a\in D \Longleftrightarrow a\notin f(a) \Longleftrightarrow a\notin D$, a contradiction.\\
\hlt{Structural Commentary}: The diagonal set is defined to disagree with each proposed row at its own index.\\
\hlt{Key Insight}: Power sets are too large to be exhausted by their base set.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:schroeder-bernstein} \hlt{Schroeder--Bernstein Theorem}\\
If there are injections $f:A\to B$ and $g:B\to A$, then there is a bijection $h:A\to B$.
\begin{proof}
Let $A_0\defeq A\setminus g(B)$, and recursively define $B_n\defeq f(A_n)$ and $A_{n+1}\defeq g(B_n)$.\\
The sets $A_n$ are pairwise disjoint: if $A_i\cap A_j\neq\emptyset$ with $i<j$, then $i=0$ already contradicts $A_j\subseteq g(B)$; if $i>0$, injectivity of $g$ and then $f$ steps backward to $A_{i-1}\cap A_{j-1}\neq\emptyset$, and iteration gives the same contradiction.\\
Since $f$ is injective, the sets $B_n=f(A_n)$ are pairwise disjoint as well.\\
Let $A_*\defeq A\setminus\bigcup\limits_{n\geq0}A_n$ and $B_*\defeq B\setminus\bigcup\limits_{n\geq0}B_n$. The map $g$ restricts to a bijection $B_*\to A_*$.\\
Indeed, $g(B_*)$ avoids every $A_n$, while if $a\in A_*$ then $a\notin A_0$, so $a=g(b)$ for a unique $b\in B$; this $b$ cannot lie in any $B_n$, otherwise $a\in A_{n+1}$.\\
Define $h(x)\defeq
\begin{cases}
f(x), & x\in\bigcup_{n\geq0}A_n,\\
g^{-1}(x), & x\in A_*,
\end{cases}$, where $g^{-1}$ denotes the inverse of $g:B_*\to A_*$.\\
This maps each $A_n$ bijectively onto $B_n$ and maps $A_*$ bijectively onto $B_*$. Hence $h$ is a bijection $A\to B$.\\
\hlt{Structural Commentary}: Two injections carve both sets into matched layers plus a common remainder.\\
\hlt{Key Insight}: Use $f$ on the escaping layers and $g^{-1}$ on the stable remainder.
\end{proof}
\end{theorem}

\begin{definition} \label{def:cardinal-comparison-and-arithmetic} \hlt{Cardinal Comparison and Arithmetic}\\
Let $A,B$ be sets, and let $\kappa=\abs{A}$, $\lambda=\abs{B}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Cardinal comparison}: $\abs{A}\leq\abs{B}$ iff $\exists$ an injection $A\to B$.
\item \hlt{Strict cardinal comparison}: $\abs{A}<\abs{B}$ iff $\abs{A}\leq\abs{B}$ and $A\not\sim_{\#}B$.
\item If $A\cap B=\emptyset$, the \hlt{cardinal sum} is $\kappa+\lambda\defeq\abs{A\cup B}$.
\item The \hlt{cardinal product} is $\kappa\lambda\defeq\abs{A\times B}$.
\item The \hlt{function set} is $A^B\defeq\{f:B\to A\}$, and the \hlt{cardinal power} is $\kappa^\lambda\defeq\abs{A^B}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:cardinal-arithmetic-reference-laws} \hlt{Cardinal Arithmetic Reference Laws}\\
Let $\kappa,\lambda,\mu$ be cardinal numbers.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\kappa+\lambda=\lambda+\kappa$ and $\kappa\lambda=\lambda\kappa$;
\item $\kappa+(\lambda+\mu)=(\kappa+\lambda)+\mu$ and $\kappa(\lambda\mu)=(\kappa\lambda)\mu$;
\item $\kappa(\lambda+\mu)=\kappa\lambda+\kappa\mu$;
\item $\kappa^{\lambda+\mu}=\kappa^\lambda\kappa^\mu$, $(\kappa^\lambda)^\mu=\kappa^{\lambda\mu}$, and $(\kappa\lambda)^\mu=\kappa^\mu\lambda^\mu$.
\end{enumerate}
\begin{proof}
Choose pairwise disjoint sets $A,B,C$ with $\abs{A}=\kappa$, $\abs{B}=\lambda$, and $\abs{C}=\mu$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Use $A\cup B\cong B\cup A$ and $A\times B\cong B\times A$.
\item Use $A\cup(B\cup C)\cong(A\cup B)\cup C$ and $A\times(B\times C)\cong(A\times B)\times C$.
\item Use $A\times(B\cup C)\cong(A\times B)\cup(A\times C)$.
\item Use $A^{B\cup C}\cong A^B\times A^C$, $(A^B)^C\cong A^{B\times C}$, and $(A\times B)^C\cong A^C\times B^C$.
\end{enumerate}
Taking cardinalities gives (i)--(iv).\\
\hlt{Structural Commentary}: The laws are canonical bijections.\\
\hlt{Key Insight}: Cardinal arithmetic is set construction up to bijection.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:infinite-cardinal-absorption-reference} \hlt{Infinite Cardinal Absorption}\\
Let $\kappa,\lambda$ be cardinals and set $\mu\defeq\max\{\kappa,\lambda\}$. If $\mu$ is infinite, then $\kappa+\lambda=\mu$.\\
If also $\kappa,\lambda>0$, then $\kappa\lambda=\mu$. In particular, $\aleph_0+\aleph_0=\aleph_0\cdot\aleph_0=\aleph_0$.
\begin{proof}
For every infinite cardinal $\nu$, the standard square law gives $\nu+\nu=\nu^2=\nu$.\\
Since $\kappa,\lambda\leq\mu$, $\mu\leq \kappa+\lambda\leq\mu+\mu=\mu$, so $\kappa+\lambda=\mu$.\\
If $\kappa,\lambda>0$, choose nonempty representatives $\abs{A}=\kappa$ and $\abs{B}=\lambda$; fixing one point in the smaller factor injects the larger factor into $A\times B$, so $\mu\leq\kappa\lambda$. Also $\kappa\lambda\leq\mu^2=\mu$, so $\kappa\lambda=\mu$.\\
\hlt{Structural Commentary}: Infinite cardinals absorb smaller summands and nonzero factors.\\
\hlt{Key Insight}: Squeeze between $\mu$ and $\mu^2=\mu$.
\end{proof}
\end{theorem}
