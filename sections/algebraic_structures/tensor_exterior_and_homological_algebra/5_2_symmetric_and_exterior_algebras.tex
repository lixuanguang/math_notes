\subsection{Symmetric and Exterior Algebras}

% This subsection develops the iterated, symmetric, exterior, and determinant
% material that follows the bilinear tensor-product core in the source.

The previous subsection shows that tensor products are the natural place where bilinear maps become linear. The same idea extends to maps with more variables, and then to tensor spaces, symmetric tensors, exterior tensors, and the determinant.

\begin{definition} \label{def:multilinear-map}\label{def:iterated-tensor-product-universal-property} \hlt{Multilinear Maps and Iterated Tensor Products}\\
Let $\mathcal{V}_1,\ldots,\mathcal{V}_n,\mathcal{W}$ be $\mathbb{F}$-vector spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A map $f:\mathcal{V}_1\times\cdots\times\mathcal{V}_n\to\mathcal{W}$ is \hlt{multilinear}, or \hlt{$n$-linear}, if for each $k$, the map $\mathbf{v}_k\mapsto f(\mathbf{v}_1,\ldots,\mathbf{v}_k,\ldots,\mathbf{v}_n)$ is linear with the other entries fixed.
\item \hlt{$\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathcal{W})$}$\defeq\{f:\mathcal{V}_1\times\cdots\times\mathcal{V}_n\to\mathcal{W}:f\text{ multilinear}\}$.
\item If $\mathcal{W}=\mathbb{F}$, elements of $\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathbb{F})$ are \hlt{multilinear forms}.
\item A pair $(\mathcal{T},t)$ is an \hlt{iterated tensor product} of $\mathcal{V}_1,\ldots,\mathcal{V}_n$ if $t\in\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathcal{T})$ and, for every $\mathbb{F}$-vector space $\mathcal{Y}$ and $f\in\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathcal{Y})$, $\exists!\widetilde{f}\in\LL(\mathcal{T},\mathcal{Y})$ such that $f=\widetilde{f}\circ t$.
\item Write $\mathcal{T}=\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$ and $\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n\defeq t(\mathbf{v}_1,\ldots,\mathbf{v}_n)$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:basic-multilinear-maps} \hlt{Basic Multilinear Maps}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A$ is an associative $\mathbb{F}$-algebra, then $m_n:A^n\to A$, $m_n(a_1,\ldots,a_n)\defeq a_1\cdots a_n$, is $n$-linear.
\item If $\dim\mathcal{V}=n$, then $\det:\mathcal{V}^n\to\mathbb{F}$, applied to coordinate columns in any basis, is alternating $n$-linear form.
\end{enumerate}
\end{example}

\begin{theorem} \label{thm:iterated-tensor-products-basic-laws} \hlt{Basic Laws for Iterated Tensor Products}\\
Let $\mathcal{V}_1,\ldots,\mathcal{V}_n,\mathcal{V},\mathcal{W}$ be $\mathbb{F}$-vector spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{Mult}_{\mathbb{F}}(\mathcal{V}_1,\ldots,\mathcal{V}_n;\mathcal{W})\cong\LL(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n,\mathcal{W})$ naturally.
\item For $1\leq r<n$, the canonical $\alpha_r$ is $\alpha_r:(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_r)\otimes(\mathcal{V}_{r+1}\otimes\cdots\otimes\mathcal{V}_n)\cong\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$,\\
where $\alpha_r((\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_r)\otimes(\mathbf{v}_{r+1}\otimes\cdots\otimes\mathbf{v}_n))=\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n$.
\item For $\pi\in S_n$, the canonical $\tau_\pi$ is $\tau_\pi:\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n\cong\mathcal{V}_{\pi(1)}\otimes\cdots\otimes\mathcal{V}_{\pi(n)}$,\\
where $\tau_\pi(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n)=\mathbf{v}_{\pi(1)}\otimes\cdots\otimes\mathbf{v}_{\pi(n)}$.
\item $\lambda:\mathbb{F}\otimes\mathcal{V}\cong\mathcal{V}$, $\lambda(r\otimes\mathbf{v})=r\mathbf{v}$, and $\rho:\mathcal{V}\otimes\mathbb{F}\cong\mathcal{V}$, $\rho(\mathbf{v}\otimes r)=r\mathbf{v}$.
\item If $\dim\mathcal{V}_i<\infty$ for all $i$, then $\dim(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n)=\prod_{i=1}^{n}\dim(\mathcal{V}_i)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $t:\mathcal{V}_1\times\cdots\times\mathcal{V}_n\to\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$ be canonical. Define $\Phi(f)=\widetilde{f}$, where $\widetilde{f}$ is the unique linear map satisfying $\widetilde{f}(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n)=f(\mathbf{v}_1,\ldots,\mathbf{v}_n)$; conversely define $\Psi(T)=T\circ t$. Then $\Psi\circ\Phi$ and $\Phi\circ\Psi$ are identity maps, so $\Phi$ is a linear bijection.
\item Set $A_r=(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_r)\otimes(\mathcal{V}_{r+1}\otimes\cdots\otimes\mathcal{V}_n)$ and $B=\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$. The rules defining $\alpha_r:A_r\to B$ and $\beta_r:B\to A_r$, $\beta_r(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n)=(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_r)\otimes(\mathbf{v}_{r+1}\otimes\cdots\otimes\mathbf{v}_n)$, are induced by multilinearity; $\alpha_r\circ\beta_r$ and $\beta_r\circ\alpha_r$ fix the spanning decomposable tensors, hence are identities.
\item Set $C_\pi=\mathcal{V}_{\pi(1)}\otimes\cdots\otimes\mathcal{V}_{\pi(n)}$. The map $p_\pi:\mathcal{V}_1\times\cdots\times\mathcal{V}_n\to C_\pi$, $p_\pi(\mathbf{v}_1,\ldots,\mathbf{v}_n)=\mathbf{v}_{\pi(1)}\otimes\cdots\otimes\mathbf{v}_{\pi(n)}$, is multilinear, so the universal property induces the unique $\tau_\pi$ with $\tau_\pi\circ t=p_\pi$.\\
The map $\sigma_\pi:C_\pi\to\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n$, $\sigma_\pi(\mathbf{v}_{\pi(1)}\otimes\cdots\otimes\mathbf{v}_{\pi(n)})=\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_n$, is induced similarly; $\sigma_\pi\tau_\pi$ and $\tau_\pi\sigma_\pi$ fix decomposable spanning tensors, hence are identities.
\item The maps $m_L:\mathbb{F}\times\mathcal{V}\to\mathcal{V}$, $m_L(r,\mathbf{v})=r\mathbf{v}$, and $m_R:\mathcal{V}\times\mathbb{F}\to\mathcal{V}$, $m_R(\mathbf{v},r)=r\mathbf{v}$, are bilinear, hence induce $\lambda$ and $\rho$. Since $\lambda(1\otimes\mathbf{v})=\mathbf{v}$ and $1\otimes r\mathbf{v}=r\otimes\mathbf{v}$, $\lambda^{-1}(\mathbf{v})=1\otimes\mathbf{v}$.\\
Since $\rho(\mathbf{v}\otimes1)=\mathbf{v}$ and $r\mathbf{v}\otimes1=\mathbf{v}\otimes r$, $\rho^{-1}(\mathbf{v})=\mathbf{v}\otimes1$.
\item Choose finite bases $B_i$ of $\mathcal{V}_i$ and set $\mathcal{B}=\{\mathbf{b}_1\otimes\cdots\otimes\mathbf{b}_n:\mathbf{b}_i\in B_i\}$. By repeated application of Theorem \ref{thm:tensor-bases-dimension-and-rank}, $\mathcal{B}$ is a basis; hence $\dim(\mathcal{V}_1\otimes\cdots\otimes\mathcal{V}_n)=\#\mathcal{B}=\prod_{i=1}^{n}\#B_i=\prod_{i=1}^{n}\dim\mathcal{V}_i$.
\end{enumerate}
\hlt{Structural Commentary}: Tensor parenthesising and ordering are canonical bookkeeping.\\
\hlt{Key Insight}: Universal properties extend pure-tensor rules uniquely.
\end{proof}
\end{theorem}

\begin{definition} \label{def:tensor-spaces-type-pq} \hlt{Tensor Spaces of Type $(p,q)$}\\
Let $\dim\mathcal{V}<\infty$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $(p,q)\in\mathbb{Z}_{\geq0}^{2}\setminus\{(0,0)\}$, with empty tensor powers equal to $\mathbb{F}$, $\mathcal{T}^{p}_{q}(\mathcal{V})\defeq\mathcal{V}^{\otimes p}\otimes(\mathcal{V}')^{\otimes q}$.
\item $\mathcal{T}^{p}_{q}(\mathcal{V})$ is the \hlt{tensor space of type $(p,q)$}; $p$ is the \hlt{contravariant type}, $q$ the \hlt{covariant type}.
\item $\mathcal{T}^{0}_{0}(\mathcal{V})\defeq\mathbb{F}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:tensor-spaces-as-multilinear-functionals} \hlt{Tensor Spaces as Multilinear Functionals}\\
Let $\dim\mathcal{V}=n<\infty$ and $p,q\in\mathbb{Z}_{\geq0}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\dim\mathcal{T}^{p}_{q}(\mathcal{V})=n^{p+q}$.
\item Let $\iota:\mathcal{V}\to\mathcal{V}''$, $\iota(\mathbf{v})(\eta)=\eta(\mathbf{v})$. Via $\iota$, natural isomorphism is $\Theta_{p,q}:\mathcal{T}^{p}_{q}(\mathcal{V})\xrightarrow{\sim}\operatorname{Mult}_{\mathbb{F}}((\mathcal{V}')^{\times p}\times\mathcal{V}^{\times q};\mathbb{F})$;\\
where $\Theta_{p,q}(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_p\otimes\varphi_1\otimes\cdots\otimes\varphi_q)(\eta_1,\ldots,\eta_p,\mathbf{x}_1,\ldots,\mathbf{x}_q)=\prod_{i=1}^{p}\eta_i(\mathbf{v}_i)\prod_{j=1}^{q}\varphi_j(\mathbf{x}_j)$,\\
with empty products equal to $1$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:iterated-tensor-products-basic-laws}, $\dim\mathcal{V}'=n$, $\dim\mathcal{T}^{p}_{q}(\mathcal{V})=\dim(\mathcal{V}^{\otimes p})\dim((\mathcal{V}')^{\otimes q})=n^pn^q=n^{p+q}$, with $n^0=1$.
\item Set $X=(\mathcal{V}')^{\otimes p}\otimes\mathcal{V}^{\otimes q}$. Since $\iota:\mathcal{V}\cong\mathcal{V}''$ and all factors are finite-dimensional,\\
$\mathcal{T}^{p}_{q}(\mathcal{V})=\mathcal{V}^{\otimes p}\otimes(\mathcal{V}')^{\otimes q} \cong((\mathcal{V}')')^{\otimes p}\otimes\mathcal{V}'^{\otimes q} \cong X' \cong\operatorname{Mult}_{\mathbb{F}}((\mathcal{V}')^{\times p}\times\mathcal{V}^{\times q};\mathbb{F})$.\\
On decomposables, the composite is $\Theta_{p,q}$; spanning gives uniqueness.
\end{enumerate}
\hlt{Structural Commentary}: Tensor elements are multilinear functionals after finite-dimensional duality.\\
\hlt{Key Insight}: Choosing which factors are active turns tensor factors into input slots.
\end{proof}
\end{proposition}

\begin{definition} \label{def:contraction-map} \hlt{Contraction}\\
Let $p,q\geq1$. The \hlt{contraction} is the unique linear map $C:\mathcal{T}^{p}_{q}(\mathcal{V})\to\mathcal{T}^{p-1}_{q-1}(\mathcal{V})$, where\\
$C(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_p\otimes\varphi_1\otimes\cdots\otimes\varphi_q)=\varphi_1(\mathbf{v}_1)\mathbf{v}_2\otimes\cdots\otimes\mathbf{v}_p\otimes\varphi_2\otimes\cdots\otimes\varphi_q$.\\
For $(i,j)$, $C_{ij}$ is $C$ conjugated by the factor permutation placing $\mathbf{v}_i,\varphi_j$ first.
\end{definition}

\begin{example} \label{ex:rank-one-operator-contraction-trace} \hlt{Rank-One Operators and Trace}\\
Under $\mathcal{V}\otimes\mathcal{V}'\cong\LL(\mathcal{V})$, $\mathbf{v}\otimes\varphi\mapsto T_{\mathbf{v},\varphi}$, where $T_{\mathbf{v},\varphi}(\mathbf{x})=\varphi(\mathbf{x})\mathbf{v}$. Hence $C(\mathbf{v}\otimes\varphi)=\varphi(\mathbf{v})=\operatorname{tr}(T_{\mathbf{v},\varphi})$.
\end{example}

\begin{definition} \label{def:special-multilinear-maps} \hlt{Symmetric, Skew-Symmetric, and Alternating Maps}\\
Let $f:\mathcal{V}^{\times n}\to\mathcal{W}$ be multilinear.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f$ is \hlt{symmetric} if interchanging any two coordinate positions does not change its value.
\item $f$ is \hlt{skew-symmetric} if interchanging any two coordinate positions multiplies its value by $-1$.
\item $f$ is \hlt{alternating} if $f(\mathbf{v}_1,\ldots,\mathbf{v}_n)=0$ whenever two coordinate entries are equal.
\end{enumerate}
If $\operatorname{char}(\mathbb{F})\neq2$, alternating and skew-symmetric are equivalent.\\
If $\operatorname{char}(\mathbb{F})=2$, skew-symmetric is the same as symmetric, and alternating is the stronger condition.
\end{definition}

\begin{definition} \label{def:permutation-action-on-tensor-powers}\label{def:symmetric-and-exterior-tensor-spaces} \hlt{Permutation Action and Symmetric/Exterior Tensor Spaces}\\
Let $\dim\mathcal{V}<\infty$, $p\in\mathbb{Z}_{\geq0}$, $\mathcal{V}^{\otimes0}=\mathbb{F}$, $T_p\defeq\mathcal{V}^{\otimes p}$, and $G_p\defeq\mathfrak{S}_p$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\sigma\in G_p$, the \hlt{permutation action} is $\lambda_{\sigma}:T_p\to T_p$, determined by $\lambda_{\sigma}(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_p)=\mathbf{v}_{\sigma(1)}\otimes\cdots\otimes\mathbf{v}_{\sigma(p)}$.
\item The \hlt{symmetric tensors} are $\operatorname{ST}^{p}(\mathcal{V})\defeq\{\mathbf{z}\in T_p:\forall\sigma\in G_p,\lambda_{\sigma}\mathbf{z}=\mathbf{z}\}$.
\item The \hlt{antisymmetric tensors} are $\operatorname{AT}^{p}(\mathcal{V})\defeq\{\mathbf{z}\in T_p:\forall\sigma\in G_p,\lambda_{\sigma}\mathbf{z}=\sign(\sigma)\mathbf{z}\}$.
\item Set $R_{\vee}\defeq\operatorname{span}\{\lambda_{\sigma}\mathbf{z}-\mathbf{z}:\sigma\in G_p,\mathbf{z}\in T_p\}$.
\item The \hlt{symmetric power} is $\operatorname{Sym}^{p}(\mathcal{V})\defeq T_p/R_{\vee}$.
\item Set $A_p\defeq\operatorname{span}\{\lambda_{\sigma}\mathbf{z}-\sign(\sigma)\mathbf{z}:\sigma\in G_p,\mathbf{z}\in T_p\}$ and $D_p\defeq\operatorname{span}\{\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_p:\exists i<j,\mathbf{v}_i=\mathbf{v}_j\}$.\\
The \hlt{exterior power} is $\bigwedge^{p}\mathcal{V}\defeq T_p/(A_p+D_p)$.
\item For quotient map $q_{\vee}$, the \hlt{symmetric product} is $q_{\vee}(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_p)=\mathbf{v}_1\vee\cdots\vee\mathbf{v}_p$
\item For quotient map $q_{\wedge}$, the \hlt{exterior product} is $q_{\wedge}(\mathbf{v}_1\otimes\cdots\otimes\mathbf{v}_p)=\mathbf{v}_1\wedge\cdots\wedge\mathbf{v}_p$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:symmetric-exterior-universal-properties} \hlt{Universal Properties of Symmetric and Exterior Powers}\\
Let $\mathcal{U}$ be a vector space, $M=\operatorname{Mult}_{\mathbb{F}}(\mathcal{V},\ldots,\mathcal{V};\mathcal{U})$, and $f^\sigma(\mathbf{v}_1,\ldots,\mathbf{v}_p)\defeq f(\mathbf{v}_{\sigma(1)},\ldots,\mathbf{v}_{\sigma(p)})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\LL(\operatorname{Sym}^{p}(\mathcal{V}),\mathcal{U})\cong M_{\vee}$, where $M_{\vee}\defeq\{f\in M:\forall\sigma\in\mathfrak{S}_p,\ f^\sigma=f\}$.
\item $\LL(\bigwedge^{p}\mathcal{V},\mathcal{U})\cong M_{\wedge}$, where $M_{\wedge}\defeq\{f\in M:\forall\sigma\in\mathfrak{S}_p,\ f^\sigma=\sign(\sigma)f,\ \forall(\mathbf{v}_1,\ldots,\mathbf{v}_p)$,\\
$(\exists i<j,\mathbf{v}_i=\mathbf{v}_j)\Rightarrow f(\mathbf{v}_1,\ldots,\mathbf{v}_p)=0\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
Let $t:\mathcal{V}^{\times p}\to\mathcal{V}^{\otimes p}$ and $q_{\vee}:\mathcal{V}^{\otimes p}\to\operatorname{Sym}^{p}(\mathcal{V})$ be canonical.\\
The map $\Phi_{\vee}(T)=Tq_{\vee}t$ sends $\LL(\operatorname{Sym}^{p}(\mathcal{V}),\mathcal{U})$ into $M_{\vee}$.\\
Conversely, for $f\in M_{\vee}$, there is a unique $\widetilde{f}:\mathcal{V}^{\otimes p}\to\mathcal{U}$ with $f=\widetilde{f}t$; since $\widetilde{f}(R_{\vee})=0$, it factors uniquely through $q_{\vee}$ as $\widetilde{f}=\overline{f}q_{\vee}$, where $\overline{f}:\operatorname{Sym}^{p}(\mathcal{V})\to\mathcal{U}$. Thus $f\mapsto\overline{f}$ is $\Phi_{\vee}^{-1}$.
\item Let $t:\mathcal{V}^{\times p}\to\mathcal{V}^{\otimes p}$ and $q_{\wedge}:\mathcal{V}^{\otimes p}\to\bigwedge^{p}\mathcal{V}$ be canonical.\\
The map $\Phi_{\wedge}(T)=Tq_{\wedge}t$ sends $\LL(\bigwedge^{p}\mathcal{V},\mathcal{U})$ into $M_{\wedge}$. Conversely, for $f\in M_{\wedge}$, the induced $\widetilde{f}:\mathcal{V}^{\otimes p}\to\mathcal{U}$ satisfies $\widetilde{f}(R_{\wedge})=0$, hence factors uniquely as $\widetilde{f}=\overline{f}q_{\wedge}$; thus $f\mapsto\overline{f}$ is $\Phi_{\wedge}^{-1}$.
\end{enumerate}
\hlt{Structural Commentary}: Symmetric and exterior powers are tensor quotients by permutation relations.\\
\hlt{Key Insight}: Symmetric and alternating multilinear maps are ordinary linear maps out of the right quotient.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:symmetric-exterior-dimensions} \hlt{Dimensions of Symmetric and Exterior Powers}\\
Let $\dim\mathcal{V}=n<\infty$ and $p\in\mathbb{Z}_{\geq0}$. Set $I_{\vee}=\{1\leq i_1\leq\cdots\leq i_p\leq n\}$ and $I_{\wedge}=\{1\leq i_1<\cdots<i_p\leq n\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\dim\operatorname{Sym}^{p}(\mathcal{V})=\#I_{\vee}=\binom{n+p-1}{p}$.
\item $\dim\bigwedge^{p}\mathcal{V}=\#I_{\wedge}=\binom{n}{p}$, with $\binom{n}{p}=0$ for $p>n$.
\end{enumerate}
\begin{proof}
Let $E=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ be a basis. For $I=(i_1,\ldots,i_p)$, set $\mathbf{e}_{I}^{\vee}=\mathbf{e}_{i_1}\vee\cdots\vee\mathbf{e}_{i_p}$ and $\mathbf{e}_{I}^{\wedge}=\mathbf{e}_{i_1}\wedge\cdots\wedge\mathbf{e}_{i_p}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
$R_{\vee}$ identifies each word with its nondecreasing representative. Thus $B_{\vee}=\{\mathbf{e}_{I}^{\vee}:I\in I_{\vee}\}$ is the quotient basis, and $\dim\operatorname{Sym}^{p}(\mathcal{V})=\#B_{\vee}=\#I_{\vee}=\binom{n+p-1}{p}$.
\item
$R_{\wedge}$ kills repeated words and identifies each remaining word with its increasing representative up to sign. Thus $B_{\wedge}=\{\mathbf{e}_{I}^{\wedge}:I\in I_{\wedge}\}$ is the quotient basis, and $\dim\bigwedge^{p}\mathcal{V}=\#B_{\wedge}=\#I_{\wedge}=\binom{n}{p}$.
\end{enumerate}
\hlt{Structural Commentary}: Symmetric powers count repetitions; exterior powers forbid repetitions.\\
\hlt{Key Insight}: Multisets give symmetric dimensions, subsets give exterior dimensions.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:symmetrization-map} \hlt{Symmetrization in Characteristic Zero}\\
Assume $\operatorname{char}(\mathbb{F})=0$. Define $S,A\in\LL(\mathcal{V}^{\otimes p})$ by $S\defeq\frac{1}{p!}\sum\limits_{\sigma\in\mathfrak{S}_p}\lambda_{\sigma}$ and $A\defeq\frac{1}{p!}\sum\limits_{\sigma\in\mathfrak{S}_p}\sign(\sigma)\lambda_{\sigma}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $S^2=S$ and $\operatorname{im}S=\operatorname{ST}^{p}(\mathcal{V})$.
\item $A^2=A$ and $\operatorname{im}A=\operatorname{AT}^{p}(\mathcal{V})$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\tau\in\mathfrak{S}_p$, $\lambda_{\tau}S=p!^{-1}\sum_{\sigma}\lambda_{\tau\sigma}=S$, hence $\operatorname{im}S\subseteq\operatorname{ST}^{p}(\mathcal{V})$.\\
If $\mathbf{z}\in\operatorname{ST}^{p}(\mathcal{V})$, then $S\mathbf{z}=\mathbf{z}$, so $\operatorname{im}S=\operatorname{ST}^{p}(\mathcal{V})$ and $S^2=S$.
\item For $\tau\in\mathfrak{S}_p$, $\lambda_{\tau}A=p!^{-1}\sum_{\sigma}\sign(\sigma)\lambda_{\tau\sigma}=\sign(\tau)A$, hence $\operatorname{im}A\subseteq\operatorname{AT}^{p}(\mathcal{V})$.\\
If $\mathbf{z}\in\operatorname{AT}^{p}(\mathcal{V})$, then $A\mathbf{z}=p!^{-1}\sum_{\sigma}\sign(\sigma)^2\mathbf{z}=\mathbf{z}$, so $\operatorname{im}A=\operatorname{AT}^{p}(\mathcal{V})$ and $A^2=A$.
\end{enumerate}
\hlt{Structural Commentary}: Averaging gives symmetry projections.\\
\hlt{Key Insight}: The factor $1/p!$ makes the averaging map fix the tensors it projects onto.
\end{proof}
\end{proposition}

\begin{definition} \label{def:graded-tensor-symmetric-exterior-algebras} \hlt{Tensor, Symmetric, and Exterior Algebras}\\
Let $\dim\mathcal{V}<\infty$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{tensor algebra} is $\mathcal{T}(\mathcal{V})\defeq\bigoplus_{p\geq0}\mathcal{V}^{\otimes p}$, with $\mathcal{V}^{\otimes0}\defeq\mathbb{F}$ and product induced by tensor concatenation.
\item The \hlt{symmetric algebra} is $\operatorname{Sym}(\mathcal{V})\defeq\bigoplus_{p\geq0}\operatorname{Sym}^{p}(\mathcal{V})$, with product induced by $\vee$.
\item The \hlt{exterior algebra} is $\bigwedge\mathcal{V}\defeq\bigoplus_{0\leq p\leq\dim\mathcal{V}}\bigwedge^{p}\mathcal{V}$, with product induced by $\wedge$.
\item Each is \hlt{graded}: degree $p$ times degree $q$ lies in degree $p+q$.
\end{enumerate}
\end{definition}

\begin{corollary} \label{cor:determinant-from-exterior-universal-property} \hlt{Determinant from the Exterior Universal Property}\\
Let $\dim\mathcal{V}=n$, $E=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$, $\Omega_E\defeq\mathbf{e}_1\wedge\cdots\wedge\mathbf{e}_n$, and $A_E(\mathbf{v}_1,\ldots,\mathbf{v}_n)=([\mathbf{v}_j]_i)_{i,j}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item There is a unique $\ell_E\in(\bigwedge^n\mathcal{V})'$ with $\ell_E(\Omega_E)=1$; hence $d_E(\mathbf{v}_1,\ldots,\mathbf{v}_n)\defeq\ell_E(\mathbf{v}_1\wedge\cdots\wedge\mathbf{v}_n)$ is the unique alternating form satisfying $d_E(\mathbf{e}_1,\ldots,\mathbf{e}_n)=1$.
\item For all $\mathbf{v}_1,\ldots,\mathbf{v}_n\in\mathcal{V}$, $d_E(\mathbf{v}_1,\ldots,\mathbf{v}_n) =\sum\limits_{\sigma\in\mathfrak{S}_n}\sign(\sigma)\prod\limits_{j=1}^{n}[\mathbf{v}_j]_{\sigma(j)} =\det A_E(\mathbf{v}_1,\ldots,\mathbf{v}_n)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:symmetric-exterior-dimensions}, $\bigwedge^n\mathcal{V}=\mathbb{F}\Omega_E$, so $\ell_E(\Omega_E)=1$ fixes a unique $\ell_E\in(\bigwedge^n\mathcal{V})'$; via Theorem \ref{thm:symmetric-exterior-universal-properties}, $\ell_E$ corresponds to $d_E(\mathbf{v}_1,\ldots,\mathbf{v}_n)=\ell_E(\mathbf{v}_1\wedge\cdots\wedge\mathbf{v}_n)$, hence $d_E$ is unique.
\item Let $\mathbf{v}_j=\sum_{i=1}^{n}a_{ij}\mathbf{e}_i$, $a_{ij}=[\mathbf{v}_j]_i$. Repeated index tuples have zero wedge; distinct tuples are $(\sigma(1),\ldots,\sigma(n))$, with wedge $\sign(\sigma)\Omega_E$. Hence $\mathbf{v}_1\wedge\cdots\wedge\mathbf{v}_n =\sum\limits_{\sigma\in\mathfrak{S}_n}\sign(\sigma)\prod\limits_{j=1}^{n}a_{\sigma(j),j}\Omega_E$, and\\
$d_E(\mathbf{v}_1,\ldots,\mathbf{v}_n) =\ell_E(\mathbf{v}_1\wedge\cdots\wedge\mathbf{v}_n) =\sum\limits_{\sigma\in\mathfrak{S}_n}\sign(\sigma)\prod\limits_{j=1}^{n}a_{\sigma(j),j}\\ =\det(a_{ij}) =\det A_E(\mathbf{v}_1,\ldots,\mathbf{v}_n)$.
\end{enumerate}
\hlt{Structural Commentary}: Determinant is the basis-normalised alternating form.\\
\hlt{Key Insight}: Top exterior dimension one turns normalisation into determinant.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:exterior-determinant-basic-properties} \hlt{Determinant Properties from Exterior Powers}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$, and set $d(A)\defeq d_E(A_{\bullet1},\ldots,A_{\bullet n})$ for $E$ the standard basis of $\mathbb{F}^n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $d(A^t)=d(A)$.
\item $d(AB)=d(A)d(B)$.
\item $A\in\operatorname{GL}(n,\mathbb{F})\Longleftrightarrow d(A)\neq0$.
\end{enumerate}
\begin{proof}
Let $\omega=\mathbf{e}_1\wedge\cdots\wedge\mathbf{e}_n$ and let $T_A$ denote the linear map with matrix $A$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Corollary \ref{cor:determinant-from-exterior-universal-property}, $d(A^t)=\sum_{\sigma}\sign(\sigma)\prod_{j=1}^{n}A_{j,\sigma(j)}$. With $\tau=\sigma^{-1}$, $\sum_{\tau}\sign(\tau)\prod_{i=1}^{n}A_{\tau(i),i}=d(A)$.
\item Since $(\bigwedge^n T_A)\omega=d(A)\omega$ and $T_{AB}=T_A\circ T_B$, applying $\bigwedge^n(T_A\circ T_B)=(\bigwedge^n T_A)(\bigwedge^n T_B)$ to $\omega$ gives $d(AB)\omega=d(A)d(B)\omega$, hence $d(AB)=d(A)d(B)$.
\item[$(iii)$]$(\Rightarrow)$ If $A\in\operatorname{GL}(n,\mathbb{F})$, then $I=AA^{-1}$, so $1=d(I)=d(A)d(A^{-1})$ by (ii); hence $d(A)\neq0$.
\item[$(iii)$]$(\Leftarrow)$ Let $C_j=A_{\bullet j}$. Suppose $\sum_{j=1}^{n}c_jC_j=0$ and $c_k\neq0$. Then $C_k=-c_k^{-1}\sum_{j\neq k}c_jC_j$, so multilinearity in the $k$th slot gives $d(A)=d(C_1,\ldots,C_k,\ldots,C_n) =-c_k^{-1}\sum\limits_{j\neq k}c_jd(C_1,\ldots,C_{k-1},C_j,C_{k+1},\ldots,C_n)=0$, since each summand has equal $j$th and $k$th entries.\\
Thus $d(A)\neq0$ implies $C_1,\ldots,C_n$ are independent, hence $A\in\operatorname{GL}(n,\mathbb{F})$.
\end{enumerate}
\hlt{Structural Commentary}: Determinant laws are top-exterior scalar laws.\\
\hlt{Key Insight}: Top exterior dimension one makes determinants multiplicative.
\end{proof}
\end{corollary}
