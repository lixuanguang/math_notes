\subsection{Linear Operators as Modules}

\begin{perspective} \label{per:operator-as-polynomial-action} \hlt{An Operator as a Polynomial Action}\\
Fixing $T\in\LL(\mathcal{V})$ lets polynomials act on vectors by $p\cdot\mathbf{v}\defeq p(T)\mathbf{v}$; $p\in\mathbb{F}[z]$, $\mathbf{v}\in\mathcal{V}$.\\
This action is still built from ordinary vector addition and scalar multiplication, but it packages the whole orbit $\mathbf{v},T\mathbf{v},T^2\mathbf{v},\ldots$ into one polynomial expression.
\end{perspective}

\begin{definition} \label{def:cyclic-subspace-and-vector-annihilator} \hlt{Cyclic Subspace and Vector Annihilator}\\
Let $T\in\LL(\mathcal{V})$ and $\mathbf{v}\in\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{$T$-cyclic subspace generated by $\mathbf{v}$} is $\mathcal{C}_T(\mathbf{v})\defeq\operatorname{span}\{T^j\mathbf{v}:j\geq0\}$
\item The vector $\mathbf{v}$ is a \hlt{cyclic vector} for $T$ if $\mathcal{C}_T(\mathbf{v})=\mathcal{V}$.
\item The \hlt{annihilator of $\mathbf{v}$ for $T$} is $\operatorname{ann}_T(\mathbf{v})\defeq\{p\in\mathbb{F}[z]:p(T)\mathbf{v}=\mathbf{0}\}$
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:cyclic-subspace-smallest-invariant-subspace} \hlt{Cyclic Subspaces Are Smallest Invariant Hosts}\\
Let $T\in\LL(\mathcal{V})$ and $\mathbf{v}\in\mathcal{V}$. Then $\mathcal{C}_T(\mathbf{v})=\{p(T)\mathbf{v}:p\in\mathbb{F}[z]\}$.\\
Moreover, $\mathcal{C}_T(\mathbf{v})$ is the smallest $T$-invariant subspace containing $\mathbf{v}$, and $\operatorname{ann}_T(\mathbf{v})$ is an ideal of $\mathbb{F}[z]$.
\begin{proof} Equality follows as $p(T)\mathbf{v}$ is a finite linear combination of the orbit vectors $T^j\mathbf{v}$, and every such finite linear combination equals $p(T)\mathbf{v}$ for some $p\in\mathbb{F}[z]$. Also $T(p(T)\mathbf{v})=(zp)(T)\mathbf{v}$, so $\mathcal{C}_T(\mathbf{v})$ is $T$-invariant.\\
If $\mathcal{U}$ is $T$-invariant and contains $\mathbf{v}$, then $T^j\mathbf{v}\in\mathcal{U}$ for every $j\geq0$, hence $\mathcal{C}_T(\mathbf{v})\subseteq\mathcal{U}$. Finally, if $p,q\in\operatorname{ann}_T(\mathbf{v})$ and $r\in\mathbb{F}[z]$, then $(p+q)(T)\mathbf{v}=\mathbf{0}$ and $(rp)(T)\mathbf{v}=r(T)p(T)\mathbf{v}=\mathbf{0}$ by Proposition \ref{prop:operator-polynomial-products}. $\operatorname{ann}_T(\mathbf{v})$ is an ideal.\\
\hlt{Structural Commentary}: A cyclic subspace is the span of the orbit generated by repeatedly applying $T$.\\
\hlt{Key Insight}: Invariance is exactly what lets the whole orbit stay inside the same host subspace.
\end{proof}
\end{proposition}

\begin{definition} \label{def:vector-order-under-operator} \hlt{Order of a Vector under an Operator}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and let $\mathbf{v}\neq\mathbf{0}$. The \hlt{order of $\mathbf{v}$ under $T$} is the unique monic polynomial $m_{\mathbf{v}}\in\operatorname{ann}_T(\mathbf{v})$ of least degree; equivalently, $\operatorname{ann}_T(\mathbf{v})=(m_{\mathbf{v}})$
\end{definition}

\begin{definition} \label{def:companion-matrix} \hlt{Companion Matrix}\\
Let $f(z)=z^n+c_{n-1}z^{n-1}+\cdots+c_1z+c_0$ be monic. For $n\geq2$, the \hlt{companion matrix} of $f$ is
\begin{align}
C_f\defeq
\begin{bmatrix}
0&0&\cdots&0&-c_0\\
1&0&\cdots&0&-c_1\\
0&1&\cdots&0&-c_2\\
\vdots&\vdots&\ddots&\vdots&\vdots\\
0&0&\cdots&1&-c_{n-1}
\end{bmatrix}.
\nonumber
\end{align}
For $n=1$, set $C_f=[-c_0]$.
\end{definition}

\begin{theorem} \label{thm:companion-matrix-has-prescribed-minimal-polynomial} \hlt{Companion Matrix Realises a Polynomial}\\
For every monic polynomial $f$, the companion matrix $C_f$ satisfies $p_{C_f}=f$ and $\mu_{C_f}=f$.
\begin{proof}
Let $\mathbf{e}_1,\ldots,\mathbf{e}_n$ be the standard basis. For $0\leq j<n$, $C_f^j\mathbf{e}_1=\mathbf{e}_{j+1}$.\\
The last-column relation gives $C_f^n\mathbf{e}_1 =-c_0\mathbf{e}_1-c_1C_f\mathbf{e}_1-\cdots-c_{n-1}C_f^{n-1}\mathbf{e}_1$. Thus $f(C_f)\mathbf{e}_1=0$.\\
Since $\mathbf{e}_j=C_f^{j-1}\mathbf{e}_1$ and $f(C_f)$ commutes with $C_f$, we get $f(C_f)\mathbf{e}_j=C_f^{j-1}f(C_f)\mathbf{e}_1=\mathbf{0} \ \forall j$, so $f(C_f)=0$.\\
If $g(z)=\sum\limits_{i=0}^{m}b_iz^i$ has $m<n$, then $g(C_f)\mathbf{e}_1=b_0\mathbf{e}_1+b_1\mathbf{e}_2+\cdots+b_m\mathbf{e}_{m+1}$, which is zero only when all $b_i=0$. Hence no nonzero polynomial of degree $<n$ annihilates $C_f$. Therefore $\deg\mu_{C_f}=n$.\\
Since $f$ is a monic degree-$n$ annihilator and $\mu_{C_f}\mid f$, we have $\mu_{C_f}=f$.\\
Cayley--Hamilton makes $p_{C_f}$ a monic degree-$n$ annihilator, so $\mu_{C_f}\mid p_{C_f}$. Both $p_{C_f}$ and $f$ are monic of degree $n$, and both are multiples of the monic degree-$n$ polynomial $\mu_{C_f}=f$, hence $p_{C_f}=f$.\\
\hlt{Structural Commentary}: The cyclic vector $\mathbf{e}_1$ generates the basis, and last column imposes relation $f(C_f)=0$.\\
\hlt{Key Insight}: The companion matrix makes $f$ the first possible annihilating relation.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:cyclic-subspace-quotient-and-companion} \label{prop:cyclic-basis-gives-companion-matrix} \hlt{Cyclic Subspaces as Polynomial Quotients}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and let $\mathbf{v}\neq\mathbf{0}$.\\
Let the order of $\mathbf{v}$ under $T$ be $m_{\mathbf{v}}(z)=z^d+c_{d-1}z^{d-1}+\cdots+c_1z+c_0$.
Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\Phi_{\mathbf{v}}:\mathbb{F}[z]\to\mathcal{C}_T(\mathbf{v})$, $\Phi_{\mathbf{v}}(p)=p(T)\mathbf{v}$, is onto with $\ker\Phi_{\mathbf{v}}=\langle m_{\mathbf{v}}\rangle$, hence $\mathcal{C}_T(\mathbf{v})\cong \mathbb{F}[z]/\langle m_{\mathbf{v}}\rangle$;
\item $\mathcal{B}_{\mathbf{v}}\defeq(\mathbf{v},T\mathbf{v},\ldots,T^{d-1}\mathbf{v})$ is a basis of $\mathcal{C}_T(\mathbf{v})$;
\item $[T|_{\mathcal{C}_T(\mathbf{v})}]_{\mathcal{B}_{\mathbf{v}}}=C_{m_{\mathbf{v}}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Surjectivity follows from $\mathcal{C}_T(\mathbf{v})=\{p(T)\mathbf{v}:p\in\mathbb{F}[z]\}$.\\
Also $\ker\Phi_{\mathbf{v}}=\operatorname{ann}_T(\mathbf{v})=\langle m_{\mathbf{v}}\rangle$ by Definition \ref{def:vector-order-under-operator}, so Theorem \ref{thm:quotient-by-kernel-is-range} gives the quotient isomorphism.
\item If $\sum_{j=0}^{d-1}a_jT^j\mathbf{v}=\mathbf{0}$, then $q(z)\defeq\sum_{j=0}^{d-1}a_jz^j\in\operatorname{ann}_T(\mathbf{v})$; minimality of $m_{\mathbf{v}}$ forces $q=0$, so $\mathcal{B}_{\mathbf{v}}$ is independent. For any $p\in\mathbb{F}[z]$, write $p=hm_{\mathbf{v}}+r$ with $\deg r<d$ or $r=0$; then $p(T)\mathbf{v}=r(T)\mathbf{v}\in\operatorname{span}\mathcal{B}_{\mathbf{v}}$. Hence $\mathcal{B}_{\mathbf{v}}$ is a basis.
\item For $0\leq j\leq d-2$, $T(T^j\mathbf{v})=T^{j+1}\mathbf{v}$.\\
Since $m_{\mathbf{v}}(T)\mathbf{v}=\mathbf{0}$, $T^d\mathbf{v}=-c_0\mathbf{v}-c_1T\mathbf{v}-\cdots-c_{d-1}T^{d-1}\mathbf{v}$, which is the last-column relation of $C_{m_{\mathbf{v}}}$.
\end{enumerate}
\hlt{Structural Commentary}: The first relation cuts out the quotient and companion block.\\
\hlt{Key Insight}: One orbit carries the whole cyclic subspace.
\end{proof}
\end{theorem}

\begin{definition} \label{def:operator-invariant-factors-elementary-divisors} \hlt{Invariant Factors and Elementary Divisors}\\
Let $T\in\LL(\mathcal{V})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{cyclic decomposition} of $T$ is a direct sum $\mathcal{V}=\bigoplus\limits_{j=1}^r\mathcal{C}_j$ of nonzero $T$-invariant cyclic subspaces.
\item If $\mathcal{C}_j\cong\mathbb{F}[z]/\langle f_j\rangle$, $f_j\in\mathbb{F}[z]$ monic and nonconstant, $f_r\mid f_{r-1}\mid\cdots\mid f_1$,\\
then $f_1,\ldots,f_r$ are the \hlt{invariant factors} of $T$.
\item Prime-power factors in factorizations of $f_j$ into monic irreducibles are the \hlt{elementary divisors} of $T$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:operator-invariant-factor-skeleton} \hlt{Invariant-Factor Skeleton for Operators}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{F}$, and let $T\in\LL(\mathcal{V})$. Equip $\mathcal{V}$ with $\mathbb{F}[z]$-module structure $p\cdot\mathbf{v}=p(T)\mathbf{v}$.\\
Then $\exists$ monic nonconstant polynomials $f_1,\ldots,f_r\in\mathbb{F}[z]$ with $f_r\mid f_{r-1}\mid\cdots\mid f_1$ and nonzero $T$-invariant cyclic subspaces $\mathcal{C}_1,\ldots,\mathcal{C}_r$ such that $\mathcal{V}=\mathcal{C}_1\oplus\cdots\oplus\mathcal{C}_r$ and $\mathcal{C}_j\cong \mathbb{F}[z]/\langle f_j\rangle$.\\
In the concatenated cyclic bases, $[T]=C_{f_1}\oplus\cdots\oplus C_{f_r}$. Moreover, $f_1=\mu_T$, the characteristic polynomial of $T$ is $f_1\cdots f_r$, and the polynomials $f_1,\ldots,f_r$ are uniquely determined by $T$.
\begin{proof}
As an $\mathbb{F}[z]$-module, $\mathcal{V}$ is finitely generated, and $\mu_T(T)=0$ makes every vector torsion.\\
Invariant-factor theorem over PID $\mathbb{F}[z]$ gives $\mathcal{V}\cong\bigoplus\limits_{j=1}^r\mathbb{F}[z]/\langle f_j\rangle$, $f_r\mid\cdots\mid f_1$, with each $f_j$ monic and nonconstant.\\
Transporting cyclic generators gives subspaces $\mathcal{C}_j$, and Theorem \ref{thm:cyclic-subspace-quotient-and-companion} gives companion block $C_{f_j}$ on each cyclic basis. The annihilator of the direct sum is $\langle\operatorname{lcm}(f_1,\ldots,f_r)\rangle=\langle f_1\rangle$, while the annihilator of the operator module is $\langle\mu_T\rangle$; hence $f_1=\mu_T$. Also $p_{C_{f_j}}=f_j$ by Theorem \ref{thm:companion-matrix-has-prescribed-minimal-polynomial}, so the block diagonal form has characteristic polynomial $\prod_{j=1}^r f_j$. Uniqueness is exactly uniqueness of invariant factors in the PID structure theorem.\\
\hlt{Structural Commentary}: Torsion module decomposition produces companion blocks.\\
\hlt{Key Insight}: Invariant factors are the polynomial-module version of choosing canonical blocks for similarity.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:elementary-divisors-and-jordan-blocks} \hlt{Elementary Divisors and Jordan Blocks}\\
Over $\mathbb{C}$, every elementary divisor has the form $(z-\lambda)^k$. The cyclic summand $\mathbb{C}[z]/\langle (z-\lambda)^k\rangle$ corresponds to one Jordan block $J_k(\lambda)$. Hence elementary divisors over $\mathbb{C}$ are precisely Jordan blocks counted with multiplicity.
\end{perspective}

\begin{perspective} \label{per:operator-module-similarity-dictionary} \hlt{Operator Modules and Similarity}\\
For each $T\in\LL(\mathcal{V})$, let $\mathcal{V}_T$ be the $\mathbb{F}[z]$-module with action $p\cdot\mathbf{v}=p(T)\mathbf{v}$. For $S,T\in\LL(\mathcal{V})$, $S\sim T \iff \mathcal{V}_S\cong\mathcal{V}_T$, as $\mathbb{F}[z]$-modules, and each cyclic summand $\mathbb{F}[z]/\langle f\rangle$ corresponds to the companion block $C_f$ by Theorem \ref{thm:cyclic-subspace-quotient-and-companion}.
\end{perspective}

\begin{definition} \label{def:rational-canonical-block-forms} \hlt{Rational Canonical Block Forms}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A matrix is in \hlt{invariant-factor rational canonical form} if $A=\operatorname{diag}(C_{s_1},\ldots,C_{s_r})$, $s_r\mid s_{r-1}\mid\cdots\mid s_1$, where each $s_j\in\mathbb{F}[z]$ is monic and nonconstant; the $s_j$ are the \hlt{invariant factors}.
\item A matrix is in \hlt{elementary-divisor rational canonical form} if $A=\operatorname{diag}(C_{q_1},\ldots,C_{q_m})$, where each $q_j$ is a power of a monic irreducible polynomial; the $q_j$ are the \hlt{elementary divisors}.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:rational-canonical-form-skeleton} \hlt{Rational Canonical Form Skeleton}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$. $\exists$ unique monic nonconstant $s_1,\ldots,s_r\in\mathbb{F}[z]$ with $s_r\mid\cdots\mid s_1$ such that $A$ is similar over $\mathbb{F}$ to $\operatorname{diag}(C_{s_1},\ldots,C_{s_r})$. Moreover, $\mu_A=s_1,\qquad p_A=s_1s_2\cdots s_r$.\\
Equivalently, the elementary divisors, counted with multiplicity, form a complete similarity invariant over $\mathbb{F}$.
\begin{proof}
Let $T_A:\mathbb{F}^n\to\mathbb{F}^n$ be $T_A\mathbf{x}=A\mathbf{x}$. By Theorem \ref{thm:operator-invariant-factor-skeleton}, $\mathbb{F}^n=\mathcal{C}_1\oplus\cdots\oplus\mathcal{C}_r$; $\mathcal{C}_j\cong\mathbb{F}[z]/\langle s_j\rangle$, $s_r\mid\cdots\mid s_1$.\\
In cyclic bases, $[T_A]=\operatorname{diag}(C_{s_1},\ldots,C_{s_r})$, so $A$ is similar to this form.\\
Uniqueness, $\mu_A=s_1$, $p_A=s_1\cdots s_r$, and the elementary-divisor classification follow from the same theorem.\\
\hlt{Structural Commentary}: Companion blocks classify similarity without requiring $p_A$ to split.\\
\hlt{Key Insight}: Similarity classes can be represented by polynomial data rather than by choosing roots.
\end{proof}
\end{theorem}
