\subsection{Group Actions}

% Source bridge: Roman, Advanced Linear Algebra, Chapter 18.
% Role here: the class-equation input needed for finite division algebras.

A group action packages the idea that elements of a group act as reversible
motions of a set. The conjugation action is the finite-group ingredient used
later in Wedderburn's theorem.

\begin{definition} \label{def:group-action-orbit-stabilizer} \hlt{Actions, Orbits, and Stabilizers}\\
Let $G$ be a group and let $\mathcal{X}$ be a set.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Left action} of $G$ on $\mathcal{X}$ is map $G\times\mathcal{X}\to\mathcal{X}$, $(g,x)\mapsto gx$, s.t. $e_Gx=x$ and $(gh)x=g(hx)$ $\forall g,h\in G$, $x\in\mathcal{X}$.
\item The \hlt{orbit} of $x\in\mathcal{X}$ is $Gx\defeq\{gx:g\in G\}$.
\item The \hlt{stabilizer} of $x\in\mathcal{X}$ is $G_x\defeq\{g\in G:gx=x\}$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:orbit-stabilizer-finite-group-actions} \hlt{Orbit-Stabilizer}\\
Let a finite group $G$ act on $\mathcal{X}$. For each $x\in\mathcal{X}$, $|Gx|=[G:G_x]=\frac{|G|}{|G_x|}$.
\begin{proof}
Define $\Phi:G/G_x\to Gx$, $\Phi(gG_x)=gx$.\\
For $g,h\in G$, $gG_x=hG_x \iff h^{-1}g\in G_x \iff h^{-1}gx=x \iff gx=hx$. Thus $\Phi$ is well defined and injective. It is surjective by definition of $Gx$, hence bijective, so $|Gx|=|G/G_x|=[G:G_x]$.\\
\hlt{Structural Commentary}: Orbits are parametrized by cosets of the stabilizer.\\
\hlt{Key Insight}: Same orbit point $\iff$ same stabilizer coset.
\end{proof}
\end{theorem}

\begin{definition} \label{def:conjugacy-class-centralizer} \hlt{Conjugacy Classes and Centralizers}\\
Let $G$ be a group.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{conjugation action} of $G$ on itself is $g\cdot x\defeq gxg^{-1}$.
\item The \hlt{conjugacy class} of $x\in G$ is $\{x\}^{G}\defeq\{gxg^{-1}:g\in G\}$.
\item The \hlt{centralizer} of $x\in G$ is $C_G(x)\defeq\{g\in G:gx=xg\}$.
\item The \hlt{center} of $G$ is $Z(G)\defeq\{g\in G:gx=xg\text{ for all }x\in G\}$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:finite-group-class-equation} \hlt{Class Equation}\\
Let $G$ be finite, $S\subseteq G$ contain one representative of each noncentral conjugacy class.\\
Then $|G|=|Z(G)|+\sum\limits_{x\in S}[G:C_G(x)]$.
\begin{proof}
The conjugacy classes form a partition $G=Z(G)\sqcup\bigsqcup\limits_{x\in S}\{x\}^{G}$.\\
Indeed, $\abs{\{x\}^{G}}=1 \iff gxg^{-1}=x$ for all $g\in G \iff x\in Z(G)$. For $x\in S$, Theorem \ref{thm:orbit-stabilizer-finite-group-actions} applied to conjugation gives $\abs{\{x\}^{G}}=[G:C_G(x)]$. Taking cardinalities yields the formula.\\
\hlt{Structural Commentary}: $G$ decomposes into central singleton classes and noncentral conjugacy classes.\\
\hlt{Key Insight}: Count conjugacy classes by orbit-stabilizer.
\end{proof}
\end{theorem}
