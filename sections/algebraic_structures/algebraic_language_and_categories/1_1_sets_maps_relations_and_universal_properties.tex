\subsection{Sets, Maps, Relations, and Universal Properties}

The elementary set, order, choice, function, and bijection vocabulary used here is collected in the foundational prerequisites. This subsection continues with set algebra, universal properties, and equivalence-based structure.

\begin{definition} \label{def:multisets-and-multiplicity} \hlt{Multisets}\\
Let $S$ be a set. A \hlt{multiset on $S$} is a pair $(S_0,m)$ such that
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $S_0\subseteq S$ is the \hlt{underlying set};
\item $m:S_0\to\Z_{>0}$ is the \hlt{multiplicity function};
\item $m(s)$ is the \hlt{multiplicity} of $s\in S_0$;
\item if $S_0$ is finite, the \hlt{size} is $\lvert m\rvert\defeq\sum_{s\in S_0}m(s)$.
\end{enumerate}
Equivalently, record $(S_0,m)$ as $\{(s,m(s)):s\in S_0\}$.
\end{definition}

\begin{example} \label{ex:multiset-eigenvalue-bookkeeping} \hlt{Multiset Bookkeeping}\\
The list $\{a,a,b,b,b,c\}$ is the multiset on $\{a,b,c\}$ with multiplicities $2,3,1$.
This language is useful whenever order is irrelevant but repetition matters, such as eigenvalues counted with algebraic multiplicity.
\end{example}

\begin{proposition} \label{prop:equivalence-relations-partitions} \hlt{Equivalence Classes Partition the Set}\\
Let $\sim$ be an equivalence relation on $X$. Then the equivalence classes form a partition of $X$.
\begin{proof}
Every $x\in X$ lies in $[x]$ since $x\sim x$. If $[x]\cap[y]\neq\emptyset$, choose $z$ in the intersection.\\
Then $x\sim z\sim y$, so $u\in[x]$ implies $u\sim y$ and $u\in[y]$. Hence $[x]\subseteq[y]$, and symmetry gives equality.\\
\hlt{Structural Commentary}: Equivalence means partition.\\
\hlt{Key Insight}: Touching classes coincide.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:number-systems-from-representatives} \hlt{Number Systems from Representatives}\\
Build by representatives, then pass to classes:
\begin{align}
\Z:\ (a,b)\mapsto[(a,b)],\qquad
\Q:\ (a,b),\,b\neq0\mapsto[(a,b)],\qquad
\R:\ \text{cuts or Cauchy data}\mapsto[\cdot]. \nonumber
\end{align}
The pattern is: choose $\sim$, prove $+,\cdot,<,\ldots$ are well defined on $X/{\sim}$, then use the classes as the number system; compare Definition \ref{def:relations-and-equivalence} and Proposition \ref{prop:quotient-space-structure}.\\
\hlt{Key Insight}: Representatives are scaffolding; the quotient laws are the interface.
\end{perspective}

\begin{example} \label{ex:supremum-need-not-be-attained} \hlt{Suprema Need Not Be Elements}\\
In $\Q$, the set $E_1=\{r\in\Q:r<0\}$ and the set $E_2=\{r\in\Q:r\leq0\}$ both have supremum $0$, but $0\notin E_1$ and $0\in E_2$. If $E=\{1/n:n=1,2,3,\ldots\}$, then $\sup E=1$ while $\inf E=0$, and the infimum is not a member of $E$.
\end{example}

\begin{theorem} \label{thm:lub-property-gives-glb-property} \hlt{Least Upper Bounds Give Greatest Lower Bounds}\\
If $S$ has the least-upper-bound property, then every nonempty bounded-below set $B\subseteq S$ has an infimum.
\begin{proof}
Let $L$ be the set of lower bounds of $B$. Then $L$ is nonempty and every $x\in B$ is an upper bound of $L$, so $\alpha=\sup L$ exists. Since every $x\in B$ bounds $L$ above, $\alpha\leq x$ for all $x\in B$; hence $\alpha$ is a lower bound of $B$.\\
If $\beta>\alpha$, then $\beta\notin L$, so $\beta$ is not a lower bound. Thus $\alpha=\inf B$.\\
\hlt{Structural Commentary}: Infima are suprema of lower bounds.\\
\hlt{Key Insight}: Completeness above gives completeness below.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:image-preimage-calculus} \hlt{Image and Preimage Calculus}\\
Let $f:X\to Y$, let $\{A_i\}_{i\in I}$ be a nonempty family of subsets of $X$, and let $\{B_i\}_{i\in I}$ be a nonempty family of subsets of $Y$. Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f\left(\bigcup_{i\in I}A_i\right)=\bigcup_{i\in I}f(A_i)$;
\item $f\left(\bigcap_{i\in I}A_i\right)\subseteq\bigcap_{i\in I}f(A_i)$;
\item $f^{-1}\left(\bigcup_{i\in I}B_i\right)=\bigcup_{i\in I}f^{-1}(B_i)$;
\item $f^{-1}\left(\bigcap_{i\in I}B_i\right)=\bigcap_{i\in I}f^{-1}(B_i)$;
\item $f^{-1}(Y\setminus B)=X\setminus f^{-1}(B)$ for every $B\subseteq Y$;
\item $f(f^{-1}(B))=B\cap\operatorname{range}(f)$ for every $B\subseteq Y$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $y\in f(\bigcup_i A_i)$ iff $y=f(x)$ for some $x\in A_i$ and some $i$, iff $y\in\bigcup_i f(A_i)$.
\item If $y=f(x)$ with $x\in\bigcap_i A_i$, then $y\in f(A_i)$ for every $i$.
\item $x\in f^{-1}(\bigcup_i B_i)$ iff $f(x)\in B_i$ for some $i$, iff $x\in\bigcup_i f^{-1}(B_i)$.
\item $x\in f^{-1}(\bigcap_i B_i)$ iff $f(x)\in B_i$ for every $i$, iff $x\in\bigcap_i f^{-1}(B_i)$.
\item For $x\in X$, $x\in f^{-1}(Y\setminus B) \Longleftrightarrow f(x)\notin B \Longleftrightarrow x\notin f^{-1}(B) \Longleftrightarrow x\in X\setminus f^{-1}(B)$
\item $y\in f(f^{-1}(B))$ iff $y=f(x)$ for some $x$ with $f(x)\in B$, iff $y\in B\cap\operatorname{range}(f)$.
\end{enumerate}
\hlt{Structural Commentary}: Preimages are Boolean-exact; images need not be.\\
\hlt{Key Insight}: Pull back conditions through $f$.
\end{proof}
\end{proposition}

\begin{example} \label{ex:square-map-image-preimage} \hlt{Image and Preimage for the Square Map}\\
Let $f:\R\to\R$ be given by $f(x)=x^2$. Then $f([-2,1])=[0,4]$, $f^{-1}([1,4])=[-2,-1]\cup[1,2]$. The map is not injective because $f(1)=f(-1)$, and it is not surjective as a map $\R\to\R$ because no real number maps to $-1$.
\end{example}

\begin{theorem} \label{thm:quotient-universal-property} \hlt{Universal Property of Quotients}\\
Let $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ be the quotient map.\\
If $L\in\LL(\mathcal{V},\mathcal{W})$ and $\mathcal{U}\subseteq\ker(L)$, then $\exists!\,\overline{L}\in\LL(\mathcal{V}/\mathcal{U},\mathcal{W})$ such that $\overline{L}\pi=L$, $\overline{L}(\mathbf{v}+\mathcal{U})=L\mathbf{v}$.\\
Moreover, $\ker(\overline{L})=\ker(L)/\mathcal{U}$ and $\operatorname{range}(\overline{L})=\operatorname{range}(L)$.
\begin{proof}
If $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$, then $\mathbf{v}-\mathbf{w}\in\mathcal{U}\subseteq\ker(L)$, so $L\mathbf{v}=L\mathbf{w}$. Hence the formula defines a linear map $\overline{L}$ and gives $\overline{L}\pi=L$. If $M\pi=L$, then $M(\mathbf{v}+\mathcal{U})=M\pi(\mathbf{v})=L\mathbf{v}=\overline{L}(\mathbf{v}+\mathcal{U})$, so $M=\overline{L}$.\\
Finally, $\mathbf{v}+\mathcal{U}\in\ker(\overline{L}) \iff L\mathbf{v}=\mathbf{0} \iff \mathbf{v}\in\ker(L)$, giving $\ker(\overline{L})=\ker(L)/\mathcal{U}$.\\
The range equality follows from $\overline{L}(\mathbf{v}+\mathcal{U})=L\mathbf{v}$.\\
\hlt{Structural Commentary}: Maps killing $\mathcal{U}$ descend uniquely to $\mathcal{V}/\mathcal{U}$.\\
\hlt{Key Insight}: Quotients are the natural domain for maps whose ambiguity contains $\mathcal{U}$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:equivalence-relation} \label{def:equivalence-classes-invariants-canonical-forms} \hlt{Equivalence Relations, Classes, Invariants, and Canonical Forms}\\
Let $\sim$ be a relation on a set $X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\sim$ is an \hlt{equivalence relation} if, for all $x,y,z\in X$, $x\sim x$, $x\sim y\Rightarrow y\sim x$, and $(x\sim y\text{ and }y\sim z)\Rightarrow x\sim z$.
\item If $\sim$ is an equivalence relation, the \hlt{equivalence class} of $x\in X$ is $[x]\defeq\{y\in X:y\sim x\}$.
\item A function $f:X\to Y$ is an \hlt{invariant} of $\sim$ if $x\sim y\Rightarrow f(x)=f(y)$, and a \hlt{complete invariant} if $x\sim y\Longleftrightarrow f(x)=f(y)$.
\item A subset $\mathcal{C}\subseteq X$ is an \hlt{set of canonical forms} for $\sim$ if each equivalence class meets $\mathcal{C}$ in exactly one element.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:similarity-equivalence-relation} \hlt{Similarity Is an Equivalence Relation}\\
Similarity is an equivalence relation on $\operatorname{Mat}(n,n,\mathbb{F})$.
\begin{proof}
For any $A$, $A=I_nAI_n^{-1}$, so similarity is reflexive. If $B=PAP^{-1}$, then $A=P^{-1}BP$, so it is symmetric. If $B=PAP^{-1}$ and $C=QBQ^{-1}$, then $C=Q(PAP^{-1})Q^{-1}=(QP)A(QP)^{-1}$ so it is transitive.\\
\hlt{Structural Commentary}: Similarity groups matrices by change of basis.\\
\hlt{Key Insight}: Identity, inverse, and composition give the three equivalence properties.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:similarity-invariants-and-canonical-forms} \hlt{Similarity Invariants and Canonical Forms}\\
Similarity is equality up to change of basis. Trace, determinant, characteristic polynomials, and minimal polynomials are similarity invariants by Theorems \ref{thm:similarity-preserves-trace-and-determinant}, \ref{thm:similarity-preserves-characteristic-polynomial}, and \ref{thm:similarity-preserves-minimal-polynomial}, but they are not generally complete.\\
Canonical forms, such as the Jordan form of Theorem \ref{thm:complex-jordan-form}, choose one structured representative of each similarity class when possible.
\end{perspective}
