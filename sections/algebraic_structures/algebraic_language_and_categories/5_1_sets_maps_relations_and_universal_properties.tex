\subsection{Sets, Maps, Relations, and Universal Properties}

\begin{definition} \label{def:analysis-sets-and-operations} \hlt{Sets and Operations}\\
A \hlt{set} is a collection. Write $x\in E$, $x\notin E$, $\emptyset$, and $A\subseteq B$; $A=B$ iff $A\subseteq B$ and $B\subseteq A$. For $A,B\subseteq X$,
\begin{align}
A\cup B &\defeq \{x\in X:x\in A\text{ or }x\in B\}, \nonumber\\
A\cap B &\defeq \{x\in X:x\in A\text{ and }x\in B\}, \nonumber\\
A\setminus B &\defeq \{x\in X:x\in A\text{ and }x\notin B\}, \nonumber\\
A^c &\defeq X\setminus A. \nonumber
\end{align}
The \hlt{power set} is $\mathcal{P}(X)\defeq\{A:A\subseteq X\}$.
\end{definition}

\begin{definition} \label{def:multisets-and-multiplicity} \hlt{Multisets}\\
Let $S$ be a set. A \hlt{multiset on $S$} is a pair $(S_0,m)$ such that
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $S_0\subseteq S$ is the \hlt{underlying set};
\item $m:S_0\to\Z_{>0}$ is the \hlt{multiplicity function};
\item $m(s)$ is the \hlt{multiplicity} of $s\in S_0$;
\item if $S_0$ is finite, the \hlt{size} is $\lvert m\rvert\defeq\sum_{s\in S_0}m(s)$.
\end{enumerate}
Equivalently, record $(S_0,m)$ as $\{(s,m(s)):s\in S_0\}$.
\end{definition}

\begin{example} \label{ex:multiset-eigenvalue-bookkeeping} \hlt{Multiset Bookkeeping}\\
The list $\{a,a,b,b,b,c\}$ is the multiset on $\{a,b,c\}$ with multiplicities $2,3,1$.
This language is useful whenever order is irrelevant but repetition matters, such as eigenvalues counted with algebraic multiplicity.
\end{example}

\begin{remark} \label{rem:set-builder-notation} \hlt{Set-Builder Notation}\\
$\{x\in X:P(x)\}$ means all $x\in X$ satisfying $P(x)$.\\
For sets $(A_n)$, $\bigcup_nA_n$ means membership in some $A_n$; $\bigcap_nA_n$, membership in every $A_n$.\\
\hlt{Key Insight}: Expand membership conditions.
\end{remark}

\begin{definition} \label{def:indexed-families-unions-intersections} \hlt{Indexed Families of Sets}\\
An \hlt{indexed family} $\{A_i\}_{i\in I}$ is one set $A_i\subseteq X$ for each index $i\in I$. Define
\begin{align}
\bigcup_{i\in I}A_i&\defeq\{x\in X:x\in A_i\text{ for some }i\in I\}, \nonumber\\
\bigcap_{i\in I}A_i&\defeq\{x\in X:x\in A_i\text{ for every }i\in I\}. \nonumber
\end{align}
\hlt{Key Insight}: Union means ``some index''; intersection means ``every index''.
\end{definition}

\begin{method} \label{meth:proof-by-containment} \hlt{Proof by Two Containments}\\
To prove $A=B$, prove both inclusions.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item To prove $A\subseteq B$, let $x\in A$ be arbitrary and use the definitions and hypotheses to show $x\in B$.
\item To prove $B\subseteq A$, let $x\in B$ be arbitrary and use the definitions and hypotheses to show $x\in A$.
\item Once both containments are proved, conclude $A=B$ by extensionality of sets.
\end{enumerate}
If both sides are described by membership conditions, this is a chain, $x\in A \Longleftrightarrow \cdots \Longleftrightarrow x\in B$.
\end{method}

\begin{proposition} \label{prop:analysis-set-algebra} \hlt{Basic Set Algebra}\\
Let $A,B,C\subseteq X$. Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\cap(B\cup C)=(A\cap B)\cup(A\cap C)$;
\item $A\cup(B\cap C)=(A\cup B)\cap(A\cup C)$;
\item $(A\cup B)^c=A^c\cap B^c$;
\item $(A\cap B)^c=A^c\cup B^c$.
\end{enumerate}
\begin{proof}
It suffices to prove (i) and (iii); the others are similar. Let $x\in X$. Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[(i)] $x\in A\cap(B\cup C) \Longleftrightarrow x\in A \text{ and } (x\in B \text{ or } x\in C) \Longleftrightarrow (x\in A\cap B) \text{ or } (x\in A\cap C) \Longleftrightarrow x\in (A\cap B)\cup(A\cap C)$
\item[(iii)] $x\in (A\cup B)^c \Longleftrightarrow x\notin A\cup B \Longleftrightarrow x\notin A \text{ and } x\notin B \Longleftrightarrow x\in A^c\cap B^c$
\end{enumerate}
\hlt{Structural Commentary}: Set algebra is membership logic.\\
\hlt{Key Insight}: Translate sets into conditions on $x$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:metric-de-morgan-laws} \hlt{Arbitrary De Morgan Laws}\\
Let $X$ be a set. For any collection $\{E_\alpha\}$ of subsets of $X$, $\left(\bigcup\limits_\alpha E_\alpha\right)^c=\bigcap\limits_\alpha E_\alpha^c$ and $\left(\bigcap\limits_\alpha E_\alpha\right)^c=\bigcup\limits_\alpha E_\alpha^c$.
\begin{proof}
The first identity follows from membership: $x\in\left(\bigcup\limits_\alpha E_\alpha\right)^c$ $\Longleftrightarrow$ $x\notin E_\alpha \ \forall \alpha$ $\Longleftrightarrow$ $x\in\bigcap\limits_\alpha E_\alpha^c$.\\
The second follows by applying the first identity to the complements $E_\alpha^c$, or by the same membership argument.\\
\hlt{Structural Commentary}: Complements exchange unions and intersections.\\
\hlt{Key Insight}: Negating ``for some'' gives ``for every'', and conversely.
\end{proof}
\end{theorem}

\begin{definition} \label{def:cartesian-product} \hlt{Cartesian Product}\\
For sets $X$ and $Y$, the \hlt{Cartesian product} is $X\times Y\defeq\{(x,y):x\in X,\ y\in Y\}$.\\
The order matters: the first coordinate comes from $X$, and the second comes from $Y$.
\end{definition}

\begin{definition} \label{def:relations-and-equivalence} \hlt{Relations and Equivalence}\\
A \hlt{relation} from $X$ to $Y$ is a subset $R\subseteq X\times Y$. If $(x,y)\in R$, one often writes $xRy$.\\
A relation $\sim$ on $X$ is an \hlt{equivalence relation} if, for all $x,y,z\in X$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $x\sim x$;
\item $x\sim y$ implies $y\sim x$;
\item $x\sim y$ and $y\sim z$ imply $x\sim z$.
\end{enumerate}
The equivalence class of $x$ is $[x]\defeq\{y\in X:y\sim x\}$.
\end{definition}

\begin{proposition} \label{prop:equivalence-relations-partitions} \hlt{Equivalence Classes Partition the Set}\\
Let $\sim$ be an equivalence relation on $X$. Then the equivalence classes form a partition of $X$.
\begin{proof}
Every $x\in X$ lies in $[x]$ since $x\sim x$. If $[x]\cap[y]\neq\emptyset$, choose $z$ in the intersection.\\
Then $x\sim z\sim y$, so $u\in[x]$ implies $u\sim y$ and $u\in[y]$. Hence $[x]\subseteq[y]$, and symmetry gives equality.\\
\hlt{Structural Commentary}: Equivalence means partition.\\
\hlt{Key Insight}: Touching classes coincide.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:number-systems-from-representatives} \hlt{Number Systems from Representatives}\\
Build by representatives, then pass to classes:
\begin{align}
\Z:\ (a,b)\mapsto[(a,b)],\qquad
\Q:\ (a,b),\,b\neq0\mapsto[(a,b)],\qquad
\R:\ \text{cuts or Cauchy data}\mapsto[\cdot]. \nonumber
\end{align}
The pattern is: choose $\sim$, prove $+,\cdot,<,\ldots$ are well defined on $X/{\sim}$, then use the classes as the number system; compare Definition \ref{def:relations-and-equivalence} and Proposition \ref{prop:quotient-space-structure}.\\
\hlt{Key Insight}: Representatives are scaffolding; the quotient laws are the interface.
\end{perspective}

\begin{principle} \label{princ:axiom-of-choice-analysis} \hlt{Choice Principle}\\
For nonempty sets $\{X_i\}_{i\in I}$, the axiom of choice asserts $\exists c:I\to\bigcup\limits_{i\in I}X_i$ such that $c(i)\in X_i\ \forall i\in I$.\\
Use direct construction for finite or explicit choices; use choice only for arbitrary simultaneous choices.
\end{principle}

\begin{definition} \label{def:ordered-set-vocabulary}\label{def:ordered-set-and-bounds}\label{def:supremum-infimum-lub-property} \hlt{Ordered Set Vocabulary}\\
Let $S$ be a set, let $E,C\subseteq S$, and let $\leq_S$ be a relation on $S$. Write $x<_Sy$ iff $x\leq_Sy$ and $x\neq y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\leq_S$ is a \hlt{partial order} if it is reflexive, antisymmetric, and transitive; then $(S,\leq_S)$ is a \hlt{poset}.
\item $\leq_S$ is an \hlt{order} if it is a partial order and any $x,y\in S$ are comparable; then $(S,\leq_S)$ is an \hlt{ordered set}.
\item $C$ is a \hlt{chain} if $\leq_S|_C$ is an order.
\item $S$ is \hlt{well-ordered} if every nonempty $Y\subseteq S$ has $\ell\in Y$ with $\ell\leq_Sy$ for all $y\in Y$.
\item $\beta\in S$ is an \hlt{upper bound} of $E$ if $x\leq_S\beta$ for all $x\in E$; lower bounds reverse the inequality.
\item $E$ is \hlt{bounded above} if it has an upper bound; bounded below is dual.
\item $\sup E$ is the least upper bound of $E$, and $\inf E$ is the greatest lower bound of $E$, when these exist.
\item $m\in E$ is \hlt{maximal} if no $x\in E$ satisfies $m<_Sx$; minimal is dual.
\item $S$ has the \hlt{least-upper-bound property} if every nonempty bounded-above $E\subseteq S$ has $\sup E\in S$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:supremum-need-not-be-attained} \hlt{Suprema Need Not Be Elements}\\
In $\Q$, the set $E_1=\{r\in\Q:r<0\}$ and the set $E_2=\{r\in\Q:r\leq0\}$ both have supremum $0$, but $0\notin E_1$ and $0\in E_2$. If $E=\{1/n:n=1,2,3,\ldots\}$, then $\sup E=1$ while $\inf E=0$, and the infimum is not a member of $E$.
\end{example}

\begin{theorem} \label{thm:lub-property-gives-glb-property} \hlt{Least Upper Bounds Give Greatest Lower Bounds}\\
If $S$ has the least-upper-bound property, then every nonempty bounded-below set $B\subseteq S$ has an infimum.
\begin{proof}
Let $L$ be the set of lower bounds of $B$. Then $L$ is nonempty and every $x\in B$ is an upper bound of $L$, so $\alpha=\sup L$ exists. Since every $x\in B$ bounds $L$ above, $\alpha\leq x$ for all $x\in B$; hence $\alpha$ is a lower bound of $B$.\\
If $\beta>\alpha$, then $\beta\notin L$, so $\beta$ is not a lower bound. Thus $\alpha=\inf B$.\\
\hlt{Structural Commentary}: Infima are suprema of lower bounds.\\
\hlt{Key Insight}: Completeness above gives completeness below.
\end{proof}
\end{theorem}

\begin{principle} \label{princ:zorns-lemma} \hlt{Zorn's Lemma}\\
If a nonempty partially ordered set $X$ has the property that every chain in $X$ has an upper bound in $X$, then $X$ contains a maximal element. This is equivalent in strength to the axiom of choice, Principle \ref{princ:axiom-of-choice-analysis}.
\end{principle}

\begin{definition} \label{def:functions-images-preimages} \hlt{Functions, Images, and Preimages}\\
Let $f:X\to Y$ be a \hlt{function}, assigning each $x\in X$ a unique $f(x)\in Y$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $X=\operatorname{dom}(f)$ is the \hlt{domain} and $Y$ is the \hlt{codomain};
\item $\operatorname{range}(f)\defeq f(X)\defeq\{f(x):x\in X\}$ is the \hlt{range} of $f$;
\item for $A\subseteq X$, $f(A)\defeq\{f(x):x\in A\}$ is the \hlt{image} of $A$;
\item for $B\subseteq Y$, $f^{-1}(B)\defeq\{x\in X:f(x)\in B\}$ is the \hlt{preimage} of $B$; no inverse function is required;
\item for $A\subseteq X$, $f|_A:A\to Y$ is the \hlt{restriction} of $f$ to $A$;
\item if $X\subseteq U$, an \hlt{extension} of $f$ to $U$ is $\widetilde f:U\to Y$ with $\widetilde f|_X=f$;
\item if $Y$ has distinguished $0$, $\operatorname{supp}(f)\defeq\{x\in X:f(x)\neq0\}$;
\item if $g:Y\to Z$, then $g\circ f:X\to Z$ is given by $(g\circ f)(x)=g(f(x))$;
\item $\operatorname{id}_X:X\to X$ denotes the \hlt{identity function}.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:image-preimage-calculus} \hlt{Image and Preimage Calculus}\\
Let $f:X\to Y$, let $\{A_i\}_{i\in I}$ be a nonempty family of subsets of $X$, and let $\{B_i\}_{i\in I}$ be a nonempty family of subsets of $Y$. Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $f\left(\bigcup_{i\in I}A_i\right)=\bigcup_{i\in I}f(A_i)$;
\item $f\left(\bigcap_{i\in I}A_i\right)\subseteq\bigcap_{i\in I}f(A_i)$;
\item $f^{-1}\left(\bigcup_{i\in I}B_i\right)=\bigcup_{i\in I}f^{-1}(B_i)$;
\item $f^{-1}\left(\bigcap_{i\in I}B_i\right)=\bigcap_{i\in I}f^{-1}(B_i)$;
\item $f^{-1}(Y\setminus B)=X\setminus f^{-1}(B)$ for every $B\subseteq Y$;
\item $f(f^{-1}(B))=B\cap\operatorname{range}(f)$ for every $B\subseteq Y$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $y\in f(\bigcup_i A_i)$ iff $y=f(x)$ for some $x\in A_i$ and some $i$, iff $y\in\bigcup_i f(A_i)$.
\item If $y=f(x)$ with $x\in\bigcap_i A_i$, then $y\in f(A_i)$ for every $i$.
\item $x\in f^{-1}(\bigcup_i B_i)$ iff $f(x)\in B_i$ for some $i$, iff $x\in\bigcup_i f^{-1}(B_i)$.
\item $x\in f^{-1}(\bigcap_i B_i)$ iff $f(x)\in B_i$ for every $i$, iff $x\in\bigcap_i f^{-1}(B_i)$.
\item For $x\in X$, $x\in f^{-1}(Y\setminus B) \Longleftrightarrow f(x)\notin B \Longleftrightarrow x\notin f^{-1}(B) \Longleftrightarrow x\in X\setminus f^{-1}(B)$
\item $y\in f(f^{-1}(B))$ iff $y=f(x)$ for some $x$ with $f(x)\in B$, iff $y\in B\cap\operatorname{range}(f)$.
\end{enumerate}
\hlt{Structural Commentary}: Preimages are Boolean-exact; images need not be.\\
\hlt{Key Insight}: Pull back conditions through $f$.
\end{proof}
\end{proposition}

\begin{definition} \label{def:injective-surjective-bijective} \hlt{Injective, Surjective, and Bijective Maps}\\
Let $f:X\to Y$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $f$ is \hlt{injective} if $f(x_1)=f(x_2)$ implies $x_1=x_2$.
\item $f$ is \hlt{surjective} if $\operatorname{range}(f)=Y$.
\item $f$ is \hlt{bijective} if it is both injective and surjective.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:bijective-iff-invertible-function} \hlt{Bijective Means Invertible}\\
A function $f:X\to Y$ is bijective iff there exists a function $g:Y\to X$ s.t. $g\circ f=\operatorname{id}_X$, $f\circ g=\operatorname{id}_Y$.\\
When such a function exists, it is unique and is denoted $f^{-1}:Y\to X$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $f$ is bijective, define $g(y)$ as the unique $x$ with $f(x)=y$. Then $g\circ f=\operatorname{id}_X$ and $f\circ g=\operatorname{id}_Y$.
\item[$(\Leftarrow)$] If such a $g$ exists, applying $g$ to $f(x_1)=f(x_2)$ gives $x_1=x_2$, and $y=f(g(y))$ gives surjectivity.
\end{enumerate}
If $g$ and $h$ are two-sided inverses, then $g(y)=g(f(h(y)))=h(y)$, so the inverse is unique.\\
\hlt{Structural Commentary}: A true inverse undoes $f$ on both sides.\\
\hlt{Key Insight}: Bijective means exactly one preimage per output.
\end{proof}
\end{theorem}

\begin{example} \label{ex:square-map-image-preimage} \hlt{Image and Preimage for the Square Map}\\
Let $f:\R\to\R$ be given by $f(x)=x^2$. Then $f([-2,1])=[0,4]$, $f^{-1}([1,4])=[-2,-1]\cup[1,2]$. The map is not injective because $f(1)=f(-1)$, and it is not surjective as a map $\R\to\R$ because no real number maps to $-1$.
\end{example}

\begin{theorem} \label{thm:quotient-universal-property} \hlt{Universal Property of Quotients}\\
Let $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ be the quotient map.\\
If $L\in\LL(\mathcal{V},\mathcal{W})$ and $\mathcal{U}\subseteq\ker(L)$, then $\exists!\,\overline{L}\in\LL(\mathcal{V}/\mathcal{U},\mathcal{W})$ such that $\overline{L}\pi=L$, $\overline{L}(\mathbf{v}+\mathcal{U})=L\mathbf{v}$.\\
Moreover, $\ker(\overline{L})=\ker(L)/\mathcal{U}$ and $\operatorname{range}(\overline{L})=\operatorname{range}(L)$.
\begin{proof}
If $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$, then $\mathbf{v}-\mathbf{w}\in\mathcal{U}\subseteq\ker(L)$, so $L\mathbf{v}=L\mathbf{w}$. Hence the formula defines a linear map $\overline{L}$ and gives $\overline{L}\pi=L$. If $M\pi=L$, then $M(\mathbf{v}+\mathcal{U})=M\pi(\mathbf{v})=L\mathbf{v}=\overline{L}(\mathbf{v}+\mathcal{U})$, so $M=\overline{L}$.\\
Finally, $\mathbf{v}+\mathcal{U}\in\ker(\overline{L}) \iff L\mathbf{v}=\mathbf{0} \iff \mathbf{v}\in\ker(L)$, giving $\ker(\overline{L})=\ker(L)/\mathcal{U}$.\\
The range equality follows from $\overline{L}(\mathbf{v}+\mathcal{U})=L\mathbf{v}$.\\
\hlt{Structural Commentary}: Maps killing $\mathcal{U}$ descend uniquely to $\mathcal{V}/\mathcal{U}$.\\
\hlt{Key Insight}: Quotients are the natural domain for maps whose ambiguity contains $\mathcal{U}$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:equivalence-relation} \label{def:equivalence-classes-invariants-canonical-forms} \hlt{Equivalence Relations, Classes, Invariants, and Canonical Forms}\\
Let $\sim$ be a relation on a set $X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\sim$ is an \hlt{equivalence relation} if, for all $x,y,z\in X$, $x\sim x$, $x\sim y\Rightarrow y\sim x$, and $(x\sim y\text{ and }y\sim z)\Rightarrow x\sim z$.
\item If $\sim$ is an equivalence relation, the \hlt{equivalence class} of $x\in X$ is $[x]\defeq\{y\in X:y\sim x\}$.
\item A function $f:X\to Y$ is an \hlt{invariant} of $\sim$ if $x\sim y\Rightarrow f(x)=f(y)$, and a \hlt{complete invariant} if $x\sim y\Longleftrightarrow f(x)=f(y)$.
\item A subset $\mathcal{C}\subseteq X$ is an \hlt{set of canonical forms} for $\sim$ if each equivalence class meets $\mathcal{C}$ in exactly one element.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:similarity-equivalence-relation} \hlt{Similarity Is an Equivalence Relation}\\
Similarity is an equivalence relation on $\operatorname{Mat}(n,n,\mathbb{F})$.
\begin{proof}
For any $A$, $A=I_nAI_n^{-1}$, so similarity is reflexive. If $B=PAP^{-1}$, then $A=P^{-1}BP$, so it is symmetric. If $B=PAP^{-1}$ and $C=QBQ^{-1}$, then $C=Q(PAP^{-1})Q^{-1}=(QP)A(QP)^{-1}$ so it is transitive.\\
\hlt{Structural Commentary}: Similarity groups matrices by change of basis.\\
\hlt{Key Insight}: Identity, inverse, and composition give the three equivalence properties.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:similarity-invariants-and-canonical-forms} \hlt{Similarity Invariants and Canonical Forms}\\
Similarity is equality up to change of basis. Trace, determinant, characteristic polynomials, and minimal polynomials are similarity invariants by Theorems \ref{thm:similarity-preserves-trace-and-determinant}, \ref{thm:similarity-preserves-characteristic-polynomial}, and \ref{thm:similarity-preserves-minimal-polynomial}, but they are not generally complete.\\
Canonical forms, such as the Jordan form of Theorem \ref{thm:complex-jordan-form}, choose one structured representative of each similarity class when possible.
\end{perspective}
