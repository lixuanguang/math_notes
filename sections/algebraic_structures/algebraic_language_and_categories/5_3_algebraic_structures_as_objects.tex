\subsection{Algebraic Structures as Objects}

\begin{perspective} \label{per:finite-support-coordinate-model} \hlt{Finite-Support Coordinate Model}\\
Proposition \ref{prop:coordinate-map-isomorphism} identifies an $n$-dimensional vector space with $\mathbb{F}^n$, and Theorem \ref{thm:linear-maps-and-matrices-same-data} identifies linear maps with matrices after bases are chosen. The space $\mathbb{F}^{(\mathbb{N})}$ is the countable standard-coordinate model with finite support. This is the field analogue of the free-module notation $(R^B)_0$.
\end{perspective}
