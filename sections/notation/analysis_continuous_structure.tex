\subsection*{Analysis and Continuous Structure Notation}

\begingroup
\small

\notationgroup{Sets and Functions in Analysis}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\mathcal{P}(X)}{power set of $X$; projection symbols use $P_{\mathcal{U}}$ or $P_j$}
\notationentry{f(A),\ f^{-1}(B)}{image of $A$ / preimage of $B$ under $f$}
\notationentry{\operatorname{range}(f),\ f|_A}{range of $f$ / restriction of $f$ to $A$}
\notationentry{\{A_i\}_{i\in I},\ \bigcup_{i\in I}A_i,\ \bigcap_{i\in I}A_i}{indexed family of sets / indexed union / indexed intersection}
\notationentry{g\circ f,\ \operatorname{id}_X}{composition of functions / identity function on $X$}
\notationentry{J_n,\ \#E,\ A\sim_{\#}B,\ [x]}{finite initial segment $J_0=\emptyset$, $J_n=\{1,\ldots,n\}$ / cardinality of a finite set / same cardinality / equivalence class of $x$ for a specified relation}
\end{tabularx}

\notationgroup{Order and Completeness}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\sup E,\ \inf E}{supremum / infimum of an ordered subset}
\notationentry{\abs{x},\ \abs{a-b}}{real or complex absolute value / distance on the real line}
\notationentry{\overline{\R}}{extended real line $\R\cup\{-\infty,+\infty\}$; the bar is conventional, not topological closure}
\notationentry{x<y,\ x\leq y}{strict order / non-strict order}
\end{tabularx}

\notationgroup{Sequences and Series}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{(a_n),\ a_n\to a}{sequence / convergence of a sequence}
\notationentry{(a_{n_k}),\ n_1<n_2<\cdots}{subsequence indexed by an increasing sequence of positive integers}
\notationentry{s_n=\sum_{k=1}^{n}a_k}{partial sums of a series}
\notationentry{\limsup a_n=s^*,\ \liminf a_n=s_*}{upper / lower limits of a real sequence}
\notationentry{\sum_{n=1}^{\infty} a_n,\ \prod_{n=1}^{\infty} a_n}{infinite series / infinite product}
\end{tabularx}

\notationgroup{Metric and Topological Notation}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{d(x,y),\ B_r(x)}{metric distance / italic open ball of radius $r$ centered at $x$}
\notationentry{B_\varepsilon(a)=V_\varepsilon(a)}{real-line open ball $(a-\varepsilon,a+\varepsilon)$; $V_\varepsilon(a)$ is Abbott's notation}
\notationentry{\overline{B}_r(x),\ \{G_\alpha\}_{\alpha\in A}}{closed ball / indexed open cover}
\notationentry{d_E,\ d_{\max},\ d_{\mathrm{sum}}}{Euclidean, maximum, and sum product metrics}
\notationentry{E^\circ,\ \overline{E},\ E^{\operatorname{acc}}}{interior, topological closure, and limit-point set of $E$}
\notationentry{\operatorname{bd}(E),\ E^c}{topological boundary and complement of $E$}
\notationentry{\diam E}{diameter of a metric set}
\notationentry{X\simeq Y}{homeomorphic metric spaces}
\notationentry{A\cap\overline{B}=\emptyset,\ \overline{A}\cap B=\emptyset}{separation condition for sets $A$ and $B$}
\notationentry{F_\sigma,\ G_\delta,\ \text{meager/thin/first category}}{countable union of closed sets / countable intersection of open sets / countable union of nowhere-dense sets}
\notationentry{\text{residual/thick},\ \text{generic}}{set containing a countable intersection of dense open sets / property holding on a residual set}
\end{tabularx}

\notationgroup{Functions, Differentiation, and Integration}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\lim_{x\to p}f(x),\ f(x+),\ f(x-)}{function limit / right and left limits}
\notationentry{C(X),\ C([a,b]),\ C_b(X),\ \norm{f}_\infty}{continuous function spaces named by domain / bounded continuous functions / supremum norm}
\notationentry{f_n\to f,\ f_n\rightrightarrows f}{pointwise convergence / uniform convergence}
\notationentry{f'(x),\ Df(a),\ D_j f_i,\ \partial_i f,\ J_f(x)}{one-variable derivative, total derivative, $j$th partial of the $i$th component, scalar partial derivative, and Jacobian determinant; $J_f$ is not the finite set $J_n$ or Jordan block $J_k(\lambda)$}
\notationentry{\nabla f,\ D_{\mathbf{u}}f,\ C^r,\ C^\infty,\ C^\omega}{gradient, directional derivative, $r$ times continuously differentiable, smooth, and analytic classes}
\notationentry{\Pi=\{x_0,\ldots,x_n\},\ c_i,\ \Delta x_i,\ \Delta\alpha_i}{partition points, tags, and interval/integrator increments}
\notationentry{R(f,\Pi,\{c_i\}),\ \delta,\ \delta(x)}{tagged Riemann sum / constant mesh bound / gauge; this uppercase $R(\cdots)$ is local to Riemann sums}
\notationentry{U(\Pi,f,\alpha),\ L(\Pi,f,\alpha)}{upper and lower Riemann--Stieltjes sums}
\notationentry{f\in\mathcal{R},\ f\in\mathcal{R}(\alpha)}{Riemann integrability class / Riemann--Stieltjes integrability class with respect to $\alpha$}
\notationentry{\int_a^b f,\ \int_a^b f\,d\alpha,\ \int_E f\,d\mu}{Riemann integral, Riemann--Stieltjes integral, and measure integral}
\notationentry{\int_a^\infty f,\ \int_{-\infty}^{\infty} f}{one-sided and two-sided improper integrals, defined by truncation limits}
\notationentry{\ell(\Pi,\gamma),\ \ell(\gamma)}{polygonal length for a partition / length of a curve}
\notationentry{I^k,\ Q^k,\ \operatorname{supp}(f)}{$k$-cell, standard $k$-simplex, and support of $f$; $I^k$ is not an identity operator}
\notationentry{dx_I,\ dx_I\wedge dx_J}{standard basic form for a multi-index $I$ / wedge product of basic forms}
\notationentry{\omega,\ d\omega,\ T^\ast\omega,\ \partial_{\mathrm{ch}}\Psi}{differential form, exterior derivative, pullback by $T$, and boundary of a chain}
\notationentry{\int_\Psi\omega,\ \partial_{\mathrm{ch}}[\mathbf{p}_0,\ldots,\mathbf{p}_k]}{integration of a form over a chain / oriented simplex boundary}
\notationentry{dV,\ dA,\ ds}{volume, area, and arc-length elements}
\end{tabularx}

\notationgroup{Power Series, Fourier, and Special Functions}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\sum c_n(x-a)^n,\ R}{power series about $a$ / radius of convergence in this context, unrelated to $R(\cdots)$ or the $R$ in $QR$}
\notationentry{I_R,\ K_\rho}{power-series interval $I_R=(-R,R)$ / closed inner interval $K_\rho=[-\rho,\rho]\subset I_R$}
\notationentry{\widehat{f}(m),\ s_N(f;x),\ s_N(f)}{trigonometric Fourier coefficient / $N$th Fourier partial sum value or function}
\notationentry{f\sim_{\mathrm{coef}}\sum \widehat{f}(n)e^{inx},\ f\sim_{\mathrm{coef}}\sum c_n\phi_n}{formal Fourier or orthonormal expansion associated to coefficients}
\notationentry{D_N(t)}{Dirichlet kernel $\sum_{m=-N}^{N}e^{imt}$; uppercase $D_N$ is not derivative notation}
\notationentry{\mathsf{E},\ \mathsf{L},\ \mathsf{C},\ \mathsf{S}}{power-series exponential, logarithm inverse, cosine, and sine}
\notationentry{\Gamma(x),\ \mathrm{B}(x,y)}{gamma function / roman beta function, distinct from balls $B_r(x)$ and matrix variables $B$}
\end{tabularx}

\notationgroup{Measure and Function Spaces}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{E\in\mathcal{M},\ E\in\mathcal{M}(\mu)}{measurable set in a measure space / Rudin measurable set generated by $\mu$}
\notationentry{\mathcal{B}(\R^n),\ \mathcal{L}^n}{Borel $\sigma$-algebra on $\R^n$ / Lebesgue measurable subsets of $\R^n$; not a basis $\mathcal{B}$, a linear-map space $\LL$, or a least-squares set $\operatorname{LS}$}
\notationentry{\mathcal{E},\ \mathcal{M}_F(\mu),\ A\triangle B,\ d_\mu(A,B)}{elementary sets, finitely $\mu$-measurable sets, symmetric difference, and Rudin set distance on sets; $\mathcal{M}$ is measure-theoretic here}
\notationentry{\mu^\ast(E),\ m^\ast(E),\ \mathbf{1}_E}{outer measure of $E$ / Lebesgue outer measure / indicator function of $E$}
\notationentry{\operatorname{vol}_n(B),\ m_n^\ast(\Omega)}{$n$-dimensional box volume / box-cover outer measure $\inf\{\sum_{j\in J}\operatorname{vol}_n(B_j):(B_j)_{j\in J}\text{ covers }\Omega,\ J\text{ at most countable}\}$}
\notationentry{m(E),\ \mu(E)}{elementary volume or Lebesgue measure for set arguments / measure of $E$; not algebraic multiplicity $m_A(\lambda)$ or minimal polynomial $\mu_A$}
\notationentry{\bowtie,\ \{f\bowtie a\}}{one of $>,\geq,<,\leq$ / corresponding level set}
\notationentry{f^+,\ f^-,\ \mathcal{S}^+(f),\ \mathcal{I}_E(s)}{positive/negative parts, simple minorants, and simple-function integral}
\notationentry{f\sim_{\mathrm{a.e.}}g\ \text{on }E,\ \text{a.e.}}{equal except on a measure-zero subset of $E$ / almost everywhere}
\notationentry{E_x,\ E^y,\ f_x,\ f^y}{slices of a set or function in product spaces}
\notationentry{\mathcal{U}f,\ F_a}{undergraph of a nonnegative function / level set $\{x:f(x)>a\}$; $\mathcal{U}f$ is not a vector subspace}
\notationentry{[f:A],\ [B:Q],\ \operatorname{dp}(E)}{average of $f$ over $A$ / density of $B$ in $Q$ / density-point set of $E$}
\notationentry{\mathscr{L}^1(\mu),\ \mathscr{L}^1_{\mathrm{loc}}(\R^n)}{Lebesgue-integrable and locally integrable functions; script $\mathscr{L}$ is distinct from $\mathcal{L}^n$ and $\LL$}
\notationentry{L^p(\mu),\ L^p(E),\ \norm{f}_p}{Lebesgue space with respect to a measure or domain, using the ambient measure when only $E$ is shown / $L^p$ norm}
\end{tabularx}
