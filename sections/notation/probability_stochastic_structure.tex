\subsection*{Probability and Stochastic Structure Notation}

\begingroup
\small

\notationgroup{Probability Spaces and Distributions}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{(\Omega,\mathcal{F},\mathbb{P}),\ \operatorname{Pr}(A)}{probability space and probability of an event; use explicit probability notation so $\mathbb{P}^n_k$ and $\mathbb{F}_0$ remain unambiguous}
\notationentry{X,Y,\ X_\ast\mathbb{P},\ \mathcal{L}(X)}{random variables, pushforward distribution, and law of $X$ when locally declared; not Lebesgue $\mathcal{L}^n$}
\notationentry{F(t),\ g(t),\ \Phi(t)}{distribution function, density, and standard normal distribution function when the context is probabilistic}
\notationentry{\mathbb{E}[X],\ \operatorname{Var}(X),\ \operatorname{Cov}(X,Y)}{expectation, variance, and covariance}
\notationentry{X\perp\!\!\!\perp Y}{independence of random variables or $\sigma$-algebras when declared}
\end{tabularx}

\notationgroup{Stochastic Processes and Markov Structure}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{(X_t)_{t\in T},\ \mathcal{F}_t}{stochastic process and filtration; this $\mathcal{F}_t$ is not the umbral algebra $\mathscr{F}$}
\notationentry{\mathbf{x},\ \mathbf{e},\ A\mathbf{e}=\mathbf{e}}{probability vector, all-ones vector, and row-stochastic Markov matrix convention used in the Markov section}
\notationentry{\pi,\ \pi A=\pi,\ A^p}{stationary distribution as a row vector when declared / $p$th power of a Markov matrix}
\notationentry{P_t,\ p_t(x,y)}{transition semigroup and transition kernel or density, when introduced}
\end{tabularx}

\endgroup
