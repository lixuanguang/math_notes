\subsection*{Probability and Stochastic Structure Notation}

\begingroup
\small

\notationgroup{Probability Spaces and Distributions}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{(\Omega,\mathcal{F},\mathbb{P}),\ \operatorname{Pr}(A)}{probability space / event probability $\operatorname{Pr}(A)=\mathbb{P}(A)$ for $A\in\mathcal{F}$; explicit probability notation keeps $\mathbb{P}^n_k$ and $\mathbb{F}_0$ unambiguous}
\notationentry{X,Y,\ X_\ast\mathbb{P},\ \mathcal{L}(X)}{random variables / pushforward law $X_\ast\mathbb{P}(B)=\mathbb{P}(X^{-1}(B))$ / law of $X$ when locally declared; not Lebesgue $\mathcal{L}^n$}
\notationentry{X\sim F,\ X\disteq Y,\ X_1,\ldots,X_n\text{ iid}}{$X$ has distribution $F$ / $X$ and $Y$ have the same distribution / independent and identically distributed sample}
\notationentry{F(t),\ g(t),\ \Phi(t)}{distribution function $F(t)=\operatorname{Pr}(X\leq t)$ / density $g$ / standard normal distribution function in probabilistic contexts}
\notationentry{\mathbb{E}[X],\ \operatorname{Var}(X),\ \operatorname{Cov}(X,Y)}{expectation / $\operatorname{Var}(X)=\mathbb{E}[(X-\mathbb{E}X)^2]$ / $\operatorname{Cov}(X,Y)=\mathbb{E}[(X-\mathbb{E}X)(Y-\mathbb{E}Y)]$, when finite}
\notationentry{X\perp\!\!\!\perp Y}{independence of random variables, equivalently independence of $\sigma(X)$ and $\sigma(Y)$ when defined}
\notationentry{X\perp\!\!\!\perp Y\mid Z}{conditional independence of $X$ and $Y$ given $Z$}
\notationentry{X_n\xrightarrow{\mathbb{P}}X,\ X_n\rightsquigarrow X,\ X_n\xrightarrow{L^p}X,\ X_n\xrightarrow{\mathrm{a.s.}}X,\ X_n\xrightarrow{qm}X}{convergence in probability / convergence in distribution / convergence in $L^p$ / almost-sure convergence / convergence in quadratic mean}
\end{tabularx}

\notationgroup{Stochastic Processes and Markov Structure}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{(X_t)_{t\in T},\ \mathcal{F}_t}{stochastic process indexed by $T$ / filtration $(\mathcal{F}_t)$ when locally declared; this $\mathcal{F}_t$ is not the umbral algebra $\mathscr{F}$}
\notationentry{\mathbf{x},\ \mathbf{e},\ A\mathbf{e}=\mathbf{e}}{probability vector $\mathbf{x}\geq0$, $\sum_i x_i=1$ / all-ones vector $\mathbf{e}=(1,\ldots,1)^t$ / row-stochastic Markov convention $A=[a_{ij}]\geq0$, $A\mathbf{e}=\mathbf{e}$}
\notationentry{\pi,\ \pi A=\pi,\ A^p}{stationary distribution written as a row probability vector / stationarity equation / $p$-step Markov matrix}
\notationentry{P_t,\ p_t(x,y)}{transition semigroup / transition probability, kernel, or density from $x$ to $y$, when introduced}
\end{tabularx}

\endgroup
