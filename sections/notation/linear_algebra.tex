\subsection*{Linear Algebra Notation}

\begingroup
\small

\notationgroup{Vector Spaces and Lists}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\mathbb{F}}{field of scalars, usually $\mathbb{R}$ or $\mathbb{C}$}
\notationentry{\mathbb{F}^{(\mathbb{N})}}{space of finitely nonzero sequences over $\mathbb{F}$}
\notationentry{\mathbf{u},\mathbf{v},\mathbf{w},\mathbf{0}}{vectors and the zero vector}
\notationentry{\mathcal{U},\mathcal{V},\mathcal{W}}{vector spaces or subspaces, depending on context}
\notationentry{I_{\mathcal{V}},\ I_n,\ 0}{identity operator on $\mathcal{V}$, identity matrix, and zero map or matrix; not the cell $I^k$ or interval $I_R$}
\notationentry{\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)}{finite ordered list, often a basis}
\notationentry{(),\ \beta_{\widehat{j}}}{empty list / list $\beta$ with the $j$th vector omitted}
\notationentry{(\beta,\mathbf{v}),\ \beta_{j\leftarrow \mathbf{v}}}{list with $\mathbf{v}$ appended / list with $j$th entry replaced by $\mathbf{v}$}
\notationentry{\operatorname{span}(E),\ \dim(\mathcal{V})}{span of $E$ / dimension of $\mathcal{V}$}
\notationentry{\operatorname{codim}_{\mathcal{V}}(\mathcal{U})}{codimension of $\mathcal{U}$ in $\mathcal{V}$, equal to $\dim(\mathcal{V}/\mathcal{U})$ when finite}
\notationentry{\bigwedge_{i\in I}\mathcal{U}_i,\ \bigvee_{i\in I}\mathcal{U}_i}{meet / join of a family of subspaces}
\notationentry{\sum_{i\in I}\mathcal{U}_i}{sum of an indexed family of subspaces, equal to the join}
\notationentry{\mathcal{U}+\mathcal{W},\ \mathcal{U}\oplus\mathcal{W},\ \mathcal{V}/\mathcal{U}}{sum, direct sum, quotient space}
\notationentry{\mathcal{V}_1\times\cdots\times\mathcal{V}_m}{product vector space with componentwise operations}
\notationentry{\mathcal{V}\cong\mathcal{W},\ \mathbf{e}_{j,k}}{isomorphic vector spaces / basis vector placed in one product component}
\notationentry{\mathbf{v}+\mathcal{U},\ \pi(\mathbf{v})}{coset modulo $\mathcal{U}$ / quotient map $\mathcal{V}\to\mathcal{V}/\mathcal{U}$ when introduced}
\end{tabularx}

\notationgroup{Linear Maps and Duality}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{T:\mathcal{V}\to\mathcal{W},\ \LL(\mathcal{V},\mathcal{W})}{linear map / space of linear maps}
\notationentry{ST,\ T|_{\mathcal{U}},\ T/\mathcal{U}}{composition $S\circ T$, restriction, and quotient operator}
\notationentry{T^{-1},\ T^m,\ p(T)}{inverse of a bijective map or operator, operator power, and polynomial applied to an operator}
\notationentry{\ker(T),\ \operatorname{range}(T)}{kernel and range of $T$}
\notationentry{\mathcal{V}_{\mathbb{C}},\ T_{\mathbb{C}}}{complexification of a real vector space / complexification of a real operator}
\notationentry{\operatorname{graph}(T),\ [S,T]}{graph of a map / commutator $ST-TS$}
\notationentry{\operatorname{Comm}(\mathcal{F}),\ \operatorname{Comm}(A)}{commutant of a matrix family / commutant of one matrix}
\notationentry{\mathcal{V}',\ \beta',\ T'}{dual space, dual basis, and dual map; prime here is not derivative notation}
\notationentry{\varphi,\psi,\ \mathcal{U}^0}{linear functionals / annihilator}
\notationentry{\mathcal{V}'',\ \Lambda_{\mathcal{V}}}{double dual / natural evaluation map}
\notationentry{[\mathbf{v}]_{\beta},\ [T]_{\gamma\leftarrow\beta}}{coordinate vector with respect to basis $\beta$ / matrix representation}
\notationentry{C_{\gamma\leftarrow\beta}}{change-of-basis matrix from $\beta$-coordinates to $\gamma$-coordinates}
\end{tabularx}

\notationgroup{Matrices and Entrywise Structure}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{A,B,C,\ \operatorname{Mat}(m,n,\mathbb{F})}{matrix variables in linear-algebra contexts / $m\times n$ matrices over $\mathbb{F}$}
\notationentry{A_{ij},\ A_{j,\cdot},\ A_{\cdot,k}}{matrix entry, $j$th row, $k$th column}
\notationentry{[a_{ij}],\ E_{j,k},\ A\mathbf{x}}{matrix written by entries / matrix unit / matrix action}
\notationentry{A^{-1},\ A^t,\ A^T,\ A^*}{inverse, transpose, alternate transpose notation, and adjoint or conjugate transpose}
\notationentry{\operatorname{diag}(\lambda_1,\ldots,\lambda_n)}{diagonal matrix with diagonal entries $\lambda_1,\ldots,\lambda_n$}
\notationentry{\overline{z},\ \operatorname{Re}z,\ \operatorname{Im}z}{complex conjugate, real part, and imaginary part; overline on sets means closure}
\notationentry{\operatorname{tr}(A),\det(A)}{trace and determinant}
\notationentry{\mathfrak{S}_n,\ \sign(\sigma),\ \operatorname{adj}(A)}{permutation group, permutation sign, adjugate matrix}
\notationentry{M_{ij},\ C_{ij},\ A_j(\mathbf{b})}{minor matrix, cofactor, and the matrix obtained from $A$ by replacing column $j$ with $\mathbf{b}$}
\notationentry{\operatorname{col}(A),\ \operatorname{row}(A)}{column space and row space of $A$}
\notationentry{\operatorname{rank}(A),\ \operatorname{nullity}(A)}{rank and nullity of $A$}
\notationentry{A\oplus B,\ A\otimes B}{matrix direct sum / Kronecker product; vector-space direct sums use subspace notation}
\notationentry{A\otimes I_q+I_n\otimes B}{Kronecker sum of square matrices $A$ and $B$ in compatible dimensions}
\notationentry{A\circ B,\ \operatorname{vec}(A)}{Hadamard product / column-stacking vectorisation}
\notationentry{M/A,\ M/D}{Schur complement of the block $A$ or $D$ in $M$; not quotient-space notation}
\notationentry{B,\ L=B^tB}{directed incidence matrix and incidence Laplacian when a graph context is active}
\end{tabularx}

\notationgroup{Inner Product Geometry}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\norm{\mathbf{v}},\ \normop{T}}{vector norm / operator norm}
\notationentry{\norm{\cdot}_1,\ \norm{\cdot}_2,\ \norm{\cdot}_{\infty}}{coordinate one-norm, Euclidean norm on vectors or spectral norm on matrices/operators, and sup norm}
\notationentry{\norm{\cdot}_F}{Frobenius norm}
\notationentry{\innerprod{\mathbf{u},\mathbf{v}},\ \innerprod[F]{A,B}}{inner product / Frobenius inner product}
\notationentry{\mathbf{u}\perp\mathbf{v},\ \mathcal{U}^{\perp},\ \delta_{ij}}{orthogonal vectors, orthogonal complement, Kronecker delta}
\notationentry{\operatorname{proj}_{\mathbf{v}}(\mathbf{u}),\ P_{\mathcal{U}}}{projection of $\mathbf{u}$ onto the line spanned by $\mathbf{v}$ / orthogonal projection onto $\mathcal{U}$}
\notationentry{d(\mathbf{v},\mathcal{U})}{distance from $\mathbf{v}$ to the set or subspace $\mathcal{U}$}
\notationentry{\operatorname{Gram}(\mathbf{u}_1,\ldots,\mathbf{u}_n)}{Gram matrix of $\mathbf{u}_1,\ldots,\mathbf{u}_n$}
\notationentry{\operatorname{GramDet}(\mathbf{u}_1,\ldots,\mathbf{u}_n)}{Gram determinant}
\notationentry{P_{\mathbf{u}},\ U_{\mathbf{w}}}{rank-one projection matrix / Householder matrix, both determined by the vector subscript}
\notationentry{\gamma=(\mathbf{e}_1,\ldots,\mathbf{e}_n)}{often an orthonormal basis}
\notationentry{\mathbf{e}=(1,\ldots,1)^t}{all-ones vector}
\notationentry{\operatorname{Ray}_A(\mathbf{x}),\ \mathbf{x}^*A\mathbf{x}}{Rayleigh quotient notation or unit-vector form / Hermitian quadratic form}
\end{tabularx}

\notationgroup{Spectral and Polynomial Structure}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{E_{\lambda}(T),\ \operatorname{spec}(T)}{eigenspace for eigenvalue $\lambda$ / spectrum}
\notationentry{\lambda_i(A),\ m_A(\lambda)}{eigenvalues in the locally declared order / algebraic multiplicity of $\lambda$ as a root of $p_A$, not measure $m(E)$}
\notationentry{\mathcal{G}_{\lambda}(T),\ J_k(\lambda)}{generalized eigenspace / Jordan block of size $k$}
\notationentry{p_A,\ q_T}{matrix characteristic polynomial / operator characteristic polynomial}
\notationentry{\mu_A,\ \mu_T,\ w_p(\lambda,A)}{minimal polynomials subscripted by matrix or operator / Weyr rank-drop count; not measure $\mu(E)$}
\notationentry{P_j,\ T=\sum_j\mu_jP_j}{spectral projections in a spectral resolution / corresponding decomposition with distinct eigenvalues $\mu_j$}
\notationentry{\sqrt{T},\ A^{1/2},\ \rho(A)}{positive square root / spectral radius of $A$, not the interval parameter in $K_\rho$}
\notationentry{\mathbb{F}[z]}{polynomials in $z$ over $\mathbb{F}$}
\notationentry{e^{tA}}{matrix exponential, defined by the convergent power series in $tA$}
\end{tabularx}

\notationgroup{Matrix Analysis and Numerical Structure}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{A\succeq0,\ A\succ0}{positive semidefinite and positive definite matrices}
\notationentry{\operatorname{in}(A),\ n_+(A),n_-(A),n_0(A)}{inertia and positive, negative, zero eigenvalue counts}
\notationentry{A=QR,\ A=LL^*,\ A=LDL^t}{QR factorization, Cholesky factorization, and symmetric $LDL^t$ factorization}
\notationentry{A=V\Sigma W^*,\ A=X\Sigma_rY^*}{singular value decomposition / compact SVD}
\notationentry{\sigma_j(A),\ \sigma_{\max}(A),\ \sigma_{\min}(A)}{singular values and extreme singular values}
\notationentry{\abs{T},\ \abs{A},\ \operatorname{vol}(\Omega)}{operator or matrix modulus / volume of a set; scalar absolute value remains $\abs{x}$}
\notationentry{A^\dagger,\ A_k}{Moore--Penrose pseudoinverse / rank-$k$ truncated SVD approximation}
\notationentry{\kappa_2(A),\ \Delta A,\ \Delta\mathbf{x}}{spectral condition number / matrix and solution perturbations}
\notationentry{\Delta(A)}{defect from normality, not a perturbation of $A$}
\notationentry{R_k(A),\ R_k^{\mathrm{off}}(A),\ R_{\max}(A)}{absolute row sum, deleted row sum, and maximum absolute row sum}
\notationentry{\operatorname{Ger}_k(A),\ \operatorname{Ger}(A)}{Gersgorin disk and Gersgorin union}
\notationentry{\operatorname{tridiag}(a,b,c)}{tridiagonal matrix with subdiagonal $a$, diagonal $b$, and superdiagonal $c$}
\notationentry{w,\ F_n,\ Q_n,\ \omega_n}{half-bandwidth / Fourier matrix / normalized Fourier matrix / primitive $n$th root used for DFT}
\notationentry{\mathcal{K}_m(A,\mathbf{b})}{Krylov subspace generated by $(A,\mathbf{b})$}
\notationentry{Q_m,\ H_m}{Krylov basis matrix / projected Hessenberg or tridiagonal matrix}
\notationentry{A=S-T,\ B=S^{-1}T,\ \mathbf{e}_k}{matrix splitting, iteration matrix, and stationary-method error vector}
\notationentry{A=D+L+U,\ B_J,\ B_{\operatorname{GS}}}{diagonal/lower/upper splitting and Jacobi/Gauss--Seidel iteration matrices}
\notationentry{\omega,\ S_\omega,\ T_\omega,\ B_\omega}{SOR parameter, SOR splitting matrices, and SOR iteration matrix}
\notationentry{M,\ M^{-1}A,\ AM^{-1}}{preconditioner and left/right preconditioned matrices}
\end{tabularx}

\endgroup
