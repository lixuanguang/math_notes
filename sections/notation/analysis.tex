\subsection*{Analysis Notation}

\begingroup
\small

\notationgroup{Sets and Functions in Analysis}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\mathcal{P}(X)}{power set of $X$}
\notationentry{f(A),\ f^{-1}(B)}{image of $A$ / preimage of $B$ under $f$}
\notationentry{\operatorname{range}(f),\ f|_A}{range of $f$ / restriction of $f$ to $A$}
\notationentry{g\circ f,\ \operatorname{id}_X}{composition of functions / identity function on $X$}
\notationentry{\#E,\ A\sim_{\#}B,\ [x]}{cardinality of a finite set / same cardinality / equivalence class}
\end{tabularx}

\notationgroup{Order and Completeness}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\sup E,\ \inf E}{supremum / infimum of an ordered subset}
\notationentry{\abs{x},\ \abs{a-b}}{absolute value or modulus / distance on the real line}
\notationentry{\overline{\R}}{extended real line $\R\cup\{-\infty,+\infty\}$}
\notationentry{x<y,\ x\leq y}{strict order / non-strict order}
\end{tabularx}

\notationgroup{Sequences and Series}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{(a_n),\ a_n\to a}{sequence / convergence of a sequence}
\notationentry{(a_{n_k}),\ n_1<n_2<\cdots}{subsequence indexed by an increasing sequence of positive integers}
\notationentry{s_n=\sum_{k=1}^{n}a_k}{partial sums of a series}
\notationentry{\limsup a_n=s^*,\ \liminf a_n=s_*}{upper / lower limits of a real sequence}
\notationentry{\sum_{n=1}^{\infty} a_n,\ \prod_{n=1}^{\infty} a_n}{infinite series / infinite product}
\end{tabularx}

\notationgroup{Metric and Topological Notation}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{d(x,y),\ B_r(x)}{metric distance / open ball of radius $r$ centered at $x$}
\notationentry{B_\varepsilon(a)=V_\varepsilon(a)}{real-line epsilon-neighborhood $(a-\varepsilon,a+\varepsilon)$; $V_\varepsilon(a)$ is Abbott's notation}
\notationentry{\overline{B}_r(x),\ \{G_\alpha\}_{\alpha\in A}}{closed ball / indexed open cover}
\notationentry{d_E,\ d_{\max},\ d_{\mathrm{sum}}}{Euclidean, maximum, and sum product metrics}
\notationentry{E^\circ,\ \overline{E},\ E^{\operatorname{acc}}}{interior, closure, and limit-point set of $E$}
\notationentry{\operatorname{bd}(E),\ E^c}{topological boundary and complement of $E$}
\notationentry{\diam E}{diameter of a metric set}
\notationentry{X\simeq Y}{homeomorphic metric spaces}
\notationentry{A\cap\overline{B}=\emptyset,\ \overline{A}\cap B=\emptyset}{separation condition for sets $A$ and $B$}
\notationentry{F_\sigma,\ G_\delta,\ \text{first category}}{countable union of closed sets / countable intersection of open sets / countable union of nowhere-dense sets}
\end{tabularx}

\notationgroup{Functions, Differentiation, and Integration}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\lim_{x\to p}f(x),\ f(x+),\ f(x-)}{function limit / right and left limits}
\notationentry{C(X),\ C([a,b]),\ C_b(X),\ \norm{f}_\infty}{continuous functions / bounded continuous functions / supremum norm}
\notationentry{f_n\to f,\ f_n\rightrightarrows f}{pointwise convergence / uniform convergence}
\notationentry{f'(x),\ Df(a),\ D_j f_i,\ \partial_i f,\ J_f(x)}{one-variable derivative, total derivative, partial derivative, and Jacobian determinant}
\notationentry{\nabla f,\ D_{\mathbf{u}}f,\ C^r,\ C^\infty,\ C^\omega}{gradient, directional derivative, $r$ times continuously differentiable, smooth, and analytic classes}
\notationentry{\Pi=\{x_0,\ldots,x_n\},\ c_i,\ \Delta x_i,\ \Delta\alpha_i}{partition points, tags, and interval/integrator increments}
\notationentry{R(f,\Pi,\{c_i\}),\ \delta,\ \delta(x)}{tagged Riemann sum / constant mesh bound / gauge}
\notationentry{U(\Pi,f,\alpha),\ L(\Pi,f,\alpha)}{upper and lower Riemann--Stieltjes sums}
\notationentry{f\in\mathcal{R},\ f\in\mathcal{R}(\alpha)}{Riemann integrable / Riemann--Stieltjes integrable with respect to $\alpha$}
\notationentry{\int_a^b f,\ \int_a^b f\,d\alpha,\ \int_E f\,d\mu}{Riemann integral, Riemann--Stieltjes integral, and measure integral}
\notationentry{\int_a^\infty f,\ \int_{-\infty}^{\infty} f}{one-sided and two-sided improper integrals, defined by truncation limits}
\notationentry{\ell(\Pi,\gamma),\ \ell(\gamma)}{polygonal length for a partition / length of a curve}
\notationentry{I^k,\ Q^k,\ \operatorname{supp}(f)}{$k$-cell, standard $k$-simplex, and support of $f$}
\notationentry{dx_I,\ dx_I\wedge dx_J}{standard basic form / wedge product of basic forms}
\notationentry{\omega,\ d\omega,\ T^\ast\omega,\ \partial_{\mathrm{ch}}\Psi}{differential form, exterior derivative, pullback, and boundary of a chain}
\notationentry{\int_\Psi\omega,\ \partial_{\mathrm{ch}}[\mathbf{p}_0,\ldots,\mathbf{p}_k]}{integration of a form over a chain / oriented simplex boundary}
\notationentry{dV,\ dA,\ ds}{volume, area, and arc-length elements}
\end{tabularx}

\notationgroup{Power Series, Fourier, and Special Functions}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\sum c_n(x-a)^n,\ R}{power series about $a$ / radius of convergence in this context}
\notationentry{I_R,\ K_\rho}{interval $I_R=(-R,R)$ / closed inner interval $K_\rho=[-\rho,\rho]\subset I_R$}
\notationentry{\widehat{f}(m),\ s_N(f;x),\ s_N(f)}{trigonometric Fourier coefficient / $N$th Fourier partial sum value or function}
\notationentry{f\sim_{\mathrm{coef}}\sum \widehat{f}(n)e^{inx},\ f\sim_{\mathrm{coef}}\sum c_n\phi_n}{formal Fourier or orthonormal expansion associated to coefficients}
\notationentry{D_N(t)}{Dirichlet kernel $\sum_{m=-N}^{N}e^{imt}$}
\notationentry{\mathsf{E},\ \mathsf{L},\ \mathsf{C},\ \mathsf{S}}{power-series exponential, logarithm inverse, cosine, and sine}
\notationentry{\Gamma(x),\ \mathrm{B}(x,y)}{gamma and beta functions}
\end{tabularx}

\notationgroup{Measure and Function Spaces}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{E\in\mathcal{M},\ E\in\mathcal{M}(\mu)}{measurable set in a measure space / Rudin measurable set generated by $\mu$}
\notationentry{\mathcal{E},\ \mathcal{M}_F(\mu),\ A\triangle B,\ d_\mu(A,B)}{elementary sets, finitely $\mu$-measurable sets, symmetric difference, and Rudin set distance}
\notationentry{\mu^\ast(E),\ \mathbf{1}_E}{outer measure of $E$ / indicator function of $E$}
\notationentry{m(E),\ \mu(E)}{elementary volume or Lebesgue measure / measure of $E$}
\notationentry{\bowtie,\ \{f\bowtie a\}}{one of $>,\geq,<,\leq$ / corresponding level set}
\notationentry{f^+,\ f^-,\ \mathcal{S}^+(f),\ \mathcal{I}_E(s)}{positive/negative parts, simple minorants, and simple-function integral}
\notationentry{f\sim_{\mathrm{a.e.}}g\ \text{on }E,\ \text{a.e.}}{equal except on a measure-zero subset of $E$ / almost everywhere}
\notationentry{\mathscr{L}^1(\mu)}{Lebesgue-integrable functions with respect to $\mu$}
\notationentry{L^p(\mu),\ L^p(E),\ \norm{f}_p}{Lebesgue space with respect to a measure or domain / $L^p$ norm}
\end{tabularx}

\endgroup
