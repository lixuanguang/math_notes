\subsection*{Algebraic Structures Notation}

\begingroup
\small

\notationgroup{Algebraic Language and Categories}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\mathcal{C},\ A\xrightarrow{f}B,\ \operatorname{id}_A}{category, morphism, and identity morphism; $\mathcal{C}$ is local unless the ambient category has been declared}
\notationentry{\operatorname{Hom}_{\mathcal{C}}(A,B),\ g\circ f}{morphism set in a category / composition of composable morphisms}
\notationentry{A\simeq B,\ F:\mathcal{C}\to\mathcal{D}}{isomorphic objects / functor between categories}
\end{tabularx}

\notationgroup{Groups, Actions, Rings, and Ideals}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{G,H,K,\ e_G,\ \langle S\rangle}{groups or subgroups, identity of $G$, and subgroup or algebraic object generated by $S$}
\notationentry{H\leq G,\ H\trianglelefteq G,\ G/H,\ [G:H]}{subgroup, normal subgroup, quotient group, and subgroup index}
\notationentry{\operatorname{ord}(g),\ \operatorname{Aut}(G)}{order of an element / automorphism group}
\notationentry{R,S,\ 0_R,\ 1_R,\ R^\times}{rings, zero and identity in $R$, and unit group of $R$}
\notationentry{Z(R),\ \operatorname{char}(R)}{center and characteristic of a ring}
\notationentry{I,J,\ \mathfrak{a},\mathfrak{p},\ R/I}{ideals, usually with fraktur for distinguished ideals such as prime ideals / quotient ring}
\notationentry{R[x],\ \mathbb{F}[z],\ \deg f}{polynomial rings and degree of a polynomial}
\notationentry{U_n,\ \Omega_n,\ \Phi_n(x),\ \varphi(n)}{$n$th roots of unity, primitive $n$th roots, $n$th cyclotomic polynomial, and Euler totient; $\Omega_n$ is algebraic here, not a domain}
\end{tabularx}

\notationgroup{Modules and Associative Algebras}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{M,N,\ S\leq M,\ M/N}{modules, submodule, and quotient module}
\notationentry{\operatorname{Hom}_R(M,N),\ \operatorname{End}_R(M)}{$R$-module homomorphisms and endomorphisms}
\notationentry{\ker\tau,\ \operatorname{im}\tau,\ \overline{\tau}}{kernel, image, and induced map on a quotient for a homomorphism $\tau$}
\notationentry{\bigoplus_i M_i,\ M=N\oplus H}{direct sum of modules / internal direct-sum decomposition}
\notationentry{\mathcal{C}_S(N),\ \mathcal{L}_\sigma,\ \mathcal{R}_\sigma}{complements of $S$ in $N$, left inverses of $\sigma$, and right inverses of $\sigma$; subscript prevents collision with $\LL$ or $\mathcal{L}^n$}
\notationentry{\mathcal{F}(\mathcal{V})}{finite-rank operators on $\mathcal{V}$ in associative-algebra contexts; not a family of functions}
\end{tabularx}

\endgroup
