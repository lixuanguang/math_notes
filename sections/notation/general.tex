\subsection*{General Notation}

\begingroup
\small

\noindent\emph{Convention.} Notation is scoped by subject, argument type, font, decoration, and any local definition in force. Bare letters are usually local variables; persistent meanings are carried by subscripts, superscripts, operators, fonts, or declared argument patterns. When a glyph is reused, the distinguishing feature is recorded below.

\notationgroup{Set, Function, and Logical Notation}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{x\in E,\ x\notin E}{membership / non-membership}
\notationentry{E\subseteq F,\ \emptyset}{subset relation / empty set}
\notationentry{X\times Y,\ f:X\to Y}{Cartesian product / function from $X$ to $Y$}
\notationentry{\Longrightarrow,\ \Longleftrightarrow}{implies / if and only if}
\notationentry{\defeq}{defined to be equal to}
\end{tabularx}

\notationgroup{Number, Size, and Scalar Notation}

\noindent\begin{tabularx}{\textwidth}{@{}p{0.38\textwidth}X@{}}
\notationentry{\N,\Z,\Q,\R,\C}{standard number systems}
\notationentry{\mathbb{F},\ \mathbb{F}_0}{ambient scalar field when not fixed as $\R$ or $\C$ / prime subfield of $\mathbb{F}$; probability notation should use explicit $\operatorname{Pr}$ or locally declared probability spaces}
\notationentry{\abs{x},\ \abs{E},\ \norm{x}}{absolute value or complex modulus for scalars; length or cardinality only when the argument is an interval or discrete set; norm of a vector, function, or operator}
\notationentry{a,b,c}{generic scalars or interval endpoints, with meaning determined by context}
\notationentry{\Phi_n(x)}{$n$th cyclotomic polynomial; this avoids conflict with $Q_n$ for normalized Fourier matrices}
\end{tabularx}

\endgroup
