\subsubsection{Spectral Theory}

Spectral theory studies an operator through algebraic expressions in the operator and through bases that reveal invariant flags. Polynomial expressions produce eigenvalues over $\mathbb{C}$, and eigenvalues make upper-triangular representations possible.

\begin{definition} \label{def:operator-powers} \hlt{Powers of an Operator}\\
Let $T\in\LL(\mathcal{V})$. For a positive integer $m$, $T^m\defeq \underbrace{T\cdots T}_{m\text{ times}}$ and $T^0\defeq I_{\mathcal{V}}$.\\
If $T$ is invertible, then $T^{-m}\defeq(T^{-1})^m$ for positive integers $m$.
\end{definition}

\begin{definition} \label{def:polynomial-applied-to-operator} \hlt{Polynomial Applied to an Operator}\\
Let $T\in\LL(\mathcal{V})$ and let $p\in\mathcal{P}(\mathbb{F})$ be $p(z)=a_0+a_1z+\cdots+a_mz^m$. Then $p(T)\defeq a_0I_{\mathcal{V}}+a_1T+\cdots+a_mT^m$.
\end{definition}

\begin{definition} \label{def:annihilating-polynomial} \hlt{Annihilating Polynomial}\\
Let $T\in\LL(\mathcal{V})$. A nonzero polynomial $p\in\mathcal{P}(\mathbb{F})$ \hlt{annihilates} $T$ if $p(T)=0$.
\end{definition}

\begin{definition} \label{def:product-of-polynomials} \hlt{Product of Polynomials}\\
If $p,q\in\mathcal{P}(\mathbb{F})$, their product $pq\in\mathcal{P}(\mathbb{F})$ is defined by $(pq)(z)=p(z)q(z)$ for $z\in\mathbb{F}$.
\end{definition}

\begin{proposition} \label{prop:operator-polynomial-products} \hlt{Multiplicative Properties of Operator Polynomials}\\
Let $p,q\in\mathcal{P}(\mathbb{F})$ and $T\in\LL(\mathcal{V})$. Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(pq)(T)=p(T)q(T)$;
\item $p(T)q(T)=q(T)p(T)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Write $p(z)=\sum_{j=0}^{m}a_jz^j$ and $q(z)=\sum_{k=0}^{n}b_kz^k$. Then $(pq)(z)=\sum\limits_{j=0}^{m}\sum\limits_{k=0}^{n}a_jb_kz^{j+k}$.\\
Substituting $T$ gives $(pq)(T)=\sum\limits_{j=0}^{m}\sum\limits_{k=0}^{n}a_jb_kT^{j+k} =\left(\sum\limits_{j=0}^{m}a_jT^j\right)\left(\sum\limits_{k=0}^{n}b_kT^k\right) =p(T)q(T)$.
\item Since polynomial multiplication is commutative, $pq=qp$.\\
By part (i), $p(T)q(T)=(pq)(T)=(qp)(T)=q(T)p(T)$.
\end{enumerate}
\hlt{Structural Commentary}: Polynomial evaluation preserves products.\\
\hlt{Key Insight}: Expanding a polynomial product works the same with $T$ as with $z$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:finite-dimensional-operators-have-annihilating-polynomials} \hlt{Existence of Annihilating Polynomials}\\
Let $\mathcal{V}$ be finite-dimensional with $n=\dim(\mathcal{V})>0$, and let $T\in\LL(\mathcal{V})$.\\
Then $T$ has a nonconstant annihilating polynomial of degree at most $n^2$.
\begin{proof}
By Theorem \ref{thm:linear-maps-and-matrices-same-data}, the vector space $\LL(\mathcal{V})$ has dimension $n^2$, so the $n^2+1$ operators $I_{\mathcal{V}},T,T^2,\ldots,T^{n^2}$ are linearly dependent. Hence there are scalars $a_0,\ldots,a_{n^2}$, not all zero, such that $\sum_{j=0}^{n^2}a_jT^j=0$.\\
If $a_1=\cdots=a_{n^2}=0$, then $a_0I_{\mathcal{V}}=0$, forcing $a_0=0$ because $\mathcal{V}\neq\{\mathbf{0}\}$, a contradiction. Thus $p(z)=\sum_{j=0}^{n^2}a_jz^j$ is a nonconstant annihilating polynomial.\\
\hlt{Structural Commentary}: Finite-dimensionality makes the powers of one operator linearly dependent inside $\LL(\mathcal{V})$.\\
\hlt{Key Insight}: A dependence among $I_{\mathcal{V}},T,\ldots,T^{n^2}$ is exactly a polynomial equation satisfied by $T$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:polynomial-eigenvector-evaluation} \hlt{Polynomial Evaluation on Eigenvectors}\\
Let $T\in\LL(\mathcal{V})$, let $p\in\mathcal{P}(\mathbb{F})$, and suppose $T\mathbf{v}=\lambda\mathbf{v}$ with $\mathbf{v}\neq\mathbf{0}$. Then $p(T)\mathbf{v}=p(\lambda)\mathbf{v}$.
\begin{proof}
Induct on $j$ to get $T^j\mathbf{v}=\lambda^j\mathbf{v}$ for every $j\geq0$.\\
If $p(z)=\sum\limits_{j=0}^{m}a_jz^j$, then $p(T)\mathbf{v} =\sum\limits_{j=0}^{m}a_jT^j\mathbf{v} =\sum\limits_{j=0}^{m}a_j\lambda^j\mathbf{v} =p(\lambda)\mathbf{v}$.\\
\hlt{Structural Commentary}: On a one-dimensional eigenspace, $T$ acts like the scalar $\lambda$.\\
\hlt{Key Insight}: Polynomial expressions in $T$ reduce to the same polynomial evaluated at $\lambda$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:annihilating-polynomials-control-spectrum} \hlt{Annihilating Polynomials Control the Spectrum}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and let $p$ be a nonconstant annihilating polynomial for $T$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Every eigenvalue of $T$ is a root of $p$.
\item If $\mathbb{F}=\mathbb{C}$, then at least one root of $p$ is an eigenvalue of $T$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $T\mathbf{v}=\lambda\mathbf{v}$ with $\mathbf{v}\neq\mathbf{0}$, then Proposition \ref{prop:polynomial-eigenvector-evaluation} gives $0=p(T)\mathbf{v}=p(\lambda)\mathbf{v}$. Since $\mathbf{v}\neq\mathbf{0}$, $p(\lambda)=0$.
\item Factor $p(z)=c(z-\lambda_1)\cdots(z-\lambda_m)$ over $\mathbb{C}$, with $c\neq0$. Then $0=p(T)=c(T-\lambda_1I_{\mathcal{V}})\cdots(T-\lambda_mI_{\mathcal{V}})$. At least one factor $T-\lambda_jI_{\mathcal{V}}$ is not injective; otherwise product would be injective. Hence $\lambda_j$ is an eigenvalue.
\end{enumerate}
\hlt{Structural Commentary}: Annihilating polynomial restricts spectrum to its roots; over $\mathbb{C}$, one root must appear.\\
\hlt{Key Insight}: Eigenvectors turn $p(T)$ into $p(\lambda)$.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:kernel-range-polynomial-in-operator-invariant} \label{prop:kernel-of-polynomial-in-operator-is-invariant} \hlt{Kernel and Range of a Polynomial in an Operator Are Invariant}\\
Let $T\in\LL(\mathcal{V})$ and $p\in\mathcal{P}(\mathbb{F})$. Then $\ker(p(T))$ and $\operatorname{range}(p(T))$ are invariant under $T$.
\begin{proof}
If $\mathbf{v}\in\ker(p(T))$, then $p(T)\mathbf{v}=\mathbf{0}$.\\
Since $p(T)$ commutes with $T$ by Proposition \ref{prop:operator-polynomial-products}, $p(T)(T\mathbf{v})=T(p(T)\mathbf{v})=T\mathbf{0}=\mathbf{0}$.\\
If $\mathbf{v}\in\operatorname{range}(p(T))$, write $\mathbf{v}=p(T)\mathbf{u}$. Then $T\mathbf{v}=Tp(T)\mathbf{u}=p(T)(T\mathbf{u})$, so $T\mathbf{v}\in\operatorname{range}(p(T))$.\\
\hlt{Structural Commentary}: Polynomial kernels and ranges are stable under the original operator.\\
\hlt{Key Insight}: Since $T$ commutes with $p(T)$, it preserves both $\ker p(T)$ and $\operatorname{range}p(T)$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:complex-operator-has-eigenvalue} \hlt{Complex Operators Have Eigenvalues}\\
Every operator on a finite-dimensional, nonzero complex vector space has an eigenvalue.
\begin{proof}
Let $T\in\LL(\mathcal{V})$. By Proposition \ref{prop:finite-dimensional-operators-have-annihilating-polynomials}, $T$ has a nonconstant annihilating polynomial. Since $\mathcal{V}$ is complex and nonzero, Theorem \ref{thm:annihilating-polynomials-control-spectrum} implies that one root of this polynomial is an eigenvalue of $T$.\\
\hlt{Structural Commentary}: Finite dimension gives an annihilating polynomial; over $\mathbb{C}$, it factors linearly.\\
\hlt{Key Insight}: One noninjective factor produces an eigenvector.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:commuting-family-common-eigenvector} \hlt{Common Eigenvector for a Commuting Family}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{C}$, and let $\mathcal{F}\subseteq\LL(\mathcal{V})$ be a nonempty commuting family: $ST=TS$ for all $S,T\in\mathcal{F}$. Then there exists $\mathbf{v}\neq\mathbf{0}$ that is an eigenvector of every operator in $\mathcal{F}$.
\begin{proof} First suppose $\mathcal{F}=\{T_1,\ldots,T_m\}$ is finite, and induct on $m$. The case $m=1$ is Theorem \ref{thm:complex-operator-has-eigenvalue}.\\
For $m>1$, choose $\lambda\in\operatorname{spec}(T_1)$ and set $\mathcal{U}=E_{\lambda}(T_1)$. If $\mathbf{u}\in\mathcal{U}$, $j\geq2$, then $(T_1-\lambda I_{\mathcal{V}})T_j\mathbf{u}=T_j(T_1-\lambda I_{\mathcal{V}})\mathbf{u}=\mathbf{0}$, so $\mathcal{U}$ is invariant under $T_2,\ldots,T_m$. Their commuting restrictions to $\mathcal{U}$ have, by induction, a common eigenvector $\mathbf{v}\in\mathcal{U}$; then $\mathbf{v}$ is also an eigenvector of $T_1$.\\
For general $\mathcal{F}$, choose $T_1,\ldots,T_m\in\mathcal{F}$ spanning $\operatorname{span}(\mathcal{F})$. The finite case gives a common eigenvector $\mathbf{v}$ for them. If $S=\sum_j a_jT_j\in\mathcal{F}$ and $T_j\mathbf{v}=\mu_j\mathbf{v}$, then $S\mathbf{v}=(\sum_j a_j\mu_j)\mathbf{v}$, so $\mathbf{v}$ is common to all of $\mathcal{F}$.\\
\hlt{Structural Commentary}: Commutation makes one eigenspace invariant, reducing the family by one.\\
\hlt{Key Insight}: Find the common eigenvector inside an eigenspace.
\end{proof}
\end{theorem}

\begin{definition} \label{def:diagonal-and-upper-triangular-matrices} \hlt{Diagonal and Upper-Triangular Matrices}\\
The \hlt{diagonal} of a square matrix is its upper-left to lower-right entries. A square matrix is \hlt{diagonal} if all off-diagonal entries are zero, and \hlt{upper triangular} if all entries below the diagonal are zero.
\end{definition}

\begin{definition} \label{def:diagonalizable-operator} \hlt{Diagonalizable Operator}\\
An operator $T\in\LL(\mathcal{V})$ is \hlt{diagonalizable} if $T$ has a diagonal matrix with respect to some basis of $\mathcal{V}$.
\end{definition}

\begin{proposition} \label{prop:eigenspaces-for-distinct-eigenvalues-form-direct-sum} \hlt{Eigenspaces for Distinct Eigenvalues Form a Direct Sum}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and suppose $\lambda_1,\ldots,\lambda_m$ are distinct eigenvalues of $T$.\\
Then $E_{\lambda_1}(T)+\cdots+E_{\lambda_m}(T)$ is a direct sum. Consequently, $\dim E_{\lambda_1}(T)+\cdots+\dim E_{\lambda_m}(T)\leq\dim(\mathcal{V})$.
\begin{proof}
Suppose $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ with $\mathbf{u}_j\in E_{\lambda_j}(T)$. If some $\mathbf{u}_j$ were nonzero, then the nonzero vectors among $\mathbf{u}_1,\ldots,\mathbf{u}_m$ would be eigenvectors corresponding to distinct eigenvalues, and Theorem \ref{thm:distinct-eigenvalues-independent-eigenvectors} would make them linearly independent. This contradicts their sum being $\mathbf{0}$. Hence every $\mathbf{u}_j=\mathbf{0}$, so the sum is direct.\\
By Theorem \ref{thm:direct-sum-iff-dimensions-add}, $\dim(E_{\lambda_1}(T)+\cdots+E_{\lambda_m}(T))=\dim E_{\lambda_1}(T)+\cdots+\dim E_{\lambda_m}(T)$.\\
The left side is the dimension of a subspace of $\mathcal{V}$, so Proposition \ref{prop:dimension-bounds-and-basis-criteria} gives the stated inequality.\\
\hlt{Structural Commentary}: Distinct eigenspaces form independent pieces.\\
\hlt{Key Insight}: A zero sum across them would contradict eigenvector independence.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:conditions-equivalent-to-diagonalizability} \hlt{Conditions Equivalent to Diagonalizability}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and let $\lambda_1,\ldots,\lambda_m$ denote the distinct eigenvalues of $T$. Set $n=\dim(\mathcal{V})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is diagonalizable;
\item $\mathcal{V}$ has a basis consisting of eigenvectors of $T$;
\item $\exists$ one-dimensional subspaces $\mathcal{U}_1,\ldots,\mathcal{U}_n$ of $\mathcal{V}$, each invariant under $T$, such that $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_n$;
\item $\mathcal{V}=E_{\lambda_1}(T)\oplus\cdots\oplus E_{\lambda_m}(T)$;
\item $\dim(\mathcal{V})=\dim E_{\lambda_1}(T)+\cdots+\dim E_{\lambda_m}(T)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $T$ is diagonalizable, then $T$ has a diagonal matrix with respect to some basis $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$. Hence $T\mathbf{v}_j=a_j\mathbf{v}_j$ for each $j$, so this basis consists of eigenvectors of $T$.
\item[$(ii)$]$\Rightarrow (i)$ If $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ is a basis of eigenvectors of $T$, then $T\mathbf{v}_j=a_j\mathbf{v}_j$ for each $j$. Thus the matrix of $T$ with respect to this basis is diagonal.
\item[$(ii)$]$\Rightarrow (iii)$ Let $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be an eigenbasis, and put $\mathcal{U}_j=\operatorname{span}(\mathbf{v}_j)$. Each $\mathcal{U}_j$ is one-dimensional and invariant under $T$ by Proposition \ref{prop:one-dimensional-invariant-subspaces-and-eigenvectors}; the basis decomposition gives $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_n$.
\item[$(iii)$]$\Rightarrow (ii)$ Choose nonzero $\mathbf{v}_j\in\mathcal{U}_j$. Proposition \ref{prop:one-dimensional-invariant-subspaces-and-eigenvectors} makes each $\mathbf{v}_j$ an eigenvector of $T$, and the direct-sum decomposition makes $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ a basis of $\mathcal{V}$.
\item[$(ii)$]$\Rightarrow (iv)$ Every vector in an eigenbasis lies in one of $E_{\lambda_1}(T),\ldots,E_{\lambda_m}(T)$, so these eigenspaces span $\mathcal{V}$. Proposition \ref{prop:eigenspaces-for-distinct-eigenvalues-form-direct-sum} shows their sum is direct.
\item[$(iv)$]$\Rightarrow (v)$ If $\mathcal{V}=E_{\lambda_1}(T)\oplus\cdots\oplus E_{\lambda_m}(T)$, then dimensions add by Theorem \ref{thm:direct-sum-iff-dimensions-add}.
\item[$(v)$]$\Rightarrow (ii)$ For each $j$, choose a basis $\beta_j$ of $E_{\lambda_j}(T)$, and concatenate these bases into a list $\beta$. The length of $\beta$ is $\dim(\mathcal{V})$. If a linear combination of vectors in $\beta$ is zero, grouping terms by eigenspace gives $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ with $\mathbf{u}_j\in E_{\lambda_j}(T)$. Proposition \ref{prop:eigenspaces-for-distinct-eigenvalues-form-direct-sum} gives $\mathbf{u}_j=\mathbf{0}$ for every $j$; since each $\beta_j$ is independent, all coefficients are zero. Thus $\beta$ is independent, Proposition \ref{prop:dimension-bounds-and-basis-criteria} makes $\beta$ a basis of $\mathcal{V}$ consisting of eigenvectors of $T$.
\end{enumerate}
\hlt{Structural Commentary}: Diagonalizability means the eigenspaces exhaust $\mathcal{V}$. Directness is automatic for distinct eigenspaces, so the real test is spanning, or equivalently the dimension count.\\
\hlt{Key Insight}: A diagonal basis is exactly an eigenbasis.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:diagonalizability-viewpoint} \hlt{Diagonalization as Exhaustion by Eigenspaces}\\
Distinct eigenspaces are already separated by Proposition \ref{prop:eigenspaces-for-distinct-eigenvalues-form-direct-sum}. Thus diagonalizability is the stronger statement that these separated pieces account for every direction in $\mathcal{V}$.
\end{perspective}

\begin{corollary} \label{cor:enough-eigenvalues-implies-diagonalizable} \hlt{Enough Eigenvalues Imply Diagonalizability}\\
If $\mathcal{V}$ is finite-dimensional and $T\in\LL(\mathcal{V})$ has $\dim(\mathcal{V})$ distinct eigenvalues, then $T$ is diagonalizable.
\begin{proof}
Let $n=\dim(\mathcal{V})$, and let $\lambda_1,\ldots,\lambda_n$ be distinct eigenvalues of $T$. For each $j$, choose an eigenvector $\mathbf{v}_j$ corresponding to $\lambda_j$. By Theorem \ref{thm:distinct-eigenvalues-independent-eigenvectors}, $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ is linearly independent. Since this list has length $n$, Proposition \ref{prop:dimension-bounds-and-basis-criteria} makes it a basis of $\mathcal{V}$. Thus $\mathcal{V}$ has an eigenbasis, so $T$ is diagonalizable by Theorem \ref{thm:conditions-equivalent-to-diagonalizability}.\\
\hlt{Structural Commentary}: One eigenvector from each distinct eigenvalue already gives a full-size independent list.\\
\hlt{Key Insight}: Once the independent eigenvectors reach $\dim(\mathcal{V})$, they must form a basis.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:upper-triangular-criteria} \hlt{Criteria for an Upper-Triangular Operator Matrix}\\
Let $T\in\LL(\mathcal{V})$ and let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis of $\mathcal{V}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $[T]_{\beta\leftarrow\beta}$ is upper triangular;
\item $T\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$ for each $j=1,\ldots,n$;
\item $\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$ is invariant under $T$ for each $j=1,\ldots,n$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$
The $j$th column of $[T]_{\beta\leftarrow\beta}$ is the coordinate column of $T\mathbf{v}_j$ in the basis $\beta$. Since the matrix is upper triangular, this column has zero entries below position $j$, so $T\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$.
\item[$(ii)$]$\Rightarrow (i)$
If $T\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$ for every $j$, then the $j$th coordinate column of $T\mathbf{v}_j$ has zero entries below position $j$. Hence $[T]_{\beta\leftarrow\beta}$ is upper triangular.
\item[$(ii)$]$\Rightarrow (iii)$
If $\mathbf{u}\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$, then linearity gives $T\mathbf{u}$ as a linear combination of $T\mathbf{v}_1,\ldots,T\mathbf{v}_j$, all of which lie in $\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$. Hence this span is invariant.
\item[$(iii)$]$\Rightarrow (ii)$
Apply invariance of $\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$ to $\mathbf{v}_j$.
\end{enumerate}
\hlt{Structural Commentary}: Upper triangularity records an invariant flag.\\
\hlt{Key Insight}: Columns below the diagonal vanish exactly when $T\mathbf{v}_j$ uses only earlier basis vectors and itself.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:complex-upper-triangularization} \hlt{Complex Upper Triangularisation}\\
Let $\mathcal{V}$ be a finite-dimensional complex vector space and let $T\in\LL(\mathcal{V})$. Then $T$ has an upper-triangular matrix with respect to some basis of $\mathcal{V}$.
\begin{proof} Induct on $n=\dim(\mathcal{V})$. If $n\leq1$, the result is immediate.\\
Suppose $n>1$ and the result holds in smaller dimensions. Choose an eigenvector $\mathbf{v}_1$ of $T$ by Theorem \ref{thm:complex-operator-has-eigenvalue}, and set $\mathcal{U}=\operatorname{span}(\mathbf{v}_1)$. Then $\mathcal{U}$ is $T$-invariant, so the quotient operator $T/\mathcal{U}$ is defined on $\mathcal{V}/\mathcal{U}$, which has dimension $n-1$. By induction, choose a basis $(\mathbf{v}_2+\mathcal{U},\ldots,\mathbf{v}_n+\mathcal{U})$ of $\mathcal{V}/\mathcal{U}$ with respect to which $T/\mathcal{U}$ is upper triangular, and choose representatives $\mathbf{v}_2,\ldots,\mathbf{v}_n\in\mathcal{V}$. Then $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ is a basis of $\mathcal{V}$: any relation descends to the quotient basis, forcing all coefficients of $\mathbf{v}_2,\ldots,\mathbf{v}_n$ to be zero, and then the coefficient of $\mathbf{v}_1$ is zero.\\
Now $T\mathbf{v}_1\in\operatorname{span}(\mathbf{v}_1)$. For $j\geq2$, upper triangularity of $T/\mathcal{U}$ gives $(T/\mathcal{U})(\mathbf{v}_j+\mathcal{U}) \in\operatorname{span}(\mathbf{v}_2+\mathcal{U},\ldots,\mathbf{v}_j+\mathcal{U})$, so $T\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$. Hence Theorem \ref{thm:upper-triangular-criteria} gives an upper-triangular matrix for $T$ in the basis $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$.\\
\hlt{Structural Commentary}: Lift an upper-triangular quotient basis above an eigenline.\\
\hlt{Key Insight}: The quotient builds the remaining invariant flag.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:upper-triangular-invertibility-criterion} \hlt{Invertibility from an Upper-Triangular Matrix}\\
Suppose $T\in\LL(\mathcal{V})$ has an upper-triangular matrix with diagonal entries $\lambda_1,\ldots,\lambda_n$ with respect to some basis of $\mathcal{V}$. Then $T$ is invertible if and only if $\lambda_j\neq0$ for every $j$.
\begin{proof}
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be such a basis.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Leftarrow)$]
If every $\lambda_j\neq0$, then upper-triangular matrix $[T]_{\beta\leftarrow\beta}$ is invertible by Theorem \ref{thm:inverse-upper-triangular}, so $T$ is invertible.
\item[$(\Rightarrow)$]
Suppose $T$ is invertible. If $\lambda_j=0$ for some $j$, set $\mathcal{U}_j=\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$. Upper triangularity gives $T(\mathcal{U}_j)\subseteq\mathcal{U}_{j-1}$ because the coefficient of $\mathbf{v}_j$ in $T\mathbf{v}_j$ is $\lambda_j=0$ and $T\mathbf{v}_i\in\mathcal{U}_{j-1}$ for $i<j$. Since $\dim(\mathcal{U}_j)>\dim(\mathcal{U}_{j-1})$, the restriction $T|_{\mathcal{U}_j}:\mathcal{U}_j\to\mathcal{U}_{j-1}$ is not injective. Hence $T$ has a nonzero kernel vector, contradicting invertibility. Thus every $\lambda_j\neq0$.
\end{enumerate}
\hlt{Structural Commentary}: Diagonal entries control each flag layer.\\
\hlt{Key Insight}: A zero diagonal entry forces a kernel vector.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:eigenvalues-of-upper-triangular-operator} \hlt{Eigenvalues from an Upper-Triangular Matrix}\\
Suppose $T\in\LL(\mathcal{V})$ has an upper-triangular matrix with diagonal entries $\lambda_1,\ldots,\lambda_n$ with respect to some basis of $\mathcal{V}$. Then the eigenvalues of $T$ are precisely $\lambda_1,\ldots,\lambda_n$.
\begin{proof}
For $\lambda\in\mathbb{F}$, the matrix of $T-\lambda I_{\mathcal{V}}$ in the same basis is upper triangular with diagonal entries $\lambda_1-\lambda,\ldots,\lambda_n-\lambda$. By Proposition \ref{prop:upper-triangular-invertibility-criterion}, $T-\lambda I_{\mathcal{V}}$ is not invertible if and only if $\lambda=\lambda_j$ for some $j$. By Theorem \ref{thm:equivalent-eigenvalue-conditions}, this is equivalent to $\lambda$ being an eigenvalue of $T$.\\
\hlt{Structural Commentary}: Scalar shifts preserve triangular form.\\
\hlt{Key Insight}: Eigenvalues occur exactly when one shifted diagonal entry becomes zero.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:upper-triangular-viewpoint} \hlt{Why Triangular Form Matters}\\
Upper-triangular form is the first useful simplification beyond finding one eigenvector. It preserves enough of the operator to read eigenvalues from the diagonal while still allowing nontrivial coupling above the diagonal.
\end{perspective}

\begin{theorem} \label{thm:complex-spectral-theorem} \hlt{Complex Spectral Theorem}\\
Let $\mathcal{V}$ be a finite-dimensional complex inner-product space and let $T\in\LL(\mathcal{V})$. The following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $T$ is normal;
\item $\mathcal{V}$ has an orthonormal basis consisting of eigenvectors of $T$;
\item $T$ has a diagonal matrix with respect to some orthonormal basis of $\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow(ii)$
By Schur triangularisation (Theorem \ref{thm:schur-triangularization}), choose an orthonormal basis $\gamma=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ in which $A=[T]_{\gamma\leftarrow\gamma}=[a_{ij}]$ is upper triangular. Clear rows inductively: once rows $<k$ are cleared, Theorem \ref{thm:normal-iff-equal-adjoint-norms} gives $\abs{a_{kk}}^2=\norm{T\mathbf{e}_k}^2=\norm{T^*\mathbf{e}_k}^2 =\abs{a_{kk}}^2+\sum_{j>k}\abs{a_{kj}}^2$.\\
Hence $a_{kj}=0$ for $j>k$. Thus $A$ is diagonal, so $\gamma$ is an orthonormal basis of eigenvectors.
\item[$(ii)$]$\Rightarrow(iii)$
In an orthonormal eigenbasis, the matrix of $T$ is diagonal.
\item[$(iii)$]$\Rightarrow(i)$
If $[T]$ is diagonal in an orthonormal basis, then $[T^*]=[T]^*$ is diagonal as well. Diagonal matrices commute, so $TT^*=T^*T$; hence $T$ is normal.
\end{enumerate}
\hlt{Structural Commentary}: Schur gives triangular form; normality kills the upper entries.\\
\hlt{Key Insight}: Equal norms of $T\mathbf{e}_j$ and $T^*\mathbf{e}_j$ force each row tail to vanish.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:invertible-quadratic-self-adjoint} \hlt{Invertible Quadratic Expressions}\\
Let $T\in\LL(\mathcal{V})$ be self-adjoint on a finite-dimensional inner-product space.\\
If $b,c\in\mathbb{R}$ and $b^2<4c$, then $T^2+bT+cI_{\mathcal{V}}$ is invertible.
\begin{proof}
Let $S=T^2+bT+cI_{\mathcal{V}}$. For $\mathbf{v}\neq\mathbf{0}$,
\begin{align}
\innerprod{S\mathbf{v},\mathbf{v}} &=\norm{T\mathbf{v}}^2+b\innerprod{T\mathbf{v},\mathbf{v}}+c\norm{\mathbf{v}}^2 \geq \norm{T\mathbf{v}}^2-\abs{b}\norm{T\mathbf{v}}\norm{\mathbf{v}}+c\norm{\mathbf{v}}^2 \nonumber \\
&=\left(\norm{T\mathbf{v}}-\frac{\abs{b}}{2}\norm{\mathbf{v}}\right)^2+\left(c-\frac{b^2}{4}\right)\norm{\mathbf{v}}^2>0 \nonumber
\end{align}
Thus $S\mathbf{v}\neq\mathbf{0}$ for every nonzero $\mathbf{v}$, so $\ker(S)=\{\mathbf{0}\}$. In finite dimensions, injectivity implies invertibility.\\
\hlt{Structural Commentary}: Negative discriminant keeps this self-adjoint quadratic positive.\\
\hlt{Key Insight}: Self-adjointness converts $\innerprod{T^2\mathbf{v},\mathbf{v}}$ into $\norm{T\mathbf{v}}^2$, so completing the square controls the kernel.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:self-adjoint-operators-have-eigenvalues} \hlt{Self-Adjoint Operators Have Eigenvalues}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be a finite-dimensional real inner-product space. Every self-adjoint $T\in\LL(\mathcal{V})$ has an eigenvalue.
\begin{proof}
Choose $\mathbf{v}\neq\mathbf{0}$ and set $n=\dim(\mathcal{V})$. The list $\mathbf{v},T\mathbf{v},\ldots,T^n\mathbf{v}$ is linearly dependent, so some nonzero real polynomial $p$ satisfies $p(T)\mathbf{v}=\mathbf{0}$. Factor $p$ over $\mathbb{R}$ as a product of real linear factors and irreducible real quadratic factors: $p(x)=a\prod_{j=1}^{M}(x^2+b_jx+c_j)\prod_{k=1}^{m}(x-\lambda_k)$, where $a\neq0$, $b_j^2<4c_j$, and $\lambda_k\in\mathbb{R}$.\\
By Proposition \ref{prop:invertible-quadratic-self-adjoint}, every quadratic factor $T^2+b_jT+c_jI_{\mathcal{V}}$ is invertible. If $m=0$, then $p(T)$ would be invertible, contradicting $p(T)\mathbf{v}=0$ with $\mathbf{v}\neq0$. Hence $m\geq1$.\\
After removing the invertible quadratic factors, the product $(T-\lambda_1I_{\mathcal{V}})\cdots(T-\lambda_mI_{\mathcal{V}})$ has nontrivial kernel. Therefore at least one factor $T-\lambda_kI_{\mathcal{V}}$ is not injective, and $\lambda_k$ is an eigenvalue of $T$.\\
\hlt{Structural Commentary}: An annihilating polynomial remains; quadratic factors are invertible.\\
\hlt{Key Insight}: The nontrivial kernel must come from a real linear factor, hence from an eigenvalue.
\end{proof}
\end{lemma}

\begin{proposition} \label{prop:self-adjoint-invariant-subspaces} \hlt{Self-Adjoint Operators and Invariant Subspaces}\\
Let $T\in\LL(\mathcal{V})$ be self-adjoint, and let $\mathcal{U}$ be a subspace of $\mathcal{V}$ invariant under $T$. Then:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{U}^{\perp}$ is invariant under $T$;
\item $T|_{\mathcal{U}}\in\LL(\mathcal{U})$ is self-adjoint;
\item $T|_{\mathcal{U}^{\perp}}\in\LL(\mathcal{U}^{\perp})$ is self-adjoint.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
If $\mathbf{v}\in\mathcal{U}^{\perp}$ and $\mathbf{u}\in\mathcal{U}$, then $\innerprod{T\mathbf{v},\mathbf{u}}=\innerprod{\mathbf{v},T\mathbf{u}}=0$ because $T\mathbf{u}\in\mathcal{U}$. Hence $T\mathbf{v}\in\mathcal{U}^{\perp}$.
\item For $\mathbf{u},\mathbf{v}\in\mathcal{U}$, $\innerprod{(T|_{\mathcal{U}})\mathbf{u},\mathbf{v}} =\innerprod{T\mathbf{u},\mathbf{v}} =\innerprod{\mathbf{u},T\mathbf{v}} =\innerprod{\mathbf{u},(T|_{\mathcal{U}})\mathbf{v}}$. Thus $T|_{\mathcal{U}}$ is self-adjoint.
\item By (i), $\mathcal{U}^{\perp}$ is invariant, so (ii)'s argument applies to $T|_{\mathcal{U}^{\perp}}$.
\end{enumerate}
\hlt{Structural Commentary}: Self-adjointness preserves orthogonal complements of invariant subspaces.\\
\hlt{Key Insight}: Move $T$ across the inner product; invariance of $\mathcal{U}$ then forces orthogonality to persist.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:real-spectral-theorem} \hlt{Real Spectral Theorem}\\
Let $\mathcal{V}$ be a finite-dimensional real inner-product space and let $T\in\LL(\mathcal{V})$. The following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $T$ is self-adjoint;
\item $\mathcal{V}$ has an orthonormal basis consisting of eigenvectors of $T$;
\item $T$ has a diagonal matrix with respect to some orthonormal basis of $\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$
Assume $T=T^*$ and induct on $n=\dim(\mathcal{V})$; $n\leq1$ is clear. For $n\geq2$, Lemma \ref{lem:self-adjoint-operators-have-eigenvalues} gives a unit eigenvector $\mathbf{u}$. Set $\mathcal{U}=\operatorname{span}(\mathbf{u})$. Since $\mathcal{U}$ is invariant, Proposition \ref{prop:self-adjoint-invariant-subspaces} makes $\mathcal{U}^{\perp}$ invariant and $T|_{\mathcal{U}^{\perp}}$ self-adjoint. Induct on $\mathcal{U}^{\perp}$ and adjoin $\mathbf{u}$.
\item[$(ii)$]$\Rightarrow (iii)$
In an orthonormal eigenbasis, the matrix of $T$ is diagonal.
\item[$(iii)$]$\Rightarrow (i)$
In an orthonormal basis where $[T]$ is real diagonal, $[T^*]=[T]^t=[T]$. Hence $T=T^*$.
\end{enumerate}
\hlt{Structural Commentary}: Split off one eigendirection, then restrict to its orthogonal complement.\\
\hlt{Key Insight}: One eigenvector reduces the dimension.
\end{proof}
\end{theorem}

\begin{definition} \label{def:positive-operator} \hlt{Positive Operator}\\
$T\in\LL(\mathcal{V})$ is \hlt{positive} if $T=T^*$ and $\innerprod{T\mathbf{v},\mathbf{v}}\geq0$ for all $\mathbf{v}\in\mathcal{V}$.
\end{definition}

\begin{definition} \label{def:square-root-operator} \hlt{Square Root of an Operator}\\
An operator $R\in\LL(\mathcal{V})$ is a \hlt{square root} of $T\in\LL(\mathcal{V})$ if $R^2=T$.
\end{definition}

\begin{theorem} \label{thm:positive-operator-characterization} \hlt{Characterization of Positive Operators}\\
For $T\in\LL(\mathcal{V})$, the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $T$ is positive;
\item $T$ is self-adjoint and every eigenvalue of $T$ is nonnegative;
\item $T$ has a positive square root;
\item $T$ has a self-adjoint square root;
\item there exists $R\in\LL(\mathcal{V})$ such that $T=R^*R$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $T\mathbf{v}=\lambda\mathbf{v}$ and $\mathbf{v}\neq0$, then $\lambda\in\mathbb{R}$ and $0\leq\innerprod{T\mathbf{v},\mathbf{v}}=\lambda\norm{\mathbf{v}}^2$ so $\lambda\geq0$.
\item[$(ii)$]$\Rightarrow (iii)$ By the spectral theorem, choose an orthonormal eigenbasis $(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ with $T\mathbf{e}_j=\lambda_j\mathbf{e}_j$ and $\lambda_j\geq0$. Define $S\mathbf{e}_j=\sqrt{\lambda_j}\mathbf{e}_j$. Then $S$ is positive and $S^2=T$.
\item[$(iii)$]$\Rightarrow (iv)$ A positive square root is self-adjoint.
\item[$(iv)$]$\Rightarrow (v)$ If $T=R^2$ with $R=R^*$, then $T=R^*R$.
\item[$(v)$]$\Rightarrow (i)$ If $T=R^*R$, then $T^*=T$ and $\innerprod{T\mathbf{v},\mathbf{v}}=\innerprod{R\mathbf{v},R\mathbf{v}}=\norm{R\mathbf{v}}^2\geq0$ for every $\mathbf{v}$, so $T$ is positive.
\end{enumerate}
\hlt{Structural Commentary}: Positivity equals nonnegative spectrum and square-root factorization.\\
\hlt{Key Insight}: In an eigenbasis, the proof is scalar: $\lambda\geq0$ iff $\sqrt{\lambda}$ exists.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:positive-square-root-unique} \hlt{Unique Positive Square Root}\\
Every positive operator on $\mathcal{V}$ has a unique positive square root.
\begin{proof}
Existence follows from Theorem \ref{thm:positive-operator-characterization}. For uniqueness, decompose $\mathcal{V}=E_{\lambda_1}(T)\oplus\cdots\oplus E_{\lambda_m}(T)$.\\
Any positive square root must equal $\sqrt{\lambda_j}I$ on $E_{\lambda_j}(T)$: indeed, if $R^2=T$, then $R$ commutes with $T$, so $E_{\lambda_j}(T)$ is $R$-invariant, and positivity plus the spectral theorem force $R|_{E_{\lambda_j}(T)}=\sqrt{\lambda_j}I$.\\
Thus two positive square roots agree on every eigenspace.\\
\hlt{Structural Commentary}: Each eigenspace determines its own scalar root.\\
\hlt{Key Insight}: Positivity selects the nonnegative root.
\end{proof}
\end{proposition}

\begin{definition} \label{def:positive-square-root-notation} \hlt{Positive Square Root Notation}\\
If $T$ is positive, then $\sqrt{T}$ denotes the unique positive square root of $T$.
\end{definition}

\begin{theorem} \label{thm:complex-isometry-description} \hlt{Description of Complex Isometries}\\
Let $\mathcal{V}$ be a finite-dimensional complex inner-product space and let $S\in\LL(\mathcal{V})$. Then $S$ is an isometry if and only if $\mathcal{V}$ has an orthonormal basis consisting of eigenvectors of $S$ whose eigenvalues all have absolute value $1$.
\begin{proof}
If $S\mathbf{e}_j=\lambda_j\mathbf{e}_j$ in an orthonormal basis with $\abs{\lambda_j}=1$, then for $\mathbf{v}=\sum_j c_j\mathbf{e}_j$,\\
$\norm{S\mathbf{v}}^2=\sum\limits_j\abs{\lambda_jc_j}^2=\sum\limits_j\abs{c_j}^2=\norm{\mathbf{v}}^2$, so $S$ is an isometry. Conversely, if $S$ is an isometry, then $S^*S=I_{\mathcal{V}}$ by Theorem \ref{thm:finite-dimensional-isometry-iff-adjoint-product-identity}. Finite-dimensional invertibility gives $SS^*=I_{\mathcal{V}}$, so $S$ is normal. By Complex Spectral Theorem, choose orthonormal eigenbasis $S\mathbf{e}_j=\lambda_j\mathbf{e}_j$. Since $\mathbf{e}_j$ is unit and $S$ preserves norms, $\abs{\lambda_j}=\norm{S\mathbf{e}_j}=1$.\\
\hlt{Structural Commentary}: Complex isometries are normal with unit-circle spectrum.\\
\hlt{Key Insight}: In an eigenbasis, isometry means $\abs{\lambda_j}=1$.
\end{proof}
\end{theorem}
