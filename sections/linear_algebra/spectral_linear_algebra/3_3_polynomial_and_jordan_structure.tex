\subsubsection{Polynomial and Jordan Structure}

Generalized eigenvectors repair the main defect of ordinary eigenvectors: an operator may not have enough eigendirections to form a basis. The replacement is to study vectors killed by powers of shifted operators.

\begin{definition} \label{def:generalized-eigenvector-and-eigenspace} \hlt{Generalized Eigenvectors and Generalized Eigenspaces}\\
Let $T\in\LL(\mathcal{V})$, $\lambda\in\mathbb{F}$. Nonzero vector $\mathbf{v}\in\mathcal{V}$ is a \hlt{generalized eigenvector} of $T$ to $\lambda$ if $(T-\lambda I_{\mathcal{V}})^j\mathbf{v}=\mathbf{0}$ for some positive integer $j$. The \hlt{generalized eigenspace} corresponding to $\lambda$ is
\begin{align}
G(\lambda,T)\defeq \{\mathbf{0}\}\cup\{\mathbf{v}\neq\mathbf{0}:\mathbf{v}\text{ is a generalized eigenvector of }T\text{ corresponding to }\lambda\}.
\nonumber
\end{align}
\end{definition}

Every eigenvector is a generalized eigenvector by taking $j=1$. The point is that higher powers of $T-\lambda I_{\mathcal{V}}$ may reveal directions that are not eigenvectors but are eventually pushed into the eigenspace.

\begin{proposition} \label{prop:kernels-of-powers-increase} \hlt{Kernels of Powers Increase}\\
Let $T\in\LL(\mathcal{V})$. Then $\{\mathbf{0}\}=\ker(T^0)\subseteq\ker(T)\subseteq\ker(T^2)\subseteq\cdots\subseteq\ker(T^k)\subseteq\ker(T^{k+1})\subseteq\cdots$.
\begin{proof}
If $\mathbf{v}\in\ker(T^k)$, then $T^k\mathbf{v}=\mathbf{0}$, hence $T^{k+1}\mathbf{v}=T(T^k\mathbf{v})=\mathbf{0}$. Thus $\ker(T^k)\subseteq\ker(T^{k+1})$ for every $k\geq0$.\\
\hlt{Structural Commentary}: Powers of $T$ form an increasing chain of kernels.\\
\hlt{Key Insight}: Once a vector is killed by $T^k$, applying one more $T$ cannot revive it.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:kernel-power-stabilization-after-equality} \hlt{Equality Forces Stabilization}\\
Let $T\in\LL(\mathcal{V})$ and $m\geq0$. If $\ker(T^m)=\ker(T^{m+1})$, then $\ker(T^m)=\ker(T^{m+1})=\ker(T^{m+2})=\cdots$.
\begin{proof}
By Proposition \ref{prop:kernels-of-powers-increase}, it suffices to prove $\ker(T^{m+k+1})\subseteq\ker(T^{m+k})$ for every $k\geq0$.\\
Take $\mathbf{v}\in\ker(T^{m+k+1})$. Then $T^{m+1}(T^k\mathbf{v})=T^{m+k+1}\mathbf{v}=\mathbf{0}$, so $T^k\mathbf{v}\in\ker(T^{m+1})=\ker(T^m)$.\\
Hence $T^{m+k}\mathbf{v}=T^m(T^k\mathbf{v})=\mathbf{0}$, proving the reverse inclusion.\\
\hlt{Structural Commentary}: Stabilization at one step propagates through the whole tail.\\
\hlt{Key Insight}: A later kernel vector can be shifted down into the already-stable kernel.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:kernel-powers-stabilize-by-dimension} \hlt{Kernels Stop Growing by $\dim(\mathcal{V})$}\\
If $\mathcal{V}$ is finite-dimensional, $T\in\LL(\mathcal{V})$, and $n=\dim(\mathcal{V})$, then $\ker(T^n)=\ker(T^{n+1})=\ker(T^{n+2})=\cdots$.
\begin{proof}
If $\ker(T^n)\neq\ker(T^{n+1})$, Prop \ref{prop:kernel-power-stabilization-after-equality} implies all inclusions $\{\mathbf{0}\}=\ker(T^0)\subsetneq\ker(T)\subsetneq\cdots\subsetneq\ker(T^{n+1})$
are strict. Each strict inclusion raises dimension by at least $1$, so $\dim\ker(T^{n+1})\geq n+1$, impossible because $\ker(T^{n+1})\subseteq\mathcal{V}$. Thus $\ker(T^n)=\ker(T^{n+1})$, and Proposition \ref{prop:kernel-power-stabilization-after-equality} gives the tail equality.\\
\hlt{Structural Commentary}: Strict kernel growth is dimension-limited.\\
\hlt{Key Insight}: Dimension bounds force the kernel chain to stabilize no later than step $\dim(\mathcal{V})$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:kernel-range-stable-direct-sum} \hlt{Stable Kernel-Range Decomposition}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and set $n=\dim(\mathcal{V})$. Then $\mathcal{V}=\ker(T^n)\oplus\operatorname{range}(T^n)$.
\begin{proof}
First show the intersection is trivial. If $\mathbf{v}\in\ker(T^n)\cap\operatorname{range}(T^n)$, then $\mathbf{v}=T^n\mathbf{u}$ and $T^n\mathbf{v}=\mathbf{0}$. Hence $T^{2n}\mathbf{u}=\mathbf{0}$, so $\mathbf{u}\in\ker(T^{2n})=\ker(T^n)$ by Theorem \ref{thm:kernel-powers-stabilize-by-dimension}. Thus $\mathbf{v}=T^n\mathbf{u}=\mathbf{0}$.\\
By Rank-Nullity Theorem \ref{thm:rank-nullity}, $\dim\ker(T^n)+\dim\operatorname{range}(T^n)=\dim(\mathcal{V})$.\\
The zero intersection and the dimension count imply the sum is direct and equals $\mathcal{V}$.\\
\hlt{Structural Commentary}: At the stable power, killed directions and surviving image directions separate cleanly.\\
\hlt{Key Insight}: Stabilization turns the usual rank-nullity count into an actual direct-sum decomposition.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:generalized-eigenspace-stabilized-kernel} \hlt{Description of Generalized Eigenspaces}\\
Let $\mathcal{V}$ be finite-dimensional, $T\in\LL(\mathcal{V})$, $\lambda\in\mathbb{F}$, and $n=\dim(\mathcal{V})$. Then $G(\lambda,T)=\ker\bigl((T-\lambda I_{\mathcal{V}})^n\bigr)$.\\
In particular, $G(\lambda,T)$ is a subspace of $\mathcal{V}$.
\begin{proof}
Set $S=T-\lambda I_{\mathcal{V}}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathbf{v}\in G(\lambda,T)$, then $S^j\mathbf{v}=\mathbf{0}$ for some $j\geq1$. If $j\leq n$, Proposition \ref{prop:kernels-of-powers-increase} gives $\mathbf{v}\in\ker(S^n)$; if $j>n$, Theorem \ref{thm:kernel-powers-stabilize-by-dimension} gives $\ker(S^j)=\ker(S^n)$.
\item[$(\Leftarrow)$] If $\mathbf{v}\in\ker(S^n)$, then either $\mathbf{v}=\mathbf{0}$ or $\mathbf{v}$ is a generalized eigenvector of $T$ corresponding to $\lambda$.
\end{enumerate}
Thus $G(\lambda,T)=\ker(S^n)$, which is a subspace.\\
\hlt{Structural Commentary}: Generalized eigenspaces are stable shifted kernels.\\
\hlt{Key Insight}: Finite dimension makes exponent $\dim(\mathcal{V})$ enough.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:generalized-eigenspaces-invariant} \hlt{Generalized Eigenspaces Are Invariant}\\
Let $\mathcal{V}$ be finite-dimensional and $T\in\LL(\mathcal{V})$. Then $G(\lambda,T)$ is invariant under $T$ for every $\lambda\in\mathbb{F}$.
\begin{proof}
By Theorem \ref{thm:generalized-eigenspace-stabilized-kernel}, $G(\lambda,T)=\ker((T-\lambda I_{\mathcal{V}})^n)$, where $n=\dim(\mathcal{V})$.\\
This is the kernel of a polynomial in $T$, so it is invariant under $T$ by Proposition \ref{prop:kernel-of-polynomial-in-operator-is-invariant}.\\
\hlt{Structural Commentary}: A generalized eigenspace is cut out by a polynomial equation in $T$.\\
\hlt{Key Insight}: Polynomial kernels commute with the operator that defines them.
\end{proof}
\end{proposition}

\begin{definition} \label{def:nilpotent-operator} \hlt{Nilpotent Operator}\\
An operator $N\in\LL(\mathcal{V})$ is \hlt{nilpotent} if $N^m=0$ for some positive integer $m$.
\end{definition}

\begin{corollary} \label{cor:nilpotent-power-dimension-zero} \hlt{Nilpotent Power Bound}\\
If $\mathcal{V}$ is finite-dimensional, $N\in\LL(\mathcal{V})$ is nilpotent, and $n=\dim(\mathcal{V})$, then $N^n=0$.
\begin{proof}
Since $N$ is nilpotent, $\ker(N^m)=\mathcal{V}$ for some $m$.\\
The kernel chain has stabilized by the $n$th step by Theorem \ref{thm:kernel-powers-stabilize-by-dimension}, so $\ker(N^n)=\mathcal{V}$. Hence $N^n=0$.\\
\hlt{Structural Commentary}: Nilpotence says some kernel in the chain is already the whole space.\\
\hlt{Key Insight}: The stabilization bound pulls that full-kernel conclusion back to exponent $\dim(\mathcal{V})$.
\end{proof}
\end{corollary}

\begin{lemma} \label{lem:scalar-plus-nilpotent-invertible} \hlt{Nonzero Scalar Plus Nilpotent Is Invertible}\\
Let $N\in\LL(\mathcal{V})$ be nilpotent and let $a\in\mathbb{F}$ with $a\neq0$. Then $aI_{\mathcal{V}}+N$ is invertible.
\begin{proof}
Choose $m$ such that $N^m=0$. Then $(aI_{\mathcal{V}}+N)\left(a^{-1}\sum\limits_{r=0}^{m-1}(-a^{-1}N)^r\right) =I_{\mathcal{V}}-(-a^{-1}N)^m=I_{\mathcal{V}}$,
and the same computation in the reverse order gives a two-sided inverse.\\
\hlt{Structural Commentary}: A nilpotent perturbation has a finite geometric-series inverse.\\
\hlt{Key Insight}: Nilpotence truncates the Neumann series exactly.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:distinct-generalized-eigenvectors-independent} \hlt{Generalized Eigenvectors for Distinct Eigenvalues Are Independent}\\
Let $\mathcal{V}$ be finite-dimensional and $T\in\LL(\mathcal{V})$. Suppose $\lambda_1,\ldots,\lambda_m$ are distinct eigenvalues of $T$, and let $\mathbf{v}_1,\ldots,\mathbf{v}_m$ be corresponding generalized eigenvectors. Then $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is linearly independent.
\begin{proof}
Let $n=\dim(\mathcal{V})$ and suppose $a_1\mathbf{v}_1+\cdots+a_m\mathbf{v}_m=\mathbf{0}$.\\
Fix $r\in\{1,\ldots,m\}$ and apply $Q_r(T)\defeq\prod\limits_{i\neq r}(T-\lambda_iI_{\mathcal{V}})^n$ to the relation.\\
For $i\neq r$, Theorem \ref{thm:generalized-eigenspace-stabilized-kernel} gives $Q_r(T)\mathbf{v}_i=\mathbf{0}$, so $a_rQ_r(T)\mathbf{v}_r=\mathbf{0}$.\\
On $G(\lambda_r,T)$, the operator $T-\lambda_iI_{\mathcal{V}}$ equals $(T-\lambda_rI_{\mathcal{V}})+(\lambda_r-\lambda_i)I_{\mathcal{V}}$. The first term is nilpotent on $G(\lambda_r,T)$ and $\lambda_r-\lambda_i\neq0$, so Lemma \ref{lem:scalar-plus-nilpotent-invertible} makes each factor invertible on $G(\lambda_r,T)$. Thus $Q_r(T)\mathbf{v}_r\neq\mathbf{0}$, forcing $a_r=0$. Since $r$ was arbitrary, all coefficients vanish.\\
\hlt{Structural Commentary}: Polynomial filters isolate one generalized eigenspace at a time.\\
\hlt{Key Insight}: Distinct eigenvalues let polynomial filters kill the other components and preserve the target one.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:nilpotent-strictly-upper-triangular-matrix} \hlt{Nilpotent Operators and Strictly Upper-Triangular Matrices}\\
Let $\mathcal{V}$ be finite-dimensional and $N\in\LL(\mathcal{V})$. Then $N$ is nilpotent if and only if $N$ has a strictly upper-triangular matrix in some basis.
\begin{proof}
Let $n=\dim(\mathcal{V})$. By Corollary \ref{cor:nilpotent-power-dimension-zero}, $\{\mathbf{0}\}\subseteq\ker(N)\subseteq\ker(N^2)\subseteq\cdots\subseteq\ker(N^n)=\mathcal{V}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Choose a basis of $\ker(N)$, extend it to a basis of $\ker(N^2)$, and continue until obtaining a basis of $\mathcal{V}$.\\
If $\mathbf{v}$ first appears in $\ker(N^k)$, then $N\mathbf{v}\in\ker(N^{k-1})$, which is spanned by earlier basis vectors. Hence the matrix of $N$ is strictly upper triangular.
\item[$(\Leftarrow)$] If the matrix of $N$ is $n\times n$ strictly upper triangular, then $N$ sends each basis vector into the span of earlier basis vectors. After $n$ applications every basis vector is sent to $\mathbf{0}$, so $N^n=0$.
\end{enumerate}
\hlt{Structural Commentary}: Kernel filtration gives adapted basis; triangular form records motion to earlier layers.\\
\hlt{Key Insight}: Nilpotence is strict upper-triangularity in a suitable basis.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:generalized-eigenvectors-as-nilpotent-data} \hlt{Eigenvalue Plus Nilpotent Part}\\
On $G(\lambda,T)$, the operator $T$ behaves as $\lambda I_{\mathcal{V}}$ plus the nilpotent operator $T-\lambda I_{\mathcal{V}}$. Generalized eigenvectors are therefore ordinary eigenvectors enriched by nilpotent chain data.
\end{perspective}

\begin{theorem} \label{thm:complex-primary-decomposition-generalized-eigenspaces} \hlt{Primary Decomposition over $\mathbb{C}$}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dim over $\mathbb{C}$, let $T\in\LL(\mathcal{V})$, and let $\lambda_1,\ldots,\lambda_m$ be the distinct eigenvalues of $T$. Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{V}=G(\lambda_1,T)\oplus\cdots\oplus G(\lambda_m,T)$;
\item each $G(\lambda_j,T)$ is $T$-invariant;
\item each $(T-\lambda_jI_{\mathcal{V}})|_{G(\lambda_j,T)}$ is nilpotent.
\end{enumerate}
\begin{proof}
Let $n=\dim(\mathcal{V})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Prove decomposition by induction on $n$. The case $n=1$ is immediate.\\
\item Put $S=T-\lambda_1I_{\mathcal{V}}$. By Theorem \ref{thm:kernel-range-stable-direct-sum}, $\mathcal{V}=\ker(S^n)\oplus\operatorname{range}(S^n) =G(\lambda_1,T)\oplus\mathcal{U}$, $\mathcal{U}\defeq\operatorname{range}(S^n)$.\\
Here $\mathcal{U}$ is $T$-invariant by Proposition \ref{prop:kernel-range-polynomial-in-operator-invariant}, and $\dim(\mathcal{U})<n$. The eigenvalues of $T|_{\mathcal{U}}$ are $\lambda_2,\ldots,\lambda_m$: $\lambda_1$ is excluded by $G(\lambda_1,T)\cap\mathcal{U}=\{\mathbf{0}\}$, and if $T\mathbf{v}=\lambda_j\mathbf{v}$ with $j\geq2$, then $S^n\mathbf{v}=(\lambda_j-\lambda_1)^n\mathbf{v}$, so $\mathbf{v}\in\mathcal{U}$.\\
If $\mathcal{U}=\{\mathbf{0}\}$, then $m=1$. Otherwise, induction gives $\mathcal{U}=G(\lambda_2,T|_{\mathcal{U}})\oplus\cdots\oplus G(\lambda_m,T|_{\mathcal{U}})$.\\
For $j\geq2$, $G(\lambda_j,T|_{\mathcal{U}})=G(\lambda_j,T)$. Indeed, the inclusion $\subseteq$ is clear; conversely, write $\mathbf{v}\in G(\lambda_j,T)$ as $\mathbf{v}=\mathbf{g}+\mathbf{u}$ with $\mathbf{g}\in G(\lambda_1,T)$ and $\mathbf{u}\in\mathcal{U}$. Invariance of both summands gives $\mathbf{g}\in G(\lambda_1,T)\cap G(\lambda_j,T)$, hence $\mathbf{g}=\mathbf{0}$ by Theorem \ref{thm:distinct-generalized-eigenvectors-independent}. Thus $\mathbf{v}\in G(\lambda_j,T|_{\mathcal{U}})$, proving (i).
\item This is Proposition \ref{prop:generalized-eigenspaces-invariant}.
\item By Theorem \ref{thm:generalized-eigenspace-stabilized-kernel}, $(T-\lambda_jI_{\mathcal{V}})^n\mathbf{v}=\mathbf{0}$ for every $\mathbf{v}\in G(\lambda_j,T)$; hence the restriction is nilpotent.
\end{enumerate}
\hlt{Structural Commentary}: Peel off one generalized eigenspace, then induct on the invariant range.\\
\hlt{Key Insight}: Stable kernel-range splitting isolates one eigenvalue at a time.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:complex-basis-of-generalized-eigenvectors} \hlt{Basis of Generalized Eigenvectors}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{C}$ and $T\in\LL(\mathcal{V})$. Then $\mathcal{V}$ has a generalized-eigenvector basis for $T$.
\begin{proof}
Let $\lambda_1,\ldots,\lambda_m$ be the distinct eigenvalues of $T$. By Theorem \ref{thm:complex-primary-decomposition-generalized-eigenspaces}, $\mathcal{V}=G(\lambda_1,T)\oplus\cdots\oplus G(\lambda_m,T)$. Choose a basis of each generalized eigenspace and concatenate the bases. Directness makes the concatenated list a basis of $\mathcal{V}$, and every nonzero vector in each list is a generalized eigenvector.\\
\hlt{Structural Commentary}: A direct sum basis is obtained by choosing bases in each summand.\\
\hlt{Key Insight}: Bases of the direct-sum pieces concatenate.

\end{proof}
\end{corollary}

\begin{definition} \label{def:eigenvalue-multiplicity-generalized-eigenspace} \hlt{Multiplicity of an Eigenvalue}\\
Let $\mathcal{V}$ be finite-dimensional and let $T\in\LL(\mathcal{V})$.\\
The \hlt{multiplicity} of an eigenvalue $\lambda$ of $T$ is $\dim G(\lambda,T)=\dim\ker\bigl((T-\lambda I_{\mathcal{V}})^{\dim(\mathcal{V})}\bigr)$.\\
This is the algebraic multiplicity in generalized-eigenspace language.
\end{definition}

\begin{proposition} \label{prop:sum-of-complex-eigenvalue-multiplicities} \hlt{Sum of Multiplicities}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be a finite-dimensional complex vector space and let $T\in\LL(\mathcal{V})$. Then the sum of the multiplicities of all eigenvalues of $T$ equals $\dim(\mathcal{V})$.
\begin{proof}
Let $\lambda_1,\ldots,\lambda_m$ be the distinct eigenvalues of $T$. By Theorem \ref{thm:complex-primary-decomposition-generalized-eigenspaces}, $\mathcal{V}=G(\lambda_1,T)\oplus\cdots\oplus G(\lambda_m,T)$.\\
Taking dimensions, by Corollary \ref{cor:dimension-direct-sum}, $\dim(\mathcal{V})=\sum\limits_{j=1}^{m}\dim G(\lambda_j,T)$, exactly the sum of the multiplicities.\\
\hlt{Structural Commentary}: Multiplicity is dimension of the generalized-eigenspace summand.\\
\hlt{Key Insight}: Multiplicities are dimensions of the direct-sum pieces.
\end{proof}
\end{proposition}

\begin{motivation} \label{mot:characteristic-and-minimal-polynomials} \hlt{Two Spectral Polynomials}\\
The characteristic polynomial records the whole primary decomposition with dimensions.\\
The minimal polynomial records the shortest polynomial relation satisfied by the operator.
\end{motivation}

\begin{definition} \label{def:matrix-characteristic-polynomial} \hlt{Characteristic Polynomial}\\
For $A\in\operatorname{Mat}(n,n,\mathbb{F})$, the \hlt{characteristic polynomial} of $A$ is $p_A(z)\defeq\det(zI_n-A)$.
\end{definition}

\begin{theorem} \label{thm:matrix-characteristic-polynomial-basic-coefficients} \hlt{Basic Form of the Characteristic Polynomial}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{F})$. Then $p_A$ is monic of degree $n$ and $p_A(z)=z^n-(\operatorname{tr}A)z^{n-1}+\cdots+(-1)^n\det A$.\\
Moreover, for $\lambda\in\mathbb{F}$, $\lambda\in\operatorname{spec}(A)$ if and only if $p_A(\lambda)=0$.
\begin{proof}
The only determinant term of degree greater than $n-2$ comes from the identity permutation:
\begin{align}
(z-a_{11})(z-a_{22})\cdots(z-a_{nn})=z^n-(a_{11}+\cdots+a_{nn})z^{n-1}+\cdots.
\nonumber
\end{align}
Thus $p_A$ is monic and the coefficient of $z^{n-1}$ is $-\operatorname{tr}A$; also $p_A(0)=\det(-A)=(-1)^n\det A$. Finally, $p_A(\lambda)=0$ if and only if $\lambda I_n-A$ is singular, equivalently $\ker(A-\lambda I_n)\neq\{\mathbf{0}\}$.\\
\hlt{Structural Commentary}: The diagonal product determines the top two coefficients.\\
\hlt{Key Insight}: Eigenvalues are precisely the roots of $\det(zI_n-A)$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:matrix-algebraic-multiplicity} \hlt{Algebraic Multiplicity}\\
For $\lambda\in\mathbb{F}$, let $m_A(\lambda)$ be the multiplicity of $\lambda$ as a root of $p_A$, with $m_A(\lambda)=0$ if $\lambda\notin\operatorname{spec}(A)$.\\
For $\lambda\in\operatorname{spec}(A)$, this is the \hlt{algebraic multiplicity} of $\lambda$. Unqualified multiplicity means algebraic multiplicity.
\end{definition}

\begin{proposition} \label{prop:upper-triangular-characteristic-polynomial} \hlt{Upper-Triangular Eigenvalues}\\
If $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{F})$ is upper triangular, then $p_A(z)=\prod\limits_{j=1}^{n}(z-a_{jj})$.\\
Thus $\operatorname{spec}(A)=\{a_{jj}:1\leq j\leq n\}$, and for $\lambda\in\operatorname{spec}(A)$, $m_A(\lambda)=\#\{j:a_{jj}=\lambda\}$.
\begin{proof}
The matrix $zI_n-A$ is upper triangular with diagonal entries $z-a_{11},\ldots,z-a_{nn}$. Its determinant is the product of those diagonal entries. The roots of this product are precisely the diagonal entries, with repetition.\\
\hlt{Structural Commentary}: Triangular form removes all off-diagonal determinant contributions.\\
\hlt{Key Insight}: Once a matrix is triangular, the spectrum is visible on the diagonal.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:similarity-preserves-characteristic-polynomial} \hlt{Similarity Preserves Characteristic Data}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$ and suppose $B=SAS^{-1}$ for some invertible $S$. Then $p_B=p_A$. In particular, for each $\lambda\in\operatorname{spec}(A)$, $m_B(\lambda)=m_A(\lambda)$ and $\dim E_{\lambda}(B)=\dim E_{\lambda}(A)$.
\begin{proof} $p_B(z) =\det(zI_n-SAS^{-1}) =\det(S(zI_n-A)S^{-1}) =\det(S)\det(zI_n-A)\det(S)^{-1} =p_A(z) \Rightarrow$\\
$m_B(\lambda)=m_A(\lambda)$. Also $B-\lambda I_n=S(A-\lambda I_n)S^{-1}$, multiply by $S$ gives isomorphism $\ker(A-\lambda I_n)\to\ker(B-\lambda I_n)$.\\
\hlt{Structural Commentary}: Similarity changes coordinates, not the shifted determinant or shifted kernel dimension.\\
\hlt{Key Insight}: Conjugation inserts $S$ and $S^{-1}$, which cancel in determinants and identify kernels.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:transpose-conjugate-adjoint-characteristic-polynomial} \hlt{Transpose, Conjugate, and Adjoint Spectra}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $p_{A^t}(z)=p_A(z)$, $p_{\overline{A}}(z)=\overline{p_A(\overline{z})}$, and $p_{A^*}(z)=\overline{p_A(\overline{z})}$.\\
Consequently, if $\lambda\in\operatorname{spec}(A)$, then $m_{A^t}(\lambda)=m_A(\lambda)$ and $m_{\overline{A}}(\overline{\lambda})=m_{A^*}(\overline{\lambda})=m_A(\lambda)$.
\begin{proof}
Use determinant identities:
\begin{align}
p_{A^t}(z)&=\det(zI_n-A^t)=\det((zI_n-A)^t)=p_A(z), \nonumber\\
p_{\overline{A}}(z)&=\det(zI_n-\overline{A})
=\det(\overline{\overline{z}I_n-A})
=\overline{\det(\overline{z}I_n-A)}
=\overline{p_A(\overline{z})}.
\nonumber
\end{align}
Since $A^*=\overline{A}^{\,t}$, the first identity applied to $\overline{A}$ gives the formula for $A^*$.\\
Multiplicity is preserved because conjugating a polynomial sends a factor $(z-\lambda)^k$ to $(z-\overline{\lambda})^k$.\\
\hlt{Structural Commentary}: Transpose preserves $p_A$; conjugation conjugates its roots.\\
\hlt{Key Insight}: The adjoint combines conjugation with transposition, so its spectrum is conjugated.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:transpose-preserves-spectrum-and-geometric-multiplicity} \hlt{Transpose Preserves Spectrum and Geometric Multiplicity}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ and $\lambda\in\mathbb{F}$. Then $\lambda\in\operatorname{spec}(A)$ $\Longleftrightarrow$ $\lambda\in\operatorname{spec}(A^t)$, and $\dim E_{\lambda}(A)=\dim E_{\lambda}(A^t)$.
\begin{proof}
Since $(A-\lambda I_n)^t=A^t-\lambda I_n$, Theorem \ref{thm:row-rank-equals-column-rank} gives $\operatorname{rank}(A-\lambda I_n)=\operatorname{rank}(A^t-\lambda I_n)$.\\
Matrix rank-nullity then gives $\dim E_{\lambda}(A) =n-\operatorname{rank}(A-\lambda I_n) =n-\operatorname{rank}(A^t-\lambda I_n) =\dim E_{\lambda}(A^t)$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\lambda\in\operatorname{spec}(A)$, then $E_{\lambda}(A)\neq\{\mathbf{0}\}$, so $E_{\lambda}(A^t)\neq\{\mathbf{0}\}$ and $\lambda\in\operatorname{spec}(A^t)$.
\item[$(\Leftarrow)$] The same argument with $A$ and $A^t$ interchanged gives the converse.
\end{enumerate}
\hlt{Structural Commentary}: Transposition preserves rank, so shifted kernels have the same dimension.\\
\hlt{Key Insight}: Spectrum and geometric multiplicity are kernel facts for $A-\lambda I_n$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:block-upper-triangular-characteristic-polynomial} \hlt{Block Upper-Triangular Characteristic Polynomial}\\
Let $A=
\begin{bmatrix}
B & C\\
0 & D
\end{bmatrix}$, $B$ and $D$ are square. Then $p_A(z)=p_B(z)p_D(z)$, so $m_A(\lambda)=m_B(\lambda)+m_D(\lambda) \ \forall\lambda\in\mathbb{F}$.
\begin{proof}
The shifted matrix is block upper triangular:
\begin{align}
zI-A=
\begin{bmatrix}
zI-B & -C\\
0 & zI-D
\end{bmatrix}.
\nonumber
\end{align}
Theorem \ref{thm:block-upper-triangular-determinant} gives $p_A(z)=\det(zI-B)\det(zI-D)=p_B(z)p_D(z)$.\\
Roots of a product are the roots of the factors, with multiplicity added.\\
\hlt{Structural Commentary}: The off-diagonal block couples vectors but not determinant factors.\\
\hlt{Key Insight}: In block upper-triangular form, spectral data lives on the diagonal blocks.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:geometric-multiplicity-bounded-by-algebraic-multiplicity} \hlt{Geometric Multiplicity Is at Most Algebraic Multiplicity}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ and let $\lambda\in\operatorname{spec}(A)$. Then $\dim E_{\lambda}(A)\leq m_A(\lambda)$.
\begin{proof}
Let $k=\dim E_{\lambda}(A)$ and choose a basis $\mathbf{x}_1,\ldots,\mathbf{x}_k$ of $E_{\lambda}(A)$. Extend it to a basis of $\mathbb{F}^n$, and let $S$ be the matrix whose columns are this basis. In the new basis,
\begin{align}
S^{-1}AS=
\begin{bmatrix}
\lambda I_k & *\\
0 & C
\end{bmatrix}.
\nonumber
\end{align}
By Theorem \ref{thm:similarity-preserves-characteristic-polynomial} and Proposition \ref{prop:block-upper-triangular-characteristic-polynomial}, $p_A(z)=p_{S^{-1}AS}(z)=(z-\lambda)^kp_C(z)$. Thus $m_A(\lambda)\geq k=\dim E_{\lambda}(A)$.\\
\hlt{Structural Commentary}: An eigenspace-first basis exposes a $\lambda I_k$ block.\\
\hlt{Key Insight}: Each independent eigenvector forces one visible factor $z-\lambda$ in the characteristic polynomial.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:diagonalizable-iff-algebraic-geometric-multiplicities-match} \hlt{Diagonalizability and Multiplicity}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $A$ is diagonalizable if and only if $\dim E_{\lambda}(A)=m_A(\lambda)$ for every $\lambda\in\operatorname{spec}(A)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $A$ is diagonalizable, then $A$ is similar to a diagonal matrix $D$. Theorem \ref{thm:similarity-preserves-characteristic-polynomial} gives $m_A(\lambda)=m_D(\lambda)$ and $\dim E_{\lambda}(A)=\dim E_{\lambda}(D)$. For diagonal $D$, both numbers count diagonal entries equal to $\lambda$.
\item[$(\Leftarrow)$] Assume $\dim E_{\lambda}(A)=m_A(\lambda)$ for every eigenvalue.\\
If $\mu_1,\ldots,\mu_d$ are distinct eigenvalues, then $\sum\limits_{i=1}^{d}m_A(\mu_i)=n$. Hence $\dim E_{\mu_1}(A)+\cdots+\dim E_{\mu_d}(A)=n$.\\
Theorem \ref{thm:conditions-equivalent-to-diagonalizability} gives an eigenbasis, so $A$ is diagonalizable.
\end{enumerate}
\hlt{Structural Commentary}: Diagonalizability means root counts match eigenspace dimensions.\\
\hlt{Key Insight}: No missing eigendirections means an eigenbasis.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:commutant-of-diagonalizable-matrix} \hlt{Commutant of a Diagonalizable Matrix}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ be diagonalizable. Suppose its distinct eigenvalues are $\mu_1,\ldots,\mu_d$, and set $n_i=m_A(\mu_i)$.\\
Write $A=S(\mu_1I_{n_1}\oplus\cdots\oplus\mu_dI_{n_d})S^{-1}$. Then $\{A\}'=\{S(X_1\oplus\cdots\oplus X_d)S^{-1}:X_i\in\operatorname{Mat}(n_i,n_i,\mathbb{C})\}$.\\
In particular, $\dim\{A\}'=n_1^2+\cdots+n_d^2$.
\begin{proof}
Let $\Lambda=\mu_1I_{n_1}\oplus\cdots\oplus\mu_dI_{n_d}$. Matrix $B$ commutes with $A$ if and only if $S^{-1}BS$ commutes with $\Lambda$.\\
By Lemma \ref{lem:block-diagonal-commutant-distinct-scalars}, this happens exactly when $S^{-1}BS=X_1\oplus\cdots\oplus X_d$ with $X_i\in\operatorname{Mat}(n_i,n_i,\mathbb{C})$.\\
Dimension count follows because conjugation by $S$ is a vector-space isomorphism and $\dim\operatorname{Mat}(n_i,n_i,\mathbb{C})=n_i^2$.\\
\hlt{Structural Commentary}: Diagonalizing $A$ reduces commutation to respecting equal-eigenvalue blocks.\\
\hlt{Key Insight}: Distinct eigenvalue blocks cannot mix under a commuting matrix.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:commutant-distinct-eigenvalues-polynomials} \hlt{Simple Spectrum Commutants Are Polynomials}\\
If $A\in\operatorname{Mat}(n,n,\mathbb{C})$ has $n$ distinct eigenvalues, then $\{A\}'=\{p(A):p\in\mathcal{P}(\mathbb{C})\}$, $\dim\{A\}'=n$.
\begin{proof}
Write $A=S\operatorname{diag}(\lambda_1,\ldots,\lambda_n)S^{-1}$ with distinct $\lambda_i$. Theorem \ref{thm:commutant-of-diagonalizable-matrix} says every $B\in\{A\}'$ has the form $S\operatorname{diag}(\xi_1,\ldots,\xi_n)S^{-1}$. Choose the Lagrange interpolation polynomial $p(z)=\sum\limits_{i=1}^{n}\xi_i\prod\limits_{j\neq i}\frac{z-\lambda_j}{\lambda_i-\lambda_j}$.\\
Then $p(\lambda_i)=\xi_i$ for every $i$, so Theorem \ref{thm:polynomial-functional-calculus-diagonalizable-matrix} gives $B=p(A)$.\\
The dimension statement is the case $n_1=\cdots=n_n=1$ of Theorem \ref{thm:commutant-of-diagonalizable-matrix}.\\
\hlt{Structural Commentary}: Simple spectrum leaves one scalar choice per eigendirection.\\
\hlt{Key Insight}: Interpolation converts arbitrary diagonal data into a polynomial in $A$.
\end{proof}
\end{corollary}

\begin{definition} \label{def:operator-characteristic-polynomial} \hlt{Characteristic Polynomial of an Operator}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{C}$, and let $T\in\LL(\mathcal{V})$. If $\lambda_1,\ldots,\lambda_m$ are the distinct eigenvalues of $T$ and $d_j=\dim G(\lambda_j,T)$, then the \hlt{characteristic polynomial} of $T$ is $q_T(z)\defeq\prod\limits_{j=1}^{m}(z-\lambda_j)^{d_j}$.
\end{definition}

\begin{proposition} \label{prop:operator-characteristic-polynomial-degree-roots} \hlt{Degree and Roots of $q_T$}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{C}$, let $T\in\LL(\mathcal{V})$, and let $q_T$ be the characteristic polynomial of $T$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\deg q_T=\dim(\mathcal{V})$.
\item The roots of $q_T$ are precisely the eigenvalues of $T$.
\end{enumerate}
\begin{proof}
Write $q_T(z)=\prod\limits_{j=1}^{m}(z-\lambda_j)^{d_j}$, where $\lambda_1,\ldots,\lambda_m$ are the distinct eigenvalues and $d_j=\dim G(\lambda_j,T)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Proposition \ref{prop:sum-of-complex-eigenvalue-multiplicities} gives $\deg q_T=\sum_jd_j=\dim(\mathcal{V})$.
\item The displayed factorization has exactly the roots $\lambda_1,\ldots,\lambda_m$.
\end{enumerate}
\hlt{Structural Commentary}: Each primary summand contributes one root factor.\\
\hlt{Key Insight}: Multiplicity is dimension.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:cayley-hamilton-operator} \hlt{Cayley--Hamilton Theorem for Operators}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{C}$, $q_T$ be characteristic polynomial of $T\in\LL(\mathcal{V})$. Then $q_T(T)=0$.
\begin{proof}
Let $\lambda_1,\ldots,\lambda_m$ be the distinct eigenvalues and set $d_j=\dim G(\lambda_j,T)$.\\
It suffices, by primary decomposition, to show $q_T(T)$ kills each $G(\lambda_j,T)$.\\
On $G(\lambda_j,T)$, the operator $T-\lambda_jI_{\mathcal{V}}$ is nilpotent, so Corollary \ref{cor:nilpotent-power-dimension-zero} gives $(T-\lambda_jI_{\mathcal{V}})^{d_j}=0$ on $G(\lambda_j,T)$.\\
Since the factors of $q_T(T)$ commute, this factor kills the whole product on $G(\lambda_j,T)$. Hence $q_T(T)=0$.\\
\hlt{Structural Commentary}: Check the polynomial on each primary summand.\\
\hlt{Key Insight}: One factor of $q_T(T)$ kills each summand.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:cayley-hamilton-matrix} \hlt{Cayley--Hamilton Theorem}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have characteristic polynomial $p_A(z)=\det(zI_n-A)$. Then $p_A(A)=0$.
\begin{proof}
Choose a Schur form $A=UTU^*$ with diagonal $\lambda_1,\ldots,\lambda_n$.\\
Since $p_A(A)=Up_A(T)U^*$ and $p_A(z)=\prod_{i=1}^{n}(z-\lambda_i)$, it suffices to prove $(T-\lambda_1I_n)(T-\lambda_2I_n)\cdots(T-\lambda_nI_n)=0$.\\
Let $P_j\defeq\prod\limits_{i=1}^{j}(T-\lambda_iI_n)$. Prove that first $j$ columns of $P_j$ are zero. This is clear for $j=1$.\\
If true for $j$, then the first $j+1$ columns of $T-\lambda_{j+1}I_n$ lie in $\operatorname{span}(\mathbf{e}_1,\ldots,\mathbf{e}_j)$, which $P_j$ kills. Hence $P_{j+1}$ has first $j+1$ zero columns, so $P_n=0$.\\
\hlt{Structural Commentary}: Schur form reduces the theorem to a triangular column-killing induction.\\
\hlt{Key Insight}: Each factor $T-\lambda_iI_n$ kills the next leading column.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:polynomial-reduction-by-characteristic-polynomial} \hlt{Reduction Modulo the Characteristic Polynomial}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ and let $f\in\mathcal{P}(\mathbb{C})$. If $f=p_Aq+r$ with $\deg r<n$, then $f(A)=r(A)$.
\begin{proof}
Evaluate division identity at $A$: $f(A)=p_A(A)q(A)+r(A)=0q(A)+r(A)=r(A)$.\\
Theorem \ref{thm:cayley-hamilton-matrix} gives $p_A(A)=0$.\\
\hlt{Structural Commentary}: Cayley--Hamilton makes polynomial expressions in $A$ reducible modulo $p_A$.\\
\hlt{Key Insight}: Every high power of $A$ can be rewritten using powers below $n$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:powers-of-matrix-span-dimension-at-most-n} \hlt{Powers Span at Most $n$ Dimensions}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $\operatorname{span}(I_n,A,A^2,\ldots)=\operatorname{span}(I_n,A,\ldots,A^{n-1})$, so this has dimension at most $n$.
\begin{proof}
For each $k\geq0$, Corollary \ref{cor:polynomial-reduction-by-characteristic-polynomial} applied to $f(z)=z^{n+k}$ gives a polynomial $r_k$ of degree at most $n-1$ such that $A^{n+k}=r_k(A)$. Thus every power of $A$ lies in $\operatorname{span}(I_n,A,\ldots,A^{n-1})$.\\
\hlt{Structural Commentary}: Polynomial reduction collapses the infinite power list to its first $n$ terms.\\
\hlt{Key Insight}: Cayley--Hamilton gives the recurrence relation for powers of $A$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:inverse-from-characteristic-polynomial} \hlt{Inverse from the Characteristic Polynomial}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have $p_A(z)=z^n+c_{n-1}z^{n-1}+\cdots+c_1z+c_0$.\\
If $A$ is invertible, then $A^{-1}=-c_0^{-1}(A^{n-1}+c_{n-1}A^{n-2}+\cdots+c_2A+c_1I_n)$.
\begin{proof}
Since $A$ is invertible, $0\notin\operatorname{spec}(A)$, so $c_0=p_A(0)=(-1)^n\det A\neq0$.\\
Cayley--Hamilton gives $A^n+c_{n-1}A^{n-1}+\cdots+c_1A+c_0I_n=0$.\\
Move $c_0I_n$ to the other side and multiply by $-c_0^{-1}A^{-1}$.\\
\hlt{Structural Commentary}: The characteristic equation contains an identity term exactly when $A$ is invertible.\\
\hlt{Key Insight}: Solve Cayley--Hamilton for $A^{-1}$ after factoring out one $A$.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:eigenvalues-of-polynomial-in-matrix} \hlt{Eigenvalues of a Polynomial in a Matrix}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$, let $p\in\mathcal{P}(\mathbb{C})$, and let $\lambda_1,\ldots,\lambda_n$ be the eigenvalues of $A$, counted with algebraic multiplicity. Then the eigenvalues of $p(A)$ are $p(\lambda_1),\ldots,p(\lambda_n)$, counted with algebraic multiplicity.
\begin{proof}
Choose a Schur form $U^*AU=T$ with diagonal $\lambda_1,\ldots,\lambda_n$. Then $p(A)=Up(T)U^*$, and $p(T)$ is upper triangular with diagonal $p(\lambda_1),\ldots,p(\lambda_n)$. Hence $p(A)$ has precisely these eigenvalues.\\
\hlt{Structural Commentary}: Schur form reduces the claim to the diagonal.\\
\hlt{Key Insight}: A polynomial preserves triangular form and applies $p$ to each diagonal entry.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:polynomial-in-matrix-invertible-spectrum-test} \hlt{Polynomial Invertibility Test}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ and $p\in\mathcal{P}(\mathbb{C})$. If $p(\lambda)\neq0$ for every $\lambda\in\operatorname{spec}(A)$, then $p(A)$ is invertible.
\begin{proof}
By Proposition \ref{prop:eigenvalues-of-polynomial-in-matrix}, $0\notin\operatorname{spec}(p(A))$. Corollary \ref{cor:matrix-invertible-iff-zero-not-spectrum} gives invertibility.\\
\hlt{Structural Commentary}: Invertibility of $p(A)$ is checked on spectral values of $A$.\\
\hlt{Key Insight}: Avoiding zero on the spectrum prevents $p(A)$ from having a kernel.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:sylvester-linear-matrix-equation} \hlt{Sylvester Linear Matrix Equation}\\
Let $A\in\operatorname{Mat}(m,m,\mathbb{C})$ and $B\in\operatorname{Mat}(n,n,\mathbb{C})$. If $\operatorname{spec}(A)\cap\operatorname{spec}(B)=\emptyset$, then for every $C\in\operatorname{Mat}(m,n,\mathbb{C})$ the equation $AX-XB=C$ has a unique solution $X\in\operatorname{Mat}(m,n,\mathbb{C})$.
\begin{proof}
Let $L(X)=AX-XB$. Since $L$ has equal finite-dimensional domain and codomain, it suffices to show $\ker L=\{0\}$. If $AX=XB$, then $p(A)X=Xp(B)$ for every polynomial $p$. \\
With $p=p_B$, Cayley--Hamilton gives $p_B(A)X=Xp_B(B)=0$. For every $\lambda\in\operatorname{spec}(A)$, disjointness gives $p_B(\lambda)\neq0$; hence $p_B(A)$ is invertible by Corollary \ref{cor:polynomial-in-matrix-invertible-spectrum-test}. Thus $X=0$, so $L$ is bijective.\\
\hlt{Structural Commentary}: Polynomial calculus pass through homogeneous solutions; disjoint spectra invert $p_B(A)$.\\
\hlt{Key Insight}: Spectral separation turns an intertwining relation into the zero relation.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:block-upper-triangular-remove-disjoint-coupling} \hlt{Removing a Spectrally Separated Off-Diagonal Block}\\
Let $M=\begin{bmatrix}A&C\\0&B\end{bmatrix}$ and $\operatorname{spec}(A)\cap\operatorname{spec}(B)=\emptyset$ Then $M$ is similar to $A\oplus B$.
\begin{proof}
By Theorem \ref{thm:sylvester-linear-matrix-equation}, choose $X$ such that $AX-XB=-C$. Then
\begin{align}
\begin{bmatrix}I&-X\\0&I\end{bmatrix}
\begin{bmatrix}A&C\\0&B\end{bmatrix}
\begin{bmatrix}I&X\\0&I\end{bmatrix}
=
\begin{bmatrix}A&AX+C-XB\\0&B\end{bmatrix}
=
\begin{bmatrix}A&0\\0&B\end{bmatrix}.
\nonumber
\end{align}
\hlt{Structural Commentary}: A block shear cancels the off-diagonal block via Sylvester's equation.\\
\hlt{Key Insight}: Spectrally separated diagonal blocks cannot carry essential upper-right coupling.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:schur-block-form-by-eigenvalue-groups} \hlt{Schur Form Grouped by Eigenvalues}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have distinct eigenvalues $\lambda_1,\ldots,\lambda_d$ with algebraic multiplicities $n_1,\ldots,n_d$.\\
Then $A$ is unitarily similar to a block upper-triangular matrix
\begin{align}
T=
\begin{bmatrix}
T_{11}&T_{12}&\cdots&T_{1d}\\
0&T_{22}&\cdots&T_{2d}\\
\vdots&\vdots&\ddots&\vdots\\
0&0&\cdots&T_{dd}
\end{bmatrix},
\qquad
T_{ii}\in\operatorname{Mat}(n_i,n_i,\mathbb{C}),
\nonumber
\end{align}
where each $T_{ii}$ is upper triangular with all diagonal entries equal to $\lambda_i$. Moreover, $A$ is similar to $T_{11}\oplus\cdots\oplus T_{dd}$.
\begin{proof}
Order the Schur diagonal as $\lambda_1,\ldots,\lambda_1,\lambda_2,\ldots,\lambda_2,\ldots,\lambda_d,\ldots,\lambda_d$.\\
Theorem \ref{thm:matrix-schur-triangularization-ordered} gives the displayed block Schur form. Since $\operatorname{spec}(T_{11})$ is disjoint from the trailing block spectrum, Corollary \ref{cor:block-upper-triangular-remove-disjoint-coupling} splits off $T_{11}$. Iterating splits off every diagonal block.\\
\hlt{Structural Commentary}: Grouped Schur form isolates spectral blocks; Sylvester shears split disjoint groups.\\
\hlt{Key Insight}: Cross-group upper blocks are coordinate artefacts.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:commuting-with-disjoint-blocks-has-zero-off-diagonal-blocks} \hlt{Commuting with Disjoint Blocks}\\
Let $A=A_{11}\oplus\cdots\oplus A_{dd}$, where the spectra of $A_{11},\ldots,A_{dd}$ are pairwise disjoint.\\
Partition $B=[B_{ij}]$ conformally with $A$. If $AB=BA$, then $B_{ij}=0$ for all $i\neq j$.
\begin{proof}
Equating $(i,j)$ blocks in $AB=BA$ gives $A_{ii}B_{ij}-B_{ij}A_{jj}=0$.\\
For $i\neq j$, the spectra of $A_{ii}$ and $A_{jj}$ are disjoint, so Theorem \ref{thm:sylvester-linear-matrix-equation} with $C=0$ gives $B_{ij}=0$.\\
\hlt{Structural Commentary}: A commuting matrix cannot mix direct-sum blocks with disjoint spectra.\\
\hlt{Key Insight}: The off-diagonal block solves a homogeneous Sylvester equation, whose only solution is zero.
\end{proof}
\end{corollary}

\begin{definition} \label{def:monic-polynomial} \hlt{Monic Polynomial}\\
A nonzero polynomial is \hlt{monic} if its highest-degree coefficient is $1$.
\end{definition}

\begin{theorem} \label{thm:operator-minimal-polynomial-existence} \hlt{Existence of the Minimal Polynomial}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dim over $\mathbb{F}$ and let $T\in\LL(\mathcal{V})$. Then $T$ has a unique least-degree monic annihilator $\mu_T$.
\begin{proof}
Proposition \ref{prop:finite-dimensional-operators-have-annihilating-polynomials} gives a nonzero annihilator; scaling makes it monic, so a least-degree monic annihilator exists. If $p$ and $q$ are two of them, then $(p-q)(T)=0$ and $\deg(p-q)<\deg p$.\\
If $p\neq q$, scaling $p-q$ gives a smaller monic annihilator, impossible. Hence $p=q$.\\
\hlt{Structural Commentary}: Existence comes from finite-dimensional dependence; uniqueness from minimality.\\
\hlt{Key Insight}: Two least monic annihilators cannot differ.
\end{proof}
\end{theorem}

\begin{definition} \label{def:operator-minimal-polynomial} \hlt{Minimal Polynomial of an Operator}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional and let $T\in\LL(\mathcal{V})$. The \hlt{minimal polynomial} of $T$, denoted $\mu_T$, is the unique monic polynomial of smallest degree such that $\mu_T(T)=0$.
\end{definition}

\begin{definition} \label{def:minimal-polynomial-matrix} \hlt{Minimal Polynomial of a Matrix}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. The \hlt{minimal polynomial} of $A$, denoted $\mu_A$, is the unique monic polynomial of smallest positive degree such that $\mu_A(A)=0$.
\end{definition}

\begin{theorem} \label{thm:minimal-polynomial-basic-properties} \hlt{Basic Properties of the Minimal Polynomial}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$, and let $\lambda_1,\ldots,\lambda_d$ be the distinct eigenvalues of $A$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu_A$ exists, is unique, and $\deg\mu_A\leq n$.
\item If a nonzero polynomial $f$ annihilates $A$, then $\mu_A$ divides $f$. In particular, $\mu_A$ divides $p_A$.
\item There are integers $q_i$ satisfying $1\leq q_i\leq m_A(\lambda_i)$ such that $\mu_A(z)=\prod\limits_{i=1}^{d}(z-\lambda_i)^{q_i}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Cayley--Hamilton supplies the monic annihilator $p_A$, so a least-degree monic annihilator $\mu_A$ exists and has degree at most $n$. If $g$ is another one, write $g=h\mu_A+r$ with $\deg r<\deg\mu_A$. Then $r(A)=0$, so minimality gives $r=0$; monicity and equal degree give $g=\mu_A$.
\item If $f(A)=0$, divide $f=q\mu_A+r$ with $\deg r<\deg\mu_A$. Evaluating at $A$ gives $r(A)=0$, hence $r=0$ and $\mu_A\mid f$. Taking $f=p_A$ gives $\mu_A\mid p_A$.
\item Since $\mu_A\mid p_A=\prod_{i=1}^{d}(z-\lambda_i)^{m_A(\lambda_i)}$, its roots are eigenvalues with exponents at most the algebraic multiplicities. If $A\mathbf{v}=\lambda_i\mathbf{v}$ and $\mathbf{v}\neq\mathbf{0}$, then $\mu_A(\lambda_i)\mathbf{v}=\mu_A(A)\mathbf{v}=0$, so every $\lambda_i$ occurs.
\end{enumerate}
\hlt{Structural Commentary}: Polynomial division makes $\mu_A$ the least monic generator of annihilators.\\
\hlt{Key Insight}: Cayley--Hamilton gives existence; eigenvectors force the roots.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:minimal-polynomial-direct-sum-disjoint-spectra} \hlt{Minimal Polynomial of a Spectrally Disjoint Direct Sum}\\
Let $A=A_1\oplus\cdots\oplus A_d$, where the spectra of $A_1,\ldots,A_d$ are pairwise disjoint. Then $\mu_A=\mu_{A_1}\mu_{A_2}\cdots\mu_{A_d}$.
\begin{proof}
Let $f=\prod\limits_i\mu_{A_i}$. Since $f(A_i)=0$ for all $i$, $f(A)=0$ and $\mu_A\mid f$. Conversely, $\mu_A(A_i)=0$, so $\mu_{A_i}\mid\mu_A$ for each $i$. Disjoint spectra make the $\mu_{A_i}$ pairwise relatively prime, hence $f\mid\mu_A$. Thus $\mu_A=f$.\\
\hlt{Structural Commentary}: Direct sums are annihilated blockwise; disjoint spectra make the block factors coprime.\\
\hlt{Key Insight}: Each spectral component contributes its own minimal-polynomial factor.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:similarity-preserves-minimal-polynomial} \hlt{Similarity Preserves the Minimal Polynomial}\\
If $A,B\in\operatorname{Mat}(n,n,\mathbb{C})$ are similar, then $p_A=p_B$ and $\mu_A=\mu_B$.
\begin{proof}
The identity $p_A=p_B$ is Theorem \ref{thm:similarity-preserves-characteristic-polynomial}. Let $A=SBS^{-1}$, then $f(A)=Sf(B)S^{-1}$ for every polynomial $f$, so $A$ and $B$ have the same annihilating polynomials. Hence $\mu_A=\mu_B$.\\
\hlt{Structural Commentary}: Similarity preserves polynomial equations.\\
\hlt{Key Insight}: Conjugation commutes with polynomial evaluation.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:diagonalizable-iff-minimal-polynomial-squarefree} \hlt{Diagonalizability and the Minimal Polynomial}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have distinct eigenvalues $\lambda_1,\ldots,\lambda_d$. Then $A$ is diagonalizable if and only if
\begin{align}
\mu_A(z)=(z-\lambda_1)\cdots(z-\lambda_d).
\nonumber
\end{align}
Equivalently, $A$ is diagonalizable if and only if $\mu_A$ is square-free.
\begin{proof}
Set $p(z)\defeq\prod_{i=1}^{d}(z-\lambda_i)$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $A=S\Lambda S^{-1}$ is diagonalizable, then $p(\Lambda)=0$, so $p(A)=0$. Since $\mu_A$ contains every factor $z-\lambda_i$ and divides $p$, we have $\mu_A=p$.
\item[$(\Leftarrow)$] Suppose $\mu_A=p$. On each primary summand $G(\lambda_i,A)$, the factors $A-\lambda_jI_n$ with $j\neq i$ are invertible by Lemma \ref{lem:scalar-plus-nilpotent-invertible}. Since $p(A)=0$, we get $A-\lambda_iI_n=0$ on $G(\lambda_i,A)$; hence every primary summand is an eigenspace, so $A$ is diagonalizable.
\end{enumerate}
\hlt{Structural Commentary}: Repeated factors of $\mu_A$ record nilpotent parts in primary blocks.\\
\hlt{Key Insight}: Square-free $\mu_A$ kills each primary block in one step.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:squarefree-annihilator-implies-diagonalizable} \hlt{Square-Free Annihilator Criterion}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. If a polynomial $f$ with no repeated roots satisfies $f(A)=0$, then $A$ is diagonalizable.
\begin{proof}
Since $\mu_A\mid f$, the minimal polynomial is square-free; Theorem \ref{thm:diagonalizable-iff-minimal-polynomial-squarefree} applies.\\
\hlt{Structural Commentary}: A square-free annihilator forces square-free $\mu_A$.\\
\hlt{Key Insight}: Square-free $\mu_A$ means diagonalizable.
\end{proof}
\end{corollary}

\begin{method} \label{meth:operator-minimal-polynomial-from-matrix} \hlt{Computing $\mu_T$ from a Matrix}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ be the matrix of $T$ in some basis.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $m=1,2,\ldots$, test the monic candidate $p_m(z)=a_0+a_1z+\cdots+a_{m-1}z^{m-1}+z^m$.
\item Solve the linear matrix equation $a_0I+a_1A+\cdots+a_{m-1}A^{m-1}=-A^m$ for $a_0,\ldots,a_{m-1}$.\\
Equivalently, vectorize $\begin{bmatrix}
\operatorname{vec}(I)&\operatorname{vec}(A)&\cdots&\operatorname{vec}(A^{m-1})
\end{bmatrix}
\begin{bmatrix}
a_0\\a_1\\ \vdots\\ a_{m-1}
\end{bmatrix}
=-\operatorname{vec}(A^m)$
\item If no solution exists, increase $m$. The first solvable candidate is $\mu_T$.
\end{enumerate}
\end{method}

\begin{method} \label{meth:minimal-polynomial-testing-from-characteristic-polynomial} \hlt{Testing a Minimal Polynomial from $p_A$}\\
Suppose $p_A(z)=\prod_i(z-\lambda_i)^{a_i}$ is known.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Write each candidate as $m_{\mathbf{q}}(z)=\prod_i(z-\lambda_i)^{q_i}$, where $1\leq q_i\leq a_i$.
\item Test exponent vectors $\mathbf{q}=(q_i)$ in increasing total degree $\sum_iq_i$.
\item For each $\mathbf{q}$, compute $m_{\mathbf{q}}(A)=\prod_i(A-\lambda_iI)^{q_i}$.
\item The first candidate with $m_{\mathbf{q}}(A)=0$ is $\mu_A$, by Theorem \ref{thm:minimal-polynomial-basic-properties}.
\end{enumerate}
\end{method}

\begin{theorem} \label{thm:operator-annihilator-iff-multiple-minimal-polynomial} \hlt{Annihilators Are Multiples of $\mu_T$}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dim, $T\in\LL(\mathcal{V})$, and $f\in\mathcal{P}(\mathbb{F})$ be nonzero. Then $f(T)=0$ if and only if $\mu_T$ divides $f$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $f(T)=0$. Divide $f=s\mu_T+r$, where $r=0$ or $\deg r<\deg\mu_T$.\\
Then $r(T)=0$, so minimality forces $r=0$. Hence $\mu_T$ divides $f$.
\item[$(\Leftarrow)$] If $f=\mu_Ts$, then $f(T)=\mu_T(T)s(T)=0$.
\end{enumerate}
\hlt{Structural Commentary}: Polynomial division reduces every annihilator to its remainder modulo $\mu_T$.\\
\hlt{Key Insight}: Minimality forces that remainder to be zero.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:operator-characteristic-multiple-minimal-polynomial} \hlt{$q_T$ Is a Multiple of $\mu_T$}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{C}$ and let $T\in\LL(\mathcal{V})$.\\
Then the characteristic polynomial $q_T$ is a polynomial multiple of the minimal polynomial $\mu_T$.
\begin{proof}
Theorem \ref{thm:cayley-hamilton-operator} gives $q_T(T)=0$. Theorem \ref{thm:operator-annihilator-iff-multiple-minimal-polynomial} then implies that $\mu_T$ divides $q_T$.\\
\hlt{Structural Commentary}: Cayley--Hamilton turns the characteristic polynomial into an annihilator.\\
\hlt{Key Insight}: The minimal polynomial divides every annihilator, hence divides $q_T$.
\end{proof}
\end{corollary}

\begin{definition} \label{def:companion-matrix} \hlt{Companion Matrix}\\
Let $f(z)=z^n+c_{n-1}z^{n-1}+\cdots+c_1z+c_0$ be monic. For $n\geq2$, the \hlt{companion matrix} of $f$ is
\begin{align}
C_f\defeq
\begin{bmatrix}
0&0&\cdots&0&-c_0\\
1&0&\cdots&0&-c_1\\
0&1&\cdots&0&-c_2\\
\vdots&\vdots&\ddots&\vdots&\vdots\\
0&0&\cdots&1&-c_{n-1}
\end{bmatrix}.
\nonumber
\end{align}
For $n=1$, set $C_f=[-c_0]$.
\end{definition}

\begin{theorem} \label{thm:companion-matrix-has-prescribed-minimal-polynomial} \hlt{Companion Matrix Realises a Polynomial}\\
For every monic polynomial $f$, the companion matrix $C_f$ satisfies $p_{C_f}=f$ and $\mu_{C_f}=f$.
\begin{proof}
Let $\mathbf{e}_1,\ldots,\mathbf{e}_n$ be the standard basis.\\
For $j<n$, $C_f^j\mathbf{e}_1=\mathbf{e}_{j+1}$, and last column is $C_f^n\mathbf{e}_1=-c_0\mathbf{e}_1-c_1C_f\mathbf{e}_1-\cdots-c_{n-1}C_f^{n-1}\mathbf{e}_1$.\\
Thus $f(C_f)\mathbf{e}_1=0$. Since $\mathbf{e}_j=C_f^{j-1}\mathbf{e}_1$ and $f(C_f)$ commutes with $C_f$, $f(C_f)=0$.\\
If $g(z)=\sum\limits_{i=0}^{m}b_iz^i$ has $m<n$, then $g(C_f)\mathbf{e}_1=b_0\mathbf{e}_1+b_1\mathbf{e}_2+\cdots+b_m\mathbf{e}_{m+1}$, which is zero only when $g=0$.\\
Hence no nonzero polynomial of degree $<n$ annihilates $C_f$. Therefore $\deg\mu_{C_f}=n$ and $\mu_{C_f}\mid f$, so $\mu_{C_f}=f$. Cayley--Hamilton makes $p_{C_f}$ a monic degree-$n$ annihilator, hence $p_{C_f}=f$.\\
\hlt{Structural Commentary}: The cyclic vector $\mathbf{e}_1$ generates the basis, and last column imposes relation $f(C_f)=0$.\\
\hlt{Key Insight}: The companion matrix makes $f$ the first possible annihilating relation.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:minimal-polynomial-complexification} \hlt{Minimal Polynomial Survives Complexification}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{R}$ and let $T\in\LL(\mathcal{V})$. Then $\mu_{T_{\mathbb{C}}}=\mu_T$ in $\mathcal{P}(\mathbb{C})$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Since $\mu_T(T)=0$, Proposition \ref{prop:complexification-operator-basic-properties} gives $\mu_T(T_{\mathbb{C}})=(\mu_T(T))_{\mathbb{C}}=0$. Hence $\mu_{T_{\mathbb{C}}}$ divides $\mu_T$ in $\mathcal{P}(\mathbb{C})$.
\item[$(\Leftarrow)$] Let $q=\mu_{T_{\mathbb{C}}}=r+is$, with $r,s\in\mathcal{P}(\mathbb{R})$. Since $q$ is monic, $r$ is monic and $\deg r=\deg q$.\\
For every $\mathbf{u}\in\mathcal{V}$, $0=q(T_{\mathbb{C}})(\mathbf{u}+i\mathbf{0})=r(T)\mathbf{u}+is(T)\mathbf{u}$. Thus $r(T)=0$, so $\deg\mu_T\leq\deg r=\deg q$. The first part gives opposite degree inequality. As $q$ divides $\mu_T$ and both are monic with equal degree, $q=\mu_T$.
\end{enumerate}
\hlt{Structural Commentary}: Annihilation passes to complexification and back by real-part extraction.\\
\hlt{Key Insight}: Evaluate $q(T_{\mathbb{C}})$ on $\mathbf{u}+i\mathbf{0}$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:real-eigenvalues-complexification} \hlt{Real Eigenvalues of a Complexification}\\
Let $\mathcal{V}$ be real, let $T\in\LL(\mathcal{V})$, and let $\lambda\in\mathbb{R}$. Then $\lambda$ is an eigenvalue of $T$ if and only if it is an eigenvalue of $T_{\mathbb{C}}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $T\mathbf{v}=\lambda\mathbf{v}$ with $\mathbf{v}\neq\mathbf{0}$, then $T_{\mathbb{C}}(\mathbf{v}+i\mathbf{0})=\lambda(\mathbf{v}+i\mathbf{0})$.
\item[$(\Leftarrow)$] If $T_{\mathbb{C}}(\mathbf{u}+i\mathbf{v})=\lambda(\mathbf{u}+i\mathbf{v})$ and $\mathbf{u}+i\mathbf{v}\neq\mathbf{0}$, then $T\mathbf{u}=\lambda\mathbf{u}$ and $T\mathbf{v}=\lambda\mathbf{v}$.\\
Since $\mathbf{u}$ or $\mathbf{v}$ is nonzero, $T$ has eigenvalue $\lambda$.
\end{enumerate}
\hlt{Structural Commentary}: Real eigenvalues survive complexification exactly.\\
\hlt{Key Insight}: For real $\lambda$, real and imaginary parts decouple.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:complexification-conjugate-generalized-eigenspaces} \hlt{Conjugation Transfers Generalized Eigenspaces}\\
Let $\mathcal{V}$ be real, let $T\in\LL(\mathcal{V})$, let $\lambda\in\mathbb{C}$, and let $k\geq0$. Then $(T_{\mathbb{C}}-\lambda I)^k(\mathbf{u}+i\mathbf{v})=\mathbf{0}$ $\Longleftrightarrow$ $(T_{\mathbb{C}}-\overline{\lambda} I)^k(\mathbf{u}-i\mathbf{v})=\mathbf{0}$.
\begin{proof}
Let $\Gamma(\mathbf{u}+i\mathbf{v})=\mathbf{u}-i\mathbf{v}$.\\
Then $\Gamma$ is a conjugate-linear bijection, $\Gamma T_{\mathbb{C}}=T_{\mathbb{C}}\Gamma$, and $\Gamma\bigl((T_{\mathbb{C}}-\lambda I)^k\mathbf{x}\bigr) =(T_{\mathbb{C}}-\overline{\lambda} I)^k\Gamma\mathbf{x}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Apply the displayed identity to $\mathbf{x}=\mathbf{u}+i\mathbf{v}$.
\item[$(\Leftarrow)$] Apply the first direction with $\overline{\lambda}$ in place of $\lambda$ and $\mathbf{u}-i\mathbf{v}$ in place of $\mathbf{u}+i\mathbf{v}$.
\end{enumerate}
\hlt{Structural Commentary}: Conjugation commutes with $T_{\mathbb{C}}$ and conjugates scalars.\\
\hlt{Key Insight}: Conjugation pairs generalized eigenspaces.
\end{proof}
\end{lemma}

\begin{corollary} \label{cor:complexification-conjugate-eigenvalue-multiplicities} \hlt{Conjugate Eigenvalues Have Equal Multiplicity}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{R}$ and let $T\in\LL(\mathcal{V})$.\\
If $\lambda$ is an eigenvalue of $T_{\mathbb{C}}$, then $\overline{\lambda}$ is an eigenvalue of $T_{\mathbb{C}}$, and $\dim_{\mathbb{C}}G(\lambda,T_{\mathbb{C}}) =\dim_{\mathbb{C}}G(\overline{\lambda},T_{\mathbb{C}})$.
\begin{proof}
Lemma \ref{lem:complexification-conjugate-generalized-eigenspaces} with $k=1$ gives the eigenvalue pairing.\\
With $k=\dim_{\mathbb{C}}\mathcal{V}_{\mathbb{C}}$, the same lemma gives a conjugate-linear bijection $G(\lambda,T_{\mathbb{C}})\longrightarrow G(\overline{\lambda},T_{\mathbb{C}})$, and $\mathbf{x}\longmapsto\Gamma\mathbf{x}$.
Conjugate-linear bijections preserve complex dimension, so the multiplicities agree.\\
\hlt{Structural Commentary}: Conjugation mirrors generalized eigenspaces.\\
\hlt{Key Insight}: Nonreal spectral data of a real operator arrives in equal-sized conjugate pairs.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:odd-dimensional-real-operator-has-eigenvalue} \hlt{Odd-Dimensional Real Operators Have Eigenvalues}\\
Every operator on a nonzero odd-dimensional real vector space has a real eigenvalue.
\begin{proof}
Let $T\in\LL(\mathcal{V})$, where $\dim_{\mathbb{R}}\mathcal{V}$ is odd. By Proposition \ref{prop:complexification-vector-space-basis}, $\dim_{\mathbb{C}}\mathcal{V}_{\mathbb{C}}$ is odd.\\
The sum of the multiplicities of all eigenvalues of $T_{\mathbb{C}}$ equals this odd dimension by Proposition \ref{prop:sum-of-complex-eigenvalue-multiplicities}. Nonreal eigenvalues contribute in conjugate pairs of equal multiplicity by Corollary \ref{cor:complexification-conjugate-eigenvalue-multiplicities}, hence contribute an even total. Therefore $T_{\mathbb{C}}$ has at least one real eigenvalue, which is an eigenvalue of $T$ by Proposition \ref{prop:real-eigenvalues-complexification}.\\
\hlt{Structural Commentary}: Odd dimension forces a real eigenvalue.\\
\hlt{Key Insight}: Parity forces at least one real spectral direction.
\end{proof}
\end{corollary}

\begin{definition} \label{def:real-operator-characteristic-polynomial} \hlt{Characteristic Polynomial of a Real Operator}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be a finite-dimensional real vector space and let $T\in\LL(\mathcal{V})$.\\
The \hlt{characteristic polynomial} of $T$ is $q_T\defeq q_{T_{\mathbb{C}}}$, the characteristic polynomial of its complexification.
\end{definition}

\begin{theorem} \label{thm:real-operator-characteristic-polynomial-properties} \hlt{Characteristic Polynomial over $\mathbb{R}$}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be a finite-dimensional real vector space, let $T\in\LL(\mathcal{V})$, and let $q_T$ be its characteristic polynomial.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $q_T\in\mathcal{P}(\mathbb{R})$.
\item $\deg q_T=\dim_{\mathbb{R}}\mathcal{V}$.
\item The real zeros of $q_T$ are precisely the eigenvalues of $T$.
\item $q_T(T)=0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Factor $q_{T_{\mathbb{C}}}$ over $\mathbb{C}$. Real eigenvalues give real linear factors. If $\lambda\notin\mathbb{R}$ has multiplicity $m$, then $\overline{\lambda}$ has multiplicity $m$ by Corollary \ref{cor:complexification-conjugate-eigenvalue-multiplicities}, and $(z-\lambda)^m(z-\overline{\lambda})^m=\bigl(z^2-2\operatorname{Re}(\lambda)z+\abs{\lambda}^2\bigr)^m$, which is real.
\item Proposition \ref{prop:operator-characteristic-polynomial-degree-roots} and Proposition \ref{prop:complexification-vector-space-basis} give $\deg q_T=\dim_{\mathbb{R}}\mathcal{V}$.
\item The real zeros of $q_T=q_{T_{\mathbb{C}}}$ are the real eigenvalues of $T_{\mathbb{C}}$ by Proposition \ref{prop:operator-characteristic-polynomial-degree-roots}, hence the eigenvalues of $T$ by Proposition \ref{prop:real-eigenvalues-complexification}.
\item Cayley--Hamilton over $\mathbb{C}$ gives $q_T(T_{\mathbb{C}})=0$. By (i), Proposition \ref{prop:complexification-operator-basic-properties}, $q_T(T_{\mathbb{C}})=(q_T(T))_{\mathbb{C}}$, so $q_T(T)=0$.
\end{enumerate}
\hlt{Structural Commentary}: Nonreal factors pair into real quadratics.\\
\hlt{Key Insight}: Complexify, then pair conjugates.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:real-characteristic-polynomial-multiple-minimal-polynomial} \hlt{$q_T$ Is a Multiple of $\mu_T$ over $\mathbb{R}$}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be a finite-dimensional real vector space and let $T\in\LL(\mathcal{V})$.\\
Then $\deg\mu_T\leq\dim_{\mathbb{R}}\mathcal{V}$, and $q_T$ is a polynomial multiple of $\mu_T$ in $\mathcal{P}(\mathbb{R})$.
\begin{proof}
Theorem \ref{thm:real-operator-characteristic-polynomial-properties} gives $q_T(T)=0$.\\
By Theorem \ref{thm:operator-annihilator-iff-multiple-minimal-polynomial}, $\mu_T$ divides $q_T$ in $\mathcal{P}(\mathbb{R})$. The degree bound follows from $\deg q_T=\dim_{\mathbb{R}}\mathcal{V}$.\\
\hlt{Structural Commentary}: Cayley--Hamilton makes $q_T$ a real annihilator of $T$.\\
\hlt{Key Insight}: The minimal polynomial divides every real annihilator.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:operator-minimal-polynomial-roots-eigenvalues} \hlt{Roots of $\mu_T$ Are Eigenvalues}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional and let $T\in\LL(\mathcal{V})$. Then the roots of $\mu_T$ are precisely the eigenvalues of $T$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mu_T(\lambda)=0$, write $\mu_T(z)=(z-\lambda)g(z)$. Since $\deg g<\deg\mu_T$, minimality gives $g(T)\neq0$; choose $\mathbf{v}$ with $\mathbf{u}=g(T)\mathbf{v}\neq\mathbf{0}$. Then $(T-\lambda I_{\mathcal{V}})\mathbf{u}=\mu_T(T)\mathbf{v}=\mathbf{0}$.
\item[$(\Leftarrow)$] If $T\mathbf{v}=\lambda\mathbf{v}$ with $\mathbf{v}\neq\mathbf{0}$, then $0=\mu_T(T)\mathbf{v}=\mu_T(\lambda)\mathbf{v}$, so $\mu_T(\lambda)=0$.
\end{enumerate}
\hlt{Structural Commentary}: A root splits off one shifted operator.\\
\hlt{Key Insight}: The quotient cannot already annihilate $T$.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:characteristic-versus-minimal-polynomial} \hlt{Characteristic vs. Minimal Polynomial}\\
The characteristic polynomial records every primary block with its dimension. The minimal polynomial records the least polynomial needed to annihilate all blocks at once.
\end{perspective}

\begin{theorem} \label{thm:complex-block-diagonal-generalized-eigenbasis} \hlt{Block Diagonal Form with Upper-Triangular Primary Blocks}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{C}$, and let $\lambda_1,\ldots,\lambda_m$ be the distinct eigenvalues of $T\in\LL(\mathcal{V})$, with multiplicities $d_1,\ldots,d_m$. Then $T$ has a matrix of the form
\begin{equation*}
\begin{bmatrix}
A_1 & & 0\\
& \ddots & \\
0 & & A_m
\end{bmatrix},
\end{equation*}
where each $A_j\in\operatorname{Mat}(d_j,d_j,\mathbb{C})$ is upper triangular with diagonal $\lambda_j,\ldots,\lambda_j$.
\begin{proof}
Set $\mathcal{U}_j=G(\lambda_j,T)$ and $N_j=(T-\lambda_jI_{\mathcal{V}})|_{\mathcal{U}_j}$.\\
By Theorem \ref{thm:complex-primary-decomposition-generalized-eigenspaces}, the $\mathcal{U}_j$ form a $T$-invariant direct-sum decomposition and each $N_j$ is nilpotent.\\
Choose a basis of each $\mathcal{U}_j$ in which $N_j$ is strictly upper triangular. Then $T|_{\mathcal{U}_j}=\lambda_jI_{\mathcal{U}_j}+N_j$ has an upper triangular matrix with diagonal $\lambda_j,\ldots,\lambda_j$. Concatenating these bases gives the desired block diagonal matrix.\\
\hlt{Structural Commentary}: Primary decomposition gives the blocks; nilpotent triangularisation shapes them.\\
\hlt{Key Insight}: On each generalized eigenspace, $T$ is eigenvalue plus nilpotent.
\end{proof}
\end{theorem}

\begin{definition} \label{def:jordan-chain-block-and-matrix} \hlt{Jordan Chains, Blocks, and Matrices}\\
For $k\geq1$, the \hlt{nilpotent Jordan block} $J_k(0)$ is the $k\times k$ matrix with $1$ on the first superdiagonal and $0$ elsewhere:
\begin{align}
J_k(0)=
\begin{bmatrix}
0&1&0&\cdots&0\\
0&0&1&\ddots&\vdots\\
\vdots&\ddots&\ddots&\ddots&0\\
0&\cdots&0&0&1\\
0&\cdots&\cdots&0&0
\end{bmatrix}.
\nonumber
\end{align}
For $\lambda\in\mathbb{C}$, set $J_k(\lambda)\defeq\lambda I_k+J_k(0)$. A \hlt{Jordan matrix} is a direct sum of Jordan blocks. A basis is a \hlt{Jordan basis} for an operator $T$ if the matrix of $T$ in that basis is a Jordan matrix.
\end{definition}

\begin{theorem} \label{thm:nilpotent-jordan-chain-basis} \hlt{Nilpotent Jordan Chain Basis}\\
Let $\mathcal{V}$ be finite-dim, $N\in\LL(\mathcal{V})$ be nilpotent. Then $\exists$ vectors $\mathbf{v}_1,\ldots,\mathbf{v}_r\in\mathcal{V}$ and nonnegative integers $m_1,\ldots,m_r$ s.t. $\mathcal{B}_N= \bigl(N^{m_1}\mathbf{v}_1,\ldots,N\mathbf{v}_1,\mathbf{v}_1,\ldots, N^{m_r}\mathbf{v}_r,\ldots,N\mathbf{v}_r,\mathbf{v}_r\bigr)$ is a basis of $\mathcal{V}$ and $N^{m_j+1}\mathbf{v}_j=\mathbf{0}$ for every $j$.
\begin{proof}
Induct on $n=\dim(\mathcal{V})$. If $N=0$, any basis works with all $m_j=0$.\\
Now assume $N\neq0$. Then $N$ is not injective, so $\operatorname{range}(N)$ is a proper subspace, and $N|_{\operatorname{range}(N)}$ is nilpotent. By induction, $\operatorname{range}(N)$ has a chain basis $\bigl(N^{m_1}\mathbf{u}_1,\ldots,N\mathbf{u}_1,\mathbf{u}_1,\ldots, N^{m_r}\mathbf{u}_r,\ldots,N\mathbf{u}_r,\mathbf{u}_r\bigr)$ with $N^{m_j+1}\mathbf{u}_j=\mathbf{0}$.\\
Choose $\mathbf{v}_j\in\mathcal{V}$ such that $N\mathbf{v}_j=\mathbf{u}_j$, and lift the chains to $\mathcal{L}= \bigl(N^{m_1+1}\mathbf{v}_1,\ldots,N\mathbf{v}_1,\mathbf{v}_1,\ldots, N^{m_r+1}\mathbf{v}_r,\ldots,N\mathbf{v}_r,\mathbf{v}_r\bigr)$.\\
This is linearly independent: applying $N$ to any dependence among its vectors gives a dependence among range-chain basis. Only coefficients of $N^{m_j+1}\mathbf{v}_j$ may remain; but $N^{m_j+1}\mathbf{v}_j=N^{m_j}\mathbf{u}_j$, so those coefficients vanish too.\\
Extend $\mathcal{L}$ to a basis by adding $\mathbf{w}_1,\ldots,\mathbf{w}_s$. For each $i$, choose $\mathbf{x}_i\in\operatorname{span}(\mathcal{L})$ s.t. $N\mathbf{x}_i=N\mathbf{w}_i$, and $\mathbf{z}_i=\mathbf{w}_i-\mathbf{x}_i$. Then $N\mathbf{z}_i=\mathbf{0}$, and replacing each $\mathbf{w}_i$ by $\mathbf{z}_i$ preserves the span. Thus $\mathcal{L},\mathbf{z}_1,\ldots,\mathbf{z}_s$ is the desired chain basis.\\
\hlt{Structural Commentary}: Build chains in $\operatorname{range}(N)$, lift them once, then add kernel chains.\\
\hlt{Key Insight}: Applying $N$ turns lifted-chain dependences into known range-chain dependences.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:complex-jordan-form} \hlt{Jordan Form over $\mathbb{C}$}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dimensional over $\mathbb{C}$ and let $T\in\LL(\mathcal{V})$. Then $\mathcal{V}$ has a Jordan basis for $T$.
\begin{proof}
Let $\lambda_1,\ldots,\lambda_m$ be the distinct eigenvalues of $T$. By Theorem \ref{thm:complex-primary-decomposition-generalized-eigenspaces}, $\mathcal{V}=G(\lambda_1,T)\oplus\cdots\oplus G(\lambda_m,T)$, and each $N_j\defeq(T-\lambda_jI_{\mathcal{V}})|_{G(\lambda_j,T)}$ is nilpotent. Apply Theorem \ref{thm:nilpotent-jordan-chain-basis} to each $N_j$. In the resulting chain basis of $G(\lambda_j,T)$, the operator $N_j$ has direct-sum matrix blocks $J_k(0)$, so $T|_{G(\lambda_j,T)}=\lambda_jI+N_j$ has the corresponding blocks $J_k(\lambda_j)$. Concatenating the chain bases over all eigenvalues gives a Jordan basis of $\mathcal{V}$.\\
\hlt{Structural Commentary}: Primary pieces decompose into nilpotent chains.\\
\hlt{Key Insight}: Jordan form is ``eigenvalue plus nilpotent chains'' on every generalized eigenspace.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:nilpotent-jordan-block-power-rank} \hlt{Powers of a Nilpotent Jordan Block}\\
For $k\geq1$ and $p\geq0$, $\operatorname{rank}\bigl(J_k(0)^p\bigr)=\max(k-p,0)$ and $\dim\ker\bigl(J_k(0)^p\bigr)=\min(p,k)$.
\begin{proof}
Since $J_k(0)$ is the left-shift, $J_k(0)^p\mathbf{e}_j=\mathbf{0}$ for $j\leq p$ and $J_k(0)^p\mathbf{e}_j=\mathbf{e}_{j-p}$ for $j>p$. Thus the image has dimension $\max(k-p,0)$, and rank-nullity gives the kernel dimension.\\
\hlt{Structural Commentary}: A nilpotent Jordan block is a left-shift on the standard basis.\\
\hlt{Key Insight}: Each power deletes one more leading layer of the chain.
\end{proof}
\end{lemma}

\begin{definition} \label{def:weyr-characteristic} \hlt{Weyr Characteristic}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ and $\lambda\in\operatorname{spec}(A)$.\\
For $p\geq1$, define $w_p(\lambda,A)\defeq \operatorname{rank}\bigl((A-\lambda I_n)^{p-1}\bigr) - \operatorname{rank}\bigl((A-\lambda I_n)^p\bigr)$.\\
The positive part of the sequence $w_1,w_2,\ldots$ is the \hlt{Weyr characteristic} of $A$ associated with $\lambda$.
\end{definition}

\begin{theorem} \label{thm:jordan-block-sizes-from-rank-drops} \hlt{Jordan Block Sizes from Rank Drops}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$, $\lambda\in\operatorname{spec}(A)$, and $p\geq1$. Then $w_p(\lambda,A)$ is the number of $\lambda$-Jordan blocks of size at least $p$; hence $w_p(\lambda,A)-w_{p+1}(\lambda,A)$ is the number of $\lambda$-blocks of size exactly $p$.
\begin{proof}
Rank is similarity-invariant, so assume $A$ is in Jordan form.\\
Blocks $J_k(\mu)$ with $\mu\neq\lambda$ contribute $0$ to the rank drop because $J_k(\mu)-\lambda I_k$ is invertible.\\
A block $J_k(\lambda)$ contributes $\operatorname{rank}\bigl(J_k(0)^{p-1}\bigr)-\operatorname{rank}\bigl(J_k(0)^p\bigr)$, which is $1$ exactly when $p\leq k$ by Lemma \ref{lem:nilpotent-jordan-block-power-rank}.\\
Summing over blocks proves the first claim; subtracting the $p+1$ count gives the exact-size count.\\
\hlt{Structural Commentary}: Rank drops count surviving chain lengths.\\
\hlt{Key Insight}: A length-$k$ block contributes to drops $1,\ldots,k$.
\end{proof}
\end{theorem}

\begin{method} \label{meth:jordan-form-from-rank-drops} \hlt{Computing Jordan Blocks by Rank Drops}\\
For each eigenvalue $\lambda$ of $A\in\operatorname{Mat}(n,n,\mathbb{C})$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Compute $r_p=\operatorname{rank}\bigl((A-\lambda I_n)^p\bigr)$ for $p=0,1,2,\ldots$ until the differences stabilize at $0$.
\item Set $w_p=r_{p-1}-r_p$. Then $w_p$ is the number of $\lambda$-blocks of size at least $p$.
\item The number of blocks $J_p(\lambda)$ is $w_p-w_{p+1}$.
\item Assemble the Jordan form by taking the direct sum over all eigenvalues.
\end{enumerate}
\end{method}

\begin{corollary} \label{cor:jordan-form-unique-up-to-block-permutation} \hlt{Uniqueness of Jordan Form}\\
If $J$ and $J'$ are Jordan matrices similar to $A\in\operatorname{Mat}(n,n,\mathbb{C})$, they differ only by a permutation of Jordan blocks.
\begin{proof}
Similarity preserves spectra and the ranks of $(X-\lambda I_n)^p$. Thus $J$ and $J'$ have the same Weyr characteristic for each eigenvalue. Theorem \ref{thm:jordan-block-sizes-from-rank-drops} then gives the same block multiplicities, so only block order may differ.\\
\hlt{Structural Commentary}: Weyr data is similarity-invariant and fixes block multiplicities.\\
\hlt{Key Insight}: Jordan form is unique modulo block order.
\end{proof}
\end{corollary}

\begin{definition} \label{def:nonderogatory-matrix} \hlt{Nonderogatory Matrix}\\
A matrix $A\in\operatorname{Mat}(n,n,\mathbb{C})$ is \hlt{nonderogatory} if its Jordan form has exactly one Jordan block for each distinct eigenvalue. Equivalently, $\dim E_{\lambda}(A)=1$ for every $\lambda\in\operatorname{spec}(A)$.
\end{definition}

\begin{lemma} \label{lem:single-jordan-block-commutant-polynomials} \hlt{Commutant of One Jordan Block}\\
If $J=J_m(\lambda)$ and $XJ=JX$, then $X=p(J)$ for some polynomial $p$ with $\deg p<m$.
\begin{proof}
It suffices to treat $N=J_m(0)$, since $J_m(\lambda)=\lambda I_m+N$.\\
The vector $\mathbf{e}_m$ is cyclic for $N$: $(\mathbf{e}_m,N\mathbf{e}_m,\ldots,N^{m-1}\mathbf{e}_m)$ is the standard basis in reverse order.\\
Choose the unique polynomial $p$ with $\deg p<m$ such that $p(N)\mathbf{e}_m=X\mathbf{e}_m$.\\
For every $r=0,\ldots,m-1$, $p(N)N^r\mathbf{e}_m=N^rp(N)\mathbf{e}_m=N^rX\mathbf{e}_m=XN^r\mathbf{e}_m$.\\
Thus $p(N)$ and $X$ agree on a basis, so $X=p(N)=q(J)$ for $q(z)=p(z-\lambda)$.\\
\hlt{Structural Commentary}: A single Jordan block is generated by one cyclic vector.\\
\hlt{Key Insight}: The cyclic vector determines the commutant.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:nonderogatory-commutant-polynomials} \hlt{Nonderogatory Commutants Are Polynomials}\\
If $A\in\operatorname{Mat}(n,n,\mathbb{C})$ is nonderogatory, then $\{A\}'=\{p(A):p\in\mathcal{P}(\mathbb{C})\}$.
\begin{proof}
The inclusion $\{p(A):p\in\mathcal{P}(\mathbb{C})\}\subseteq\{A\}'$ is immediate.\\
For the reverse inclusion, replace $A$ by its Jordan form $J=J_{m_1}(\lambda_1)\oplus\cdots\oplus J_{m_d}(\lambda_d)$, where the $\lambda_i$ are distinct. If $XJ=JX$ and $X=[X_{ij}]$ is partitioned conformally with $J$, Corollary \ref{cor:commuting-with-disjoint-blocks-has-zero-off-diagonal-blocks} gives $X_{ij}=0$ for $i\neq j$. Each diagonal block $X_{ii}$ commutes with $J_{m_i}(\lambda_i)$, so Lemma \ref{lem:single-jordan-block-commutant-polynomials} gives $X_{ii}=p_i(J_{m_i}(\lambda_i))$ for some $\deg p_i<m_i$.\\
Because the polynomials $(z-\lambda_i)^{m_i}$ are pairwise coprime, Hermite interpolation gives a polynomial $p$ such that $p\equiv p_i\pmod{(z-\lambda_i)^{m_i}}$ for every $i$. Then $X=p(J)$. Conjugating back to $A$ proves the claim.\\
\hlt{Structural Commentary}: Disjoint eigenvalue blocks force block diagonality; one Jordan block per eigenvalue makes each diagonal commutant polynomial.\\
\hlt{Key Insight}: Hermite interpolation glues the separate block polynomials into one polynomial in $A$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:reversal-matrix} \hlt{Reversal Matrix}\\
For $k\geq1$, the \hlt{reversal matrix} $R_k\in\operatorname{Mat}(k,k,\mathbb{C})$ is defined by
\begin{align}
(R_k)_{ij}=
\begin{cases}
1,& i+j=k+1,\\
0,& i+j\neq k+1.
\end{cases}
\nonumber
\end{align}
Thus $R_k^t=R_k$ and $R_k^2=I_k$.
\end{definition}

\begin{lemma} \label{lem:jordan-block-transpose-reversal-similarity} \hlt{Reversal Conjugates a Jordan Block to Its Transpose}\\
For every $k\geq1$ and $\lambda\in\mathbb{C}$, $J_k(\lambda)^t=R_kJ_k(\lambda)R_k$. Moreover, $R_kJ_k(\lambda)$ and $J_k(\lambda)R_k$ are symmetric.
\begin{proof}
Let $N=J_k(0)$. Since $R_k$ reverses standard basis order, $R_kNR_k=N^t$.\\
Adding scalar gives $R_kJ_k(\lambda)R_k=J_k(\lambda)^t$. As $R_k=R_k^t=R_k^{-1}$, this identity gives $J_k(\lambda)^tR_k=R_kJ_k(\lambda)$ and $R_kJ_k(\lambda)^t=J_k(\lambda)R_k$, so $R_kJ_k(\lambda)$ and $J_k(\lambda)R_k$ are symmetric.\\
\hlt{Structural Commentary}: Reversal changes the superdiagonal shift into the subdiagonal shift.\\
\hlt{Key Insight}: Transposition reads the same Jordan chain backwards.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:matrix-similar-to-transpose-and-symmetric-factorizations} \hlt{Transpose Similarity and Symmetric Factorisations}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ is similar to $A^t$ via a symmetric similarity matrix.
\item $A=BC=DE$, where $B,C,D,E$ are symmetric and $B,E$ are invertible.
\end{enumerate}
\begin{proof}
Choose a Jordan form $A=SJS^{-1}$, and let $R$ be the direct sum of the reversal matrices matching the Jordan blocks of $J$. By Lemma \ref{lem:jordan-block-transpose-reversal-similarity}, $J^t=RJR$, $R^t=R$, $R^2=I_n$, and both $RJ$ and $JR$ are symmetric.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
Set $P=(S^{-1})^tRS^{-1}$. Then $P^t=P$ and $PAP^{-1}=(S^{-1})^tRJR S^t=(S^{-1})^tJ^tS^t=A^t$.
\item
Using $R^2=I_n$, $A=(SRS^t)\bigl((S^{-1})^tRJ S^{-1}\bigr)$ and $A=(SJR S^t)\bigl((S^{-1})^tR S^{-1}\bigr)$.\\
All four factors are symmetric because $R,RJ,JR$ are symmetric; $SRS^t$ and $(S^{-1})^tRS^{-1}$ are invertible.
\end{enumerate}
\hlt{Structural Commentary}: Reduce to Jordan blocks, reverse chains, then conjugate back.\\
\hlt{Key Insight}: Symmetric involutions produce both transpose similarity and symmetric factors.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:conjugate-similarity-jordan-block-criterion} \hlt{Conjugate Similarity Criterion}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $A$ is similar to $\overline{A}$ if and only if each $J_k(\lambda)$ occurs in the Jordan form of $A$ with the same multiplicity as $J_k(\overline{\lambda})$.
\begin{proof}
Let $J$ be a Jordan form of $A$. Then $\overline{J}$ is a Jordan form of $\overline{A}$, obtained by replacing each $J_k(\lambda)$ by $J_k(\overline{\lambda})$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $A$ is similar to $\overline{A}$, then $J$ and $\overline{J}$ are similar.\\
By Corollary \ref{cor:jordan-form-unique-up-to-block-permutation}, their Jordan blocks have same multiplicities.
\item[$(\Leftarrow)$] If the multiplicities agree, then $J$ and $\overline{J}$ differ only by a permutation of blocks. Hence $A$ is similar to $\overline{A}$.
\end{enumerate}
\hlt{Structural Commentary}: Conjugation changes eigenvalue labels but not block sizes.\\
\hlt{Key Insight}: Non-real Jordan blocks must occur in conjugate pairs.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:exponential-of-jordan-block} \hlt{Exponential of a Jordan Block}\\
For $k\geq1$, $\lambda\in\mathbb{C}$, and $t\in\mathbb{C}$, $e^{tJ_k(\lambda)} =e^{\lambda t}\sum\limits_{r=0}^{k-1}\frac{t^r}{r!}J_k(0)^r$.
\begin{proof}
Since $J_k(\lambda)=\lambda I_k+J_k(0)$ and the two summands commute, $e^{tJ_k(\lambda)}=e^{\lambda tI_k}e^{tJ_k(0)}$.\\
The first factor is $e^{\lambda t}I_k$, and the second exponential truncates because $J_k(0)^k=0$.\\
\hlt{Structural Commentary}: The scalar part supplies $e^{\lambda t}$; the nilpotent part supplies a finite polynomial in $t$.\\
\hlt{Key Insight}: Nilpotence turns the exponential series into a finite sum.
\end{proof}
\end{proposition}

\begin{method} \label{meth:solve-linear-ode-by-jordan-form} \hlt{Solving $\mathbf{x}'(t)=A\mathbf{x}(t)$ by Jordan Form}\\
To solve $\mathbf{x}'(t)=A\mathbf{x}(t)$ with $\mathbf{x}(0)=\mathbf{x}_0$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Put $A=SJS^{-1}$, where $J=\bigoplus_{i=1}^rJ_{k_i}(\lambda_i)$.
\item Set $\mathbf{y}(t)=S^{-1}\mathbf{x}(t)$ and $\mathbf{y}_0=S^{-1}\mathbf{x}_0$. Then $\mathbf{y}'(t)=J\mathbf{y}(t)$ and $\mathbf{x}(t)=S\mathbf{y}(t)$.
\item Split $\mathbf{y}=(\mathbf{y}_1,\ldots,\mathbf{y}_r)$ according to the Jordan blocks. Then $\mathbf{y}_i'(t)=J_{k_i}(\lambda_i)\mathbf{y}_i(t)$.
\item For each block, use Proposition \ref{prop:exponential-of-jordan-block}: $\mathbf{y}_i(t) =e^{tJ_{k_i}(\lambda_i)}\mathbf{y}_i(0) =e^{\lambda_i t}\sum\limits_{q=0}^{k_i-1}\frac{t^q}{q!}J_{k_i}(0)^q\mathbf{y}_i(0)$.
\item Reassemble $\mathbf{y}(t)$ and transform back: $\mathbf{x}(t)=S\mathbf{y}(t)$.
\end{enumerate}
Thus a size-$k$ block with eigenvalue $\lambda$ contributes entries of the form $e^{\lambda t}p(t)$, where $\deg p<k$.
\end{method}

\begin{definition} \label{def:spectral-radius-and-convergent-matrix} \hlt{Spectral Radius and Convergent Matrix}\\
For $A\in\operatorname{Mat}(n,n,\mathbb{C})$, the \hlt{spectral radius} is $\rho(A)\defeq\max\{\abs{\lambda}:\lambda\in\operatorname{spec}(A)\}$.\\
We say $A^p\to0$ if every entry of $A^p$ converges to $0$ as $p\to\infty$.
\end{definition}

\begin{proposition} \label{prop:powers-of-jordan-block} \hlt{Powers of a Jordan Block}\\
For $p\geq0$, $J_k(\lambda)^p =\sum\limits_{r=0}^{\min(p,k-1)}\binom{p}{r}\lambda^{p-r}J_k(0)^r$. Thus $J_k(\lambda)^p\to0$ if and only if $\abs{\lambda}<1$.
\begin{proof} Formula follows from binomial theorem applied to $J_k(\lambda)=\lambda I_k+J_k(0)$, with truncation from $J_k(0)^k=0$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $J_k(\lambda)^p\to0$, then its diagonal entries $\lambda^p$ tend to $0$, so $\abs{\lambda}<1$.
\item[$(\Leftarrow)$] If $\abs{\lambda}<1$, then each coefficient $\binom{p}{r}\lambda^{p-r} \rightarrow 0$ for fixed $r$; hence every term in finite expansion tends to $0$.
\end{enumerate}
\hlt{Structural Commentary}: Jordan powers are exponential-polynomial terms.\\
\hlt{Key Insight}: Polynomial growth cannot defeat exponential decay when $\abs{\lambda}<1$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:matrix-powers-converge-to-zero-spectral-radius} \hlt{Matrix Powers Converge to Zero}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $A^p\to0$ if and only if $\rho(A)<1$.
\begin{proof}
Choose a Jordan form $A=SJS^{-1}$.\\
Since $S$ and $S^{-1}$ are fixed, $A^p\to0$ if and only if every Jordan block power in $J^p$ converges to $0$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Each block power tends to $0$, so Proposition \ref{prop:powers-of-jordan-block} gives $\abs{\lambda}<1$ for every eigenvalue. Thus $\rho(A)<1$.
\item[$(\Leftarrow)$] If $\rho(A)<1$, then every eigenvalue satisfies $\abs{\lambda}<1$. Proposition \ref{prop:powers-of-jordan-block} makes each block power tend to $0$.
\end{enumerate}
\hlt{Structural Commentary}: Similarity reduces matrix-power convergence to Jordan block-power convergence.\\
\hlt{Key Insight}: The largest eigenvalue modulus controls the asymptotic size of all blocks.
\end{proof}
\end{theorem}

\begin{definition} \label{def:power-bounded-matrix} \hlt{Power Bounded Matrix}\\
A matrix $A\in\operatorname{Mat}(n,n,\mathbb{C})$ is \hlt{power bounded} if for each entry position $(i,j)$, the sequence of entries of $A^p$ in that position is bounded for $p=1,2,\ldots$.
\end{definition}

\begin{theorem} \label{thm:matrix-power-bounded-jordan-criterion} \hlt{Power Bounded Jordan Criterion}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ is power bounded.
\item $\rho(A)\leq1$, and every eigenvalue $\lambda$ with $\abs{\lambda}=1$ has only $1\times1$ Jordan blocks.
\item $\rho(A)\leq1$, and $\dim E_{\lambda}(A)=m_A(\lambda)$ for every $\lambda\in\operatorname{spec}(A)$ with $\abs{\lambda}=1$.
\end{enumerate}
\begin{proof}
Work in Jordan form.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow(ii)$: If $\abs{\lambda}>1$, the diagonal entries $\lambda^p$ are unbounded.\\
If $\abs{\lambda}=1$ and $J_k(\lambda)$ has $k\geq2$, then the first superdiagonal contains $p\lambda^{p-1}$, also unbounded.
\item[$(ii)$]$\Rightarrow(i)$: Blocks with $\abs{\lambda}<1$ converge to $0$ by Proposition \ref{prop:powers-of-jordan-block}; $1\times1$ blocks with $\abs{\lambda}=1$ are bounded. Finite direct sums and fixed similarities preserve boundedness.
\item[$(ii)$]$\Leftrightarrow(iii)$: For fixed $\lambda$, $\dim E_{\lambda}(A)$ counts Jordan blocks, while $m_A(\lambda)$ counts their total size.\\
Equality holds exactly when every block has size $1$.
\end{enumerate}
\hlt{Structural Commentary}: Unit-circle eigenvalues are harmless only when their nilpotent parts vanish.\\
\hlt{Key Insight}: Nontrivial Jordan blocks on the unit circle create polynomial growth.
\end{proof}
\end{theorem}

\begin{definition} \label{def:probability-vector-and-markov-matrix} \hlt{Probability Vectors and Markov Matrices}\\
A vector $\mathbf{x}\in\mathbb{R}^n$ is a \hlt{probability vector} if its entries are nonnegative and sum to $1$. Let $\mathbf{e}\defeq(1,\ldots,1)^t$. A matrix $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{R})$ is a \hlt{Markov matrix} if $a_{ij}\geq0$ for all $i,j$ and $A\mathbf{e}=\mathbf{e}$; equivalently, every row sum is $1$.
\end{definition}

\begin{lemma} \label{lem:markov-matrix-basic-properties} \hlt{Basic Properties of Markov Matrices}\\
If $A\in\operatorname{Mat}(n,n,\mathbb{R})$ is Markov, then $1\in\operatorname{spec}(A)$, and each $A^p$ is Markov with entries in $[0,1]$.
\begin{proof}
Since $A\mathbf{e}=\mathbf{e}$, $1$ is an eigenvalue. Inductively, $A^p$ has nonnegative entries and $A^p\mathbf{e}=\mathbf{e}$, so $A^p$ is Markov. Each nonnegative row sums to $1$, so all entries lie in $[0,1]$.\\
\hlt{Structural Commentary}: Markov structure is closed under powers.\\
\hlt{Key Insight}: Nonnegative row sums of $1$ bound every entry.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:markov-matrix-unit-circle-semisimple} \hlt{Unit-Circle Spectrum of Markov Matrices}\\
If $A\in\operatorname{Mat}(n,n,\mathbb{R})$ is Markov, then $\rho(A)=1$, and $\dim E_{\lambda}(A)=m_A(\lambda)$ for every $\lambda\in\operatorname{spec}(A)$ with $\abs{\lambda}=1$.
\begin{proof}
Lemma \ref{lem:markov-matrix-basic-properties} makes $A$ power bounded. Theorem \ref{thm:matrix-power-bounded-jordan-criterion} gives $\rho(A)\leq1$ and $\dim E_{\lambda}(A)=m_A(\lambda)$ on $\abs{\lambda}=1$. Since $A\mathbf{e}=\mathbf{e}$, $1\in\operatorname{spec}(A)$, so $\rho(A)\geq1$. Thus $\rho(A)=1$.\\
\hlt{Structural Commentary}: Bounded Markov powers rule out unit-circle Jordan growth.\\
\hlt{Key Insight}: $A\mathbf{e}=\mathbf{e}$ forces the eigenvalue $1$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:positive-markov-stationary-distribution} \hlt{Positive Markov Matrices Converge to a Stationary Distribution}\\
Let $n\geq2$ and let $A\in\operatorname{Mat}(n,n,\mathbb{R})$ be a Markov matrix with positive entries. Then there is a unique probability vector $\mathbf{x}$ with $A^t\mathbf{x}=\mathbf{x}$; its entries are positive and $\lim\limits_{p\to\infty}A^p=\mathbf{e}\mathbf{x}^t$.
\begin{proof}
Let $A\mathbf{y}=\lambda\mathbf{y}$ with $\abs{\lambda}=1$, and choose $k$ with $\abs{y_k}=\norm{\mathbf{y}}_{\infty}$.\\
Positivity and the $k$th row sum give $\norm{\mathbf{y}}_{\infty} =\abs{\lambda y_k} =\abs{\sum\limits_{j=1}^{n}a_{kj}y_j} \leq\sum\limits_{j=1}^{n}a_{kj}\abs{y_j} \leq\norm{\mathbf{y}}_{\infty}$.\\
Equality forces all $\abs{y_j}=\norm{\mathbf{y}}_{\infty}$ and all $y_j$ to have the same argument. Hence $\mathbf{y}=c\mathbf{e}$, and $A\mathbf{e}=\mathbf{e}$ gives $\lambda=1$. Thus $E_1(A)=\operatorname{span}(\mathbf{e})$ and no other eigenvalue lies on the unit circle.\\
By Theorem \ref{thm:markov-matrix-unit-circle-semisimple}, the Jordan form is $[1]\oplus J$ with $\rho(J)<1$. Choose $S=[\mathbf{e}\ Y]$ such that $A=S([1]\oplus J)S^{-1}$, and let $\mathbf{x}^t$ be the first row of $S^{-1}$. Then $\mathbf{x}^tA=\mathbf{x}^t$ and $\mathbf{x}^t\mathbf{e}=1$. Since $J^p\to0$, $A^p=S([1]\oplus J^p)S^{-1}\to \mathbf{e}\mathbf{x}^t$.\\
The nonnegative powers $A^p$ imply $\mathbf{x}\geq0$; since $A^t\mathbf{x}=\mathbf{x}$ and $A^t$ has positive entries, in fact $\mathbf{x}>0$. Finally, $\dim E_1(A^t)=\dim E_1(A)=1$, so $\mathbf{x}^t\mathbf{e}=1$ gives uniqueness among probability vectors.\\
\hlt{Structural Commentary}: Positivity leaves one simple unit-circle block; all other blocks decay.\\
\hlt{Key Insight}: The limit is the rank-one projection $\mathbf{e}\mathbf{x}^t$.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:identity-plus-nilpotent-square-root} \hlt{Identity Plus Nilpotent Has a Square Root}\\
Let $\mathcal{V}$ be a complex vector space and let $N\in\LL(\mathcal{V})$ be nilpotent. Then $I_{\mathcal{V}}+N$ has a square root.
\begin{proof}
Choose $m\geq1$ such that $N^m=0$. If $m=1$, then $N=0$ and $I_{\mathcal{V}}$ is a square root.\\
Otherwise seek $R=I_{\mathcal{V}}+a_1N+\cdots+a_{m-1}N^{m-1}$. With $a_0=1$, choose $a_1,\ldots,a_{m-1}$ recursively so the coefficients of $R^2$ match those of $I_{\mathcal{V}}+N$: $2a_1=1$, and $2a_k+\sum\limits_{r=1}^{k-1}a_ra_{k-r}=0\quad (2\leq k\leq m-1)$\\
Terms of degree $\geq m$ vanish because $N^m=0$, so $R^2=I_{\mathcal{V}}+N$.\\
\hlt{Structural Commentary}: Nilpotence reduces the square-root series to a finite polynomial.\\
\hlt{Key Insight}: Match coefficients recursively.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:complex-invertible-operator-has-square-root} \hlt{Invertible Complex Operators Have Square Roots}\\
Let $\mathcal{V}$ be finite-dimensional over $\mathbb{C}$ and let $T\in\LL(\mathcal{V})$ be invertible. Then $T$ has a square root.
\begin{proof}
Let $\mathcal{U}_j=G(\lambda_j,T)$ for the distinct eigenvalues of $T$. Primary decomposition gives $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_m$, and invertibility gives $\lambda_j\neq0$. Put $N_j=(T-\lambda_jI_{\mathcal{V}})|_{\mathcal{U}_j}$.\\
Then $T|_{\mathcal{U}_j} =\lambda_j\left(I_{\mathcal{U}_j}+\lambda_j^{-1}N_j\right)$ with $\lambda_j^{-1}N_j$ nilpotent. By Lemma \ref{lem:identity-plus-nilpotent-square-root}, choose $B_j^2=I_{\mathcal{U}_j}+\lambda_j^{-1}N_j$; choose $\alpha_j^2=\lambda_j$ and set $R_j=\alpha_jB_j$. Then $R_j^2=T|_{\mathcal{U}_j}$. Define $R=R_1\oplus\cdots\oplus R_m$ on the direct sum. Then $R^2=T$.\\
\hlt{Structural Commentary}: Take square roots on primary blocks, then recombine.\\
\hlt{Key Insight}: Invertibility makes every scalar block nonzero.
\end{proof}
\end{theorem}
