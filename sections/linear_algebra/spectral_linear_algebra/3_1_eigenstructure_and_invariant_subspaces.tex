\subsubsection{Eigenstructure and Invariant Subspaces}

The guiding idea of spectral linear algebra is to understand an operator by finding subspaces that it maps into themselves. On such subspaces, the operator can be studied without leaving the smaller space.

\begin{definition} \label{def:invariant-subspace} \hlt{Invariant Subspace}\\
Let $T\in\LL(\mathcal{V})$. A subspace $\mathcal{U}$ of $\mathcal{V}$ is \hlt{invariant under $T$} if $T\mathbf{u}\in\mathcal{U}$ for every $\mathbf{u}\in\mathcal{U}$.
\end{definition}

Equivalently, $\mathcal{U}$ is invariant under $T$ exactly when the restriction $T|_{\mathcal{U}}$ is an operator on $\mathcal{U}$.

\begin{definition} \label{def:eigenvalue-eigenvector} \hlt{Eigenvalue and Eigenvector}\\
Let $T\in\LL(\mathcal{V})$. A scalar $\lambda\in\mathbb{F}$ is an \hlt{eigenvalue} of $T$ if there exists $\mathbf{v}\in\mathcal{V}$ with $\mathbf{v}\neq\mathbf{0}$ and $T\mathbf{v}=\lambda\mathbf{v}$. Such a vector $\mathbf{v}$ is an \hlt{eigenvector} of $T$ corresponding to $\lambda$.
\end{definition}

\begin{definition} \label{def:eigenspace} \hlt{Eigenspace}\\
Let $T\in\LL(\mathcal{V})$ and $\lambda\in\mathbb{F}$. The \hlt{eigenspace} of $T$ corresponding to $\lambda$ is $E_{\lambda}(T)\defeq\ker(T-\lambda I_{\mathcal{V}})$. Thus $\lambda$ is an eigenvalue of $T$ if and only if $E_{\lambda}(T)\neq\{\mathbf{0}\}$. Equivalently, $E_{\lambda}(T)$ is the nullspace of the shifted operator $T-\lambda I_{\mathcal{V}}$.
\end{definition}

\begin{definition} \label{def:spectrum-and-geometric-multiplicity} \hlt{Spectrum and Geometric Multiplicity}\\
Let $T\in\LL(\mathcal{V})$. The \hlt{spectrum} of $T$ is $\operatorname{spec}(T)\defeq\{\lambda\in\mathbb{F}:\lambda\text{ is an eigenvalue of }T\}$.\\
The \hlt{geometric multiplicity} of $\lambda\in\mathbb{F}$ is $\dim E_{\lambda}(T)$. For a square matrix $A\in\operatorname{Mat}(n,n,\mathbb{F})$, $E_{\lambda}(A)\defeq\ker(A-\lambda I_n)$ and $\operatorname{spec}(A)$ denotes the spectrum of the induced operator on $\mathbb{F}^n$.
\end{definition}

\begin{proposition} \label{prop:one-dimensional-invariant-subspaces-and-eigenvectors} \hlt{One-Dimensional Invariant Subspaces and Eigenvectors}\\
Let $T\in\LL(\mathcal{V})$, $\mathbf{v}\neq\mathbf{0}$. Then $\operatorname{span}(\mathbf{v})$ is invariant under $T$ if and only if there exists $\lambda\in\mathbb{F}$ such that $T\mathbf{v}=\lambda\mathbf{v}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\operatorname{span}(\mathbf{v})$ is invariant under $T$, then $T\mathbf{v}\in\operatorname{span}(\mathbf{v})$, so $T\mathbf{v}=\lambda\mathbf{v}$ for some $\lambda\in\mathbb{F}$.
\item[$(\Leftarrow)$] If $T\mathbf{v}=\lambda\mathbf{v}$ and $\mathbf{u}=a\mathbf{v}\in\operatorname{span}(\mathbf{v})$, then $T\mathbf{u}=aT\mathbf{v}=a\lambda\mathbf{v}\in\operatorname{span}(\mathbf{v})$.
\end{enumerate}
\hlt{Structural Commentary}: One-dimensional invariance means rescaling one direction.\\
\hlt{Key Insight}: In dimension one, staying inside the subspace means being a scalar multiple of the same vector.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:eigenvectors-as-kernel-vectors} \hlt{Eigenvectors as Kernel Vectors}\\
Let $T\in\LL(\mathcal{V})$, $\lambda\in\mathbb{F}$. Vector $\mathbf{v}$ is eigenvector of $T$ corresponding to $\lambda$ if and only if $\mathbf{v}\in\ker(T-\lambda I_{\mathcal{V}})$, $\mathbf{v}\neq\mathbf{0}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathbf{v}$ is an eigenvector corresponding to $\lambda$, then $\mathbf{v}\neq\mathbf{0}$ and $T\mathbf{v}=\lambda\mathbf{v}$, so $(T-\lambda I_{\mathcal{V}})\mathbf{v}=\mathbf{0}$.
\item[$(\Leftarrow)$] If $\mathbf{v}\in\ker(T-\lambda I_{\mathcal{V}})$ and $\mathbf{v}\neq\mathbf{0}$, then $(T-\lambda I_{\mathcal{V}})\mathbf{v}=\mathbf{0}$, so $T\mathbf{v}=\lambda\mathbf{v}$.
\end{enumerate}
\hlt{Structural Commentary}: Eigenvectors are kernels of shifted operators.\\
\hlt{Key Insight}: Subtract $\lambda I_{\mathcal{V}}$ to convert $T\mathbf{v}=\lambda\mathbf{v}$ into homogeneous form.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:equivalent-eigenvalue-conditions} \hlt{Equivalent Conditions for an Eigenvalue}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and let $\lambda\in\mathbb{F}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\lambda$ is an eigenvalue of $T$;
\item $T-\lambda I_{\mathcal{V}}$ is not injective;
\item $T-\lambda I_{\mathcal{V}}$ is not surjective;
\item $T-\lambda I_{\mathcal{V}}$ is not invertible.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $\lambda$ is an eigenvalue, Proposition \ref{prop:eigenvectors-as-kernel-vectors} gives nonzero vector in $\ker(T-\lambda I_{\mathcal{V}})$, so $T-\lambda I_{\mathcal{V}}$ not injective.
\item[$(ii)$]$\Rightarrow (i)$ If $T-\lambda I_{\mathcal{V}}$ not injective, then $\ker(T-\lambda I_{\mathcal{V}})$ contains nonzero vector, so $\lambda$ is eigenvalue (Prop \ref{prop:eigenvectors-as-kernel-vectors}).
\item[$(ii)$]$\Rightarrow (iii)$ If $T-\lambda I_{\mathcal{V}}$ were surjective, Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences} would make it injective, a contradiction.
\item[$(iii)$]$\Rightarrow (iv)$ If $T-\lambda I_{\mathcal{V}}$ were invertible, then it would be surjective, a contradiction.
\item[$(iv)$]$\Rightarrow (ii)$ If $T-\lambda I_{\mathcal{V}}$ were injective, Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences} would make it invertible, a contradiction.
\end{enumerate}
\hlt{Structural Commentary}: In finite dimensions, noninvertibility has equivalent forms.\\
\hlt{Key Insight}: A scalar $\lambda$ is spectral precisely when $T-\lambda I_{\mathcal{V}}$ has a nontrivial kernel.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:distinct-eigenvalues-independent-eigenvectors} \hlt{Eigenvectors for Distinct Eigenvalues Are Independent}\\
Let $T\in\LL(\mathcal{V})$. Suppose $\lambda_1,\ldots,\lambda_m$ are distinct eigenvalues of $T$, and let $\mathbf{v}_1,\ldots,\mathbf{v}_m$ be corresponding eigenvectors. Then $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is linearly independent.
\begin{proof}
Suppose the list is linearly dependent. Let $k$ be smallest positive integer s.t. $\mathbf{v}_k\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_{k-1})$.\\
Then $k\geq2$, $(\mathbf{v}_1,\ldots,\mathbf{v}_{k-1})$ is linearly independent, and $\exists$ scalars $a_1,\ldots,a_{k-1}$ s.t. $\mathbf{v}_k=a_1\mathbf{v}_1+\cdots+a_{k-1}\mathbf{v}_{k-1}$.\\
Applying $T$ gives $\lambda_k\mathbf{v}_k=a_1\lambda_1\mathbf{v}_1+\cdots+a_{k-1}\lambda_{k-1}\mathbf{v}_{k-1}$.
Multiplying first displayed equation by $\lambda_k$ and subtracting gives $0=a_1(\lambda_k-\lambda_1)\mathbf{v}_1+\cdots+a_{k-1}(\lambda_k-\lambda_{k-1})\mathbf{v}_{k-1}$.
Since $\mathbf{v}_1,\ldots,\mathbf{v}_{k-1}$ are linearly independent and $\lambda_k\neq\lambda_i$ for $i<k$, all $a_i=0$. Hence $\mathbf{v}_k=\mathbf{0}$, contradicting that $\mathbf{v}_k$ is an eigenvector.\\
\hlt{Structural Commentary}: A smallest dependence isolates the first new eigenvector that depends on earlier ones. Applying $T$ and subtracting separates the distinct eigenvalue factors.\\
\hlt{Key Insight}: Distinct eigenvalues make the same dependence relation impossible after applying $T$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:number-of-distinct-eigenvalues} \hlt{Number of Distinct Eigenvalues}\\
If $\mathcal{V}$ is finite-dimensional and $T\in\LL(\mathcal{V})$, then $T$ has at most $\dim(\mathcal{V})$ distinct eigenvalues.
\begin{proof}
Choose one eigenvector for each distinct eigenvalue of $T$. By Theorem \ref{thm:distinct-eigenvalues-independent-eigenvectors}, these eigenvectors form a linearly independent list. A linearly independent list in $\mathcal{V}$ has length at most $\dim(\mathcal{V})$ by Theorem \ref{thm:finitedimlenofvec}.\\
\hlt{Structural Commentary}: Distinct eigenvalues give independent directions.\\
\hlt{Key Insight}: Count one eigenvector per eigenspace.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:invertible-iff-zero-not-eigenvalue} \hlt{Invertibility and the Zero Eigenvalue}\\
Let $\mathcal{V}$ be finite-dimensional and let $T\in\LL(\mathcal{V})$. Then $T$ is invertible if and only if $0\notin\operatorname{spec}(T)$.
\begin{proof}
Apply Theorem \ref{thm:equivalent-eigenvalue-conditions} with $\lambda=0$. Then $0$ is an eigenvalue exactly when $T-0I_{\mathcal{V}}=T$ is not invertible.\\
\hlt{Structural Commentary}: The zero eigenvalue is the spectral form of singularity.\\
\hlt{Key Insight}: Invertibility is equivalent to having trivial kernel, and $E_0(T)=\ker T$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:matrix-invertible-iff-zero-not-spectrum} \hlt{Matrix Invertibility and the Spectrum}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$. Then $A$ is invertible if and only if $0\notin\operatorname{spec}(A)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $A$ is invertible, then $\ker(A)=\{\mathbf{0}\}$, so $0$ is not an eigenvalue of $A$.
\item[$(\Leftarrow)$] If $0\notin\operatorname{spec}(A)$, then $\ker(A)=\{\mathbf{0}\}$. Corollary \ref{cor:matrix-rank-nullity} gives full rank, hence $A$ is invertible.
\end{enumerate}
\hlt{Structural Commentary}: Matrix invertibility is the coordinate form of trivial kernel.\\
\hlt{Key Insight}: A zero eigenvalue is exactly a nonzero solution of $A\mathbf{x}=\mathbf{0}$.
\end{proof}
\end{corollary}

\begin{definition} \label{def:diagonal-and-upper-triangular-matrices} \hlt{Diagonal and Upper-Triangular Matrices}\\
The \hlt{diagonal} of a square matrix is its upper-left to lower-right entries. A square matrix is \hlt{diagonal} if all off-diagonal entries are zero, and \hlt{upper triangular} if all entries below the diagonal are zero.
\end{definition}

\begin{definition} \label{def:diagonalizable-operator} \hlt{Diagonalizable Operator}\\
An operator $T\in\LL(\mathcal{V})$ is \hlt{diagonalizable} if $T$ has a diagonal matrix with respect to some basis of $\mathcal{V}$.
\end{definition}

\begin{proposition} \label{prop:eigenspaces-for-distinct-eigenvalues-form-direct-sum} \hlt{Eigenspaces for Distinct Eigenvalues Form a Direct Sum}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and suppose $\lambda_1,\ldots,\lambda_m$ are distinct eigenvalues of $T$.\\
Then $E_{\lambda_1}(T)+\cdots+E_{\lambda_m}(T)$ is a direct sum. Consequently, $\dim E_{\lambda_1}(T)+\cdots+\dim E_{\lambda_m}(T)\leq\dim(\mathcal{V})$.
\begin{proof}
Suppose $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ with $\mathbf{u}_j\in E_{\lambda_j}(T)$. If some $\mathbf{u}_j$ were nonzero, then the nonzero vectors among $\mathbf{u}_1,\ldots,\mathbf{u}_m$ would be eigenvectors corresponding to distinct eigenvalues, and Theorem \ref{thm:distinct-eigenvalues-independent-eigenvectors} would make them linearly independent. This contradicts their sum being $\mathbf{0}$. Hence every $\mathbf{u}_j=\mathbf{0}$, so the sum is direct.\\
By Theorem \ref{thm:direct-sum-iff-dimensions-add}, $\dim(E_{\lambda_1}(T)+\cdots+E_{\lambda_m}(T))=\dim E_{\lambda_1}(T)+\cdots+\dim E_{\lambda_m}(T)$.\\
The left side is the dimension of a subspace of $\mathcal{V}$, so Proposition \ref{prop:dimension-bounds-and-basis-criteria} gives the stated inequality.\\
\hlt{Structural Commentary}: Distinct eigenspaces form independent pieces.\\
\hlt{Key Insight}: A zero sum across them would contradict eigenvector independence.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:conditions-equivalent-to-diagonalizability} \hlt{Conditions Equivalent to Diagonalizability}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and let $\lambda_1,\ldots,\lambda_m$ denote the distinct eigenvalues of $T$. Set $n=\dim(\mathcal{V})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is diagonalizable;
\item $\mathcal{V}$ has a basis consisting of eigenvectors of $T$;
\item $\exists$ one-dimensional subspaces $\mathcal{U}_1,\ldots,\mathcal{U}_n$ of $\mathcal{V}$, each invariant under $T$, such that $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_n$;
\item $\mathcal{V}=E_{\lambda_1}(T)\oplus\cdots\oplus E_{\lambda_m}(T)$;
\item $\dim(\mathcal{V})=\dim E_{\lambda_1}(T)+\cdots+\dim E_{\lambda_m}(T)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $T$ is diagonalizable, then $T$ has a diagonal matrix with respect to some basis $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$. Hence $T\mathbf{v}_j=a_j\mathbf{v}_j$ for each $j$, so this basis consists of eigenvectors of $T$.
\item[$(ii)$]$\Rightarrow (i)$ If $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ is a basis of eigenvectors of $T$, then $T\mathbf{v}_j=a_j\mathbf{v}_j$ for each $j$. Thus the matrix of $T$ with respect to this basis is diagonal.
\item[$(ii)$]$\Rightarrow (iii)$ Let $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be an eigenbasis, and put $\mathcal{U}_j=\operatorname{span}(\mathbf{v}_j)$. Each $\mathcal{U}_j$ is one-dimensional and invariant under $T$ by Proposition \ref{prop:one-dimensional-invariant-subspaces-and-eigenvectors}; the basis decomposition gives $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_n$.
\item[$(iii)$]$\Rightarrow (ii)$ Choose nonzero $\mathbf{v}_j\in\mathcal{U}_j$. Proposition \ref{prop:one-dimensional-invariant-subspaces-and-eigenvectors} makes each $\mathbf{v}_j$ an eigenvector of $T$, and the direct-sum decomposition makes $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ a basis of $\mathcal{V}$.
\item[$(ii)$]$\Rightarrow (iv)$ Every vector in an eigenbasis lies in one of $E_{\lambda_1}(T),\ldots,E_{\lambda_m}(T)$, so these eigenspaces span $\mathcal{V}$. Proposition \ref{prop:eigenspaces-for-distinct-eigenvalues-form-direct-sum} shows their sum is direct.
\item[$(iv)$]$\Rightarrow (v)$ If $\mathcal{V}=E_{\lambda_1}(T)\oplus\cdots\oplus E_{\lambda_m}(T)$, then dimensions add by Theorem \ref{thm:direct-sum-iff-dimensions-add}.
\item[$(v)$]$\Rightarrow (ii)$ For each $j$, choose a basis $\beta_j$ of $E_{\lambda_j}(T)$, and concatenate these bases into a list $\beta$. The length of $\beta$ is $\dim(\mathcal{V})$. If a linear combination of vectors in $\beta$ is zero, grouping terms by eigenspace gives $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ with $\mathbf{u}_j\in E_{\lambda_j}(T)$. Proposition \ref{prop:eigenspaces-for-distinct-eigenvalues-form-direct-sum} gives $\mathbf{u}_j=\mathbf{0}$ for every $j$; since each $\beta_j$ is independent, all coefficients are zero. Thus $\beta$ is independent, Proposition \ref{prop:dimension-bounds-and-basis-criteria} makes $\beta$ a basis of $\mathcal{V}$ consisting of eigenvectors of $T$.
\end{enumerate}
\hlt{Structural Commentary}: Diagonalizability means the eigenspaces exhaust $\mathcal{V}$. Directness is automatic for distinct eigenspaces, so the real test is spanning, or equivalently the dimension count.\\
\hlt{Key Insight}: A diagonal basis is exactly an eigenbasis.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:diagonalizability-viewpoint} \hlt{Diagonalization as Exhaustion by Eigenspaces}\\
Distinct eigenspaces are already separated by Proposition \ref{prop:eigenspaces-for-distinct-eigenvalues-form-direct-sum}. Thus diagonalizability is the stronger statement that these separated pieces account for every direction in $\mathcal{V}$.
\end{perspective}

\begin{corollary} \label{cor:enough-eigenvalues-implies-diagonalizable} \hlt{Enough Eigenvalues Imply Diagonalizability}\\
If $\mathcal{V}$ is finite-dimensional and $T\in\LL(\mathcal{V})$ has $\dim(\mathcal{V})$ distinct eigenvalues, then $T$ is diagonalizable.
\begin{proof}
Let $n=\dim(\mathcal{V})$, and let $\lambda_1,\ldots,\lambda_n$ be distinct eigenvalues of $T$. For each $j$, choose an eigenvector $\mathbf{v}_j$ corresponding to $\lambda_j$. By Theorem \ref{thm:distinct-eigenvalues-independent-eigenvectors}, $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ is linearly independent. Since this list has length $n$, Proposition \ref{prop:dimension-bounds-and-basis-criteria} makes it a basis of $\mathcal{V}$. Thus $\mathcal{V}$ has an eigenbasis, so $T$ is diagonalizable by Theorem \ref{thm:conditions-equivalent-to-diagonalizability}.\\
\hlt{Structural Commentary}: One eigenvector from each distinct eigenvalue already gives a full-size independent list.\\
\hlt{Key Insight}: Once the independent eigenvectors reach $\dim(\mathcal{V})$, they must form a basis.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:polynomial-functional-calculus-diagonalizable-matrix} \hlt{Polynomial Functional Calculus for Diagonalizable Matrices}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ be diagonalizable and write $A=S\Lambda S^{-1}$, where $\Lambda=\operatorname{diag}(\lambda_1,\ldots,\lambda_n)$.\\
If $p\in\mathcal{P}(\mathbb{F})$, then $p(A)=S\operatorname{diag}(p(\lambda_1),\ldots,p(\lambda_n))S^{-1}$.
\begin{proof}
For every $j\geq0$, $A^j=S\Lambda^jS^{-1}$.\\
If $p(z)=\sum_j c_jz^j$, then $p(A)=\sum\limits_j c_jA^j =S\left(\sum\limits_j c_j\Lambda^j\right)S^{-1} =S p(\Lambda) S^{-1}$.\\
Since $\Lambda$ is diagonal, $p(\Lambda)=\operatorname{diag}(p(\lambda_1),\ldots,p(\lambda_n))$.\\
\hlt{Structural Commentary}: Diagonalization reduces $p(A)$ to scalar evaluation on eigenvalues.\\
\hlt{Key Insight}: Compute functions of $A$ by applying the function to the eigenvalues.
\end{proof}
\end{theorem}

\begin{definition} \label{def:matrix-commutant} \hlt{Commutant}\\
For $\emptyset\neq\mathcal{F}\subseteq\operatorname{Mat}(n,n,\mathbb{F})$, the \hlt{commutant} of $\mathcal{F}$ is $\mathcal{F}'\defeq\{X\in\operatorname{Mat}(n,n,\mathbb{F}): XA=AX\ \forall A\in\mathcal{F}\}$.\\
For a single matrix $A$, write $\{A\}'$.
\end{definition}

\begin{theorem} \label{thm:matrix-invariant-subspace-projection-block-criteria} \hlt{Invariant Subspaces, Blocks, and Projections}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$, let $\mathcal{U}$ be an $r$-dimensional subspace of $\mathbb{F}^n$ with $1\leq r\leq n-1$, and let $Q=[Q_1\ Q_2]$ be unitary, where the columns of $Q_1$ and $Q_2$ are orthonormal bases of $\mathcal{U}$ and $\mathcal{U}^{\perp}$, respectively. Let $P=Q_1Q_1^*$ be the orthogonal projection onto $\mathcal{U}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{U}$ is invariant under $A$;
\item $Q^*AQ=\begin{bmatrix}B & X\\ 0 & C\end{bmatrix}$ for some compatible matrices $B,X,C$;
\item $PAP=AP$.
\end{enumerate}
\begin{proof} Let $M=Q^*AQ=\begin{bmatrix}B&X\\Y&C\end{bmatrix}$, $E=Q^*PQ=I_r\oplus0_{n-r}$. Here $Y=Q_2^*AQ_1$ and $\ker(Q_2^*)=\operatorname{col}(Q_1)=\mathcal{U}$.
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ If $A\mathcal{U}\subseteq\mathcal{U}$, then the columns of $AQ_1$ lie in $\ker(Q_2^*)$, so $Y=0$.
\item[$(ii)$] $\Rightarrow (iii)$ If $Y=0$, then $EME=\begin{bmatrix}B&0\\0&0\end{bmatrix}$ and $ME=\begin{bmatrix}B&0\\0&0\end{bmatrix}$.\\
Thus $Q^*(PAP)Q=EME=ME=Q^*(AP)Q$, so $PAP=AP$.
\item[$(iii)$] $\Rightarrow (i)$ If $PAP=AP$, then $EME=ME$. Since $EME=\begin{bmatrix}B&0\\0&0\end{bmatrix}$ and $ME=\begin{bmatrix}B&0\\Y&0\end{bmatrix}$, we get $Y=0$.\\
Hence $AQ_1$ has columns in $\ker(Q_2^*)=\mathcal{U}$, so $A\mathcal{U}\subseteq\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: In adapted coordinates, the lower-left block is the component of $A\mathcal{U}$ in $\mathcal{U}^{\perp}$.\\
\hlt{Key Insight}: $PAP=AP$ exactly kills this component.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:reducing-subspace-projection-commutation} \hlt{Reducing Subspaces and Commuting Projections}\\
Maintain the notation of Theorem \ref{thm:matrix-invariant-subspace-projection-block-criteria}. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{U}$ is invariant under both $A$ and $A^*$;
\item $Q^*AQ=B\oplus C$ for some $B\in\operatorname{Mat}(r,r,\mathbb{F})$ and $C\in\operatorname{Mat}(n-r,n-r,\mathbb{F})$;
\item $PA=AP$.
\end{enumerate}
\begin{proof} Let $M=Q^*AQ=\begin{bmatrix}B&X\\Y&C\end{bmatrix}$ and $E=Q^*PQ=I_r\oplus0_{n-r}$.\\
Then $Q^*A^*Q=M^*$, $Q^*(PA)Q=EM$, and $Q^*(AP)Q=ME$.
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ By Theorem \ref{thm:matrix-invariant-subspace-projection-block-criteria}, $A$-invariance gives $Y=0$; apply to $A^*$ gives $X=0$. Thus $M=B\oplus C$.
\item[$(ii)$] $\Rightarrow (iii)$ If $M=B\oplus C$, then $EM=ME$, hence $PA=AP$.
\item[$(iii)$] $\Rightarrow (i)$ If $PA=AP$, then $EM=ME$. Since $EM=\begin{bmatrix}B&X\\0&0\end{bmatrix}$ and $ME=\begin{bmatrix}B&0\\Y&0\end{bmatrix}$, we get $X=Y=0$.\\
Hence $\mathcal{U}$ is invariant under both $A$ and $A^*$.
\end{enumerate}
\hlt{Structural Commentary}: Reducing subspaces are exactly block diagonal decompositions in adapted coordinates.\\
\hlt{Key Insight}: Commutation with $P$ kills both off-diagonal blocks.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:upper-triangular-criteria} \hlt{Criteria for an Upper-Triangular Operator Matrix}\\
Let $T\in\LL(\mathcal{V})$ and let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis of $\mathcal{V}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $[T]_{\beta\leftarrow\beta}$ is upper triangular;
\item $T\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$ for each $j=1,\ldots,n$;
\item $\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$ is invariant under $T$ for each $j=1,\ldots,n$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$
The $j$th column of $[T]_{\beta\leftarrow\beta}$ is the coordinate column of $T\mathbf{v}_j$ in the basis $\beta$. Since the matrix is upper triangular, this column has zero entries below position $j$, so $T\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$.
\item[$(ii)$]$\Rightarrow (i)$
If $T\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$ for every $j$, then the $j$th coordinate column of $T\mathbf{v}_j$ has zero entries below position $j$. Hence $[T]_{\beta\leftarrow\beta}$ is upper triangular.
\item[$(ii)$]$\Rightarrow (iii)$
If $\mathbf{u}\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$, then linearity gives $T\mathbf{u}$ as a linear combination of $T\mathbf{v}_1,\ldots,T\mathbf{v}_j$, all of which lie in $\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$. Hence this span is invariant.
\item[$(iii)$]$\Rightarrow (ii)$
Apply invariance of $\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$ to $\mathbf{v}_j$.
\end{enumerate}
\hlt{Structural Commentary}: Upper triangularity records an invariant flag.\\
\hlt{Key Insight}: Columns below the diagonal vanish exactly when $T\mathbf{v}_j$ uses only earlier basis vectors and itself.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:upper-triangular-orthonormal-basis} \hlt{Upper-Triangular Form in an Orthonormal Basis}\\
Let $\mathcal{V}$ be a finite-dimensional inner-product space and let $T\in\LL(\mathcal{V})$. If $T$ has an upper-triangular matrix with respect to some basis of $\mathcal{V}$, then $T$ has an upper-triangular matrix with respect to some orthonormal basis of $\mathcal{V}$.
\begin{proof}
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis for which $[T]_{\beta\leftarrow\beta}$ is upper triangular. By Theorem \ref{thm:upper-triangular-criteria}, each subspace
\begin{align}
\mathcal{U}_j=\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j),\qquad j=1,\ldots,n,
\nonumber
\end{align}
is invariant under $T$. Apply Gram--Schmidt to $\beta$ and obtain an orthonormal basis $\gamma=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ with
\begin{align}
\operatorname{span}(\mathbf{e}_1,\ldots,\mathbf{e}_j)=\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)=\mathcal{U}_j.
\nonumber
\end{align}
Thus each $\operatorname{span}(\mathbf{e}_1,\ldots,\mathbf{e}_j)$ is invariant under $T$, and Theorem \ref{thm:upper-triangular-criteria} shows that $[T]_{\gamma\leftarrow\gamma}$ is upper triangular.\\
\hlt{Structural Commentary}: Upper triangularity is preservation of a nested flag of subspaces. Gram--Schmidt preserves that flag while making the basis orthonormal.\\
\hlt{Key Insight}: Triangular form depends on the flag, not on the particular basis used inside the flag.
\end{proof}
\end{corollary}

\begin{definition} \label{def:restriction-and-quotient-operators} \hlt{Restriction and Quotient Operators}\\
Let $T\in\LL(\mathcal{V})$ and let $\mathcal{U}$ be invariant under $T$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{restriction operator} $T|_{\mathcal{U}}\in\LL(\mathcal{U})$ is defined by $T|_{\mathcal{U}}(\mathbf{u})=T\mathbf{u}$ for $\mathbf{u}\in\mathcal{U}$.
\item The \hlt{quotient operator} $T/\mathcal{U}\in\LL(\mathcal{V}/\mathcal{U})$ is defined by $(T/\mathcal{U})(\mathbf{v}+\mathcal{U})=T\mathbf{v}+\mathcal{U}$, $\mathbf{v}\in\mathcal{V}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:restriction-and-quotient-operators-well-defined} \hlt{Restriction and Quotient Operators Are Well-Defined}\\
Let $T\in\LL(\mathcal{V})$ and let $\mathcal{U}$ be invariant under $T$. Then $T|_{\mathcal{U}}\in\LL(\mathcal{U})$ and $T/\mathcal{U}\in\LL(\mathcal{V}/\mathcal{U})$.
\begin{proof} As $\mathcal{U}$ invariant under $T$, $T\mathbf{u}\in\mathcal{U}$ whenever $\mathbf{u}\in\mathcal{U}$, so $T|_{\mathcal{U}}$ maps $\mathcal{U}$ into itself. Linearity inherited from $T$.\\
For the quotient operator, suppose $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$. Then $\mathbf{v}-\mathbf{w}\in\mathcal{U}$. Invariance gives $T(\mathbf{v}-\mathbf{w})\in\mathcal{U}$, so $T\mathbf{v}-T\mathbf{w}\in\mathcal{U}$ and hence $T\mathbf{v}+\mathcal{U}=T\mathbf{w}+\mathcal{U}$. Thus $T/\mathcal{U}$ is well-defined. Linearity follows from the quotient operations and linearity of $T$.\\
\hlt{Structural Commentary}: Restriction stays inside $\mathcal{U}$; quotient collapses $\mathcal{U}$.\\
\hlt{Key Insight}: Invariance makes both operators well-defined.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:restriction-quotient-perspective} \hlt{Two Smaller Views of an Operator}\\
When $\mathcal{U}$ is invariant under $T$, the operators $T|_{\mathcal{U}}$ and $T/\mathcal{U}$ split attention between what happens inside $\mathcal{U}$ and what remains after $\mathcal{U}$ is collapsed. These two views often simplify an operator, even though they need not determine the whole operator by themselves.
\end{perspective}
