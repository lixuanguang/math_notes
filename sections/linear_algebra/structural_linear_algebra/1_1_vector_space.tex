\subsubsection{Vector Spaces and Linear Structure}

The guiding idea is that a vector space is not a list of coordinates first. It is a space of objects that can be added and scaled, and coordinates only appear after a basis is chosen.

\begin{definition} \hlt{Vector Space}\\
Let $\mathbb{F}$ be a field. A vector space over $\mathbb{F}$ is a set $\mathcal{V}$ together with operations of vector addition and scalar multiplication such that, for all $\mathbf{u}, \mathbf{v}, \mathbf{w} \in \mathcal{V}$ and all $a,b \in \mathbb{F}$, the following axioms hold:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item vector addition is commutative: $\mathbf{u} + \mathbf{v} = \mathbf{v} + \mathbf{u}$;
\item vector addition is associative: $(\mathbf{u} + \mathbf{v}) + \mathbf{w} = \mathbf{u} + (\mathbf{v} + \mathbf{w})$;
\item there exists an additive identity $\mathbf{0} \in \mathcal{V}$ such that $\mathbf{0} + \mathbf{v} = \mathbf{v}$;
\item for each $\mathbf{v} \in \mathcal{V}$ there exists an additive inverse $-\mathbf{v} \in \mathcal{V}$ such that $\mathbf{v} + (-\mathbf{v}) = \mathbf{0}$;
\item scalar multiplication has identity: $1\mathbf{v} = \mathbf{v}$;
\item scalar multiplication is compatible with multiplication in $\mathbb{F}$: $(ab)\mathbf{v} = a(b\mathbf{v})$;
\item scalar multiplication distributes over vector addition: $a(\mathbf{u} + \mathbf{v}) = a\mathbf{u} + a\mathbf{v}$;
\item scalar multiplication distributes over addition in $\mathbb{F}$: $(a+b)\mathbf{v} = a\mathbf{v} + b\mathbf{v}$.
\end{enumerate}
\end{definition}

\begin{definition}
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. If $\mathcal{V} = \{ \mathbf{0} \}$, then $\mathcal{V}$ is a \hlt{zero vector space}; if $\mathcal{V} \neq \{ \mathbf{0} \}$, then $\mathcal{V}$ is a \hlt{nonzero vector space}.
\end{definition}

\begin{theorem} \label{thm:basicvectorspaceconsequences} \hlt{Basic Consequences of Vector Space Axioms}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then the additive identity and additive inverses are unique. Moreover, for all $a \in \mathbb{F}$ and $\mathbf{v} \in \mathcal{V}$, $0\mathbf{v} = \mathbf{0}$, $a\mathbf{0} = \mathbf{0}$, $(-1)\mathbf{v} = -\mathbf{v}$ and $a\mathbf{v} = \mathbf{0}$ if and only if $a = 0$ or $\mathbf{v} = \mathbf{0}$.
\begin{proof}
If $\mathbf{0}$ and $\mathbf{0}'$ are additive identities, then $\mathbf{0} = \mathbf{0} + \mathbf{0}' = \mathbf{0}'$, so the additive identity is unique.\\
If $\mathbf{w}$ and $\mathbf{w}'$ are additive inverses of $\mathbf{v}$, then $\mathbf{w} = \mathbf{w} + \mathbf{0} = \mathbf{w} + (\mathbf{v} + \mathbf{w}') = (\mathbf{w} + \mathbf{v}) + \mathbf{w}' = \mathbf{0} + \mathbf{w}' = \mathbf{w}'$, so additive inverses are unique. Since $0\mathbf{v} = (0+0)\mathbf{v} = 0\mathbf{v} + 0\mathbf{v}$, cancellation gives $0\mathbf{v} = \mathbf{0}$.\\
Similarly, $a\mathbf{0} = a(\mathbf{0}+\mathbf{0}) = a\mathbf{0} + a\mathbf{0}$, so $a\mathbf{0} = \mathbf{0}$. Also $\mathbf{v} + (-1)\mathbf{v} = 1\mathbf{v} + (-1)\mathbf{v} = (1-1)\mathbf{v} = 0\mathbf{v} = \mathbf{0}$, hence $(-1)\mathbf{v}$ is the additive inverse of $\mathbf{v}$, and therefore $(-1)\mathbf{v} = -\mathbf{v}$.\\
Finally, if $a = 0$ or $\mathbf{v} = \mathbf{0}$, then the preceding identities give $a\mathbf{v} = \mathbf{0}$. Conversely, suppose $a\mathbf{v} = \mathbf{0}$ and $a \neq 0$. Then $a^{-1}$ exists in $\mathbb{F}$, and $\mathbf{v} = 1\mathbf{v} = (a^{-1}a)\mathbf{v} = a^{-1}(a\mathbf{v}) = a^{-1}\mathbf{0} = \mathbf{0}$.\\
\hlt{Structural Commentary}: These identities are the basic algebraic hygiene of vector spaces. They let later arguments use cancellation, zero vectors, and inverses without reopening the axioms each time.
\end{proof}
\end{theorem}
\hlt{Key Insight}: The vector space axioms force the zero and inverse behaviour to be rigid, not optional structure.

\begin{corollary} \label{cor:negativeoneactsasinverse}
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then $(-1)\mathbf{v} = -\mathbf{v}$ for every $\mathbf{v} \in \mathcal{V}$.
\begin{proof}
Let $\mathbf{v} \in \mathcal{V}$. We show that $(-1)\mathbf{v} + \mathbf{v} = \mathbf{0}$. By the vector space axioms and Theorem \ref{thm:basicvectorspaceconsequences},
\begin{align}
(-1)\mathbf{v} + \mathbf{v} = (-1)\mathbf{v} + 1\mathbf{v} = (-1+1)\mathbf{v} = 0\mathbf{v} = \mathbf{0} \nonumber
\end{align}
Thus $(-1)\mathbf{v}$ is the additive inverse of $\mathbf{v}$, so $(-1)\mathbf{v} = -\mathbf{v}$.\\
\hlt{Structural Commentary}: Scalar multiplication by $-1$ is exactly the same operation as taking the additive inverse.
\end{proof}
\end{corollary}
\hlt{Key Insight}: Negation is not an extra operation; it is already encoded by the field action.

\begin{definition} A \hlt{subspace} of a vector space $\mathcal{V}$ over $\mathbb{F}$ is a subset $\mathcal{U} \subseteq \mathcal{V}$ that is itself a vector space over $\mathbb{F}$ with the same vector addition and scalar multiplication as in $\mathcal{V}$.
\end{definition}

\begin{theorem} \label{thm:subspacetest} \hlt{Subspace Test}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$ and let $\mathcal{U}$ be a nonempty subset of $\mathcal{V}$. Then $\mathcal{U}$ is a subspace of $\mathcal{V}$ if and only if $c\mathbf{u} + \mathbf{v} \in \mathcal{U}$ for all $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and all $c \in \mathbb{F}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[]
\item[$(\Rightarrow)$] If $\mathcal{U}$ is a subspace, then $c\mathbf{u}+\mathbf{v} \in \mathcal{U}$ for all $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and all $c \in \mathbb{F}$ because $\mathcal{U}$ is closed under scalar multiplication and vector addition.
\item[$(\Leftarrow)$] Suppose that $c\mathbf{u}+\mathbf{v} \in \mathcal{U}$ whenever $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and $c \in \mathbb{F}$. We verify that $\mathcal{U}$ is a vector space with the operations inherited from $\mathcal{V}$. Since $\mathcal{U}$ is nonempty, let $\mathbf{u} \in \mathcal{U}$. By Corollary \ref{cor:negativeoneactsasinverse}, $\mathbf{0} = (-1)\mathbf{u}+\mathbf{u} \in \mathcal{U}$. If $\mathbf{u} \in \mathcal{U}$, then $-\mathbf{u} = (-1)\mathbf{u}+\mathbf{0} \in \mathcal{U}$. If $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and $c \in \mathbb{F}$, then $c\mathbf{u} = c\mathbf{u}+\mathbf{0} \in \mathcal{U}$ and $\mathbf{u}+\mathbf{v} = 1\mathbf{u}+\mathbf{v} \in \mathcal{U}$. Thus $\mathcal{U}$ contains $\mathbf{0}$ and is closed under additive inverses, scalar multiplication, and vector addition; the remaining vector space axioms are inherited from $\mathcal{V}$. Hence $\mathcal{U}$ is a subspace.
\end{enumerate}\hfill \\
\hlt{Structural Commentary}: The theorem compresses the subspace axioms into one closure condition. This is the practical test used whenever a subset is proposed as a linear space.
\end{proof}
\end{theorem}
\hlt{Key Insight}: To prove a subset is a subspace, prove it survives one affine-looking linear combination.

\begin{proposition} \label{prop:intersectionsubspaces} \hlt{Intersection of Subspaces}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$. Then $\mathcal{U}\cap\mathcal{W} = \left\{ \mathbf{v} : \mathbf{v}\in\mathcal{U} \text{ and } \mathbf{v}\in\mathcal{W} \right\}$ is a subspace of $\mathcal{V}$.
\begin{proof}
Since $\mathbf{0}\in\mathcal{U}$ and $\mathbf{0}\in\mathcal{W}$, we have $\mathbf{0}\in\mathcal{U}\cap\mathcal{W}$, so the intersection is nonempty. If $\mathbf{u},\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$ and $c\in\mathbb{F}$, then $c\mathbf{u}+\mathbf{v}$ lies in both $\mathcal{U}$ and $\mathcal{W}$ because both are subspaces. Hence $c\mathbf{u}+\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$, and Theorem \ref{thm:subspacetest} shows that $\mathcal{U}\cap\mathcal{W}$ is a subspace of $\mathcal{V}$.\\
\hlt{Structural Commentary}: Intersections preserve every closure condition shared by the subspaces. The subspace test packages this into one check.
\end{proof}
\end{proposition}
\hlt{Key Insight}: Common vectors of two subspaces still form a subspace because the shared part remains closed under linear combinations.

\begin{definition} \hlt{Lists, Linear Combinations, and Span}\\
A \hlt{list} of vectors in $\mathcal{V}$ is a nonempty finite ordered sequence $\beta = (\mathbf{v}_1,\ldots,\mathbf{v}_m)$ in $\mathcal{V}$, denoted by a Greek letter.\\
If $\mathbf{v}_1, \ldots, \mathbf{v}_m \in \mathcal{V}$ and $a_1, \ldots, a_m \in \mathbb{F}$, then $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m$ is a \hlt{linear combination} of $\mathbf{v}_1, \ldots, \mathbf{v}_m$.\\
The \hlt{span} of a nonempty subset $E \subseteq \mathcal{V}$ is $\operatorname{span}(E) \defeq \left\{ a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m : m \in \N,\ a_j \in \mathbb{F},\ \mathbf{v}_j \in E \right\}$.\\
By convention, $\operatorname{span}(\emptyset) = \{ \mathbf{0} \}$. If $\beta = (\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is a list, then $\operatorname{span}(\beta)
= \operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_m)
\defeq \operatorname{span}(\{\mathbf{v}_1,\ldots,\mathbf{v}_m\})$.\\
A subset $E \subseteq \mathcal{V}$ \hlt{spans} $\mathcal{V}$, or is a \hlt{spanning set} for $\mathcal{V}$, if $\operatorname{span}(E)=\mathcal{V}$.\\
A list $\beta$ in $\mathcal{V}$ \hlt{spans} $\mathcal{V}$, or is a \hlt{spanning list} for $\mathcal{V}$, if $\operatorname{span}(\beta)=\mathcal{V}$.
\end{definition}

\begin{theorem} \label{thm:spanclosure} \hlt{Basic Properties of Span}\\
Let $E \subseteq \mathcal{V}$. Then:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\operatorname{span}(E)$ is a subspace of $\mathcal{V}$.
\item[$(ii)$] $E \subseteq \operatorname{span}(E)$.
\item[$(iii)$] $E=\operatorname{span}(E)$ if and only if $E$ is a subspace of $\mathcal{V}$.
\item[$(iv)$] $\operatorname{span}(\operatorname{span}(E))=\operatorname{span}(E)$.
\end{enumerate}
Consequently, $\operatorname{span}(E)$ is the smallest subspace of $\mathcal{V}$ containing $E$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[]
\item[$(i)$] If $E=\emptyset$, then $\operatorname{span}(E)=\{\mathbf{0}\}$, so $\operatorname{span}(E)$ is a subspace. Suppose $E\neq\emptyset$. If $\mathbf{x},\mathbf{y}\in\operatorname{span}(E)$ and $c\in\mathbb{F}$, then $\mathbf{x}$ and $\mathbf{y}$ are finite linear combinations of vectors in $E$, hence $c\mathbf{x}+\mathbf{y}$ is again a finite linear combination of vectors in $E$. Thus $c\mathbf{x}+\mathbf{y}\in\operatorname{span}(E)$, and Theorem \ref{thm:subspacetest} shows that $\operatorname{span}(E)$ is a subspace.
\item[$(ii)$] Each $\mathbf{v}\in E$ satisfies $\mathbf{v}=1\mathbf{v}\in\operatorname{span}(E)$, so $E\subseteq\operatorname{span}(E)$.
\item[$(iii)$] If $E=\operatorname{span}(E)$, then $(i)$ shows that $E$ is a subspace. Conversely, if $E$ is a subspace, then $E$ is closed under finite linear combinations, so $\operatorname{span}(E)\subseteq E$; together with $(ii)$ this gives $E=\operatorname{span}(E)$.
\item[$(iv)$] By $(i)$, $\operatorname{span}(E)$ is a subspace. Applying $(iii)$ to the subspace $\operatorname{span}(E)$ gives $\operatorname{span}(\operatorname{span}(E))=\operatorname{span}(E)$.
\end{enumerate} \hfill \\
If $\mathcal{U} \subseteq \mathcal{V}$ containing $E$, closure under vector addition and scalar multiplication forces every finite linear combination of vectors in $E$ to lie in $\mathcal{U}$. Hence $\operatorname{span}(E)\subseteq\mathcal{U}$, so $\operatorname{span}(E)$ is the smallest subspace containing $E$.\\
\hlt{Structural Commentary}: Taking span is a closure operation. It contains the original set, produces a subspace, and becomes stable after one application.
\end{proof}
\end{theorem}
\hlt{Key Insight}: Span is the smallest linear environment forced by a set of vectors.

\begin{proposition} \label{prop:spanmonotonicity} \hlt{Monotonicity of Span}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subsets of $\mathcal{V}$. If $\mathcal{U}\subseteq\mathcal{W}$, then $\operatorname{span}(\mathcal{U})\subseteq\operatorname{span}(\mathcal{W})$.
\begin{proof}
If $\mathcal{U}=\emptyset$, then $\operatorname{span}(\mathcal{U})=\{\mathbf{0}\}\subseteq\operatorname{span}(\mathcal{W})$.\\
Suppose $\mathcal{U}\neq\emptyset$. Every vector in $\operatorname{span}(\mathcal{U})$ is a finite linear combination of vectors in $\mathcal{U}$; since $\mathcal{U}\subseteq\mathcal{W}$, the same expression is a finite linear combination of vectors in $\mathcal{W}$. Hence $\operatorname{span}(\mathcal{U})\subseteq\operatorname{span}(\mathcal{W})$.\\
\hlt{Structural Commentary}: Enlarging the generating set can only enlarge the collection of linear combinations.
\end{proof}
\end{proposition}
\hlt{Key Insight}: Span is monotone with respect to inclusion of generating sets.

\begin{definition} A list $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ in $\mathcal{V}$ is \hlt{linearly independent} if $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = \mathbf{0}$ implies $a_1 = \cdots = a_m = 0$. Otherwise the list is linearly dependent.
\end{definition}

\begin{lemma} \label{lem:lineardependencelemma} \hlt{Linear Dependence Lemma}\\
If $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly dependent and $\mathbf{v}_1 \neq \mathbf{0}$, then there exists $j \in \{2, \ldots, m\}$ such that
\begin{align}
\mathbf{v}_j \in \operatorname{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{j-1}), \nonumber
\end{align}
and removing $\mathbf{v}_j$ does not change the span of the list.
\begin{proof}
Because the list is dependent, there exist scalars $a_1, \ldots, a_m$, not all zero, such that
\begin{align}
a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = \mathbf{0}. \nonumber
\end{align}
Choose $j$ to be the largest index such that $a_j \neq 0$. Then
\begin{align}
\mathbf{v}_j = -\frac{a_1}{a_j}\mathbf{v}_1 - \cdots - \frac{a_{j-1}}{a_j}\mathbf{v}_{j-1}, \nonumber
\end{align}
so $\mathbf{v}_j$ lies in the span of the preceding vectors. Any linear combination involving $\mathbf{v}_j$ can therefore be rewritten using only the remaining vectors, so removing $\mathbf{v}_j$ does not alter the span.
\hlt{Structural Commentary}: A dependence relation identifies a redundant vector. Choosing the largest nonzero coefficient makes the redundancy compatible with the order of the list.
\end{proof}
\end{lemma}
\hlt{Key Insight}: Linear dependence means at least one vector can be discarded without losing span.

\begin{theorem} \label{thm:finitedimlenofvec}
In a finite-dimensional vector space, the length of every linearly independent list is at most the length of every spanning list.
\begin{proof}
Let $(\mathbf{u}_1, \ldots, \mathbf{u}_m)$ be linearly independent and let $(\mathbf{w}_1, \ldots, \mathbf{w}_n)$ span $\mathcal{V}$. Starting from the spanning list $(\mathbf{w}_1, \ldots, \mathbf{w}_n)$, insert $\mathbf{u}_1, \ldots, \mathbf{u}_m$ one at a time and, whenever dependence appears, remove a vector from the old spanning list using Lemma \ref{lem:lineardependencelemma}. After $m$ steps we still have a spanning list of length $n$ containing $\mathbf{u}_1, \ldots, \mathbf{u}_m$, hence $m \leq n$.
\hlt{Structural Commentary}: This comparison theorem is the counting engine behind dimension. It says independence and spanning place opposite pressure on list length.
\end{proof}
\end{theorem}
\hlt{Key Insight}: Independent lists cannot be longer than spanning lists in finite dimensions.

\begin{definition} \hlt{Basis}\\
A basis of $\mathcal{V}$ is a list of vectors that is linearly independent and spans $\mathcal{V}$.
\end{definition}

\begin{proposition}
A list $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$ if and only if every $\mathbf{v} \in \mathcal{V}$ can be written uniquely as
\begin{align}
\mathbf{v} = a_1\mathbf{v}_1 + \cdots + a_n\mathbf{v}_n. \nonumber
\end{align}
\hlt{Structural Commentary}: A basis is exactly a coordinate system: spanning gives existence and independence gives uniqueness.
\end{proposition}
\hlt{Key Insight}: Basis vectors are useful because they make every vector have one and only one coordinate description.

\begin{theorem} \hlt{Reducing Spanning Lists}\\
Every spanning list in a finite-dimensional vector space can be reduced to a basis.
\begin{proof}
Repeatedly remove any vector that lies in the span of the preceding vectors. Lemma \ref{lem:lineardependencelemma} shows that the span does not change, and when no such vector remains the list is linearly independent. The resulting list is therefore a basis.
\hlt{Structural Commentary}: A spanning list may contain redundancy, but finite-dimensionality lets us remove it until only essential generators remain.
\end{proof}
\end{theorem}
\hlt{Key Insight}: Every finite spanning list hides a basis inside it.

\begin{theorem} \hlt{Extending Independent Lists}\\
Every linearly independent list in a finite-dimensional vector space can be extended to a basis.
\begin{proof}
Take any finite spanning list and append its vectors one at a time to the independent list, keeping only those that are not already in the span. This preserves linear independence and eventually yields a spanning list, hence a basis.
\hlt{Structural Commentary}: Independent lists are partial coordinate systems. Extension fills in the missing directions without destroying independence.
\end{proof}
\end{theorem}
\hlt{Key Insight}: Every finite-dimensional independent list can be completed to a basis.

\begin{corollary}
Every finite-dimensional vector space has a basis, and every subspace of a finite-dimensional vector space is finite-dimensional.
\hlt{Structural Commentary}: Finite-dimensional spaces and their subspaces admit finite coordinate descriptions.
\end{corollary}
\hlt{Key Insight}: Finite-dimensionality passes downward to subspaces.

\begin{theorem} \hlt{Basis Length Is Well-Defined}\\
Any two bases of a finite-dimensional vector space have the same length.
\begin{proof}
Each basis is linearly independent and spanning, so Theorem \ref{thm:finitedimlenofvec} applied in both directions gives equality of lengths.
\hlt{Structural Commentary}: This result makes dimension well-defined. Without it, the length of a basis would depend on the chosen coordinates.
\end{proof}
\end{theorem}
\hlt{Key Insight}: Dimension is an intrinsic number, not an artifact of a particular basis.

\begin{definition} \hlt{Dimension}\\
If $\mathcal{V}$ is finite-dimensional, its dimension is the common length of any basis of $\mathcal{V}$, denoted $\dim(\mathcal{V})$.
\end{definition}

\begin{proposition}
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U} \subseteq \mathcal{V}$ is a subspace, then
\begin{align}
\dim(\mathcal{U}) \leq \dim(\mathcal{V}). \nonumber
\end{align}
Moreover, a spanning list of length $\dim(\mathcal{V})$ is automatically a basis, and a linearly independent list of length $\dim(\mathcal{V})$ is automatically a basis.
\hlt{Structural Commentary}: Dimension bounds the size of subspaces and identifies when a list is exactly the right size to be a basis.
\end{proposition}
\hlt{Key Insight}: In dimension $n$, either $n$ independent vectors or $n$ spanning vectors already give a basis.

\begin{definition} \hlt{Sums of Subspaces}\\
If $\mathcal{U}_1, \ldots, \mathcal{U}_m$ are subspaces of $\mathcal{V}$, define
\begin{align}
\mathcal{U}_1 + \cdots + \mathcal{U}_m
\defeq
\left\{ \mathbf{u}_1 + \cdots + \mathbf{u}_m : \mathbf{u}_j \in \mathcal{U}_j \right\}. \nonumber
\end{align}
In particular, for subspaces $\mathcal{U},\mathcal{W}\subseteq\mathcal{V}$,
\begin{align}
\mathcal{U}+\mathcal{W}
= \operatorname{span}(\mathcal{U}\cup\mathcal{W})
= \left\{ \mathbf{u}+\mathbf{w} : \mathbf{u}\in\mathcal{U},\ \mathbf{w}\in\mathcal{W} \right\}. \nonumber
\end{align}
If every vector in this sum has a unique such decomposition, then the sum is direct and we write
\begin{align}
\mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m. \nonumber
\end{align}
For two subspaces, $\mathcal{U}+\mathcal{W}$ is direct if and only if $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$; in this case it is denoted $\mathcal{U}\oplus\mathcal{W}$.
\end{definition}

\begin{proposition} \label{prop:uniquenessdirectsumtwosubspaces} \hlt{Uniqueness in a Direct Sum}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$ with $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$. Then each vector in $\mathcal{U}\oplus\mathcal{W}$ has a unique representation as $\mathbf{u}+\mathbf{w}$, where $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{W}$.
\begin{proof}
Existence follows from the definition of $\mathcal{U}+\mathcal{W}$. For uniqueness, suppose
\begin{align}
\mathbf{u}_1+\mathbf{w}_1=\mathbf{u}_2+\mathbf{w}_2,
\qquad
\mathbf{u}_1,\mathbf{u}_2\in\mathcal{U},\quad \mathbf{w}_1,\mathbf{w}_2\in\mathcal{W}. \nonumber
\end{align}
Then $\mathbf{u}_1-\mathbf{u}_2=\mathbf{w}_2-\mathbf{w}_1$. The left side lies in $\mathcal{U}$ and the right side lies in $\mathcal{W}$, so this common vector lies in $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$. Hence $\mathbf{u}_1=\mathbf{u}_2$ and $\mathbf{w}_1=\mathbf{w}_2$.\\
\hlt{Structural Commentary}: Trivial intersection prevents cancellation between the two subspaces. Thus a sum records its $\mathcal{U}$-part and $\mathcal{W}$-part separately.
\end{proof}
\end{proposition}
\hlt{Key Insight}: Directness is uniqueness of decomposition: no nonzero vector can be shifted from one summand to the other.

\begin{proposition} \label{prop:sumsubspacessmallest} \hlt{Sum of Subspaces Is the Smallest Containing Subspace}\\
Suppose $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is the smallest subspace of $\mathcal{V}$ containing $\mathcal{U}_1,\ldots,\mathcal{U}_m$.
\begin{proof}
First, $\mathbf{0} \in \mathcal{U}_1+\cdots+\mathcal{U}_m$. If $\mathbf{x}=\mathbf{u}_1+\cdots+\mathbf{u}_m$ and $\mathbf{y}=\mathbf{v}_1+\cdots+\mathbf{v}_m$ lie in $\mathcal{U}_1+\cdots+\mathcal{U}_m$, then for any $c \in \mathbb{F}$,
\begin{align}
c\mathbf{x}+\mathbf{y}
= (c\mathbf{u}_1+\mathbf{v}_1)+\cdots+(c\mathbf{u}_m+\mathbf{v}_m) \in \mathcal{U}_1+\cdots+\mathcal{U}_m, \nonumber
\end{align}
because each $\mathcal{U}_j$ is a subspace. By Theorem \ref{thm:subspacetest}, $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a subspace of $\mathcal{V}$.\\
Clearly each $\mathcal{U}_j$ is contained in $\mathcal{U}_1+\cdots+\mathcal{U}_m$ by setting all other summands equal to $\mathbf{0}$. Conversely, if $\mathcal{W}$ is any subspace of $\mathcal{V}$ containing $\mathcal{U}_1,\ldots,\mathcal{U}_m$, then every sum $\mathbf{u}_1+\cdots+\mathbf{u}_m$ with $\mathbf{u}_j \in \mathcal{U}_j$ lies in $\mathcal{W}$. Hence $\mathcal{U}_1+\cdots+\mathcal{U}_m \subseteq \mathcal{W}$, so the sum is the smallest such subspace.
\hlt{Structural Commentary}: Sums of subspaces behave like unions after enforcing linear closure. The smallest subspace containing several pieces is obtained by allowing finite sums of vectors from those pieces.
\end{proof}
\end{proposition}
\hlt{Key Insight}: The sum of subspaces is the linear closure of those subspaces taken together.

\begin{proposition}
For subspaces $\mathcal{U}_1, \ldots, \mathcal{U}_m \subseteq \mathcal{V}$, the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$;
\item every $\mathbf{v} \in \mathcal{V}$ has a unique representation $\mathbf{v} = \mathbf{u}_1 + \cdots + \mathbf{u}_m$ with $\mathbf{u}_j \in \mathcal{U}_j$;
\item if $\mathbf{u}_1 + \cdots + \mathbf{u}_m = \mathbf{0}$ with $\mathbf{u}_j \in \mathcal{U}_j$, then $\mathbf{u}_1 = \cdots = \mathbf{u}_m = \mathbf{0}$.
\end{enumerate}
\hlt{Structural Commentary}: Direct sums are sums without overlap. The zero-representation condition is often the easiest way to test uniqueness.
\end{proposition}
\hlt{Key Insight}: A sum is direct precisely when there is no nontrivial way for the pieces to cancel each other.

\begin{remark} \hlt{Products and Direct Sums}\\
For finitely many vector spaces $\mathcal{V}_1, \ldots, \mathcal{V}_m$, the cartesian product
\begin{align}
\mathcal{V}_1 \times \cdots \times \mathcal{V}_m \nonumber
\end{align}
with componentwise operations is again a vector space. In the finite setting this is also the external direct sum, so products and direct sums coincide as vector spaces. This viewpoint is often the cleanest way to understand block decompositions later on.
\end{remark}

\begin{theorem} \hlt{Dimension Formula for a Sum}
If $\mathcal{U}, \mathcal{W}$ are finite-dimensional subspaces of $\mathcal{V}$, then
\begin{align}
\dim(\mathcal{U} + \mathcal{W})
=
\dim(\mathcal{U}) + \dim(\mathcal{W}) - \dim(\mathcal{U} \cap \mathcal{W}). \nonumber
\end{align}
\begin{proof}
Choose a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r)$ of $\mathcal{U} \cap \mathcal{W}$, extend it to a basis
\begin{align}
(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p) \nonumber
\end{align}
of $\mathcal{U}$, and to a basis
\begin{align}
(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{w}_1, \ldots, \mathbf{w}_q) \nonumber
\end{align}
of $\mathcal{W}$. Then
\begin{align}
(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p, \mathbf{w}_1, \ldots, \mathbf{w}_q) \nonumber
\end{align}
is a basis of $\mathcal{U} + \mathcal{W}$, giving
\begin{align}
\dim(\mathcal{U} + \mathcal{W}) = r + p + q
=
(r+p) + (r+q) - r. \nonumber
\end{align}
\hlt{Structural Commentary}: The formula corrects double-counting: vectors in $\mathcal{U} \cap \mathcal{W}$ are counted once in each subspace.
\end{proof}
\end{theorem}
\hlt{Key Insight}: The dimension of a sum is the sum of dimensions minus the overlap.

\begin{corollary}
If $\mathcal{U} \cap \mathcal{W} = \{ \mathbf{0} \}$, then
\begin{align}
\dim(\mathcal{U} \oplus \mathcal{W}) = \dim(\mathcal{U}) + \dim(\mathcal{W}). \nonumber
\end{align}
\hlt{Structural Commentary}: When the overlap is trivial, there is no double-counting in the dimension formula.
\end{corollary}
\hlt{Key Insight}: Dimensions add exactly for direct sums.

\begin{proposition} \hlt{Complements Exist in Finite Dimensions}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U} \subseteq \mathcal{V}$ is a subspace, then there exists a subspace $\mathcal{W}$ such that
\begin{align}
\mathcal{V} = \mathcal{U} \oplus \mathcal{W}. \nonumber
\end{align}
\begin{proof}
Choose a basis of $\mathcal{U}$ and extend it to a basis of $\mathcal{V}$. Let $\mathcal{W}$ be the span of the added basis vectors.
\hlt{Structural Commentary}: Complements split a finite-dimensional space into a chosen subspace and the remaining directions.
\end{proof}
\end{proposition}
\hlt{Key Insight}: Every finite-dimensional subspace can be paired with another subspace to recover the whole space as a direct sum.

\begin{definition} \hlt{Quotient Space}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$. For $\mathbf{v} \in \mathcal{V}$, the coset of $\mathbf{v}$ modulo $\mathcal{U}$ is
\begin{align}
\mathbf{v} + \mathcal{U} \defeq \left\{ \mathbf{v} + \mathbf{u} : \mathbf{u} \in \mathcal{U} \right\}. \nonumber
\end{align}
The quotient space $\mathcal{V}/\mathcal{U}$ is the set of all cosets $\mathbf{v} + \mathcal{U}$.
\end{definition}

\begin{proposition} \hlt{Structure of the Quotient}\\
Addition and scalar multiplication on $\mathcal{V}/\mathcal{U}$ defined by
\begin{align}
(\mathbf{v} + \mathcal{U}) + (\mathbf{w} + \mathcal{U}) \defeq (\mathbf{v} + \mathbf{w}) + \mathcal{U},
\qquad
a(\mathbf{v} + \mathcal{U}) \defeq a\mathbf{v} + \mathcal{U} \nonumber
\end{align}
are well-defined, and with these operations $\mathcal{V}/\mathcal{U}$ is a vector space.
\begin{proof}
If $\mathbf{v} + \mathcal{U} = \mathbf{v}' + \mathcal{U}$ and $\mathbf{w} + \mathcal{U} = \mathbf{w}' + \mathcal{U}$, then $\mathbf{v}-\mathbf{v}' \in \mathcal{U}$ and $\mathbf{w}-\mathbf{w}' \in \mathcal{U}$. Because $\mathcal{U}$ is a subspace,
\begin{align}
(\mathbf{v}+\mathbf{w}) - (\mathbf{v}'+\mathbf{w}') \in \mathcal{U}
\quad\text{and}\quad
a\mathbf{v} - a\mathbf{v}' \in \mathcal{U}, \nonumber
\end{align}
so the operations do not depend on the chosen representatives. The vector space axioms then follow from the corresponding axioms in $\mathcal{V}$.
\hlt{Structural Commentary}: Quotients inherit linear structure precisely because changing representatives only changes vectors by elements of $\mathcal{U}$.
\end{proof}
\end{proposition}
\hlt{Key Insight}: A quotient space treats vectors differing by an element of $\mathcal{U}$ as the same vector.

\begin{definition} \hlt{Quotient Map}\\
The map
\begin{align}
\pi : \mathcal{V} \to \mathcal{V}/\mathcal{U},
\qquad
\pi(\mathbf{v}) = \mathbf{v} + \mathcal{U}, \nonumber
\end{align}
is called the quotient map.
\end{definition}

\begin{proposition}
The quotient map $\pi$ is linear, surjective, and
\begin{align}
\ker(\pi) = \mathcal{U}. \nonumber
\end{align}
\hlt{Structural Commentary}: The quotient map is the canonical linear map that collapses $\mathcal{U}$ to zero and keeps track of cosets.
\end{proposition}
\hlt{Key Insight}: Quotient maps turn subspaces into kernels.

\begin{theorem} \hlt{Dimension of a Quotient}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U}$ is a subspace, then
\begin{align}
\dim(\mathcal{V}/\mathcal{U}) = \dim(\mathcal{V}) - \dim(\mathcal{U}). \nonumber
\end{align}
\begin{proof}
Choose a basis
\begin{align}
(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{w}_1, \ldots, \mathbf{w}_k) \nonumber
\end{align}
of $\mathcal{V}$ such that $(\mathbf{u}_1, \ldots, \mathbf{u}_m)$ is a basis of $\mathcal{U}$. Then every coset in $\mathcal{V}/\mathcal{U}$ has a unique form
\begin{align}
a_1(\mathbf{w}_1 + \mathcal{U}) + \cdots + a_k(\mathbf{w}_k + \mathcal{U}), \nonumber
\end{align}
so $(\mathbf{w}_1 + \mathcal{U}, \ldots, \mathbf{w}_k + \mathcal{U})$ is a basis of $\mathcal{V}/\mathcal{U}$. Hence
\begin{align}
\dim(\mathcal{V}/\mathcal{U}) = k = \dim(\mathcal{V}) - \dim(\mathcal{U}). \nonumber
\end{align}
\hlt{Structural Commentary}: The quotient removes the directions belonging to $\mathcal{U}$ and retains the complementary degrees of freedom.
\end{proof}
\end{theorem}
\hlt{Key Insight}: Passing to $\mathcal{V}/\mathcal{U}$ subtracts the dimension of the collapsed subspace.

\begin{remark} \hlt{Structural Viewpoint}\\
Passing to a quotient means declaring every vector in $\mathcal{U}$ to be invisible. Direct sums split a space into independent pieces; quotients collapse a chosen piece to zero. These are complementary ways of extracting structure, and they reappear constantly in the study of linear maps through kernels, ranges, and induced maps.
\end{remark}
