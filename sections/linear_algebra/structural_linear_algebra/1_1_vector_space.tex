\subsubsection{Vector Spaces and Linear Structure}

% Axler 1.A–1.C, 2.A-2.C, 3.E
% Gracia/Horn 1.1-1.6, 2.1-2.2

The guiding idea is that a vector space is not a list of coordinates first. It is a space of objects that can be added and scaled, and coordinates only appear after a basis is chosen.

\begin{definition} \hlt{Vector Space}\\
Let $\mathbb{F}$ be a field. A vector space over $\mathbb{F}$ is a set $\mathcal{V}$ together with operations of vector addition and scalar multiplication such that, for all $\mathbf{u}, \mathbf{v}, \mathbf{w} \in \mathcal{V}$ and all $a,b \in \mathbb{F}$, the following axioms hold:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item vector addition is commutative: $\mathbf{u} + \mathbf{v} = \mathbf{v} + \mathbf{u}$;
\item vector addition is associative: $(\mathbf{u} + \mathbf{v}) + \mathbf{w} = \mathbf{u} + (\mathbf{v} + \mathbf{w})$;
\item there exists an additive identity $\mathbf{0} \in \mathcal{V}$ such that $\mathbf{0} + \mathbf{v} = \mathbf{v}$;
\item for each $\mathbf{v} \in \mathcal{V}$ there exists an additive inverse $-\mathbf{v} \in \mathcal{V}$ such that $\mathbf{v} + (-\mathbf{v}) = \mathbf{0}$;
\item scalar multiplication has identity: $1\mathbf{v} = \mathbf{v}$;
\item scalar multiplication is compatible with multiplication in $\mathbb{F}$: $(ab)\mathbf{v} = a(b\mathbf{v})$;
\item scalar multiplication distributes over vector addition: $a(\mathbf{u} + \mathbf{v}) = a\mathbf{u} + a\mathbf{v}$;
\item scalar multiplication distributes over addition in $\mathbb{F}$: $(a+b)\mathbf{v} = a\mathbf{v} + b\mathbf{v}$.
\end{enumerate}
If $\mathcal{V} = \{ \mathbf{0} \}$, then $\mathcal{V}$ is a \hlt{zero vector space}; if $\mathcal{V} \neq \{ \mathbf{0} \}$, then $\mathcal{V}$ is a \hlt{nonzero vector space}.
\end{definition}

\begin{definition} \hlt{Finitely Nonzero Sequence Space}\\
Let $\mathbb{F}^{(\mathbb{N})}$ denote the set of all sequences $(x_1,x_2,\ldots)$ of elements of $\mathbb{F}$ such that $x_j=0$ for all but finitely many $j\in\mathbb{N}$. Addition and scalar multiplication are defined componentwise.
\end{definition}

\begin{proposition} \label{prop:finitely-nonzero-sequences-vector-space} \hlt{Finitely Nonzero Sequences Form a Vector Space}\\
The set $\mathbb{F}^{(\mathbb{N})}$ is a vector space over $\mathbb{F}$ under componentwise addition and scalar multiplication.
\begin{proof}
The zero sequence belongs to $\mathbb{F}^{(\mathbb{N})}$. If $\mathbf{x}=(x_1,x_2,\ldots)$ and $\mathbf{y}=(y_1,y_2,\ldots)$ are in $\mathbb{F}^{(\mathbb{N})}$ and $a\in\mathbb{F}$, then the nonzero entries of $a\mathbf{x}+\mathbf{y}$ can occur only among the finitely many indices where $\mathbf{x}$ or $\mathbf{y}$ is nonzero. Thus $a\mathbf{x}+\mathbf{y}\in\mathbb{F}^{(\mathbb{N})}$, and the vector space axioms are inherited componentwise from $\mathbb{F}$.\\
\hlt{Structural Commentary}: Finite support is preserved by finite linear combinations.\\
\hlt{Key Insight}: Componentwise operations cannot create infinitely many new nonzero entries.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:basicvectorspaceconsequences} \hlt{Basic Consequences of Vector Space Axioms}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then the additive identity and additive inverses are unique. Moreover, for all $a \in \mathbb{F}$ and $\mathbf{v} \in \mathcal{V}$, $0\mathbf{v} = \mathbf{0}$, $a\mathbf{0} = \mathbf{0}$, $(-1)\mathbf{v} = -\mathbf{v}$ and $a\mathbf{v} = \mathbf{0}$ if and only if $a = 0$ or $\mathbf{v} = \mathbf{0}$.
\begin{proof}
If $\mathbf{0}$ and $\mathbf{0}'$ are additive identities, then $\mathbf{0} = \mathbf{0} + \mathbf{0}' = \mathbf{0}'$, so the additive identity is unique.\\
If $\mathbf{w}$ and $\mathbf{w}'$ are additive inverses of $\mathbf{v}$, then $\mathbf{w} = \mathbf{w} + \mathbf{0} = \mathbf{w} + (\mathbf{v} + \mathbf{w}') = (\mathbf{w} + \mathbf{v}) + \mathbf{w}' = \mathbf{0} + \mathbf{w}' = \mathbf{w}'$, so additive inverses are unique. Since $0\mathbf{v} = (0+0)\mathbf{v} = 0\mathbf{v} + 0\mathbf{v}$, cancellation gives $0\mathbf{v} = \mathbf{0}$.\\
Similarly, $a\mathbf{0} = a(\mathbf{0}+\mathbf{0}) = a\mathbf{0} + a\mathbf{0}$, so $a\mathbf{0} = \mathbf{0}$. Also $\mathbf{v} + (-1)\mathbf{v} = 1\mathbf{v} + (-1)\mathbf{v} = (1-1)\mathbf{v} = 0\mathbf{v} = \mathbf{0}$, hence $(-1)\mathbf{v}$ is the additive inverse of $\mathbf{v}$, and therefore $(-1)\mathbf{v} = -\mathbf{v}$.\\
Finally, if $a = 0$ or $\mathbf{v} = \mathbf{0}$, then the preceding identities give $a\mathbf{v} = \mathbf{0}$. Conversely, suppose $a\mathbf{v} = \mathbf{0}$ and $a \neq 0$. Then $a^{-1}$ exists in $\mathbb{F}$, and $\mathbf{v} = 1\mathbf{v} = (a^{-1}a)\mathbf{v} = a^{-1}(a\mathbf{v}) = a^{-1}\mathbf{0} = \mathbf{0}$.\\
\hlt{Structural Commentary}: These identities are the basic algebraic hygiene of vector spaces. They let later arguments use cancellation, zero vectors, and inverses without reopening the axioms each time.\\
\hlt{Key Insight}: The vector space axioms force the zero and inverse behaviour to be rigid, not optional structure.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:negativeoneactsasinverse}
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then $(-1)\mathbf{v} = -\mathbf{v}$ for every $\mathbf{v} \in \mathcal{V}$.
\begin{proof}
Let $\mathbf{v} \in \mathcal{V}$. We show that $(-1)\mathbf{v} + \mathbf{v} = \mathbf{0}$. By the vector space axioms and Theorem \ref{thm:basicvectorspaceconsequences},
\begin{align}
(-1)\mathbf{v} + \mathbf{v} = (-1)\mathbf{v} + 1\mathbf{v} = (-1+1)\mathbf{v} = 0\mathbf{v} = \mathbf{0} \nonumber
\end{align}
Thus $(-1)\mathbf{v}$ is the additive inverse of $\mathbf{v}$, so $(-1)\mathbf{v} = -\mathbf{v}$.\\
\hlt{Structural Commentary}: Scalar multiplication by $-1$ is exactly same operation as taking additive inverse.\\
\hlt{Key Insight}: Negation is not an extra operation; it is already encoded by the field action.
\end{proof}
\end{corollary}

\begin{definition} A \hlt{subspace} of a vector space $\mathcal{V}$ over $\mathbb{F}$ is a subset $\mathcal{U} \subseteq \mathcal{V}$ that is itself a vector space over $\mathbb{F}$ with the same vector addition and scalar multiplication as in $\mathcal{V}$.
\end{definition}

\begin{theorem} \label{thm:subspacetest} \hlt{Subspace Test}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$ and let $\mathcal{U}$ be a nonempty subset of $\mathcal{V}$. Then $\mathcal{U}$ is a subspace of $\mathcal{V}$ if and only if $c\mathbf{u} + \mathbf{v} \in \mathcal{U}$ for all $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and all $c \in \mathbb{F}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathcal{U}$ is a subspace, then $c\mathbf{u}+\mathbf{v} \in \mathcal{U}$ for all $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and all $c \in \mathbb{F}$ because $\mathcal{U}$ is closed under scalar multiplication and vector addition.
\item[$(\Leftarrow)$] Suppose that $c\mathbf{u}+\mathbf{v} \in \mathcal{U}$ whenever $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and $c \in \mathbb{F}$. We verify that $\mathcal{U}$ is a vector space with the operations inherited from $\mathcal{V}$. Since $\mathcal{U}$ is nonempty, let $\mathbf{u} \in \mathcal{U}$. By Corollary \ref{cor:negativeoneactsasinverse}, $\mathbf{0} = (-1)\mathbf{u}+\mathbf{u} \in \mathcal{U}$. If $\mathbf{u} \in \mathcal{U}$, then $-\mathbf{u} = (-1)\mathbf{u}+\mathbf{0} \in \mathcal{U}$. If $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and $c \in \mathbb{F}$, then $c\mathbf{u} = c\mathbf{u}+\mathbf{0} \in \mathcal{U}$ and $\mathbf{u}+\mathbf{v} = 1\mathbf{u}+\mathbf{v} \in \mathcal{U}$. Thus $\mathcal{U}$ contains $\mathbf{0}$ and is closed under additive inverses, scalar multiplication, and vector addition; the remaining vector space axioms are inherited from $\mathcal{V}$. Hence $\mathcal{U}$ is a subspace.
\end{enumerate}
\hlt{Structural Commentary}: The theorem compresses the subspace axioms into one closure condition. This is the practical test used whenever a subset is proposed as a linear space.\\
\hlt{Key Insight}: To prove a subset is a subspace, prove it survives one affine-looking linear combination.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:intersectionsubspaces} \hlt{Intersection of Subspaces}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$. Then $\mathcal{U}\cap\mathcal{W} = \left\{ \mathbf{v} : \mathbf{v}\in\mathcal{U} \text{ and } \mathbf{v}\in\mathcal{W} \right\}$ is a subspace of $\mathcal{V}$.
\begin{proof}
Since $\mathbf{0}\in\mathcal{U}$ and $\mathbf{0}\in\mathcal{W}$, we have $\mathbf{0}\in\mathcal{U}\cap\mathcal{W}$, so the intersection is nonempty. If $\mathbf{u},\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$ and $c\in\mathbb{F}$, then $c\mathbf{u}+\mathbf{v}$ lies in both $\mathcal{U}$ and $\mathcal{W}$ because both are subspaces. Hence $c\mathbf{u}+\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$, and Theorem \ref{thm:subspacetest} shows that $\mathcal{U}\cap\mathcal{W}$ is a subspace of $\mathcal{V}$.\\
\hlt{Structural Commentary}: Intersections preserve every closure condition shared by the subspaces. The subspace test packages this into one check.\\
\hlt{Key Insight}: Common vectors of two subspaces still form a subspace because the shared part remains closed under linear combinations.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Lists, Linear Combinations, and Span}\\
A \hlt{list} of vectors in $\mathcal{V}$ is a finite ordered sequence $\beta = (\mathbf{v}_1,\ldots,\mathbf{v}_m)$ in $\mathcal{V}$; the empty list is denoted $()$.\\
If $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is nonempty and $a_1,\ldots,a_m\in\mathbb{F}$, then $a_1\mathbf{v}_1+\cdots+a_m\mathbf{v}_m$ is a \hlt{linear combination} of $\beta$.\\
For $E\subseteq\mathcal{V}$, the \hlt{span} of $E$ is $\operatorname{span}(E)\defeq\{a_1\mathbf{v}_1+\cdots+a_m\mathbf{v}_m:m\in\N,\ a_j\in\mathbb{F},\ \mathbf{v}_j\in E\}$. $\operatorname{span}(\emptyset)=\{\mathbf{0}\}$.\\
If $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is a nonempty list, then $\operatorname{span}(\beta) = \operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_m)
\defeq \operatorname{span}(\{\mathbf{v}_1,\ldots,\mathbf{v}_m\})$.\\
A subset $E \subseteq \mathcal{V}$ \hlt{spans} $\mathcal{V}$, or is a \hlt{spanning set} for $\mathcal{V}$, if $\operatorname{span}(E)=\mathcal{V}$.\\
A list $\beta$ in $\mathcal{V}$ \hlt{spans} $\mathcal{V}$, or is a \hlt{spanning list} for $\mathcal{V}$, if $\operatorname{span}(\beta)=\mathcal{V}$.
\end{definition}

\begin{theorem} \label{thm:span-smallest-containing-subspace} \hlt{Span is the Smallest Containing Subspace}\\
Let $E\subseteq\mathcal{V}$. Then $\operatorname{span}(E)$ is the smallest subspace of $\mathcal{V}$ containing $E$.
\begin{proof}
If $E=\emptyset$, then $\operatorname{span}(E)=\{\mathbf{0}\}$, which is a subspace. Suppose $E\neq\emptyset$. If $\mathbf{x},\mathbf{y}\in\operatorname{span}(E)$ and $c\in\mathbb{F}$, then $\mathbf{x}$ and $\mathbf{y}$ are finite linear combinations of vectors in $E$. Hence $c\mathbf{x}+\mathbf{y}$ is also a finite linear combination of vectors in $E$, so $c\mathbf{x}+\mathbf{y}\in\operatorname{span}(E)$. By Theorem \ref{thm:subspacetest}, $\operatorname{span}(E)$ is a subspace of $\mathcal{V}$.\\
Each $\mathbf{v}\in E$ satisfies $\mathbf{v}=1\mathbf{v}\in\operatorname{span}(E)$, so $E\subseteq\operatorname{span}(E)$. If $\mathcal{U}$ is a subspace of $\mathcal{V}$ with $E\subseteq\mathcal{U}$, then closure under vector addition and scalar multiplication forces every finite linear combination of vectors in $E$ to lie in $\mathcal{U}$. Thus $\operatorname{span}(E)\subseteq\mathcal{U}$. Therefore $\operatorname{span}(E)$ is the smallest subspace of $\mathcal{V}$ containing $E$.\\
\hlt{Structural Commentary}: Span takes a set of vectors and closes it under all finite linear combinations. The result is precisely the smallest subspace forced by those vectors.\\
\hlt{Key Insight}: Nothing beyond linear closure added. Every containing subspace must already contain the span.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:spanmonotonicity} \hlt{Monotonicity of Span}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subsets of $\mathcal{V}$. If $\mathcal{U}\subseteq\mathcal{W}$, then $\operatorname{span}(\mathcal{U})\subseteq\operatorname{span}(\mathcal{W})$.
\begin{proof}
If $\mathcal{U}=\emptyset$, then $\operatorname{span}(\mathcal{U})=\{\mathbf{0}\}\subseteq\operatorname{span}(\mathcal{W})$.\\
Suppose $\mathcal{U}\neq\emptyset$. Every vector in $\operatorname{span}(\mathcal{U})$ is a finite linear combination of vectors in $\mathcal{U}$; since $\mathcal{U}\subseteq\mathcal{W}$, the same expression is a finite linear combination of vectors in $\mathcal{W}$. Hence $\operatorname{span}(\mathcal{U})\subseteq\operatorname{span}(\mathcal{W})$.\\
\hlt{Structural Commentary}: Enlarging the generating set can only enlarge the collection of linear combinations.\\
\hlt{Key Insight}: Span is monotone with respect to inclusion of generating sets.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Linear Independence} \\
A list $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ in $\mathcal{V}$ is \hlt{linearly independent} if $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = \mathbf{0}$ implies $a_1 = \cdots = a_m = 0$. Otherwise the list is linearly dependent.
\end{definition}

\begin{proposition} \label{prop:uniquecoefficientsindependentlist} \hlt{Unique Coefficients for Independent Lists}\\
Let $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ be a linearly independent list in $\mathcal{V}$. Then $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = b_1\mathbf{v}_1 + \cdots + b_m\mathbf{v}_m$ if and only if $a_j=b_j$ for each $j=1,\ldots,m$.
\begin{proof}
The displayed equality is equivalent to $(a_1-b_1)\mathbf{v}_1 + \cdots + (a_m-b_m)\mathbf{v}_m = \mathbf{0}$.\\
If $a_j=b_j$ for every $j$, this relation is immediate. Conversely, linear independence forces $a_j-b_j=0$ for every $j$, hence $a_j=b_j$ for every $j$.\\
\hlt{Structural Commentary}: Linear independence turns a vector equality into coefficient equality. It prevents two different coordinate lists from representing the same linear combination.\\
\hlt{Key Insight}: Independent vectors do not allow hidden cancellation among their coefficients.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:appendingtodependentlist} \hlt{Appending to a Dependent List}\\
Let $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ be a linearly dependent list in $\mathcal{V}$. Then $\forall \mathbf{v}\in\mathcal{V}$, the list $(\mathbf{v}_1, \ldots, \mathbf{v}_m, \mathbf{v})$ is linearly dependent.
\begin{proof}
Since $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly dependent, $\exists$ scalars $a_1,\ldots,a_m$, not all zero, s.t. $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = \mathbf{0}$.\\
For any $\mathbf{v}\in\mathcal{V}$, $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m + 0\mathbf{v} = \mathbf{0}$ and the scalars $a_1,\ldots,a_m,0$ are not all zero.\\
\hlt{Structural Commentary}: Dependence is inherited by longer lists because the old nontrivial relation survives after assigning coefficient $0$ to every new vector.\\
\hlt{Key Insight}: Once a list has a redundancy, adding more vectors cannot remove it.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Omitting a Vector from a List}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be a list in $\mathcal{V}$ with $m\geq 2$. For $j\in\{1,\ldots,m\}$, the list obtained from $\beta$ by omitting $\mathbf{v}_j$ is denoted $(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$.
\end{definition}

\begin{proposition} \label{prop:omittingfromindependentlist} \hlt{Omitting from an Independent List}\\
Let $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be a linearly independent list in $\mathcal{V}$ with $m\geq 2$. Then for each $j\in\{1,\ldots,m\}$, the list $(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$ is linearly independent.
\begin{proof}
Assign coefficient $0$ to $\mathbf{v}_j$. Then any zero relation in the sublist becomes a zero relation in the independent full list, so all coefficients vanish.\\
\hlt{Structural Commentary}: Linear independence passes to shorter lists because deleting a vector cannot create a new cancellation relation.\\
\hlt{Key Insight}: Every sublist of an independent list remains independent.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:adjoiningandremovingvectors} \hlt{Adjoining and Removing Vectors from Lists}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be a list in $\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\beta$ is linearly independent and $\mathbf{v}\notin\operatorname{span}(\beta)$, then $(\mathbf{v}_1,\ldots,\mathbf{v}_m,\mathbf{v})$ is linearly independent.
\item If $a_1\mathbf{v}_1+\cdots+a_m\mathbf{v}_m=\mathbf{0}$ is a nontrivial relation and $a_j\neq 0$, then $\operatorname{span}(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m) = \operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_m)$.\\
Consequently, if $\beta$ spans $\mathcal{V}$, then $(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$ also spans $\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Suppose $a_1\mathbf{v}_1+\cdots+a_m\mathbf{v}_m+a\mathbf{v}=\mathbf{0}$.\\
If $a\neq 0$, then $\mathbf{v}=-a^{-1}(a_1\mathbf{v}_1+\cdots+a_m\mathbf{v}_m)\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_m)$, a contradiction. Hence $a=0$, and linear independence of $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ gives $a_1=\cdots=a_m=0$.
\item
Since $a_j\neq 0$, $\mathbf{v}_j=-\sum\limits_{\substack{i=1\\i\neq j}}^m\frac{a_i}{a_j}\mathbf{v}_i$, so $\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$. Thus every linear combination using $\mathbf{v}_j$ can be rewritten without it, while the reverse inclusion is immediate. Hence the two spans are equal, and the spanning conclusion follows.\\
\end{enumerate}
\hlt{Structural Commentary}: These are the two basic list moves: add a genuinely new direction, or remove a vector already forced by a dependence relation. Together they control how independence and span change.\\
\hlt{Key Insight}: A vector outside the span increases independence; a vector solved from a dependence relation can be discarded without changing span.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:replacement-lemma} \hlt{Replacement Lemma}\\
Let $\mathcal{V}$ be nonzero, let $\beta=(\mathbf{u}_1,\ldots,\mathbf{u}_r)$ be a spanning list of $\mathcal{V}$, and let $\mathbf{v}\in\mathcal{V}$ be nonzero.\\
Suppose $\mathbf{v}=c_1\mathbf{u}_1+\cdots+c_r\mathbf{u}_r$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Some $c_j\neq 0$.
\item If $c_j\neq 0$, then $(\mathbf{v},\mathbf{u}_1,\ldots,\widehat{\mathbf{u}_j},\ldots,\mathbf{u}_r)$ spans $\mathcal{V}$; if $\beta$ is a basis, this replacement list is a basis.
\item If $r\geq 2$, $\beta$ is a basis, and $\mathbf{v}\notin\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_k)$ for some $k\in\{1,\ldots,r-1\}$, then $\exists$ $j\in\{k+1,\ldots,r\}$ such that $(\mathbf{v},\mathbf{u}_1,\ldots,\mathbf{u}_k,\mathbf{u}_{k+1},\ldots,\widehat{\mathbf{u}_j},\ldots,\mathbf{u}_r)$ is a basis of $\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If every $c_j=0$, then $\mathbf{v}=\mathbf{0}$, a contradiction.
\item Since $c_j\neq 0$, $\mathbf{u}_j=c_j^{-1}\left(\mathbf{v}-\sum_{i\neq j}c_i\mathbf{u}_i\right)$, so replacing $\mathbf{u}_j$ by $\mathbf{v}$ preserves span. If $\beta$ is a basis and $a\mathbf{v}+\sum_{i\neq j}b_i\mathbf{u}_i=\mathbf{0}$, substituting $\mathbf{v}=\sum_i c_i\mathbf{u}_i$ gives a relation in $\beta$ whose $\mathbf{u}_j$-coefficient is $ac_j$. Since $c_j\neq 0$, $a=0$; then all $b_i=0$.
\item If all $c_{k+1},\ldots,c_r = 0$, then $\mathbf{v}\in\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_k)$. Thus some $c_j\neq 0$ with $j>k$, (ii) gives desired basis.
\end{enumerate}
\hlt{Structural Commentary}: Replacement swaps one spanning direction without losing span.\\
\hlt{Key Insight}: A nonzero coefficient identifies the removable vector.
\end{proof}
\end{lemma}


\begin{lemma} \label{lem:lineardependencelemma} \hlt{Linear Dependence Lemma}\\
If $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly dependent and $\mathbf{v}_1 \neq \mathbf{0}$, then there exists $j \in \{2, \ldots, m\}$ such that $\mathbf{v}_j \in \operatorname{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{j-1})$ and removing $\mathbf{v}_j$ does not change the span of the list.
\begin{proof}
Because the list is dependent, there exist scalars $a_1, \ldots, a_m$, not all zero, such that $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = \mathbf{0}$.\\
Since $\mathbf{v}_1\neq\mathbf{0}$, not all of $a_2,\ldots,a_m$ can be zero; otherwise $a_1\mathbf{v}_1=\mathbf{0}$ would force $a_1=0$. Choose $j$ to be the largest index such that $a_j \neq 0$. Then $j\geq 2$ and $\mathbf{v}_j = -\frac{a_1}{a_j}\mathbf{v}_1 - \cdots - \frac{a_{j-1}}{a_j}\mathbf{v}_{j-1}$, so $\mathbf{v}_j$ lies in the span of the preceding vectors. Proposition \ref{prop:adjoiningandremovingvectors} shows that removing $\mathbf{v}_j$ does not change the span of the list.\\
\hlt{Structural Commentary}: This lemma is the ordered version of removing a dependent vector. Choosing the largest nonzero coefficient ensures the removable vector occurs after the first vector.\\
\hlt{Key Insight}: Linear dependence means at least one later vector can be discarded without losing span.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:finitedimlenofvec}
In a finite-dimensional vector space, the length of every linearly independent list is at most the length of every spanning list.
\begin{proof}
Let $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ be linearly independent and let $(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ span $\mathcal{V}$. Suppose $m>n$. Starting from $(\mathbf{w}_1,\ldots,\mathbf{w}_n)$, inductively build a spanning list of length $n$ whose first $k$ entries are $\mathbf{u}_1,\ldots,\mathbf{u}_k$.\\
For $k<n$, express $\mathbf{u}_{k+1}$ in the current spanning list. Some coefficient after the first $k$ entries is nonzero; otherwise $\mathbf{u}_{k+1}\in\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_k)$. Lemma \ref{lem:replacement-lemma}(ii) replaces that entry by $\mathbf{u}_{k+1}$ while preserving span.\\
After $n$ steps, $(\mathbf{u}_1,\ldots,\mathbf{u}_n)$ spans $\mathcal{V}$, so $\mathbf{u}_{n+1}\in\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$, contradicting independence. Hence $m\leq n$.\\
\hlt{Structural Commentary}: This comparison theorem is the counting engine behind dimension. It says independence and spanning place opposite pressure on list length.\\
\hlt{Key Insight}: Independent lists cannot be longer than spanning lists in finite dimensions.
\end{proof}
\end{theorem}


\begin{definition} \hlt{Basis}\\
A list $\beta$ in $\mathcal{V}$ is a \hlt{basis} of $\mathcal{V}$ if $\beta$ is linearly independent and $\operatorname{span}(\beta)=\mathcal{V}$.
\end{definition}

\begin{proposition} \label{prop:basis-unique-coordinate-representation} \hlt{Unique Coordinate Representation}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a list in $\mathcal{V}$. Then $\beta$ is a basis of $\mathcal{V}$ if and only if every $\mathbf{v}\in\mathcal{V}$ can be written uniquely as
\begin{align}
\mathbf{v}=a_1\mathbf{v}_1+\cdots+a_n\mathbf{v}_n,\qquad a_1,\ldots,a_n\in\mathbb{F}. \nonumber
\end{align}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $\beta$ is a basis of $\mathcal{V}$. Since $\operatorname{span}(\beta)=\mathcal{V}$, every $\mathbf{v}\in\mathcal{V}$ has a representation of the displayed form. If $a_1\mathbf{v}_1+\cdots+a_n\mathbf{v}_n=b_1\mathbf{v}_1+\cdots+b_n\mathbf{v}_n$, then $(a_1-b_1)\mathbf{v}_1+\cdots+(a_n-b_n)\mathbf{v}_n=\mathbf{0}$. Linear independence of $\beta$ gives $a_j=b_j$ for each $j$, so the representation is unique.
\item[$(\Leftarrow)$] Suppose every $\mathbf{v}\in\mathcal{V}$ has a unique representation of the displayed form. Existence gives $\operatorname{span}(\beta)=\mathcal{V}$. If $a_1\mathbf{v}_1+\cdots+a_n\mathbf{v}_n=\mathbf{0}$, then this representation of $\mathbf{0}$ must equal the representation with all coefficients $0$. Hence $a_1=\cdots=a_n=0$, so $\beta$ is linearly independent. Therefore $\beta$ is a basis of $\mathcal{V}$.
\end{enumerate}
\hlt{Structural Commentary}: A basis is a coordinate system: spanning gives existence of coordinates, and linear independence gives uniqueness.\\
\hlt{Key Insight}: Coordinates are meaningful when every vector has one representation and no hidden alternative.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:reducing-spanning-lists} \hlt{Reducing Spanning Lists}\\
Every spanning list in a finite-dimensional vector space contains a sublist that is a basis.
\begin{proof}
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a spanning list of $\mathcal{V}$. If $\beta$ is linearly independent, then $\beta$ is already a basis. Otherwise, remove a redundant vector without changing the span: if $\mathbf{v}_1=\mathbf{0}$, remove $\mathbf{v}_1$; if $\mathbf{v}_1\neq\mathbf{0}$, apply Lemma \ref{lem:lineardependencelemma}. Repeating this finite process gives a sublist that still spans $\mathcal{V}$ and is linearly independent.\\
\hlt{Structural Commentary}: A spanning list may contain redundant vectors. Removing only vectors forced by the others preserves span until a basis remains.\\
\hlt{Key Insight}: Spanning gives enough vectors; reduction deletes exactly the unnecessary ones.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:extending-independent-lists} \hlt{Extending Independent Lists}\\
Every linearly independent list in a finite-dimensional vector space can be extended to a basis.
\begin{proof}
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be linearly independent in $\mathcal{V}$, and choose a spanning list $(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ of $\mathcal{V}$. Starting with $\mathcal{B}=\beta$, inspect $\mathbf{w}_1,\ldots,\mathbf{w}_n$ in order and append $\mathbf{w}_j$ exactly when $\mathbf{w}_j\notin\operatorname{span}(\mathcal{B})$. Each append preserves linear independence by Proposition \ref{prop:adjoiningandremovingvectors}. After all $\mathbf{w}_j$ have been inspected, every vector in the spanning list lies in $\operatorname{span}(\mathcal{B})$, so $\mathcal{B}$ spans $\mathcal{V}$.\\
\hlt{Structural Commentary}: An independent list has no redundancy, but may miss directions. Appending only vectors outside its span fills those directions while preserving independence.\\
\hlt{Key Insight}: A finite-dimensional independent list becomes a basis once enough new directions are added.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:finite-dimensional-spaces-and-subspaces-have-bases}
Every finite-dimensional vector space has a basis, and every subspace of a finite-dimensional vector space is finite-dimensional.
\begin{proof}
Since $\mathcal{V}$ is finite-dimensional, it has a finite spanning list; Theorem \ref{thm:reducing-spanning-lists} reduces it to a basis. Let $\mathcal{U}\subseteq\mathcal{V}$ be a subspace. If $\mathcal{U}=\{\mathbf{0}\}$, then $\mathcal{U}$ is finite-dimensional. Otherwise, build a linearly independent list in $\mathcal{U}$ by adjoining vectors outside the current span.\\
Theorem \ref{thm:finitedimlenofvec} bounds the length of this list by the length of any spanning list of $\mathcal{V}$, so the process must terminate. At termination, the list spans $\mathcal{U}$ and hence is a basis of $\mathcal{U}$.\\
\hlt{Structural Commentary}: Finite-dimensional spaces have finite coordinate systems, and subspaces cannot contain longer independent lists than the ambient space permits.\\
\hlt{Key Insight}: Finite-dimensionality passes downward to subspaces.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:basis-length-well-defined} \hlt{Basis Length Is Well-Defined}\\
Any two bases of a finite-dimensional vector space have the same length.
\begin{proof}
Let $\beta_1=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ and $\beta_2=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ be bases of $\mathcal{V}$. Since $\beta_1$ is linearly independent and $\beta_2$ spans $\mathcal{V}$, Theorem \ref{thm:finitedimlenofvec} gives $m\leq n$. Reversing the roles of the two bases gives $n\leq m$. Hence $m=n$.\\
\hlt{Structural Commentary}: Bases are simultaneously independent and spanning, so the comparison theorem traps their lengths in both directions. This makes dimension well-defined.\\
\hlt{Key Insight}: Dimension is an intrinsic number, not an artefact of a particular basis.
\end{proof}
\end{theorem}

\begin{definition} \hlt{Dimension}\\
The zero vector space has \hlt{dimension} $0$. If $\mathcal{V}$ has a basis of length $n > 0 \in \Z$, then $\mathcal{V}$ is \hlt{$n$-dimensional} and $\dim(\mathcal{V})=n$. A vector space is \hlt{finite-dimensional} if it has finite dimension; otherwise it is \hlt{infinite-dimensional}.
\end{definition}

\begin{proposition} \label{prop:dimension-bounds-and-basis-criteria} \hlt{Dimension Bounds and Basis Criteria}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U} \subseteq \mathcal{V}$ is a subspace, then $\dim(\mathcal{U}) \leq \dim(\mathcal{V})$, with equality if and only if $\mathcal{U}=\mathcal{V}$. Moreover, a spanning list of length $\dim(\mathcal{V})$ is automatically a basis, and a linearly independent list of length $\dim(\mathcal{V})$ is automatically a basis.
\begin{proof}
Let $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ be a basis of $\mathcal{U}$ and $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ a basis of $\mathcal{V}$. Theorem \ref{thm:finitedimlenofvec} gives $m\leq n$, so $\dim(\mathcal{U})\leq\dim(\mathcal{V})$. If $\mathcal{U}=\mathcal{V}$, equality is clear. Conversely, if $m=n$, then the basis of $\mathcal{U}$ is an independent list in $\mathcal{V}$ of length $\dim(\mathcal{V})$; extending it to a basis of $\mathcal{V}$ adds no vectors, so it already spans $\mathcal{V}$. Hence $\mathcal{U}=\mathcal{V}$.\\
A spanning list of length $\dim(\mathcal{V})$ reduces to a basis without removing vectors, and an independent list of that length extends to a basis without adding vectors.\\
\hlt{Structural Commentary}: Dimension bounds subspaces and detects full-size lists. Equality of dimensions forces a subspace to be the whole space.\\
\hlt{Key Insight}: In dimension $n$, $n$ independent or $n$ spanning vectors already form a basis.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Sums of Subspaces}\\
If $\mathcal{U}_1, \ldots, \mathcal{U}_m$ are subspaces of $\mathcal{V}$, define $\mathcal{U}_1 + \cdots + \mathcal{U}_m \defeq \left\{ \mathbf{u}_1 + \cdots + \mathbf{u}_m : \mathbf{u}_j \in \mathcal{U}_j \right\}$.
In particular, for subspaces $\mathcal{U},\mathcal{W}\subseteq\mathcal{V}$, $\mathcal{U}+\mathcal{W} = \operatorname{span}(\mathcal{U}\cup\mathcal{W}) = \left\{ \mathbf{u}+\mathbf{w} : \mathbf{u}\in\mathcal{U},\ \mathbf{w}\in\mathcal{W} \right\}$.
If every vector in this sum has a unique such decomposition, then the sum is direct and we write $\mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$.\\
For two subspaces, $\mathcal{U}+\mathcal{W}$ is direct if and only if $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$; in this case it is denoted $\mathcal{U}\oplus\mathcal{W}$.
\end{definition}

\begin{proposition} \label{prop:uniqueness-direct-sum-two-subspaces} \hlt{Uniqueness in a Direct Sum}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$ with $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$. Then each vector in $\mathcal{U}\oplus\mathcal{W}$ has a unique representation $\mathbf{u}+\mathbf{w}$ with $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{W}$.
\begin{proof}
Existence follows from the definition of $\mathcal{U}+\mathcal{W}$. For uniqueness, suppose $\mathbf{u}_1+\mathbf{w}_1=\mathbf{u}_2+\mathbf{w}_2$, $\mathbf{u}_1,\mathbf{u}_2\in\mathcal{U}$, $\mathbf{w}_1,\mathbf{w}_2\in\mathcal{W}$. Then $\mathbf{u}_1-\mathbf{u}_2=\mathbf{w}_2-\mathbf{w}_1 \in \mathcal{U}, \mathcal{W}$, hence equals $\mathbf{0}$. Thus $\mathbf{u}_1=\mathbf{u}_2$ and $\mathbf{w}_1=\mathbf{w}_2$.\\
\hlt{Structural Commentary}: Trivial intersection prevents cancellation between summands.\\
\hlt{Key Insight}: Directness means unique decomposition.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:sum-subspaces-smallest-containing-subspace} \hlt{Sum of Subspaces Is the Smallest Containing Subspace}\\
Suppose $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is the smallest subspace of $\mathcal{V}$ containing $\mathcal{U}_1,\ldots,\mathcal{U}_m$.
\begin{proof}
Since $\mathbf{0}\in\mathcal{U}_j$ for each $j$, we have $\mathbf{0}\in\mathcal{U}_1+\cdots+\mathcal{U}_m$. If $\mathbf{x}=\mathbf{u}_1+\cdots+\mathbf{u}_m$ and $\mathbf{y}=\mathbf{v}_1+\cdots+\mathbf{v}_m$ lie in $\mathcal{U}_1+\cdots+\mathcal{U}_m$, then for any $c \in \mathbb{F}$, $c\mathbf{x}+\mathbf{y} = (c\mathbf{u}_1+\mathbf{v}_1)+\cdots+(c\mathbf{u}_m+\mathbf{v}_m) \in \mathcal{U}_1+\cdots+\mathcal{U}_m$, because $c\mathbf{u}_j+\mathbf{v}_j\in\mathcal{U}_j$ for each $j$. By Theorem \ref{thm:subspacetest}, $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a subspace of $\mathcal{V}$.\\
Each $\mathcal{U}_j$ is contained in $\mathcal{U}_1+\cdots+\mathcal{U}_m$ by taking all other summands to be $\mathbf{0}$. If $\mathcal{W}$ is any subspace of $\mathcal{V}$ containing $\mathcal{U}_1,\ldots,\mathcal{U}_m$, then closure under addition gives $\mathbf{u}_1+\cdots+\mathbf{u}_m\in\mathcal{W}$ whenever $\mathbf{u}_j\in\mathcal{U}_j$. Hence $\mathcal{U}_1+\cdots+\mathcal{U}_m\subseteq\mathcal{W}$, so the sum is the smallest such subspace.\\
\hlt{Structural Commentary}: Sums of subspaces are obtained by closing several subspaces under finite addition. Minimality follows because every containing subspace is already closed under those sums.\\
\hlt{Key Insight}: The sum adds no extra structure beyond the linear combinations forced by the given subspaces.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:direct-sum-equivalent-conditions} \hlt{Equivalent Conditions for a Direct Sum}\\
For subspaces $\mathcal{U}_1, \ldots, \mathcal{U}_m \subseteq \mathcal{V}$, the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$;
\item every $\mathbf{v} \in \mathcal{V}$ has a unique representation $\mathbf{v} = \mathbf{u}_1 + \cdots + \mathbf{u}_m$ with $\mathbf{u}_j \in \mathcal{U}_j$;
\item $\mathcal{V}=\mathcal{U}_1+\cdots+\mathcal{U}_m$, and if $\mathbf{u}_1 + \cdots + \mathbf{u}_m = \mathbf{0}$ with $\mathbf{u}_j \in \mathcal{U}_j$, then $\mathbf{u}_1 = \cdots = \mathbf{u}_m = \mathbf{0}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman)*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ This follows from the definition of direct sum.
\item[$(ii)$]$\Rightarrow (iii)$ Every vector lies in $\mathcal{U}_1+\cdots+\mathcal{U}_m$, so $\mathcal{V}=\mathcal{U}_1+\cdots+\mathcal{U}_m$. If $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$, then this representation of $\mathbf{0}$ must equal $\mathbf{0}+\cdots+\mathbf{0}$ by uniqueness, so each $\mathbf{u}_j=\mathbf{0}$.
\item[$(iii)$]$\Rightarrow (i)$ Since $\mathcal{V}=\mathcal{U}_1+\cdots+\mathcal{U}_m$, every $\mathbf{v}\in\mathcal{V}$ has at least one representation. If $\mathbf{v}=\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{w}_1+\cdots+\mathbf{w}_m$, then $(\mathbf{u}_1-\mathbf{w}_1)+\cdots+(\mathbf{u}_m-\mathbf{w}_m)=\mathbf{0}$. By iii., $\mathbf{u}_j-\mathbf{w}_j=\mathbf{0}$ for every $j$, so the representation is unique. Hence $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_m$.
\end{enumerate}
\hlt{Structural Commentary}: Direct sums require both spanning by the summands and no cancellation among them. The zero-representation condition isolates the no-cancellation part.\\
\hlt{Key Insight}: Uniqueness is equivalent to saying the only way the summands add to zero is trivially.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Product of Vector Spaces}\\
Let $\mathcal{V}_1,\ldots,\mathcal{V}_m$ be vector spaces over $\mathbb{F}$. Their \hlt{product} is $\mathcal{V}_1\times\cdots\times\mathcal{V}_m \defeq \{(\mathbf{v}_1,\ldots,\mathbf{v}_m):\mathbf{v}_j\in\mathcal{V}_j\}$.\\
Addition, scalar multiplication defined componentwise: $(\mathbf{u}_1,\ldots,\mathbf{u}_m)+(\mathbf{v}_1,\ldots,\mathbf{v}_m) \defeq (\mathbf{u}_1+\mathbf{v}_1,\ldots,\mathbf{u}_m+\mathbf{v}_m)$ and $a(\mathbf{v}_1,\ldots,\mathbf{v}_m) \defeq (a\mathbf{v}_1,\ldots,a\mathbf{v}_m)$.
\end{definition}

\begin{proposition} \label{prop:product-vector-space} \hlt{Products Are Vector Spaces}\\
If $\mathcal{V}_1,\ldots,\mathcal{V}_m$ are vector spaces over $\mathbb{F}$, then $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ is a vector space over $\mathbb{F}$ under componentwise ops.
\begin{proof}
The additive identity is $(\mathbf{0}_1,\ldots,\mathbf{0}_m)$, where $\mathbf{0}_j$ is the zero vector of $\mathcal{V}_j$, and additive inverse of $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is $(-\mathbf{v}_1,\ldots,-\mathbf{v}_m)$. All remaining vector space axioms hold componentwise because they hold in each $\mathcal{V}_j$.\\
\hlt{Structural Commentary}: Product spaces collect several vector spaces into one coordinatewise structure.\\
\hlt{Key Insight}: Every vector space axiom reduces to the same axiom in each component.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-product} \hlt{Dimension of a Product}\\
If $\mathcal{V}_1,\ldots,\mathcal{V}_m$ are finite-dimensional vector spaces over $\mathbb{F}$, then $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ is finite-dimensional and
\begin{align}
\dim(\mathcal{V}_1\times\cdots\times\mathcal{V}_m) = \dim(\mathcal{V}_1)+\cdots+\dim(\mathcal{V}_m) \nonumber
\end{align}
\begin{proof}
For each $j$, let $(\mathbf{v}_{j,1},\ldots,\mathbf{v}_{j,n_j})$ be a basis of $\mathcal{V}_j$. For every $j$ and $k$, let $\mathbf{e}_{j,k}$ denote the element of $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ whose $j$th component is $\mathbf{v}_{j,k}$ and whose other components are zero.\\
The list of all $\mathbf{e}_{j,k}$ spans the product because each component can be expanded in its own basis. If a linear combination of the $\mathbf{e}_{j,k}$ is zero, then looking at the $j$th component gives a linear relation among $(\mathbf{v}_{j,1},\ldots,\mathbf{v}_{j,n_j})$, so all coefficients in that component are zero. Hence the list is linearly independent and therefore is a basis of the product. Its length is $n_1+\cdots+n_m$.\\
\hlt{Structural Commentary}: A basis for a product is formed by placing each component basis into its own slot.\\
\hlt{Key Insight}: Product dimension counts the independent directions contributed by each factor.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:dimension-formula-for-sum} \hlt{Dimension Formula for a Sum}\\
If $\mathcal{U}, \mathcal{W}$ are finite-dimensional subspaces of $\mathcal{V}$, then $\dim(\mathcal{U} + \mathcal{W}) = \dim(\mathcal{U}) + \dim(\mathcal{W}) - \dim(\mathcal{U} \cap \mathcal{W})$.
\begin{proof}
Choose a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r)$ of $\mathcal{U} \cap \mathcal{W}$, extend it to a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p)$ of $\mathcal{U}$, and to a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{w}_1, \ldots, \mathbf{w}_q)$ of $\mathcal{W}$. Then $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p, \mathbf{w}_1, \ldots, \mathbf{w}_q) $ spans $\mathcal{U}+\mathcal{W}$ because every vector in the sum is the sum of one vector from $\mathcal{U}$ and one vector from $\mathcal{W}$.\\
To prove linear independence, suppose $a_1\mathbf{e}_1+\cdots+a_r\mathbf{e}_r+b_1\mathbf{u}_1+\cdots+b_p\mathbf{u}_p+c_1\mathbf{w}_1+\cdots+c_q\mathbf{w}_q=\mathbf{0}$.
Then $a_1\mathbf{e}_1+\cdots+a_r\mathbf{e}_r+b_1\mathbf{u}_1+\cdots+b_p\mathbf{u}_p$ lies in $\mathcal{U}\cap\mathcal{W}$. Since $(\mathbf{e}_1,\ldots,\mathbf{e}_r)$ is a basis of $\mathcal{U}\cap\mathcal{W}$ and $(\mathbf{e}_1,\ldots,\mathbf{e}_r,\mathbf{u}_1,\ldots,\mathbf{u}_p)$ is a basis of $\mathcal{U}$, we get $b_1=\cdots=b_p=0$. The remaining relation lies in the basis $(\mathbf{e}_1,\ldots,\mathbf{e}_r,\mathbf{w}_1,\ldots,\mathbf{w}_q)$ of $\mathcal{W}$, so $a_1=\cdots=a_r=c_1=\cdots=c_q=0$. Thus the displayed list is a basis of $\mathcal{U}+\mathcal{W}$.\\
Therefore $\dim(\mathcal{U} + \mathcal{W}) = r + p + q = (r+p) + (r+q) - r$.\\
\hlt{Structural Commentary}: The formula corrects double-counting: vectors in $\mathcal{U} \cap \mathcal{W}$ are counted once in each subspace. Extending an intersection basis separates shared directions from new directions.\\
\hlt{Key Insight}: The dimension of a sum is the sum of dimensions minus the overlap.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:dimension-direct-sum} \hlt{Dimension of a Direct Sum}\\
If $\mathcal{U}$ and $\mathcal{W}$ are finite-dimensional subspaces of $\mathcal{V}$ with $\mathcal{U} \cap \mathcal{W} = \{ \mathbf{0} \}$, then $\dim(\mathcal{U} \oplus \mathcal{W}) = \dim(\mathcal{U}) + \dim(\mathcal{W})$.
\begin{proof}
Since $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$, the sum is direct and $\dim(\mathcal{U}\cap\mathcal{W})=0$. The result follows from Theorem \ref{thm:dimension-formula-for-sum}.\\
\hlt{Structural Commentary}: When the overlap is trivial, there is no double-counting in the dimension formula. Direct sums are exactly the case where the overlap term vanishes.\\
\hlt{Key Insight}: Dimensions add exactly for direct sums.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:direct-sum-iff-dimensions-add} \hlt{Direct Sums and Dimension}\\
Suppose $\mathcal{V}$ is finite-dimensional and $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a direct sum if and only if $\dim(\mathcal{U}_1+\cdots+\mathcal{U}_m) = \dim(\mathcal{U}_1)+\cdots+\dim(\mathcal{U}_m)$.
\begin{proof}
For each $j$, choose a basis $\beta_j$ of $\mathcal{U}_j$, and concatenate these bases into a list $\beta$.\\
This list spans $\mathcal{U}_1+\cdots+\mathcal{U}_m$ and has length $\dim(\mathcal{U}_1)+\cdots+\dim(\mathcal{U}_m)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If the sum is direct, any linear relation among the vectors in $\beta$ gives a zero decomposition $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ with $\mathbf{u}_j\in\mathcal{U}_j$. Directness gives $\mathbf{u}_j=\mathbf{0}$ for every $j$, and independence of each $\beta_j$ gives all coefficients zero. Thus $\beta$ is a basis of the sum, and the dimensions add.
\item[$(\Leftarrow)$] If the dimensions add, then $\beta$ is a spanning list whose length equals the dimension of the sum, so Proposition \ref{prop:dimension-bounds-and-basis-criteria} makes $\beta$ a basis. A zero decomposition $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ expands to a zero linear relation in this basis, so every $\mathbf{u}_j=\mathbf{0}$. Hence the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: Direct sums add dimensions without hidden overlap.\\
\hlt{Key Insight}: Equality of dimensions means the concatenated bases have no linear redundancy.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:large-subspaces-intersect} \hlt{Large Subspaces Must Intersect}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of a finite-dimensional vector space $\mathcal{V}$, and let $k$ be a positive integer.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item If $\dim(\mathcal{U})+\dim(\mathcal{W})>\dim(\mathcal{V})$, then $\mathcal{U}\cap\mathcal{W}$ contains a nonzero vector.
\item If $\dim(\mathcal{U})+\dim(\mathcal{W})\geq\dim(\mathcal{V})+k$, then $\mathcal{U}\cap\mathcal{W}$ contains $k$ linearly independent vectors.
\end{enumerate}
\begin{proof}
Since $\mathcal{U}+\mathcal{W}\subseteq\mathcal{V}$, Theorem \ref{thm:dimension-formula-for-sum} gives
\begin{align}
\dim(\mathcal{U}\cap\mathcal{W}) =\dim(\mathcal{U})+\dim(\mathcal{W})-\dim(\mathcal{U}+\mathcal{W}) \geq \dim(\mathcal{U})+\dim(\mathcal{W})-\dim(\mathcal{V}) \nonumber
\end{align}
If $\dim(\mathcal{U})+\dim(\mathcal{W})>\dim(\mathcal{V})$, the right-hand side is at least $1$, so the intersection contains a nonzero vector. If $\dim(\mathcal{U})+\dim(\mathcal{W})\geq\dim(\mathcal{V})+k$, then $\dim(\mathcal{U}\cap\mathcal{W})\geq k$, so intersection contains $k$ linearly independent vectors.\\
\hlt{Structural Commentary}: Excess dimension appears as overlap.\\
\hlt{Key Insight}: Two large subspaces cannot fit inside $\mathcal{V}$ without sharing directions.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:complements-exist-finite-dimensions} \hlt{Complements Exist in Finite Dimensions}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U} \subseteq \mathcal{V}$ is a subspace, then there exists a subspace $\mathcal{W}$ such that $\mathcal{V} = \mathcal{U} \oplus \mathcal{W}$.
\begin{proof}
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ of $\mathcal{U}$ and extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_n)$ of $\mathcal{V}$.\\
Let $\mathcal{W}=\operatorname{span}(\mathbf{w}_1,\ldots,\mathbf{w}_n)$. Every vector in $\mathcal{V}$ is a linear combination of this extended basis, so $\mathcal{V}=\mathcal{U}+\mathcal{W}$. If $\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$, then $\mathbf{v}$ has one representation using only $\mathbf{u}_1,\ldots,\mathbf{u}_m$ and another using only $\mathbf{w}_1,\ldots,\mathbf{w}_n$. Uniqueness of coefficients in the basis of $\mathcal{V}$ forces $\mathbf{v}=\mathbf{0}$. Hence $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.\\
\hlt{Structural Commentary}: Complements split a space into a subspace and the remaining basis directions.\\
\hlt{Key Insight}: Extend a basis of $\mathcal{U}$ to split $\mathcal{V}$.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Affine Subset and Parallel Subspaces}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$. For $\mathbf{v}\in\mathcal{V}$, the subset $\mathbf{v}+\mathcal{U}\defeq\{\mathbf{v}+\mathbf{u}:\mathbf{u}\in\mathcal{U}\}$ is an \hlt{affine subset} of $\mathcal{V}$ parallel to $\mathcal{U}$.
\end{definition}

\begin{proposition} \label{prop:parallel-affine-subsets-equal-or-disjoint} \hlt{Parallel Affine Subsets Are Equal or Disjoint}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$ and let $\mathbf{v},\mathbf{w}\in\mathcal{V}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathbf{v}-\mathbf{w}\in\mathcal{U}$;
\item $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$;
\item $(\mathbf{v}+\mathcal{U})\cap(\mathbf{w}+\mathcal{U})\neq\emptyset$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman)*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $\mathbf{v}-\mathbf{w}\in\mathcal{U}$ and $\mathbf{u}\in\mathcal{U}$, then $\mathbf{v}+\mathbf{u}=\mathbf{w}+((\mathbf{v}-\mathbf{w})+\mathbf{u})\in\mathbf{w}+\mathcal{U}$. Thus $\mathbf{v}+\mathcal{U}\subseteq\mathbf{w}+\mathcal{U}$, and the reverse inclusion is similar.
\item[$(ii)$]$\Rightarrow (iii)$ Equal affine subsets have the same elements.
\item[$(iii)$]$\Rightarrow (i)$ If $\mathbf{x}\in(\mathbf{v}+\mathcal{U})\cap(\mathbf{w}+\mathcal{U})$, then $\mathbf{x}=\mathbf{v}+\mathbf{u}_1=\mathbf{w}+\mathbf{u}_2$ for some $\mathbf{u}_1,\mathbf{u}_2\in\mathcal{U}$. Hence $\mathbf{v}-\mathbf{w}=\mathbf{u}_2-\mathbf{u}_1\in\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: Parallel affine subsets either coincide or do not meet.\\
\hlt{Key Insight}: Intersection is controlled by the difference of base points.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Quotient Space}\\
Let $\mathcal{U} \subseteq \mathcal{V}$. The \hlt{quotient space} $\mathcal{V}/\mathcal{U}$ is the set of all affine subsets of $\mathcal{V}$ parallel to $\mathcal{U}$ $\mathcal{V}/\mathcal{U}\defeq\{\mathbf{v}+\mathcal{U}:\mathbf{v}\in\mathcal{V}\}$.
\end{definition}

\begin{proposition} \label{prop:quotient-space-structure} \hlt{Structure of the Quotient}\\
Addition and scalar multiplication on $\mathcal{V}/\mathcal{U}$ defined by $(\mathbf{v} + \mathcal{U}) + (\mathbf{w} + \mathcal{U}) \defeq (\mathbf{v} + \mathbf{w}) + \mathcal{U}$, $a(\mathbf{v} + \mathcal{U}) \defeq a\mathbf{v} + \mathcal{U}$ are well-defined, and with these operations $\mathcal{V}/\mathcal{U}$ is a vector space.
\begin{proof}
If $\mathbf{v} + \mathcal{U} = \mathbf{v}' + \mathcal{U}$ and $\mathbf{w} + \mathcal{U} = \mathbf{w}' + \mathcal{U}$, then $\mathbf{v}-\mathbf{v}' \in \mathcal{U}$ and $\mathbf{w}-\mathbf{w}' \in \mathcal{U}$.\\
Since $\mathcal{U}$ is a subspace, $(\mathbf{v}+\mathbf{w}) - (\mathbf{v}'+\mathbf{w}') \in \mathcal{U}$, and $a\mathbf{v} - a\mathbf{v}' \in \mathcal{U}$ so the operations do not depend on the chosen representatives. The vector space axioms are inherited from $\mathcal{V}$ under these operations.\\
\hlt{Structural Commentary}: Quotients preserve operations modulo $\mathcal{U}$.\\
\hlt{Key Insight}: Cosets identify vectors differing by elements of $\mathcal{U}$.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Quotient Map}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$. The \hlt{quotient map} is the map $\pi : \mathcal{V} \to \mathcal{V}/\mathcal{U}$ defined by $\pi(\mathbf{v})=\mathbf{v}+\mathcal{U}$.
\end{definition}

\begin{proposition} \label{prop:quotient-map-properties} \hlt{Properties of the Quotient Map}\\
The quotient map $\pi$ is linear, surjective, and $\ker(\pi) = \mathcal{U}$.
\begin{proof}
For $\mathbf{v},\mathbf{w}\in\mathcal{V}$ and $a\in\mathbb{F}$, $\pi(a\mathbf{v}+\mathbf{w})=(a\mathbf{v}+\mathbf{w})+\mathcal{U} =a(\mathbf{v}+\mathcal{U})+(\mathbf{w}+\mathcal{U}) =a\pi(\mathbf{v})+\pi(\mathbf{w})$ so $\pi$ is linear. It is surjective because every coset in $\mathcal{V}/\mathcal{U}$ has the form $\mathbf{v}+\mathcal{U}=\pi(\mathbf{v})$. Finally, $\ker(\pi)=\{\mathbf{v}\in\mathcal{V}:\mathbf{v}+\mathcal{U}=\mathcal{U}\}=\mathcal{U}$.\\
\hlt{Structural Commentary}: The quotient map collapses $\mathcal{U}$ to zero and records cosets.\\
\hlt{Key Insight}: Quotient maps realise subspaces as kernels.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-quotient} \hlt{Dimension of a Quotient}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U}$ is a subspace, then $\dim(\mathcal{V}/\mathcal{U}) = \dim(\mathcal{V}) - \dim(\mathcal{U})$.
\begin{proof}
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ of $\mathcal{U}$ and extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_k)$ of $\mathcal{V}$.\\
The cosets $\mathbf{w}_1+\mathcal{U},\ldots,\mathbf{w}_k+\mathcal{U}$ span $\mathcal{V}/\mathcal{U}$ because the $\mathbf{u}_j$-terms vanish in the quotient.\\
If $a_1(\mathbf{w}_1+\mathcal{U})+\cdots+a_k(\mathbf{w}_k+\mathcal{U})=\mathcal{U}$, then $a_1\mathbf{w}_1+\cdots+a_k\mathbf{w}_k\in\mathcal{U}$, so a linear relation in the extended basis forces $a_1=\cdots=a_k=0$. Hence these cosets form a basis of $\mathcal{V}/\mathcal{U}$, and its length is $k=\dim(\mathcal{V})-\dim(\mathcal{U})$.\\
\hlt{Structural Commentary}: Quotiening removes the directions in $\mathcal{U}$ and keeps the added basis directions.\\
\hlt{Key Insight}: Passing to $\mathcal{V}/\mathcal{U}$ subtracts the collapsed dimension.
\end{proof}
\end{theorem}

This section develops two complementary ways to organise a vector space: build from smaller pieces by spans and direct sums, or collapse a chosen subspace by passing to a quotient. Bases and dimension measure exactly what is added, split off, or made invisible.
