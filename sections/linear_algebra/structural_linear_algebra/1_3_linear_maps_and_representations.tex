\subsection{Linear Maps and Representations}

% **Role:** algebraic language of linear structure.
%
% Linear maps, kernels, ranges, rank-nullity, isomorphisms, matrices of maps, change of basis, similarity.

Linear maps preserve vector-space structure. Kernels, ranges, rank-nullity, matrices, change of basis, and similarity describe them without making coordinates primary.

\begin{definition} \label{def:linear-map} \hlt{Linear Map}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be vector spaces over the same field $\mathbb{F}$.\\
A function $T:\mathcal{V}\to\mathcal{W}$ is \hlt{linear} if $T(a\mathbf{u}+b\mathbf{v})=aT\mathbf{u}+bT\mathbf{v}$ for all $\mathbf{u},\mathbf{v}\in\mathcal{V}$ and $a,b\in\mathbb{F}$.\\
The set of all linear maps from $\mathcal{V}$ to $\mathcal{W}$ is denoted $\LL(\mathcal{V},\mathcal{W})$. If $\mathcal{V}=\mathcal{W}$, elements of $\LL(\mathcal{V})$ are \hlt{operators} on $\mathcal{V}$.
\end{definition}

\begin{definition} \hlt{Zero and Identity Maps}\\
For vector spaces $\mathcal{V}$ and $\mathcal{W}$, the \hlt{zero map} from $\mathcal{V}$ to $\mathcal{W}$ is $0_{\mathcal{V},\mathcal{W}}:\mathcal{V}\to\mathcal{W}$, where $0_{\mathcal{V},\mathcal{W}}\mathbf{v}=\mathbf{0}_{\mathcal{W}}$ for all $\mathbf{v}\in\mathcal{V}$.\\
The \hlt{identity map} on $\mathcal{V}$ is $I_{\mathcal{V}}:\mathcal{V}\to\mathcal{V}$, where $I_{\mathcal{V}}\mathbf{v}=\mathbf{v}$ for all $\mathbf{v}\in\mathcal{V}$.
\end{definition}

\begin{proposition} \label{prop:linear-maps-send-zero-to-zero} \hlt{Linear Maps Send Zero to Zero}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$, then $T\mathbf{0}_{\mathcal{V}}=\mathbf{0}_{\mathcal{W}}$.
\begin{proof}
Linearity gives $T\mathbf{0}_{\mathcal{V}}=T(\mathbf{0}_{\mathcal{V}}+\mathbf{0}_{\mathcal{V}})=T\mathbf{0}_{\mathcal{V}}+T\mathbf{0}_{\mathcal{V}}$. Adding $-T\mathbf{0}_{\mathcal{V}}$ to both sides gives $T\mathbf{0}_{\mathcal{V}}=\mathbf{0}_{\mathcal{W}}$.\\
\hlt{Structural Commentary}: Linearity preserves zero.\\
\hlt{Key Insight}: A linear map cannot move zero away from zero.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Operations on Linear Maps}\\
For $S,T\in\LL(\mathcal{V},\mathcal{W})$ and $a\in\mathbb{F}$, define pointwise $(S+T)\mathbf{v}\defeq S\mathbf{v}+T\mathbf{v}$ and $(aT)\mathbf{v}\defeq a(T\mathbf{v})$ for every $\mathbf{v}\in\mathcal{V}$.
\end{definition}

\begin{proposition} \label{prop:linear-maps-form-vector-space} \hlt{Linear Maps Form a Vector Space}\\
With the operations above, $\LL(\mathcal{V},\mathcal{W})$ is a vector space over $\mathbb{F}$.
\begin{proof}
If $S,T\in\LL(\mathcal{V},\mathcal{W})$ and $a,c\in\mathbb{F}$, then for $\mathbf{u},\mathbf{v}\in\mathcal{V}$,
\begin{align}
(aS+T)(c\mathbf{u}+\mathbf{v})
=a(cS\mathbf{u}+S\mathbf{v})+(cT\mathbf{u}+T\mathbf{v})
=c(aS+T)\mathbf{u}+(aS+T)\mathbf{v}. \nonumber
\end{align}
Thus $\LL(\mathcal{V},\mathcal{W})$ is closed under linear combinations. The zero map is the additive identity, $-T$ is the additive inverse of $T$, and the remaining vector-space axioms hold pointwise because they hold in $\mathcal{W}$.\\
\hlt{Structural Commentary}: Linear map operations are pointwise.\\
\hlt{Key Insight}: Pointwise sums and scalar multiples preserve linearity.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Product of Linear Maps}\\
For $T\in\LL(\mathcal{U},\mathcal{V})$ and $S\in\LL(\mathcal{V},\mathcal{W})$, the \hlt{product} $ST\in\LL(\mathcal{U},\mathcal{W})$ is defined by $(ST)\mathbf{u}\defeq S(T\mathbf{u})$ for all $\mathbf{u}\in\mathcal{U}$.
\end{definition}

\begin{proposition} \label{prop:linear-map-product-algebra} \hlt{Algebra of Linear Map Products}\\
If $T\in\LL(\mathcal{U},\mathcal{V})$ and $S\in\LL(\mathcal{V},\mathcal{W})$, then $ST\in\LL(\mathcal{U},\mathcal{W})$. Whenever defined, $(RS)T=R(ST)$, $TI_{\mathcal{U}}=T=I_{\mathcal{V}}T$, $(S_1+S_2)T=S_1T+S_2T$, and $S(T_1+T_2)=ST_1+ST_2$.
\begin{proof}
For $\mathbf{u},\mathbf{x}\in\mathcal{U}$ and $a\in\mathbb{F}$, $(ST)(a\mathbf{u}+\mathbf{x})=S(aT\mathbf{u}+T\mathbf{x})=aS(T\mathbf{u})+S(T\mathbf{x})=a(ST)\mathbf{u}+(ST)\mathbf{x}$, hence $ST\in\LL(\mathcal{U},\mathcal{W})$. For each remaining identity $A=B$ and admissible $\mathbf{z}$, $A\mathbf{z}=B\mathbf{z}$ by associativity of evaluation, identity maps, or distributivity in the target space; hence $A=B$.\\
\hlt{Structural Commentary}: Composition chains linear actions, while addition distributes through evaluation.\\
\hlt{Key Insight}: Linear map algebra is function composition plus pointwise vector-space algebra.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Graph of a Function}\\
For a function $T:\mathcal{V}\to\mathcal{W}$, the \hlt{graph} of $T$ is $\operatorname{graph}(T)\defeq\{(\mathbf{v},T\mathbf{v}):\mathbf{v}\in\mathcal{V}\}\subseteq\mathcal{V}\times\mathcal{W}$.
\end{definition}

\begin{proposition} \label{prop:graph-subspace-characterizes-linearity} \hlt{Linearity and Graphs}\\
A function $T:\mathcal{V}\to\mathcal{W}$ is linear if and only if $\operatorname{graph}(T)$ is a subspace of $\mathcal{V}\times\mathcal{W}$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $T$ is linear, then $(\mathbf{0},\mathbf{0})\in\operatorname{graph}(T)$.\\
If $(\mathbf{u},T\mathbf{u}),(\mathbf{v},T\mathbf{v})\in\operatorname{graph}(T)$ and $a\in\mathbb{F}$, then $a(\mathbf{u},T\mathbf{u})+(\mathbf{v},T\mathbf{v})=(a\mathbf{u}+\mathbf{v},T(a\mathbf{u}+\mathbf{v}))\in\operatorname{graph}(T)$.\\
Thus $\operatorname{graph}(T)$ is a subspace.
\item[$(\Leftarrow)$] If $\operatorname{graph}(T)$ is a subspace, then for $\mathbf{u},\mathbf{v}\in\mathcal{V}$ and $a\in\mathbb{F}$, the vector
$a(\mathbf{u},T\mathbf{u})+(\mathbf{v},T\mathbf{v})=(a\mathbf{u}+\mathbf{v},aT\mathbf{u}+T\mathbf{v})$
lies in $\operatorname{graph}(T)$. Hence $T(a\mathbf{u}+\mathbf{v})=aT\mathbf{u}+T\mathbf{v}$, so $T$ is linear.
\end{enumerate}
\hlt{Structural Commentary}: A graph records a function as a subset of a product space.\\
\hlt{Key Insight}: Closure of the graph is exactly linearity of the function.
\end{proof}
\end{proposition}

\begin{principle} \hlt{Coordinate-Free Principle}\\
A linear map is completely determined by what it does to a basis. Coordinates do not create the map; they only encode it once bases have been chosen.
\end{principle}

\begin{proposition} \label{prop:linear-map-determined-by-basis} \hlt{Linear Maps Are Determined by a Basis}\\
If $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$ and $\mathbf{w}_1, \ldots, \mathbf{w}_n \in \mathcal{W}$ are arbitrary, then $\exists$ a unique linear map $T : \mathcal{V} \to \mathcal{W}$ such that $T(\mathbf{v}_j) = \mathbf{w}_j$ for $j=1,\ldots,n$.
\begin{proof}
Every $\mathbf{v} \in \mathcal{V}$ has a unique expression $\mathbf{v} = a_1\mathbf{v}_1 + \cdots + a_n\mathbf{v}_n$. Define $T(\mathbf{v}) \defeq a_1\mathbf{w}_1 + \cdots + a_n\mathbf{w}_n$.\\
This is well-defined because basis coordinates are unique, and linearity follows from coordinate expansions. Any linear map with the prescribed basis values must agree with this formula, so it is unique.\\
\hlt{Structural Commentary}: A basis reduces a linear map to finitely many choices.\\
\hlt{Key Insight}: Once basis images are fixed, linearity determines every other value.
\end{proof}
\end{proposition}

\begin{definition} \label{def:kernel-and-range-linear-map} \hlt{Kernel and Range}\\
For $T\in\LL(\mathcal{V},\mathcal{W})$, define
$\ker(T)\defeq\{\mathbf{v}\in\mathcal{V}:T\mathbf{v}=\mathbf{0}_{\mathcal{W}}\}$ and
$\operatorname{range}(T)\defeq\{T\mathbf{v}:\mathbf{v}\in\mathcal{V}\}$.\\
The kernel is also called the \hlt{nullspace} of $T$.
\end{definition}

\begin{definition} \hlt{Injective and Surjective Linear Maps}\\
A function $T:\mathcal{V}\to\mathcal{W}$ is \hlt{injective} if $T\mathbf{u}=T\mathbf{v}$ implies $\mathbf{u}=\mathbf{v}$. It is \hlt{surjective} if $\operatorname{range}(T)=\mathcal{W}$.
\end{definition}

\begin{proposition} \label{prop:kernel-and-range-are-subspaces} \hlt{Kernel and Range Are Subspaces}\\
If $T \in \LL(\mathcal{V}, \mathcal{W})$, then $\ker(T)$ is a subspace of $\mathcal{V}$ and $\operatorname{range}(T)$ is a subspace of $\mathcal{W}$.
\begin{proof}
Since $T\mathbf{0}_{\mathcal{V}}=\mathbf{0}_{\mathcal{W}}$, the kernel is nonempty.\\
If $\mathbf{u},\mathbf{v}\in\ker(T)$ and $a\in\mathbb{F}$, then $T(a\mathbf{u}+\mathbf{v})=aT\mathbf{u}+T\mathbf{v}=\mathbf{0}_{\mathcal{W}}$, so $a\mathbf{u}+\mathbf{v}\in\ker(T)$.\\
For the range, if $T\mathbf{u},T\mathbf{v}\in\operatorname{range}(T)$ and $a\in\mathbb{F}$, then $aT\mathbf{u}+T\mathbf{v}=T(a\mathbf{u}+\mathbf{v})\in\operatorname{range}(T)$.\\
\hlt{Structural Commentary}: The kernel and range are the two canonical subspaces attached to a linear map.\\
\hlt{Key Insight}: Linearity makes both preimages of zero and images of vectors closed under linear combinations.
\end{proof}
\end{proposition}

\begin{definition} \label{def:projection-onto-subspace-along-complement} \hlt{Projection Along a Complement}\\
An operator $P\in\LL(\mathcal{V})$ is an \hlt{algebraic projection}, or \hlt{idempotent operator}, if $P^2=P$.\\
If $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, the \hlt{projection onto $\mathcal{U}$ along $\mathcal{W}$} is $\rho_{\mathcal{U},\mathcal{W}}\in\LL(\mathcal{V})$ where $\rho_{\mathcal{U},\mathcal{W}}(\mathbf{u}+\mathbf{w})\defeq\mathbf{u}$ for $\mathbf{u}\in\mathcal{U}$, $\mathbf{w}\in\mathcal{W}$.
\end{definition}

\begin{proposition} \label{prop:algebraic-projection-range-kernel-idempotent} \hlt{Algebraic Projections and Direct-Sum Splittings}\\
For $P\in\LL(\mathcal{V})$, $P$ is an algebraic projection if and only if $\mathcal{V}=\operatorname{range}(P)\oplus\ker(P)$ and $P=\rho_{\operatorname{range}(P),\ker(P)}$.\\
In particular, if $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$ and $\rho=\rho_{\mathcal{U},\mathcal{W}}$, then $\rho$ is an algebraic projection, $\operatorname{range}(\rho)=\mathcal{U}$, and $\ker(\rho)=\mathcal{W}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $P$ is an algebraic projection, so $P^2=P$. For every $\mathbf{v}\in\mathcal{V}$, $\mathbf{v}=P\mathbf{v}+(\mathbf{v}-P\mathbf{v})$, with $P\mathbf{v}\in\operatorname{range}(P)$ and $\mathbf{v}-P\mathbf{v}\in\ker(P)$. If $\mathbf{x}\in\operatorname{range}(P)\cap\ker(P)$, then $\mathbf{x}=P\mathbf{y}$ and $\mathbf{0}=P\mathbf{x}=P^2\mathbf{y}=\mathbf{x}$. Thus the sum is direct. Also $P$ fixes $\operatorname{range}(P)$ and kills $\ker(P)$, so $P=\rho_{\operatorname{range}(P),\ker(P)}$.
\item[$(\Leftarrow)$] Assume the right-hand side. For $\mathbf{v}=\mathbf{x}+\mathbf{y}$ with $\mathbf{x}\in\operatorname{range}(P)$ and $\mathbf{y}\in\ker(P)$, $P^2\mathbf{v}=P\mathbf{x}=\mathbf{x}=P\mathbf{v}$.\\
Thus $P$ is an algebraic projection. For $\rho_{\mathcal{U},\mathcal{W}}$, the formula $\rho_{\mathcal{U},\mathcal{W}}(\mathbf{u}+\mathbf{w})=\mathbf{u}$ gives range $\mathcal{U}$ and kernel $\mathcal{W}$, and the equivalence above shows that $\rho$ is an algebraic projection.
\end{enumerate}
\hlt{Structural Commentary}: Idempotents keep one summand and kill its complement.\\
\hlt{Key Insight}: Direct-sum coordinates turn a projection into $I\oplus0$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:complementary-algebraic-projections} \hlt{Complementary Algebraic Projections}\\
If $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, then $\rho_{\mathcal{U},\mathcal{W}}+\rho_{\mathcal{W},\mathcal{U}}=I_{\mathcal{V}}$, and $\rho_{\mathcal{U},\mathcal{W}}\rho_{\mathcal{W},\mathcal{U}} =\rho_{\mathcal{W},\mathcal{U}}\rho_{\mathcal{U},\mathcal{W}}=0$.
\begin{proof}
For $\mathbf{v}=\mathbf{u}+\mathbf{w}$, $\mathbf{u}\in\mathcal{U}$, $\mathbf{w}\in\mathcal{W}$, $(\rho_{\mathcal{U},\mathcal{W}}+\rho_{\mathcal{W},\mathcal{U}})\mathbf{v} =\mathbf{u}+\mathbf{w} =\mathbf{v}$,\\
and $\rho_{\mathcal{U},\mathcal{W}}\rho_{\mathcal{W},\mathcal{U}}\mathbf{v} =\rho_{\mathcal{U},\mathcal{W}}\mathbf{w} =\mathbf{0}$; $\rho_{\mathcal{W},\mathcal{U}}\rho_{\mathcal{U},\mathcal{W}}\mathbf{v} =\rho_{\mathcal{W},\mathcal{U}}\mathbf{u} =\mathbf{0}$. Thus the stated operator identities hold.\\
\hlt{Structural Commentary}: A direct-sum decomposition gives two complementary idempotents.\\
\hlt{Key Insight}: Projection onto one summand kills the other summand.
\end{proof}
\end{proposition}

\begin{definition} \label{def:algebraically-orthogonal-projections-and-resolutions} \hlt{Orthogonal Projections in the Algebraic Sense}\\
Two algebraic projections $\rho,\sigma\in\LL(\mathcal{V})$ are \hlt{algebraically orthogonal}, written $\rho\perp\sigma$, if $\rho\sigma=\sigma\rho=0$.\\
\hlt{Resolution of identity} on $\mathcal{V}$ is a finite family of algebraic projections $(\rho_i)_{i=1}^{k}$ with $\rho_i\perp\rho_j$ for $i\neq j$ and $\sum\limits_{i=1}^{k}\rho_i=I_{\mathcal{V}}$.
\end{definition}

\begin{theorem} \label{thm:resolution-of-identity-gives-direct-sum} \hlt{Resolution Ranges Form a Direct Sum}\\
If $(\rho_i)_{i=1}^{k}$ is a resolution of $I_{\mathcal{V}}$ and $R_i=\operatorname{range}(\rho_i)$, then $\mathcal{V}=\bigoplus\limits_{i=1}^{k}R_i$, and $\ker(\rho_i)=\bigoplus\limits_{j\neq i}R_j$.
\begin{proof}
Since $\sum\limits_{i=1}^{k}\rho_i=I_{\mathcal{V}}$, $\mathbf{v}=\sum\limits_{i=1}^{k}\rho_i\mathbf{v}\in\sum\limits_{i=1}^{k}R_i$.\\
If $\sum\limits_{i=1}^{k}\mathbf{x}_i=\mathbf{0}$ with $\mathbf{x}_i\in R_i$, then for each $1\leq m\leq k$, $\mathbf{0}=\rho_m\sum\limits_{i=1}^{k}\mathbf{x}_i=\mathbf{x}_m$, as $\rho_m$ fixes $R_m$ and kills $R_i$ for $i\neq m$.\\
Thus $\mathcal{V}=\bigoplus\limits_{i=1}^{k}R_i$. Also, for $\mathbf{v}=\sum\limits_{j=1}^{k}\mathbf{x}_j$ with $\mathbf{x}_j\in R_j$, $\rho_i\mathbf{v}=\mathbf{x}_i$, so $\ker(\rho_i)=\bigoplus\limits_{j\neq i}R_j$.\\
\hlt{Structural Commentary}: A resolution decomposes the identity into independent idempotent pieces.\\
\hlt{Key Insight}: Pairwise zero products are the operator form of directness.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:direct-sum-gives-resolution-of-identity} \hlt{Direct Sums Give Resolutions}\\
If $\mathcal{V}=\bigoplus_{i=1}^{k}\mathcal{U}_i$ and $\rho_i=\rho_{\mathcal{U}_i,\bigoplus_{j\neq i}\mathcal{U}_j}$, then $(\rho_i)_{i=1}^{k}$ is a resolution of $I_{\mathcal{V}}$.
\begin{proof}
For $\mathbf{v}=\sum_{i=1}^{k}\mathbf{u}_i$, $\mathbf{u}_i\in\mathcal{U}_i$, $\sum\limits_{i=1}^{k}\rho_i\mathbf{v} =\sum\limits_{i=1}^{k}\mathbf{u}_i =\mathbf{v}$. Thus $\sum\limits_{i=1}^{k}\rho_i=I_{\mathcal{V}}$.\\
If $i\neq j$, then $\rho_i\rho_j\mathbf{v}=\rho_i\mathbf{u}_j=\mathbf{0}$ and $\rho_j\rho_i\mathbf{v}=\rho_j\mathbf{u}_i=\mathbf{0}$, so $\rho_i\perp\rho_j$.\\
\hlt{Structural Commentary}: Direct-sum coordinates produce mutually annihilating coordinate projections.\\
\hlt{Key Insight}: Coordinate projections add to the identity and kill all other summands.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:algebra-of-projections} \hlt{Algebra of Projections}\\
Let $\rho,\sigma\in\LL(\mathcal{V})$ be algebraic projections and $\operatorname{char}(\mathbb{F})\neq2$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\rho+\sigma$ is an algebraic projection $\Longleftrightarrow \rho\perp\sigma$.\\
Then $\operatorname{range}(\rho+\sigma)=\operatorname{range}(\rho)\oplus\operatorname{range}(\sigma)$; $\ker(\rho+\sigma)=\ker(\rho)\cap\ker(\sigma)$
\item $\rho-\sigma$ is an algebraic projection $\Longleftrightarrow \rho\sigma=\sigma\rho=\sigma$.\\
Then $\operatorname{range}(\rho-\sigma)=\operatorname{range}(\rho)\cap\ker(\sigma)$; $\ker(\rho-\sigma)=\ker(\rho)\oplus\operatorname{range}(\sigma)$.
\item If $\rho\sigma=\sigma\rho$, then $\rho\sigma$ is algebraic projection and $\operatorname{range}(\rho\sigma)=\operatorname{range}(\rho)\cap\operatorname{range}(\sigma)$; $\ker(\rho\sigma)=\ker(\rho)+\ker(\sigma)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Put $A=\rho\sigma$, $B=\sigma\rho$, and $C=\rho\sigma\rho$. Since $(\rho+\sigma)^2=\rho+\sigma \Longleftrightarrow A+B=0$,\\
$A+B=0$ gives $A+C=\rho(A+B)=0$ and $C+B=(A+B)\rho=0$, hence $B=A$ and $2A=0$.\\
Thus $A=B=0$. The converse is immediate. Assume $\rho\perp\sigma$. Then $\operatorname{range}(\rho+\sigma)\subseteq\operatorname{range}(\rho)+\operatorname{range}(\sigma)$, while $(\rho+\sigma)(\mathbf{x}+\mathbf{y})=\mathbf{x}+\mathbf{y}$ for $\mathbf{x}\in\operatorname{range}(\rho)$, $\mathbf{y}\in\operatorname{range}(\sigma)$. Also $\operatorname{range}(\rho)\cap\operatorname{range}(\sigma)=\{\mathbf{0}\}$, since $\mathbf{z}\in\operatorname{range}(\rho)\cap\operatorname{range}(\sigma)$ and $\mathbf{z}=\sigma\mathbf{w}$ imply $\mathbf{z}=\rho\mathbf{z}=\rho\sigma\mathbf{w}=\mathbf{0}$. Finally, $(\rho+\sigma)\mathbf{v}=0 \Longleftrightarrow \rho\mathbf{v}=0=\sigma\mathbf{v}$.
\item Since $P^2=P\Longleftrightarrow (I_{\mathcal{V}}-P)^2=I_{\mathcal{V}}-P$, $\rho-\sigma$ is an algebraic projection $\Longleftrightarrow$ $(I_{\mathcal{V}}-\rho)+\sigma$ is an algebraic projection. By (i), this is equivalent to $(I_{\mathcal{V}}-\rho)\sigma=\sigma(I_{\mathcal{V}}-\rho)=0$, i.e. $\rho\sigma=\sigma\rho=\sigma$.\\
Assume $\rho\sigma=\sigma\rho=\sigma$ and set $Q=\rho-\sigma$. Then $Q\mathbf{v}\in\operatorname{range}(\rho)\cap\ker(\sigma)$; $Q\mathbf{x}=\mathbf{x}$, $\mathbf{x}\in\operatorname{range}(\rho)\cap\ker(\sigma)$, so $\operatorname{range}(Q)=\operatorname{range}(\rho)\cap\ker(\sigma)$. Also $Q$ kills both $\ker(\rho)$ and $\operatorname{range}(\sigma)$, and $\ker(\rho)\cap\operatorname{range}(\sigma)=\{\mathbf{0}\}$.\\
If $\mathbf{v}\in\ker(Q)$, then $\rho\mathbf{v}=\sigma\mathbf{v}$ and $\mathbf{v}=(\mathbf{v}-\rho\mathbf{v})+\sigma\mathbf{v} \in\ker(\rho)\oplus\operatorname{range}(\sigma)$. Hence $\ker(Q)=\ker(\rho)\oplus\operatorname{range}(\sigma)$.
\item If $\rho\sigma=\sigma\rho$, then $(\rho\sigma)^2=\rho^2\sigma^2=\rho\sigma$. Moreover $\mathbf{x}\in\operatorname{range}(\rho\sigma)$ $\Longleftrightarrow$ $\rho\mathbf{x}=\mathbf{x}=\sigma\mathbf{x}$,\\
so $\operatorname{range}(\rho\sigma)=\operatorname{range}(\rho)\cap\operatorname{range}(\sigma)$. Finally, $\ker(\rho)+\ker(\sigma)\subseteq\ker(\rho\sigma)$, and if $\rho\sigma\mathbf{v}=0$,\\
then $\mathbf{v}=(\mathbf{v}-\sigma\mathbf{v})+\sigma\mathbf{v}$; $\mathbf{v}-\sigma\mathbf{v}\in\ker(\sigma)$, $\sigma\mathbf{v}\in\ker(\rho)$. Thus $\ker(\rho\sigma)=\ker(\rho)+\ker(\sigma)$.
\end{enumerate}
\hlt{Structural Commentary}: Operator identities encode range-kernel geometry.\\
\hlt{Key Insight}: Idempotence is a rigid equation; expanding it forces the cross terms to encode the geometry.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:shift-operators-finitely-nonzero} \hlt{Shift Operators on Finitely Nonzero Sequences}\\
Define $R,S:\mathbb{F}^{(\mathbb{N})}\to\mathbb{F}^{(\mathbb{N})}$ by $R(x_1,x_2,\ldots)=(0,x_1,x_2,\ldots)$, $S(x_1,x_2,\ldots)=(x_2,x_3,\ldots)$. Then $R,S\in\LL(\mathbb{F}^{(\mathbb{N})})$.\\
Also, $\ker(R)=\{\mathbf{0}\}$, $\operatorname{range}(R)=\{(x_1,x_2,\ldots)\in\mathbb{F}^{(\mathbb{N})}:x_1=0\}$, $\ker(S)=\operatorname{span}((1,0,0,\ldots))$, $\operatorname{range}(S)=\mathbb{F}^{(\mathbb{N})}$.
\begin{proof}
Both $R$ and $S$ preserve finite support, so they map $\mathbb{F}^{(\mathbb{N})}$ into itself.\\
Since both maps act componentwise by shifting entries, $R(a\mathbf{x}+\mathbf{y})=aR\mathbf{x}+R\mathbf{y}$ and $S(a\mathbf{x}+\mathbf{y})=aS\mathbf{x}+S\mathbf{y}$ for all $\mathbf{x},\mathbf{y}\in\mathbb{F}^{(\mathbb{N})}$ and $a\in\mathbb{F}$. Thus $R$ and $S$ are linear.\\
The map $R$ has zero kernel as $R\mathbf{x}=\mathbf{0}$ forces every entry of $\mathbf{x}$ to be zero, and its range is exactly the set of finitely nonzero sequences with first entry zero. For $S$, $S\mathbf{x}=\mathbf{0}$ forces $x_2=x_3=\cdots=0$, so $\ker(S)=\operatorname{span}((1,0,0,\ldots))$. Finally, for any $(y_1,y_2,\ldots)\in\mathbb{F}^{(\mathbb{N})}$, we have $S(0,y_1,y_2,\ldots)=(y_1,y_2,\ldots)$, so $S$ is surjective.\\
\hlt{Structural Commentary}: Shift operators expose behavior that finite-dimensional maps cannot have.\\
\hlt{Key Insight}: The right shift is injective but not surjective, while the left shift is surjective but not injective.
\end{proof}
\end{proposition}

\begin{example} \label{ex:shift-operators-show-operator-algebra-noncommutative} \hlt{Shift Operators Show Noncommutativity}\\
For the shifts $R,S\in\LL(\mathbb{F}^{(\mathbb{N})})$ from Proposition \ref{prop:shift-operators-finitely-nonzero}, $SR=I_{\mathbb{F}^{(\mathbb{N})}}$ and $RS(x_1,x_2,\ldots)=(0,x_2,x_3,\ldots)$.\\
Hence $RS\neq SR$, for instance on $(1,0,0,\ldots)$. Therefore $\LL(\mathbb{F}^{(\mathbb{N})})$ is not commutative under composition.
\end{example}

\begin{proposition} \label{prop:injective-iff-kernel-zero} \hlt{Injectivity and the Kernel}\\
A linear map $T \in \LL(\mathcal{V}, \mathcal{W})$ is injective if and only if $\ker(T) = \{ \mathbf{0} \}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $T$ is injective and $\mathbf{v}\in\ker(T)$, then $T\mathbf{v}=\mathbf{0}=T\mathbf{0}$, so $\mathbf{v}=\mathbf{0}$.
\item[$(\Leftarrow)$] If $\ker(T)=\{\mathbf{0}\}$ and $T\mathbf{v}=T\mathbf{w}$, then $T(\mathbf{v}-\mathbf{w})=\mathbf{0}$. Hence $\mathbf{v}-\mathbf{w}\in\ker(T)$, so $\mathbf{v}=\mathbf{w}$.
\end{enumerate}
\hlt{Structural Commentary}: The kernel measures exactly where a linear map collapses vectors to zero.\\
\hlt{Key Insight}: Injectivity is equivalent to collapsing only the zero vector.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:range-spanned-by-images-of-basis} \hlt{Range from Basis Images}\\
Let $\mathcal{V}$ be finite-dimensional with basis $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then $\operatorname{range}(T)=\operatorname{span}(T\mathbf{v}_1,\ldots,T\mathbf{v}_n)$.\\
In particular, $\operatorname{range}(T)$ is finite-dimensional and $\dim(\operatorname{range}(T))\leq n$.
\begin{proof}
If $\mathbf{v}=c_1\mathbf{v}_1+\cdots+c_n\mathbf{v}_n$, then $T\mathbf{v}=c_1T\mathbf{v}_1+\cdots+c_nT\mathbf{v}_n$.\\
Thus every vector in $\operatorname{range}(T)$ lies in $\operatorname{span}(T\mathbf{v}_1,\ldots,T\mathbf{v}_n)$, and the reverse inclusion is immediate from the definition of range. The dimension bound follows because this spanning list has length $n$.\\
\hlt{Structural Commentary}: The range is generated by the images of any basis of the domain.\\
\hlt{Key Insight}: Linear maps carry a basis-spanning formula directly into a range-spanning formula.
\end{proof}
\end{theorem}

\begin{definition} \label{def:rank-and-nullity-linear-map} \hlt{Rank and Nullity}\\
For a finite-dimensional domain, the rank of $T$ is $\dim(\operatorname{range}(T))$ and the nullity of $T$ is $\dim(\ker(T))$.
\end{definition}

\begin{theorem} \label{thm:rank-nullity} \hlt{Rank-Nullity Theorem}\\
If $\mathcal{V}$ is finite-dim and $T \in \LL(\mathcal{V}, \mathcal{W})$, then $\operatorname{range}(T)$ is finite-dim and $\dim(\mathcal{V}) = \dim(\ker(T)) + \dim(\operatorname{range}(T))$.
\begin{proof}
By Proposition \ref{prop:kernel-and-range-are-subspaces} and Corollary \ref{cor:finite-dimensional-spaces-and-subspaces-have-bases}, $\ker(T)$ has a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$.\\
Extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_r)$ of $\mathcal{V}$ by Theorem \ref{thm:extending-independent-lists}.\\
If $\mathbf{v}=\sum\limits_{i=1}^m a_i\mathbf{u}_i+\sum\limits_{j=1}^r b_j\mathbf{w}_j$, then $T\mathbf{v}=\sum\limits_{j=1}^r b_jT\mathbf{w}_j$, because $T\mathbf{u}_i=\mathbf{0}$. Hence $(T\mathbf{w}_1,\ldots,T\mathbf{w}_r)$ spans $\operatorname{range}(T)$.\\
If $\sum\limits_{j=1}^r c_jT\mathbf{w}_j=\mathbf{0}$, then $\sum\limits_{j=1}^r c_j\mathbf{w}_j\in\ker(T)$, so for some scalars $d_i$, $\sum\limits_{j=1}^r c_j\mathbf{w}_j=\sum\limits_{i=1}^m d_i\mathbf{u}_i$.\\
Thus $\sum\limits_{j=1}^r c_j\mathbf{w}_j-\sum\limits_{i=1}^m d_i\mathbf{u}_i=\mathbf{0}$, and linear independence of extended basis forces every $c_j$ and $d_i$ to be zero.\\
Hence $(T\mathbf{w}_1,\ldots,T\mathbf{w}_r)$ is a basis of $\operatorname{range}(T)$. Thus $\dim(\ker(T))=m$, $\dim(\operatorname{range}(T))=r$, and $\dim(\mathcal{V})=m+r$.\\
\hlt{Structural Commentary}: An adapted basis splits $\mathcal{V}$ into kernel directions and directions detected by $T$.\\
\hlt{Key Insight}: Rank-nullity says every domain dimension is either killed by $T$ or appears in its range.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:map-to-smaller-space-not-injective} \hlt{Maps to Smaller Spaces Are Not Injective}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional vector spaces with $\dim(\mathcal{V})>\dim(\mathcal{W})$.\\
Then no linear map from $\mathcal{V}$ to $\mathcal{W}$ is injective.
\begin{proof}
Let $T\in\LL(\mathcal{V},\mathcal{W})$. By Theorem \ref{thm:rank-nullity}, $\dim(\ker(T))=\dim(\mathcal{V})-\dim(\operatorname{range}(T)) \geq \dim(\mathcal{V})-\dim(\mathcal{W})>0$.\\
Thus $\ker(T)$ contains a nonzero vector, so $T$ is not injective by Proposition \ref{prop:injective-iff-kernel-zero}.\\
\hlt{Structural Commentary}: Excess domain dimension forces collapse.\\
\hlt{Key Insight}: The dimension deficit must appear inside the kernel.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:map-to-larger-space-not-surjective} \hlt{Maps to Larger Spaces Are Not Surjective}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional vector spaces with $\dim(\mathcal{V})<\dim(\mathcal{W})$.\\
Then no linear map from $\mathcal{V}$ to $\mathcal{W}$ is surjective.
\begin{proof}
Let $T\in\LL(\mathcal{V},\mathcal{W})$. By Theorem \ref{thm:rank-nullity}, $\dim(\operatorname{range}(T))=\dim(\mathcal{V})-\dim(\ker(T))\leq\dim(\mathcal{V})<\dim(\mathcal{W})$.\\
Hence $\operatorname{range}(T)$ is a proper subspace of $\mathcal{W}$, so $T$ is not surjective.\\
\hlt{Structural Commentary}: A smaller domain cannot cover a larger codomain.\\
\hlt{Key Insight}: The range can never have dimension larger than the domain.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:equal-dimension-kernel-range} \hlt{Equal-Dimension Kernel-Range Criterion}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional vector spaces with $\dim(\mathcal{V})=\dim(\mathcal{W})$, and let $T\in\LL(\mathcal{V},\mathcal{W})$.\\
Then $\ker(T)=\{\mathbf{0}\}$ if and only if $\operatorname{range}(T)=\mathcal{W}$. Equivalently, $T$ is injective if and only if $T$ is surjective.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\ker(T)=\{\mathbf{0}\}$, then rank-nullity gives $\dim(\operatorname{range}(T))=\dim(\mathcal{V})=\dim(\mathcal{W})$.\\
Since $\operatorname{range}(T)$ is a subspace of $\mathcal{W}$ with the same dimension, $\operatorname{range}(T)=\mathcal{W}$.
\item[$(\Leftarrow)$] If $\operatorname{range}(T)=\mathcal{W}$, then rank-nullity gives $\dim(\ker(T))=\dim(\mathcal{V})-\dim(\mathcal{W})=0$, hence $\ker(T)=\{\mathbf{0}\}$.
\end{enumerate}
\hlt{Structural Commentary}: Equal finite dimensions make injectivity and surjectivity equivalent.\\
\hlt{Key Insight}: In equal dimensions, having no kernel loss is exactly having full range.
\end{proof}
\end{corollary}

\begin{definition} \hlt{Invertible Linear Map and Isomorphism}\\
A linear map $T \in \LL(\mathcal{V}, \mathcal{W})$ is \hlt{invertible} if there exists $S \in \LL(\mathcal{W}, \mathcal{V})$ such that $ST = I_{\mathcal{V}}$ and $TS = I_{\mathcal{W}}$.\\
An invertible linear map is called an \hlt{isomorphism}, and if one exists we write $\mathcal{V} \cong \mathcal{W}$.
\end{definition}

\begin{proposition} \label{prop:inverse-linear-map-is-unique} \hlt{Inverse Is Unique}\\
An invertible linear map has a unique inverse. Thus if $T$ is invertible, its inverse is denoted $T^{-1}$.
\begin{proof}
Suppose $S_1,S_2\in\LL(\mathcal{W},\mathcal{V})$ are inverses of $T\in\LL(\mathcal{V},\mathcal{W})$.\\
Then $S_1 = S_1I_{\mathcal{W}} = S_1(TS_2) = (S_1T)S_2 = I_{\mathcal{V}}S_2 = S_2$. Hence the inverse is unique.\\
\hlt{Structural Commentary}: A two-sided inverse is forced by the composition identities.\\
\hlt{Key Insight}: Associativity lets the two candidate inverses meet around $T$ and collapse to the same map.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:inverse-of-product-linear-maps} \hlt{Inverse of a Product Reverses Order}\\
Let $T\in\LL(\mathcal{U},\mathcal{V})$ and $S\in\LL(\mathcal{V},\mathcal{W})$ be invertible. Then $ST$ is invertible and $(ST)^{-1}=T^{-1}S^{-1}$.
\begin{proof}
Since $T^{-1}\in\LL(\mathcal{V},\mathcal{U})$ and $S^{-1}\in\LL(\mathcal{W},\mathcal{V})$, the product $T^{-1}S^{-1}\in\LL(\mathcal{W},\mathcal{U})$ is defined.\\
Moreover, $(T^{-1}S^{-1})(ST)=T^{-1}(S^{-1}S)T=T^{-1}T=I_{\mathcal{U}}$ and $(ST)(T^{-1}S^{-1})=S(TT^{-1})S^{-1}=SS^{-1}=I_{\mathcal{W}}$.\\
Thus $T^{-1}S^{-1}$ is a two-sided inverse of $ST$, so $(ST)^{-1}=T^{-1}S^{-1}$.\\
\hlt{Structural Commentary}: To undo a composition, undo the last map first.\\
\hlt{Key Insight}: Composition order reverses under inversion.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:linear-map-invertible-iff-bijective} \hlt{Invertibility and Bijectivity}\\
A linear map is invertible if and only if it is both injective and surjective.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Let $S$ be an inverse of $T$. If $T\mathbf{v}=T\mathbf{w}$, then $\mathbf{v}=ST\mathbf{v}=ST\mathbf{w}=\mathbf{w}$, so $T$ is injective. Also every $\mathbf{w}$ in the codomain satisfies $\mathbf{w}=T(S\mathbf{w})$, so $T$ is surjective.
\item[$(\Leftarrow)$] Suppose $T$ is injective and surjective. Define $S\mathbf{w}$ to be the unique vector $\mathbf{v}$ with $T\mathbf{v}=\mathbf{w}$.\\
Since both sides below map under $T$ to $a\mathbf{w}_1+\mathbf{w}_2$, $S(a\mathbf{w}_1+\mathbf{w}_2)=aS\mathbf{w}_1+S\mathbf{w}_2$. Thus $S$ is linear and satisfies $ST=I$ and $TS=I$, so $T$ is invertible.
\end{enumerate}
\hlt{Structural Commentary}: Bijectivity gives a unique reverse map, and linearity makes it a linear inverse.\\
\hlt{Key Insight}: A linear map is invertible exactly when it is one-to-one and onto.
\end{proof}
\end{proposition}

\begin{definition} \label{def:linear-map-morphism-terminology-vector-spaces} \hlt{Morphism Terminology for Linear Maps}\\
For $T\in\LL(\mathcal{V},\mathcal{W})$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is a \hlt{homomorphism}.
\item If $\mathcal{V}=\mathcal{W}$, then $T$ is an \hlt{endomorphism}.
\item If $\ker(T)=\{\mathbf{0}\}$, then $T$ is a \hlt{monomorphism}.
\item If $\operatorname{range}(T)=\mathcal{W}$, then $T$ is an \hlt{epimorphism}.
\item If $\ker(T)=\{\mathbf{0}\}$ and $\operatorname{range}(T)=\mathcal{W}$, then $T$ is an \hlt{isomorphism}.
\item If $\mathcal{V}=\mathcal{W}$ and $T$ is invertible, then $T$ is an \hlt{automorphism}.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:rank-nullity-complement-view} \hlt{Rank-Nullity as a Complement Isomorphism}\\
Let $T\in\LL(\mathcal{V},\mathcal{W})$, and suppose $\mathcal{V}=\ker(T)\oplus\mathcal{M}$. Then $T|_{\mathcal{M}}:\mathcal{M}\to\operatorname{range}(T)$ is an isomorphism.
\begin{proof}
Indeed, $\ker(T|_{\mathcal{M}})=\mathcal{M}\cap\ker(T)=\{\mathbf{0}\}$, and every $T\mathbf{v}$ equals $T\mathbf{m}$ for the decomposition $\mathbf{v}=\mathbf{k}+\mathbf{m}$.\\
Thus rank-nullity counts $\mathcal{V}=\ker(T)\oplus\mathcal{M}$ with $\mathcal{M}\cong\operatorname{range}(T)$.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Quotient Map}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$. The \hlt{quotient map} is the map $\pi : \mathcal{V} \to \mathcal{V}/\mathcal{U}$ defined by $\pi(\mathbf{v})=\mathbf{v}+\mathcal{U}$.
\end{definition}

\begin{proposition} \label{prop:quotient-map-properties} \hlt{Properties of the Quotient Map}\\
The quotient map $\pi$ is linear, surjective, and $\ker(\pi) = \mathcal{U}$.
\begin{proof}
For $\mathbf{v},\mathbf{w}\in\mathcal{V}$ and $a\in\mathbb{F}$, $\pi(a\mathbf{v}+\mathbf{w})=(a\mathbf{v}+\mathbf{w})+\mathcal{U} =a(\mathbf{v}+\mathcal{U})+(\mathbf{w}+\mathcal{U}) =a\pi(\mathbf{v})+\pi(\mathbf{w})$ so $\pi$ is linear. It is surjective because every coset in $\mathcal{V}/\mathcal{U}$ has the form $\mathbf{v}+\mathcal{U}=\pi(\mathbf{v})$. Finally, $\ker(\pi)=\{\mathbf{v}\in\mathcal{V}:\mathbf{v}+\mathcal{U}=\mathcal{U}\}=\mathcal{U}$.\\
\hlt{Structural Commentary}: The quotient map collapses $\mathcal{U}$ to zero and records cosets.\\
\hlt{Key Insight}: Quotient maps realise subspaces as kernels.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:quotient-factorisation-through-projection} \hlt{Universal Property of the Quotient Map}\\
Let $\mathcal{U}\leq\mathcal{V}$, $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$, and $T\in\LL(\mathcal{V},\mathcal{W})$ with $\mathcal{U}\leq\ker(T)$.\\
Then $\exists!\,\overline{T}\in\LL(\mathcal{V}/\mathcal{U},\mathcal{W})
$ such that $\overline{T}\pi=T$, $\overline{T}(\mathbf{v}+\mathcal{U})=T\mathbf{v}$, and $\ker(\overline{T})=\ker(T)/\mathcal{U}$, $\operatorname{range}(\overline{T})=\operatorname{range}(T)$.
\begin{proof}
Define $\overline{T}(\mathbf{v}+\mathcal{U})=T\mathbf{v}$. If $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$, then $\mathbf{v}-\mathbf{w}\in\mathcal{U}\leq\ker(T)$, so $T\mathbf{v}=T\mathbf{w}$; hence $\overline{T}$ is well-defined.\\
Also $\overline{T}\big(a(\mathbf{v}+\mathcal{U})+b(\mathbf{w}+\mathcal{U})\big) =T(a\mathbf{v}+b\mathbf{w}) =aT\mathbf{v}+bT\mathbf{w}$.\\
By definition, $\overline{T}\pi=T$. If $S\pi=T$, then $S(\mathbf{v}+\mathcal{U})=S\pi(\mathbf{v})=T\mathbf{v}=\overline{T}(\mathbf{v}+\mathcal{U})$, so $S=\overline{T}$.\\
Also $\mathbf{v}+\mathcal{U}\in\ker(\overline{T}) \Longleftrightarrow T\mathbf{v}=\mathbf{0} \Longleftrightarrow \mathbf{v}\in\ker(T)$.\\
So $\ker(\overline{T})=\ker(T)/\mathcal{U}$, and $\operatorname{range}(\overline{T})=\{\overline{T}(\mathbf{v}+\mathcal{U}):\mathbf{v}\in\mathcal{V}\}=\operatorname{range}(T)$.\\
\hlt{Structural Commentary}: Maps killing $\mathcal{U}$ factor through $\mathcal{V}/\mathcal{U}$.\\
\hlt{Key Insight}: Quotients are universal domains for maps with prescribed kernel ambiguity.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:subspace-correspondence-quotient} \hlt{Subspace Correspondence for Quotients}\\
Let $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ be the quotient map. The assignments $\mathcal{T}\longmapsto \mathcal{T}/\mathcal{U}$ and $\mathcal{Y}\longmapsto \pi^{-1}(\mathcal{Y})$ are inverse inclusion-preserving bijections between $\{\mathcal{T}$ subspace of $\mathcal{V}:\mathcal{U}\subseteq\mathcal{T}\}$ and $\{\mathcal{Y}$ subspace of $\mathcal{V}/\mathcal{U}\}$.
\begin{proof}
If $\mathcal{U}\subseteq\mathcal{T}\subseteq\mathcal{V}$, then $\mathcal{T}/\mathcal{U}$ is a subspace of $\mathcal{V}/\mathcal{U}$.\\
If $\mathcal{Y}\subseteq\mathcal{V}/\mathcal{U}$ is a subspace, then $\pi^{-1}(\mathcal{Y})$ is a subspace of $\mathcal{V}$ and contains $\ker(\pi)=\mathcal{U}$.\\
For $\mathcal{U}\subseteq\mathcal{T}\subseteq\mathcal{V}$, $\pi^{-1}(\mathcal{T}/\mathcal{U})=\mathcal{T}$, since $\mathbf{v}+\mathcal{U}=\mathbf{t}+\mathcal{U}$ with $\mathbf{t}\in\mathcal{T}$ implies $\mathbf{v}-\mathbf{t}\in\mathcal{U}\subseteq\mathcal{T}$.\\
For $\mathcal{Y}\subseteq\mathcal{V}/\mathcal{U}$, surjectivity gives $\pi\left(\pi^{-1}(\mathcal{Y})\right)=\mathcal{Y}$, that is, $\pi^{-1}(\mathcal{Y})/\mathcal{U}=\mathcal{Y}$\\
Both assignments preserve inclusion by monotonicity of images and preimages.\\
\hlt{Structural Commentary}: Subspaces above $\mathcal{U}$ are exactly the subspaces visible after collapsing $\mathcal{U}$.\\
\hlt{Key Insight}: The quotient map translates containment above $\mathcal{U}$ into containment inside $\mathcal{V}/\mathcal{U}$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:quotient-by-kernel-is-range} \hlt{Quotient by the Kernel}\\
Let $T\in\LL(\mathcal{V},\mathcal{W})$. Then $\widetilde{T}(\mathbf{v}+\ker(T))\defeq T\mathbf{v}$ determines a well-defined linear map $\widetilde{T}:\mathcal{V}/\ker(T)\to\mathcal{W}$. Also,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\widetilde{T}$ is injective;
\item $\operatorname{range}(\widetilde{T})=\operatorname{range}(T)$;
\item $\mathcal{V}/\ker(T)\cong\operatorname{range}(T)$.
\end{enumerate}
\begin{proof}
If $\mathbf{v}+\ker(T)=\mathbf{v}'+\ker(T)$, then $\mathbf{v}-\mathbf{v}'\in\ker(T)$, so $T\mathbf{v}=T\mathbf{v}'$. Thus $\widetilde{T}$ is well-defined.\\
For cosets and $a\in\mathbb{F}$, $\widetilde{T}\big(a(\mathbf{v}+\ker(T))+(\mathbf{v}'+\ker(T))\big) =T(a\mathbf{v}+\mathbf{v}') =aT\mathbf{v}+T\mathbf{v}'$, so $\widetilde{T}$ is linear.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\widetilde{T}(\mathbf{v}+\ker(T))=\mathbf{0}$, then $T\mathbf{v}=\mathbf{0}$, so $\mathbf{v}\in\ker(T)$ and $\mathbf{v}+\ker(T)=\ker(T)$, the zero coset.\\
Hence $\ker(\widetilde{T})=\{\ker(T)\}$ and $\widetilde{T}$ is injective.
\item For every coset $\mathbf{v}+\ker(T)$, the value $\widetilde{T}(\mathbf{v}+\ker(T))=T\mathbf{v}$ lies in $\operatorname{range}(T)$.\\
Conversely, every vector in $\operatorname{range}(T)$ has the form $T\mathbf{v}=\widetilde{T}(\mathbf{v}+\ker(T))$. Hence the ranges are equal.
\item Let $\widehat{T}:\mathcal{V}/\ker(T)\to\operatorname{range}(T)$ be the same formula as $\widetilde{T}$ with codomain restricted to $\operatorname{range}(T)$.\\
The linearity proved above and parts (i)--(ii) show that $\widehat{T}$ is linear, injective, and surjective, so Proposition \ref{prop:linear-map-invertible-iff-bijective} gives $\mathcal{V}/\ker(T)\cong\operatorname{range}(T)$.
\end{enumerate}
\hlt{Structural Commentary}: Quotienting by the kernel removes exactly the ambiguity of $T$.\\
\hlt{Key Insight}: Every linear map factors as a quotient followed by an isomorphism onto its range.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:quotient-by-complement-is-complement} \hlt{Quotient by a Complemented Subspace}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$ with $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$. Then $J:\mathcal{W}\to\mathcal{V}/\mathcal{U}$, $J(\mathbf{w})=\mathbf{w}+\mathcal{U}$ is an isomorphism.\\
Hence every complement of $\mathcal{U}$ is isomorphic to $\mathcal{V}/\mathcal{U}$.
\begin{proof}
The map is linear by the quotient operations from Proposition \ref{prop:quotient-space-structure}.\\
If $J(\mathbf{w})=\mathcal{U}$, then $\mathbf{w}\in\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$, so $J$ is injective.\\
If $\mathbf{v}+\mathcal{U}\in\mathcal{V}/\mathcal{U}$, write $\mathbf{v}=\mathbf{u}+\mathbf{w}$ with $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{W}$. Then $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}=J(\mathbf{w})$, so $J$ is surjective.\\
\hlt{Structural Commentary}: A complement gives canonical representatives for quotient cosets.\\
\hlt{Key Insight}: Collapsing $\mathcal{U}$ leaves exactly the complementary directions visible.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:quotient-of-direct-sum-by-direct-sum} \hlt{Quotient of a Direct Sum}\\
If $\mathcal{V}=\mathcal{V}_1\oplus\mathcal{V}_2$ and $\mathcal{U}_j\leq\mathcal{V}_j$ for $j=1,2$, then $\frac{\mathcal{V}}{\mathcal{U}_1\oplus\mathcal{U}_2} \cong \frac{\mathcal{V}_1}{\mathcal{U}_1}\oplus \frac{\mathcal{V}_2}{\mathcal{U}_2}$.
\begin{proof}
Define, using the unique decomposition\\
$\mathbf{v}=\mathbf{v}_1+\mathbf{v}_2$,
$\Phi:\mathcal{V}\to (\mathcal{V}_1/\mathcal{U}_1)\oplus(\mathcal{V}_2/\mathcal{U}_2)$ and $\Phi(\mathbf{v}_1+\mathbf{v}_2) =(\mathbf{v}_1+\mathcal{U}_1,\mathbf{v}_2+\mathcal{U}_2)$.\\
For $\mathbf{x}=\mathbf{x}_1+\mathbf{x}_2$, $\mathbf{y}=\mathbf{y}_1+\mathbf{y}_2$, and $a,b\in\mathbb{F}$, $\Phi(a\mathbf{x}+b\mathbf{y}) =a\Phi(\mathbf{x})+b\Phi(\mathbf{y})$, so $\Phi$ is linear.\\
It is surjective since $(\mathbf{v}_1+\mathcal{U}_1,\mathbf{v}_2+\mathcal{U}_2)=\Phi(\mathbf{v}_1+\mathbf{v}_2)$. Finally, $\ker(\Phi)=\{\mathbf{v}_1+\mathbf{v}_2:\mathbf{v}_1\in\mathcal{U}_1,\ \mathbf{v}_2\in\mathcal{U}_2\} =\mathcal{U}_1\oplus\mathcal{U}_2$.\\
The result follows from Theorem \ref{thm:quotient-by-kernel-is-range}.\\
\hlt{Structural Commentary}: Quotienting respects direct-sum coordinates.\\
\hlt{Key Insight}: Direct-sum coordinates make quotienting componentwise.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:quotients-and-complements-are-not-functorial-warnings} \hlt{Quotient and Complement Warnings}\\
If $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, then $\mathcal{W}\cong\mathcal{V}/\mathcal{U}$, but complements are not functorial in $\mathcal{U}$.\\
For $\mathcal{V}$ countably infinite-dimensional with basis $(\mathbf{e}_n)_{n\geq1}$, put $\mathcal{S}=\operatorname{span}(\mathbf{e}_n:n\geq2)$.\\
Then $\mathcal{S}\cong\mathcal{V}$ and $\mathcal{V}=\mathcal{V}\oplus\{\mathbf{0}\} =\mathcal{S}\oplus\mathbb{F}\mathbf{e}_1$; $\mathcal{V}/\mathcal{V}=\{\mathbf{0}\}$, $\mathcal{V}/\mathcal{S}\cong\mathbb{F}\mathbf{e}_1$.
\end{perspective}

\begin{definition} \label{def:codimension-subspace} \hlt{Codimension}\\
If $\mathcal{U}$ is a subspace of $\mathcal{V}$ and $\dim(\mathcal{V}/\mathcal{U})<\infty$, define $\operatorname{codim}_{\mathcal{V}}(\mathcal{U})\defeq\dim(\mathcal{V}/\mathcal{U})$.\\
Thus, if $\mathcal{V}$ is finite-dimensional, Theorem \ref{thm:dimension-of-quotient} gives $\operatorname{codim}_{\mathcal{V}}(\mathcal{U})=\dim(\mathcal{V})-\dim(\mathcal{U})$.\\
If $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, Proposition \ref{prop:quotient-by-complement-is-complement} gives $\operatorname{codim}_{\mathcal{V}}(\mathcal{U})=\dim(\mathcal{W})$.
\end{definition}

\begin{perspective} \label{per:quotients-split-over-fields} \hlt{Quotients Split Over Fields}\\
In finite-dimensional vector spaces, choosing a complement $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$ identifies $\mathcal{V}/\mathcal{U}$ with $\mathcal{W}$. The quotient map collapses $\mathcal{U}$ and keeps the complementary directions.
\end{perspective}

\begin{proposition} \label{prop:finite-dimensional-one-sided-inverses-split} \hlt{One-Sided Inverses Split in Finite Dimensions}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional vector spaces and let $T\in\LL(\mathcal{V},\mathcal{W})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $T$ is injective, then there exists $L\in\LL(\mathcal{W},\mathcal{V})$ such that $LT=I_{\mathcal{V}}$.
\item If $T$ is surjective, then there exists $R\in\LL(\mathcal{W},\mathcal{V})$ such that $TR=I_{\mathcal{W}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $\mathcal{H}\subseteq\mathcal{W}$ with $\mathcal{W}=\operatorname{range}(T)\oplus\mathcal{H}$. For $\mathbf{w}=T\mathbf{v}+\mathbf{h}$, define $L\mathbf{w}=\mathbf{v}$. This is well-defined since the direct-sum decomposition is unique and $T$ is injective. It is linear, and $LT\mathbf{v}=\mathbf{v}$ for every $\mathbf{v}\in\mathcal{V}$.
\item Choose $\mathcal{H}\subseteq\mathcal{V}$ with $\mathcal{V}=\ker(T)\oplus\mathcal{H}$. Then $T|_{\mathcal{H}}:\mathcal{H}\to\mathcal{W}$ is injective since $\mathcal{H}\cap\ker(T)=\{\mathbf{0}\}$, and surjective since each $\mathbf{v}=\mathbf{k}+\mathbf{h}$ satisfies $T\mathbf{v}=T\mathbf{h}$. Let $R=(T|_{\mathcal{H}})^{-1}$, viewed as a map $\mathcal{W}\to\mathcal{V}$.\\
Then $R$ is linear and $TR=I_{\mathcal{W}}$.
\end{enumerate}
\hlt{Structural Commentary}: Complements supply the missing directions on which a one-sided inverse is chosen.\\
\hlt{Key Insight}: Finite-dimensional vector spaces split kernels and ranges.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:isomorphisms-preserve-bases} \hlt{Isomorphisms Preserve Bases}\\
Let $T:\mathcal{V}\to\mathcal{W}$ be an isomorphism and let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a finite list in $\mathcal{V}$. Set $T\beta=(T\mathbf{v}_1,\ldots,T\mathbf{v}_n)$.\\
The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\beta$ is a basis of $\mathcal{V}$.
\item $T\beta$ is a basis of $\mathcal{W}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $\mathbf{w}\in\mathcal{W}$, choose $\mathbf{v}\in\mathcal{V}$ with $\mathbf{w}=T\mathbf{v}$ and write $\mathbf{v}=\sum_j a_j\mathbf{v}_j$. Then $\mathbf{w}=\sum_j a_jT\mathbf{v}_j$, so $T\beta$ spans $\mathcal{W}$. If $\sum_j a_jT\mathbf{v}_j=\mathbf{0}$, applying $T^{-1}$ gives $\sum_j a_j\mathbf{v}_j=\mathbf{0}$; hence all $a_j=0$. Thus $T\beta$ is a basis.
\item[$(ii)$]$\Rightarrow (i)$
Assume $T\beta$ is a basis of $\mathcal{W}$. If $\mathbf{v}\in\mathcal{V}$, write $T\mathbf{v}=\sum_j a_jT\mathbf{v}_j$; applying $T^{-1}$ gives $\mathbf{v}=\sum_j a_j\mathbf{v}_j$, so $\beta$ spans $\mathcal{V}$. If $\sum_j a_j\mathbf{v}_j=\mathbf{0}$, applying $T$ gives $\sum_j a_jT\mathbf{v}_j=\mathbf{0}$; hence all $a_j=0$. Thus $\beta$ is linearly independent, and therefore a basis of $\mathcal{V}$.
\end{enumerate}
\hlt{Structural Commentary}: Isomorphisms preserve exactly the spanning and independence data that define a basis.\\
\hlt{Key Insight}: An isomorphism changes the ambient space, not the linear relations among vectors.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:finite-dimensional-operator-invertible-equivalences} \hlt{Finite-Dimensional Invertibility Criteria}\\
If $\mathcal{V}$ is finite-dimensional and $T \in \LL(\mathcal{V})$, then the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $T$ is invertible;
\item $T$ is injective;
\item $T$ is surjective.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman)*.]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$ follows from Proposition \ref{prop:linear-map-invertible-iff-bijective}.
\item[$(ii)$] $\Rightarrow (iii)$: If $T$ is injective, then $\ker(T)=\{\mathbf{0}\}$.\\
Rank-nullity gives $\dim(\operatorname{range}(T))=\dim(\mathcal{V})$, hence $\operatorname{range}(T)=\mathcal{V}$, so $T$ is surjective.
\item[$(iii)$] $\Rightarrow (i)$: If $T$ is surjective, rank-nullity gives $\dim(\ker(T))=0$, so $T$ is injective.\\
Thus $T$ is bijective, hence invertible by Proposition \ref{prop:linear-map-invertible-iff-bijective}.
\end{enumerate}
\hlt{Structural Commentary}: For operators on finite-dimensional spaces, kernel and range dimensions trade off exactly. Injectivity and surjectivity therefore force each other.\\
\hlt{Key Insight}: In equal finite dimensions, losing no information is the same as reaching the whole space.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:kernel-obstruction-to-invertibility} \hlt{Kernel Obstruction to Invertibility}\\
By Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences} and Proposition \ref{prop:injective-iff-kernel-zero}, a finite-dimensional operator is invertible exactly when its kernel is zero. Any nonzero $\mathbf{v}\in\ker T$ therefore blocks invertibility.
\end{perspective}

\begin{theorem} \label{thm:finite-dimensional-isomorphic-iff-same-dimension} \hlt{Isomorphism and Dimension}\\
Two finite-dimensional vector spaces are isomorphic if and only if they have the same dimension.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $T:\mathcal{V}\to\mathcal{W}$ is an isomorphism and $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ is a basis of $\mathcal{V}$, then Proposition \ref{prop:isomorphisms-preserve-bases} shows that $(T\mathbf{v}_1,\ldots,T\mathbf{v}_n)$ is a basis of $\mathcal{W}$.
\item[$(\Leftarrow)$] If both spaces have dimension $n$, choose bases $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ and $(\mathbf{w}_1,\ldots,\mathbf{w}_n)$.\\
By Proposition \ref{prop:linear-map-determined-by-basis}, there is a linear map $T$ with $T\mathbf{v}_j=\mathbf{w}_j$ for every $j$. Then $T(\sum a_j\mathbf{v}_j)=\sum a_j\mathbf{w}_j$, so $\ker(T)=\{\mathbf{0}\}$ and $\operatorname{range}(T)=\mathcal{W}$. Hence $T$ is an isomorphism.
\end{enumerate}
\hlt{Structural Commentary}: Dimension classifies finite-dimensional spaces up to isomorphism.\\
\hlt{Key Insight}: Matching bases give an isomorphism; any isomorphism carries bases to bases.
\end{proof}
\end{theorem}

\begin{definition} \label{def:matrices-and-matrix-operations} \hlt{Matrices and Matrix Operations}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Matrix space}: for $m,n\in\mathbb{Z}_{>0}$, $\operatorname{Mat}(m,n,\mathbb{F})=\{[A_{j,k}]_{1\leq j\leq m,1\leq k\leq n}:A_{j,k}\in\mathbb{F}\}$.
\item \hlt{Entries, rows, columns}: for $A=[A_{j,k}]\in\operatorname{Mat}(m,n,\mathbb{F})$,\\
$A_{j,k}$ is the $(j,k)$-entry, $A_{j,\cdot}$ is row $j$, and $A_{\cdot,k}$ is column $k$.
\item \hlt{Transpose}: for $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $A^t\in\operatorname{Mat}(n,m,\mathbb{F})$ is defined by $(A^t)_{k,j}=A_{j,k}$.
\item \hlt{Main diagonal}: for $A=[A_{j,k}]\in\operatorname{Mat}(m,n,\mathbb{F})$ and $r=\min(m,n)$, $\operatorname{diag}(A)=(A_{1,1},\ldots,A_{r,r})$.
\item \hlt{Symmetry}: for $A\in\operatorname{Mat}(n,n,\mathbb{F})$, $A$ is symmetric if $A^t=A$ and skew-symmetric if $A^t=-A$.
\item \hlt{Triangularity}: for $A\in\operatorname{Mat}(n,n,\mathbb{F})$, $A$ is upper triangular if $A_{j,k}=0$ for $j>k$,\\
lower triangular if $A_{j,k}=0$ for $j<k$, and diagonal if $A_{j,k}=0$ for $j\neq k$.
\item \hlt{Linear operations}: for $A,C\in\operatorname{Mat}(m,n,\mathbb{F})$ and $a\in\mathbb{F}$, $(A+C)_{j,k}=A_{j,k}+C_{j,k}$ and $(aA)_{j,k}=aA_{j,k}$.
\item \hlt{Matrix product}: for $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $B\in\operatorname{Mat}(n,p,\mathbb{F})$, $AB\in\operatorname{Mat}(m,p,\mathbb{F})$ is $(AB)_{j,k}\defeq\sum\limits_{r=1}^{n}A_{j,r}B_{r,k}$.
\item \hlt{Identity and inverse}: $I_n=[\delta_{j,k}]\in\operatorname{Mat}(n,n,\mathbb{F})$; for $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$, $B$ is a two-sided inverse of $A$ if $AB=I_n=BA$, and $A$ is invertible if such $B$ exists.
\end{enumerate}
\end{definition}

\begin{lemma} \label{lem:two-sided-inverse-uniqueness} \hlt{Uniqueness of Two-Sided Inverses}\\
Let $A,B,C\in\operatorname{Mat}(n,n,\mathbb{F})$. If $AB=I_n$ and $BC=I_n$, then $A=C$.
\begin{proof}
Using $BC=I_n$ and $AB=I_n$, $A=AI_n=A(BC)=(AB)C=I_nC=C$.\\
\hlt{Structural Commentary}: A common middle inverse forces the left and right inverses to coincide.\\
\hlt{Key Insight}: Associativity turns $AB=I_n=BC$ into $A=C$.
\end{proof}
\end{lemma}

\begin{definition} \hlt{Inverse Matrix Notation}\\
If $A\in\operatorname{Mat}(n,n,\mathbb{F})$ is invertible, Lemma \ref{lem:two-sided-inverse-uniqueness} shows that its two-sided inverse is unique.\\
This inverse is denoted $A^{-1}$, and the set of invertible matrices in $\operatorname{Mat}(n,n,\mathbb{F})$ is denoted $\operatorname{Mat}(n,n,\mathbb{F})^{\times}$.
\end{definition}

\begin{proposition} \label{prop:matrix-spaces-are-vector-spaces} \hlt{Matrix Spaces}\\
For positive integers $m,n$, $\operatorname{Mat}(m,n,\mathbb{F})$ is a vector space over $\mathbb{F}$ with dimension $mn$.
\begin{proof}
Entrywise addition and scalar multiplication satisfy the vector space axioms because $\mathbb{F}$ does. For $1\leq j\leq m$ and $1\leq k\leq n$, let $E_{j,k}$ be the matrix with $1$ in entry $(j,k)$ and $0$ elsewhere.\\
Every $A=[A_{j,k}]$ has the unique expansion $A=\sum_{j=1}^m\sum\limits_{k=1}^n A_{j,k}E_{j,k}$.\\
Thus the $mn$ matrices $E_{j,k}$ form a basis, so $\dim(\operatorname{Mat}(m,n,\mathbb{F}))=mn$.\\
\hlt{Structural Commentary}: A matrix is a finite array of independent scalar slots.\\
\hlt{Key Insight}: The standard entry matrices isolate those slots one at a time.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:matrix-product-entry-row-column} \hlt{Matrix Product Entries Are Row Times Column}\\
If $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $B\in\operatorname{Mat}(n,p,\mathbb{F})$, then $(AB)_{j,k}=A_{j,\cdot}B_{\cdot,k}$ for $1\leq j\leq m$ and $1\leq k\leq p$.
\begin{proof}
The row-column product $A_{j,\cdot}B_{\cdot,k}$ is exactly $\sum_{r=1}^n A_{j,r}B_{r,k}$, which is the definition of $(AB)_{j,k}$.\\
\hlt{Structural Commentary}: Matrix multiplication is assembled entry-by-entry from row-column pairings.\\
\hlt{Key Insight}: Each entry of $AB$ sees one row of $A$ and one column of $B$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:matrix-vector-product-column-combination} \hlt{Matrix-Vector Products as Column Combinations}\\
If $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{x}=(x_1,\ldots,x_n)^t\in\mathbb{F}^n$, then $A\mathbf{x}=x_1\mathbf{a}_1+\cdots+x_n\mathbf{a}_n$.
\begin{proof}
The $j$th entry of $A\mathbf{x}$ is $\sum_{k=1}^n A_{j,k}x_k$, which is also the $j$th entry of $x_1\mathbf{a}_1+\cdots+x_n\mathbf{a}_n$.\\
\hlt{Structural Commentary}: Multiplying by a column selects a linear combination of the columns of $A$.\\
\hlt{Key Insight}: The coordinates of $\mathbf{x}$ are precisely the column coefficients.
\end{proof}
\end{proposition}

\begin{definition} \label{def:column-space-matrix-rank-nullity} \hlt{Column Space, Matrix Rank, and Nullity}\\
Let $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(m,n,\mathbb{F})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{col}(A)\defeq\operatorname{span}(\mathbf{a}_1,\ldots,\mathbf{a}_n)\subseteq\mathbb{F}^m$, and $\operatorname{row}(A)\defeq\operatorname{col}(A^t)\subseteq\mathbb{F}^n$.
\item $\operatorname{rank}(A)\defeq\dim(\operatorname{col}(A))$; column rank is $\dim(\operatorname{col}(A))$, and row rank is $\dim(\operatorname{row}(A))$.
\item $A$ has \hlt{full rank} if $\operatorname{rank}(A)=\min(m,n)$, \hlt{full column rank} if $\operatorname{rank}(A)=n$, and \hlt{full row rank} if $\operatorname{rank}(A)=m$.
\item $\ker(A)\defeq\{\mathbf{x}\in\mathbb{F}^n:A\mathbf{x}=\mathbf{0}\}$ is the \hlt{nullspace}, and $\operatorname{nullity}(A)\defeq\dim(\ker(A))$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:linear-system-solvability-and-affine-solution-set} \hlt{Solvability and Complete Solution Sets}\\
Let $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(m,n,\mathbb{F})$ and let $\mathbf{b}\in\mathbb{F}^m$. Then $A\mathbf{x}=\mathbf{b}$ is solvable if and only if $\mathbf{b}\in\operatorname{col}(A)$.\\
If $\mathbf{x}_p$ is one solution, then every solution is of the form $\mathbf{x}=\mathbf{x}_p+\mathbf{x}_0$, where $\mathbf{x}_0\in\ker(A)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $A\mathbf{x}=\mathbf{b}$ for some $\mathbf{x}=(x_1,\ldots,x_n)^T$, then $\mathbf{b}=A\mathbf{x}=\sum\limits_{j=1}^n x_j\mathbf{a}_j\in\operatorname{col}(A)$.
\item[$(\Leftarrow)$] If $\mathbf{b}\in\operatorname{col}(A)$, then $\mathbf{b}=\sum_{j=1}^n x_j\mathbf{a}_j$ for some scalars $x_j$. Hence, for $\mathbf{x}=(x_1,\ldots,x_n)^T$, we have $A\mathbf{x}=\mathbf{b}$.
\end{enumerate}
Now suppose $A\mathbf{x}_p=\mathbf{b}$. For any $\mathbf{x}$, $A\mathbf{x}=\mathbf{b}\iff A(\mathbf{x}-\mathbf{x}_p)=\mathbf{0}\iff \mathbf{x}-\mathbf{x}_p\in\ker(A)$.\\
Thus every solution is exactly of the form $\mathbf{x}=\mathbf{x}_p+\mathbf{x}_0$, where $\mathbf{x}_0\in\ker(A)$.\\
\hlt{Structural Commentary}: A consistent system is an affine shift of $\ker(A)$ by any particular solution.\\
\hlt{Key Insight}: Column space decides existence; kernel decides nonuniqueness.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:matrix-product-by-columns} \hlt{Matrix Products by Columns}\\
If $A\in\operatorname{Mat}(m,r,\mathbb{F})$ and $B=[\mathbf{b}_1\ \cdots\ \mathbf{b}_n]\in\operatorname{Mat}(r,n,\mathbb{F})$, then $AB=[A\mathbf{b}_1\ \cdots\ A\mathbf{b}_n]$.
\begin{proof}
For each $k$, the $k$th column of $AB$ has entries $\sum_{\ell=1}^r A_{j,\ell}B_{\ell,k}$, which are exactly the entries of $A\mathbf{b}_k$.\\
\hlt{Structural Commentary}: Matrix products apply $A$ to columns of $B$.\\
\hlt{Key Insight}: The columns of $AB$ are obtained by applying $A$ to the columns of $B$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:matrix-product-outer-product-decomposition} \hlt{Matrix Products as Sums of Outer Products}\\
Let $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_r]\in\operatorname{Mat}(m,r,\mathbb{F})$, and write $B\in\operatorname{Mat}(r,n,\mathbb{F})$ by rows as
\begin{align}
B=
\begin{bmatrix}
\mathbf{b}_1^t\\
\vdots\\
\mathbf{b}_r^t
\end{bmatrix}.
\nonumber
\end{align}
Then $AB=\mathbf{a}_1\mathbf{b}_1^t+\cdots+\mathbf{a}_r\mathbf{b}_r^t$.
\begin{proof}
The $(j,k)$ entry of $\mathbf{a}_\ell\mathbf{b}_\ell^t$ is $A_{j,\ell}B_{\ell,k}$. Summing over $\ell=1,\ldots,r$ gives $\sum_{\ell=1}^r A_{j,\ell}B_{\ell,k}=(AB)_{j,k}$.\\
\hlt{Structural Commentary}: Column-row products decompose a matrix product into single column-row pieces.\\
\hlt{Key Insight}: Each shared index $\ell$ contributes one outer product.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:elementary-transpose-rules} \hlt{Elementary Transpose Rules}\\
Let $A,C\in\operatorname{Mat}(m,n,\mathbb{F})$ and let $a\in\mathbb{F}$. Then $(A^t)^t=A$; $(A+C)^t=A^t+C^t$; and $(aA)^t=aA^t$.
\begin{proof}
For all admissible indices, $((A^t)^t)_{j,k}=(A^t)_{k,j}=A_{j,k}$, so $(A^t)^t=A$.\\
Also $((A+C)^t)_{j,k}=(A+C)_{k,j}=A_{k,j}+C_{k,j}=(A^t+C^t)_{j,k}$ and $((aA)^t)_{j,k}=(aA)_{k,j}=aA_{k,j}=(aA^t)_{j,k}$.\\
\hlt{Structural Commentary}: Transpose swaps indices; entrywise algebra is unchanged.\\
\hlt{Key Insight}: Transpose is linear and involutive.
\end{proof}
\end{proposition}

\begin{example} \label{ex:matrix-space-symmetric-skew-direct-sum} \hlt{Symmetric and Skew-Symmetric Decomposition}\\
Assume $\operatorname{char}(\mathbb{F})\neq2$, $M=\operatorname{Mat}(n,n,\mathbb{F})$, let $\operatorname{Sym}_n(\mathbb{F})=\{A\in M:A^t=A\}$, $\operatorname{Skew}_n(\mathbb{F})=\{A\in M:A^t=-A\}$.\\
For the map $\tau:M\to M$ defined by $\tau(A)=A^t$, linearity follows from Proposition \ref{prop:elementary-transpose-rules}. Hence $\operatorname{Sym}_n(\mathbb{F})=\ker(\tau-I)$ and $\operatorname{Skew}_n(\mathbb{F})=\ker(\tau+I)$, so both are subspaces of $M$.\\
For every $A\in M$, $A=\frac12(A+A^t)+\frac12(A-A^t) \in\operatorname{Sym}_n(\mathbb{F})+\operatorname{Skew}_n(\mathbb{F})$.\\
Also $A\in\operatorname{Sym}_n(\mathbb{F})\cap\operatorname{Skew}_n(\mathbb{F})$ implies $A=A^t=-A$, hence $A=0$. Thus $M=\operatorname{Sym}_n(\mathbb{F})\oplus\operatorname{Skew}_n(\mathbb{F})$.\\
\hlt{Structural Commentary}: Transpose splits a matrix into its even and odd parts.\\
\hlt{Key Insight}: The directness of the split uses the ability to divide by $2$.
\end{example}

\begin{proposition} \label{prop:transpose-reverses-products} \hlt{Transpose Reverses Products}\\
If $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $B\in\operatorname{Mat}(n,p,\mathbb{F})$, then $(AB)^t=B^tA^t$.
\begin{proof}
For $1\leq j\leq p$ and $1\leq k\leq m$, $((AB)^t)_{j,k}=(AB)_{k,j}=\sum\limits_{\ell=1}^n A_{k,\ell}B_{\ell,j}=\sum\limits_{\ell=1}^n (B^t)_{j,\ell}(A^t)_{\ell,k}=(B^tA^t)_{j,k}$.\\
\hlt{Structural Commentary}: Transposition swaps the order of composition encoded by matrix multiplication.\\
\hlt{Key Insight}: Rows become columns, so the product order reverses.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:matrix-inverse-product-and-transpose} \hlt{Matrix Inverses and Transposes Reverse Products}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$ be invertible. Then $AB$, $A^t$ are invertible, and $(AB)^{-1}=B^{-1}A^{-1}$; $(A^t)^{-1}=(A^{-1})^t$.
\begin{proof}
First, $(B^{-1}A^{-1})(AB)=B^{-1}(A^{-1}A)B=B^{-1}B=I_n$; $(AB)(B^{-1}A^{-1})=A(BB^{-1})A^{-1}=AA^{-1}=I_n$.\\
Hence $(AB)^{-1}=B^{-1}A^{-1}$. By Proposition \ref{prop:transpose-reverses-products}, $(A^{-1})^tA^t=(AA^{-1})^t=I_n$; $A^t(A^{-1})^t=(A^{-1}A)^t=I_n$.\\
\hlt{Structural Commentary}: Inversion and transposition both reverse multiplication order.\\
\hlt{Key Insight}: Apply the defining inverse equations and reverse products.
\end{proof}
\end{corollary}

\begin{definition} \hlt{Coordinate Vector}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis of a finite-dimensional vector space $\mathcal{V}$.\\
If $\mathbf{v}=c_1\mathbf{v}_1+\cdots+c_n\mathbf{v}_n$, then the \hlt{$\beta$-coordinate vector} of $\mathbf{v}$ is
\begin{align}
[\mathbf{v}]_{\beta}\defeq
\begin{bmatrix}
c_1\\
\vdots\\
c_n
\end{bmatrix}
\in\mathbb{F}^n. \nonumber
\end{align}
The function $[\cdot]_{\beta}:\mathcal{V}\to\mathbb{F}^n$ is the \hlt{$\beta$-basis representation function}.
\end{definition}

\begin{proposition} \label{prop:coordinate-map-isomorphism} \hlt{Coordinate Maps Are Linear Bijections}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis of a finite-dimensional vector space $\mathcal{V}$.\\
Then the coordinate map $[\cdot]_{\beta}:\mathcal{V}\to\mathbb{F}^n$ is linear and bijective.
\begin{proof}
If $\mathbf{u}=a_1\mathbf{v}_1+\cdots+a_n\mathbf{v}_n$ and $\mathbf{v}=b_1\mathbf{v}_1+\cdots+b_n\mathbf{v}_n$, then for $c\in\mathbb{F}$,
\begin{align}
[c\mathbf{u}+\mathbf{v}]_{\beta} =
\begin{bmatrix}
ca_1+b_1\\
\vdots\\
ca_n+b_n
\end{bmatrix}
= c[\mathbf{u}]_{\beta}+[\mathbf{v}]_{\beta} \nonumber
\end{align}
Thus $[\cdot]_{\beta}$ is linear. It is injective because $[\mathbf{v}]_{\beta}=\mathbf{0}$ forces all basis coefficients of $\mathbf{v}$ to be zero, and it is surjective because every column $(c_1,\ldots,c_n)^t\in\mathbb{F}^n$ equals $[c_1\mathbf{v}_1+\cdots+c_n\mathbf{v}_n]_{\beta}$.\\
\hlt{Structural Commentary}: A basis converts abstract vectors into coordinate columns without losing structure.\\
\hlt{Key Insight}: Coordinates give a structure-preserving copy of the vector space, not extra information.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Matrix Representation}\\
Let $\beta = (\mathbf{v}_1, \ldots, \mathbf{v}_n)$ be a basis of $\mathcal{V}$ and $\gamma = (\mathbf{w}_1, \ldots, \mathbf{w}_m)$ a basis of $\mathcal{W}$. For $T \in \LL(\mathcal{V}, \mathcal{W})$, the matrix of $T$ relative to $(\beta,\gamma)$ is $[T]_{\gamma \leftarrow \beta} \defeq
\begin{bmatrix}
[T(\mathbf{v}_1)]_{\gamma} & \cdots & [T(\mathbf{v}_n)]_{\gamma}
\end{bmatrix}
\in \operatorname{Mat}(m,n,\mathbb{F})$.
\end{definition}

\begin{proposition} \label{prop:stacked-nullspaces-and-concatenated-column-spaces} \hlt{Stacks Intersect Kernels, Concatenations Sum Columns}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $B\in\operatorname{Mat}(p,n,\mathbb{F})$, $C\in\operatorname{Mat}(m,r,\mathbb{F})$, and $D\in\operatorname{Mat}(m,s,\mathbb{F})$.\\
Then $\ker\begin{bmatrix} A\\ B\end{bmatrix}=\ker(A)\cap\ker(B)$; and $\operatorname{col}\begin{bmatrix} C & D\end{bmatrix} =\operatorname{col}(C)+\operatorname{col}(D)$.
\begin{proof}
For $\mathbf{x}\in\mathbb{F}^n$, $\mathbf{x}\in\ker\begin{bmatrix}A\\ B\end{bmatrix} \iff \begin{bmatrix}A\\ B\end{bmatrix}\mathbf{x}=\mathbf{0} \iff A\mathbf{x}=\mathbf{0}\ \text{and}\ B\mathbf{x}=\mathbf{0} \iff \mathbf{x}\in\ker(A)\cap\ker(B)$.\\
For $\mathbf{y}\in\mathbb{F}^m$, $\mathbf{y}\in\operatorname{col}\begin{bmatrix}C&D\end{bmatrix} \iff \exists\,\mathbf{u}\in\mathbb{F}^r,\ \mathbf{v}\in\mathbb{F}^s$ such that $\mathbf{y}=\begin{bmatrix}C&D\end{bmatrix}\begin{bmatrix}\mathbf{u}\\ \mathbf{v}\end{bmatrix}$\\
$\iff \exists\,\mathbf{u},\mathbf{v}$ such that $\mathbf{y}=C\mathbf{u}+D\mathbf{v} \iff \mathbf{y}\in\operatorname{col}(C)+\operatorname{col}(D)$.\\
\hlt{Structural Commentary}: Stacking intersects constraints; concatenation adds column directions.\\
\hlt{Key Insight}: Stacks encode intersections of nullspaces, while concatenations encode sums of column spaces.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:matrix-representation-respects-linear-combinations} \hlt{Matrices Respect Linear Combinations of Maps}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis of $\mathcal{V}$ and $\gamma$ be a basis of $\mathcal{W}$.\\
 If $S,T\in\LL(\mathcal{V},\mathcal{W})$ and $a\in\mathbb{F}$, then $[aS+T]_{\gamma \leftarrow \beta}=a[S]_{\gamma \leftarrow \beta}+[T]_{\gamma \leftarrow \beta}$.
\begin{proof}
The $j$th column of $[aS+T]_{\gamma \leftarrow \beta}$ is $[(aS+T)\mathbf{v}_j]_{\gamma}=[aS\mathbf{v}_j+T\mathbf{v}_j]_{\gamma}=a[S\mathbf{v}_j]_{\gamma}+[T\mathbf{v}_j]_{\gamma}$, which is the $j$th column of $a[S]_{\gamma \leftarrow \beta}+[T]_{\gamma \leftarrow \beta}$.\\
\hlt{Structural Commentary}: Matrix representation records basis images column-by-column.\\
\hlt{Key Insight}: Linear combinations of maps become entrywise linear combinations of their matrices.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Linear Map Induced by a Matrix}\\
For $A\in\operatorname{Mat}(m,n,\mathbb{F})$, the \hlt{linear map induced by $A$} is the map $T_A:\mathbb{F}^n\to\mathbb{F}^m$ defined by $T_A(\mathbf{x})=A\mathbf{x}$.
\end{definition}

\begin{proposition} \label{prop:matrix-induced-map-linear} \hlt{Matrix-Induced Maps Are Linear}\\
If $A\in\operatorname{Mat}(m,n,\mathbb{F})$, then $T_A:\mathbb{F}^n\to\mathbb{F}^m$ is linear.
\begin{proof}
For $\mathbf{x},\mathbf{y}\in\mathbb{F}^n$ and $a\in\mathbb{F}$, $T_A(a\mathbf{x}+\mathbf{y})=A(a\mathbf{x}+\mathbf{y})=aA\mathbf{x}+A\mathbf{y}=aT_A(\mathbf{x})+T_A(\mathbf{y})$.\\
\hlt{Structural Commentary}: Matrix multiplication preserves addition and scalar multiplication.\\
\hlt{Key Insight}: Every matrix defines a canonical linear map between coordinate spaces.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:matrix-rank-nullity} \hlt{Matrix Rank-Nullity}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$. Then $\dim(\ker(A))+\dim(\operatorname{col}(A))=n$ and $\operatorname{nullity}(A)+\operatorname{rank}(A)=n$.\\
If $m=n$, then $\ker(A)=\{\mathbf{0}\}$ if and only if $\operatorname{col}(A)=\mathbb{F}^n$.
\begin{proof}
Apply Theorem \ref{thm:rank-nullity} to the induced linear map $T_A:\mathbb{F}^n\to\mathbb{F}^m$. By definition, $\ker(T_A)=\ker(A)$ and $\operatorname{range}(T_A)=\operatorname{col}(A)$, giving the displayed identities. When $m=n$, the remaining equivalence follows as:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\ker(A)=\{\mathbf{0}\}$, then $\dim(\operatorname{col}(A))=n$. Since $\operatorname{col}(A)\subseteq\mathbb{F}^n$, Proposition \ref{prop:dimension-bounds-and-basis-criteria} gives $\operatorname{col}(A)=\mathbb{F}^n$.
\item[$(\Leftarrow)$] If $\operatorname{col}(A)=\mathbb{F}^n$, then $\dim(\operatorname{col}(A))=n$, so the identity gives $\dim(\ker(A))=0$. Hence $\ker(A)=\{\mathbf{0}\}$.
\end{enumerate}
\hlt{Structural Commentary}: Matrix rank-nullity is ordinary rank-nullity written in coordinates.\\
\hlt{Key Insight}: Kernel loss plus column-space dimension equals the number of columns.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:row-rank-equals-column-rank} \hlt{Row Rank Equals Column Rank}\\
For $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $\dim(\operatorname{row}(A))=\dim(\operatorname{col}(A))$. Equivalently, $\operatorname{rank}(A^t)=\operatorname{rank}(A)$.
\begin{proof}
Let $r=\operatorname{rank}(A)$. If $r=0$, then $\operatorname{col}(A)=\{\mathbf{0}\}$, so every column of $A$ is zero and $A=0$; the result follows. Suppose $r>0$. Choose columns $\mathbf{a}_{j_1},\ldots,\mathbf{a}_{j_r}$ forming a basis of $\operatorname{col}(A)$, and set $B=[\mathbf{a}_{j_1}\ \cdots\ \mathbf{a}_{j_r}]\in\operatorname{Mat}(m,r,\mathbb{F})$.\\
For each column $\mathbf{a}_k$ of $A$, there is a unique $\mathbf{x}_k\in\mathbb{F}^r$ such that $\mathbf{a}_k=B\mathbf{x}_k$. With $X=[\mathbf{x}_1\ \cdots\ \mathbf{x}_n]\in\operatorname{Mat}(r,n,\mathbb{F})$, we have $A=BX$. Therefore $A^t=X^tB^t$, so every column of $A^t$ is a linear combination of the $r$ columns of $X^t$. Hence $\operatorname{rank}(A^t)\leq r=\operatorname{rank}(A)$. Applying the same argument to $A^t$ gives $\operatorname{rank}(A)\leq\operatorname{rank}(A^t)$.\\
\hlt{Structural Commentary}: Column rank controls row rank by factoring $A$ through a basis of its column space.\\
\hlt{Key Insight}: Rows and columns measure the same intrinsic rank, viewed from transposed coordinates.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:row-column-rank-different-hosts} \hlt{Row and Column Spaces Have Different Hosts}\\
For $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $\operatorname{row}(A)\subseteq\mathbb{F}^n$ and $\operatorname{col}(A)\subseteq\mathbb{F}^m$.\\
Hence, when $m\neq n$, Theorem \ref{thm:row-rank-equals-column-rank} asserts only $\dim\operatorname{row}(A)=\dim\operatorname{col}(A)$, not equality of subspaces.
\end{perspective}

\begin{theorem} \label{thm:matrix-four-fundamental-subspaces-dimensions} \hlt{Four Fundamental Matrix Subspaces}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and let $r=\operatorname{rank}(A)$. Then $\operatorname{row}(A),\ker(A) \subseteq \mathbb{F}^n$; $\operatorname{col}(A),\ker(A^t) \subseteq \mathbb{F}^m$,\\
and $\dim(\operatorname{col}(A))=\dim(\operatorname{row}(A))=r$; $\dim(\ker(A))=n-r$; $\dim(\ker(A^t))=m-r$.
\begin{proof}
The inclusions follow from the ambient dimensions of the rows and columns, and from the domains of $A$ and $A^t$. By definition, $\dim(\operatorname{col}(A))=r$; by Theorem \ref{thm:row-rank-equals-column-rank}, $\dim(\operatorname{row}(A))=r$.\\
Rank-nullity gives $\dim(\ker(A))=n-\operatorname{rank}(A)=n-r$. Since $\operatorname{rank}(A^t)=\dim(\operatorname{col}(A^t))=\dim(\operatorname{row}(A))=r$, rank-nullity for $A^t:\mathbb{F}^m\to\mathbb{F}^n$ gives $\dim(\ker(A^t))=m-\operatorname{rank}(A^t)=m-r$.\\
\hlt{Structural Commentary}: One rank determines all four fundamental dimensions.\\
\hlt{Key Insight}: Rank counts what survives; nullity counts what is killed.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:homogeneous-system-more-variables-than-equations} \hlt{Homogeneous Systems with More Variables Than Equations}\\
If $A\in\operatorname{Mat}(m,n,\mathbb{F})$ with $n>m$, then the homogeneous system $A\mathbf{x}=\mathbf{0}$ has a nonzero solution.
\begin{proof}
The induced map $T_A:\mathbb{F}^n\to\mathbb{F}^m$ has domain dimension larger than codomain dimension, so Corollary \ref{cor:map-to-smaller-space-not-injective} shows that $T_A$ is not injective.\\
Hence $\ker(A)=\ker(T_A)$ contains a nonzero vector $\mathbf{x}$, and this vector satisfies $A\mathbf{x}=\mathbf{0}$.\\
\hlt{Structural Commentary}: Extra variables force a nontrivial kernel.\\
\hlt{Key Insight}: A nontrivial kernel is exactly a nonzero homogeneous solution.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:inhomogeneous-system-more-equations-than-variables} \hlt{Inhomogeneous Systems with More Equations Than Variables}\\
If $A\in\operatorname{Mat}(m,n,\mathbb{F})$ with $m>n$, then there exists $\mathbf{b}\in\mathbb{F}^m$ such that $A\mathbf{x}=\mathbf{b}$ has no solution.
\begin{proof}
The induced map $T_A:\mathbb{F}^n\to\mathbb{F}^m$ has domain dimension smaller than codomain dimension, so Corollary \ref{cor:map-to-larger-space-not-surjective} shows that $T_A$ is not surjective. Choose $\mathbf{b}\in\mathbb{F}^m\setminus\operatorname{range}(T_A)$. Then no $\mathbf{x}\in\mathbb{F}^n$ satisfies $A\mathbf{x}=\mathbf{b}$.\\
\hlt{Structural Commentary}: Too few variables cannot reach every possible right-hand side.\\
\hlt{Key Insight}: A missing range vector becomes an inconsistent inhomogeneous system.
\end{proof}
\end{corollary}

\begin{definition} \label{def:elementary-row-operations-and-matrices} \hlt{Elementary Row Operations and Elementary Matrices}\\
For row symbols $R_i$, the \hlt{elementary row operations} are:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Row scaling}: $R_i\leftarrow cR_i$, $c\in\mathbb{F}^{\times}$;
\item \hlt{Row swap}: $R_i\leftrightarrow R_j$, $i\neq j$;
\item \hlt{Row replacement}: $R_i\leftarrow R_i+cR_j$, $i\neq j$, $c\in\mathbb{F}$.
\end{enumerate}
An \hlt{elementary matrix} is $E=\epsilon(I_m)$, where $\epsilon$ is one elementary row operation.
\end{definition}

\begin{proposition} \label{prop:elementary-matrices-perform-row-operations} \hlt{Elementary Matrices Perform Row Operations}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\epsilon$ is an elementary row operation and $E_{\epsilon}=\epsilon(I_m)$, then $E_{\epsilon}A=\epsilon(A)$.
\item $E_{\epsilon}^{-1}=E_{\epsilon^{-1}}$; hence every elementary matrix is invertible with elementary inverse.
\item If $F=E_{\epsilon}^t\in\operatorname{Mat}(n,n,\mathbb{F})$, then $AF=(\epsilon(A^t))^t$.\\
Hence right multiplication performs the corresponding column operation.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For each row index $r$, $\operatorname{row}_r(E_{\epsilon}A)=\operatorname{row}_r(E_{\epsilon})A$.\\
So the rows of $E_{\epsilon}A$ are exactly the $\epsilon$-linear combinations of the rows of $A$.
\item Since $\epsilon^{-1}$ is elementary, $E_{\epsilon^{-1}}E_{\epsilon} =E_{\epsilon^{-1}}\epsilon(I_m) =\epsilon^{-1}(\epsilon(I_m)) =I_m$, and similarly $E_{\epsilon}E_{\epsilon^{-1}}=I_m$.
\item By (i) and Proposition \ref{prop:transpose-reverses-products}, $(AF)^t=E_{\epsilon}A^t=\epsilon(A^t)$, so $AF=(\epsilon(A^t))^t$.
\end{enumerate}
\hlt{Structural Commentary}: Elementary matrices encode row and column operations by multiplication.\\
\hlt{Key Insight}: Row operations are left multiplication; column operations are right multiplication.
\end{proof}
\end{proposition}

\begin{definition} \label{def:row-equivalence-and-rref} \hlt{Row Equivalence and Reduced Row Echelon Form}\\
For $A,B\in\operatorname{Mat}(m,n,\mathbb{F})$, write $A\sim_{\mathrm{row}}B$ if $B=E_k\cdots E_1A$ for elementary $E_i\in\operatorname{Mat}(m,m,\mathbb{F})$ and some $k\geq0$.\\
A matrix $R=[r_{ij}]$ is in \hlt{reduced row echelon form} if its nonzero rows have pivot columns $p_1,\ldots,p_s$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item rows $1,\ldots,s$ are nonzero and rows $s+1,\ldots,m$ are zero;
\item $r_{i,p_i}=1$ and $r_{i,j}=0$ for $j<p_i$;
\item $p_1<\cdots<p_s$;
\item $r_{\ell,p_i}=0$ for all $\ell\neq i$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:rref-existence-uniqueness-and-elementary-factorization} \hlt{Reduced Row Echelon Form}\\
For every $A\in\operatorname{Mat}(m,n,\mathbb{F})$, there is a unique RREF matrix $\operatorname{rref}(A)$ with $A\sim_{\mathrm{row}}\operatorname{rref}(A)$.\\
If $R=\operatorname{rref}(A)$, then $R=E_k\cdots E_1A$ for elementary $E_i\in\operatorname{Mat}(m,m,\mathbb{F})$.
\begin{proof}
Gaussian elimination successively chooses a pivot column, swaps/scales to a pivot $1$, clears the pivot column, and continues on the remaining submatrix. Proposition \ref{prop:elementary-matrices-perform-row-operations} gives $R=E_k\cdots E_1A$ with $R$ in RREF.\\
For uniqueness, row operations preserve $\ker(A)=\{\mathbf{x}:A\mathbf{x}=\mathbf{0}\}$. For an RREF matrix, the pivot set and pivot-variable formulas are determined by this kernel.\\
Hence two row-equivalent RREF matrices have the same pivot columns and rows.\\
\hlt{Structural Commentary}: RREF is the canonical row-operation representative of a matrix.\\
\hlt{Key Insight}: Elimination gives existence; the solution space pins down the reduced representative.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:invertible-matrices-row-reduce-to-identity} \hlt{Invertible Matrices Row-Reduce to the Identity}\\
If $P\in\operatorname{Mat}(m,m,\mathbb{F})$ is invertible, then $\operatorname{rref}(P)=I_m$. Hence there are elementary matrices $E_1,\ldots,E_k\in\operatorname{Mat}(m,m,\mathbb{F})$ such that $E_k\cdots E_1P=I_m$.
\begin{proof}
Let $R=\operatorname{rref}(P)$. By Theorem \ref{thm:rref-existence-uniqueness-and-elementary-factorization}, $R=E_k\cdots E_1P$ for elementary matrices $E_i$. Since each $E_i$ is invertible by Proposition \ref{prop:elementary-matrices-perform-row-operations}, left multiplication by $E_k\cdots E_1$ preserves the kernel, so $\ker(R)=\ker(P)$. Because $P$ is invertible, $P\mathbf{x}=\mathbf{0}$ implies $\mathbf{x}=P^{-1}P\mathbf{x}=\mathbf{0}$; hence $\ker(R)=\{\mathbf{0}\}$.\\
Suppose the nonzero rows of $R$ have pivot columns $p_1<\cdots<p_s$. If $s<m$, choose a non-pivot column $j$. Set the non-pivot variable $x_j=1$, set all other non-pivot variables equal to $0$, and define the pivot variables recursively from $i=s$ down to $1$ so that the $i$th row equation is satisfied. This gives a nonzero $\mathbf{x}$ with $R\mathbf{x}=\mathbf{0}$, contradicting $\ker(R)=\{\mathbf{0}\}$. Thus $s=m$.\\
Since $p_1<\cdots<p_m$ are $m$ columns in an $m\times m$ matrix, $p_i=i$ for every $i$. The RREF conditions then force the $i$th pivot column to be the $i$th standard coordinate column, so $R=I_m$.
\end{proof}
\end{lemma}

\begin{corollary} \label{cor:row-equivalence-characterization} \hlt{Row Equivalence Characterization}\\
For $A,B\in\operatorname{Mat}(m,n,\mathbb{F})$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\sim_{\mathrm{row}}B$.
\item $\operatorname{rref}(A)=\operatorname{rref}(B)$.
\item $\exists\,P\in\operatorname{Mat}(m,m,\mathbb{F})^{\times}$ such that $B=PA$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ Row-equivalent matrices have the same unique RREF by Theorem \ref{thm:rref-existence-uniqueness-and-elementary-factorization}.
\item[$(ii)$]$\Rightarrow (iii)$ If $R=\operatorname{rref}(A)=\operatorname{rref}(B)$, then $R=E_k\cdots E_1A=F_{\ell}\cdots F_1B$.\\
Since each $F_j$ is invertible, $B=(F_{\ell}\cdots F_1)^{-1}E_k\cdots E_1A$.
\item[$(iii)$]$\Rightarrow (i)$ If $B=PA$ and $P\in\operatorname{Mat}(m,m,\mathbb{F})^{\times}$, Lemma \ref{lem:invertible-matrices-row-reduce-to-identity} gives elementary matrices with $E_k\cdots E_1P=I_m$.\\
Hence $P=E_1^{-1}\cdots E_k^{-1}$, so $B$ is obtained from $A$ by elementary row operations.
\end{enumerate}
\hlt{Structural Commentary}: Row equivalence is invertible left multiplication.\\
\hlt{Key Insight}: Products of elementary matrices are the coordinate record of row reduction.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:matrix-representation-action} \hlt{Matrix Representation Acts on Coordinates}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis of $\mathcal{V}$, let $\gamma$ be a basis of $\mathcal{W}$, and let $T\in\LL(\mathcal{V},\mathcal{W})$.\\
Then for every $\mathbf{v} \in \mathcal{V}$, $[T(\mathbf{v})]_{\gamma} = [T]_{\gamma \leftarrow \beta}[\mathbf{v}]_{\beta}$.
\begin{proof}
Write $\mathbf{v} = a_1\mathbf{v}_1 + \cdots + a_n\mathbf{v}_n$. Then linearity gives $T(\mathbf{v}) = a_1T(\mathbf{v}_1) + \cdots + a_nT(\mathbf{v}_n)$ and the right-hand side is exactly the linear combination of the columns of $[T]_{\gamma \leftarrow \beta}$ with coefficients $a_1,\ldots,a_n$.\\
\hlt{Structural Commentary}: Matrix representation turns applying $T$ into multiplying coordinate columns.\\
\hlt{Key Insight}: The columns of $[T]_{\gamma \leftarrow \beta}$ are the coordinate images of the basis vectors.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:invertible-matrix-columns-basis} \hlt{Columns of an Invertible Matrix Form a Basis}\\
Let $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(n,n,\mathbb{F})$ be invertible. Then $(\mathbf{a}_1,\ldots,\mathbf{a}_n)$ is a basis of $\mathbb{F}^n$.
\begin{proof}
The columns span $\mathbb{F}^n$ because for any $\mathbf{y}\in\mathbb{F}^n$, writing $\mathbf{x}=A^{-1}\mathbf{y}$ gives $\mathbf{y}=A\mathbf{x}=x_1\mathbf{a}_1+\cdots+x_n\mathbf{a}_n$.\\
They are linearly independent because if $c_1\mathbf{a}_1+\cdots+c_n\mathbf{a}_n=\mathbf{0}$, then $A\mathbf{c}=\mathbf{0}$, so $\mathbf{c}=A^{-1}\mathbf{0}=\mathbf{0}$.\\
\hlt{Structural Commentary}: An invertible matrix is a coordinate isomorphism of $\mathbb{F}^n$. Its columns are the images of the standard coordinate directions.\\
\hlt{Key Insight}: Invertibility forces the columns to span and prevents nontrivial column relations.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:square-one-sided-inverse-is-two-sided} \hlt{One-Sided Inverses of Square Matrices}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$. If $AB=I_n$, then $BA=I_n$.
\begin{proof}
If $B\mathbf{x}=\mathbf{0}$, then $\mathbf{x}=AB\mathbf{x}=\mathbf{0}$, so the induced map $T_B:\mathbb{F}^n\to\mathbb{F}^n$ is injective. By Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences}, $T_B$ is surjective. For any $\mathbf{y}\in\mathbb{F}^n$, choose $\mathbf{x}$ with $\mathbf{y}=B\mathbf{x}$. Then $BA\mathbf{y}=BAB\mathbf{x}=B\mathbf{x}=\mathbf{y}$.\\
Thus $BA$ acts as the identity on every vector of $\mathbb{F}^n$, so $BA=I_n$.\\
\hlt{Structural Commentary}: In finite dimensions, injectivity and surjectivity coincide for square linear maps. A left inverse makes $B$ injective, hence onto, which forces the missing inverse relation.\\
\hlt{Key Insight}: A one-sided inverse of a square matrix is automatically two-sided.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:composition-and-matrix-multiplication} \hlt{Composition and Matrix Multiplication}\\
If $T \in \LL(\mathcal{U}, \mathcal{V})$ and $S \in \LL(\mathcal{V}, \mathcal{W})$, then $[ST]_{\delta \leftarrow \alpha} = [S]_{\delta \leftarrow \beta}[T]_{\beta \leftarrow \alpha}$ for any bases $\alpha$ of $\mathcal{U}$, $\beta$ of $\mathcal{V}$, and $\delta$ of $\mathcal{W}$.
\begin{proof}
For every $\mathbf{u}\in\mathcal{U}$, $[(ST)\mathbf{u}]_{\delta}=[S]_{\delta \leftarrow \beta}[T\mathbf{u}]_{\beta}=[S]_{\delta \leftarrow \beta}[T]_{\beta \leftarrow \alpha}[\mathbf{u}]_{\alpha}$.\\
Since coordinate columns $[\mathbf{u}]_{\alpha}$ range over all of $\mathbb{F}^{\dim(\mathcal{U})}$, the representing matrices are equal.\\
\hlt{Structural Commentary}: Matrix multiplication is the coordinate shadow of composition of linear maps.\\
\hlt{Key Insight}: Applying $T$ then $S$ becomes multiplying first by the matrix of $T$, then by the matrix of $S$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:linear-maps-and-matrices-same-data} \hlt{Linear Maps and Matrices Are the Same Data}\\
If $\dim(\mathcal{V}) = n$ and $\dim(\mathcal{W}) = m$, then choosing bases of $\mathcal{V}$ and $\mathcal{W}$ gives isomorphism $\LL(\mathcal{V}, \mathcal{W}) \cong \operatorname{Mat}(m,n,\mathbb{F})$.\\
Consequently, $\dim(\LL(\mathcal{V}, \mathcal{W})) = mn$.
\begin{proof}
Fix bases $\beta$ of $\mathcal{V}$ and $\gamma$ of $\mathcal{W}$. The map $T\mapsto [T]_{\gamma \leftarrow \beta}$ is linear. It is injective because a linear map whose matrix is zero sends every basis vector of $\mathcal{V}$ to $\mathbf{0}$, hence is the zero map. It is surjective because, for any $A=[a_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$, there is a unique linear map satisfying $T(\mathbf{v}_j)=a_{1j}\mathbf{w}_1+\cdots+a_{mj}\mathbf{w}_m$, $j=1,\ldots,n$.\\
Thus $\LL(\mathcal{V},\mathcal{W})\cong\operatorname{Mat}(m,n,\mathbb{F})$, whose standard entry matrices form a basis of length $mn$.\\
\hlt{Structural Commentary}: Choosing bases converts every linear map into one and only one array of scalars.\\
\hlt{Key Insight}: A matrix is exactly the list of coordinate images of the domain basis vectors.
\end{proof}
\end{theorem}

\begin{motivation} \hlt{Coordinate Encoding}\\
This is the key bridge between abstraction and computation. The linear map is basis-free; the matrix is what the map looks like after coordinates have been chosen.
\end{motivation}

\begin{definition} \hlt{Change of Basis Matrix}\\
Let $\beta$ and $\gamma$ be two bases of the same vector space $\mathcal{V}$. The change-of-basis matrix from $\beta$-coordinates to $\gamma$-coordinates is $C_{\gamma \leftarrow \beta} \defeq [I_{\mathcal{V}}]_{\gamma \leftarrow \beta}$. Thus $[\mathbf{v}]_{\gamma} = C_{\gamma \leftarrow \beta}[\mathbf{v}]_{\beta}$ for every $\mathbf{v} \in \mathcal{V}$.
\end{definition}

\begin{theorem} \label{thm:change-of-basis-matrices} \hlt{Change-of-Basis Matrices}\\
Let $\mathcal{V}$ be $n$-dimensional and let $\beta$ be a basis of $\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\gamma$ is a basis of $\mathcal{V}$, then $C_{\gamma \leftarrow \beta}$ is invertible and $(C_{\gamma \leftarrow \beta})^{-1}=C_{\beta \leftarrow \gamma}$.
\item If $S\in\operatorname{Mat}(n,n,\mathbb{F})$ is invertible, then there is a basis $\gamma$ of $\mathcal{V}$ such that $S=C_{\gamma \leftarrow \beta}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman)*.]
\setlength{\itemsep}{0pt}
\item[$(i)$] For any $\mathbf{v}\in\mathcal{V}$, $C_{\beta \leftarrow \gamma}C_{\gamma \leftarrow \beta}[\mathbf{v}]_{\beta}=[\mathbf{v}]_{\beta}$, $C_{\gamma \leftarrow \beta}C_{\beta \leftarrow \gamma}[\mathbf{v}]_{\gamma}=[\mathbf{v}]_{\gamma}$.\\
Since $[\cdot]_{\beta}$ and $[\cdot]_{\gamma}$ are onto $\mathbb{F}^n$, both products equal $I_n$. Hence $C_{\gamma \leftarrow \beta}$ is invertible with inverse $C_{\beta \leftarrow \gamma}$.
\item[$(ii)$] Write $S^{-1}=[\mathbf{s}_1\ \cdots\ \mathbf{s}_n]$. For each $j$, choose the unique $\mathbf{w}_j\in\mathcal{V}$ with $[\mathbf{w}_j]_{\beta}=\mathbf{s}_j$, and set $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$.\\
Since columns of $S^{-1}$ form a basis of $\mathbb{F}^n$ and $[\cdot]_{\beta}$ is an isomorphism, $\gamma$ is a basis of $\mathcal{V}$.\\
Moreover $C_{\beta \leftarrow \gamma}=S^{-1}$, so $C_{\gamma \leftarrow \beta}=S$.
\end{enumerate}
\hlt{Structural Commentary}: Change of basis is an invertible coordinate rewrite.\\
\hlt{Key Insight}: Every basis change is reversible, and every invertible matrix is one.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:basis-columns-form-invertible-matrix} \hlt{Basis Columns Give an Invertible Matrix}\\
If $(\mathbf{a}_1,\ldots,\mathbf{a}_n)$ is a basis of $\mathbb{F}^n$, then $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(n,n,\mathbb{F})$ is invertible.
\begin{proof}
Let $\epsilon=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ be the standard basis and $\alpha=(\mathbf{a}_1,\ldots,\mathbf{a}_n)$. Then
$C_{\epsilon \leftarrow \alpha} =
\begin{bmatrix}
[\mathbf{a}_1]_{\epsilon} & \cdots & [\mathbf{a}_n]_{\epsilon}
\end{bmatrix} =A$.\\
Since $\alpha$ is a basis, Theorem \ref{thm:change-of-basis-matrices} gives that $A$ is invertible.\\
\hlt{Structural Commentary}: Basis columns form a change-of-basis matrix.\\
\hlt{Key Insight}: Changing coordinates is reversible.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:complete-full-column-rank-matrix} \hlt{Completing a Full-Column-Rank Matrix}\\
If $X\in\operatorname{Mat}(m,n,\mathbb{F})$ has rank $n<m$, then $\exists$ $X'\in\operatorname{Mat}(m,m-n,\mathbb{F})$ s.t. $\begin{bmatrix}X & X'\end{bmatrix}\in\operatorname{Mat}(m,m,\mathbb{F})$ is invertible.
\begin{proof}
Write $X=[\mathbf{x}_1\ \cdots\ \mathbf{x}_n]$. Since $\operatorname{rank}(X)=n$, the columns $\mathbf{x}_1,\ldots,\mathbf{x}_n$ are linearly independent in $\mathbb{F}^m$. By Theorem \ref{thm:extending-independent-lists}, extend this list to a basis $(\mathbf{x}_1,\ldots,\mathbf{x}_n,\mathbf{x}_{n+1},\ldots,\mathbf{x}_m)$ of $\mathbb{F}^m$.\\
Set $X'=[\mathbf{x}_{n+1}\ \cdots\ \mathbf{x}_m]$. Corollary \ref{cor:basis-columns-form-invertible-matrix} shows that $\begin{bmatrix}X & X'\end{bmatrix}$ is invertible.\\
\hlt{Structural Commentary}: Full column rank means the existing columns are independent directions. Extending them to a basis turns the rectangular matrix into an invertible square matrix.\\
\hlt{Key Insight}: Independent columns can always be completed to a basis in finite dimensions.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:square-matrix-invertible-iff-full-rank} \hlt{Invertibility and Full Rank}\\
For $A\in\operatorname{Mat}(n,n,\mathbb{F})$, $A$ is invertible $\Longleftrightarrow$ $\operatorname{rank}(A)=n$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $A$ is invertible, its columns form a basis of $\mathbb{F}^n$ by Theorem \ref{thm:invertible-matrix-columns-basis}; hence $\operatorname{rank}(A)=n$.
\item[$(\Leftarrow)$] If $\operatorname{rank}(A)=n$, then $\operatorname{col}(A)=\mathbb{F}^n$.\\
Thus the columns of $A$ form a basis by Proposition \ref{prop:full-dimension-basis-criteria}, so Corollary \ref{cor:basis-columns-form-invertible-matrix} gives that $A$ is invertible.
\end{enumerate}
\hlt{Structural Commentary}: Full rank is square-matrix invertibility.\\
\hlt{Key Insight}: Invertibility means the columns fill $\mathbb{F}^n$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:transpose-invertible-iff} \hlt{Transpose Preserves Invertibility}\\
For $A\in\operatorname{Mat}(n,n,\mathbb{F})$, the matrix $A$ is invertible if and only if $A^t$ is invertible.
\begin{proof}
By Theorem \ref{thm:row-rank-equals-column-rank}, $\operatorname{rank}(A)=\operatorname{rank}(A^t)$. Thus $\operatorname{rank}(A)=n$ if and only if $\operatorname{rank}(A^t)=n$.\\
Corollary \ref{cor:square-matrix-invertible-iff-full-rank} gives the result.\\
\hlt{Structural Commentary}: Transposition exchanges rows and columns without changing rank.\\
\hlt{Key Insight}: A square matrix is invertible exactly when either its rows or columns have full rank.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:full-rank-products-preserve-rank} \hlt{Full-Rank Products Preserve Rank}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$. If $X\in\operatorname{Mat}(p,m,\mathbb{F})$ has full column rank and $Y\in\operatorname{Mat}(n,q,\mathbb{F})$ has full row rank, then $\operatorname{rank}(A)=\operatorname{rank}(XAY)$. This holds when $X$ and $Y$ are square invertible matrices of appropriate sizes.
\begin{proof}
Because $X$ has full column rank, the induced map $L_X:\mathbb{F}^m\to\mathbb{F}^p$, $L_X(\mathbf{u})=X\mathbf{u}$, has kernel $\{\mathbf{0}\}$.\\
Thus $\mathbf{z}\in\ker(XA) \iff X(A\mathbf{z})=\mathbf{0} \iff A\mathbf{z}=\mathbf{0} \iff \mathbf{z}\in\ker(A)$.\\
Since $XA$ and $A$ have the same number of columns, Corollary \ref{cor:matrix-rank-nullity} gives $\operatorname{rank}(XA)=\operatorname{rank}(A)$.\\
Because $Y$ has full row rank, $Y^t$ has full column rank. Applying the left-multiplication result to $A^t$ and $Y^t$, then using Theorem \ref{thm:row-rank-equals-column-rank}, gives $\operatorname{rank}(AY)=\operatorname{rank}((AY)^t) =\operatorname{rank}(Y^tA^t) =\operatorname{rank}(A^t) =\operatorname{rank}(A)$.\\
Applying the left-multiplication result once more to $AY$ gives $\operatorname{rank}(XAY)=\operatorname{rank}(AY)=\operatorname{rank}(A)$.\\
\hlt{Structural Commentary}: Multiplication by a full-column-rank matrix on the left loses no output information, while full row rank on the right loses no input information.\\
\hlt{Key Insight}: Rank changes only when multiplication introduces a new kernel or collapses the row side.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:two-sided-change-of-basis-for-linear-maps} \hlt{Change of Bases for a Linear Map}\\
Let $T\in\LL(\mathcal{V},\mathcal{W})$. Let $\beta,\beta'$ be bases of $\mathcal{V}$ and let $\gamma,\gamma'$ be bases of $\mathcal{W}$. Then $[T]_{\gamma'\leftarrow\beta'} =C_{\gamma'\leftarrow\gamma}[T]_{\gamma\leftarrow\beta}C_{\beta\leftarrow\beta'}$.
\begin{proof}
For every $\mathbf{v}\in\mathcal{V}$, $[T\mathbf{v}]_{\gamma'} =C_{\gamma'\leftarrow\gamma}[T\mathbf{v}]_{\gamma} =C_{\gamma'\leftarrow\gamma}[T]_{\gamma\leftarrow\beta}[\mathbf{v}]_{\beta} =C_{\gamma'\leftarrow\gamma}[T]_{\gamma\leftarrow\beta}C_{\beta\leftarrow\beta'}[\mathbf{v}]_{\beta'}$.\\
Since $[\cdot]_{\beta'}$ is onto, the representing matrices are equal.\\
\hlt{Structural Commentary}: Basis changes act by $C_{\gamma'\leftarrow\gamma}$ on the left and $C_{\beta\leftarrow\beta'}$ on the right.\\
\hlt{Key Insight}: Domain changes act on input coordinates; codomain changes act on output coordinates.
\end{proof}
\end{theorem}

\begin{definition} \label{def:matrix-equivalence} \hlt{Matrix Equivalence}\\
For $A,B\in\operatorname{Mat}(m,n,\mathbb{F})$, write $A\sim_{\mathrm{eq}} B$ if $\exists$ $P\in\operatorname{Mat}(m,m,\mathbb{F})^{\times}$ and $Q\in\operatorname{Mat}(n,n,\mathbb{F})^{\times}$ s.t. $B=PAQ$.
\end{definition}

\begin{proposition} \label{prop:matrix-equivalence-basic-properties} \hlt{Matrix Equivalence and Elementary Operations}\\
The relation $\sim_{\mathrm{eq}}$ is an equivalence relation on $\operatorname{Mat}(m,n,\mathbb{F})$. Moreover, $A\sim_{\mathrm{eq}}B$ if and only if $B$ can be obtained from $A$ by finitely many elementary row and column operations.
\begin{proof}
Reflexivity follows from $A=I_mAI_n$. If $B=PAQ$ with $P,Q$ invertible, then $A=P^{-1}BQ^{-1}$, so symmetry holds. If $B=P_1AQ_1$ and $C=P_2BQ_2$, then $C=(P_2P_1)A(Q_1Q_2)$, and products of invertible matrices are invertible by Corollary \ref{cor:matrix-inverse-product-and-transpose}; hence transitivity holds.\\
Each elementary row operation is left multiplication by an invertible elementary matrix by Proposition \ref{prop:elementary-matrices-perform-row-operations}. Each elementary column operation is right multiplication by the transpose of an invertible elementary matrix, again by Proposition \ref{prop:elementary-matrices-perform-row-operations} and Corollary \ref{cor:matrix-inverse-product-and-transpose}. Thus any finite sequence of elementary row and column operations has the form $B=PAQ$ with $P,Q$ invertible, so $A\sim_{\mathrm{eq}}B$.\\
Conversely, suppose $A\sim_{\mathrm{eq}}B$, so $B=PAQ$ with $P,Q$ invertible. Lemma \ref{lem:invertible-matrices-row-reduce-to-identity} gives products of elementary matrices expressing $P$ and $Q$ as products of elementary factors. Left multiplication by the factors of $P$ is a sequence of elementary row operations, and right multiplication by the factors of $Q$ is a sequence of elementary column operations. Hence $B$ can be obtained from $A$ by finitely many elementary row and column operations.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:matrix-equivalence-same-linear-map} \hlt{Equivalent Matrices Represent the Same Linear Map}\\
Let $\dim\mathcal{V}=n$, $\dim\mathcal{W}=m$, and $A,B\in\operatorname{Mat}(m,n,\mathbb{F})$.\\
Then $A\sim_{\mathrm{eq}}B$ $\Longleftrightarrow$ $\exists\,T\in\LL(\mathcal{V},\mathcal{W})$ with $A=[T]_{\gamma\leftarrow\beta},\ B=[T]_{\gamma'\leftarrow\beta'}$ for some bases $\beta,\beta'$ of $\mathcal{V}$ and $\gamma,\gamma'$ of $\mathcal{W}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Let $B=PAQ$ with $P,Q$ invertible. Choose bases $\beta,\gamma$ and define $T\in\LL(\mathcal{V},\mathcal{W})$ by $[T]_{\gamma\leftarrow\beta}=A$.\\
By Theorem \ref{thm:change-of-basis-matrices}, choose basis $\gamma'$ with $P=C_{\gamma'\leftarrow\gamma}$, and basis $\beta'$ with $Q^{-1}=C_{\beta'\leftarrow\beta}$. Then $Q=C_{\beta\leftarrow\beta'}$.\\
Then Theorem \ref{thm:two-sided-change-of-basis-for-linear-maps} gives $[T]_{\gamma'\leftarrow\beta'}=PAQ=B$.
\item[$(\Leftarrow)$] If $A=[T]_{\gamma\leftarrow\beta}$ and $B=[T]_{\gamma'\leftarrow\beta'}$, then $B=C_{\gamma'\leftarrow\gamma}AC_{\beta\leftarrow\beta'}$, so $A\sim_{\mathrm{eq}}B$.
\end{enumerate}
\hlt{Structural Commentary}: Equivalence is two-sided basis change.\\
\hlt{Key Insight}: Same map; different coordinates.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:rank-classifies-matrix-equivalence} \hlt{Rank Classifies Matrix Equivalence}\\
For $0\leq r\leq\min(m,n)$, set $J_r^{m,n}\defeq
\begin{bmatrix}
I_r & 0\\
0 & 0
\end{bmatrix}
\in\operatorname{Mat}(m,n,\mathbb{F})$.\\
For $A,B\in\operatorname{Mat}(m,n,\mathbb{F})$, $A\sim_{\mathrm{eq}}J_{\operatorname{rank}(A)}^{m,n}$ and $A\sim_{\mathrm{eq}}B\Longleftrightarrow \operatorname{rank}(A)=\operatorname{rank}(B)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $B=PAQ$ with $P,Q$ invertible, then $\operatorname{rank}(B)=\operatorname{rank}(A)$ by Theorem \ref{thm:full-rank-products-preserve-rank}.
\item[$(\Leftarrow)$] For any $X$ with $\operatorname{rank}(X)=r$, let $T_X:\mathbb{F}^n\to\mathbb{F}^m$ be $T_X(\mathbf{x})=X\mathbf{x}$.\\
Choose $\mathcal{M}_X$ with $\mathbb{F}^n=\ker(T_X)\oplus\mathcal{M}_X$; then $T_X|_{\mathcal{M}_X}:\mathcal{M}_X\to\operatorname{range}(T_X)$ is an isomorphism.\\
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_r)$ of $\mathcal{M}_X$, extend it by a basis of $\ker(T_X)$ to $\beta_X$, and choose $\gamma_X$ beginning with $T_X\mathbf{u}_1,\ldots,T_X\mathbf{u}_r$. Then $[T_X]_{\gamma_X\leftarrow\beta_X}=J_r^{m,n}$, so $X\sim_{\mathrm{eq}}J_r^{m,n}$.\\
If $\operatorname{rank}(A)=\operatorname{rank}(B)=r$, then $A\sim_{\mathrm{eq}}J_r^{m,n}$ and $B\sim_{\mathrm{eq}}J_r^{m,n}$. By Proposition \ref{prop:matrix-equivalence-basic-properties}, symmetry gives $J_r^{m,n}\sim_{\mathrm{eq}}B$, and transitivity gives $A\sim_{\mathrm{eq}}B$.
\end{enumerate}
\hlt{Structural Commentary}: Rank is the invariant of two-sided equivalence.\\
\hlt{Key Insight}: Adapt bases to kernel and range.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:linear-map-rank-normal-form} \hlt{Rank Normal Form for Linear Maps}\\
Let $T\in\LL(\mathcal{V},\mathcal{W})$, $\dim\mathcal{V}=n$, $\dim\mathcal{W}=m$, and $\operatorname{rank}(T)=r$. Then $\exists$ bases $\beta,\gamma$ with $[T]_{\gamma\leftarrow\beta}=J_r^{m,n}$.
\begin{proof}
For initial bases $\beta_0,\gamma_0$, let $A=[T]_{\gamma_0\leftarrow\beta_0}$.\\
By Theorem \ref{thm:rank-classifies-matrix-equivalence}, $A\sim_{\mathrm{eq}}J_r^{m,n}$; Theorem \ref{thm:matrix-equivalence-same-linear-map} gives bases $\beta,\gamma$ with $[T]_{\gamma\leftarrow\beta}=J_r^{m,n}$.\\
\hlt{Structural Commentary}: Rank gives the two-basis normal form.\\
\hlt{Key Insight}: Rank normalizes the map.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:linear-map-equivalence-by-rank} \hlt{Linear Maps Classified by Rank}\\
For $S,T\in\LL(\mathcal{V},\mathcal{W})$, write $S\sim_{\mathrm{bc}}T$ if $\exists$ bases $\beta_S,\beta_T$ of $\mathcal{V}$ and $\gamma_S,\gamma_T$ of $\mathcal{W}$ such that $[S]_{\gamma_S\leftarrow\beta_S}=[T]_{\gamma_T\leftarrow\beta_T}$.\\
Then $S\sim_{\mathrm{bc}}T$ $\Longleftrightarrow$ $\operatorname{rank}(S)=\operatorname{rank}(T)$
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $[S]_{\gamma_S\leftarrow\beta_S}=[T]_{\gamma_T\leftarrow\beta_T}$, then the two maps have the same matrix rank, hence the same rank.
\item[$(\Leftarrow)$] If $\operatorname{rank}(S)=\operatorname{rank}(T)=r$, Corollary \ref{cor:linear-map-rank-normal-form} gives bases with $[S]_{\gamma_S\leftarrow\beta_S}=J_r^{m,n}=[T]_{\gamma_T\leftarrow\beta_T}$.
\end{enumerate}
\hlt{Structural Commentary}: Rank is the invariant of basis-change equivalence.\\
\hlt{Key Insight}: Same rank means same normal form.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:change-of-basis-formula} \hlt{Change of Basis Formula}\\
Let $\mathcal{V}$ be finite-dimensional, let $T \in \LL(\mathcal{V})$, and let $\beta$ be a basis of $\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\gamma$ is a basis of $\mathcal{V}$ and $S=C_{\gamma \leftarrow \beta}$, then $[T]_{\gamma \leftarrow \gamma} = S[T]_{\beta \leftarrow \beta}S^{-1}$.
\item If $S\in\operatorname{Mat}(n,n,\mathbb{F})$ is invertible, then there is a basis $\gamma$ of $\mathcal{V}$ such that $[T]_{\gamma \leftarrow \gamma} = S[T]_{\beta \leftarrow \beta}S^{-1}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Suppose $\gamma$ is a basis and $S=C_{\gamma \leftarrow \beta}$. By Theorem \ref{thm:change-of-basis-matrices}, $S^{-1}=C_{\beta \leftarrow \gamma}$.\\
For every $\mathbf{v}\in\mathcal{V}$, $[T\mathbf{v}]_{\gamma} =S[T\mathbf{v}]_{\beta} =S[T]_{\beta \leftarrow \beta}[\mathbf{v}]_{\beta} =S[T]_{\beta \leftarrow \beta}S^{-1}[\mathbf{v}]_{\gamma}$.\\
Since $[\cdot]_{\gamma}$ is onto $\mathbb{F}^n$, the representing matrices are equal.
\item If $S$ is invertible, Theorem \ref{thm:change-of-basis-matrices} supplies a basis $\gamma$ with $S=C_{\gamma \leftarrow \beta}$, and part (i) applies.
\end{enumerate}
\hlt{Structural Commentary}: Changing basis conjugates operator matrices.\\
\hlt{Key Insight}: Similarity is a coordinate rewrite.
\end{proof}
\end{theorem}

\begin{definition} \label{def:similar-matrices} \hlt{Similarity}\\
Matrices $A,B \in \operatorname{Mat}(n,n,\mathbb{F})$ are similar if there exists an invertible matrix $P$ such that $B = PAP^{-1}$.
\end{definition}

\begin{proposition} \label{prop:similarity-equivalence-relation} \hlt{Similarity Is an Equivalence Relation}\\
Similarity is an equivalence relation on $\operatorname{Mat}(n,n,\mathbb{F})$.
\begin{proof}
For any $A$, $A=I_nAI_n^{-1}$, so similarity is reflexive. If $B=PAP^{-1}$, then $A=P^{-1}BP$, so it is symmetric. If $B=PAP^{-1}$ and $C=QBQ^{-1}$, then $C=Q(PAP^{-1})Q^{-1}=(QP)A(QP)^{-1}$ so it is transitive.\\
\hlt{Structural Commentary}: Similarity groups matrices by change of basis.\\
\hlt{Key Insight}: Identity, inverse, and composition give the three equivalence properties.
\end{proof}
\end{proposition}

\begin{perspective} \hlt{Similarity as Same Operator, New Coordinates}\\
Similarity does not mean two operators are merely related; it means they are the same operator written in two different bases. This is the structural content behind change of basis.
\end{perspective}

\begin{remark} \label{rem:similarity-convention} \hlt{Similarity Convention}\\
The forms $PAP^{-1}$ and $M^{-1}AM$ are the same convention with $P=M^{-1}$. In Theorem \ref{thm:change-of-basis-formula}, this means choosing opposite coordinate-change directions.
\end{remark}

\begin{corollary} \label{cor:similar-iff-same-linear-operator} \hlt{Similarity as One Operator in Two Bases}\\
For $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ and $B$ are similar.
\item There are an $n$-dim vector space $\mathcal{V}$, bases $\beta,\gamma$ of $\mathcal{V}$, and $T\in\LL(\mathcal{V})$ such that $A=[T]_{\beta \leftarrow \beta}$ and $B=[T]_{\gamma \leftarrow \gamma}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman)*.]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow (ii)$: Suppose $B=PAP^{-1}$ with $P$ invertible.\\
Let $\mathcal{V}=\mathbb{F}^n$, let $\beta$ be the standard basis, and let $T_A(\mathbf{x})=A\mathbf{x}$. By Theorem \ref{thm:change-of-basis-matrices}, choose $\gamma$ with $P=C_{\gamma \leftarrow \beta}$. Then $[T_A]_{\beta \leftarrow \beta}=A$, and Theorem \ref{thm:change-of-basis-formula} gives $[T_A]_{\gamma \leftarrow \gamma}=B$.
\item[$(ii)$] $\Rightarrow (i)$: If $A=[T]_{\beta \leftarrow \beta}$ and $B=[T]_{\gamma \leftarrow \gamma}$, then $B=C_{\gamma \leftarrow \beta}AC_{\beta \leftarrow \gamma}$ with $C_{\gamma \leftarrow \beta}$ invertible.
\end{enumerate}
\hlt{Structural Commentary}: Similar matrices are one operator in different bases.\\
\hlt{Key Insight}: Conjugation records a change of basis.
\end{proof}
\end{corollary}

\begin{definition} \label{def:similar-linear-operators} \hlt{Similarity of Operators}\\
For $T\in\LL(\mathcal{V})$ and $S\in\LL(\mathcal{W})$, $T\sim S$ $\Longleftrightarrow$ $\exists\,\Phi:\mathcal{V}\xrightarrow{\sim}\mathcal{W}$ such that $ S=\Phi T\Phi^{-1}$.\\
If $\mathcal{V}=\mathcal{W}$, then $\Phi\in\operatorname{Aut}(\mathcal{V})$ and similarity is conjugacy in $\LL(\mathcal{V})$.
\end{definition}

\begin{proposition} \label{prop:operator-similarity-and-common-matrices} \hlt{Operator Similarity and Common Matrix Representatives}\\
Let $\dim\mathcal{V}=\dim\mathcal{W}=n$, $T\in\LL(\mathcal{V})$, and $S\in\LL(\mathcal{W})$.\\
Then $T\sim S$ $\Longleftrightarrow$ $\exists\,A\in\operatorname{Mat}(n,n,\mathbb{F})$ with $A=[T]_{\beta\leftarrow\beta}=[S]_{\gamma\leftarrow\gamma}$ for some bases $\beta$ of $\mathcal{V}$ and $\gamma$ of $\mathcal{W}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Let $S=\Phi T\Phi^{-1}$ and $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$. Put $\gamma=(\Phi\mathbf{v}_1,\ldots,\Phi\mathbf{v}_n)$.\\
Then $\gamma$ is a basis and, for all $j$, $S(\Phi\mathbf{v}_j)=\Phi T\mathbf{v}_j$, so $[S]_{\gamma\leftarrow\gamma}=[T]_{\beta\leftarrow\beta}$.
\item[$(\Leftarrow)$] Suppose $[T]_{\beta\leftarrow\beta}=[S]_{\gamma\leftarrow\gamma}$. Define $\Phi:\mathcal{V}\to\mathcal{W}$ by $\Phi(\beta_j)=\gamma_j$.\\
Then $\Phi$ is an isomorphism and $\Phi T\beta_j=S\Phi\beta_j$ for every $j$, hence $\Phi T=S\Phi$. Thus $S=\Phi T\Phi^{-1}$.
\end{enumerate}
\hlt{Structural Commentary}: Similarity is equality of matrices after identifying bases.\\
\hlt{Key Insight}: A common matrix is an operator-level conjugacy written in bases.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:similarity-preserves-scalar-shifts} \hlt{Similarity and Scalar Shifts}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A$ and $B$ are similar, then $A-\lambda I_n$ and $B-\lambda I_n$ are similar for every $\lambda\in\mathbb{F}$.
\item If $A-\lambda I_n$ and $B-\lambda I_n$ are similar for some $\lambda\in\mathbb{F}$, then $A$ and $B$ are similar.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $B=PAP^{-1}$ with $P$ invertible, then $B-\lambda I_n=PAP^{-1}-\lambda PP^{-1}=P(A-\lambda I_n)P^{-1}$.
\item If $B-\lambda I_n=P(A-\lambda I_n)P^{-1}$ with $P$ invertible, then adding $\lambda I_n=\lambda PP^{-1}$ gives $B=PAP^{-1}$.
\end{enumerate}
\hlt{Structural Commentary}: Scalar shifts commute with change of basis.\\
\hlt{Key Insight}: The identity matrix is basis-independent.
\end{proof}
\end{theorem}
