\subsection{Duality and Multilinear Structure}

% **Role:** algebraic language of linear structure.
%
% Dual spaces, dual maps, annihilators, transpose as dual map, bilinear forms, tensor products as later extension.

Duality studies vectors through scalar-valued tests. Functionals, dual maps, annihilators, transposes, bilinear forms, and tensor products extend linear structure into multilinear language.

\begin{definition} \hlt{Linear Functional and Dual Space}\\
A \hlt{linear functional} on $\mathcal{V}$ is a linear map from $\mathcal{V}$ to $\mathbb{F}$. The \hlt{dual space} of $\mathcal{V}$ is $\mathcal{V}' \defeq \LL(\mathcal{V},\mathbb{F})$.
\end{definition}

\begin{example} \label{ex:evaluation-and-integral-functionals} \hlt{Evaluation and Integral Functionals}\\
The evaluation map $\operatorname{ev}_0:\mathbb{F}[x]\to\mathbb{F}$, $p\mapsto p(0)$ is a linear functional, since $\operatorname{ev}_0(\alpha p+\beta q)=\alpha p(0)+\beta q(0)$ for all $\alpha,\beta\in\mathbb{F}$ and $p,q\in\mathbb{F}[x]$.
If $C[a,b]$ is the real vector space of continuous functions on $[a,b]$, then $I:C[a,b]\to\mathbb{R}$, $I(f)\defeq\int_a^b f(x)\,dx$ is a linear functional, since $I(\alpha f+\beta g)=\alpha I(f)+\beta I(g)$ $\forall \alpha,\beta\in\mathbb{R}$ and $f,g\in C[a,b]$.
\end{example}

\begin{definition} \hlt{Dual Basis}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis of $\mathcal{V}$. The \hlt{dual basis} of $\beta$ is the list $\beta'=(\varphi_1,\ldots,\varphi_n)$ in $\mathcal{V}'$ defined by
\begin{align}
\varphi_j(\mathbf{v}_k)=
\begin{cases}
1, & k=j,\\
0, & k\neq j.
\end{cases}
\nonumber
\end{align}
Existence and uniqueness of each $\varphi_j$ follow from Proposition \ref{prop:linear-map-determined-by-basis}.
\end{definition}

\begin{theorem} \label{thm:dual-basis-and-dimension} \hlt{Dual Basis and Dimension of the Dual Space}\\
Let $\mathcal{V}$ be finite-dimensional with basis $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$, and let $\beta'=(\varphi_1,\ldots,\varphi_n)$ be the dual basis. Then $\beta'$ is a basis of $\mathcal{V}'$. In particular, $\dim(\mathcal{V}')=\dim(\mathcal{V})$.
\begin{proof}
First suppose $a_1\varphi_1+\cdots+a_n\varphi_n=0$. Evaluating at $\mathbf{v}_j$ gives $a_j=0$ for each $j$, so $\beta'$ is linearly independent. Now let $\varphi\in\mathcal{V}'$. For $\mathbf{v}=c_1\mathbf{v}_1+\cdots+c_n\mathbf{v}_n$, $\left(\sum\limits_{j=1}^n \varphi(\mathbf{v}_j)\varphi_j\right)(\mathbf{v}) =\sum\limits_{j=1}^n \varphi(\mathbf{v}_j)c_j =\varphi(\mathbf{v})$.\\
Thus $\varphi=\sum_{j=1}^n \varphi(\mathbf{v}_j)\varphi_j$, so $\beta'$ spans $\mathcal{V}'$.\\
\hlt{Structural Commentary}: A basis of $\mathcal{V}$ induces the dual coordinate system.\\
\hlt{Key Insight}: A linear functional is determined exactly by its values on a basis.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:coordinate-functionals-for-hamel-basis} \hlt{Coordinate Functionals for a Hamel Basis}\\
Let $\mathcal{B}=\{\mathbf{b}_i\}_{i\in I}$ be a Hamel basis of $\mathcal{V}$, define $\varepsilon_i\in\mathcal{V}'$ by $\varepsilon_i\left(\sum\limits_{j\in F}a_j\mathbf{b}_j\right)\defeq a_i$, $F\subseteq I$ finite, $a_i=0$ if $i\notin F$.\\
Then $\{\varepsilon_i\}_{i\in I}$ is linearly independent in $\mathcal{V}'$. Moreover, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\operatorname{span}\{\varepsilon_i:i\in I\}=\mathcal{V}'$;
\item $\abs{I}<\infty$;
\item $\dim(\mathcal{V})<\infty$.
\end{enumerate}
\begin{proof}
If $\sum\limits_{r=1}^n c_r\varepsilon_{i_r}=0$ with distinct $i_r$, then $0=\left(\sum\limits_{r=1}^n c_r\varepsilon_{i_r}\right)(\mathbf{b}_{i_s})=c_s$ $\forall s$. Hence $\{\varepsilon_i\}_{i\in I}$ linearly independent.
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow$ $(ii)$. Suppose $\operatorname{span}\{\varepsilon_i:i\in I\}=\mathcal{V}'$. Put $\operatorname{supp}_{\mathcal{B}}(\psi)\defeq\{i\in I:\psi(\mathbf{b}_i)\neq0\}$.\\
If $I$ were infinite, define $\varphi\in\mathcal{V}'$ by $\varphi(\mathbf{b}_i)=1$ for all $i\in I$.\\
Then $\operatorname{supp}_{\mathcal{B}}(\varphi)=I$, while $\operatorname{supp}_{\mathcal{B}}(\psi)$ is finite for every $\psi\in\operatorname{span}\{\varepsilon_i:i\in I\}$, a contradiction.
\item[$(ii)$] $\Rightarrow$ $(iii)$. Since $\mathcal{B}$ is a basis, $\dim(\mathcal{V})=\abs{I}<\infty$.
\item[$(iii)$] $\Rightarrow$ $(i)$. If $\dim(\mathcal{V})<\infty$, then Theorem \ref{thm:dual-basis-and-dimension} gives $\operatorname{span}\{\varepsilon_i:i\in I\}=\mathcal{V}'$.
\end{enumerate}
\hlt{Structural Commentary}: Coordinate spans are finite-support.\\
\hlt{Key Insight}: Infinite-dimensional coordinate functionals are independent, not spanning.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-algebraic-dual} \hlt{Dimension of the Algebraic Dual}\\
For every $\mathbb{F}$-vector space $\mathcal{V}$, $\dim(\mathcal{V})\leq\dim(\mathcal{V}')$. Moreover, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\dim(\mathcal{V})=\dim(\mathcal{V}')$;
\item $\dim(\mathcal{V})<\infty$.
\end{enumerate}
\begin{proof}
Let $\mathcal{B}=\{\mathbf{b}_i\}_{i\in I}$ be a Hamel basis. By Proposition \ref{prop:coordinate-functionals-for-hamel-basis}, $\{\varepsilon_i\}_{i\in I}\subseteq\mathcal{V}'$ is linearly independent; hence $\dim(\mathcal{V})=\abs{I}\leq\dim(\mathcal{V}')$. For the equality criterion:
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow$ $(ii)$. Suppose $\dim(\mathcal{V})=\dim(\mathcal{V}')$. If $I$ were infinite, then $\mathcal{V}\cong(\mathbb{F}^{I})_0$ and $\mathcal{V}'\cong\mathbb{F}^{I}$.\\
Let $\mathbb{P}\subseteq\mathbb{F}$ be the prime subfield. Since $\mathbb{P}$ is countable, $\dim_{\mathbb{P}}\big((\mathbb{P}^{I})_0\big)=\abs{I}$ and $\abs{\mathbb{P}^{I}}>\abs{I}$ by Cantor, so $\dim_{\mathbb{P}}(\mathbb{P}^{I})>\abs{I}$; hence $\dim_{\mathbb{P}}\big((\mathbb{P}^{I})_0\big)=\abs{I}< \dim_{\mathbb{P}}(\mathbb{P}^{I})$.\\
Every $\mathbb{P}$-linearly independent family in $\mathbb{P}^{I}$ is $\mathbb{F}$-linearly independent in $\mathbb{F}^{I}$, by expanding coefficients in a $\mathbb{P}$-basis of $\mathbb{F}$. Thus $\dim_{\mathbb{F}}(\mathcal{V}) =\dim_{\mathbb{F}}\big((\mathbb{F}^{I})_0\big) < \dim_{\mathbb{F}}(\mathbb{F}^{I}) =\dim_{\mathbb{F}}(\mathcal{V}')$, a contradiction. Hence $\dim(\mathcal{V})<\infty$.
\item[$(ii)$] $\Rightarrow$ $(i)$. If $\dim(\mathcal{V})<\infty$, then Theorem \ref{thm:dual-basis-and-dimension} gives $\dim(\mathcal{V})=\dim(\mathcal{V}')$.
\end{enumerate}
\hlt{Structural Commentary}: Algebraic duals have arbitrary support.\\
\hlt{Key Insight}: Infinite algebraic duals are bigger than their original spaces.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:functionals-separate-points-finite-dimensional} \label{prop:functionals-separate-points} \hlt{Functionals Separate Points}\\
Let $\mathcal{V}$ be a vector space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathbf{v}\neq\mathbf{0}$, then there exists $\varphi\in\mathcal{V}'$ such that $\varphi(\mathbf{v})\neq0$.
\item A vector $\mathbf{v}\in\mathcal{V}$ is zero if and only if $\varphi(\mathbf{v})=0$ for every $\varphi\in\mathcal{V}'$.
\item If $\varphi\in\mathcal{V}'$ and $\varphi(\mathbf{x})\neq0$, then $\mathcal{V}=\operatorname{span}(\mathbf{x})\oplus\ker(\varphi)$.
\item If nonzero $\varphi,\psi\in\mathcal{V}'$ have the same kernel, then $\varphi=a\psi$ for some nonzero $a\in\mathbb{F}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Extend $\{\mathbf{v}\}$ to a Hamel basis $\mathcal{B}$ of $\mathcal{V}$ by Theorem \ref{thm:hamel-basis-extension}. Define $\varphi$ on $\mathcal{B}$ by $\varphi(\mathbf{v})=1$ and $\varphi(\mathbf{b})=0$ for $\mathbf{b}\in\mathcal{B}\setminus\{\mathbf{v}\}$, then extend linearly to finite basis expansions.
\item If $\mathbf{v}=\mathbf{0}$, then $\varphi(\mathbf{v})=0$ for every $\varphi\in\mathcal{V}'$. Conversely, if $\varphi(\mathbf{v})=0$ for every $\varphi\in\mathcal{V}'$ and $\mathbf{v}\neq\mathbf{0}$, then (i) gives some $\psi\in\mathcal{V}'$ with $\psi(\mathbf{v})\neq0$, a contradiction.
\item Since $\varphi(\mathbf{x})\neq0$, $\operatorname{span}(\mathbf{x})\cap\ker(\varphi)=\{\mathbf{0}\}$.\\
For any $\mathbf{v}\in\mathcal{V}$, $\mathbf{v}=\frac{\varphi(\mathbf{v})}{\varphi(\mathbf{x})}\mathbf{x}
       +\left(\mathbf{v}-\frac{\varphi(\mathbf{v})}{\varphi(\mathbf{x})}\mathbf{x}\right)$, and the second summand lies in $\ker(\varphi)$.
\item Let $\mathcal{K}=\ker(\varphi)=\ker(\psi)$ and choose $\mathbf{x}\notin\mathcal{K}$. By (iii), every $\mathbf{v}\in\mathcal{V}$ can be written $\mathbf{v}=c\mathbf{x}+\mathbf{k}$ with $\mathbf{k}\in\mathcal{K}$. Setting $a=\varphi(\mathbf{x})/\psi(\mathbf{x})$ gives $\varphi(\mathbf{v})=c\varphi(\mathbf{x})=a\,c\psi(\mathbf{x})=a\psi(\mathbf{v})$.
\end{enumerate}
\hlt{Structural Commentary}: Dual vectors separate points and define hyperplanes.\\
\hlt{Key Insight}: A nonzero functional splits the space into one visible direction plus its kernel.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:nonzero-functional-hyperplane-kernel} \hlt{Kernels of Nonzero Functionals}\\
If $0\neq\varphi\in\mathcal{V}'$, then $\varphi$ is surjective onto $\mathbb{F}$, $\mathcal{V}/\ker(\varphi)\cong\mathbb{F}$, and $\operatorname{codim}_{\mathcal{V}}(\ker\varphi)=1$.
\begin{proof}
Choose $\mathbf{x}$ with $\varphi(\mathbf{x})\neq0$. For each $a\in\mathbb{F}$, $\varphi\bigl(a\varphi(\mathbf{x})^{-1}\mathbf{x}\bigr)=a$, so $\operatorname{im}\varphi=\mathbb{F}$.\\
Hence, by Theorem \ref{thm:quotient-by-kernel-is-range}, $\mathcal{V}/\ker(\varphi)\cong\operatorname{im}\varphi=\mathbb{F}$ and $\operatorname{codim}_{\mathcal{V}}(\ker\varphi)=\dim(\mathcal{V}/\ker\varphi)=1$.\\
\hlt{Structural Commentary}: A nonzero functional sees exactly one scalar direction and kills a hyperplane.\\
\hlt{Key Insight}: Kernels of nonzero functionals have codimension one.
\end{proof}
\end{corollary}

\begin{definition} \label{def:algebraic-double-dual-and-natural-map} \hlt{Algebraic Double Dual and Natural Map}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{algebraic double dual} of $\mathcal{V}$ is $\mathcal{V}''\defeq(\mathcal{V}')'$.
\item The \hlt{natural evaluation map} is $\Lambda_{\mathcal{V}}:\mathcal{V}\to\mathcal{V}''$; $\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)=\varphi(\mathbf{v})$.
\item $\mathcal{V}$ is \hlt{algebraically reflexive} if $\Lambda_{\mathcal{V}}$ is surjective.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:algebraic-reflexivity-natural-map} \hlt{Algebraic Reflexivity}\\
Natural map $\Lambda_{\mathcal{V}}\in\LL(\mathcal{V},\mathcal{V}'')$ is linear and injective. It is an isomorphism if and only if $\mathcal{V}$ is finite-dimensional.
\begin{proof}
For $\mathbf{v}\in\mathcal{V}$, $\Lambda_{\mathcal{V}}(\mathbf{v}):\varphi\mapsto\varphi(\mathbf{v})$ is linear on $\mathcal{V}'$.\\
For $a,b\in\mathbb{F}$ and $\mathbf{u},\mathbf{v}\in\mathcal{V}$, $\Lambda_{\mathcal{V}}(a\mathbf{u}+b\mathbf{v})(\varphi) =a\Lambda_{\mathcal{V}}(\mathbf{u})(\varphi)+b\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)$, $\varphi\in\mathcal{V}'$, so $\Lambda_{\mathcal{V}}\in\LL(\mathcal{V},\mathcal{V}'')$.\\
If $\mathbf{v}\in\ker(\Lambda_{\mathcal{V}})$, then $\varphi(\mathbf{v})=0$ for all $\varphi\in\mathcal{V}'$, hence $\mathbf{v}=\mathbf{0}$ by Proposition \ref{prop:functionals-separate-points}. Thus $\ker(\Lambda_{\mathcal{V}})=\{\mathbf{0}\}$.\\
For the isomorphism criterion:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\Lambda_{\mathcal{V}}$ is an isomorphism and $\dim(\mathcal{V})=\infty$, then Theorem \ref{thm:dimension-of-algebraic-dual} gives $\dim(\mathcal{V}'')>\dim(\mathcal{V}')>\dim(\mathcal{V})$, contradicting $\mathcal{V}\cong\mathcal{V}''$. Hence $\dim(\mathcal{V})<\infty$.
\item[$(\Leftarrow)$] If $\dim(\mathcal{V})<\infty$, then $\dim(\mathcal{V}'')=\dim(\mathcal{V}')=\dim(\mathcal{V})$ by Theorem \ref{thm:dual-basis-and-dimension}.\\
Since $\Lambda_{\mathcal{V}}$ is injective, it is an isomorphism.
\end{enumerate}
\hlt{Structural Commentary}: $\Lambda_{\mathcal{V}}$ embeds vectors as evaluation functionals.\\
\hlt{Key Insight}: Finite-dimensional spaces are exactly the algebraically reflexive vector spaces.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:double-dual-map-identifies-with-original-map} \hlt{Double Dual Maps Identify with Original Maps}\\
Let $\dim\mathcal{V},\dim\mathcal{W}<\infty$ and $T\in\LL(\mathcal{V},\mathcal{W})$.\\
Under $\Lambda_{\mathcal{V}}:\mathcal{V}\cong\mathcal{V}''$ and $\Lambda_{\mathcal{W}}:\mathcal{W}\cong\mathcal{W}''$, $T''\Lambda_{\mathcal{V}}=\Lambda_{\mathcal{W}}T$,  so $T''$ identifies with $T$.
\begin{proof}
For $\mathbf{v}\in\mathcal{V}$ and $\psi\in\mathcal{W}'$, $\bigl(T''\Lambda_{\mathcal{V}}(\mathbf{v})\bigr)(\psi) =\Lambda_{\mathcal{V}}(\mathbf{v})(T'\psi) =(T'\psi)(\mathbf{v}) =\psi(T\mathbf{v}) =\Lambda_{\mathcal{W}}(T\mathbf{v})(\psi)$.\\
Hence $T''\Lambda_{\mathcal{V}}=\Lambda_{\mathcal{W}}T$.\\
\hlt{Structural Commentary}: Double dualisation commutes with natural evaluation.\\
\hlt{Key Insight}: The double dual square commutes because both sides evaluate every functional the same way.
\end{proof}
\end{corollary}

\begin{definition} \label{def:dual-map} \hlt{Dual Map}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$, the \hlt{dual map} of $T$ is the map $T':\mathcal{W}'\to\mathcal{V}'$ defined by $T'\varphi \defeq \varphi T$ for every $\varphi\in\mathcal{W}'$.\\
Equivalently, $(T'\varphi)(\mathbf{v})=\varphi(T\mathbf{v})$.
\end{definition}

\begin{perspective} \label{per:dual-map-versus-inner-product-adjoint} \hlt{Dual Map and Inner-Product Adjoint}\\
For $T:\mathcal{V}\to\mathcal{W}$, the pullback $\varphi\mapsto\varphi\circ T$ is the \hlt{dual map} $T':\mathcal{W}'\to\mathcal{V}'$. Also called \hlt{operator adjoint} and written $T^\times$. It is distinct from the inner-product adjoint $T^*$ of Definition \ref{def:adjoint-linear-map}, which requires inner products.
\end{perspective}

\begin{proposition} \label{prop:dual-map-is-linear} \hlt{Dual Maps Are Linear}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$, then $T'\in\LL(\mathcal{W}',\mathcal{V}')$.
\begin{proof}
For $\varphi,\psi\in\mathcal{W}'$, $a\in\mathbb{F}$, and $\mathbf{v}\in\mathcal{V}$, $\left(T'(a\varphi+\psi)\right)(\mathbf{v}) =(a\varphi+\psi)(T\mathbf{v}) =a\varphi(T\mathbf{v})+\psi(T\mathbf{v}) =\left(aT'\varphi+T'\psi\right)(\mathbf{v})$.\\
\hlt{Structural Commentary}: The dual map pulls functionals back along $T$ and preserves linear structure.\\
\hlt{Key Insight}: Dual maps reverse direction because a functional can test $T\mathbf{v}$ only after $T$ is applied.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:algebra-of-dual-maps} \hlt{Algebra of Dual Maps}\\
The dual operation has the following properties:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $S,T\in\LL(\mathcal{V},\mathcal{W})$ and $a\in\mathbb{F}$, then $(aS+T)'=aS'+T'$.
\item If $T\in\LL(\mathcal{U},\mathcal{V})$ and $S\in\LL(\mathcal{V},\mathcal{W})$, then $(ST)'=T'S'$.
\item $(I_{\mathcal{V}})'=I_{\mathcal{V}'}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\varphi\in\mathcal{W}'$ and $\mathbf{v}\in\mathcal{V}$, $\left((aS+T)'\varphi\right)(\mathbf{v}) =\varphi(aS\mathbf{v}+T\mathbf{v}) =a(S'\varphi)(\mathbf{v})+(T'\varphi)(\mathbf{v})$, so $(aS+T)'=aS'+T'$.
\item For $\varphi\in\mathcal{W}'$ and $\mathbf{u}\in\mathcal{U}$, $\left((ST)'\varphi\right)(\mathbf{u}) =\varphi(S(T\mathbf{u})) =\left(T'(S'\varphi)\right)(\mathbf{u})$, so $(ST)'=T'S'$.
\item For $\varphi\in\mathcal{V}'$ and $\mathbf{v}\in\mathcal{V}$, $(I_{\mathcal{V}}'\varphi)(\mathbf{v})=\varphi(I_{\mathcal{V}}\mathbf{v})=\varphi(\mathbf{v})$, so $(I_{\mathcal{V}})'=I_{\mathcal{V}'}$.
\end{enumerate}
\hlt{Structural Commentary}: Duality is linear on maps and contravariant under composition.\\
\hlt{Key Insight}: Pullbacks compose in the opposite order.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:dual-map-inverse} \hlt{Dual Map of an Inverse}\\
If $T\in\LL(\mathcal{V},\mathcal{W})$ is an isomorphism, then $T'$ is an isomorphism and $(T^{-1})'=(T')^{-1}$
\begin{proof}
Since $T^{-1}\in\LL(\mathcal{W},\mathcal{V})$, Theorem \ref{thm:algebra-of-dual-maps} gives $T'(T^{-1})'=(T^{-1}T)'=(I_{\mathcal{V}})'=I_{\mathcal{V}'}$\\
and $(T^{-1})'T'=(TT^{-1})'=(I_{\mathcal{W}})'=I_{\mathcal{W}'}$. Thus $(T^{-1})'$ is inverse of $T'$, $T'$ is isomorphism and $(T^{-1})'=(T')^{-1}$.\\
\hlt{Structural Commentary}: Duality preserves inverses contravariantly.\\
\hlt{Key Insight}: Invertibility survives duality.
\end{proof}
\end{corollary}

\begin{definition} \hlt{Annihilator}\\
If $\mathcal{U}\subseteq\mathcal{V}$, the \hlt{annihilator} of $\mathcal{U}$ is $\mathcal{U}^0\defeq\{\varphi\in\mathcal{V}':\varphi(\mathbf{u})=0 \text{ for every } \mathbf{u}\in\mathcal{U}\}$.
\end{definition}

\begin{proposition} \label{prop:annihilator-is-subspace} \hlt{Annihilators Are Subspaces}\\
If $\mathcal{U}\subseteq\mathcal{V}$, then $\mathcal{U}^0$ is a subspace of $\mathcal{V}'$.
\begin{proof}
The zero functional lies in $\mathcal{U}^0$. If $\varphi,\psi\in\mathcal{U}^0$, $a\in\mathbb{F}$, and $\mathbf{u}\in\mathcal{U}$, then $(a\varphi+\psi)(\mathbf{u})=a\varphi(\mathbf{u})+\psi(\mathbf{u})=0$.\\
Thus $a\varphi+\psi\in\mathcal{U}^0$, and Theorem \ref{thm:subspacetest} shows that $\mathcal{U}^0$ is a subspace of $\mathcal{V}'$.\\
\hlt{Structural Commentary}: The annihilator collects all linear tests that vanish on a chosen set. Closure follows because vanishing is preserved by linear combinations of tests.\\
\hlt{Key Insight}: Annihilators are kernels of evaluation constraints.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-annihilator} \hlt{Dimension of the Annihilator}\\
Let $\mathcal{V}$ be finite-dimensional and let $\mathcal{U}$ be a subspace of $\mathcal{V}$. Then $\dim(\mathcal{U})+\dim(\mathcal{U}^0)=\dim(\mathcal{V})$.
\begin{proof}
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ of $\mathcal{U}$ and extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_k)$ of $\mathcal{V}$.\\
Let $(\varphi_1,\ldots,\varphi_m,\psi_1,\ldots,\psi_k)$ be the corresponding dual basis.\\
By Theorem \ref{thm:dual-basis-and-dimension}, every $\varphi\in\mathcal{V}'$ has the expansion $\varphi=\sum\limits_{j=1}^m \varphi(\mathbf{u}_j)\varphi_j+\sum\limits_{\ell=1}^k \varphi(\mathbf{w}_\ell)\psi_\ell$.\\
If $\varphi\in\mathcal{U}^0$, then $\varphi(\mathbf{u}_j)=0$ for every $j$, so $\varphi\in\operatorname{span}(\psi_1,\ldots,\psi_k)$.\\
Conversely, each $\psi_\ell$ vanishes on every $\mathbf{u}_j$, hence on $\mathcal{U}=\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_m)$. Therefore $\mathcal{U}^0=\operatorname{span}(\psi_1,\ldots,\psi_k)$.\\
The list $(\psi_1,\ldots,\psi_k)$ is linearly independent as a sublist of the dual basis, so $\dim(\mathcal{U}^0)=k=\dim(\mathcal{V})-\dim(\mathcal{U})$.\\
\hlt{Structural Commentary}: Extending a basis of $\mathcal{U}$ separates directions inside $\mathcal{U}$ from complementary directions. The annihilator is dual to the complementary part.\\
\hlt{Key Insight}: To vanish on $\mathcal{U}$, a functional can only use dual coordinates attached to directions outside $\mathcal{U}$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:annihilator-lattice-identities} \hlt{Annihilator Lattice Identities}\\
Let $\mathcal{V}$ be finite-dimensional, and let $\mathcal{S},\mathcal{T}$ be subspaces of $\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathcal{S}\subseteq\mathcal{T}$, then $\mathcal{T}^0\subseteq\mathcal{S}^0$.
\item $(\mathcal{S}+\mathcal{T})^0=\mathcal{S}^0\cap\mathcal{T}^0$.
\item $(\mathcal{S}\cap\mathcal{T})^0=\mathcal{S}^0+\mathcal{T}^0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\varphi\in\mathcal{T}^0$ and $\mathbf{s}\in\mathcal{S}$, then $\mathbf{s}\in\mathcal{T}$, so $\varphi(\mathbf{s})=0$. Hence $\varphi\in\mathcal{S}^0$.
\item For $\varphi\in\mathcal{V}'$, $\varphi\in(\mathcal{S}+\mathcal{T})^0$ if and only if $\varphi(\mathbf{s}+\mathbf{t})=0$ for all $\mathbf{s}\in\mathcal{S}$ and $\mathbf{t}\in\mathcal{T}$, if and only if $\varphi\in\mathcal{S}^0\cap\mathcal{T}^0$.
\item Since $\mathcal{S}\cap\mathcal{T}\subseteq\mathcal{S},\mathcal{T}$, part (i) gives $\mathcal{S}^0+\mathcal{T}^0\subseteq(\mathcal{S}\cap\mathcal{T})^0$. By part (ii), Theorem \ref{thm:dimension-formula-for-sum}, and Theorem \ref{thm:dimension-of-annihilator},\\
$\dim(\mathcal{S}^0+\mathcal{T}^0) =\dim(\mathcal{S}^0)+\dim(\mathcal{T}^0)-\dim(\mathcal{S}^0\cap\mathcal{T}^0) =\dim(\mathcal{V})-\dim(\mathcal{S}\cap\mathcal{T}) =\dim\big((\mathcal{S}\cap\mathcal{T})^0\big)$.\\
The inclusion is therefore equality.
\end{enumerate}
\hlt{Structural Commentary}: Annihilation is an order-reversing lattice duality.\\
\hlt{Key Insight}: Vanishing on more vectors gives fewer functionals.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:double-annihilator-finite-dimensional} \hlt{Double Annihilator}\\
Let $\dim\mathcal{V}<\infty$ and $\emptyset\neq M\subseteq\mathcal{V}$. Under $\Lambda_{\mathcal{V}}:\mathcal{V}\cong\mathcal{V}''$, $M^{00}=\Lambda_{\mathcal{V}}(\operatorname{span}M)$.\\
In particular, if $\mathcal{S}\leq\mathcal{V}$, then $\mathcal{S}^{00}=\Lambda_{\mathcal{V}}(\mathcal{S})\cong\mathcal{S}$.
\begin{proof}
If $\mathbf{v}\in\operatorname{span}M$ and $\varphi\in M^0$, then $\varphi(\mathbf{v})=0$, so $\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)=0$. Hence $\Lambda_{\mathcal{V}}(\operatorname{span}M)\subseteq M^{00}$.\\
Conversely, let $\eta\in M^{00}$. By Theorem \ref{thm:algebraic-reflexivity-natural-map}, $\eta=\Lambda_{\mathcal{V}}(\mathbf{v})$ for a unique $\mathbf{v}\in\mathcal{V}$.\\
If $\mathbf{v}\notin\operatorname{span}M$, choose $\varphi\in\mathcal{V}'$ with $\varphi|_{\operatorname{span}M}=0$ and $\varphi(\mathbf{v})=1$.\\
Then $\varphi\in M^0$ but $\eta(\varphi)=\Lambda_{\mathcal{V}}(\mathbf{v})(\varphi)=1$, contradicting $\eta\in M^{00}$. Thus $\mathbf{v}\in\operatorname{span}M$, so $M^{00}\subseteq\Lambda_{\mathcal{V}}(\operatorname{span}M)$.\\
\hlt{Structural Commentary}: Double annihilation returns the span under $\Lambda_{\mathcal{V}}$.\\
\hlt{Key Insight}: Linear functionals separate any vector outside a finite-dimensional span.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:dual-kernel-range-annihilator} \hlt{Kernel and Range of the Dual Map}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\ker(T')=(\operatorname{range}(T))^0$;
\item $\dim(\ker(T'))=\dim(\ker(T))+\dim(\mathcal{W})-\dim(\mathcal{V})$;
\item $\operatorname{range}(T')=(\ker(T))^0$;
\item $\dim(\operatorname{range}(T'))=\dim(\operatorname{range}(T))$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\varphi\in\mathcal{W}'$, $T'\varphi=0$ if and only if $\varphi(T\mathbf{v})=0$ for every $\mathbf{v}\in\mathcal{V}$, if and only if $\varphi$ vanishes on $\operatorname{range}(T)$.
\item By (i), Theorem \ref{thm:dimension-of-annihilator}, and rank-nullity,\\
$\dim(\ker(T'))=\dim(\mathcal{W})-\dim(\operatorname{range}(T)) =\dim(\ker(T))+\dim(\mathcal{W})-\dim(\mathcal{V})$.
\item If $\varphi=T'\psi$ and $\mathbf{v}\in\ker(T)$, then $\varphi(\mathbf{v})=\psi(T\mathbf{v})=0$, so $\operatorname{range}(T')\subseteq(\ker(T))^0$. Also, by rank-nullity for $T'$, (ii), Theorem \ref{thm:dual-basis-and-dimension}, and Theorem \ref{thm:dimension-of-annihilator}, $\dim(\operatorname{range}(T'))=\dim(\mathcal{V})-\dim(\ker(T)) =\dim\big((\ker(T))^0\big)$.\\
Thus $\operatorname{range}(T')=(\ker(T))^0$.
\item By (iii), Theorem \ref{thm:dimension-of-annihilator}, and rank-nullity, $\dim(\operatorname{range}(T'))=\dim(\mathcal{V})-\dim(\ker(T)) =\dim(\operatorname{range}(T))$
\end{enumerate}
\hlt{Structural Commentary}: Duality exchanges kernels and ranges through annihilators.\\
\hlt{Key Insight}: A functional dies after pulling back by $T$ exactly when it vanishes on everything $T$ can reach.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:quotient-dual-annihilator-isomorphism} \hlt{Dual of a Quotient as an Annihilator}\\
Let $\mathcal{V}$ be finite-dimensional, let $\mathcal{U}\leq\mathcal{V}$, and let $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ be the quotient map.\\
Then $\ker(\pi')=\{0\}$ and $\operatorname{range}(\pi')=\mathcal{U}^0$. Hence $\pi'$ identifies $(\mathcal{V}/\mathcal{U})'$ with $\mathcal{U}^0$, so $(\mathcal{V}/\mathcal{U})'\cong\mathcal{U}^0$.
\begin{proof}
By Theorem \ref{thm:dimension-of-quotient}, $\mathcal{V}/\mathcal{U}$ is finite-dimensional. Since $\pi$ is surjective and $\ker(\pi)=\mathcal{U}$, Corollary \ref{cor:surjective-iff-dual-injective} gives $\ker(\pi')=\{0\}$, and Theorem \ref{thm:dual-kernel-range-annihilator} gives $\operatorname{range}(\pi')=(\ker(\pi))^0=\mathcal{U}^0$.\\
\hlt{Structural Commentary}: A functional on the quotient is exactly a functional on $\mathcal{V}$ that cannot see $\mathcal{U}$.\\
\hlt{Key Insight}: Pulling back along the quotient map identifies $(\mathcal{V}/\mathcal{U})'$ with $\mathcal{U}^0$.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:direct-sum-dual-annihilator} \hlt{Direct Sums and Annihilators}\\
Let $\mathcal{V}$ be finite-dimensional, and suppose $\mathcal{V}=\mathcal{S}\oplus\mathcal{T}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Extension by zero on $\mathcal{S}$ gives an isomorphism $\mathcal{T}'\cong\mathcal{S}^0$.
\item $\mathcal{V}'=\mathcal{S}^0\oplus\mathcal{T}^0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Define $E:\mathcal{T}'\to\mathcal{S}^0$ by $(E\varphi)(\mathbf{s}+\mathbf{t})=\varphi(\mathbf{t})$. Since $\mathcal{V}=\mathcal{S}\oplus\mathcal{T}$, this is well-defined and linear. Restriction $R:\mathcal{S}^0\to\mathcal{T}'$, $R\psi=\psi|_{\mathcal{T}}$, satisfies $RE=I_{\mathcal{T}'}$ and $ER=I_{\mathcal{S}^0}$, so $E$ is an isomorphism.
\item By Theorem \ref{thm:annihilator-lattice-identities}, $\mathcal{S}^0+\mathcal{T}^0=(\mathcal{S}\cap\mathcal{T})^0=\{\mathbf{0}\}^0=\mathcal{V}'$, and $\mathcal{S}^0\cap\mathcal{T}^0=(\mathcal{S}+\mathcal{T})^0=\mathcal{V}^0=\{0\}$.\\
Thus $\mathcal{V}'=\mathcal{S}^0\oplus\mathcal{T}^0$.
\end{enumerate}
\hlt{Structural Commentary}: A direct sum decomposes the dual by complementary annihilators.\\
\hlt{Key Insight}: Direct sums split the dual space by annihilators.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:range-membership-dual-alternative} \hlt{Range Membership Dual Alternative}\\
Let $\mathcal{V},\mathcal{W}$ be finite-dimensional, $T\in\LL(\mathcal{V},\mathcal{W})$, and $\mathbf{w}\in\mathcal{W}$. Exactly one holds:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathbf{w}\in\operatorname{range}(T)$;
\item there exists $\psi\in\mathcal{W}'$ such that $\psi T=0$ and $\psi(\mathbf{w})\neq0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The alternatives are incompatible: if $\mathbf{w}=T\mathbf{v}$ and $\psi T=0$, then $\psi(\mathbf{w})=\psi(T\mathbf{v})=0$.
\item Suppose $\mathbf{w}\notin\operatorname{range}(T)$. Let $\mathcal{U}=\operatorname{range}(T)$, choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_r)$ of $\mathcal{U}$, and extend $(\mathbf{u}_1,\ldots,\mathbf{u}_r,\mathbf{w})$ to a basis of $\mathcal{W}$. Define $\psi\in\mathcal{W}'$ by $\psi(\mathbf{u}_j)=0$, $\psi(\mathbf{w})=1$, and $\psi=0$ on the added basis vectors.\\
Then $\psi$ vanishes on $\operatorname{range}(T)$, so $\psi T=0$, while $\psi(\mathbf{w})=1$.
\end{enumerate}
\hlt{Structural Commentary}: A missing range vector is detected by a dual test that vanishes on the whole range.\\
\hlt{Key Insight}: Either a vector is reached by the map, or a functional separates it from the range.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:matrix-equation-left-nullspace-alternative} \hlt{Matrix Equation Alternative}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{b}\in\mathbb{F}^m$. Exactly one holds:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\mathbf{x}=\mathbf{b}$ for some $\mathbf{x}\in\mathbb{F}^n$;
\item $\mathbf{y}^tA=\mathbf{0}^t$ and $\mathbf{y}^t\mathbf{b}\neq0$ for some $\mathbf{y}\in\mathbb{F}^m$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorem \ref{thm:range-membership-dual-alternative}, for $T_A:\mathbb{F}^n\to\mathbb{F}^m$ and $\mathbf{w}=\mathbf{b}$, $\mathbf{b}\in\operatorname{range}(T_A)$ is exactly $A\mathbf{x}=\mathbf{b}$ for some $\mathbf{x}\in\mathbb{F}^n$.
\item Under the standard dual identification, $\psi\in(\mathbb{F}^m)'$ has the form $\psi(\mathbf{z})=\mathbf{y}^t\mathbf{z}$.\\
Thus $\psi T_A=0$ iff $\mathbf{y}^tA=\mathbf{0}^t$, and $\psi(\mathbf{b})\neq0$ iff $\mathbf{y}^t\mathbf{b}\neq0$.
\end{enumerate}
\hlt{Structural Commentary}: Inconsistency is certified by a left-null vector.\\
\hlt{Key Insight}: Left null vectors are the coordinate form of dual separating functionals.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:surjective-iff-dual-injective} \hlt{Surjectivity and Dual Injectivity}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then $T$ is surjective if and only if $T'$ is injective.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $T$ is surjective. Then $\operatorname{range}(T)=\mathcal{W}$, so Theorem \ref{thm:dual-kernel-range-annihilator} gives $\ker(T')=\mathcal{W}^0=\{0\}$.
\item[$(\Leftarrow)$] Suppose $T'$ is injective. Then $\ker(T')=\{0\}$, so Theorem \ref{thm:dual-kernel-range-annihilator} gives $(\operatorname{range}(T))^0=\{0\}$. By Theorem \ref{thm:dimension-of-annihilator}, $\dim(\operatorname{range}(T))=\dim(\mathcal{W})$, hence $\operatorname{range}(T)=\mathcal{W}$.
\end{enumerate}
\hlt{Structural Commentary}: A map is onto exactly when no nonzero functional can vanish on its range. Dual injectivity detects this absence of invisible codomain directions.\\
\hlt{Key Insight}: Surjectivity of $T$ becomes injectivity of the pullback map $T'$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:injective-iff-dual-surjective} \hlt{Injectivity and Dual Surjectivity}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional, and let $T\in\LL(\mathcal{V},\mathcal{W})$. Then $T$ is injective if and only if $T'$ is surjective.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $T$ is injective. Then $\ker(T)=\{\mathbf{0}\}$, so Theorem \ref{thm:dual-kernel-range-annihilator} gives $\operatorname{range}(T')=(\ker(T))^0=\mathcal{V}'$.
\item[$(\Leftarrow)$] Suppose $T'$ is surjective. Then $\operatorname{range}(T')=\mathcal{V}'$, so Theorem \ref{thm:dual-kernel-range-annihilator} gives $(\ker(T))^0=\mathcal{V}'$. By Theorem \ref{thm:dimension-of-annihilator}, this forces $\dim(\ker(T))=0$, hence $\ker(T)=\{\mathbf{0}\}$.
\end{enumerate}
\hlt{Structural Commentary}: Injectivity says there are no nonzero domain directions killed by $T$. The dual statement says every test on the domain comes by pulling back a test from the codomain.\\
\hlt{Key Insight}: No kernel directions means no obstruction to extending functionals backward through $T$.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:matrix-of-dual-map-is-transpose} \hlt{Matrix of the Dual Map Is the Transpose}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional, let $T\in\LL(\mathcal{V},\mathcal{W})$, and let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_m)$ be bases of $\mathcal{V}$ and $\mathcal{W}$. If $\beta'$ and $\gamma'$ are the corresponding dual bases, then $[T']_{\beta'\leftarrow\gamma'}=\left([T]_{\gamma\leftarrow\beta}\right)^t$.
\begin{proof}
Let $A=[T]_{\gamma\leftarrow\beta}$ and $C=[T']_{\beta'\leftarrow\gamma'}$. Write $\gamma'=(\psi_1,\ldots,\psi_m)$ and $\beta'=(\varphi_1,\ldots,\varphi_n)$.\\
For $1\leq j\leq m$, $T'\psi_j=\sum\limits_{r=1}^n C_{r,j}\varphi_r$.
Evaluating at $\mathbf{v}_k$ gives $C_{k,j}=(T'\psi_j)(\mathbf{v}_k)=\psi_j(T\mathbf{v}_k)$.\\
Since $T\mathbf{v}_k=\sum\limits_{r=1}^m A_{r,k}\mathbf{w}_r$, we get $\psi_j(T\mathbf{v}_k)=A_{j,k}$. Hence $C_{k,j}=A_{j,k}$ for all $j,k$, so $C=A^t$.\\
\hlt{Structural Commentary}: Duality turns the matrix of a map into the matrix of its pullback on coordinate functionals. The row-column indexing is exactly transposition.\\
\hlt{Key Insight}: Coefficients of $T\mathbf{v}_k$ become values of the dual basis functionals on $T\mathbf{v}_k$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:bilinear-form-and-matrix} \hlt{Bilinear Forms and Matrices}\\
A map $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ is \hlt{bilinear} if it is linear in each variable separately.\\
For bases $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ of $\mathcal{V}$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ of $\mathcal{W}$, the \hlt{matrix of $B$ in the bases $\beta,\gamma$} is $[B]_{\beta,\gamma}=[B(\mathbf{v}_i,\mathbf{w}_j)]$. If $\mathbf{x}=[\mathbf{v}]_{\beta}$ and $\mathbf{y}=[\mathbf{w}]_{\gamma}$, then $B(\mathbf{v},\mathbf{w})=\mathbf{x}^t [B]_{\beta,\gamma}\mathbf{y}$.
\end{definition}

\begin{proposition} \label{prop:bilinear-forms-linearize-through-duals} \hlt{Bilinear Forms Linearize through Duals}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear, and define $\Phi_B:\mathcal{V}\to\mathcal{W}'$ where $(\Phi_B\mathbf{v})(\mathbf{w})=B(\mathbf{v},\mathbf{w})$,\\
and $\Psi_B:\mathcal{W}\to\mathcal{V}'$ where $(\Psi_B\mathbf{w})(\mathbf{v})=B(\mathbf{v},\mathbf{w})$. Then $\Phi_B,\Psi_B$ are linear.\\
For bases $\beta,\gamma$ and $M=[B]_{\beta,\gamma}$, $[\Phi_B]_{\gamma'\leftarrow\beta}=M^t$; and $[\Psi_B]_{\beta'\leftarrow\gamma}=M$.
\begin{proof}
For fixed $\mathbf{v}$ and $\mathbf{w}$, bilinearity gives $\Phi_B\mathbf{v}\in\mathcal{W}'$ and $\Psi_B\mathbf{w}\in\mathcal{V}'$. For $\mathbf{v}_1,\mathbf{v}_2\in\mathcal{V}$, $a\in\mathbb{F}$, and every $\mathbf{w}\in\mathcal{W}$, $(\Phi_B(a\mathbf{v}_1+\mathbf{v}_2))(\mathbf{w}) =B(a\mathbf{v}_1+\mathbf{v}_2,\mathbf{w}) =a(\Phi_B\mathbf{v}_1)(\mathbf{w})+(\Phi_B\mathbf{v}_2)(\mathbf{w})$, so $\Phi_B$ is linear.\\
For $\mathbf{w}_1,\mathbf{w}_2\in\mathcal{W}$, $a\in\mathbb{F}$, and every $\mathbf{v}\in\mathcal{V}$, $(\Psi_B(a\mathbf{w}_1+\mathbf{w}_2))(\mathbf{v}) =B(\mathbf{v},a\mathbf{w}_1+\mathbf{w}_2) =a(\Psi_B\mathbf{w}_1)(\mathbf{v})+(\Psi_B\mathbf{w}_2)(\mathbf{v})$,  so $\Psi_B$ is linear. Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$.\\
The $j$th $\gamma'$-coordinate of $\Phi_B(\mathbf{v}_i)$ is $(\Phi_B\mathbf{v}_i)(\mathbf{w}_j)=B(\mathbf{v}_i,\mathbf{w}_j)=M_{i,j}$, so column $i$ of $[\Phi_B]_{\gamma'\leftarrow\beta}$ is row $i$ of $M$ transposed, hence $[\Phi_B]_{\gamma'\leftarrow\beta}=M^t$. Likewise, the $i$th $\beta'$-coordinate of $\Psi_B(\mathbf{w}_j)$ is $M_{i,j}$, so $[\Psi_B]_{\beta'\leftarrow\gamma}=M$.\\
\hlt{Structural Commentary}: A bilinear form becomes linear once one input is frozen. Tracking both freezes gives two dual-valued linear maps whose matrices are transposes of one another.\\
\hlt{Key Insight}: Bilinearity is linearity into a dual space.
\end{proof}
\end{proposition}

\begin{definition} \label{def:bilinear-left-right-radicals} \hlt{Radicals and Nonsingularity of a Bilinear Pairing}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{left} and \hlt{right radicals} are
\[
\operatorname{rad}_{L}(B)\defeq\{\mathbf{v}\in\mathcal{V}:B(\mathbf{v},\mathcal{W})=0\},
\qquad
\operatorname{rad}_{R}(B)\defeq\{\mathbf{w}\in\mathcal{W}:B(\mathcal{V},\mathbf{w})=0\}.
\]
\item $B$ is \hlt{left nondegenerate} if $\operatorname{rad}_{L}(B)=\{\mathbf{0}\}$ and \hlt{right nondegenerate} if $\operatorname{rad}_{R}(B)=\{\mathbf{0}\}$.
\item $B$ is \hlt{nonsingular} or \hlt{perfect} if $\Phi_B:\mathcal{V}\to\mathcal{W}'$ and $\Psi_B:\mathcal{W}\to\mathcal{V}'$ from Proposition \ref{prop:bilinear-forms-linearize-through-duals} are isomorphisms.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:bilinear-radicals-matrix-and-dual-maps} \hlt{Radicals, Matrices, and Dual Maps}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear, $\dim\mathcal{V}=m$, $\dim\mathcal{W}=n$, and $M=[B]_{\beta,\gamma}$ in ordered bases $\beta,\gamma$.\\
Then $[\operatorname{rad}_{L}(B)]_{\beta}=\ker(M^t)$ and $[\operatorname{rad}_{R}(B)]_{\gamma}=\ker(M)$,\\
therefore $\dim\operatorname{rad}_{L}(B)=m-\operatorname{rank}(M)$ and $\dim\operatorname{rad}_{R}(B)=n-\operatorname{rank}(M)$.\\
Moreover, $\ker(\Phi_B)=\operatorname{rad}_{L}(B)$ and $\ker(\Psi_B)=\operatorname{rad}_{R}(B)$.
\begin{proof}
For $\mathbf{x}=[\mathbf{v}]_{\beta}$ and $\mathbf{y}=[\mathbf{w}]_{\gamma}$, $B(\mathbf{v},\mathbf{w})=\mathbf{x}^{t}M\mathbf{y}$.\\
Thus $\mathbf{v}\in\operatorname{rad}_{L}(B)$ iff $\mathbf{x}^{t}M\mathbf{y}=0$ for all $\mathbf{y}$ iff $M^t\mathbf{x}=0$. Similarly, $\mathbf{w}\in\operatorname{rad}_{R}(B)$ iff $M\mathbf{y}=0$. Rank-nullity gives dimension formulas. Finally, $\mathbf{v}\in\ker(\Phi_B)$ $\Longleftrightarrow$ $B(\mathbf{v},\mathcal{W})=0$ $\Longleftrightarrow$ $\mathbf{v}\in\operatorname{rad}_{L}(B)$; similarly $\ker(\Psi_B)=\operatorname{rad}_{R}(B)$.\\
\hlt{Structural Commentary}: Radicals are the kernel obstructions in each slot.\\
\hlt{Key Insight}: In coordinates, radicals are matrix nullspaces.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:square-bilinear-pairing-equivalences} \hlt{Square Pairing Nonsingularity Criteria}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be bilinear with $\dim\mathcal{V}=\dim\mathcal{W}=n$, and let $M=[B]_{\beta,\gamma}$ in ordered bases $\beta,\gamma$.\\
Then the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $B$ is left nondegenerate;
\item $B$ is right nondegenerate;
\item $M$ is invertible;
\item $B$ is perfect.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$] $\Rightarrow$ $(ii)$. If $\operatorname{rad}_{L}(B)=0$, then by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}, $\ker(M^t)=0$ $\Rightarrow$ $\operatorname{rank}(M)=n$ $\Rightarrow$ $\ker(M)=0$.
\item[$(ii)$] $\Rightarrow$ $(iii)$. If $\operatorname{rad}_{R}(B)=0$, then $\ker(M)=0$ by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}; since $M\in\operatorname{Mat}(n,n,\mathbb{F})$, $M$ is invertible.
\item[$(iii)$] $\Rightarrow$ $(iv)$. If $M$ is invertible, then $\ker(M)=\ker(M^t)=0$, so $\operatorname{rad}_{L}(B)=\operatorname{rad}_{R}(B)=0$.\\
Hence $\ker(\Phi_B)=\ker(\Psi_B)=0$ by Proposition \ref{prop:bilinear-radicals-matrix-and-dual-maps}.\\
Since $\dim\mathcal{V}=\dim\mathcal{W}'=\dim\mathcal{W}=\dim\mathcal{V}'=n$, $\Phi_B$ and $\Psi_B$ are isomorphisms; hence $B$ is perfect.
\item[$(iv)$] $\Rightarrow$ $(i)$. If $B$ is perfect, then $\Phi_B$ is an isomorphism. Thus $\operatorname{rad}_{L}(B)=\ker(\Phi_B)=0$.
\end{enumerate}
\hlt{Structural Commentary}: In square pairings, one-sided nondegeneracy is invertibility.\\
\hlt{Key Insight}: Perfect pairings are exactly invertible matrices viewed without coordinates.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:perfect-pairing-represents-functionals} \hlt{Perfect Pairings Represent Functionals}\\
Let $B:\mathcal{V}\times\mathcal{W}\to\mathbb{F}$ be perfect.\\
Then $\forall\psi\in\mathcal{W}'\ \exists!\mathbf{v}\in\mathcal{V}: \psi(\mathbf{w})=B(\mathbf{v},\mathbf{w})$, $\mathbf{w}\in\mathcal{W}$; and $\forall\varphi\in\mathcal{V}'\ \exists!\mathbf{w}\in\mathcal{W}: \varphi(\mathbf{v})=B(\mathbf{v},\mathbf{w})$, $\mathbf{v}\in\mathcal{V}$.
\begin{proof}
Since $B$ is perfect, $\Phi_B:\mathcal{V}\to\mathcal{W}'$ and $\Psi_B:\mathcal{W}\to\mathcal{V}'$ are isomorphisms.\\
Thus $\psi=\Phi_B(\mathbf{v})$ and $\varphi=\Psi_B(\mathbf{w})$ for unique $\mathbf{v},\mathbf{w}$, which gives the two formulas.\\
\hlt{Structural Commentary}: Perfect pairings identify each space with the other's dual.\\
\hlt{Key Insight}: Functionals are represented by vectors exactly when the appropriate pairing map is onto.
\end{proof}
\end{corollary}

\begin{definition} \label{def:matrix-congruence} \hlt{Matrix Congruence}\\
Matrices $M,N\in\operatorname{Mat}(n,n,\mathbb{F})$ are \hlt{congruent} if $\exists\,P\in\operatorname{Mat}(n,n,\mathbb{F})$ invertible such that $N=P^tMP$.
\end{definition}

\begin{proposition} \label{prop:square-bilinear-form-change-of-basis-congruence} \hlt{Basis Change for a Square Bilinear Form}\\
Let $\mathcal{V}$ be finite-dimensional, let $B:\mathcal{V}\times\mathcal{V}\to\mathbb{F}$ be bilinear, and let $\beta,\gamma$ be ordered bases of $\mathcal{V}$.\\
Set $M_{\beta}\defeq [B]_{\beta,\beta}$; $M_{\gamma}\defeq [B]_{\gamma,\gamma}$, and $C\defeq C_{\beta\leftarrow\gamma}$.\\
Then $M_{\gamma}=C^tM_{\beta}C$. Hence $M_{\beta}$ and $M_{\gamma}$ are congruent.
\begin{proof}
For $\mathbf{x},\mathbf{y}\in\mathcal{V}$, $[\mathbf{x}]_{\beta}=C[\mathbf{x}]_{\gamma}$ and $[\mathbf{y}]_{\beta}=C[\mathbf{y}]_{\gamma}$. Therefore $B(\mathbf{x},\mathbf{y}) =[\mathbf{x}]_{\beta}^{t}M_{\beta}[\mathbf{y}]_{\beta} =[\mathbf{x}]_{\gamma}^{t}C^tM_{\beta}C[\mathbf{y}]_{\gamma}$.\\
By uniqueness of the representing matrix in the basis $\gamma$, $M_{\gamma}=C^tM_{\beta}C$.\\
Since $C$ is invertible by Theorem \ref{thm:change-of-basis-matrices}, Definition \ref{def:matrix-congruence} gives congruence.\\
\hlt{Structural Commentary}: Operators change by similarity; square bilinear forms change by congruence because both input slots are rewritten.\\
\hlt{Key Insight}: A basis change acts on a square bilinear form from both sides.
\end{proof}
\end{proposition}
