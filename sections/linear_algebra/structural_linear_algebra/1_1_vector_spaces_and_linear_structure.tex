\subsection{Vector Spaces and Linear Structure}

% **Role:** algebraic language of linear structure.
%
% Fields, vector spaces, subspaces, span, independence, bases, dimension, coordinates.

Vector spaces encode addition and scalar multiplication. Subspaces, spans, bases, dimension, and coordinates describe how linear structure is generated and measured.

\begin{definition} \label{def:vector-space} \hlt{Vector Space}
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.vector_space_axioms}
\leanok\\
Let $\mathbb{F}$ be a field. A vector space over $\mathbb{F}$ is a set $\mathcal{V}$ with vector addition and scalar multiplication such that, $\forall\,\mathbf{u}, \mathbf{v}, \mathbf{w} \in \mathcal{V}$ and $\forall\,a,b \in \mathbb{F}$, the following axioms hold:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item vector addition is commutative: $\mathbf{u} + \mathbf{v} = \mathbf{v} + \mathbf{u}$;
\item vector addition is associative: $(\mathbf{u} + \mathbf{v}) + \mathbf{w} = \mathbf{u} + (\mathbf{v} + \mathbf{w})$;
\item there exists an additive identity $\mathbf{0} \in \mathcal{V}$ such that $\mathbf{0} + \mathbf{v} = \mathbf{v}$;
\item for each $\mathbf{v} \in \mathcal{V}$ there exists an additive inverse $-\mathbf{v} \in \mathcal{V}$ such that $\mathbf{v} + (-\mathbf{v}) = \mathbf{0}$;
\item scalar multiplication has identity: $1\mathbf{v} = \mathbf{v}$;
\item scalar multiplication is compatible with multiplication in $\mathbb{F}$: $(ab)\mathbf{v} = a(b\mathbf{v})$;
\item scalar multiplication distributes over vector addition: $a(\mathbf{u} + \mathbf{v}) = a\mathbf{u} + a\mathbf{v}$;
\item scalar multiplication distributes over addition in $\mathbb{F}$: $(a+b)\mathbf{v} = a\mathbf{v} + b\mathbf{v}$.
\end{enumerate}
If $\mathcal{V}=\{\mathbf{0}\}$, then $\mathcal{V}$ is a \hlt{zero vector space}; otherwise $\mathcal{V}$ is a \hlt{nonzero vector space}.
\end{definition}

\begin{definition} \label{def:finitely-nonzero-sequence-space} \hlt{Finitely Nonzero Sequence Space}
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.FinitelyNonzeroSequenceSpace}
\leanok\\
Let $\mathbb{F}^{(\mathbb{N})}$ denote the set of sequences $(x_1,x_2,\ldots)$ in $\mathbb{F}$ with finite support, i.e. $x_j=0$ for all but finitely many $j\in\mathbb{N}$. Addition and scalar multiplication are componentwise.
\end{definition}

\begin{proposition} \label{prop:finitely-nonzero-sequences-vector-space} \hlt{Finitely Nonzero Sequences Form a Vector Space}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.finitely_nonzero_sequences_vector_space}
\leanok
The set $\mathbb{F}^{(\mathbb{N})}$ is a vector space over $\mathbb{F}$ under componentwise addition and scalar multiplication.
\begin{proof}
\leanok
The zero sequence has finite support.\\
If $\mathbf{x},\mathbf{y}\in\mathbb{F}^{(\mathbb{N})}$ and $a\in\mathbb{F}$, then $a\mathbf{x}+\mathbf{y}$ can be nonzero only where $\mathbf{x}$ or $\mathbf{y}$ is nonzero, a finite set of indices.\\
Hence $a\mathbf{x}+\mathbf{y}\in\mathbb{F}^{(\mathbb{N})}$, and the vector space axioms are inherited componentwise from $\mathbb{F}$.\\
\hlt{Structural Commentary}: Finite support is preserved by finite linear combinations.\\
\hlt{Key Insight}: Componentwise operations cannot create infinitely many new nonzero entries.
\end{proof}
\end{proposition}

\begin{example} \label{ex:roman-vector-space-examples} \hlt{Standard Vector Space Models}\\
Let $\mathbb{F}$ be a field and let $X$ be a set. The following are $\mathbb{F}$-vector spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathbb{F}^{X}=\{f:X\to\mathbb{F}\}$, with pointwise operations
$(f+g)(x)=f(x)+g(x)$ and $(af)(x)=a f(x)$.
\item $\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbb{F}^n$, with entrywise operations. For $\mathbb{F}=\mathbb{F}_q$, set $V(n,q):=\mathbb{F}_q^n$.
\item $\operatorname{Seq}(\mathbb{F})=\mathbb{F}^{\mathbb{N}}$, with componentwise operations; $\mathbb{F}^{(\mathbb{N})}$ is its finite-support subspace.
\item Over $\mathbb{C}$, the sequence spaces $c_0=\{(z_n):z_n\to0\}$, $\ell^\infty=\{(z_n):\sup_n\abs{z_n}<\infty\}$, and, for $1\leq p<\infty$, $\ell^p=\left\{(z_n):\sum\limits_{n=1}^{\infty}\abs{z_n}^{p}<\infty\right\}$ are vector spaces under componentwise operations; for $\ell^p$, closure under addition follows from Minkowski's inequality.
\end{enumerate}
\hlt{Structural Commentary}: The vector-space axioms are independent of finite-dimensional coordinates; function and sequence spaces give canonical infinite-dimensional models.\\
\hlt{Key Insight}: Pointwise or componentwise operations inherit linear structure from the scalar field.
\end{example}

\begin{theorem} \label{thm:basicvectorspaceconsequences} \hlt{Basic Consequences of Vector Space Axioms}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.basic_vector_space_consequences}
\leanok
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item the additive identity is unique;
\item for each $\mathbf{v}\in\mathcal{V}$, the additive inverse of $\mathbf{v}$ is unique;
\item $\forall \mathbf{v}\in\mathcal{V}$, $0\mathbf{v}=\mathbf{0}$;
\item $\forall a\in\mathbb{F}$, $a\mathbf{0}=\mathbf{0}$;
\item $\forall a\in\mathbb{F}$ and $\mathbf{v}\in\mathcal{V}$, $a\mathbf{v}=\mathbf{0}\iff a=0$ or $\mathbf{v}=\mathbf{0}$.
\end{enumerate}
\begin{proof}
\leanok
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathbf{0}$ and $\mathbf{0}'$ are additive identities, then $\mathbf{0}=\mathbf{0}+\mathbf{0}'=\mathbf{0}'$.
\item If $\mathbf{w},\mathbf{w}'$ are additive inverses of $\mathbf{v}$, then $\mathbf{w}=\mathbf{w}+\mathbf{0}=\mathbf{w}+(\mathbf{v}+\mathbf{w}') =(\mathbf{w}+\mathbf{v})+\mathbf{w}'=\mathbf{0}+\mathbf{w}'=\mathbf{w}'$.
\item $0\mathbf{v}=(0+0)\mathbf{v}=0\mathbf{v}+0\mathbf{v}$; adding $-(0\mathbf{v})$ gives $0\mathbf{v}=\mathbf{0}$.
\item $a\mathbf{0}=a(\mathbf{0}+\mathbf{0})=a\mathbf{0}+a\mathbf{0}$; adding $-(a\mathbf{0})$ gives $a\mathbf{0}=\mathbf{0}$.
\item[$(v)$] ($\Rightarrow$) Suppose $a\mathbf{v}=\mathbf{0}$. If $a=0$, the first alternative holds.\\
If $a\neq0$, then $\mathbf{v}=1\mathbf{v}=(a^{-1}a)\mathbf{v}=a^{-1}(a\mathbf{v})=a^{-1}\mathbf{0}=\mathbf{0}$.
\item[$(v)$] ($\Leftarrow$) If $a=0$ or $\mathbf{v}=\mathbf{0}$, then (iii) or (iv) gives $a\mathbf{v}=\mathbf{0}$.
\end{enumerate}
\hlt{Structural Commentary}: These identities make cancellation, zeros, and inverses available.\\
\hlt{Key Insight}: The vector space axioms force the zero and inverse behaviour to be rigid, not optional structure.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:negativeoneactsasinverse} \hlt{Scalar Multiplication by $-1$ Is Negation}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.negative_one_smul_eq_neg_vector}
\leanok
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then $\forall\,\mathbf{v}\in\mathcal{V}$, $(-1)\mathbf{v}=-\mathbf{v}$.
\begin{proof}
\leanok
For $\mathbf{v}\in\mathcal{V}$, $(-1)\mathbf{v} + \mathbf{v} = (-1)\mathbf{v} + 1\mathbf{v} = (-1+1)\mathbf{v} = 0\mathbf{v} = \mathbf{0}$.\\
Hence $(-1)\mathbf{v}$ is an additive inverse of $\mathbf{v}$; by uniqueness, $(-1)\mathbf{v} = -\mathbf{v}$.\\
\hlt{Structural Commentary}: Scalar multiplication by $-1$ is exactly the same operation as taking an additive inverse.\\
\hlt{Key Insight}: Negation is not an extra operation; it is already encoded by the field action.
\end{proof}
\end{corollary}

\begin{definition} \label{def:subspace} \hlt{Subspace}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.Subspace}
\leanok
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. A subset $\mathcal{U}\subseteq\mathcal{V}$ is a \hlt{subspace}, written $\mathcal{U}\leq\mathcal{V}$, if $\mathcal{U}$ is a vector space over $\mathbb{F}$ under the operations inherited from $\mathcal{V}$.
\end{definition}

\begin{theorem} \label{thm:subspacetest} \hlt{Subspace Test}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.subspace_test}
\leanok
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$ and let $\mathcal{U}$ be a nonempty subset of $\mathcal{V}$.\\
Then $\mathcal{U}$ is a subspace of $\mathcal{V}$ iff $\forall\,\mathbf{u},\mathbf{v}\in\mathcal{U}$ and $\forall\,c\in\mathbb{F}$, one has $c\mathbf{u}+\mathbf{v}\in\mathcal{U}$.
\begin{proof}
\leanok
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathcal{U}\leq\mathcal{V}$, then closure under scalar multiplication and addition gives $c\mathbf{u}+\mathbf{v}\in\mathcal{U}$ for all $\mathbf{u},\mathbf{v}\in\mathcal{U}$ and $c\in\mathbb{F}$.
\item[$(\Leftarrow)$] Assume the stated closure condition. Choose $\mathbf{u}_0\in\mathcal{U}$. Then $\mathbf{0}=(-1)\mathbf{u}_0+\mathbf{u}_0\in\mathcal{U}$. For $\mathbf{u},\mathbf{v}\in\mathcal{U}$ and $c\in\mathbb{F}$, the identities $-\mathbf{u}=(-1)\mathbf{u}+\mathbf{0}$, $c\mathbf{u}=c\mathbf{u}+\mathbf{0}$, and $\mathbf{u}+\mathbf{v}=1\mathbf{u}+\mathbf{v}$ give closure under additive inverses, scalar multiplication, and addition. The remaining axioms are inherited from $\mathcal{V}$, so $\mathcal{U}\leq\mathcal{V}$.
\end{enumerate}
\hlt{Structural Commentary}: One closure condition captures the subspace axioms.\\
\hlt{Key Insight}: To prove a subset is a subspace, prove it survives one affine-looking linear combination.
\end{proof}
\end{theorem}

\begin{definition} \label{def:linear-code-and-hamming-weight} \hlt{Linear Code and Hamming Weight}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{ambient code space} is $V(n,q)\defeq\mathbb{F}_q^n$.
\item A \hlt{linear code} of length $n$ over $\mathbb{F}_q$ is a \hlt{subspace} $C\leq V(n,q)$.
\item For $\mathbf{x}=(x_1,\ldots,x_n)\in\mathbb{F}_q^n$, its \hlt{Hamming weight} is $\operatorname{wt}(\mathbf{x})\defeq\#\{j:1\leq j\leq n,\ x_j\neq0\}$.
\end{enumerate}
\end{definition}

\begin{example} \label{ex:linear-codes-even-weight-subspace} \hlt{Even-Weight Code}\\
In $V(n,2)=\mathbb{F}_2^n$, let $E_n=\left\{\mathbf{x}\in\mathbb{F}_2^n:\sum\limits_{j=1}^{n}x_j=0\right\}$.
The condition $\sum\limits_{j=1}^{n}x_j=0$ is equivalent to $\operatorname{wt}(\mathbf{x})$ being even. If $\mathbf{x},\mathbf{y}\in E_n$ and $c\in\mathbb{F}_2$, then $\sum\limits_{j=1}^{n}(cx_j+y_j)=c\sum\limits_{j=1}^{n}x_j+\sum\limits_{j=1}^{n}y_j=0$, so $c\mathbf{x}+\mathbf{y}\in E_n$. Hence $E_n\leq V(n,2)$ by Theorem \ref{thm:subspacetest}, and $E_n$ is a linear code.\\
\hlt{Structural Commentary}: The even-weight code is the parity-zero subspace in $V(n,2)$.\\
\hlt{Key Insight}: Parity is linear over $\mathbb{F}_2$.
\end{example}

\begin{proposition} \label{prop:intersectionsubspaces} \hlt{Intersection of Subspaces}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$. Then $\mathcal{U}\cap\mathcal{W} = \left\{ \mathbf{v} : \mathbf{v}\in\mathcal{U} \text{ and } \mathbf{v}\in\mathcal{W} \right\}$ is a subspace of $\mathcal{V}$.
\begin{proof}
Since $\mathbf{0}\in\mathcal{U}$ and $\mathbf{0}\in\mathcal{W}$, we have $\mathbf{0}\in\mathcal{U}\cap\mathcal{W}$, so the intersection is nonempty.\\
If $\mathbf{u},\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$ and $c\in\mathbb{F}$, then $c\mathbf{u}+\mathbf{v}$ lies in both $\mathcal{U}$ and $\mathcal{W}$ because each is a subspace.\\
Thus $c\mathbf{u}+\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$, and Theorem \ref{thm:subspacetest} gives $\mathcal{U}\cap\mathcal{W}\leq\mathcal{V}$.\\
\hlt{Structural Commentary}: Intersections inherit shared closure.\\
\hlt{Key Insight}: Common vectors remain closed under linear combinations.
\end{proof}
\end{proposition}

\begin{definition} \label{def:linear-combinations-and-span} \hlt{Lists, Linear Combinations, and Span}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{list} in $\mathcal{V}$ is $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$; the empty list is $()$.
\item For $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ and $a_1,\ldots,a_m\in\mathbb{F}$, a \hlt{linear combination} of $\beta$ is $\sum\limits_{j=1}^{m}a_j\mathbf{v}_j$.
\item For $E\subseteq\mathcal{V}$, the \hlt{span} of $E$ is
$\operatorname{span}(E)\defeq\left\{\sum\limits_{j=1}^{m}a_j\mathbf{v}_j:m\in\N,\ a_j\in\mathbb{F},\ \mathbf{v}_j\in E\right\}$, with $\operatorname{span}(\emptyset)=\{\mathbf{0}\}$.
\item For $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$, the \hlt{span} of $\beta$ is $\operatorname{span}(\beta)\defeq\operatorname{span}(\{\mathbf{v}_1,\ldots,\mathbf{v}_m\})$.
\item $E\subseteq\mathcal{V}$ \hlt{spans} $\mathcal{V}$ iff $\operatorname{span}(E)=\mathcal{V}$; a list $\beta$ \hlt{spans} $\mathcal{V}$ iff $\operatorname{span}(\beta)=\mathcal{V}$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:span-smallest-containing-subspace} \hlt{Span is the Smallest Containing Subspace}\\
Let $E\subseteq\mathcal{V}$. Then $\operatorname{span}(E)$ is the smallest subspace of $\mathcal{V}$ containing $E$.
\begin{proof}
If $E=\emptyset$, then $\operatorname{span}(E)=\{\mathbf{0}\}$, which is a subspace. Suppose $E\neq\emptyset$. For $\mathbf{x},\mathbf{y}\in\operatorname{span}(E)$ and $c\in\mathbb{F}$, both $\mathbf{x}$ and $\mathbf{y}$ are finite linear combinations of vectors in $E$, so $c\mathbf{x}+\mathbf{y}$ is also such a combination. Hence $c\mathbf{x}+\mathbf{y}\in\operatorname{span}(E)$, and Theorem \ref{thm:subspacetest} makes $\operatorname{span}(E)$ a subspace.\\
Each $\mathbf{v}\in E$ satisfies $\mathbf{v}=1\mathbf{v}\in\operatorname{span}(E)$, so $E\subseteq\operatorname{span}(E)$. If $\mathcal{U}\leq\mathcal{V}$ and $E\subseteq\mathcal{U}$, then closure under finite linear combinations gives $\operatorname{span}(E)\subseteq\mathcal{U}$. Thus $\operatorname{span}(E)$ is the smallest subspace containing $E$.\\
\hlt{Structural Commentary}: Span closes a set of vectors under finite linear combinations, giving exactly the subspace forced by those vectors.\\
\hlt{Key Insight}: Nothing beyond linear closure is added. Every containing subspace must already contain span.
\end{proof}
\end{theorem}

\begin{method} \label{meth:subspace-diagnostic} \hlt{Subspace Diagnostic}\\
Let $S\subseteq\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item To prove $S\leq\mathcal{V}$, show $S\neq\varnothing$ and $\forall\,\mathbf{u},\mathbf{v}\in S,\ \forall\,c\in\mathbb{F}$, $c\mathbf{u}+\mathbf{v}\in S$.
\item To disprove $S\leq\mathcal{V}$, exhibit $\mathbf{u},\mathbf{v}\in S$ and $c\in\mathbb{F}$ with $c\mathbf{u}+\mathbf{v}\notin S$; $\mathbf{0}\in S$ alone is insufficient.
\item The generated subspace is $\langle S\rangle=\operatorname{span}(S)=\left\{\sum\limits_{i=1}^{n}a_i\mathbf{s}_i:n\in\mathbb{N},\ a_i\in\mathbb{F},\ \mathbf{s}_i\in S\right\}$.
\end{enumerate}
\end{method}

\begin{proposition} \label{prop:spanmonotonicity} \hlt{Monotonicity of Span}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subsets of $\mathcal{V}$. If $\mathcal{U}\subseteq\mathcal{W}$, then $\operatorname{span}(\mathcal{U})\subseteq\operatorname{span}(\mathcal{W})$.
\begin{proof}
If $\mathcal{U}=\emptyset$, then $\operatorname{span}(\mathcal{U})=\{\mathbf{0}\}\subseteq\operatorname{span}(\mathcal{W})$.\\
Suppose $\mathcal{U}\neq\emptyset$. Every vector in $\operatorname{span}(\mathcal{U})$ is a finite linear combination of vectors in $\mathcal{U}$; since $\mathcal{U}\subseteq\mathcal{W}$, the same expression is a finite linear combination of vectors in $\mathcal{W}$. Hence $\operatorname{span}(\mathcal{U})\subseteq\operatorname{span}(\mathcal{W})$.\\
\hlt{Structural Commentary}: Enlarging the generating set can only enlarge the collection of linear combinations.\\
\hlt{Key Insight}: Span is monotone with respect to inclusion of generating sets.
\end{proof}
\end{proposition}

\begin{definition} \label{def:linear-independence-basis-dimension} \hlt{Linear Independence, Bases, and Dimension}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A list $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is \hlt{linearly independent} if $\sum\limits_{j=1}^{m}a_j\mathbf{v}_j=\mathbf{0}\Rightarrow a_1=\cdots=a_m=0$.\\
Otherwise it is \hlt{linearly dependent}. The empty list is linearly independent.
\item A list $\beta$ is a \hlt{basis} of $\mathcal{V}$ if $\beta$ is linearly independent and $\operatorname{span}(\beta)=\mathcal{V}$.
\item $\mathcal{V}$ is \hlt{finite-dimensional} if some finite list $\beta$ satisfies $\operatorname{span}(\beta)=\mathcal{V}$; otherwise $\mathcal{V}$ is \hlt{infinite-dimensional}.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:uniquecoefficientsindependentlist} \hlt{Unique Coefficients for Independent Lists}\\
Let $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ be a linearly independent list in $\mathcal{V}$.\\
Then $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = b_1\mathbf{v}_1 + \cdots + b_m\mathbf{v}_m$ iff $a_j=b_j$ for each $j=1,\ldots,m$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Subtracting the two sides gives $(a_1-b_1)\mathbf{v}_1+\cdots+(a_m-b_m)\mathbf{v}_m=\mathbf{0}$.\\
Linear independence forces $a_j-b_j=0$ for each $j$, hence $a_j=b_j$.
\item[$(\Leftarrow)$] If $a_j=b_j$ for each $j$, then the two linear combinations have identical coefficients and are equal.
\end{enumerate}
\hlt{Structural Commentary}: Linear independence makes coefficients in a linear combination unique.\\
\hlt{Key Insight}: Independent vectors allow no hidden coefficient cancellation.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:appendingtodependentlist} \hlt{Appending to a Dependent List}\\
Let $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ be a linearly dependent list in $\mathcal{V}$.\\
Then for each $\mathbf{v}\in\mathcal{V}$, the list $(\mathbf{v}_1, \ldots, \mathbf{v}_m, \mathbf{v})$ is linearly dependent.
\begin{proof}
Since $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly dependent, $\exists$ scalars $a_1,\ldots,a_m$, not all zero, with $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = \mathbf{0}$.\\
Appending coefficient $0$ gives $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m + 0\mathbf{v} = \mathbf{0}$, and the scalars $a_1,\ldots,a_m,0$ are not all zero.\\
\hlt{Structural Commentary}: Longer lists inherit dependence; set new coefficients to $0$.\\
\hlt{Key Insight}: Once a list has a redundancy, adding more vectors cannot remove it.
\end{proof}
\end{proposition}

\begin{definition} \label{def:omitting-vector-from-list} \hlt{Omitting a Vector from a List}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be a list in $\mathcal{V}$ with $m\geq 2$.\\
For $j\in\{1,\ldots,m\}$, the list obtained from $\beta$ by omitting $\mathbf{v}_j$ is denoted $(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$.
\end{definition}

\begin{proposition} \label{prop:omittingfromindependentlist} \hlt{Omitting from an Independent List}\\
If $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is linearly independent and $m\geq2$, then $\beta_{\widehat{j}}\defeq(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$ is linearly independent for each $j\in\{1,\ldots,m\}$.
\begin{proof}
Fix $j$. Any relation $\sum\limits_{i\neq j}a_i\mathbf{v}_i=\mathbf{0}$ on $\beta_{\widehat{j}}$ becomes a relation on $\beta$ after setting $a_j=0$. Independence of $\beta$ gives $a_i=0$ for all $i$, so the sublist is independent.\\
\hlt{Structural Commentary}: Deletion cannot create cancellation.\\
\hlt{Key Insight}: Every sublist of an independent list remains independent.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:adjoiningandremovingvectors} \hlt{Adjoining and Removing Vectors from Lists}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be a list in $\mathcal{V}$. Write $(\beta,\mathbf{v})\defeq(\mathbf{v}_1,\ldots,\mathbf{v}_m,\mathbf{v})$ and $\beta_{\widehat{j}}\defeq(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\beta$ is linearly independent and $\mathbf{v}\notin\operatorname{span}(\beta)$, then $(\beta,\mathbf{v})$ is linearly independent.
\item If $\sum\limits_{i=1}^m a_i\mathbf{v}_i=\mathbf{0}$ is nontrivial and $a_j\neq 0$, then $\operatorname{span}(\beta_{\widehat{j}})=\operatorname{span}(\beta)$.\\
Hence $\operatorname{span}(\beta)=\mathcal{V}$ implies $\operatorname{span}(\beta_{\widehat{j}})=\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Suppose $\sum\limits_{i=1}^m a_i\mathbf{v}_i+a\mathbf{v}=\mathbf{0}$. If $a\neq0$, then $\mathbf{v}=-a^{-1}\sum\limits_{i=1}^m a_i\mathbf{v}_i\in\operatorname{span}(\beta)$, a contradiction.\\
Hence $a=0$, and independence of $\beta$ gives $a_1=\cdots=a_m=0$.
\item Solving the relation for $\mathbf{v}_j$ gives $\mathbf{v}_j=-a_j^{-1}\sum\limits_{i\neq j}a_i\mathbf{v}_i\in\operatorname{span}(\beta_{\widehat{j}})$. Thus every vector of $\beta$ lies in $\operatorname{span}(\beta_{\widehat{j}})$, so $\operatorname{span}(\beta)\subseteq\operatorname{span}(\beta_{\widehat{j}})$. The reverse inclusion is immediate because $\beta_{\widehat{j}}$ is a sublist of $\beta$.
\end{enumerate}
\hlt{Structural Commentary}: New directions preserve independence; redundant vectors preserve span.\\
\hlt{Key Insight}: Add new directions; remove redundant ones.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:replacement-lemma} \hlt{Replacement Lemma}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$, let $\beta=(\mathbf{u}_1,\ldots,\mathbf{u}_r)$ span $\mathcal{V}$, and let $\mathbf{0}\neq\mathbf{v}=\sum\limits_{i=1}^r c_i\mathbf{u}_i$.\\
For $j\in\{1,\ldots,r\}$, set $\beta_{j\leftarrow\mathbf{v}}\defeq(\mathbf{u}_1,\ldots,\mathbf{u}_{j-1},\mathbf{v},\mathbf{u}_{j+1},\ldots,\mathbf{u}_r)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $c_j\neq0$ for some $j$.
\item If $c_j\neq0$, then $\operatorname{span}(\beta_{j\leftarrow\mathbf{v}})=\mathcal{V}$; if $\beta$ is a basis, then $\beta_{j\leftarrow\mathbf{v}}$ is a basis.
\item If $r\geq2$, $\beta$ is a basis, and $\mathbf{v}\notin\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_k)$ for some $1\leq k<r$, then $\beta_{j\leftarrow\mathbf{v}}$ is a basis for some $j>k$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If all $c_j=0$, then $\mathbf{v}=\mathbf{0}$.
\item Since $\mathbf{u}_j=c_j^{-1}\left(\mathbf{v}-\sum\limits_{i\neq j}c_i\mathbf{u}_i\right)$, every vector of $\beta$ lies in $\operatorname{span}(\beta_{j\leftarrow\mathbf{v}})$.\\
Conversely, $\mathbf{v}\in\operatorname{span}(\beta)$, so every vector of $\beta_{j\leftarrow\mathbf{v}}$ lies in $\operatorname{span}(\beta)$. Hence $\operatorname{span}(\beta_{j\leftarrow\mathbf{v}})=\mathcal{V}$.\\
If $\beta$ is a basis and $\sum\limits_{i<j}b_i\mathbf{u}_i+a\mathbf{v}+\sum\limits_{i>j}b_i\mathbf{u}_i=\mathbf{0}$, substitution gives $\sum\limits_{i\neq j}(b_i+ac_i)\mathbf{u}_i+ac_j\mathbf{u}_j=\mathbf{0}$. Independence of $\beta$ gives $ac_j=0$ and $b_i+ac_i=0$ for $i\neq j$.\\
Since $c_j\neq0$, $a=0$, and then every $b_i=0$. Thus $\beta_{j\leftarrow\mathbf{v}}$ is linearly independent.
\item Since $\mathbf{v}\notin\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_k)$, some coefficient among $c_{k+1},\ldots,c_r$ is nonzero.\\
Choose $j>k$ with $c_j\neq0$; by (ii), $\beta_{j\leftarrow\mathbf{v}}$ is a basis.
\end{enumerate}
\hlt{Structural Commentary}: Replacement preserves span.\\
\hlt{Key Insight}: Replace where the coefficient is nonzero.
\end{proof}
\end{lemma}


\begin{lemma} \label{lem:lineardependencelemma} \hlt{Linear Dependence Lemma}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be linearly dependent with $\mathbf{v}_1\neq\mathbf{0}$, and set $\beta_{\widehat{j}}\defeq(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$.\\
Then some $j\in\{2,\ldots,m\}$ satisfies $\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_{j-1})$ and $\operatorname{span}(\beta_{\widehat{j}})=\operatorname{span}(\beta)$.
\begin{proof}
Choose a nontrivial relation $\sum\limits_{i=1}^m a_i\mathbf{v}_i=\mathbf{0}$ and let $j=\max\{i:a_i\neq0\}$.\\
If $j=1$, then $a_1\mathbf{v}_1=\mathbf{0}$ with $a_1\neq0$, so $\mathbf{v}_1=\mathbf{0}$, contrary to hypothesis. Hence $j\geq2$.\\
Since $a_i=0$ for $i>j$, we have $\mathbf{v}_j=-a_j^{-1}\sum\limits_{i=1}^{j-1}a_i\mathbf{v}_i$, so $\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_{j-1})$.\\
Substituting this expression for $\mathbf{v}_j$ rewrites every linear combination of $\beta$ without $\mathbf{v}_j$, so $\operatorname{span}(\beta)\subseteq\operatorname{span}(\beta_{\widehat{j}})$. The reverse inclusion follows because $\beta_{\widehat{j}}$ is a sublist of $\beta$. Hence the spans are equal.\\
\hlt{Structural Commentary}: The last nonzero coefficient gives the removable vector.\\
\hlt{Key Insight}: Dependence gives a later vector removable without losing span.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:finitedimlenofvec} \hlt{Independent Lists Are No Longer Than Spanning Lists}\\
If $\mathcal{V}$ is finite-dimensional, $\alpha=(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ is linearly independent, and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ spans $\mathcal{V}$, then $m\leq n$.
\begin{proof}
Suppose $m>n$.\\
If $n=0$, then $\mathcal{V}=\operatorname{span}(\gamma)=\{\mathbf{0}\}$, contradicting independence of the nonempty list $\alpha$; hence $n\geq1$.\\
Construct spanning lists $\gamma_k$ of length $n$ that contain $\mathbf{u}_1,\ldots,\mathbf{u}_k$ for $0\leq k\leq n$. Set $\gamma_0=\gamma$. Given $\gamma_k$ with $k<n$, write $\mathbf{u}_{k+1}$ in $\operatorname{span}(\gamma_k)$. Since $\mathbf{u}_{k+1}\notin\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_k)$, some coefficient on an entry of $\gamma_k$ outside $\mathbf{u}_1,\ldots,\mathbf{u}_k$ is nonzero. Replacing that entry by $\mathbf{u}_{k+1}$ preserves span by Lemma \ref{lem:replacement-lemma}(ii); call the result $\gamma_{k+1}$.\\
Then $\gamma_n$ spans $\mathcal{V}$ and contains $\mathbf{u}_1,\ldots,\mathbf{u}_n$, so $\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_n)=\mathcal{V}$.\\
Since $m>n$, this puts $\mathbf{u}_{n+1}$ in the span of earlier vectors, contradicting independence of $\alpha$.\\
\hlt{Structural Commentary}: Independent length is bounded by spanning length.\\
\hlt{Key Insight}: Independent lists cannot be longer than spanning lists in finite dimensions.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:basis-unique-coordinate-representation} \hlt{Unique Coordinate Representation}\\
$\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ is a basis of $\mathcal{V}$ iff each $\mathbf{v}\in\mathcal{V}$ has a unique coefficient vector $(a_1,\ldots,a_n)\in\mathbb{F}^n$ with $\mathbf{v}=\sum\limits_{j=1}^n a_j\mathbf{v}_j$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Spanning gives a representation of each $\mathbf{v}\in\mathcal{V}$. If $\mathbf{v}=\sum\limits_{j=1}^{n}a_j\mathbf{v}_j=\sum\limits_{j=1}^{n}b_j\mathbf{v}_j$, then Proposition \ref{prop:uniquecoefficientsindependentlist} gives $a_j=b_j$ for each $j$.
\item[$(\Leftarrow)$] Existence gives $\operatorname{span}(\beta)=\mathcal{V}$. If $\sum\limits_{j=1}^{n}a_j\mathbf{v}_j=\mathbf{0}$, compare with $\mathbf{0}=\sum\limits_{j=1}^{n}0\mathbf{v}_j$; uniqueness gives $a_j=0$ for each $j$, so $\beta$ is independent.
\end{enumerate}
\hlt{Structural Commentary}: A basis is exactly existence and uniqueness of coordinates.\\
\hlt{Key Insight}: Coordinates are basis coefficients.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:reducing-spanning-lists} \hlt{Reducing Spanning Lists}\\
Every spanning list in a finite-dimensional vector space contains a sublist that is a basis.
\begin{proof}
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a spanning list of $\mathcal{V}$. If $\beta$ is independent, it is a basis. Otherwise remove one vector without changing span: omit $\mathbf{v}_1$ if $\mathbf{v}_1=\mathbf{0}$, and otherwise use Lemma \ref{lem:lineardependencelemma} to remove a suitable $\mathbf{v}_j$. The resulting sublist still spans $\mathcal{V}$ and has length one less. Repeat this step. The length strictly decreases, so the process stops with a spanning independent sublist, hence a basis.\\
\hlt{Structural Commentary}: Delete redundant vectors while preserving span.\\
\hlt{Key Insight}: Spanning gives enough vectors; reduction deletes exactly the unnecessary ones.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:extending-independent-lists} \hlt{Extending Independent Lists}\\
Every linearly independent list in a finite-dimensional vector space can be extended to a basis.
\begin{proof}
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be linearly independent in $\mathcal{V}$, and choose a spanning list $(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ of $\mathcal{V}$.\\
Initialize $\mathcal{B}=\beta$. Inspect $\mathbf{w}_1,\ldots,\mathbf{w}_n$ in order, appending $\mathbf{w}_j$ exactly when $\mathbf{w}_j\notin\operatorname{span}(\mathcal{B})$. Each append preserves independence by Proposition \ref{prop:adjoiningandremovingvectors}.\\
After the inspection, each $\mathbf{w}_j$ lies in $\operatorname{span}(\mathcal{B})$: it was either already there or was appended. Since the $\mathbf{w}_j$ span $\mathcal{V}$ and $\operatorname{span}(\mathcal{B})$ is a subspace, every vector of $\mathcal{V}$ lies in $\operatorname{span}(\mathcal{B})$.\\
Thus $\mathcal{B}$ is a spanning independent list, hence a basis.\\
\hlt{Structural Commentary}: An independent list has no redundancy, but may miss directions. Appending only vectors outside its span fills those directions while preserving independence.\\
\hlt{Key Insight}: A finite-dimensional independent list becomes a basis once enough new directions are added.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:finite-dimensional-spaces-and-subspaces-have-bases} \hlt{Bases and Subspaces in Finite Dimension}\\
If $\mathcal{V}$ is finite-dimensional, then $\mathcal{V}$ has a basis. Moreover, $\mathcal{U}\leq\mathcal{V}$ implies $\mathcal{U}$ is finite-dimensional.
\begin{proof}
Choose a finite spanning list $\gamma$ of $\mathcal{V}$. By Theorem \ref{thm:reducing-spanning-lists}, some sublist of $\gamma$ is a basis of $\mathcal{V}$.\\
Let $\mathcal{U}\leq\mathcal{V}$. If $\mathcal{U}=\{\mathbf{0}\}$, then $()$ is a basis. Otherwise start with $\beta=()$ and append vectors from $\mathcal{U}\setminus\operatorname{span}(\beta)$ while possible. Proposition \ref{prop:adjoiningandremovingvectors} preserves independence, while Theorem \ref{thm:finitedimlenofvec} bounds the length by the length of $\gamma$; hence the process terminates. At termination, $\mathcal{U}\subseteq\operatorname{span}(\beta)$, so $\beta$ is a finite basis of $\mathcal{U}$.\\
\hlt{Structural Commentary}: Finite-dimensional spaces have finite coordinate systems, and subspaces cannot contain longer independent lists than the ambient space permits.\\
\hlt{Key Insight}: Finite-dimensionality passes downward to subspaces.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:basis-length-well-defined} \hlt{Basis Length Is Well-Defined}\\
Any two bases of a finite-dimensional vector space have the same length.
\begin{proof}
Let $\beta_1=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ and $\beta_2=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ be bases of $\mathcal{V}$. Theorem \ref{thm:finitedimlenofvec} applied to $\beta_1$ against $\beta_2$ gives $m\leq n$, and reversing the roles gives $n\leq m$. Hence $m=n$.\\
\hlt{Structural Commentary}: Bases are simultaneously independent and spanning, so the comparison theorem traps their lengths in both directions. This makes dimension well-defined.\\
\hlt{Key Insight}: Dimension is an intrinsic number, not an artefact of a particular basis.
\end{proof}
\end{theorem}

\begin{definition} \label{def:dimension} \hlt{Dimension}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathcal{V}=\{\mathbf{0}\}$, then $\dim(\mathcal{V})\defeq0$.
\item If $\mathcal{V}$ has a basis of length $n\geq1$, then $\mathcal{V}$ is \hlt{$n$-dimensional} and $\dim(\mathcal{V})\defeq n$.
\end{enumerate}
Theorem \ref{thm:basis-length-well-defined} makes $\dim(\mathcal{V})$ well-defined.
\end{definition}

\begin{proposition} \label{prop:dimension-bounds-and-basis-criteria} \hlt{Subspace Dimension Criterion}\\
Let $\mathcal{V}$ be finite-dimensional and let $\mathcal{U}\leq\mathcal{V}$. Then $\dim(\mathcal{U})\leq\dim(\mathcal{V})$, and equality holds iff $\mathcal{U}=\mathcal{V}$.
\begin{proof}
Let $\delta=(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ be a basis of $\mathcal{U}$ and let $\beta$ be a basis of $\mathcal{V}$, where $n=\dim(\mathcal{V})$.\\
Since $\delta$ is independent in $\mathcal{V}$ and $\beta$ spans $\mathcal{V}$, Theorem \ref{thm:finitedimlenofvec} gives $m\leq n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathcal{U}=\mathcal{V}$, then any basis of $\mathcal{U}$ is a basis of $\mathcal{V}$, so $\dim(\mathcal{U})=\dim(\mathcal{V})$.
\item[$(\Leftarrow)$] If $\dim(\mathcal{U})=\dim(\mathcal{V})$, then $\delta$ is independent in $\mathcal{V}$ of length $n$. By Theorem \ref{thm:extending-independent-lists}, extending $\delta$ to a basis of $\mathcal{V}$ adds no vector, since every basis has length $n$. Hence $\mathcal{U}=\operatorname{span}(\delta)=\mathcal{V}$.
\end{enumerate}
\hlt{Structural Commentary}: Dimension bounds subspaces; equality forces fullness.\\
\hlt{Key Insight}: Equal dimension subspaces fill the ambient space.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:full-dimension-basis-criteria} \hlt{Full-Dimension Basis Criteria}\\
Let $\mathcal{V}$ be finite-dimensional and set $n\defeq\dim(\mathcal{V})$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item if $\gamma$ spans $\mathcal{V}$ and has length $n$, then $\gamma$ is a basis;
\item if $\alpha$ is linearly independent in $\mathcal{V}$ and has length $n$, then $\alpha$ is a basis.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] By Theorem \ref{thm:reducing-spanning-lists}, $\gamma$ contains a basis sublist. Since every basis has length $n$, no proper sublist can have length $n$; hence $\gamma$ itself is a basis.
\item[$(ii)$] Extend $\alpha$ to a basis of $\mathcal{V}$ by Theorem \ref{thm:extending-independent-lists}. Since every basis has length $n$ and $\alpha$ already has length $n$, no vector is added; hence $\alpha$ is a basis.
\end{enumerate}
\hlt{Structural Commentary}: Full-length spanning or independent lists cannot be shortened or extended.\\
\hlt{Key Insight}: In dimension $n$, $n$ independent or $n$ spanning vectors already form a basis.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:basis-equivalent-characterizations} \hlt{Basis Characterizations}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a finite list in $\mathcal{V}$, and write $\beta^\frown\mathbf{x}=(\mathbf{v}_1,\ldots,\mathbf{v}_n,\mathbf{x})$. Then the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\beta$ is a basis of $\mathcal{V}$;
\item each $\mathbf{x}\in\mathcal{V}$ has a unique coefficient vector $(a_1,\ldots,a_n)\in\mathbb{F}^n$ with $\mathbf{x}=\sum\limits_{j=1}^{n}a_j\mathbf{v}_j$;
\item $\operatorname{span}(\beta)=\mathcal{V}$, and no proper sublist of $\beta$ spans $\mathcal{V}$;
\item $\beta$ is linearly independent, and $\beta^\frown\mathbf{x}$ is linearly dependent for each $\mathbf{x}\in\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ This is Proposition \ref{prop:basis-unique-coordinate-representation}.
\item[$(ii)$]$\Rightarrow (iii)$ Existence gives $\operatorname{span}(\beta)=\mathcal{V}$.\\
If a proper sublist $\gamma\subsetneq\beta$ spanned $\mathcal{V}$ and omitted the occurrence $\mathbf{v}_j$, then $\mathbf{v}_j$ would have both its standard $\beta$-coordinate representation and one using only $\gamma$, contradicting uniqueness.
\item[$(iii)$]$\Rightarrow (iv)$ If $\beta$ were dependent, Proposition \ref{prop:adjoiningandremovingvectors} would remove some $\mathbf{v}_j$ without changing $\operatorname{span}(\beta)$, contradicting minimality. Thus $\beta$ is independent.\\
Since $\operatorname{span}(\beta)=\mathcal{V}$, writing $\mathbf{x}$ as a linear combination of $\beta$ gives a nontrivial relation on $\beta^\frown\mathbf{x}$.
\item[$(iv)$]$\Rightarrow (i)$ If $\operatorname{span}(\beta)\neq\mathcal{V}$, choose $\mathbf{x}\notin\operatorname{span}(\beta)$. Then Proposition \ref{prop:adjoiningandremovingvectors} makes $\beta^\frown\mathbf{x}$ independent, contradicting (iv). Hence $\beta$ spans $\mathcal{V}$, and with independence from (iv), it is a basis.
\end{enumerate}
\hlt{Structural Commentary}: A basis is simultaneously just enough to span and too large to extend independently.\\
\hlt{Key Insight}: Spanning and independence are dual minimality and maximality conditions.
\end{proof}
\end{theorem}

\begin{definition} \label{def:hamel-basis} \hlt{Hamel Basis}\\
A subset $\mathcal{B}\subseteq\mathcal{V}$ is a \hlt{Hamel basis} of $\mathcal{V}$ if it is linearly independent and $\operatorname{span}(\mathcal{B})=\mathcal{V}$. Equivalently, each $\mathbf{v}\in\mathcal{V}$ has a unique finitely supported coefficient function $a:\mathcal{B}\to\mathbb{F}$ with $\mathbf{v}=\sum\limits_{\mathbf{b}\in\mathcal{B}}a(\mathbf{b})\mathbf{b}$.\\
Finite ordered bases are orderings of finite Hamel bases.
\end{definition}

\begin{theorem} \label{thm:hamel-basis-extension} \hlt{Hamel Basis Extension}\\
Let $I\subseteq S\subseteq\mathcal{V}$, with $I$ linearly independent and $\operatorname{span}(S)=\mathcal{V}$.\\
Assuming Zorn's lemma, there exists a Hamel basis $\mathcal{B}$ of $\mathcal{V}$ such that $I\subseteq\mathcal{B}\subseteq S$. Consequently, every nonzero vector space has a basis; every independent set extends to a basis; every spanning set contains a basis.
\begin{proof}
Let $\mathscr{P}=\{E:I\subseteq E\subseteq S,\ E\ \text{linearly independent}\}$, ordered by inclusion. Then $I\in\mathscr{P}$.\\
If $\mathcal{C}\subseteq\mathscr{P}$ is a nonempty chain, put $U=\bigcup\limits_{E\in\mathcal{C}}E$. In any finite relation $\sum\limits_{k=1}^{m}a_k\mathbf{u}_k=\mathbf{0}$ in $U$, the finitely many vectors $\mathbf{u}_k$ lie in one member of the chain; independence there gives $a_k=0$ for all $k$.\\
Thus $U\in\mathscr{P}$ and is an upper bound for $\mathcal{C}$. The empty chain is bounded by $I$.\\
By Zorn's lemma, Principle \ref{princ:zorns-lemma}, choose maximal $\mathcal{B}\in\mathscr{P}$.\\
If $s\in S\setminus\operatorname{span}(\mathcal{B})$, then $\mathcal{B}\cup\{s\}$ is linearly independent, contradicting maximality. Hence $S\subseteq\operatorname{span}(\mathcal{B})$, so $\operatorname{span}(\mathcal{B})=\mathcal{V}$ and $\mathcal{B}$ is a Hamel basis. The consequences follow by taking $S=\mathcal{V}$ or $I=\emptyset$.\\
\hlt{Structural Commentary}: Finite extension uses exchange; arbitrary extension uses maximality.\\
\hlt{Key Insight}: Maximal independence inside a spanning set forces spanning.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:arbitrary-bases-same-cardinality} \hlt{Dimension as Basis Cardinality}\\
If $\mathcal{B}$ and $\mathcal{C}$ are Hamel bases of $\mathcal{V}$, then $\abs{\mathcal{B}}=\abs{\mathcal{C}}$. Hence $\dim(\mathcal{V}):=\abs{\mathcal{B}}$ is well-defined for any Hamel basis $\mathcal{B}$.
\begin{proof}
If one basis is finite, say $\abs{\mathcal{B}}=n$, then $\mathcal{C}$ cannot contain $n+1$ distinct vectors, since such a finite sublist would be independent while $\mathcal{B}$ spans $\mathcal{V}$, contradicting Theorem \ref{thm:finitedimlenofvec}. Hence $\mathcal{C}$ is finite, and Theorem \ref{thm:basis-length-well-defined} gives $\abs{\mathcal{C}}=n$. Thus assume both bases are infinite. For $\mathbf{c}\in\mathcal{C}$, let $\operatorname{supp}_{\mathcal{B}}(\mathbf{c})\subseteq\mathcal{B}$ be its finite $\mathcal{B}$-support.\\
Every $\mathbf{b}\in\mathcal{B}$ appears in some $\operatorname{supp}_{\mathcal{B}}(\mathbf{c})$: otherwise all vectors in $\operatorname{span}(\mathcal{C})$ would have zero $\mathbf{b}$-coordinate, contradicting $\operatorname{span}(\mathcal{C})=\mathcal{V}$. Hence $\mathcal{B}\subseteq\bigcup\limits_{\mathbf{c}\in\mathcal{C}}\operatorname{supp}_{\mathcal{B}}(\mathbf{c})$. Thus $\abs{\mathcal{B}}\leq\sum\limits_{\mathbf{c}\in\mathcal{C}}\abs{\operatorname{supp}_{\mathcal{B}}(\mathbf{c})}\leq\abs{\mathcal{C}}\aleph_0=\abs{\mathcal{C}}$ by Theorem \ref{thm:infinite-cardinal-absorption-reference}. Interchanging $\mathcal{B}$ and $\mathcal{C}$ gives $\abs{\mathcal{C}}\leq\abs{\mathcal{B}}$; Theorem \ref{thm:schroeder-bernstein} gives equality.\\
\hlt{Structural Commentary}: Even in infinite dimensions, a basis size is intrinsic.\\
\hlt{Key Insight}: Each basis vector is detected inside finite supports relative to the other basis.
\end{proof}
\end{theorem}
