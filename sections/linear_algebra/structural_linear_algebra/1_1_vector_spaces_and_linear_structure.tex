\subsection{Vector Spaces and Linear Structure}

% **Role:** algebraic language of linear structure.
%
% Fields, vector spaces, subspaces, span, independence, bases, dimension, coordinates.

Vector spaces are objects that can be added and scaled. Subspaces, spans, bases, dimension, and coordinates describe how they are generated and measured.

\begin{definition} \label{def:vector-space} \hlt{Vector Space}
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.vector_space_axioms}
\leanok\\
Let $\mathbb{F}$ be a field. A vector space over $\mathbb{F}$ is a set $\mathcal{V}$ together with operations of vector addition and scalar multiplication such that, for all $\mathbf{u}, \mathbf{v}, \mathbf{w} \in \mathcal{V}$ and all $a,b \in \mathbb{F}$, the following axioms hold:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item vector addition is commutative: $\mathbf{u} + \mathbf{v} = \mathbf{v} + \mathbf{u}$;
\item vector addition is associative: $(\mathbf{u} + \mathbf{v}) + \mathbf{w} = \mathbf{u} + (\mathbf{v} + \mathbf{w})$;
\item there exists an additive identity $\mathbf{0} \in \mathcal{V}$ such that $\mathbf{0} + \mathbf{v} = \mathbf{v}$;
\item for each $\mathbf{v} \in \mathcal{V}$ there exists an additive inverse $-\mathbf{v} \in \mathcal{V}$ such that $\mathbf{v} + (-\mathbf{v}) = \mathbf{0}$;
\item scalar multiplication has identity: $1\mathbf{v} = \mathbf{v}$;
\item scalar multiplication is compatible with multiplication in $\mathbb{F}$: $(ab)\mathbf{v} = a(b\mathbf{v})$;
\item scalar multiplication distributes over vector addition: $a(\mathbf{u} + \mathbf{v}) = a\mathbf{u} + a\mathbf{v}$;
\item scalar multiplication distributes over addition in $\mathbb{F}$: $(a+b)\mathbf{v} = a\mathbf{v} + b\mathbf{v}$.
\end{enumerate}
If $\mathcal{V} = \{ \mathbf{0} \}$, then $\mathcal{V}$ is a \hlt{zero vector space}; if $\mathcal{V} \neq \{ \mathbf{0} \}$, then $\mathcal{V}$ is a \hlt{nonzero vector space}.
\end{definition}

\begin{definition} \label{def:finitely-nonzero-sequence-space} \hlt{Finitely Nonzero Sequence Space}
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.FinitelyNonzeroSequenceSpace}
\leanok\\
Let $\mathbb{F}^{(\mathbb{N})}$ denote the set of all sequences $(x_1,x_2,\ldots)$ of elements of $\mathbb{F}$ such that $x_j=0$ for all but finitely many $j\in\mathbb{N}$. Addition and scalar multiplication are defined componentwise.
\end{definition}

\begin{proposition} \label{prop:finitely-nonzero-sequences-vector-space} \hlt{Finitely Nonzero Sequences Form a Vector Space}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.finitely_nonzero_sequences_vector_space}
\leanok
The set $\mathbb{F}^{(\mathbb{N})}$ is a vector space over $\mathbb{F}$ under componentwise addition and scalar multiplication.
\begin{proof}
\leanok
The zero sequence belongs to $\mathbb{F}^{(\mathbb{N})}$. If $\mathbf{x}=(x_1,x_2,\ldots)$ and $\mathbf{y}=(y_1,y_2,\ldots)$ are in $\mathbb{F}^{(\mathbb{N})}$ and $a\in\mathbb{F}$, then the nonzero entries of $a\mathbf{x}+\mathbf{y}$ can occur only among the finitely many indices where $\mathbf{x}$ or $\mathbf{y}$ is nonzero. Thus $a\mathbf{x}+\mathbf{y}\in\mathbb{F}^{(\mathbb{N})}$, and the vector space axioms are inherited componentwise from $\mathbb{F}$.\\
\hlt{Structural Commentary}: Finite support is preserved by finite linear combinations.\\
\hlt{Key Insight}: Componentwise operations cannot create infinitely many new nonzero entries.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:basicvectorspaceconsequences} \hlt{Basic Consequences of Vector Space Axioms}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.basic_vector_space_consequences}
\leanok
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then the additive identity and additive inverses are unique. Moreover, for all $a \in \mathbb{F}$ and $\mathbf{v} \in \mathcal{V}$, $0\mathbf{v} = \mathbf{0}$, $a\mathbf{0} = \mathbf{0}$, and $a\mathbf{v} = \mathbf{0}$ if and only if $a = 0$ or $\mathbf{v} = \mathbf{0}$.
\begin{proof}
\leanok
If $\mathbf{0}$ and $\mathbf{0}'$ are additive identities, then $\mathbf{0} = \mathbf{0} + \mathbf{0}' = \mathbf{0}'$, so the additive identity is unique.\\
If $\mathbf{w}$ and $\mathbf{w}'$ are additive inverses of $\mathbf{v}$, then $\mathbf{w} = \mathbf{w} + \mathbf{0} = \mathbf{w} + (\mathbf{v} + \mathbf{w}') = (\mathbf{w} + \mathbf{v}) + \mathbf{w}' = \mathbf{0} + \mathbf{w}' = \mathbf{w}'$, so additive inverses are unique. Since $0\mathbf{v} = (0+0)\mathbf{v} = 0\mathbf{v} + 0\mathbf{v}$, cancellation gives $0\mathbf{v} = \mathbf{0}$.\\
Similarly, $a\mathbf{0} = a(\mathbf{0}+\mathbf{0}) = a\mathbf{0} + a\mathbf{0}$, so $a\mathbf{0} = \mathbf{0}$. If $a = 0$ or $\mathbf{v} = \mathbf{0}$, then the preceding identities give $a\mathbf{v} = \mathbf{0}$. Conversely, suppose $a\mathbf{v} = \mathbf{0}$ and $a \neq 0$. Then $a^{-1}$ exists in $\mathbb{F}$, and $\mathbf{v} = 1\mathbf{v} = (a^{-1}a)\mathbf{v} = a^{-1}(a\mathbf{v}) = a^{-1}\mathbf{0} = \mathbf{0}$.\\
\hlt{Structural Commentary}: These identities are the basic algebraic hygiene of vector spaces. They let later arguments use cancellation, zero vectors, and inverses without reopening the axioms each time.\\
\hlt{Key Insight}: The vector space axioms force the zero and inverse behaviour to be rigid, not optional structure.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:negativeoneactsasinverse}
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.negative_one_smul_eq_neg_vector}
\leanok
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. Then $(-1)\mathbf{v} = -\mathbf{v}$ for every $\mathbf{v} \in \mathcal{V}$.
\begin{proof}
\leanok
Let $\mathbf{v} \in \mathcal{V}$. We show that $(-1)\mathbf{v} + \mathbf{v} = \mathbf{0}$. By the vector space axioms and Theorem \ref{thm:basicvectorspaceconsequences},
\begin{align}
(-1)\mathbf{v} + \mathbf{v} = (-1)\mathbf{v} + 1\mathbf{v} = (-1+1)\mathbf{v} = 0\mathbf{v} = \mathbf{0} \nonumber
\end{align}
Thus $(-1)\mathbf{v}$ is the additive inverse of $\mathbf{v}$, so $(-1)\mathbf{v} = -\mathbf{v}$.\\
\hlt{Structural Commentary}: Scalar multiplication by $-1$ is exactly the same operation as taking an additive inverse.\\
\hlt{Key Insight}: Negation is not an extra operation; it is already encoded by the field action.
\end{proof}
\end{corollary}

\begin{definition} \label{def:subspace}
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.Subspace}
\leanok
A \hlt{subspace} of a vector space $\mathcal{V}$ over $\mathbb{F}$ is a subset $\mathcal{U} \subseteq \mathcal{V}$ that is itself a vector space over $\mathbb{F}$ with the same vector addition and scalar multiplication as in $\mathcal{V}$.
\end{definition}

\begin{theorem} \label{thm:subspacetest} \hlt{Subspace Test}\\
\lean{MathNotes.LinearAlgebra.StructuralLinearAlgebra.VectorSpacesAndLinearStructure.subspace_test}
\leanok
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$ and let $\mathcal{U}$ be a nonempty subset of $\mathcal{V}$. Then $\mathcal{U}$ is a subspace of $\mathcal{V}$ if and only if $c\mathbf{u} + \mathbf{v} \in \mathcal{U}$ for all $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and all $c \in \mathbb{F}$.
\begin{proof}
\leanok
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathcal{U}$ is a subspace, then $c\mathbf{u}+\mathbf{v} \in \mathcal{U}$ for all $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and all $c \in \mathbb{F}$ because $\mathcal{U}$ is closed under scalar multiplication and vector addition.
\item[$(\Leftarrow)$] Suppose that $c\mathbf{u}+\mathbf{v} \in \mathcal{U}$ whenever $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and $c \in \mathbb{F}$. We verify that $\mathcal{U}$ is a vector space with the operations inherited from $\mathcal{V}$. Since $\mathcal{U}$ is nonempty, let $\mathbf{u} \in \mathcal{U}$. By Corollary \ref{cor:negativeoneactsasinverse}, $\mathbf{0} = (-1)\mathbf{u}+\mathbf{u} \in \mathcal{U}$. If $\mathbf{u} \in \mathcal{U}$, then $-\mathbf{u} = (-1)\mathbf{u}+\mathbf{0} \in \mathcal{U}$. If $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ and $c \in \mathbb{F}$, then $c\mathbf{u} = c\mathbf{u}+\mathbf{0} \in \mathcal{U}$ and $\mathbf{u}+\mathbf{v} = 1\mathbf{u}+\mathbf{v} \in \mathcal{U}$. Thus $\mathcal{U}$ contains $\mathbf{0}$ and is closed under additive inverses, scalar multiplication, and vector addition; the remaining vector space axioms are inherited from $\mathcal{V}$. Hence $\mathcal{U}$ is a subspace.
\end{enumerate}
\hlt{Structural Commentary}: The theorem compresses the subspace axioms into one closure condition. This is the practical test used whenever a subset is proposed as a linear space.\\
\hlt{Key Insight}: To prove a subset is a subspace, prove it survives one affine-looking linear combination.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:intersectionsubspaces} \hlt{Intersection of Subspaces}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$. Then $\mathcal{U}\cap\mathcal{W} = \left\{ \mathbf{v} : \mathbf{v}\in\mathcal{U} \text{ and } \mathbf{v}\in\mathcal{W} \right\}$ is a subspace of $\mathcal{V}$.
\begin{proof}
Since $\mathbf{0}\in\mathcal{U}$ and $\mathbf{0}\in\mathcal{W}$, we have $\mathbf{0}\in\mathcal{U}\cap\mathcal{W}$, so the intersection is nonempty. If $\mathbf{u},\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$ and $c\in\mathbb{F}$, then $c\mathbf{u}+\mathbf{v}$ lies in both $\mathcal{U}$ and $\mathcal{W}$ because both are subspaces. Hence $c\mathbf{u}+\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$, and Theorem \ref{thm:subspacetest} shows that $\mathcal{U}\cap\mathcal{W}$ is a subspace of $\mathcal{V}$.\\
\hlt{Structural Commentary}: Intersections preserve every closure condition shared by the subspaces. The subspace test packages this into one check.\\
\hlt{Key Insight}: Common vectors of two subspaces still form a subspace because the shared part remains closed under linear combinations.
\end{proof}
\end{proposition}

\begin{definition} \label{def:linear-combinations-and-span} \hlt{Lists, Linear Combinations, and Span}\\
A \hlt{list} of vectors in $\mathcal{V}$ is a finite ordered sequence $\beta = (\mathbf{v}_1,\ldots,\mathbf{v}_m)$ in $\mathcal{V}$; the empty list is denoted $()$.\\
If $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is nonempty and $a_1,\ldots,a_m\in\mathbb{F}$, then $a_1\mathbf{v}_1+\cdots+a_m\mathbf{v}_m$ is a \hlt{linear combination} of $\beta$.\\
For $E\subseteq\mathcal{V}$, the \hlt{span} of $E$ is $\operatorname{span}(E)\defeq\{a_1\mathbf{v}_1+\cdots+a_m\mathbf{v}_m:m\in\N,\ a_j\in\mathbb{F},\ \mathbf{v}_j\in E\}$. $\operatorname{span}(\emptyset)=\{\mathbf{0}\}$.\\
If $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is a nonempty list, then $\operatorname{span}(\beta) = \operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_m)
\defeq \operatorname{span}(\{\mathbf{v}_1,\ldots,\mathbf{v}_m\})$.\\
A subset $E \subseteq \mathcal{V}$ \hlt{spans} $\mathcal{V}$, or is a \hlt{spanning set} for $\mathcal{V}$, if $\operatorname{span}(E)=\mathcal{V}$.\\
A list $\beta$ in $\mathcal{V}$ \hlt{spans} $\mathcal{V}$, or is a \hlt{spanning list} for $\mathcal{V}$, if $\operatorname{span}(\beta)=\mathcal{V}$.
\end{definition}

\begin{theorem} \label{thm:span-smallest-containing-subspace} \hlt{Span is the Smallest Containing Subspace}\\
Let $E\subseteq\mathcal{V}$. Then $\operatorname{span}(E)$ is the smallest subspace of $\mathcal{V}$ containing $E$.
\begin{proof}
If $E=\emptyset$, then $\operatorname{span}(E)=\{\mathbf{0}\}$, which is a subspace. Suppose $E\neq\emptyset$. If $\mathbf{x},\mathbf{y}\in\operatorname{span}(E)$ and $c\in\mathbb{F}$, then $\mathbf{x}$ and $\mathbf{y}$ are finite linear combinations of vectors in $E$. Hence $c\mathbf{x}+\mathbf{y}$ is also a finite linear combination of vectors in $E$, so $c\mathbf{x}+\mathbf{y}\in\operatorname{span}(E)$. By Theorem \ref{thm:subspacetest}, $\operatorname{span}(E)$ is a subspace of $\mathcal{V}$.\\
Each $\mathbf{v}\in E$ satisfies $\mathbf{v}=1\mathbf{v}\in\operatorname{span}(E)$, so $E\subseteq\operatorname{span}(E)$. If $\mathcal{U}$ is a subspace of $\mathcal{V}$ with $E\subseteq\mathcal{U}$, then closure under vector addition and scalar multiplication forces every finite linear combination of vectors in $E$ to lie in $\mathcal{U}$. Thus $\operatorname{span}(E)\subseteq\mathcal{U}$. Therefore $\operatorname{span}(E)$ is the smallest subspace of $\mathcal{V}$ containing $E$.\\
\hlt{Structural Commentary}: Span takes a set of vectors and closes it under all finite linear combinations. The result is precisely the smallest subspace forced by those vectors.\\
\hlt{Key Insight}: Nothing beyond linear closure is added. Every containing subspace must already contain span.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:spanmonotonicity} \hlt{Monotonicity of Span}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subsets of $\mathcal{V}$. If $\mathcal{U}\subseteq\mathcal{W}$, then $\operatorname{span}(\mathcal{U})\subseteq\operatorname{span}(\mathcal{W})$.
\begin{proof}
If $\mathcal{U}=\emptyset$, then $\operatorname{span}(\mathcal{U})=\{\mathbf{0}\}\subseteq\operatorname{span}(\mathcal{W})$.\\
Suppose $\mathcal{U}\neq\emptyset$. Every vector in $\operatorname{span}(\mathcal{U})$ is a finite linear combination of vectors in $\mathcal{U}$; since $\mathcal{U}\subseteq\mathcal{W}$, the same expression is a finite linear combination of vectors in $\mathcal{W}$. Hence $\operatorname{span}(\mathcal{U})\subseteq\operatorname{span}(\mathcal{W})$.\\
\hlt{Structural Commentary}: Enlarging the generating set can only enlarge the collection of linear combinations.\\
\hlt{Key Insight}: Span is monotone with respect to inclusion of generating sets.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Linear Independence} \\
A list $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ in $\mathcal{V}$ is \hlt{linearly independent} if $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = \mathbf{0}$ implies $a_1 = \cdots = a_m = 0$. The empty list is linearly independent. Any list that is not linearly independent is linearly dependent.
\end{definition}

\begin{proposition} \label{prop:uniquecoefficientsindependentlist} \hlt{Unique Coefficients for Independent Lists}\\
Let $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ be a linearly independent list in $\mathcal{V}$.\\
Then $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = b_1\mathbf{v}_1 + \cdots + b_m\mathbf{v}_m$ if and only if $a_j=b_j$ for each $j=1,\ldots,m$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Subtracting the two sides gives $(a_1-b_1)\mathbf{v}_1+\cdots+(a_m-b_m)\mathbf{v}_m=\mathbf{0}$.\\
Linear independence forces $a_j-b_j=0$ for every $j$, so $a_j=b_j$.
\item[$(\Leftarrow)$] If $a_j=b_j$ for every $j$, then the two linear combinations have identical coefficients and are equal.
\end{enumerate}
\hlt{Structural Commentary}: Linear independence makes coefficients in a linear combination unique.\\
\hlt{Key Insight}: Independent vectors allow no hidden coefficient cancellation.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:appendingtodependentlist} \hlt{Appending to a Dependent List}\\
Let $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ be a linearly dependent list in $\mathcal{V}$.\\
Then for every $\mathbf{v}\in\mathcal{V}$, the list $(\mathbf{v}_1, \ldots, \mathbf{v}_m, \mathbf{v})$ is linearly dependent.
\begin{proof}
Since $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly dependent, $\exists$ scalars $a_1,\ldots,a_m$, not all zero, s.t. $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m = \mathbf{0}$.\\
For any $\mathbf{v}\in\mathcal{V}$, $a_1\mathbf{v}_1 + \cdots + a_m\mathbf{v}_m + 0\mathbf{v} = \mathbf{0}$ and the scalars $a_1,\ldots,a_m,0$ are not all zero.\\
\hlt{Structural Commentary}: Longer lists inherit dependence; set new coefficients to $0$.\\
\hlt{Key Insight}: Once a list has a redundancy, adding more vectors cannot remove it.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Omitting a Vector from a List}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be a list in $\mathcal{V}$ with $m\geq 2$.\\
For $j\in\{1,\ldots,m\}$, the list obtained from $\beta$ by omitting $\mathbf{v}_j$ is denoted $(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$.
\end{definition}

\begin{proposition} \label{prop:omittingfromindependentlist} \hlt{Omitting from an Independent List}\\
If $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is linearly independent and $m\geq 2$, then $(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$ is linearly independent for every $j\in\{1,\ldots,m\}$.
\begin{proof}
Fix $j$, and suppose $\sum\limits_{\substack{i=1\\ i\neq j}}^m a_i\mathbf{v}_i=\mathbf{0}$. Set $a_j=0$.\\
Then $\sum\limits_{i=1}^m a_i\mathbf{v}_i=\mathbf{0}$, so independence of $\beta$ gives $a_i=0$ for every $i\neq j$.\\
\hlt{Structural Commentary}: Shorter lists inherit independence; deletion cannot create cancellation.\\
\hlt{Key Insight}: Every sublist of an independent list remains independent.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:adjoiningandremovingvectors} \hlt{Adjoining and Removing Vectors from Lists}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be a list in $\mathcal{V}$. Write $(\beta,\mathbf{v})\defeq(\mathbf{v}_1,\ldots,\mathbf{v}_m,\mathbf{v})$ and $\beta_{\widehat{j}}\defeq(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\beta$ is linearly independent and $\mathbf{v}\notin\operatorname{span}(\beta)$, then $(\beta,\mathbf{v})$ is linearly independent.
\item If $\sum\limits_{i=1}^m a_i\mathbf{v}_i=\mathbf{0}$ is nontrivial, $a_j\neq 0$, then $\operatorname{span}(\beta_{\widehat{j}})=\operatorname{span}(\beta)$. Hence $\operatorname{span}(\beta)=\mathcal{V}$ implies $\operatorname{span}(\beta_{\widehat{j}})=\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\sum\limits_{i=1}^m a_i\mathbf{v}_i+a\mathbf{v}=\mathbf{0}$ and $a\neq0$, then $\mathbf{v}=-a^{-1}\sum\limits_{i=1}^m a_i\mathbf{v}_i\in\operatorname{span}(\beta)$, a contradiction.
\item Since $a_j\neq0$, $\mathbf{v}_j=-\sum\limits_{\substack{i=1\\i\neq j}}^m\frac{a_i}{a_j}\mathbf{v}_i\in\operatorname{span}(\beta_{\widehat{j}})$. Hence $\operatorname{span}(\beta)\subseteq\operatorname{span}(\beta_{\widehat{j}})$; the reverse inclusion is immediate.\\
\end{enumerate}
\hlt{Structural Commentary}: New directions preserve independence; redundant vectors preserve span.\\
\hlt{Key Insight}: Add new directions; remove redundant ones.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:replacement-lemma} \hlt{Replacement Lemma}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$, let $\beta=(\mathbf{u}_1,\ldots,\mathbf{u}_r)$ span $\mathcal{V}$, and let $\mathbf{0}\neq\mathbf{v}=\sum\limits_{i=1}^r c_i\mathbf{u}_i$.\\
For $j\in\{1,\ldots,r\}$, set $\beta_{j\leftarrow\mathbf{v}}\defeq(\mathbf{u}_1,\ldots,\mathbf{u}_{j-1},\mathbf{v},\mathbf{u}_{j+1},\ldots,\mathbf{u}_r)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $c_j\neq0$ for some $j$.
\item If $c_j\neq0$, then $\operatorname{span}(\beta_{j\leftarrow\mathbf{v}})=\mathcal{V}$; if $\beta$ is a basis, then $\beta_{j\leftarrow\mathbf{v}}$ is a basis.
\item If $r\geq2$, $\beta$ is a basis, and $\mathbf{v}\notin\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_k)$ for some $1\leq k<r$, then $\beta_{j\leftarrow\mathbf{v}}$ is a basis for some $j>k$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If all $c_j=0$, then $\mathbf{v}=\mathbf{0}$.
\item Since $\mathbf{u}_j=c_j^{-1}\left(\mathbf{v}-\sum\limits_{i\neq j}c_i\mathbf{u}_i\right)$, every vector of $\beta$ lies in $\operatorname{span}(\beta_{j\leftarrow\mathbf{v}})$, while every vector of $\beta_{j\leftarrow\mathbf{v}}$ lies in $\operatorname{span}(\beta)$. Hence $\operatorname{span}(\beta_{j\leftarrow\mathbf{v}})=\mathcal{V}$. If $\beta$ is a basis and $\sum\limits_{i<j}b_i\mathbf{u}_i+a\mathbf{v}+\sum\limits_{i>j}b_i\mathbf{u}_i=\mathbf{0}$, then substitution gives $\sum\limits_{i\neq j}(b_i+ac_i)\mathbf{u}_i+ac_j\mathbf{u}_j=\mathbf{0}$. Linear independence of $\beta$ gives $ac_j=0$ and $b_i+ac_i=0$ for $i\neq j$.\\
Since $c_j\neq0$, $a=0$, and then every $b_i=0$. Thus $\beta_{j\leftarrow\mathbf{v}}$ is also linearly independent.
\item If $c_i=0 \ \forall i>k$, then $\mathbf{v}\in\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_k)$, contradiction. Thus some $c_j\neq0$ with $j>k$, and (ii) applies.
\end{enumerate}
\hlt{Structural Commentary}: Replacement preserves span.\\
\hlt{Key Insight}: Replace where the coefficient is nonzero.
\end{proof}
\end{lemma}


\begin{lemma} \label{lem:lineardependencelemma} \hlt{Linear Dependence Lemma}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be linearly dependent with $\mathbf{v}_1\neq\mathbf{0}$, and set $\beta_{\widehat{j}}\defeq(\mathbf{v}_1,\ldots,\widehat{\mathbf{v}_j},\ldots,\mathbf{v}_m)$.\\
Then $\exists j\in\{2,\ldots,m\}$ such that $\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_{j-1})$ and $\operatorname{span}(\beta_{\widehat{j}})=\operatorname{span}(\beta)$.
\begin{proof}
Choose a nontrivial relation $\sum\limits_{i=1}^m a_i\mathbf{v}_i=\mathbf{0}$ and let $j=\max\{i:a_i\neq0\}$.\\
If $j=1$, then $a_1\mathbf{v}_1=\mathbf{0}$ with $a_1\neq0$, so $\mathbf{v}_1=\mathbf{0}$, contrary to hypothesis. Hence $j\geq2$.\\
Since $a_i=0$ for $i>j$, we have $\mathbf{v}_j=-\sum\limits_{i=1}^{j-1}\frac{a_i}{a_j}\mathbf{v}_i$, so $\mathbf{v}_j\in\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_{j-1})$.\\
Therefore every linear combination of $\beta$ can be rewritten without $\mathbf{v}_j$, which gives $\operatorname{span}(\beta)\subseteq\operatorname{span}(\beta_{\widehat{j}})$.\\
The reverse inclusion is immediate because $\beta_{\widehat{j}}$ is a sublist of $\beta$.\\
\hlt{Structural Commentary}: The last nonzero coefficient gives the removable vector.\\
\hlt{Key Insight}: Dependence gives a later vector removable without losing span.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:finitedimlenofvec} \hlt{Independent Lists Are No Longer Than Spanning Lists}\\
If $\mathcal{V}$ is finite-dimensional, $\alpha=(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ is linearly independent, and $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ spans $\mathcal{V}$, then $m\leq n$.
\begin{proof}
Suppose $m>n$. If $n=0$, then $\operatorname{span}(\gamma)=\{\mathbf{0}\}$, so $\mathcal{V}=\{\mathbf{0}\}$ and no nonempty independent list exists; hence $n\geq1$. Starting from $\gamma$, we construct for each $0\leq k\leq n$ a spanning list of length $n$ containing $\mathbf{u}_1,\ldots,\mathbf{u}_k$.\\
The case $k=0$ is $\gamma$. Suppose the list has been constructed for some $k<n$. Write $\mathbf{u}_{k+1}$ as a linear combination of its entries. If all coefficients of entries other than $\mathbf{u}_1,\ldots,\mathbf{u}_k$ were zero, then $\mathbf{u}_{k+1}\in\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_k)$, contradicting the linear independence of $\alpha$. Thus some remaining entry has nonzero coefficient; replacing that entry by $\mathbf{u}_{k+1}$ preserves span by Lemma \ref{lem:replacement-lemma}(ii). This proves the induction.\\
For $k=n$, the resulting spanning list has length $n$ and contains $\mathbf{u}_1,\ldots,\mathbf{u}_n$; hence it is exactly these vectors, up to order. Thus $\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_n)=\mathcal{V}$.\\
Since $m>n$, this puts $\mathbf{u}_{n+1}$ in $\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$, contradicting the linear independence of $\alpha$.\\
\hlt{Structural Commentary}: Independent length is bounded by spanning length.\\
\hlt{Key Insight}: Independent lists cannot be longer than spanning lists in finite dimensions.
\end{proof}
\end{theorem}


\begin{definition} \hlt{Basis}\\
A list $\beta$ in $\mathcal{V}$ is a \hlt{basis} of $\mathcal{V}$ if it is linearly independent and $\operatorname{span}(\beta)=\mathcal{V}$.
\end{definition}

\begin{proposition} \label{prop:basis-unique-coordinate-representation} \hlt{Unique Coordinate Representation}\\
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$. Then $\beta$ is a basis of $\mathcal{V}$ iff each $\mathbf{v}\in\mathcal{V}$ has a unique $(a_1,\ldots,a_n)\in\mathbb{F}^n$ with $\mathbf{v}=\sum\limits_{j=1}^n a_j\mathbf{v}_j$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Spanning gives existence; Proposition \ref{prop:uniquecoefficientsindependentlist} gives uniqueness.
\item[$(\Leftarrow)$] Existence gives $\operatorname{span}(\beta)=\mathcal{V}$. If $\sum\limits_{j=1}^n a_j\mathbf{v}_j=\mathbf{0}$, uniqueness of zero representation gives $a_1=\cdots=a_n=0$.
\end{enumerate}
\hlt{Structural Commentary}: A basis is exactly existence and uniqueness of coordinates.\\
\hlt{Key Insight}: Coordinates are basis coefficients.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:reducing-spanning-lists} \hlt{Reducing Spanning Lists}\\
Every spanning list in a finite-dimensional vector space contains a sublist that is a basis.
\begin{proof}
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a spanning list of $\mathcal{V}$. If $\beta$ is linearly independent, it is a basis. Otherwise, remove one vector without changing the span: if $\mathbf{v}_1=\mathbf{0}$, omit $\mathbf{v}_1$; if $\mathbf{v}_1\neq\mathbf{0}$, Lemma \ref{lem:lineardependencelemma} gives a removable $\mathbf{v}_j$. The new list is a sublist of $\beta$, still spans $\mathcal{V}$, and has length one less.\\
Repeat this step while the current spanning list is dependent. Since the length decreases each time, the process stops after at most $n$ removals. At termination the remaining sublist still spans $\mathcal{V}$ and is linearly independent, hence is a basis.\\
\hlt{Structural Commentary}: A spanning list may contain redundant vectors. Removing only vectors forced by the others preserves span until a basis remains.\\
\hlt{Key Insight}: Spanning gives enough vectors; reduction deletes exactly the unnecessary ones.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:extending-independent-lists} \hlt{Extending Independent Lists}\\
Every linearly independent list in a finite-dimensional vector space can be extended to a basis.
\begin{proof}
Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be linearly independent in $\mathcal{V}$, and choose a spanning list $(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ of $\mathcal{V}$.\\
Starting with $\mathcal{B}=\beta$, inspect $\mathbf{w}_1,\ldots,\mathbf{w}_n$ in order and append $\mathbf{w}_j$ exactly when $\mathbf{w}_j\notin\operatorname{span}(\mathcal{B})$. Each append preserves linear independence by Proposition \ref{prop:adjoiningandremovingvectors}.\\
After the inspection, each $\mathbf{w}_j$ lies in $\operatorname{span}(\mathcal{B})$: it was either already there when inspected, or it was appended to $\mathcal{B}$. Since the $\mathbf{w}_j$ span $\mathcal{V}$, every vector of $\mathcal{V}$ is a linear combination of vectors in $\operatorname{span}(\mathcal{B})$, and hence lies in $\operatorname{span}(\mathcal{B})$. Thus $\mathcal{B}$ is a spanning independent list, so it is a basis.\\
\hlt{Structural Commentary}: An independent list has no redundancy, but may miss directions. Appending only vectors outside its span fills those directions while preserving independence.\\
\hlt{Key Insight}: A finite-dimensional independent list becomes a basis once enough new directions are added.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:finite-dimensional-spaces-and-subspaces-have-bases}
Every finite-dimensional vector space has a basis, and every subspace of a finite-dimensional vector space is finite-dimensional.
\begin{proof}
Since $\mathcal{V}$ is finite-dimensional, it has a finite spanning list; Theorem \ref{thm:reducing-spanning-lists} reduces it to a basis.\\
Let $\mathcal{U}\subseteq\mathcal{V}$ be a subspace. If $\mathcal{U}=\{\mathbf{0}\}$, then the empty list is a basis of $\mathcal{U}$.\\
Otherwise, begin with any nonzero vector of $\mathcal{U}$, and repeatedly append a vector of $\mathcal{U}$ outside the span of the current list if such a vector exists. Proposition \ref{prop:adjoiningandremovingvectors} keeps the list linearly independent. Theorem \ref{thm:finitedimlenofvec} bounds its length by the length of any spanning list of $\mathcal{V}$, so this finite process must terminate. When it terminates, no vector of $\mathcal{U}$ lies outside the span of the list; hence the list spans $\mathcal{U}$ and is a basis of $\mathcal{U}$.\\
\hlt{Structural Commentary}: Finite-dimensional spaces have finite coordinate systems, and subspaces cannot contain longer independent lists than the ambient space permits.\\
\hlt{Key Insight}: Finite-dimensionality passes downward to subspaces.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:basis-length-well-defined} \hlt{Basis Length Is Well-Defined}\\
Any two bases of a finite-dimensional vector space have the same length.
\begin{proof}
Let $\beta_1=(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ and $\beta_2=(\mathbf{w}_1,\ldots,\mathbf{w}_n)$ be bases of $\mathcal{V}$. Since $\beta_1$ is linearly independent and $\beta_2$ spans $\mathcal{V}$, Theorem \ref{thm:finitedimlenofvec} gives $m\leq n$. Reversing the roles of the two bases gives $n\leq m$. Hence $m=n$.\\
\hlt{Structural Commentary}: Bases are simultaneously independent and spanning, so the comparison theorem traps their lengths in both directions. This makes dimension well-defined.\\
\hlt{Key Insight}: Dimension is an intrinsic number, not an artefact of a particular basis.
\end{proof}
\end{theorem}

\begin{definition} \hlt{Dimension}\\
The zero vector space has \hlt{dimension} $0$. If $\mathcal{V}$ has a basis of positive integer length $n$, then $\mathcal{V}$ is \hlt{$n$-dimensional} and $\dim(\mathcal{V})=n$. A vector space is \hlt{finite-dimensional} if it has finite dimension; otherwise it is \hlt{infinite-dimensional}.
\end{definition}

\begin{proposition} \label{prop:dimension-bounds-and-basis-criteria} \hlt{Dimension Bounds and Basis Criteria}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U} \subseteq \mathcal{V}$ is a subspace, then $\dim(\mathcal{U}) \leq \dim(\mathcal{V})$, with equality if and only if $\mathcal{U}=\mathcal{V}$. Moreover, a spanning list of length $\dim(\mathcal{V})$ is automatically a basis, and a linearly independent list of length $\dim(\mathcal{V})$ is automatically a basis.
\begin{proof}
Let $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ be a basis of $\mathcal{U}$ and $(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ a basis of $\mathcal{V}$.\\
Theorem \ref{thm:finitedimlenofvec} gives $m\leq n$, so $\dim(\mathcal{U})\leq\dim(\mathcal{V})$. If $\mathcal{U}=\mathcal{V}$, equality is clear.\\
Conversely, if $m=n$, then the basis of $\mathcal{U}$ is an independent list in $\mathcal{V}$ of length $\dim(\mathcal{V})$; extending it to a basis of $\mathcal{V}$ adds no vectors, so it already spans $\mathcal{V}$. Hence $\mathcal{U}=\mathcal{V}$. Let $\gamma$ be a spanning list of length $n=\dim(\mathcal{V})$.\\
By Theorem \ref{thm:reducing-spanning-lists}, $\gamma$ contains a sublist that is a basis of $\mathcal{V}$. Any proper sublist has length $<n$, impossible by Theorem \ref{thm:basis-length-well-defined}; hence no vector was removed and $\gamma$ is itself a basis.\\
Let $\alpha$ be a linearly independent list of length $n$. By Theorem \ref{thm:extending-independent-lists}, $\alpha$ extends to a basis of $\mathcal{V}$.\\
Adding any vector would produce a basis of length $>n$, again contradicting Theorem \ref{thm:basis-length-well-defined}; hence no vector was added and $\alpha$ is itself a basis.\\
\hlt{Structural Commentary}: Dimension bounds subspaces and detects full-size lists. Equality of dimensions forces a subspace to be the whole space.\\
\hlt{Key Insight}: In dimension $n$, $n$ independent or $n$ spanning vectors already form a basis.
\end{proof}
\end{proposition}
