\subsection{Direct Sums, Quotients, and Decomposition}

% **Role:** algebraic language of linear structure.
%
% Sums, direct sums, complements, quotient spaces, induced maps, dimension formulas.

Direct sums split spaces into complementary pieces; quotients collapse selected directions. Dimension formulas keep both constructions controlled.

\begin{definition} \hlt{Sums of Subspaces}\\
If $\mathcal{U}_1, \ldots, \mathcal{U}_m$ are subspaces of $\mathcal{V}$, define $\mathcal{U}_1 + \cdots + \mathcal{U}_m \defeq \left\{ \mathbf{u}_1 + \cdots + \mathbf{u}_m : \mathbf{u}_j \in \mathcal{U}_j \right\}$.\\
In particular, for subspaces $\mathcal{U},\mathcal{W}\subseteq\mathcal{V}$, $\mathcal{U}+\mathcal{W} = \operatorname{span}(\mathcal{U}\cup\mathcal{W}) = \left\{ \mathbf{u}+\mathbf{w} : \mathbf{u}\in\mathcal{U},\ \mathbf{w}\in\mathcal{W} \right\}$.\\
If every vector in this sum has a unique such decomposition, then the sum is direct and we write $\mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$.\\
For two subspaces, $\mathcal{U}+\mathcal{W}$ is direct if and only if $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$; in this case it is denoted $\mathcal{U}\oplus\mathcal{W}$.
\end{definition}

\begin{perspective} \label{per:sum-not-union} \hlt{Sum Is Not Union}\\
The union $\mathcal{U}\cup\mathcal{W}$ keeps vectors from either subspace; the sum $\mathcal{U}+\mathcal{W}$ adds vectors from both. Unless one subspace contains the other, $\mathcal{U}\cup\mathcal{W}$ is not closed under addition.
\end{perspective}

\begin{theorem} \label{thm:finite-union-proper-subspaces-infinite-field} \hlt{Finite Unions of Proper Subspaces}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be over an infinite field $\mathbb{F}$. If $\mathcal{U}_1,\ldots,\mathcal{U}_n\subsetneq\mathcal{V}$ are subspaces, then $\mathcal{V}\neq\bigcup\limits_{j=1}^{n}\mathcal{U}_j$.
\begin{proof}
Assume a minimal cover $\mathcal{V}=\bigcup\limits_{j=1}^{n}\mathcal{U}_j$ by proper subspaces, so $\mathcal{U}_i\not\subseteq\bigcup\limits_{j\neq i}\mathcal{U}_j$.\\
Choose $\mathbf{w}\in\mathcal{U}_1\setminus\bigcup\limits_{j=2}^{n}\mathcal{U}_j$, $\mathbf{v}\in\mathcal{V}\setminus\mathcal{U}_1$
Let $A=\mathbf{v}+\mathbb{F}\mathbf{w}$. Then $\abs{A}=\abs{\mathbb{F}}=\infty$, $A\cap\mathcal{U}_1=\emptyset$, and for $j\geq2$, $\abs{A\cap\mathcal{U}_j}\leq1$, since two points $\mathbf{v}+r_1\mathbf{w},\mathbf{v}+r_2\mathbf{w}\in\mathcal{U}_j$ with $r_1\neq r_2$ imply $\mathbf{w}\in\mathcal{U}_j$.\\
Hence $\abs{A\cap\bigcup\limits_{j=1}^{n}\mathcal{U}_j}\leq n-1<\infty$, contradicting $A\subseteq\mathcal{V}=\bigcup\limits_{j=1}^{n}\mathcal{U}_j$.\\
\hlt{Structural Commentary}: Over an infinite field, finitely many proper linear pieces cannot cover every affine line.\\
\hlt{Key Insight}: One affine line has too many points for finitely many proper intersections.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:sum-versus-direct-sum-existence} \hlt{Sums Always Exist; Directness Is Extra}\\
For subspaces $\mathcal{U},\mathcal{W}\subseteq\mathcal{V}$, the sum $\mathcal{U}+\mathcal{W}$ is always a subspace by Proposition \ref{prop:sum-subspaces-smallest-containing-subspace}. Writing $\mathcal{U}\oplus\mathcal{W}$ asserts the extra condition $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$, equivalently uniqueness of the decomposition $\mathbf{u}+\mathbf{w}$.
\end{perspective}

\begin{proposition} \label{prop:uniqueness-direct-sum-two-subspaces} \hlt{Uniqueness in a Direct Sum}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$ with $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$. Then each vector in $\mathcal{U}\oplus\mathcal{W}$ has a unique representation $\mathbf{u}+\mathbf{w}$ with $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{W}$.
\begin{proof}
Existence follows from the definition of $\mathcal{U}+\mathcal{W}$. For uniqueness, suppose $\mathbf{u}_1+\mathbf{w}_1=\mathbf{u}_2+\mathbf{w}_2$, $\mathbf{u}_1,\mathbf{u}_2\in\mathcal{U}$, $\mathbf{w}_1,\mathbf{w}_2\in\mathcal{W}$. Then $\mathbf{u}_1-\mathbf{u}_2=\mathbf{w}_2-\mathbf{w}_1 \in \mathcal{U}, \mathcal{W}$, hence equals $\mathbf{0}$. Thus $\mathbf{u}_1=\mathbf{u}_2$ and $\mathbf{w}_1=\mathbf{w}_2$.\\
\hlt{Structural Commentary}: Trivial intersection prevents cancellation between summands.\\
\hlt{Key Insight}: Directness means unique decomposition.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:sum-subspaces-smallest-containing-subspace} \hlt{Sum of Subspaces Is the Smallest Containing Subspace}\\
Suppose $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is the smallest subspace of $\mathcal{V}$ containing $\mathcal{U}_1,\ldots,\mathcal{U}_m$.
\begin{proof}
Since $\mathbf{0}\in\mathcal{U}_j$ for each $j$, we have $\mathbf{0}\in\mathcal{U}_1+\cdots+\mathcal{U}_m$. If $\mathbf{x}=\mathbf{u}_1+\cdots+\mathbf{u}_m$ and $\mathbf{y}=\mathbf{v}_1+\cdots+\mathbf{v}_m$ lie in $\mathcal{U}_1+\cdots+\mathcal{U}_m$, then for any $c \in \mathbb{F}$, $c\mathbf{x}+\mathbf{y} = (c\mathbf{u}_1+\mathbf{v}_1)+\cdots+(c\mathbf{u}_m+\mathbf{v}_m) \in \mathcal{U}_1+\cdots+\mathcal{U}_m$, because $c\mathbf{u}_j+\mathbf{v}_j\in\mathcal{U}_j$ for each $j$. By Theorem \ref{thm:subspacetest}, $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a subspace of $\mathcal{V}$.\\
Each $\mathcal{U}_j$ is contained in $\mathcal{U}_1+\cdots+\mathcal{U}_m$ by taking all other summands to be $\mathbf{0}$. If $\mathcal{W}$ is any subspace of $\mathcal{V}$ containing $\mathcal{U}_1,\ldots,\mathcal{U}_m$, then closure under addition gives $\mathbf{u}_1+\cdots+\mathbf{u}_m\in\mathcal{W}$ whenever $\mathbf{u}_j\in\mathcal{U}_j$. Hence $\mathcal{U}_1+\cdots+\mathcal{U}_m\subseteq\mathcal{W}$, so the sum is the smallest such subspace.\\
\hlt{Structural Commentary}: Sums of subspaces are obtained by closing several subspaces under finite addition. Minimality follows because every containing subspace is already closed under those sums.\\
\hlt{Key Insight}: The sum adds no extra structure beyond the linear combinations forced by the given subspaces.
\end{proof}
\end{proposition}

\begin{definition} \label{def:subspace-meet-and-join} \hlt{Meet and Join of Subspaces}\\
For subspaces $\{\mathcal{U}_i:i\in I\}$ of $\mathcal{V}$, define $\bigwedge\limits_{i\in I}\mathcal{U}_i\defeq\bigcap\limits_{i\in I}\mathcal{U}_i$; and $\bigvee\limits_{i\in I}\mathcal{U}_i\defeq\sum\limits_{i\in I}\mathcal{U}_i\defeq\operatorname{span}\left(\bigcup\limits_{i\in I}\mathcal{U}_i\right)$.\\
The empty meet is $\mathcal{V}$, and the empty join is $\{\mathbf{0}\}$.
\end{definition}

\begin{proposition} \label{prop:subspaces-form-complete-lattice} \hlt{Subspace Meet and Join}\\
For subspaces $\{\mathcal{U}_i:i\in I\}$ of $\mathcal{V}$, ordered by inclusion, $\bigwedge\limits_{i\in I}\mathcal{U}_i=\inf\{\mathcal{U}_i:i\in I\}$; and $\bigvee\limits_{i\in I}\mathcal{U}_i=\sup\{\mathcal{U}_i:i\in I\}$.
\begin{proof}
If $I=\emptyset$, conventions give vacuous bounds $\mathcal{V}$ and $\{\mathbf{0}\}$. Assume $I\neq\emptyset$.\\
The meet $M=\bigcap_{i\in I}\mathcal{U}_i$ is a subspace by Proposition \ref{prop:intersectionsubspaces}; moreover $M\subseteq\mathcal{U}_i$ for all $i$, and any subspace $\mathcal{W}$ with $\mathcal{W}\subseteq\mathcal{U}_i$ for all $i$ satisfies $\mathcal{W}\subseteq M$. The join $J=\operatorname{span}\left(\bigcup_{i\in I}\mathcal{U}_i\right)$ is a subspace by Theorem \ref{thm:span-smallest-containing-subspace}; moreover $\mathcal{U}_i\subseteq J$ for all $i$, and any subspace $\mathcal{W}$ with $\mathcal{U}_i\subseteq\mathcal{W}$ for all $i$ contains $\bigcup_{i\in I}\mathcal{U}_i$, hence contains $J$.\\
\hlt{Structural Commentary}: Meet enforces common constraints; join closes the combined directions under span.\\
\hlt{Key Insight}: Meet is intersection, while join is span of the union.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:subspace-lattice-complete-modular} \hlt{Subspace Lattice}\\
Let $\operatorname{Sub}(\mathcal{V})$ be the set of subspaces of $\mathcal{V}$, ordered by inclusion. Then $\operatorname{Sub}(\mathcal{V})$ is a complete modular lattice with bottom $\{\mathbf{0}\}$, top $\mathcal{V}$, and $\bigwedge\limits_{i\in I}\mathcal{U}_i=\bigcap\limits_{i\in I}\mathcal{U}_i$, $\bigvee\limits_{i\in I}\mathcal{U}_i=\sum\limits_{i\in I}\mathcal{U}_i=\operatorname{span}\left(\bigcup\limits_{i\in I}\mathcal{U}_i\right)$.\\
Equivalently, if $\mathcal{U}\subseteq\mathcal{S}$, then $\mathcal{S}\cap(\mathcal{T}+\mathcal{U})=(\mathcal{S}\cap\mathcal{T})+\mathcal{U}$.
\begin{proof}
Completeness and the meet/join formulas are Proposition \ref{prop:subspaces-form-complete-lattice}. Assume $\mathcal{U}\subseteq\mathcal{S}$.\\
Then $(\mathcal{S}\cap\mathcal{T})+\mathcal{U}\subseteq\mathcal{S}\cap(\mathcal{T}+\mathcal{U})$. Conversely, if $\mathbf{x}\in\mathcal{S}\cap(\mathcal{T}+\mathcal{U})$, write $\mathbf{x}=\mathbf{t}+\mathbf{u}$ with $\mathbf{t}\in\mathcal{T}$ and $\mathbf{u}\in\mathcal{U}\subseteq\mathcal{S}$.\\
Then $\mathbf{t}=\mathbf{x}-\mathbf{u}\in\mathcal{S}\cap\mathcal{T}$, so $\mathbf{x}\in(\mathcal{S}\cap\mathcal{T})+\mathcal{U}$.\\
\hlt{Structural Commentary}: The subspace lattice is complete and modular, not generally distributive.\\
\hlt{Key Insight}: If one summand already lies inside $\mathcal{S}$, intersecting with $\mathcal{S}$ can be pushed through the sum.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:direct-sum-equivalent-conditions} \hlt{Equivalent Conditions for a Direct Sum}\\
For subspaces $\mathcal{U}_1, \ldots, \mathcal{U}_m \subseteq \mathcal{V}$, the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$;
\item every $\mathbf{v} \in \mathcal{V}$ has a unique representation $\mathbf{v} = \mathbf{u}_1 + \cdots + \mathbf{u}_m$ with $\mathbf{u}_j \in \mathcal{U}_j$;
\item $\mathcal{V}=\mathcal{U}_1+\cdots+\mathcal{U}_m$, and if $\mathbf{u}_1 + \cdots + \mathbf{u}_m = \mathbf{0}$ with $\mathbf{u}_j \in \mathcal{U}_j$, then $\mathbf{u}_1 = \cdots = \mathbf{u}_m = \mathbf{0}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman)*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ This follows from the definition of direct sum.
\item[$(ii)$]$\Rightarrow (iii)$ Every vector lies in $\mathcal{U}_1+\cdots+\mathcal{U}_m$, so $\mathcal{V}=\mathcal{U}_1+\cdots+\mathcal{U}_m$. If $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$, then this representation of $\mathbf{0}$ must equal $\mathbf{0}+\cdots+\mathbf{0}$ by uniqueness, so each $\mathbf{u}_j=\mathbf{0}$.
\item[$(iii)$]$\Rightarrow (i)$ Since $\mathcal{V}=\mathcal{U}_1+\cdots+\mathcal{U}_m$, every $\mathbf{v}\in\mathcal{V}$ has at least one representation. If $\mathbf{v}=\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{w}_1+\cdots+\mathbf{w}_m$, then $(\mathbf{u}_1-\mathbf{w}_1)+\cdots+(\mathbf{u}_m-\mathbf{w}_m)=\mathbf{0}$. By iii., $\mathbf{u}_j-\mathbf{w}_j=\mathbf{0}$ for every $j$, so the representation is unique. Hence $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_m$.
\end{enumerate}
\hlt{Structural Commentary}: Direct sums require both spanning by the summands and no cancellation among them. The zero-representation condition isolates the no-cancellation part.\\
\hlt{Key Insight}: Uniqueness is equivalent to saying the only way the summands add to zero is trivially.
\end{proof}
\end{proposition}

\begin{definition} \label{def:internal-direct-sum-family} \hlt{Internal Direct Sum of a Family}\\
Let $\{\mathcal{U}_i:i\in I\}$ be subspaces of $\mathcal{V}$.\\
Then $\mathcal{V}=\bigoplus\limits_{i\in I}\mathcal{U}_i$, if every $\mathbf{v}\in\mathcal{V}$ has a unique finitely supported family $(\mathbf{u}_i)_{i\in I}$ with $\mathbf{u}_i\in\mathcal{U}_i$ such that $\mathbf{v}=\sum\limits_{i\in I}\mathbf{u}_i$.\\
Equivalently, $\mathcal{V}=\sum\limits_{i\in I}\mathcal{U}_i$ and $\sum\limits_{i\in F}\mathbf{u}_i=\mathbf{0}$, with $F\subseteq I$ finite and $\mathbf{u}_i\in\mathcal{U}_i$, implies $\mathbf{u}_i=\mathbf{0}$ for all $i\in F$.
\end{definition}

\begin{proposition} \label{prop:internal-direct-sum-family-criterion} \hlt{Family Direct-Sum Criterion}\\
For $\{\mathcal{U}_i\}_{i\in I}\subseteq\operatorname{Sub}(\mathcal{V})$, $\sum\limits_{i\in I}\mathcal{U}_i$ is direct $\Longleftrightarrow$ $\forall i\in I$, $\mathcal{U}_i\cap\sum\limits_{j\in I\setminus\{i\}}\mathcal{U}_j=\{\mathbf{0}\}$
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathbf{x}\in\mathcal{U}_i\cap\sum\limits_{j\in I\setminus\{i\}}\mathcal{U}_j$, then $\mathbf{x}=\sum\limits_{j\in F}\mathbf{u}_j$ for some finite $F\subseteq I\setminus\{i\}$ and $\mathbf{u}_j\in\mathcal{U}_j$.\\
Directness applied to $\mathbf{x}-\sum\limits_{j\in F}\mathbf{u}_j=\mathbf{0}$ gives $\mathbf{x}=\mathbf{0}$.
\item[$(\Leftarrow)$] If $\sum\limits_{k=1}^{m}\mathbf{u}_{i_k}=\mathbf{0}$ with distinct $i_k$ and $\mathbf{u}_{i_k}\in\mathcal{U}_{i_k}$, then for each $k$, $\mathbf{u}_{i_k}=-\sum\limits_{\ell\neq k}\mathbf{u}_{i_\ell}\in\mathcal{U}_{i_k}\cap\sum\limits_{j\in I\setminus\{i_k\}}\mathcal{U}_j=\{\mathbf{0}\}$.\\
Thus all finite zero-relations are trivial, so the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: Family directness requires each summand to miss all others together.\\
\hlt{Key Insight}: Directness is subspace independence.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Product of Vector Spaces}\\
Let $\mathcal{V}_1,\ldots,\mathcal{V}_m$ be vector spaces over $\mathbb{F}$. Their \hlt{product} is $\mathcal{V}_1\times\cdots\times\mathcal{V}_m \defeq \{(\mathbf{v}_1,\ldots,\mathbf{v}_m):\mathbf{v}_j\in\mathcal{V}_j\}$.\\
Addition, scalar multiplication defined componentwise: $(\mathbf{u}_1,\ldots,\mathbf{u}_m)+(\mathbf{v}_1,\ldots,\mathbf{v}_m) \defeq (\mathbf{u}_1+\mathbf{v}_1,\ldots,\mathbf{u}_m+\mathbf{v}_m)$ and $a(\mathbf{v}_1,\ldots,\mathbf{v}_m) \defeq (a\mathbf{v}_1,\ldots,a\mathbf{v}_m)$.
\end{definition}

\begin{proposition} \label{prop:product-vector-space} \hlt{Products Are Vector Spaces}\\
If $\mathcal{V}_1,\ldots,\mathcal{V}_m$ are vector spaces over $\mathbb{F}$, then $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ is a vector space under componentwise operations.
\begin{proof}
The additive identity is $(\mathbf{0}_1,\ldots,\mathbf{0}_m)$, where $\mathbf{0}_j$ is the zero vector of $\mathcal{V}_j$, and additive inverse of $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is $(-\mathbf{v}_1,\ldots,-\mathbf{v}_m)$. All remaining vector space axioms hold componentwise because they hold in each $\mathcal{V}_j$.\\
\hlt{Structural Commentary}: Product spaces collect several vector spaces into one coordinatewise structure.\\
\hlt{Key Insight}: Every vector space axiom reduces to the same axiom in each component.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-product} \hlt{Dimension of a Product}\\
If $\mathcal{V}_1,\ldots,\mathcal{V}_m$ are finite-dimensional vector spaces over $\mathbb{F}$, then $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ is finite-dimensional and
\begin{align}
\dim(\mathcal{V}_1\times\cdots\times\mathcal{V}_m) = \dim(\mathcal{V}_1)+\cdots+\dim(\mathcal{V}_m) \nonumber
\end{align}
\begin{proof}
For each $j$, let $(\mathbf{v}_{j,1},\ldots,\mathbf{v}_{j,n_j})$ be a basis of $\mathcal{V}_j$. For every $j$ and $k$, let $\mathbf{e}_{j,k}$ denote the element of $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ whose $j$th component is $\mathbf{v}_{j,k}$ and whose other components are zero.\\
The list of all $\mathbf{e}_{j,k}$ spans the product because each component can be expanded in its own basis. If a linear combination of the $\mathbf{e}_{j,k}$ is zero, then looking at the $j$th component gives a linear relation among $(\mathbf{v}_{j,1},\ldots,\mathbf{v}_{j,n_j})$, so all coefficients in that component are zero. Hence the list is linearly independent and therefore is a basis of the product. Its length is $n_1+\cdots+n_m$.\\
\hlt{Structural Commentary}: A basis for a product is formed by placing each component basis into its own slot.\\
\hlt{Key Insight}: Product dimension counts the independent directions contributed by each factor.
\end{proof}
\end{theorem}

\begin{definition} \label{def:external-products-and-direct-sums-family} \hlt{External Product and Direct Sum}\\
Let $\{\mathcal{V}_i:i\in I\}$ be $\mathbb{F}$-vector spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{external direct product} is $\prod\limits_{i\in I}\mathcal{V}_i =\{\mathbf{v}:I\to\bigcup\limits_{i\in I}\mathcal{V}_i:\mathbf{v}(i)\in\mathcal{V}_i\}$.
\item Operations are componentwise: $(\mathbf{u}+\mathbf{v})(i)=\mathbf{u}(i)+\mathbf{v}(i)$ and $(a\mathbf{v})(i)=a\mathbf{v}(i)$.
\item The \hlt{support} is $\operatorname{supp}(\mathbf{v})=\{i\in I:\mathbf{v}(i)\neq\mathbf{0}\}$.\\
The \hlt{external direct sum} is $\bigoplus\limits_{i\in I}^{\mathrm{ext}}\mathcal{V}_i =\{\mathbf{v}\in\prod\limits_{i\in I}\mathcal{V}_i:\abs{\operatorname{supp}(\mathbf{v})}<\infty\}$. If $\abs{I}<\infty$, then $\bigoplus\limits_{i\in I}^{\mathrm{ext}}\mathcal{V}_i=\prod\limits_{i\in I}\mathcal{V}_i$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:external-direct-sum-family-vector-space} \hlt{External Sums Are Finite-Support Products}\\
With componentwise operations, $\prod\limits_{i\in I}\mathcal{V}_i$ is an $\mathbb{F}$-vector space, $\bigoplus\limits_{i\in I}^{\mathrm{ext}}\mathcal{V}_i =\{\mathbf{v}\in\prod\limits_{i\in I}\mathcal{V}_i:\abs{\operatorname{supp}(\mathbf{v})}<\infty\} \leq \prod\limits_{i\in I}\mathcal{V}_i$.
\begin{proof}
Let $P=\prod_{i\in I}\mathcal{V}_i$. For $\mathbf{u},\mathbf{v}\in P$ and $a,b\in\mathbb{F}$, $(a\mathbf{u}+b\mathbf{v})(i)=a\mathbf{u}(i)+b\mathbf{v}(i)\in\mathcal{V}_i$.\\
The zero $0_P(i)=0_i$, inverses $(-\mathbf{v})(i)=-\mathbf{v}(i)$, and all axioms are coordinatewise.\\
Let $S=\bigoplus_{i\in I}^{\mathrm{ext}}\mathcal{V}_i$. If $\mathbf{u},\mathbf{v}\in S$ and $a,b\in\mathbb{F}$, then $\operatorname{supp}(a\mathbf{u}+b\mathbf{v})\subseteq \operatorname{supp}(\mathbf{u})\cup\operatorname{supp}(\mathbf{v})$, so $a\mathbf{u}+b\mathbf{v}\in S$.\\
Also $0_P\in S$. Hence $S\leq P$.\\
\hlt{Structural Commentary}: $\prod_i\mathcal{V}_i$: arbitrary support; $\bigoplus_i^{\mathrm{ext}}\mathcal{V}_i$: finite support.\\
\hlt{Key Insight}: Finite support is exactly what finite linear combinations preserve.
\end{proof}
\end{proposition}


\begin{proposition} \label{prop:direct-sum-map-injective} \hlt{Direct Sum Map}\\
Suppose $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Define $\Gamma:\mathcal{U}_1\times\cdots\times\mathcal{U}_m\to\mathcal{U}_1+\cdots+\mathcal{U}_m$ by $\Gamma(\mathbf{u}_1,\ldots,\mathbf{u}_m)=\mathbf{u}_1+\cdots+\mathbf{u}_m$.\\
Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a direct sum if and only if $\Gamma$ is injective.
\begin{proof}
The map $\Gamma$ is linear, and its kernel consists of all $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ such that $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a direct sum and $\Gamma(\mathbf{u}_1,\ldots,\mathbf{u}_m)=\mathbf{0}$, uniqueness of the zero decomposition forces each $\mathbf{u}_j=\mathbf{0}$. Hence $\ker(\Gamma)=\{\mathbf{0}\}$, so $\Gamma$ is injective.
\item[$(\Leftarrow)$] If $\Gamma$ is injective and $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$, then $(\mathbf{u}_1,\ldots,\mathbf{u}_m)\in\ker(\Gamma)$, so every $\mathbf{u}_j=\mathbf{0}$. Thus the only zero decomposition is trivial, which is the direct-sum condition.
\end{enumerate}
\hlt{Structural Commentary}: The sum map packages all decompositions into one linear map.\\
\hlt{Key Insight}: Directness is injectivity of the map that adds the components.
\end{proof}
\end{proposition}


\begin{theorem} \label{thm:dimension-formula-for-sum} \hlt{Dimension Formula for a Sum}\\
If $\mathcal{U}, \mathcal{W}$ are finite-dimensional subspaces of $\mathcal{V}$, then $\dim(\mathcal{U} + \mathcal{W}) = \dim(\mathcal{U}) + \dim(\mathcal{W}) - \dim(\mathcal{U} \cap \mathcal{W})$.
\begin{proof}
Choose a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r)$ of $\mathcal{U} \cap \mathcal{W}$, extend it to a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p)$ of $\mathcal{U}$, and to a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{w}_1, \ldots, \mathbf{w}_q)$ of $\mathcal{W}$. Then $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p, \mathbf{w}_1, \ldots, \mathbf{w}_q) $ spans $\mathcal{U}+\mathcal{W}$ because every vector in the sum is the sum of one vector from $\mathcal{U}$ and one vector from $\mathcal{W}$.\\
To prove linear independence, suppose $a_1\mathbf{e}_1+\cdots+a_r\mathbf{e}_r+b_1\mathbf{u}_1+\cdots+b_p\mathbf{u}_p+c_1\mathbf{w}_1+\cdots+c_q\mathbf{w}_q=\mathbf{0}$.
Then $a_1\mathbf{e}_1+\cdots+a_r\mathbf{e}_r+b_1\mathbf{u}_1+\cdots+b_p\mathbf{u}_p$ lies in $\mathcal{U}\cap\mathcal{W}$. Since $(\mathbf{e}_1,\ldots,\mathbf{e}_r)$ is a basis of $\mathcal{U}\cap\mathcal{W}$ and $(\mathbf{e}_1,\ldots,\mathbf{e}_r,\mathbf{u}_1,\ldots,\mathbf{u}_p)$ is a basis of $\mathcal{U}$, we get $b_1=\cdots=b_p=0$. The remaining relation lies in the basis $(\mathbf{e}_1,\ldots,\mathbf{e}_r,\mathbf{w}_1,\ldots,\mathbf{w}_q)$ of $\mathcal{W}$, so $a_1=\cdots=a_r=c_1=\cdots=c_q=0$. Thus the displayed list is a basis of $\mathcal{U}+\mathcal{W}$.\\
Therefore $\dim(\mathcal{U} + \mathcal{W}) = r + p + q = (r+p) + (r+q) - r$.\\
\hlt{Structural Commentary}: The formula corrects double-counting: vectors in $\mathcal{U} \cap \mathcal{W}$ are counted once in each subspace. Extending an intersection basis separates shared directions from new directions.\\
\hlt{Key Insight}: The dimension of a sum is the sum of dimensions minus the overlap.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:dimension-direct-sum} \hlt{Dimension of a Direct Sum}\\
If $\mathcal{U}$ and $\mathcal{W}$ are finite-dimensional subspaces of $\mathcal{V}$ with $\mathcal{U} \cap \mathcal{W} = \{ \mathbf{0} \}$, then $\dim(\mathcal{U} \oplus \mathcal{W}) = \dim(\mathcal{U}) + \dim(\mathcal{W})$.
\begin{proof}
Since $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$, the sum is direct and $\dim(\mathcal{U}\cap\mathcal{W})=0$. The result follows from Theorem \ref{thm:dimension-formula-for-sum}.\\
\hlt{Structural Commentary}: When the overlap is trivial, there is no double-counting in the dimension formula. Direct sums are exactly the case where the overlap term vanishes.\\
\hlt{Key Insight}: Dimensions add exactly for direct sums.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:direct-sum-iff-dimensions-add} \hlt{Direct Sums and Dimension}\\
Suppose $\mathcal{V}$ is finite-dimensional and $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a direct sum if and only if $\dim(\mathcal{U}_1+\cdots+\mathcal{U}_m) = \dim(\mathcal{U}_1)+\cdots+\dim(\mathcal{U}_m)$.
\begin{proof}
For each $j$, choose a basis $\beta_j$ of $\mathcal{U}_j$, and concatenate these bases into a list $\beta$.\\
This list spans $\mathcal{U}_1+\cdots+\mathcal{U}_m$ and has length $\dim(\mathcal{U}_1)+\cdots+\dim(\mathcal{U}_m)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If the sum is direct, any linear relation among the vectors in $\beta$ gives a zero decomposition $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ with $\mathbf{u}_j\in\mathcal{U}_j$. Directness gives $\mathbf{u}_j=\mathbf{0}$ for every $j$, and independence of each $\beta_j$ gives all coefficients zero. Thus $\beta$ is a basis of the sum, and the dimensions add.
\item[$(\Leftarrow)$] If the dimensions add, then $\beta$ is a spanning list whose length equals the dimension of the sum, so Proposition \ref{prop:dimension-bounds-and-basis-criteria} makes $\beta$ a basis. A zero decomposition $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ expands to a zero linear relation in this basis, so every $\mathbf{u}_j=\mathbf{0}$. Hence the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: Direct sums add dimensions without hidden overlap.\\
\hlt{Key Insight}: Equality of dimensions means the concatenated bases have no linear redundancy.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:large-subspaces-intersect} \hlt{Large Subspaces Must Intersect}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of a finite-dimensional vector space $\mathcal{V}$, and let $k$ be a positive integer.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item If $\dim(\mathcal{U})+\dim(\mathcal{W})>\dim(\mathcal{V})$, then $\mathcal{U}\cap\mathcal{W}$ contains a nonzero vector.
\item If $\dim(\mathcal{U})+\dim(\mathcal{W})\geq\dim(\mathcal{V})+k$, then $\mathcal{U}\cap\mathcal{W}$ contains $k$ linearly independent vectors.
\end{enumerate}
\begin{proof}
Since $\mathcal{U}+\mathcal{W}\subseteq\mathcal{V}$, Theorem \ref{thm:dimension-formula-for-sum} gives
\begin{align}
\dim(\mathcal{U}\cap\mathcal{W}) =\dim(\mathcal{U})+\dim(\mathcal{W})-\dim(\mathcal{U}+\mathcal{W}) \geq \dim(\mathcal{U})+\dim(\mathcal{W})-\dim(\mathcal{V}) \nonumber
\end{align}
If $\dim(\mathcal{U})+\dim(\mathcal{W})>\dim(\mathcal{V})$, the right-hand side is at least $1$, so the intersection contains a nonzero vector. If $\dim(\mathcal{U})+\dim(\mathcal{W})\geq\dim(\mathcal{V})+k$, then $\dim(\mathcal{U}\cap\mathcal{W})\geq k$, so intersection contains $k$ linearly independent vectors.\\
\hlt{Structural Commentary}: Excess dimension appears as overlap.\\
\hlt{Key Insight}: Two large subspaces cannot fit inside $\mathcal{V}$ without sharing directions.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:complements-exist-finite-dimensions} \hlt{Complements Exist in Finite Dimensions}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U} \subseteq \mathcal{V}$ is a subspace, then there exists a subspace $\mathcal{W}$ such that $\mathcal{V} = \mathcal{U} \oplus \mathcal{W}$.
\begin{proof}
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ of $\mathcal{U}$ and extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_n)$ of $\mathcal{V}$.\\
Let $\mathcal{W}=\operatorname{span}(\mathbf{w}_1,\ldots,\mathbf{w}_n)$. Every vector in $\mathcal{V}$ is a linear combination of this extended basis, so $\mathcal{V}=\mathcal{U}+\mathcal{W}$. If $\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$, then $\mathbf{v}$ has one representation using only $\mathbf{u}_1,\ldots,\mathbf{u}_m$ and another using only $\mathbf{w}_1,\ldots,\mathbf{w}_n$. Uniqueness of coefficients in the basis of $\mathcal{V}$ forces $\mathbf{v}=\mathbf{0}$. Hence $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.\\
\hlt{Structural Commentary}: Complements split a space into a subspace and the remaining basis directions.\\
\hlt{Key Insight}: Extend a basis of $\mathcal{U}$ to split $\mathcal{V}$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:complements-exist-arbitrary-vector-spaces} \hlt{Complements Exist}\\
Assuming Zorn's lemma, for every $\mathcal{U}\leq\mathcal{V}$, $\exists\,\mathcal{W}\leq\mathcal{V}$ such that $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.
\begin{proof}
By Theorem \ref{thm:hamel-basis-extension}, choose a Hamel basis $\mathcal{B}_{\mathcal{U}}$ of $\mathcal{U}$ and a Hamel basis $\mathcal{B}\supseteq\mathcal{B}_{\mathcal{U}}$ of $\mathcal{V}$.\\
Put $\mathcal{C}=\mathcal{B}\setminus\mathcal{B}_{\mathcal{U}}$ and $\mathcal{W}=\operatorname{span}(\mathcal{C})$. For $\mathbf{v}\in\mathcal{V}$, its finite $\mathcal{B}$-expansion splits, with finite $F_{\mathcal{U}}\subseteq\mathcal{B}_{\mathcal{U}}$ and $F_{\mathcal{C}}\subseteq\mathcal{C}$,\\
as $\mathbf{v} =\sum\limits_{b\in F_{\mathcal{U}}}a_b b+\sum\limits_{c\in F_{\mathcal{C}}}a_c c \in \mathcal{U}+\mathcal{W}$.
If $\mathbf{x}\in\mathcal{U}\cap\mathcal{W}$, then its $\mathcal{B}$-coordinates are supported in both $\mathcal{B}_{\mathcal{U}}$ and $\mathcal{C}$; uniqueness gives $\mathbf{x}=\mathbf{0}$. Hence $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.\\
\hlt{Structural Commentary}: Extend a basis of $\mathcal{U}$; span the new directions.\\
\hlt{Key Insight}: The uniqueness of basis expansion becomes the uniqueness of direct-sum decomposition.
\end{proof}
\end{theorem}

\begin{definition} \label{def:projection-onto-subspace-along-complement} \hlt{Projection Along a Complement}\\
Let $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$. The \hlt{projection onto $\mathcal{U}$ along $\mathcal{W}$} is operator $\rho_{\mathcal{U},\mathcal{W}}\in\LL(\mathcal{V})$, $\rho_{\mathcal{U},\mathcal{W}}(\mathbf{u}+\mathbf{w})\defeq \mathbf{u}$, $\mathbf{u}\in\mathcal{U}$, $\mathbf{w}\in\mathcal{W}$.
\end{definition}

\begin{proposition} \label{prop:algebraic-projection-range-kernel-idempotent} \hlt{Algebraic Projections Are Idempotents}\\
For $P\in\LL(\mathcal{V})$, $P^2=P$ $\Longleftrightarrow$ $\mathcal{V}=\operatorname{range}(P)\oplus\ker(P)$ and $P=\rho_{\operatorname{range}(P),\ker(P)}$.\\
In particular, if $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$ and $\rho=\rho_{\mathcal{U},\mathcal{W}}$, then $\rho^2=\rho$, $\operatorname{range}(\rho)=\mathcal{U}$, $\ker(\rho)=\mathcal{W}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $P^2=P$. For every $\mathbf{v}\in\mathcal{V}$, $\mathbf{v}=P\mathbf{v}+(\mathbf{v}-P\mathbf{v})$, with $P\mathbf{v}\in\operatorname{range}(P)$ and $\mathbf{v}-P\mathbf{v}\in\ker(P)$. If $\mathbf{x}\in\operatorname{range}(P)\cap\ker(P)$, then $\mathbf{x}=P\mathbf{y}$ and $\mathbf{0}=P\mathbf{x}=P^2\mathbf{y}=\mathbf{x}$. Thus the sum is direct.\\
Also $P$ fixes $\operatorname{range}(P)$ and kills $\ker(P)$, so $P=\rho_{\operatorname{range}(P),\ker(P)}$.
\item[$(\Leftarrow)$] Assume the right-hand side. For $\mathbf{v}=\mathbf{x}+\mathbf{y}$ with $\mathbf{x}\in\operatorname{range}(P)$ and $\mathbf{y}\in\ker(P)$, $P^2\mathbf{v}=P\mathbf{x}=\mathbf{x}=P\mathbf{v}$.\\
For $\rho_{\mathcal{U},\mathcal{W}}$, the formula $\rho_{\mathcal{U},\mathcal{W}}(\mathbf{u}+\mathbf{w})=\mathbf{u}$ gives range $\mathcal{U}$ and kernel $\mathcal{W}$; the implication gives $\rho^2=\rho$.
\end{enumerate}
\hlt{Structural Commentary}: Idempotents keep one summand and kill its complement.\\
\hlt{Key Insight}: Direct-sum coordinates turn a projection into $I\oplus0$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:complementary-algebraic-projections} \hlt{Complementary Algebraic Projections}\\
If $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, then $\rho_{\mathcal{U},\mathcal{W}}+\rho_{\mathcal{W},\mathcal{U}}=I_{\mathcal{V}}$, and $\rho_{\mathcal{U},\mathcal{W}}\rho_{\mathcal{W},\mathcal{U}} =\rho_{\mathcal{W},\mathcal{U}}\rho_{\mathcal{U},\mathcal{W}}=0$.
\begin{proof}
For $\mathbf{v}=\mathbf{u}+\mathbf{w}$, $\mathbf{u}\in\mathcal{U}$, $\mathbf{w}\in\mathcal{W}$, $(\rho_{\mathcal{U},\mathcal{W}}+\rho_{\mathcal{W},\mathcal{U}})\mathbf{v} =\mathbf{u}+\mathbf{w} =\mathbf{v}$,\\
and $\rho_{\mathcal{U},\mathcal{W}}\rho_{\mathcal{W},\mathcal{U}}\mathbf{v} =\rho_{\mathcal{U},\mathcal{W}}\mathbf{w} =\mathbf{0}$; $\rho_{\mathcal{W},\mathcal{U}}\rho_{\mathcal{U},\mathcal{W}}\mathbf{v} =\rho_{\mathcal{W},\mathcal{U}}\mathbf{u} =\mathbf{0}$. Thus the stated operator identities hold.\\
\hlt{Structural Commentary}: A direct-sum decomposition gives two complementary idempotents.\\
\hlt{Key Insight}: Projection onto one summand kills the other summand.
\end{proof}
\end{proposition}

\begin{definition} \label{def:algebraically-orthogonal-projections-and-resolutions} \hlt{Orthogonal Projections in the Algebraic Sense}\\
Two projections $\rho,\sigma\in\LL(\mathcal{V})$ are \hlt{algebraically orthogonal}, written $\rho\perp\sigma$, if $\rho\sigma=\sigma\rho=0$.\\
A \hlt{resolution of the identity} on $\mathcal{V}$ is a finite family of projections $(\rho_i)_{i=1}^{k}$ with $\rho_i\perp\rho_j$ for $i\neq j$ and $\sum\limits_{i=1}^{k}\rho_i=I_{\mathcal{V}}$.
\end{definition}

\begin{theorem} \label{thm:resolution-of-identity-gives-direct-sum} \hlt{Resolution Ranges Form a Direct Sum}\\
If $(\rho_i)_{i=1}^{k}$ is a resolution of $I_{\mathcal{V}}$ and $R_i=\operatorname{range}(\rho_i)$, then $\mathcal{V}=\bigoplus\limits_{i=1}^{k}R_i$, and $\ker(\rho_i)=\bigoplus\limits_{j\neq i}R_j$.
\begin{proof}
Since $\sum\limits_{i=1}^{k}\rho_i=I_{\mathcal{V}}$, $\mathbf{v}=\sum\limits_{i=1}^{k}\rho_i\mathbf{v}\in\sum\limits_{i=1}^{k}R_i$.\\
If $\sum\limits_{i=1}^{k}\mathbf{x}_i=\mathbf{0}$ with $\mathbf{x}_i\in R_i$, then for each $1\leq m\leq k$, $\mathbf{0}=\rho_m\sum\limits_{i=1}^{k}\mathbf{x}_i=\mathbf{x}_m$, as $\rho_m$ fixes $R_m$ and kills $R_i$ for $i\neq m$.\\
Thus $\mathcal{V}=\bigoplus\limits_{i=1}^{k}R_i$. Also, for $\mathbf{v}=\sum\limits_{j=1}^{k}\mathbf{x}_j$ with $\mathbf{x}_j\in R_j$, $\rho_i\mathbf{v}=\mathbf{x}_i$, so $\ker(\rho_i)=\bigoplus\limits_{j\neq i}R_j$.\\
\hlt{Structural Commentary}: A resolution decomposes the identity into independent idempotent pieces.\\
\hlt{Key Insight}: Pairwise zero products are the operator form of directness.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:direct-sum-gives-resolution-of-identity} \hlt{Direct Sums Give Resolutions}\\
If $\mathcal{V}=\bigoplus_{i=1}^{k}\mathcal{U}_i$ and $\rho_i=\rho_{\mathcal{U}_i,\bigoplus_{j\neq i}\mathcal{U}_j}$, then $(\rho_i)_{i=1}^{k}$ is a resolution of $I_{\mathcal{V}}$.
\begin{proof}
For $\mathbf{v}=\sum_{i=1}^{k}\mathbf{u}_i$, $\mathbf{u}_i\in\mathcal{U}_i$, $\sum\limits_{i=1}^{k}\rho_i\mathbf{v} =\sum\limits_{i=1}^{k}\mathbf{u}_i =\mathbf{v}$. Thus $\sum\limits_{i=1}^{k}\rho_i=I_{\mathcal{V}}$.\\
If $i\neq j$, then $\rho_i\rho_j\mathbf{v}=\rho_i\mathbf{u}_j=\mathbf{0}$ and $\rho_j\rho_i\mathbf{v}=\rho_j\mathbf{u}_i=\mathbf{0}$, so $\rho_i\perp\rho_j$.\\
\hlt{Structural Commentary}: Direct-sum coordinates produce mutually annihilating coordinate projections.\\
\hlt{Key Insight}: Coordinate projections add to the identity and kill all other summands.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:algebra-of-projections} \hlt{Algebra of Projections}\\
Let $\rho,\sigma\in\LL(\mathcal{V})$ be projections and $\operatorname{char}(\mathbb{F})\neq2$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\rho+\sigma$ is a projection $\Longleftrightarrow \rho\perp\sigma$.\\
Then $\operatorname{range}(\rho+\sigma)=\operatorname{range}(\rho)\oplus\operatorname{range}(\sigma)$; $\ker(\rho+\sigma)=\ker(\rho)\cap\ker(\sigma)$
\item $\rho-\sigma$ is a projection $\Longleftrightarrow \rho\sigma=\sigma\rho=\sigma$.\\
Then $\operatorname{range}(\rho-\sigma)=\operatorname{range}(\rho)\cap\ker(\sigma)$; $\ker(\rho-\sigma)=\ker(\rho)\oplus\operatorname{range}(\sigma)$.
\item If $\rho\sigma=\sigma\rho$, then $\rho\sigma$ is a projection and $\operatorname{range}(\rho\sigma)=\operatorname{range}(\rho)\cap\operatorname{range}(\sigma)$; $\ker(\rho\sigma)=\ker(\rho)+\ker(\sigma)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Put $A=\rho\sigma$, $B=\sigma\rho$, and $C=\rho\sigma\rho$. Since $(\rho+\sigma)^2=\rho+\sigma \Longleftrightarrow A+B=0$,\\
$A+B=0$ gives $A+C=\rho(A+B)=0$ and $C+B=(A+B)\rho=0$, hence $B=A$ and $2A=0$.\\
Thus $A=B=0$. The converse is immediate. Assume $\rho\perp\sigma$. Then $\operatorname{range}(\rho+\sigma)\subseteq\operatorname{range}(\rho)+\operatorname{range}(\sigma)$, while $(\rho+\sigma)(\mathbf{x}+\mathbf{y})=\mathbf{x}+\mathbf{y}$ for $\mathbf{x}\in\operatorname{range}(\rho)$, $\mathbf{y}\in\operatorname{range}(\sigma)$. Also $\operatorname{range}(\rho)\cap\operatorname{range}(\sigma)=\{\mathbf{0}\}$, since $\mathbf{z}\in\operatorname{range}(\rho)\cap\operatorname{range}(\sigma)$ and $\mathbf{z}=\sigma\mathbf{w}$ imply $\mathbf{z}=\rho\mathbf{z}=\rho\sigma\mathbf{w}=\mathbf{0}$. Finally, $(\rho+\sigma)\mathbf{v}=0 \Longleftrightarrow \rho\mathbf{v}=0=\sigma\mathbf{v}$.
\item Since $P^2=P\Longleftrightarrow (I_{\mathcal{V}}-P)^2=I_{\mathcal{V}}-P$, $\rho-\sigma$ is a projection $\Longleftrightarrow$ $(I_{\mathcal{V}}-\rho)+\sigma$ is a projection.\
By (i), this is equivalent to $(I_{\mathcal{V}}-\rho)\sigma=\sigma(I_{\mathcal{V}}-\rho)=0$, i.e. $\rho\sigma=\sigma\rho=\sigma$.\\
Assume $\rho\sigma=\sigma\rho=\sigma$ and set $Q=\rho-\sigma$. Then $Q\mathbf{v}\in\operatorname{range}(\rho)\cap\ker(\sigma)$; $Q\mathbf{x}=\mathbf{x}$, $\mathbf{x}\in\operatorname{range}(\rho)\cap\ker(\sigma)$, so $\operatorname{range}(Q)=\operatorname{range}(\rho)\cap\ker(\sigma)$. Also $Q$ kills both $\ker(\rho)$ and $\operatorname{range}(\sigma)$, and $\ker(\rho)\cap\operatorname{range}(\sigma)=\{\mathbf{0}\}$.\\
If $\mathbf{v}\in\ker(Q)$, then $\rho\mathbf{v}=\sigma\mathbf{v}$ and $\mathbf{v}=(\mathbf{v}-\rho\mathbf{v})+\sigma\mathbf{v} \in\ker(\rho)\oplus\operatorname{range}(\sigma)$. Hence $\ker(Q)=\ker(\rho)\oplus\operatorname{range}(\sigma)$.
\item If $\rho\sigma=\sigma\rho$, then $(\rho\sigma)^2=\rho^2\sigma^2=\rho\sigma$. Moreover $\mathbf{x}\in\operatorname{range}(\rho\sigma)$ $\Longleftrightarrow$ $\rho\mathbf{x}=\mathbf{x}=\sigma\mathbf{x}$,\\
so $\operatorname{range}(\rho\sigma)=\operatorname{range}(\rho)\cap\operatorname{range}(\sigma)$. Finally, $\ker(\rho)+\ker(\sigma)\subseteq\ker(\rho\sigma)$, and if $\rho\sigma\mathbf{v}=0$,\\
then $\mathbf{v}=(\mathbf{v}-\sigma\mathbf{v})+\sigma\mathbf{v}$; $\mathbf{v}-\sigma\mathbf{v}\in\ker(\sigma)$, $\sigma\mathbf{v}\in\ker(\rho)$. Thus $\ker(\rho\sigma)=\ker(\rho)+\ker(\sigma)$.
\end{enumerate}
\hlt{Structural Commentary}: Operator identities encode range-kernel geometry.\\
\hlt{Key Insight}: Idempotence is a rigid equation; expanding it forces the cross terms to encode the geometry.
\end{proof}
\end{theorem}

\begin{definition} \label{def:affine-subset-parallel-subspace} \hlt{Affine Subset and Parallel Subspaces}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$. For $\mathbf{v}\in\mathcal{V}$, the subset $\mathbf{v}+\mathcal{U}\defeq\{\mathbf{v}+\mathbf{u}:\mathbf{u}\in\mathcal{U}\}$ is an \hlt{affine subset} of $\mathcal{V}$ parallel to $\mathcal{U}$.
\end{definition}

\begin{proposition} \label{prop:parallel-affine-subsets-equal-or-disjoint} \hlt{Parallel Affine Subsets Are Equal or Disjoint}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$ and let $\mathbf{v},\mathbf{w}\in\mathcal{V}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathbf{v}-\mathbf{w}\in\mathcal{U}$;
\item $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$;
\item $(\mathbf{v}+\mathcal{U})\cap(\mathbf{w}+\mathcal{U})\neq\emptyset$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman)*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $\mathbf{v}-\mathbf{w}\in\mathcal{U}$ and $\mathbf{u}\in\mathcal{U}$, then $\mathbf{v}+\mathbf{u}=\mathbf{w}+((\mathbf{v}-\mathbf{w})+\mathbf{u})\in\mathbf{w}+\mathcal{U}$. Thus $\mathbf{v}+\mathcal{U}\subseteq\mathbf{w}+\mathcal{U}$, and the reverse inclusion is similar.
\item[$(ii)$]$\Rightarrow (iii)$ Equal affine subsets have the same elements.
\item[$(iii)$]$\Rightarrow (i)$ If $\mathbf{x}\in(\mathbf{v}+\mathcal{U})\cap(\mathbf{w}+\mathcal{U})$, then $\mathbf{x}=\mathbf{v}+\mathbf{u}_1=\mathbf{w}+\mathbf{u}_2$ for some $\mathbf{u}_1,\mathbf{u}_2\in\mathcal{U}$. Hence $\mathbf{v}-\mathbf{w}=\mathbf{u}_2-\mathbf{u}_1\in\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: Parallel affine subsets either coincide or do not meet.\\
\hlt{Key Insight}: Intersection is controlled by the difference of base points.
\end{proof}
\end{proposition}

\begin{definition} \label{def:quotient-space} \hlt{Quotient Space}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$.\\
The \hlt{quotient space} $\mathcal{V}/\mathcal{U}$ is the set of all affine subsets of $\mathcal{V}$ parallel to $\mathcal{U}$: $\mathcal{V}/\mathcal{U}\defeq\{\mathbf{v}+\mathcal{U}:\mathbf{v}\in\mathcal{V}\}$.
\end{definition}

\begin{perspective} \label{per:quotient-cosets-as-equivalence-classes} \hlt{Cosets as Equivalence Classes}\\
The quotient relation is $\mathbf{v}\sim\mathbf{w}$ when $\mathbf{v}-\mathbf{w}\in\mathcal{U}$. Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint} shows that the equivalence class of $\mathbf{v}$ is exactly the coset $\mathbf{v}+\mathcal{U}$. Thus the quotient space remembers vectors only up to directions in $\mathcal{U}$.
\end{perspective}

\begin{proposition} \label{prop:quotient-relation-congruence} \hlt{Quotient Relation Is a Linear Congruence}\\
Let $\mathcal{U}\leq\mathcal{V}$ and define $\mathbf{v}\equiv_{\mathcal{U}}\mathbf{w}\Longleftrightarrow \mathbf{v}-\mathbf{w}\in\mathcal{U}$.\\
Then $\equiv_{\mathcal{U}}$ is an equivalence relation and, for $a,b\in\mathbb{F}$, $\mathbf{v}_1\equiv_{\mathcal{U}}\mathbf{v}_2$, $\mathbf{w}_1\equiv_{\mathcal{U}}\mathbf{w}_2$ $\Longrightarrow$ $a\mathbf{v}_1+b\mathbf{w}_1\equiv_{\mathcal{U}} a\mathbf{v}_2+b\mathbf{w}_2$.
\begin{proof}
Since $\mathcal{U}\leq\mathcal{V}$, $\mathbf{0}\in\mathcal{U}$, $-\mathcal{U}=\mathcal{U}$, and $\mathcal{U}+\mathcal{U}\subseteq\mathcal{U}$; hence $\equiv_{\mathcal{U}}$ is reflexive, symmetric, and transitive.\\
If $\mathbf{v}_1\equiv_{\mathcal{U}}\mathbf{v}_2$ and $\mathbf{w}_1\equiv_{\mathcal{U}}\mathbf{w}_2$, then $(a\mathbf{v}_1+b\mathbf{w}_1)-(a\mathbf{v}_2+b\mathbf{w}_2) =a(\mathbf{v}_1-\mathbf{v}_2)+b(\mathbf{w}_1-\mathbf{w}_2)\in\mathcal{U}$.\\
\hlt{Structural Commentary}: Subspace congruence makes quotient operations well-defined.\\
\hlt{Key Insight}: Subspace closure is exactly the well-definedness mechanism for quotient operations.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:quotient-space-structure} \hlt{Structure of the Quotient}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$. Addition and scalar multiplication on $\mathcal{V}/\mathcal{U}$ defined by $(\mathbf{v} + \mathcal{U}) + (\mathbf{w} + \mathcal{U}) \defeq (\mathbf{v} + \mathbf{w}) + \mathcal{U}$ and $a(\mathbf{v} + \mathcal{U}) \defeq a\mathbf{v} + \mathcal{U}$ are well-defined, and with these operations $\mathcal{V}/\mathcal{U}$ is a vector space.
\begin{proof}
If $\mathbf{v} + \mathcal{U} = \mathbf{v}' + \mathcal{U}$ and $\mathbf{w} + \mathcal{U} = \mathbf{w}' + \mathcal{U}$, then $\mathbf{v}-\mathbf{v}' \in \mathcal{U}$ and $\mathbf{w}-\mathbf{w}' \in \mathcal{U}$.\\
Since $\mathcal{U}$ is a subspace, $(\mathbf{v}+\mathbf{w}) - (\mathbf{v}'+\mathbf{w}') \in \mathcal{U}$, and $a\mathbf{v} - a\mathbf{v}' \in \mathcal{U}$ so the operations do not depend on the chosen representatives. The vector space axioms are inherited from $\mathcal{V}$ under these operations.\\
\hlt{Structural Commentary}: Quotients preserve operations modulo $\mathcal{U}$.\\
\hlt{Key Insight}: Cosets identify vectors differing by elements of $\mathcal{U}$.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Quotient Map}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$. The \hlt{quotient map} is the map $\pi : \mathcal{V} \to \mathcal{V}/\mathcal{U}$ defined by $\pi(\mathbf{v})=\mathbf{v}+\mathcal{U}$.
\end{definition}

\begin{proposition} \label{prop:quotient-map-properties} \hlt{Properties of the Quotient Map}\\
The quotient map $\pi$ is linear, surjective, and $\ker(\pi) = \mathcal{U}$.
\begin{proof}
For $\mathbf{v},\mathbf{w}\in\mathcal{V}$ and $a\in\mathbb{F}$, $\pi(a\mathbf{v}+\mathbf{w})=(a\mathbf{v}+\mathbf{w})+\mathcal{U} =a(\mathbf{v}+\mathcal{U})+(\mathbf{w}+\mathcal{U}) =a\pi(\mathbf{v})+\pi(\mathbf{w})$ so $\pi$ is linear. It is surjective because every coset in $\mathcal{V}/\mathcal{U}$ has the form $\mathbf{v}+\mathcal{U}=\pi(\mathbf{v})$. Finally, $\ker(\pi)=\{\mathbf{v}\in\mathcal{V}:\mathbf{v}+\mathcal{U}=\mathcal{U}\}=\mathcal{U}$.\\
\hlt{Structural Commentary}: The quotient map collapses $\mathcal{U}$ to zero and records cosets.\\
\hlt{Key Insight}: Quotient maps realise subspaces as kernels.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:quotient-factorisation-through-projection} \hlt{Universal Property of the Quotient Map}\\
Let $\mathcal{U}\leq\mathcal{V}$, $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$, and $T\in\LL(\mathcal{V},\mathcal{W})$ with $\mathcal{U}\leq\ker(T)$.\\
Then $\exists!\,\overline{T}\in\LL(\mathcal{V}/\mathcal{U},\mathcal{W})
$ such that $\overline{T}\pi=T$, $\overline{T}(\mathbf{v}+\mathcal{U})=T\mathbf{v}$, and $\ker(\overline{T})=\ker(T)/\mathcal{U}$, $\operatorname{range}(\overline{T})=\operatorname{range}(T)$.
\begin{proof}
Define $\overline{T}(\mathbf{v}+\mathcal{U})=T\mathbf{v}$. If $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$, then $\mathbf{v}-\mathbf{w}\in\mathcal{U}\leq\ker(T)$, so $T\mathbf{v}=T\mathbf{w}$; hence $\overline{T}$ is well-defined.\\
Also $\overline{T}\big(a(\mathbf{v}+\mathcal{U})+b(\mathbf{w}+\mathcal{U})\big) =T(a\mathbf{v}+b\mathbf{w}) =aT\mathbf{v}+bT\mathbf{w}$.\\
By definition, $\overline{T}\pi=T$. If $S\pi=T$, then $S(\mathbf{v}+\mathcal{U})=S\pi(\mathbf{v})=T\mathbf{v}=\overline{T}(\mathbf{v}+\mathcal{U})$, so $S=\overline{T}$.\\
Also $\mathbf{v}+\mathcal{U}\in\ker(\overline{T}) \Longleftrightarrow T\mathbf{v}=\mathbf{0} \Longleftrightarrow \mathbf{v}\in\ker(T)$.\\
So $\ker(\overline{T})=\ker(T)/\mathcal{U}$, and $\operatorname{range}(\overline{T})=\{\overline{T}(\mathbf{v}+\mathcal{U}):\mathbf{v}\in\mathcal{V}\}=\operatorname{range}(T)$.\\
\hlt{Structural Commentary}: Maps killing $\mathcal{U}$ factor through $\mathcal{V}/\mathcal{U}$.\\
\hlt{Key Insight}: Quotients are universal domains for maps with prescribed kernel ambiguity.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:subspace-correspondence-quotient} \hlt{Subspace Correspondence for Quotients}\\
Let $\pi:\mathcal{V}\to\mathcal{V}/\mathcal{U}$ be the quotient map. The assignments $\mathcal{T}\longmapsto \mathcal{T}/\mathcal{U}$ and $\mathcal{Y}\longmapsto \pi^{-1}(\mathcal{Y})$ are inverse inclusion-preserving bijections between $\{\mathcal{T}$ subspace of $\mathcal{V}:\mathcal{U}\subseteq\mathcal{T}\}$ and $\{\mathcal{Y}$ subspace of $\mathcal{V}/\mathcal{U}\}$.
\begin{proof}
If $\mathcal{U}\subseteq\mathcal{T}\subseteq\mathcal{V}$, then $\mathcal{T}/\mathcal{U}$ is a subspace of $\mathcal{V}/\mathcal{U}$.\\
If $\mathcal{Y}\subseteq\mathcal{V}/\mathcal{U}$ is a subspace, then $\pi^{-1}(\mathcal{Y})$ is a subspace of $\mathcal{V}$ and contains $\ker(\pi)=\mathcal{U}$.\\
For $\mathcal{U}\subseteq\mathcal{T}\subseteq\mathcal{V}$, $\pi^{-1}(\mathcal{T}/\mathcal{U})=\mathcal{T}$, since $\mathbf{v}+\mathcal{U}=\mathbf{t}+\mathcal{U}$ with $\mathbf{t}\in\mathcal{T}$ implies $\mathbf{v}-\mathbf{t}\in\mathcal{U}\subseteq\mathcal{T}$.\\
For $\mathcal{Y}\subseteq\mathcal{V}/\mathcal{U}$, surjectivity gives $\pi\left(\pi^{-1}(\mathcal{Y})\right)=\mathcal{Y}$, that is, $\pi^{-1}(\mathcal{Y})/\mathcal{U}=\mathcal{Y}$\\
Both assignments preserve inclusion by monotonicity of images and preimages.\\
\hlt{Structural Commentary}: Subspaces above $\mathcal{U}$ are exactly the subspaces visible after collapsing $\mathcal{U}$.\\
\hlt{Key Insight}: The quotient map translates containment above $\mathcal{U}$ into containment inside $\mathcal{V}/\mathcal{U}$.
\end{proof}
\end{theorem}

\begin{definition} \hlt{Induced Map on a Quotient}\\
For $T\in\LL(\mathcal{V},\mathcal{W})$, define $\widetilde{T}:\mathcal{V}/\ker(T)\to\mathcal{W}$ by $\widetilde{T}(\mathbf{v}+\ker(T))=T\mathbf{v}$.
\end{definition}

\begin{theorem} \label{thm:quotient-by-kernel-is-range} \hlt{Quotient by the Kernel}\\
Let $T\in\LL(\mathcal{V},\mathcal{W})$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\widetilde{T}$ is a well-defined linear map from $\mathcal{V}/\ker(T)$ to $\mathcal{W}$;
\item $\widetilde{T}$ is injective;
\item $\operatorname{range}(\widetilde{T})=\operatorname{range}(T)$;
\item $\mathcal{V}/\ker(T)\cong\operatorname{range}(T)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathbf{v}+\ker(T)=\mathbf{v}'+\ker(T)$, then $\mathbf{v}-\mathbf{v}'\in\ker(T)$, so $T\mathbf{v}=T\mathbf{v}'$. Thus $\widetilde{T}$ is well-defined.\\
For cosets and $a\in\mathbb{F}$, $\widetilde{T}\big(a(\mathbf{v}+\ker(T))+(\mathbf{v}'+\ker(T))\big) =T(a\mathbf{v}+\mathbf{v}') =aT\mathbf{v}+T\mathbf{v}'$, so $\widetilde{T}$ is linear.
\item If $\widetilde{T}(\mathbf{v}+\ker(T))=\mathbf{0}$, then $T\mathbf{v}=\mathbf{0}$, so $\mathbf{v}\in\ker(T)$ and $\mathbf{v}+\ker(T)=\ker(T)$, the zero coset.\\
Hence $\ker(\widetilde{T})=\{\ker(T)\}$ and $\widetilde{T}$ is injective.
\item For every coset $\mathbf{v}+\ker(T)$, the value $\widetilde{T}(\mathbf{v}+\ker(T))=T\mathbf{v}$ lies in $\operatorname{range}(T)$.\\
Conversely, every vector in $\operatorname{range}(T)$ has the form $T\mathbf{v}=\widetilde{T}(\mathbf{v}+\ker(T))$. Hence the ranges are equal.
\item Let $\widehat{T}:\mathcal{V}/\ker(T)\to\operatorname{range}(T)$ be the same formula as $\widetilde{T}$ with codomain restricted to $\operatorname{range}(T)$.\\
(i)--(iii) show that $\widehat{T}$ is linear, injective, and surjective, so Proposition \ref{prop:linear-map-invertible-iff-bijective} gives $\mathcal{V}/\ker(T)\cong\operatorname{range}(T)$.
\end{enumerate}
\hlt{Structural Commentary}: Quotienting by the kernel removes exactly the ambiguity of $T$.\\
\hlt{Key Insight}: Every linear map factors as a quotient followed by an isomorphism onto its range.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:quotient-by-complement-is-complement} \hlt{Quotient by a Complemented Subspace}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$ with $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$. Then $J:\mathcal{W}\to\mathcal{V}/\mathcal{U}$, $J(\mathbf{w})=\mathbf{w}+\mathcal{U}$ is an isomorphism.\\
Hence every complement of $\mathcal{U}$ is isomorphic to $\mathcal{V}/\mathcal{U}$.
\begin{proof}
The map is linear by the quotient operations from Proposition \ref{prop:quotient-space-structure}.\\
If $J(\mathbf{w})=\mathcal{U}$, then $\mathbf{w}\in\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$, so $J$ is injective.\\
If $\mathbf{v}+\mathcal{U}\in\mathcal{V}/\mathcal{U}$, write $\mathbf{v}=\mathbf{u}+\mathbf{w}$ with $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{W}$. Then $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}=J(\mathbf{w})$, so $J$ is surjective.\\
\hlt{Structural Commentary}: A complement gives canonical representatives for quotient cosets.\\
\hlt{Key Insight}: Collapsing $\mathcal{U}$ leaves exactly the complementary directions visible.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:quotient-of-direct-sum-by-direct-sum} \hlt{Quotient of a Direct Sum}\\
If $\mathcal{V}=\mathcal{V}_1\oplus\mathcal{V}_2$ and $\mathcal{U}_j\leq\mathcal{V}_j$ for $j=1,2$, then $\frac{\mathcal{V}}{\mathcal{U}_1\oplus\mathcal{U}_2} \cong \frac{\mathcal{V}_1}{\mathcal{U}_1}\oplus \frac{\mathcal{V}_2}{\mathcal{U}_2}$.
\begin{proof}
Define, using the unique decomposition $\mathbf{v}=\mathbf{v}_1+\mathbf{v}_2$,
$\Phi:\mathcal{V}\to (\mathcal{V}_1/\mathcal{U}_1)\oplus(\mathcal{V}_2/\mathcal{U}_2)$ and $\Phi(\mathbf{v}_1+\mathbf{v}_2) =(\mathbf{v}_1+\mathcal{U}_1,\mathbf{v}_2+\mathcal{U}_2)$.\\
For $\mathbf{x}=\mathbf{x}_1+\mathbf{x}_2$, $\mathbf{y}=\mathbf{y}_1+\mathbf{y}_2$, and $a,b\in\mathbb{F}$, $\Phi(a\mathbf{x}+b\mathbf{y}) =a\Phi(\mathbf{x})+b\Phi(\mathbf{y})$, so $\Phi$ is linear.\\
It is surjective since $(\mathbf{v}_1+\mathcal{U}_1,\mathbf{v}_2+\mathcal{U}_2)=\Phi(\mathbf{v}_1+\mathbf{v}_2)$.\\
Finally, $\ker(\Phi)=\{\mathbf{v}_1+\mathbf{v}_2:\mathbf{v}_1\in\mathcal{U}_1,\ \mathbf{v}_2\in\mathcal{U}_2\} =\mathcal{U}_1\oplus\mathcal{U}_2$.\\
The result follows from Theorem \ref{thm:quotient-by-kernel-is-range}.\\
\hlt{Structural Commentary}: Quotienting respects direct-sum coordinates.\\
\hlt{Key Insight}: Direct-sum coordinates make quotienting componentwise.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:quotients-and-complements-are-not-functorial-warnings} \hlt{Quotient and Complement Warnings}\\
If $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, then $\mathcal{W}\cong\mathcal{V}/\mathcal{U}$, but complements are not functorial in $\mathcal{U}$.\\
For $\mathcal{V}$ countably infinite-dimensional with basis $(\mathbf{e}_n)_{n\geq1}$, put $\mathcal{S}=\operatorname{span}(\mathbf{e}_n:n\geq2)$.\\
Then $\mathcal{S}\cong\mathcal{V}$ and $\mathcal{V}=\mathcal{V}\oplus\{\mathbf{0}\} =\mathcal{S}\oplus\mathbb{F}\mathbf{e}_1$; $\mathcal{V}/\mathcal{V}=\{\mathbf{0}\}$, $\mathcal{V}/\mathcal{S}\cong\mathbb{F}\mathbf{e}_1$.
\end{perspective}


\begin{theorem} \label{thm:dimension-of-quotient} \hlt{Dimension of a Quotient}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U}$ is a subspace, then $\dim(\mathcal{V}/\mathcal{U}) = \dim(\mathcal{V}) - \dim(\mathcal{U})$.
\begin{proof}
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ of $\mathcal{U}$ and extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_k)$ of $\mathcal{V}$.\\
The cosets $\mathbf{w}_1+\mathcal{U},\ldots,\mathbf{w}_k+\mathcal{U}$ span $\mathcal{V}/\mathcal{U}$ because the $\mathbf{u}_j$-terms vanish in the quotient.\\
If $a_1(\mathbf{w}_1+\mathcal{U})+\cdots+a_k(\mathbf{w}_k+\mathcal{U})=\mathcal{U}$, then $a_1\mathbf{w}_1+\cdots+a_k\mathbf{w}_k\in\mathcal{U}$, so a linear relation in the extended basis forces $a_1=\cdots=a_k=0$. Hence these cosets form a basis of $\mathcal{V}/\mathcal{U}$, and its length is $k=\dim(\mathcal{V})-\dim(\mathcal{U})$.\\
\hlt{Structural Commentary}: Quotiening removes the directions in $\mathcal{U}$ and keeps the added basis directions.\\
\hlt{Key Insight}: Passing to $\mathcal{V}/\mathcal{U}$ subtracts the collapsed dimension.
\end{proof}
\end{theorem}

\begin{definition} \label{def:codimension-subspace} \hlt{Codimension}\\
If $\mathcal{U}$ is a subspace of $\mathcal{V}$ and $\dim(\mathcal{V}/\mathcal{U})<\infty$, define $\operatorname{codim}_{\mathcal{V}}(\mathcal{U})\defeq\dim(\mathcal{V}/\mathcal{U})$.\\
Thus, if $\mathcal{V}$ is finite-dimensional, Theorem \ref{thm:dimension-of-quotient} gives $\operatorname{codim}_{\mathcal{V}}(\mathcal{U})=\dim(\mathcal{V})-\dim(\mathcal{U})$.\\
If $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$, Proposition \ref{prop:quotient-by-complement-is-complement} gives $\operatorname{codim}_{\mathcal{V}}(\mathcal{U})=\dim(\mathcal{W})$.
\end{definition}

\begin{perspective} \label{per:quotients-split-over-fields} \hlt{Quotients Split Over Fields}\\
In finite-dimensional vector spaces, choosing a complement $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$ identifies $\mathcal{V}/\mathcal{U}$ with $\mathcal{W}$. The quotient map collapses $\mathcal{U}$ and keeps the complementary directions.
\end{perspective}

This block develops two complementary ways to organise a vector space: build from smaller pieces by spans and direct sums, or collapse a chosen subspace by passing to a quotient. Bases and dimension measure exactly what is added, split off, or made invisible.
