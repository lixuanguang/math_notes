\subsection{Direct Sums, Quotients, and Decomposition}

% **Role:** algebraic language of linear structure.
%
% Sums, direct sums, complements, quotient spaces, induced maps, dimension formulas.

Direct sums split spaces into complementary pieces; quotients collapse selected directions. Dimension formulas keep both constructions controlled.

\begin{definition} \hlt{Sums of Subspaces}\\
Let $\mathcal{U}_1,\ldots,\mathcal{U}_m\leq\mathcal{V}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{sum} of $\mathcal{U}_1,\ldots,\mathcal{U}_m$ is
$\mathcal{U}_1+\cdots+\mathcal{U}_m\defeq\{\mathbf{u}_1+\cdots+\mathbf{u}_m:\mathbf{u}_j\in\mathcal{U}_j\}$.
\item For subspaces $\mathcal{U},\mathcal{W}\leq\mathcal{V}$,
$\mathcal{U}+\mathcal{W}=\operatorname{span}(\mathcal{U}\cup\mathcal{W})=\{\mathbf{u}+\mathbf{w}:\mathbf{u}\in\mathcal{U},\ \mathbf{w}\in\mathcal{W}\}$.
\item The sum is \hlt{direct} if every vector in it has a unique decomposition as $\mathbf{u}_1+\cdots+\mathbf{u}_m$ with $\mathbf{u}_j\in\mathcal{U}_j$; in this case it is denoted $\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_m$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:sum-subspaces-smallest-containing-subspace} \hlt{Sum of Subspaces Is the Smallest Containing Subspace}\\
Suppose $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is the smallest subspace of $\mathcal{V}$ containing $\mathcal{U}_1,\ldots,\mathcal{U}_m$.
\begin{proof}
Since $\mathbf{0}\in\mathcal{U}_j$ for each $j$, we have $\mathbf{0}\in\mathcal{U}_1+\cdots+\mathcal{U}_m$. If $\mathbf{x}=\mathbf{u}_1+\cdots+\mathbf{u}_m$ and $\mathbf{y}=\mathbf{v}_1+\cdots+\mathbf{v}_m$ lie in $\mathcal{U}_1+\cdots+\mathcal{U}_m$, then for any $c \in \mathbb{F}$, $c\mathbf{x}+\mathbf{y} = (c\mathbf{u}_1+\mathbf{v}_1)+\cdots+(c\mathbf{u}_m+\mathbf{v}_m) \in \mathcal{U}_1+\cdots+\mathcal{U}_m$, because $c\mathbf{u}_j+\mathbf{v}_j\in\mathcal{U}_j$ for each $j$. By Theorem \ref{thm:subspacetest}, $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a subspace of $\mathcal{V}$.\\
Each $\mathcal{U}_j$ is contained in $\mathcal{U}_1+\cdots+\mathcal{U}_m$ by taking all other summands to be $\mathbf{0}$. If $\mathcal{W}$ is any subspace of $\mathcal{V}$ containing $\mathcal{U}_1,\ldots,\mathcal{U}_m$, then closure under addition gives $\mathbf{u}_1+\cdots+\mathbf{u}_m\in\mathcal{W}$ whenever $\mathbf{u}_j\in\mathcal{U}_j$. Hence $\mathcal{U}_1+\cdots+\mathcal{U}_m\subseteq\mathcal{W}$, so the sum is the smallest such subspace.\\
\hlt{Structural Commentary}: Sums of subspaces are obtained by closing several subspaces under finite addition. Minimality follows because every containing subspace is already closed under those sums.\\
\hlt{Key Insight}: The sum adds no extra structure beyond the linear combinations forced by the given subspaces.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:sum-not-union} \hlt{Sum Is Not Union}\\
The union $\mathcal{U}\cup\mathcal{W}$ keeps vectors from either subspace; the sum $\mathcal{U}+\mathcal{W}$ adds vectors from both. Unless one subspace contains the other, $\mathcal{U}\cup\mathcal{W}$ is not closed under addition.
\end{perspective}

\begin{theorem} \label{thm:finite-union-proper-subspaces-infinite-field} \hlt{Finite Unions of Proper Subspaces}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be over an infinite field $\mathbb{F}$. If $\mathcal{U}_1,\ldots,\mathcal{U}_n\subsetneq\mathcal{V}$ are subspaces, then $\mathcal{V}\neq\bigcup\limits_{j=1}^{n}\mathcal{U}_j$.
\begin{proof}
Assume for contradiction that $\mathcal{V}=\bigcup\limits_{j=1}^{n}\mathcal{U}_j$. After deleting redundant subspaces and reindexing, assume the cover is minimal, so $\mathcal{U}_i\not\subseteq\bigcup\limits_{j\neq i}\mathcal{U}_j$. If $n=1$, then $\mathcal{V}=\mathcal{U}_1$, contradicting $\mathcal{U}_1\subsetneq\mathcal{V}$; hence $n\geq2$.\\
Choose $\mathbf{w}\in\mathcal{U}_1\setminus\bigcup\limits_{j=2}^{n}\mathcal{U}_j$ and $\mathbf{v}\in\mathcal{V}\setminus\mathcal{U}_1$. Since $\mathbf{0}\in\mathcal{U}_j$ for every $j$, we have $\mathbf{w}\neq\mathbf{0}$.\\
Let $A=\{\mathbf{v}+r\mathbf{w}:r\in\mathbb{F}\}$. Then $\abs{A}=\abs{\mathbb{F}}=\infty$, $A\cap\mathcal{U}_1=\emptyset$, and for $j\geq2$, $\abs{A\cap\mathcal{U}_j}\leq1$, since two points $\mathbf{v}+r_1\mathbf{w},\mathbf{v}+r_2\mathbf{w}\in\mathcal{U}_j$ with $r_1\neq r_2$ imply $\mathbf{w}\in\mathcal{U}_j$.\\
Hence $\abs{A\cap\bigcup\limits_{j=1}^{n}\mathcal{U}_j}\leq n-1<\infty$, contradicting $A\subseteq\mathcal{V}=\bigcup\limits_{j=1}^{n}\mathcal{U}_j$.\\
\hlt{Structural Commentary}: Over an infinite field, finite proper subspaces cannot cover a well-chosen affine line.\\
\hlt{Key Insight}: A single one-parameter translate has too many points for finitely many proper intersections.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:sum-versus-direct-sum-existence} \hlt{Sums Always Exist; Directness Is Extra}\\
For subspaces $\mathcal{U},\mathcal{W}\subseteq\mathcal{V}$, the sum $\mathcal{U}+\mathcal{W}$ is always a subspace by Proposition \ref{prop:sum-subspaces-smallest-containing-subspace}. Writing $\mathcal{U}\oplus\mathcal{W}$ asserts the extra condition $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$, equivalently uniqueness of the decomposition $\mathbf{u}+\mathbf{w}$.
\end{perspective}

\begin{proposition} \label{prop:uniqueness-direct-sum-two-subspaces} \hlt{Two-Subspace Directness Criterion}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of $\mathcal{V}$. Then $\mathcal{U}+\mathcal{W}$ is direct if and only if $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $\mathcal{U}+\mathcal{W}$ is direct. If $\mathbf{x}\in\mathcal{U}\cap\mathcal{W}$, then $\mathbf{x}=\mathbf{x}+\mathbf{0}=\mathbf{0}+\mathbf{x}$ gives two decompositions of $\mathbf{x}$ with one term in each subspace. Uniqueness gives $\mathbf{x}=\mathbf{0}$, so $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$.
\item[$(\Leftarrow)$] Suppose $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$. Every vector in $\mathcal{U}+\mathcal{W}$ has at least one representation $\mathbf{u}+\mathbf{w}$. If $\mathbf{u}_1+\mathbf{w}_1=\mathbf{u}_2+\mathbf{w}_2$, with $\mathbf{u}_1,\mathbf{u}_2\in\mathcal{U}$ and $\mathbf{w}_1,\mathbf{w}_2\in\mathcal{W}$, then $\mathbf{u}_1-\mathbf{u}_2=\mathbf{w}_2-\mathbf{w}_1\in\mathcal{U}\cap\mathcal{W}$, hence equals $\mathbf{0}$.\\
Thus $\mathbf{u}_1=\mathbf{u}_2$ and $\mathbf{w}_1=\mathbf{w}_2$, so the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: Trivial intersection prevents cancellation between summands.\\
\hlt{Key Insight}: Directness means unique decomposition.
\end{proof}
\end{proposition}

\begin{definition} \label{def:subspace-meet-and-join} \hlt{Subspace Poset, Meet, and Join}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{subspace poset} of $\mathcal{V}$ is
$\operatorname{Sub}(\mathcal{V})\defeq\{\mathcal{U}\subseteq\mathcal{V}:\mathcal{U}\leq\mathcal{V}\}$, ordered by inclusion.
\item For $\{\mathcal{U}_i:i\in I\}\subseteq\operatorname{Sub}(\mathcal{V})$, the \hlt{meet} is
$\bigwedge\limits_{i\in I}\mathcal{U}_i\defeq\bigcap\limits_{i\in I}\mathcal{U}_i$.
\item For $\{\mathcal{U}_i:i\in I\}\subseteq\operatorname{Sub}(\mathcal{V})$, the \hlt{join} is
$\bigvee\limits_{i\in I}\mathcal{U}_i\defeq\sum\limits_{i\in I}\mathcal{U}_i\defeq\operatorname{span}\left(\bigcup\limits_{i\in I}\mathcal{U}_i\right)$.
\item The empty meet is $\mathcal{V}$, and the empty join is $\{\mathbf{0}\}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:subspaces-form-complete-lattice} \hlt{Subspace Meet and Join}\\
For subspaces $\{\mathcal{U}_i:i\in I\}$ of $\mathcal{V}$, ordered by inclusion, $\bigwedge\limits_{i\in I}\mathcal{U}_i=\inf\{\mathcal{U}_i:i\in I\}$; and $\bigvee\limits_{i\in I}\mathcal{U}_i=\sup\{\mathcal{U}_i:i\in I\}$.
\begin{proof}
If $I=\emptyset$, conventions give vacuous bounds $\mathcal{V}$ and $\{\mathbf{0}\}$. Assume $I\neq\emptyset$.\\
The meet $M=\bigcap_{i\in I}\mathcal{U}_i$ is a subspace by Proposition \ref{prop:intersectionsubspaces}; moreover $M\subseteq\mathcal{U}_i$ for all $i$, and any subspace $\mathcal{W}$ with $\mathcal{W}\subseteq\mathcal{U}_i$ for all $i$ satisfies $\mathcal{W}\subseteq M$. The join $J=\operatorname{span}\left(\bigcup_{i\in I}\mathcal{U}_i\right)$ is a subspace by Theorem \ref{thm:span-smallest-containing-subspace}; moreover $\mathcal{U}_i\subseteq J$ for all $i$, and any subspace $\mathcal{W}$ with $\mathcal{U}_i\subseteq\mathcal{W}$ for all $i$ contains $\bigcup_{i\in I}\mathcal{U}_i$, hence contains $J$.\\
\hlt{Structural Commentary}: Meet enforces common constraints; join closes the combined directions under span.\\
\hlt{Key Insight}: Meet is intersection, while join is span of the union.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:subspace-lattice-complete-modular} \hlt{Subspace Lattice}\\
The subspace poset $\operatorname{Sub}(\mathcal{V})$ is a complete modular lattice.\\
Explicitly, for every $\{\mathcal{U}_i:i\in I\}\subseteq\operatorname{Sub}(\mathcal{V})$,
$\bot=\{\mathbf{0}\}$, $\top=\mathcal{V}$, $\bigwedge\limits_{i\in I}\mathcal{U}_i=\bigcap\limits_{i\in I}\mathcal{U}_i$, $\bigvee\limits_{i\in I}\mathcal{U}_i=\sum\limits_{i\in I}\mathcal{U}_i=\operatorname{span}\left(\bigcup\limits_{i\in I}\mathcal{U}_i\right)$.\\
Moreover, if $\mathcal{U}\subseteq\mathcal{S}$, then $\mathcal{S}\cap(\mathcal{T}+\mathcal{U})=(\mathcal{S}\cap\mathcal{T})+\mathcal{U}$.
\begin{proof}
Proposition \ref{prop:subspaces-form-complete-lattice} gives arbitrary meets and joins with the displayed formulas; the empty meet and join give $\top=\mathcal{V}$ and $\bot=\{\mathbf{0}\}$. It remains to prove modularity. Assume $\mathcal{U}\subseteq\mathcal{S}$. If $\mathbf{x}\in(\mathcal{S}\cap\mathcal{T})+\mathcal{U}$, then $\mathbf{x}=\mathbf{y}+\mathbf{u}$ with $\mathbf{y}\in\mathcal{S}\cap\mathcal{T}$ and $\mathbf{u}\in\mathcal{U}$, hence $\mathbf{x}\in\mathcal{S}\cap(\mathcal{T}+\mathcal{U})$. Conversely, if $\mathbf{x}\in\mathcal{S}\cap(\mathcal{T}+\mathcal{U})$, write $\mathbf{x}=\mathbf{t}+\mathbf{u}$ with $\mathbf{t}\in\mathcal{T}$ and $\mathbf{u}\in\mathcal{U}$. Since $\mathbf{x}\in\mathcal{S}$ and $\mathbf{u}\in\mathcal{U}\subseteq\mathcal{S}$, $\mathbf{t}=\mathbf{x}-\mathbf{u}\in\mathcal{S}\cap\mathcal{T}$, so $\mathbf{x}\in(\mathcal{S}\cap\mathcal{T})+\mathcal{U}$.\\
\hlt{Structural Commentary}: The subspace lattice is complete and modular, not generally distributive.\\
\hlt{Key Insight}: If one summand already lies inside $\mathcal{S}$, intersecting with $\mathcal{S}$ can be pushed through the sum.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:direct-sum-equivalent-conditions} \hlt{Equivalent Conditions for a Direct Sum}\\
For subspaces $\mathcal{U}_1, \ldots, \mathcal{U}_m \subseteq \mathcal{V}$, the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$;
\item every $\mathbf{v} \in \mathcal{V}$ has a unique representation $\mathbf{v} = \mathbf{u}_1 + \cdots + \mathbf{u}_m$ with $\mathbf{u}_j \in \mathcal{U}_j$;
\item $\mathcal{V}=\mathcal{U}_1+\cdots+\mathcal{U}_m$, and if $\mathbf{u}_1 + \cdots + \mathbf{u}_m = \mathbf{0}$ with $\mathbf{u}_j \in \mathcal{U}_j$, then $\mathbf{u}_1 = \cdots = \mathbf{u}_m = \mathbf{0}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman)*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ This follows from the definition of direct sum.
\item[$(ii)$]$\Rightarrow (iii)$ Every vector lies in $\mathcal{U}_1+\cdots+\mathcal{U}_m$, so $\mathcal{V}=\mathcal{U}_1+\cdots+\mathcal{U}_m$. If $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$, then this representation of $\mathbf{0}$ must equal $\mathbf{0}+\cdots+\mathbf{0}$ by uniqueness, so each $\mathbf{u}_j=\mathbf{0}$.
\item[$(iii)$]$\Rightarrow (i)$ Since $\mathcal{V}=\mathcal{U}_1+\cdots+\mathcal{U}_m$, every $\mathbf{v}\in\mathcal{V}$ has at least one representation.\\
If $\mathbf{v}=\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{w}_1+\cdots+\mathbf{w}_m$, then $(\mathbf{u}_1-\mathbf{w}_1)+\cdots+(\mathbf{u}_m-\mathbf{w}_m)=\mathbf{0}$. By iii., $\mathbf{u}_j-\mathbf{w}_j=\mathbf{0}$ for every $j$, so the representation is unique. Hence $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_m$.
\end{enumerate}
\hlt{Structural Commentary}: Direct sums require both spanning by the summands and no cancellation among them. The zero-representation condition isolates the no-cancellation part.\\
\hlt{Key Insight}: Uniqueness is equivalent to saying the only way the summands add to zero is trivially.
\end{proof}
\end{proposition}

\begin{definition} \label{def:internal-direct-sum-family} \hlt{Support and Internal Direct Sum of a Family}\\
Let $I$ be an index set. For each $i\in I$, let $\mathcal{X}_i$ be a vector space with zero vector $\mathbf{0}_i$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Support.} For $\mathbf{x}=(\mathbf{x}_i)_{i\in I}\in\prod\limits_{i\in I}\mathcal{X}_i$, define $\operatorname{supp}(\mathbf{x})\defeq\{i\in I:\mathbf{x}_i\neq\mathbf{0}_i\}$.\\
\item The indexed family $\mathbf{x}$ is \hlt{finitely supported} iff $\abs{\operatorname{supp}(\mathbf{x})}<\infty$.
\item \hlt{Finite sum.} If all $\mathcal{X}_i$ lie in a common vector space and $\mathbf{x}$ is finitely supported, define $\sum\limits_{i\in I}\mathbf{x}_i\defeq\sum\limits_{i\in\operatorname{supp}(\mathbf{x})}\mathbf{x}_i$.
\item \hlt{Internal direct sum.} For $\{\mathcal{U}_i:i\in I\}\subseteq\operatorname{Sub}(\mathcal{V})$, $\mathcal{V}=\bigoplus\limits_{i\in I}\mathcal{U}_i$ means every $\mathbf{v}\in\mathcal{V}$ has a unique finitely supported $\mathbf{u}=(\mathbf{u}_i)_{i\in I}$ with $\mathbf{u}_i\in\mathcal{U}_i$ and $\mathbf{v}=\sum\limits_{i\in I}\mathbf{u}_i$.
\end{enumerate}
Equivalently, $\mathcal{V}=\sum\limits_{i\in I}\mathcal{U}_i$ and every finite relation $\sum\limits_{i\in F}\mathbf{u}_i=\mathbf{0}$, with $F\subseteq I$ finite and $\mathbf{u}_i\in\mathcal{U}_i$, has $\mathbf{u}_i=\mathbf{0}$ $\forall i\in F$.
\end{definition}

\begin{proposition} \label{prop:internal-direct-sum-family-criterion} \hlt{Family Direct-Sum Criterion}\\
For $\{\mathcal{U}_i\}_{i\in I}\subseteq\operatorname{Sub}(\mathcal{V})$, $\sum\limits_{i\in I}\mathcal{U}_i$ is direct $\Longleftrightarrow$ $\forall i\in I$, $\mathcal{U}_i\cap\sum\limits_{j\in I\setminus\{i\}}\mathcal{U}_j=\{\mathbf{0}\}$
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathbf{x}\in\mathcal{U}_i\cap\sum\limits_{j\in I\setminus\{i\}}\mathcal{U}_j$, then $\mathbf{x}=\sum\limits_{j\in F}\mathbf{u}_j$ for some finite $F\subseteq I\setminus\{i\}$ and $\mathbf{u}_j\in\mathcal{U}_j$.\\
Directness applied to $\mathbf{x}-\sum\limits_{j\in F}\mathbf{u}_j=\mathbf{0}$ gives $\mathbf{x}=\mathbf{0}$.
\item[$(\Leftarrow)$] If $\sum\limits_{k=1}^{m}\mathbf{u}_{i_k}=\mathbf{0}$ with distinct $i_k$ and $\mathbf{u}_{i_k}\in\mathcal{U}_{i_k}$, then for each $k$, $\mathbf{u}_{i_k}=-\sum\limits_{\ell\neq k}\mathbf{u}_{i_\ell}\in\mathcal{U}_{i_k}\cap\sum\limits_{j\in I\setminus\{i_k\}}\mathcal{U}_j=\{\mathbf{0}\}$.\\
Thus all finite zero-relations are trivial, so the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: Family directness requires each summand to miss all others together.\\
\hlt{Key Insight}: Directness is subspace independence.
\end{proof}
\end{proposition}

\begin{definition} \hlt{Product of Vector Spaces}\\
Let $\mathcal{V}_1,\ldots,\mathcal{V}_m$ be vector spaces over $\mathbb{F}$. Their \hlt{product} is $\mathcal{V}_1\times\cdots\times\mathcal{V}_m \defeq \{(\mathbf{v}_1,\ldots,\mathbf{v}_m):\mathbf{v}_j\in\mathcal{V}_j\}$.\\
Addition, scalar multiplication defined componentwise: $(\mathbf{u}_1,\ldots,\mathbf{u}_m)+(\mathbf{v}_1,\ldots,\mathbf{v}_m) \defeq (\mathbf{u}_1+\mathbf{v}_1,\ldots,\mathbf{u}_m+\mathbf{v}_m)$ and $a(\mathbf{v}_1,\ldots,\mathbf{v}_m) \defeq (a\mathbf{v}_1,\ldots,a\mathbf{v}_m)$.
\end{definition}

\begin{proposition} \label{prop:product-vector-space} \hlt{Products Are Vector Spaces}\\
If $\mathcal{V}_1,\ldots,\mathcal{V}_m$ are vector spaces over $\mathbb{F}$, then $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ is a vector space under componentwise operations.
\begin{proof}
The additive identity is $(\mathbf{0}_1,\ldots,\mathbf{0}_m)$, where $\mathbf{0}_j$ is the zero vector of $\mathcal{V}_j$, and additive inverse of $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is $(-\mathbf{v}_1,\ldots,-\mathbf{v}_m)$. All remaining vector space axioms hold componentwise because they hold in each $\mathcal{V}_j$.\\
\hlt{Structural Commentary}: Product spaces collect several vector spaces into one coordinatewise structure.\\
\hlt{Key Insight}: Every vector space axiom reduces to the same axiom in each component.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-product} \hlt{Dimension of a Product}\\
If $\mathcal{V}_1,\ldots,\mathcal{V}_m$ are finite-dimensional vector spaces over $\mathbb{F}$, then $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ is finite-dimensional and
\begin{align}
\dim(\mathcal{V}_1\times\cdots\times\mathcal{V}_m) = \dim(\mathcal{V}_1)+\cdots+\dim(\mathcal{V}_m) \nonumber
\end{align}
\begin{proof}
For each $j$, let $(\mathbf{v}_{j,1},\ldots,\mathbf{v}_{j,n_j})$ be a basis of $\mathcal{V}_j$. For every $j$ and $k$, let $\mathbf{e}_{j,k}$ denote the element of $\mathcal{V}_1\times\cdots\times\mathcal{V}_m$ whose $j$th component is $\mathbf{v}_{j,k}$ and whose other components are zero.\\
The list of all $\mathbf{e}_{j,k}$ spans the product because each component can be expanded in its own basis. If a linear combination of the $\mathbf{e}_{j,k}$ is zero, then looking at the $j$th component gives a linear relation among $(\mathbf{v}_{j,1},\ldots,\mathbf{v}_{j,n_j})$, so all coefficients in that component are zero. Hence the list is linearly independent and therefore is a basis of the product. Its length is $n_1+\cdots+n_m$.\\
\hlt{Structural Commentary}: A basis for a product is formed by placing each component basis into its own slot.\\
\hlt{Key Insight}: Product dimension counts the independent directions contributed by each factor.
\end{proof}
\end{theorem}

\begin{definition} \label{def:external-products-and-direct-sums-family} \hlt{External Product and Direct Sum}\\
Let $\{\mathcal{V}_i:i\in I\}$ be $\mathbb{F}$-vector spaces.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The \hlt{external direct product} is $\prod\limits_{i\in I}\mathcal{V}_i =\{\mathbf{v}:I\to\bigcup\limits_{i\in I}\mathcal{V}_i:\mathbf{v}(i)\in\mathcal{V}_i\}$.
\item Operations are componentwise: $(\mathbf{u}+\mathbf{v})(i)=\mathbf{u}(i)+\mathbf{v}(i)$ and $(a\mathbf{v})(i)=a\mathbf{v}(i)$.
\item The \hlt{external direct sum} is $\bigoplus\limits_{i\in I}^{\mathrm{ext}}\mathcal{V}_i =\{\mathbf{v}\in\prod\limits_{i\in I}\mathcal{V}_i:\abs{\operatorname{supp}(\mathbf{v})}<\infty\}$. If $\abs{I}<\infty$, then $\bigoplus\limits_{i\in I}^{\mathrm{ext}}\mathcal{V}_i=\prod\limits_{i\in I}\mathcal{V}_i$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:external-direct-sum-family-vector-space} \hlt{External Sums Are Finite-Support Products}\\
With componentwise operations, $\prod\limits_{i\in I}\mathcal{V}_i$ is an $\mathbb{F}$-vector space, $\bigoplus\limits_{i\in I}^{\mathrm{ext}}\mathcal{V}_i =\{\mathbf{v}\in\prod\limits_{i\in I}\mathcal{V}_i:\abs{\operatorname{supp}(\mathbf{v})}<\infty\} \leq \prod\limits_{i\in I}\mathcal{V}_i$.
\begin{proof}
Let $P=\prod_{i\in I}\mathcal{V}_i$. For $\mathbf{u},\mathbf{v}\in P$ and $a,b\in\mathbb{F}$, $(a\mathbf{u}+b\mathbf{v})(i)=a\mathbf{u}(i)+b\mathbf{v}(i)\in\mathcal{V}_i$.\\
The zero $0_P(i)=0_i$, inverses $(-\mathbf{v})(i)=-\mathbf{v}(i)$, and all axioms are coordinatewise.\\
Let $S=\bigoplus_{i\in I}^{\mathrm{ext}}\mathcal{V}_i$. If $\mathbf{u},\mathbf{v}\in S$ and $a,b\in\mathbb{F}$, then $\operatorname{supp}(a\mathbf{u}+b\mathbf{v})\subseteq \operatorname{supp}(\mathbf{u})\cup\operatorname{supp}(\mathbf{v})$, so $a\mathbf{u}+b\mathbf{v}\in S$.\\
Also $0_P\in S$. Hence $S\leq P$.\\
\hlt{Structural Commentary}: $\prod_i\mathcal{V}_i$: arbitrary support; $\bigoplus_i^{\mathrm{ext}}\mathcal{V}_i$: finite support.\\
\hlt{Key Insight}: Finite support is exactly what finite linear combinations preserve.
\end{proof}
\end{proposition}


\begin{proposition} \label{prop:direct-sum-map-injective} \hlt{Direct Sum Map}\\
Suppose $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Define $\Gamma:\mathcal{U}_1\times\cdots\times\mathcal{U}_m\to\mathcal{U}_1+\cdots+\mathcal{U}_m$ by $\Gamma(\mathbf{u}_1,\ldots,\mathbf{u}_m)=\mathbf{u}_1+\cdots+\mathbf{u}_m$.\\
Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a direct sum if and only if $\Gamma$ is injective.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is direct. If $\Gamma(\mathbf{u}_1,\ldots,\mathbf{u}_m)=\mathbf{0}$, then $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$, so directness gives $\mathbf{u}_j=\mathbf{0}$ for all $j$. Hence $\ker\Gamma=\{(\mathbf{0},\ldots,\mathbf{0})\}$, and $\Gamma$ is injective.
\item[$(\Leftarrow)$] Suppose $\Gamma$ is injective. If $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ with $\mathbf{u}_j\in\mathcal{U}_j$, then $\Gamma(\mathbf{u}_1,\ldots,\mathbf{u}_m)=\Gamma(\mathbf{0},\ldots,\mathbf{0})$. Injectivity gives $\mathbf{u}_j=\mathbf{0}$ for all $j$, so every zero decomposition is trivial. Hence the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: The addition map identifies decompositions precisely through its kernel.\\
\hlt{Key Insight}: Directness is injectivity of $\Gamma$.
\end{proof}
\end{proposition}


\begin{theorem} \label{thm:dimension-formula-for-sum} \hlt{Dimension Formula for a Sum}\\
If $\mathcal{U}, \mathcal{W}$ are finite-dimensional subspaces of $\mathcal{V}$, then $\dim(\mathcal{U} + \mathcal{W}) = \dim(\mathcal{U}) + \dim(\mathcal{W}) - \dim(\mathcal{U} \cap \mathcal{W})$.
\begin{proof}
Choose a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r)$ of $\mathcal{U} \cap \mathcal{W}$, extend it to a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p)$ of $\mathcal{U}$, and to a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{w}_1, \ldots, \mathbf{w}_q)$ of $\mathcal{W}$. Then $(\mathbf{e}_1, \ldots, \mathbf{e}_r, \mathbf{u}_1, \ldots, \mathbf{u}_p, \mathbf{w}_1, \ldots, \mathbf{w}_q) $ spans $\mathcal{U}+\mathcal{W}$ because every vector in the sum is the sum of one vector from $\mathcal{U}$ and one vector from $\mathcal{W}$.\\
To prove linear independence, suppose $a_1\mathbf{e}_1+\cdots+a_r\mathbf{e}_r+b_1\mathbf{u}_1+\cdots+b_p\mathbf{u}_p+c_1\mathbf{w}_1+\cdots+c_q\mathbf{w}_q=\mathbf{0}$.
Then $a_1\mathbf{e}_1+\cdots+a_r\mathbf{e}_r+b_1\mathbf{u}_1+\cdots+b_p\mathbf{u}_p$ lies in $\mathcal{U}\cap\mathcal{W}$. Since $(\mathbf{e}_1,\ldots,\mathbf{e}_r)$ is a basis of $\mathcal{U}\cap\mathcal{W}$ and $(\mathbf{e}_1,\ldots,\mathbf{e}_r,\mathbf{u}_1,\ldots,\mathbf{u}_p)$ is a basis of $\mathcal{U}$, we get $b_1=\cdots=b_p=0$. The remaining relation lies in the basis $(\mathbf{e}_1,\ldots,\mathbf{e}_r,\mathbf{w}_1,\ldots,\mathbf{w}_q)$ of $\mathcal{W}$, so $a_1=\cdots=a_r=c_1=\cdots=c_q=0$. Thus the displayed list is a basis of $\mathcal{U}+\mathcal{W}$.\\
Therefore $\dim(\mathcal{U} + \mathcal{W}) = r + p + q = (r+p) + (r+q) - r$.\\
\hlt{Structural Commentary}: The formula corrects double-counting: vectors in $\mathcal{U} \cap \mathcal{W}$ are counted once in each subspace. Extending an intersection basis separates shared directions from new directions.\\
\hlt{Key Insight}: The dimension of a sum is the sum of dimensions minus the overlap.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:dimension-direct-sum} \hlt{Dimension of a Direct Sum}\\
If $\mathcal{U}$ and $\mathcal{W}$ are finite-dimensional subspaces of $\mathcal{V}$ with $\mathcal{U} \cap \mathcal{W} = \{ \mathbf{0} \}$, then $\dim(\mathcal{U} \oplus \mathcal{W}) = \dim(\mathcal{U}) + \dim(\mathcal{W})$.
\begin{proof}
Since $\mathcal{U}\cap\mathcal{W}=\{\mathbf{0}\}$, the sum is direct and $\dim(\mathcal{U}\cap\mathcal{W})=0$. The result follows from Theorem \ref{thm:dimension-formula-for-sum}.\\
\hlt{Structural Commentary}: When the overlap is trivial, there is no double-counting in the dimension formula. Direct sums are exactly the case where the overlap term vanishes.\\
\hlt{Key Insight}: Dimensions add exactly for direct sums.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:direct-sum-iff-dimensions-add} \hlt{Direct Sums and Dimension}\\
Suppose $\mathcal{V}$ is finite-dimensional and $\mathcal{U}_1,\ldots,\mathcal{U}_m$ are subspaces of $\mathcal{V}$. Then $\mathcal{U}_1+\cdots+\mathcal{U}_m$ is a direct sum if and only if $\dim(\mathcal{U}_1+\cdots+\mathcal{U}_m) = \dim(\mathcal{U}_1)+\cdots+\dim(\mathcal{U}_m)$.
\begin{proof}
For each $j$, choose a basis $\beta_j$ of $\mathcal{U}_j$, and concatenate these bases into a list $\beta$.\\
This list spans $\mathcal{U}_1+\cdots+\mathcal{U}_m$ and has length $\dim(\mathcal{U}_1)+\cdots+\dim(\mathcal{U}_m)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If the sum is direct, any linear relation among the vectors in $\beta$ gives a zero decomposition $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ with $\mathbf{u}_j\in\mathcal{U}_j$. Directness gives $\mathbf{u}_j=\mathbf{0}$ for every $j$, and independence of each $\beta_j$ gives all coefficients zero. Thus $\beta$ is a basis of the sum, and the dimensions add.
\item[$(\Leftarrow)$] If the dimensions add, then $\beta$ is a spanning list whose length equals the dimension of the sum, so Proposition \ref{prop:full-dimension-basis-criteria} makes $\beta$ a basis. A zero decomposition $\mathbf{u}_1+\cdots+\mathbf{u}_m=\mathbf{0}$ expands to a zero linear relation in this basis, so every $\mathbf{u}_j=\mathbf{0}$. Hence the sum is direct.
\end{enumerate}
\hlt{Structural Commentary}: Direct sums add dimensions without hidden overlap.\\
\hlt{Key Insight}: Equality of dimensions means the concatenated bases have no linear redundancy.
\end{proof}
\end{theorem}

\begin{definition} \label{def:adapted-basis-direct-sum} \hlt{Basis Adapted to a Direct Sum}\\
Suppose $\mathcal{V}=\mathcal{U}_1\oplus\cdots\oplus\mathcal{U}_m$ and each $\beta_j$ is a basis of $\mathcal{U}_j$. The concatenated basis $\beta=(\beta_1,\ldots,\beta_m)$ of $\mathcal{V}$ is called a \hlt{basis adapted to the direct-sum decomposition}.\\
Equivalently, the vectors in an adapted basis are grouped by the summands whose direct sum is $\mathcal{V}$.
\end{definition}

\begin{corollary} \label{cor:large-subspaces-intersect} \hlt{Large Subspaces Must Intersect}\\
Let $\mathcal{U}$ and $\mathcal{W}$ be subspaces of a finite-dimensional vector space $\mathcal{V}$, and let $k$ be a positive integer.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item If $\dim(\mathcal{U})+\dim(\mathcal{W})>\dim(\mathcal{V})$, then $\mathcal{U}\cap\mathcal{W}$ contains a nonzero vector.
\item If $\dim(\mathcal{U})+\dim(\mathcal{W})\geq\dim(\mathcal{V})+k$, then $\mathcal{U}\cap\mathcal{W}$ contains $k$ linearly independent vectors.
\end{enumerate}
\begin{proof}
Since $\mathcal{U}+\mathcal{W}\subseteq\mathcal{V}$, Theorem \ref{thm:dimension-formula-for-sum} gives
\begin{align}
\dim(\mathcal{U}\cap\mathcal{W}) =\dim(\mathcal{U})+\dim(\mathcal{W})-\dim(\mathcal{U}+\mathcal{W}) \geq \dim(\mathcal{U})+\dim(\mathcal{W})-\dim(\mathcal{V}) \nonumber
\end{align}
If $\dim(\mathcal{U})+\dim(\mathcal{W})>\dim(\mathcal{V})$, the right-hand side is at least $1$, so the intersection contains a nonzero vector. If $\dim(\mathcal{U})+\dim(\mathcal{W})\geq\dim(\mathcal{V})+k$, then $\dim(\mathcal{U}\cap\mathcal{W})\geq k$, so intersection contains $k$ linearly independent vectors.\\
\hlt{Structural Commentary}: Excess dimension appears as overlap.\\
\hlt{Key Insight}: Two large subspaces cannot fit inside $\mathcal{V}$ without sharing directions.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:complements-exist-finite-dimensions} \hlt{Complements Exist in Finite Dimensions}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U} \subseteq \mathcal{V}$ is a subspace, then there exists a subspace $\mathcal{W}$ such that $\mathcal{V} = \mathcal{U} \oplus \mathcal{W}$.
\begin{proof}
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ of $\mathcal{U}$ and extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_n)$ of $\mathcal{V}$.\\
Let $\mathcal{W}=\operatorname{span}(\mathbf{w}_1,\ldots,\mathbf{w}_n)$. Every vector in $\mathcal{V}$ is a linear combination of this extended basis, so $\mathcal{V}=\mathcal{U}+\mathcal{W}$. If $\mathbf{v}\in\mathcal{U}\cap\mathcal{W}$, then $\mathbf{v}$ has one representation using only $\mathbf{u}_1,\ldots,\mathbf{u}_m$ and another using only $\mathbf{w}_1,\ldots,\mathbf{w}_n$. Uniqueness of coefficients in the basis of $\mathcal{V}$ forces $\mathbf{v}=\mathbf{0}$. Hence $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.\\
\hlt{Structural Commentary}: Complements split a space into a subspace and the remaining basis directions.\\
\hlt{Key Insight}: Extend a basis of $\mathcal{U}$ to split $\mathcal{V}$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:complements-exist-arbitrary-vector-spaces} \hlt{Complements Exist}\\
Assuming Zorn's lemma, for every $\mathcal{U}\leq\mathcal{V}$, $\exists\,\mathcal{W}\leq\mathcal{V}$ such that $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.
\begin{proof}
By Theorem \ref{thm:hamel-basis-extension}, choose a Hamel basis $\mathcal{B}_{\mathcal{U}}$ of $\mathcal{U}$ and a Hamel basis $\mathcal{B}\supseteq\mathcal{B}_{\mathcal{U}}$ of $\mathcal{V}$.\\
Put $\mathcal{C}=\mathcal{B}\setminus\mathcal{B}_{\mathcal{U}}$ and $\mathcal{W}=\operatorname{span}(\mathcal{C})$. For $\mathbf{v}\in\mathcal{V}$, its finite $\mathcal{B}$-expansion splits, with finite $F_{\mathcal{U}}\subseteq\mathcal{B}_{\mathcal{U}}$ and $F_{\mathcal{C}}\subseteq\mathcal{C}$,\\
as $\mathbf{v} =\sum\limits_{b\in F_{\mathcal{U}}}a_b b+\sum\limits_{c\in F_{\mathcal{C}}}a_c c \in \mathcal{U}+\mathcal{W}$.
If $\mathbf{x}\in\mathcal{U}\cap\mathcal{W}$, then its $\mathcal{B}$-coordinates are supported in both $\mathcal{B}_{\mathcal{U}}$ and $\mathcal{C}$; uniqueness gives $\mathbf{x}=\mathbf{0}$. Hence $\mathcal{V}=\mathcal{U}\oplus\mathcal{W}$.\\
\hlt{Structural Commentary}: Extend a basis of $\mathcal{U}$; span the new directions.\\
\hlt{Key Insight}: The uniqueness of basis expansion becomes the uniqueness of direct-sum decomposition.
\end{proof}
\end{theorem}

\begin{definition} \label{def:affine-subset-parallel-subspace} \hlt{Affine Subset and Parallel Subspaces}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$. For $\mathbf{v}\in\mathcal{V}$, the subset $\mathbf{v}+\mathcal{U}\defeq\{\mathbf{v}+\mathbf{u}:\mathbf{u}\in\mathcal{U}\}$ is an \hlt{affine subset} of $\mathcal{V}$ parallel to $\mathcal{U}$.
\end{definition}

\begin{proposition} \label{prop:parallel-affine-subsets-equal-or-disjoint} \hlt{Parallel Affine Subsets Are Equal or Disjoint}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$ and let $\mathbf{v},\mathbf{w}\in\mathcal{V}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathbf{v}-\mathbf{w}\in\mathcal{U}$;
\item $\mathbf{v}+\mathcal{U}=\mathbf{w}+\mathcal{U}$;
\item $(\mathbf{v}+\mathcal{U})\cap(\mathbf{w}+\mathcal{U})\neq\emptyset$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman)*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $\mathbf{v}-\mathbf{w}\in\mathcal{U}$ and $\mathbf{u}\in\mathcal{U}$, then $\mathbf{v}+\mathbf{u}=\mathbf{w}+((\mathbf{v}-\mathbf{w})+\mathbf{u})\in\mathbf{w}+\mathcal{U}$. Thus $\mathbf{v}+\mathcal{U}\subseteq\mathbf{w}+\mathcal{U}$, and the reverse inclusion is similar.
\item[$(ii)$]$\Rightarrow (iii)$ Equal affine subsets have the same elements.
\item[$(iii)$]$\Rightarrow (i)$ If $\mathbf{x}\in(\mathbf{v}+\mathcal{U})\cap(\mathbf{w}+\mathcal{U})$, then $\mathbf{x}=\mathbf{v}+\mathbf{u}_1=\mathbf{w}+\mathbf{u}_2$ for some $\mathbf{u}_1,\mathbf{u}_2\in\mathcal{U}$. Hence $\mathbf{v}-\mathbf{w}=\mathbf{u}_2-\mathbf{u}_1\in\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: Parallel affine subsets either coincide or do not meet.\\
\hlt{Key Insight}: Intersection is controlled by the difference of base points.
\end{proof}
\end{proposition}

\begin{definition} \label{def:quotient-space} \hlt{Quotient Space}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$.\\
The \hlt{quotient space} $\mathcal{V}/\mathcal{U}$ is the set of all affine subsets of $\mathcal{V}$ parallel to $\mathcal{U}$: $\mathcal{V}/\mathcal{U}\defeq\{\mathbf{v}+\mathcal{U}:\mathbf{v}\in\mathcal{V}\}$.
\end{definition}

\begin{perspective} \label{per:quotient-cosets-as-equivalence-classes} \hlt{Cosets as Equivalence Classes}\\
The quotient relation is $\mathbf{v}\sim\mathbf{w}$ when $\mathbf{v}-\mathbf{w}\in\mathcal{U}$. Proposition \ref{prop:parallel-affine-subsets-equal-or-disjoint} shows that the equivalence class of $\mathbf{v}$ is exactly the coset $\mathbf{v}+\mathcal{U}$. Thus the quotient space remembers vectors only up to directions in $\mathcal{U}$.
\end{perspective}

\begin{proposition} \label{prop:quotient-relation-congruence} \hlt{Quotient Relation Is a Linear Congruence}\\
Let $\mathcal{U}\leq\mathcal{V}$ and define $\mathbf{v}\equiv_{\mathcal{U}}\mathbf{w}\Longleftrightarrow \mathbf{v}-\mathbf{w}\in\mathcal{U}$.\\
Then $\equiv_{\mathcal{U}}$ is an equivalence relation and, for $a,b\in\mathbb{F}$, $\mathbf{v}_1\equiv_{\mathcal{U}}\mathbf{v}_2$, $\mathbf{w}_1\equiv_{\mathcal{U}}\mathbf{w}_2$ $\Longrightarrow$ $a\mathbf{v}_1+b\mathbf{w}_1\equiv_{\mathcal{U}} a\mathbf{v}_2+b\mathbf{w}_2$.
\begin{proof}
Since $\mathcal{U}\leq\mathcal{V}$, $\mathbf{0}\in\mathcal{U}$, $-\mathcal{U}=\mathcal{U}$, and $\mathcal{U}+\mathcal{U}\subseteq\mathcal{U}$; hence $\equiv_{\mathcal{U}}$ is reflexive, symmetric, and transitive.\\
If $\mathbf{v}_1\equiv_{\mathcal{U}}\mathbf{v}_2$ and $\mathbf{w}_1\equiv_{\mathcal{U}}\mathbf{w}_2$, then $(a\mathbf{v}_1+b\mathbf{w}_1)-(a\mathbf{v}_2+b\mathbf{w}_2) =a(\mathbf{v}_1-\mathbf{v}_2)+b(\mathbf{w}_1-\mathbf{w}_2)\in\mathcal{U}$.\\
\hlt{Structural Commentary}: Subspace congruence makes quotient operations well-defined.\\
\hlt{Key Insight}: Subspace closure is exactly the well-definedness mechanism for quotient operations.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:quotient-space-structure} \hlt{Structure of the Quotient}\\
Let $\mathcal{U}$ be a subspace of $\mathcal{V}$. Addition and scalar multiplication on $\mathcal{V}/\mathcal{U}$ defined by $(\mathbf{v} + \mathcal{U}) + (\mathbf{w} + \mathcal{U}) \defeq (\mathbf{v} + \mathbf{w}) + \mathcal{U}$ and $a(\mathbf{v} + \mathcal{U}) \defeq a\mathbf{v} + \mathcal{U}$ are well-defined, and with these operations $\mathcal{V}/\mathcal{U}$ is a vector space.
\begin{proof}
If $\mathbf{v} + \mathcal{U} = \mathbf{v}' + \mathcal{U}$ and $\mathbf{w} + \mathcal{U} = \mathbf{w}' + \mathcal{U}$, then $\mathbf{v}-\mathbf{v}' \in \mathcal{U}$ and $\mathbf{w}-\mathbf{w}' \in \mathcal{U}$.\\
Since $\mathcal{U}$ is a subspace, $(\mathbf{v}+\mathbf{w}) - (\mathbf{v}'+\mathbf{w}') \in \mathcal{U}$, and $a\mathbf{v} - a\mathbf{v}' \in \mathcal{U}$ so the operations do not depend on the chosen representatives. The vector space axioms are inherited from $\mathcal{V}$ under these operations.\\
\hlt{Structural Commentary}: Quotients preserve operations modulo $\mathcal{U}$.\\
\hlt{Key Insight}: Cosets identify vectors differing by elements of $\mathcal{U}$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:dimension-of-quotient} \hlt{Dimension of a Quotient}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U}$ is a subspace, then $\dim(\mathcal{V}/\mathcal{U}) = \dim(\mathcal{V}) - \dim(\mathcal{U})$.
\begin{proof}
Choose a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m)$ of $\mathcal{U}$ and extend it to a basis $(\mathbf{u}_1,\ldots,\mathbf{u}_m,\mathbf{w}_1,\ldots,\mathbf{w}_k)$ of $\mathcal{V}$.\\
The cosets $\mathbf{w}_1+\mathcal{U},\ldots,\mathbf{w}_k+\mathcal{U}$ span $\mathcal{V}/\mathcal{U}$ because the $\mathbf{u}_j$-terms vanish in the quotient.\\
If $a_1(\mathbf{w}_1+\mathcal{U})+\cdots+a_k(\mathbf{w}_k+\mathcal{U})=\mathcal{U}$, then $a_1\mathbf{w}_1+\cdots+a_k\mathbf{w}_k\in\mathcal{U}$, so a linear relation in the extended basis forces $a_1=\cdots=a_k=0$. Hence these cosets form a basis of $\mathcal{V}/\mathcal{U}$, and its length is $k=\dim(\mathcal{V})-\dim(\mathcal{U})$.\\
\hlt{Structural Commentary}: Quotiening removes the directions in $\mathcal{U}$ and keeps the added basis directions.\\
\hlt{Key Insight}: Passing to $\mathcal{V}/\mathcal{U}$ subtracts the collapsed dimension.
\end{proof}
\end{theorem}

This block develops two complementary ways to organise a vector space: build from smaller pieces by spans and direct sums, or collapse a chosen subspace by passing to a quotient. Bases and dimension measure exactly what is added, split off, or made invisible.
