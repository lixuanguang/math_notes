\subsection{Orthogonal Complements and Projections}

% **Role:** linear algebra with metric and angle structure.
%
% Orthogonal complements, projection theorem, orthogonal projections, best approximation.

Orthogonal complements turn orthogonality into a subspace operation; projections turn it into canonical decompositions and best approximations.

\begin{definition} \label{def:orthogonal-subsets-and-direct-sums}\label{def:orthogonal-complement} \hlt{Orthogonality, Complements, and Direct Sums}\\
Let $\mathcal{X},\mathcal{Y},\mathcal{U}\subseteq\mathcal{V}$ and let $\mathcal{S},\mathcal{T}\leq\mathcal{V}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{X}$ and $\mathcal{Y}$ are \hlt{orthogonal}, written $\mathcal{X}\perp\mathcal{Y}$, if $\innerprod{\mathbf{x},\mathbf{y}}=0$ for all $\mathbf{x}\in\mathcal{X}$ and $\mathbf{y}\in\mathcal{Y}$.
\item The \hlt{orthogonal complement} of $\mathcal{U}$ is $\mathcal{U}^{\perp}\defeq\{\mathbf{v}\in\mathcal{V}:\innerprod{\mathbf{v},\mathbf{u}}=0\text{ for all }\mathbf{u}\in\mathcal{U}\}$.
\item $\mathcal{V}$ is the \hlt{orthogonal direct sum} of $\mathcal{S}$ and $\mathcal{T}$, written $\mathcal{V}=\mathcal{S}\odot\mathcal{T}$, if $\mathcal{V}=\mathcal{S}\oplus\mathcal{T}$ and $\mathcal{S}\perp\mathcal{T}$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:basic-properties-orthogonal-complement} \hlt{Basic Properties of Orthogonal Complements}\\
For subsets $\mathcal{U},\mathcal{W}\subseteq\mathcal{V}$,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{U}^{\perp}$ is a subspace of $\mathcal{V}$;
\item $\{\mathbf{0}\}^{\perp}=\mathcal{V}$;
\item $\mathcal{V}^{\perp}=\{\mathbf{0}\}$;
\item $\mathcal{U}\cap\mathcal{U}^{\perp}\subseteq\{\mathbf{0}\}$;
\item if $\mathcal{U}\subseteq\mathcal{W}$, then $\mathcal{W}^{\perp}\subseteq\mathcal{U}^{\perp}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item We have $\mathbf{0}\in\mathcal{U}^{\perp}$. If $\mathbf{v},\mathbf{w}\in\mathcal{U}^{\perp}$ and $a\in\mathbb{F}$, then for every $\mathbf{u}\in\mathcal{U}$, $\innerprod{a\mathbf{v}+\mathbf{w},\mathbf{u}}=a\innerprod{\mathbf{v},\mathbf{u}}+\innerprod{\mathbf{w},\mathbf{u}}=0$.
\item For every $\mathbf{v}\in\mathcal{V}$, $\innerprod{\mathbf{v},\mathbf{0}}=0$, so $\mathbf{v}\in\{\mathbf{0}\}^{\perp}$.
\item If $\mathbf{v}\in\mathcal{V}^{\perp}$, then $\innerprod{\mathbf{v},\mathbf{v}}=0$, so $\mathbf{v}=\mathbf{0}$.
\item If $\mathbf{v}\in\mathcal{U}\cap\mathcal{U}^{\perp}$, then $\innerprod{\mathbf{v},\mathbf{v}}=0$, so $\mathbf{v}=\mathbf{0}$.
\item If $\mathbf{v}\in\mathcal{W}^{\perp}$, then $\mathbf{v}$ is orthogonal to every vector in $\mathcal{W}$, hence to every vector in $\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: Orthogonal complements are solution spaces of homogeneous inner-product equations. Larger input sets impose more equations, so their complements become smaller.\\
\hlt{Key Insight}: Orthogonality constraints are linear constraints.
\end{proof}
\end{proposition}

\begin{remark} \label{rem:orthogonal-complements-definiteness-warning} \hlt{Definiteness Matters}\\
For a bilinear form $B$, $\mathcal{U}^{\perp}=\{\mathbf{v}:B(\mathbf{v},\mathbf{u})=0\ \forall\mathbf{u}\in\mathcal{U}\}$ is still a subspace.\\
Without positive definiteness, however, one may have $\mathcal{V}^{\perp}\neq\{\mathbf{0}\}$ or $\mathcal{U}\cap\mathcal{U}^{\perp}\neq\{\mathbf{0}\}$.\\
Thus $\mathcal{V}=\mathcal{U}\oplus\mathcal{U}^{\perp}$ is an inner-product result, not a consequence of bilinearity alone.
\end{remark}

\begin{theorem} \label{thm:orthogonal-direct-sum-decomposition} \hlt{Direct Sum with the Orthogonal Complement}\\
If $\mathcal{U}$ is a finite-dimensional subspace of $\mathcal{V}$, then $\mathcal{V}=\mathcal{U}\oplus\mathcal{U}^{\perp}$.
\begin{proof}
Let $(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ be an orthonormal basis of $\mathcal{U}$. For $\mathbf{v}\in\mathcal{V}$, define $\mathbf{u}=\sum\limits_{j=1}^{m}\innerprod{\mathbf{v},\mathbf{e}_j}\mathbf{e}_j$ and set $\mathbf{w}=\mathbf{v}-\mathbf{u}$. Clearly $\mathbf{u}\in\mathcal{U}$. For each $k$, $\innerprod{\mathbf{w},\mathbf{e}_k} =\innerprod{\mathbf{v},\mathbf{e}_k} -\sum\limits_{j=1}^{m}\innerprod{\mathbf{v},\mathbf{e}_j}\innerprod{\mathbf{e}_j,\mathbf{e}_k} =0$.\\
If $\mathbf{x}\in\mathcal{U}$, write $\mathbf{x}=\sum_{k=1}^m a_k\mathbf{e}_k$. Then $\innerprod{\mathbf{w},\mathbf{x}}=\sum_{k=1}^m \overline{a_k}\innerprod{\mathbf{w},\mathbf{e}_k}=0$, so $\mathbf{w}\in\mathcal{U}^{\perp}$ and $\mathbf{v}=\mathbf{u}+\mathbf{w}$.\\
Hence $\mathcal{V}=\mathcal{U}+\mathcal{U}^{\perp}$. The intersection is $\{\mathbf{0}\}$ by Proposition \ref{prop:basic-properties-orthogonal-complement}, so the sum is direct.\\
\hlt{Structural Commentary}: Each vector splits into a $\mathcal{U}$-component and an orthogonal residual.\\
\hlt{Key Insight}: Projection plus residual gives the canonical direct-sum decomposition.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:orthogonal-direct-sum-characterization} \hlt{Orthogonal Direct Sum Characterization}\\
For subspaces $\mathcal{S},\mathcal{T}\subseteq\mathcal{V}$, the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{V}=\mathcal{S}\odot\mathcal{T}$.
\item $\mathcal{V}=\mathcal{S}\oplus\mathcal{T}$ and $\mathcal{T}=\mathcal{S}^{\perp}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $\mathcal{V}=\mathcal{S}\odot\mathcal{T}$, then $\mathcal{V}=\mathcal{S}\oplus\mathcal{T}$ and $\mathcal{T}\subseteq\mathcal{S}^{\perp}$. For $\mathbf{v}\in\mathcal{S}^{\perp}$, write $\mathbf{v}=\mathbf{s}+\mathbf{t}$ with $\mathbf{s}\in\mathcal{S}$, $\mathbf{t}\in\mathcal{T}$.\\
Then $\mathbf{s}=\mathbf{v}-\mathbf{t}\in\mathcal{S}^{\perp}$, so $\mathbf{s}\in\mathcal{S}\cap\mathcal{S}^{\perp}=\{\mathbf{0}\}$ and $\mathbf{v}\in\mathcal{T}$. Hence $\mathcal{T}=\mathcal{S}^{\perp}$.
\item[$(ii)$]$\Rightarrow (i)$ If $\mathcal{V}=\mathcal{S}\oplus\mathcal{T}$ and $\mathcal{T}=\mathcal{S}^{\perp}$, then $\mathcal{S}\perp\mathcal{T}$, so $\mathcal{V}=\mathcal{S}\odot\mathcal{T}$.
\end{enumerate}
\hlt{Structural Commentary}: Orthogonal complements are maximal orthogonal partners.\\
\hlt{Key Insight}: A full orthogonal summand equals the complement.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:dimension-of-orthogonal-complement} \hlt{Dimension of the Orthogonal Complement}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U}$ is a subspace of $\mathcal{V}$, then $\dim(\mathcal{U}^{\perp})=\dim(\mathcal{V})-\dim(\mathcal{U})$.
\begin{proof}
Since $\mathcal{V}$ is finite-dimensional, $\mathcal{U}$ is finite-dimensional by Corollary \ref{cor:finite-dimensional-spaces-and-subspaces-have-bases}. Theorem \ref{thm:orthogonal-direct-sum-decomposition} gives $\mathcal{V}=\mathcal{U}\oplus\mathcal{U}^{\perp}$. Thus Corollary \ref{cor:dimension-direct-sum} gives $\dim(\mathcal{V})=\dim(\mathcal{U})+\dim(\mathcal{U}^{\perp})$.\\
\hlt{Structural Commentary}: Dimensions add across the orthogonal split.\\
\hlt{Key Insight}: Orthogonal complements contain exactly the dimensions not used by $\mathcal{U}$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:double-orthogonal-complement} \hlt{Double Orthogonal Complement}\\
If $\mathcal{U}$ is a finite-dimensional subspace of $\mathcal{V}$, then $\mathcal{U}=(\mathcal{U}^{\perp})^{\perp}$.
\begin{proof}
If $\mathbf{u}\in\mathcal{U}$, then $\mathbf{u}$ is orthogonal to every vector in $\mathcal{U}^{\perp}$, so $\mathcal{U}\subseteq(\mathcal{U}^{\perp})^{\perp}$. Conversely, let $\mathbf{v}\in(\mathcal{U}^{\perp})^{\perp}$. By Theorem \ref{thm:orthogonal-direct-sum-decomposition}, write $\mathbf{v}=\mathbf{u}+\mathbf{w}$ with $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{U}^{\perp}$. Since $\mathcal{U}\subseteq(\mathcal{U}^{\perp})^{\perp}$, we have $\mathbf{w}=\mathbf{v}-\mathbf{u}\in(\mathcal{U}^{\perp})^{\perp}$.\\
Hence $\mathbf{w}\in\mathcal{U}^{\perp}\cap(\mathcal{U}^{\perp})^{\perp}$, so $\mathbf{w}=\mathbf{0}$ and $\mathbf{v}=\mathbf{u}\in\mathcal{U}$.\\
\hlt{Structural Commentary}: The orthogonal split leaves no extra direction beyond $\mathcal{U}$.\\
\hlt{Key Insight}: Taking the orthogonal complement twice recovers the original finite-dimensional subspace.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:double-orthogonal-complement-of-finite-span} \hlt{Double Complement of a Finite Span}\\
If $\mathcal{X}\subseteq\mathcal{V}$ and $\operatorname{span}(\mathcal{X})$ is finite-dimensional, then $(\mathcal{X}^{\perp})^{\perp}=\operatorname{span}(\mathcal{X})$.
\begin{proof}
Let $W=\operatorname{span}(\mathcal{X})$. As $\mathcal{X}^{\perp}=W^{\perp}$, $W$ is finite-dimensional, by Corollary \ref{cor:double-orthogonal-complement}, $(\mathcal{X}^{\perp})^{\perp}=(W^{\perp})^{\perp}=W$.\\
\hlt{Structural Commentary}: Orthogonal complements depend only on the span of the tested set.\\
\hlt{Key Insight}: Double orthogonality recovers the finite-dimensional span, not merely the original subset.
\end{proof}
\end{corollary}

\begin{perspective} \label{per:orthogonal-subspaces-versus-complements} \hlt{Orthogonal Versus Complementary}\\
Orthogonality gives $\mathcal{W}\subseteq\mathcal{U}^{\perp}$ and $\mathcal{U}\subseteq\mathcal{W}^{\perp}$, not necessarily equality. In finite dimensions, equality follows once $\dim(\mathcal{U})+\dim(\mathcal{W})=\dim(\mathcal{V})$ by Corollary \ref{cor:dimension-of-orthogonal-complement}.
\end{perspective}

\begin{proposition} \label{prop:orthogonal-projection-onto-line}\label{prop:orthogonal-decomposition-one-vector} \hlt{Orthogonal Projection onto a Line}\\
Let $\mathbf{u},\mathbf{v}\in\mathcal{V}$ with $\mathbf{v}\neq\mathbf{0}$. Define $c\defeq\frac{\innerprod{\mathbf{u},\mathbf{v}}}{\norm{\mathbf{v}}^2}$, $\mathbf{w}\defeq\mathbf{u}-c\mathbf{v}$. Then $\mathbf{w}\perp\mathbf{v}$ and $\mathbf{u}=c\mathbf{v}+\mathbf{w}$.\\
The vector $\operatorname{proj}_{\mathbf{v}}(\mathbf{u})\defeq c\mathbf{v} =\innerprod{\mathbf{u},\frac{\mathbf{v}}{\norm{\mathbf{v}}}}\frac{\mathbf{v}}{\norm{\mathbf{v}}}$ is the \hlt{orthogonal projection} of $\mathbf{u}$ onto $\operatorname{span}(\mathbf{v})$.
\begin{proof}
The decomposition $\mathbf{u}=c\mathbf{v}+\mathbf{w}$ follows from the definition of $\mathbf{w}$.\\
Moreover, $\innerprod{\mathbf{w},\mathbf{v}} =\innerprod{\mathbf{u}-c\mathbf{v},\mathbf{v}} =\innerprod{\mathbf{u},\mathbf{v}}-c\innerprod{\mathbf{v},\mathbf{v}} =\innerprod{\mathbf{u},\mathbf{v}} -\frac{\innerprod{\mathbf{u},\mathbf{v}}}{\norm{\mathbf{v}}^2}\norm{\mathbf{v}}^2 =0$. Thus $\mathbf{w}\perp\mathbf{v}$.\\
\hlt{Structural Commentary}: Choose $c$ so the remainder $\mathbf{u}-c\mathbf{v}$ has zero inner product with $\mathbf{v}$.\\
\hlt{Key Insight}: Projection is the part detected by pairing with the direction vector.
\end{proof}
\end{proposition}

\begin{definition} \label{def:orthogonal-projection-subspace} \hlt{Orthogonal Projection onto a Subspace}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of $\mathcal{V}$.\\
For $\mathbf{v}\in\mathcal{V}$, write the unique decomposition $\mathbf{v}=\mathbf{u}+\mathbf{w}$, $\mathbf{u}\in\mathcal{U}$, $\mathbf{w}\in\mathcal{U}^{\perp}$, from Theorem \ref{thm:orthogonal-direct-sum-decomposition}.\\
The \hlt{orthogonal projection} of $\mathcal{V}$ onto $\mathcal{U}$ is the map $P_{\mathcal{U}}:\mathcal{V}\to\mathcal{V}$ defined by $P_{\mathcal{U}}\mathbf{v}=\mathbf{u}$.
\end{definition}

\begin{proposition} \label{prop:orthogonal-projection-properties} \hlt{Properties of Orthogonal Projection}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of $\mathcal{V}$. Then:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $P_{\mathcal{U}}\in\LL(\mathcal{V})$;
\item $P_{\mathcal{U}}\mathbf{u}=\mathbf{u}$ for every $\mathbf{u}\in\mathcal{U}$;
\item $P_{\mathcal{U}}\mathbf{w}=\mathbf{0}$ for every $\mathbf{w}\in\mathcal{U}^{\perp}$;
\item $\operatorname{range}(P_{\mathcal{U}})=\mathcal{U}$;
\item $\ker(P_{\mathcal{U}})=\mathcal{U}^{\perp}$, i.e. the kernel/nullspace of $P_{\mathcal{U}}$ is $\mathcal{U}^{\perp}$;
\item $\mathbf{v}-P_{\mathcal{U}}\mathbf{v}\in\mathcal{U}^{\perp}$ for every $\mathbf{v}\in\mathcal{V}$;
\item $P_{\mathcal{U}}^2=P_{\mathcal{U}}$;
\item $\norm{P_{\mathcal{U}}\mathbf{v}}\leq\norm{\mathbf{v}}$ for every $\mathbf{v}\in\mathcal{V}$;
\item if $(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ is an orthonormal basis of $\mathcal{U}$, then $P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{m}\innerprod{\mathbf{v},\mathbf{e}_j}\mathbf{e}_j$, $\mathbf{v}\in\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For linearity, let $a\in\mathbb{F}$ and write $\mathbf{v}_r=\mathbf{u}_r+\mathbf{w}_r$ with $\mathbf{u}_r\in\mathcal{U}$ and $\mathbf{w}_r\in\mathcal{U}^{\perp}$ for $r=1,2$.\\
Then $a\mathbf{v}_1+\mathbf{v}_2=(a\mathbf{u}_1+\mathbf{u}_2)+(a\mathbf{w}_1+\mathbf{w}_2)$, where first summand lies in $\mathcal{U}$ and second in $\mathcal{U}^{\perp}$.\\
By uniqueness of the orthogonal decomposition, $P_{\mathcal{U}}(a\mathbf{v}_1+\mathbf{v}_2)=aP_{\mathcal{U}}\mathbf{v}_1+P_{\mathcal{U}}\mathbf{v}_2$.
\item This follows from $\mathbf{u}=\mathbf{u}+\mathbf{0}$ with $\mathbf{u}\in\mathcal{U}$ and $\mathbf{0}\in\mathcal{U}^{\perp}$.
\item This follows from $\mathbf{w}=\mathbf{0}+\mathbf{w}$ with $\mathbf{0}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{U}^{\perp}$.
\item The definition gives $\operatorname{range}(P_{\mathcal{U}})\subseteq\mathcal{U}$, and (ii) gives $\mathcal{U}\subseteq\operatorname{range}(P_{\mathcal{U}})$.
\item Part (iii) gives $\mathcal{U}^{\perp}\subseteq\ker(P_{\mathcal{U}})$.\\
Conversely, if $P_{\mathcal{U}}\mathbf{v}=\mathbf{0}$, then $\mathbf{v}=\mathbf{0}+(\mathbf{v}-P_{\mathcal{U}}\mathbf{v})$ with $\mathbf{v}-P_{\mathcal{U}}\mathbf{v}\in\mathcal{U}^{\perp}$, so $\mathbf{v}\in\mathcal{U}^{\perp}$.
\item This is exactly the residual term in the decomposition $\mathbf{v}=P_{\mathcal{U}}\mathbf{v}+(\mathbf{v}-P_{\mathcal{U}}\mathbf{v})$.
\item By (ii), $P_{\mathcal{U}}(P_{\mathcal{U}}\mathbf{v})=P_{\mathcal{U}}\mathbf{v}$ for every $\mathbf{v}\in\mathcal{V}$.
\item Write $\mathbf{v}=\mathbf{u}+\mathbf{w}$ with $\mathbf{u}\in\mathcal{U}$, $\mathbf{w}\in\mathcal{U}^{\perp}$. Then $\mathbf{u}\perp\mathbf{w}$ and $\norm{\mathbf{v}}^2=\norm{\mathbf{u}}^2+\norm{\mathbf{w}}^2\geq\norm{\mathbf{u}}^2=\norm{P_{\mathcal{U}}\mathbf{v}}^2$.\\
Since norms are nonnegative, $\norm{P_{\mathcal{U}}\mathbf{v}}\leq\norm{\mathbf{v}}$.
\item This is the explicit $\mathcal{U}$-component constructed in the proof of Theorem \ref{thm:orthogonal-direct-sum-decomposition}.
\end{enumerate}
\hlt{Structural Commentary}: Orthogonal projection keeps the $\mathcal{U}$-component and discards the $\mathcal{U}^{\perp}$-component.\\
\hlt{Key Insight}: $P_{\mathcal{U}}$ is the identity on $\mathcal{U}$ and zero on $\mathcal{U}^{\perp}$.
\end{proof}
\end{proposition}

\begin{definition} \label{def:rank-one-projection-matrix} \hlt{Rank-One Projection Matrix}\\
Let $\mathbf{u}\in\mathbb{F}^n$ be a unit vector. The \hlt{rank-one projection matrix} onto $\operatorname{span}(\mathbf{u})$ is $P_{\mathbf{u}}\defeq \mathbf{u}\mathbf{u}^*\in\operatorname{Mat}(n,n,\mathbb{F})$, where $^*$ denotes conjugate transpose. For every $\mathbf{x}\in\mathbb{F}^n$, $P_{\mathbf{u}}\mathbf{x}=\innerprod[\mathbb{F}^n]{\mathbf{x},\mathbf{u}}\mathbf{u}$.
\end{definition}

\begin{proposition} \label{prop:rank-one-projection-matrix-properties} \hlt{Properties of Rank-One Projection Matrices}\\
Let $\mathbf{u}\in\mathbb{F}^n$ be a unit vector. Then:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\operatorname{range}(P_{\mathbf{u}})=\operatorname{span}(\mathbf{u})$;
\item $\ker(P_{\mathbf{u}})=\operatorname{span}(\mathbf{u})^{\perp}$;
\item $P_{\mathbf{u}}^*=P_{\mathbf{u}}$;
\item $P_{\mathbf{u}}^2=P_{\mathbf{u}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The formula $P_{\mathbf{u}}\mathbf{x}=\innerprod{\mathbf{x},\mathbf{u}}\mathbf{u}$ gives $\operatorname{range}(P_{\mathbf{u}})\subseteq\operatorname{span}(\mathbf{u})$, and $P_{\mathbf{u}}\mathbf{u}=\mathbf{u}$ gives the reverse inclusion.
\item The same formula gives $P_{\mathbf{u}}\mathbf{x}=0$ exactly when $\innerprod{\mathbf{x},\mathbf{u}}\mathbf{u}=\mathbf{0}$.\\
Since $\mathbf{u}\neq\mathbf{0}$, this is equivalent to $\innerprod{\mathbf{x},\mathbf{u}}=0$, i.e. $\mathbf{x}\in\operatorname{span}(\mathbf{u})^{\perp}$.
\item Since $P_{\mathbf{u}}=\mathbf{u}\mathbf{u}^*$, $P_{\mathbf{u}}^*=(\mathbf{u}\mathbf{u}^*)^*=\mathbf{u}\mathbf{u}^*=P_{\mathbf{u}}$.
\item Because $\mathbf{u}^*\mathbf{u}=\norm{\mathbf{u}}_2^2=1$, $P_{\mathbf{u}}^2=\mathbf{u}(\mathbf{u}^*\mathbf{u})\mathbf{u}^*=\mathbf{u}\mathbf{u}^*=P_{\mathbf{u}}$.
\end{enumerate}
\hlt{Structural Commentary}: A rank-one projection keeps the $\mathbf{u}$ direction and kills its orthogonal complement.\\
\hlt{Key Insight}: The outer product $\mathbf{u}\mathbf{u}^*$ encodes projection onto one unit direction.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:orthogonal-projection-minimizes-distance} \hlt{Orthogonal Projection Minimizes Distance}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of $\mathcal{V}$. If $\mathbf{v}\in\mathcal{V}$ and $\mathbf{u}\in\mathcal{U}$, then $\norm{\mathbf{v}-P_{\mathcal{U}}\mathbf{v}}\leq\norm{\mathbf{v}-\mathbf{u}}$.\\
Equality holds if and only if $\mathbf{u}=P_{\mathcal{U}}\mathbf{v}$.
\begin{proof}
Set $\mathbf{p}=P_{\mathcal{U}}\mathbf{v}$, $\mathbf{r}=\mathbf{v}-\mathbf{p}\in\mathcal{U}^{\perp}$, and $\mathbf{h}=\mathbf{p}-\mathbf{u}\in\mathcal{U}$.\\
Then $\mathbf{r}\perp\mathbf{h}$, so $\norm{\mathbf{v}-\mathbf{u}}^2 =\norm{\mathbf{r}+\mathbf{h}}^2 =\norm{\mathbf{r}}^2+\norm{\mathbf{h}}^2 \geq\norm{\mathbf{r}}^2 =\norm{\mathbf{v}-P_{\mathcal{U}}\mathbf{v}}^2$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If equality holds, then $\norm{\mathbf{h}}=0$, hence $\mathbf{u}=\mathbf{p}=P_{\mathcal{U}}\mathbf{v}$.
\item[$(\Leftarrow)$] If $\mathbf{u}=P_{\mathcal{U}}\mathbf{v}$, then equality is immediate.
\end{enumerate}
\hlt{Structural Commentary}: Projection error is orthogonal to every in-subspace correction.\\
\hlt{Key Insight}: The projection is the unique nearest point.
\end{proof}
\end{theorem}

\begin{definition} \label{def:fourier-coefficients-and-expansion} \hlt{Fourier Coefficients and Expansion}\\
Let $\mathcal{O}=(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ be an orthonormal list and $\mathbf{v}\in\mathcal{V}$.\\
The \hlt{Fourier coefficients} of $\mathbf{v}$ relative to $\mathcal{O}$ are $c_j(\mathbf{v})\defeq\innerprod{\mathbf{v},\mathbf{e}_j}$, $1\leq j\leq m$.\\
The \hlt{Fourier expansion} of $\mathbf{v}$ onto $\mathcal{O}$ is $\widehat{\mathbf{v}}_{\mathcal{O}} \defeq \sum\limits_{j=1}^{m}c_j(\mathbf{v})\mathbf{e}_j$.
\end{definition}

\begin{theorem} \label{thm:fourier-expansion-best-approximation-finite} \hlt{Fourier Expansion onto a Finite Orthonormal Set}\\
Let $\mathcal{O}=(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ be orthonormal, let $\mathcal{S}=\operatorname{span}(\mathcal{O})$, and let $\mathbf{v}\in\mathcal{V}$. Then:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\widehat{\mathbf{v}}_{\mathcal{O}}$ is the unique $\mathbf{s}\in\mathcal{S}$ with $\mathbf{v}-\mathbf{s}\in\mathcal{S}^{\perp}$;
\item $\widehat{\mathbf{v}}_{\mathcal{O}}=\operatorname*{arg\,min}_{\mathbf{s}\in\mathcal{S}}\norm{\mathbf{v}-\mathbf{s}}$;
\item $\norm{\widehat{\mathbf{v}}_{\mathcal{O}}}\leq\norm{\mathbf{v}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $1\leq k\leq m$, $\innerprod{\mathbf{v}-\widehat{\mathbf{v}}_{\mathcal{O}},\mathbf{e}_k} =c_k(\mathbf{v})-\sum\limits_{j=1}^{m}c_j(\mathbf{v})\delta_{jk} =0$. Hence $\mathbf{v}-\widehat{\mathbf{v}}_{\mathcal{O}}\in\mathcal{S}^{\perp}$.\\
\item If $\mathbf{s}\in\mathcal{S}$ also satisfies $\mathbf{v}-\mathbf{s}\in\mathcal{S}^{\perp}$, then $\mathbf{s}-\widehat{\mathbf{v}}_{\mathcal{O}}\in\mathcal{S}\cap\mathcal{S}^{\perp}=\{\mathbf{0}\}$.
\item By (i), $\widehat{\mathbf{v}}_{\mathcal{O}}=P_{\mathcal{S}}\mathbf{v}$; Theorem \ref{thm:orthogonal-projection-minimizes-distance} gives the minimizer.
\item By Theorem \ref{thm:bessel-inequality-finite-orthonormal-list}, $\norm{\widehat{\mathbf{v}}_{\mathcal{O}}}^2=\sum_{j=1}^{m}\abs{c_j(\mathbf{v})}^2\leq\norm{\mathbf{v}}^2$, hence $\norm{\widehat{\mathbf{v}}_{\mathcal{O}}}\leq\norm{\mathbf{v}}$.
\end{enumerate}
\hlt{Structural Commentary}: Fourier expansion is orthogonal projection in orthonormal coordinates.\\
\hlt{Key Insight}: The coefficients are obtained by pairing with the basis vectors because all cross terms vanish.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:finite-orthonormal-basis-characterizations} \hlt{Characterizing Finite Orthonormal Bases}\\
Let $\mathcal{O}=(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ be orthonormal, let $\mathcal{S}=\operatorname{span}(\mathcal{O})$, and write $c_j(\mathbf{v})$ and $\widehat{\mathbf{v}}\defeq\widehat{\mathbf{v}}_{\mathcal{O}}$ as in Definition \ref{def:fourier-coefficients-and-expansion}. The following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{S}=\mathcal{V}$.
\item $\mathcal{S}^{\perp}=\{\mathbf{0}\}$.
\item $\widehat{\mathbf{v}}=\mathbf{v}$ for all $\mathbf{v}\in\mathcal{V}$.
\item $\norm{\widehat{\mathbf{v}}}=\norm{\mathbf{v}}$ for all $\mathbf{v}\in\mathcal{V}$.
\item For all $\mathbf{v},\mathbf{w}\in\mathcal{V}$, $\innerprod{\mathbf{v},\mathbf{w}} = \sum\limits_{j=1}^{m}c_j(\mathbf{v})\overline{c_j(\mathbf{w})}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $\mathcal{S}=\mathcal{V}$, then $\mathcal{S}^{\perp}=\mathcal{V}^{\perp}=\{\mathbf{0}\}$.
\item[$(ii)$]$\Rightarrow (iii)$ By Theorem \ref{thm:fourier-expansion-best-approximation-finite}, $\mathbf{v}-\widehat{\mathbf{v}}\in\mathcal{S}^{\perp}=\{\mathbf{0}\}$.
\item[$(iii)$]$\Rightarrow (i)$ Since $\widehat{\mathbf{v}}\in\mathcal{S}$ for all $\mathbf{v}$, condition $(iii)$ gives $\mathcal{V}\subseteq\mathcal{S}$, hence $\mathcal{S}=\mathcal{V}$.
\item[$(iii)$]$\Rightarrow (iv)$ If $\widehat{\mathbf{v}}=\mathbf{v}$, then $\norm{\widehat{\mathbf{v}}}=\norm{\mathbf{v}}$.
\item[$(iv)$]$\Rightarrow (iii)$ The decomposition $\mathbf{v}=\widehat{\mathbf{v}}+(\mathbf{v}-\widehat{\mathbf{v}})$ is orthogonal, so $\norm{\mathbf{v}}^2=\norm{\widehat{\mathbf{v}}}^2+\norm{\mathbf{v}-\widehat{\mathbf{v}}}^2$.\\
Together with $(iv)$, this gives $\mathbf{v}=\widehat{\mathbf{v}}$.
\item[$(iii)$]$\Rightarrow (v)$ If $\widehat{\mathbf{v}}=\mathbf{v}$ and $\widehat{\mathbf{w}}=\mathbf{w}$, then orthonormality gives $\innerprod{\mathbf{v},\mathbf{w}}=\sum_{j=1}^{m}c_j(\mathbf{v})\overline{c_j(\mathbf{w})}$.
\item[$(v)$]$\Rightarrow (iv)$ Put $\mathbf{w}=\mathbf{v}$ and use $\norm{\widehat{\mathbf{v}}}^2=\sum_{j=1}^{m}\abs{c_j(\mathbf{v})}^2$ to get $\norm{\mathbf{v}}=\norm{\widehat{\mathbf{v}}}$.
\end{enumerate}
\hlt{Structural Commentary}: Each condition says projection onto $\mathcal{S}$ loses no component.\\
\hlt{Key Insight}: Basis means lossless projection.
\end{proof}
\end{theorem}

\begin{definition} \label{def:distance-to-subspace} \hlt{Distance to a Subspace}\\
Let $\mathcal{U}$ be a nonempty subset of an inner-product space $\mathcal{V}$.\\
The \hlt{distance} from $\mathbf{v}\in\mathcal{V}$ to $\mathcal{U}$ is $d(\mathbf{v},\mathcal{U})\defeq \inf_{\mathbf{u}\in\mathcal{U}}\norm{\mathbf{v}-\mathbf{u}}$.\\
If $\mathcal{U}$ is finite-dimensional, Theorem \ref{thm:orthogonal-projection-minimizes-distance} gives $d(\mathbf{v},\mathcal{U})=\norm{\mathbf{v}-P_{\mathcal{U}}\mathbf{v}}$.
\end{definition}

\begin{perspective} \label{per:projection-theorem-finite-dimensional-hypothesis} \hlt{Why the Finite-Dimensional Hypothesis Matters}\\
In $\ell^2(\mathbb{F})$, let $\mathcal{S}=\mathbb{F}^{(\mathbb{N})}=\operatorname{span}\{\mathbf{e}_j:j\geq1\}$.\\
If $\mathbf{x}\in\mathcal{S}^{\perp}$, then $x_j=\innerprod{\mathbf{x},\mathbf{e}_j}=0\quad(\forall j)$, so $\mathcal{S}^{\perp}=\{\mathbf{0}\}$. Since $\mathcal{S}\subsetneq\ell^2(\mathbb{F})$, $\mathcal{S}\oplus\mathcal{S}^{\perp}\neq\ell^2(\mathbb{F})$.
\end{perspective}
