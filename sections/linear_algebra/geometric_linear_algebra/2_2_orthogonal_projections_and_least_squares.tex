\subsubsection{Orthogonal Projections and Least Squares}

Orthogonal projections turn inner-product geometry into approximation theory. The projection theorem identifies best approximations by orthogonal residuals, leading to least squares, normal equations, full-rank projection formulas, and minimum-norm solutions.

\begin{proposition} \label{prop:orthogonal-projection-onto-line}\label{prop:orthogonal-decomposition-one-vector} \hlt{Orthogonal Projection onto a Line}\\
Let $\mathbf{u},\mathbf{v}\in\mathcal{V}$ with $\mathbf{v}\neq\mathbf{0}$. Define $c\defeq\frac{\innerprod{\mathbf{u},\mathbf{v}}}{\norm{\mathbf{v}}^2}$, $\mathbf{w}\defeq\mathbf{u}-c\mathbf{v}$. Then $\mathbf{w}\perp\mathbf{v}$ and $\mathbf{u}=c\mathbf{v}+\mathbf{w}$.\\
The vector $\operatorname{proj}_{\mathbf{v}}(\mathbf{u})\defeq c\mathbf{v} =\innerprod{\mathbf{u},\frac{\mathbf{v}}{\norm{\mathbf{v}}}}\frac{\mathbf{v}}{\norm{\mathbf{v}}}$ is the \hlt{orthogonal projection} of $\mathbf{u}$ onto $\operatorname{span}(\mathbf{v})$.
\begin{proof}
The decomposition $\mathbf{u}=c\mathbf{v}+\mathbf{w}$ follows from the definition of $\mathbf{w}$.\\
Moreover, $\innerprod{\mathbf{w},\mathbf{v}} =\innerprod{\mathbf{u}-c\mathbf{v},\mathbf{v}} =\innerprod{\mathbf{u},\mathbf{v}}-c\innerprod{\mathbf{v},\mathbf{v}} =\innerprod{\mathbf{u},\mathbf{v}} -\frac{\innerprod{\mathbf{u},\mathbf{v}}}{\norm{\mathbf{v}}^2}\norm{\mathbf{v}}^2 =0$. Thus $\mathbf{w}\perp\mathbf{v}$.\\
\hlt{Structural Commentary}: Choose $c$ so the remainder $\mathbf{u}-c\mathbf{v}$ has zero inner product with $\mathbf{v}$.\\
\hlt{Key Insight}: Projection is the part detected by pairing with the direction vector.
\end{proof}
\end{proposition}

\begin{definition} \label{def:orthogonal-projection-subspace} \hlt{Orthogonal Projection onto a Subspace}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of $\mathcal{V}$. For $\mathbf{v}\in\mathcal{V}$, write the unique decomposition $\mathbf{v}=\mathbf{u}+\mathbf{w}$, $\mathbf{u}\in\mathcal{U}$, $\mathbf{w}\in\mathcal{U}^{\perp}$, from Theorem \ref{thm:orthogonal-direct-sum-decomposition}. The \hlt{orthogonal projection} of $\mathcal{V}$ onto $\mathcal{U}$ is the map $P_{\mathcal{U}}:\mathcal{V}\to\mathcal{V}$ defined by $P_{\mathcal{U}}\mathbf{v}=\mathbf{u}$.
\end{definition}

\begin{proposition} \label{prop:orthogonal-projection-properties} \hlt{Properties of Orthogonal Projection}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of $\mathcal{V}$. Then:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $P_{\mathcal{U}}\in\LL(\mathcal{V})$;
\item $P_{\mathcal{U}}\mathbf{u}=\mathbf{u}$ for every $\mathbf{u}\in\mathcal{U}$;
\item $P_{\mathcal{U}}\mathbf{w}=\mathbf{0}$ for every $\mathbf{w}\in\mathcal{U}^{\perp}$;
\item $\operatorname{range}(P_{\mathcal{U}})=\mathcal{U}$;
\item $\ker(P_{\mathcal{U}})=\mathcal{U}^{\perp}$, i.e. the kernel/nullspace of $P_{\mathcal{U}}$ is $\mathcal{U}^{\perp}$;
\item $\mathbf{v}-P_{\mathcal{U}}\mathbf{v}\in\mathcal{U}^{\perp}$ for every $\mathbf{v}\in\mathcal{V}$;
\item $P_{\mathcal{U}}^2=P_{\mathcal{U}}$;
\item $\norm{P_{\mathcal{U}}\mathbf{v}}\leq\norm{\mathbf{v}}$ for every $\mathbf{v}\in\mathcal{V}$;
\item if $(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ is an orthonormal basis of $\mathcal{U}$, then $P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{m}\innerprod{\mathbf{v},\mathbf{e}_j}\mathbf{e}_j$, $\mathbf{v}\in\mathcal{V}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For linearity, write $\mathbf{v}_r=\mathbf{u}_r+\mathbf{w}_r$ with $\mathbf{u}_r\in\mathcal{U}$ and $\mathbf{w}_r\in\mathcal{U}^{\perp}$ for $r=1,2$.\\
Then $a\mathbf{v}_1+\mathbf{v}_2=(a\mathbf{u}_1+\mathbf{u}_2)+(a\mathbf{w}_1+\mathbf{w}_2)$, where first summand lies in $\mathcal{U}$ and second in $\mathcal{U}^{\perp}$.
Therefore $P_{\mathcal{U}}(a\mathbf{v}_1+\mathbf{v}_2)=aP_{\mathcal{U}}\mathbf{v}_1+P_{\mathcal{U}}\mathbf{v}_2$.
\item This follows from $\mathbf{u}=\mathbf{u}+\mathbf{0}$ with $\mathbf{u}\in\mathcal{U}$ and $\mathbf{0}\in\mathcal{U}^{\perp}$.
\item This follows from $\mathbf{w}=\mathbf{0}+\mathbf{w}$ with $\mathbf{0}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{U}^{\perp}$.
\item The definition gives $\operatorname{range}(P_{\mathcal{U}})\subseteq\mathcal{U}$, and (ii) gives $\mathcal{U}\subseteq\operatorname{range}(P_{\mathcal{U}})$.
\item Part (iii) gives $\mathcal{U}^{\perp}\subseteq\ker(P_{\mathcal{U}})$.\\
Conversely, if $P_{\mathcal{U}}\mathbf{v}=\mathbf{0}$, then $\mathbf{v}=\mathbf{0}+(\mathbf{v}-P_{\mathcal{U}}\mathbf{v})$ with $\mathbf{v}-P_{\mathcal{U}}\mathbf{v}\in\mathcal{U}^{\perp}$, so $\mathbf{v}\in\mathcal{U}^{\perp}$.
\item This is exactly the residual term in the decomposition $\mathbf{v}=P_{\mathcal{U}}\mathbf{v}+(\mathbf{v}-P_{\mathcal{U}}\mathbf{v})$.
\item By (ii), $P_{\mathcal{U}}(P_{\mathcal{U}}\mathbf{v})=P_{\mathcal{U}}\mathbf{v}$ for every $\mathbf{v}\in\mathcal{V}$.
\item Write $\mathbf{v}=\mathbf{u}+\mathbf{w}$ with $\mathbf{u}\in\mathcal{U}$, $\mathbf{w}\in\mathcal{U}^{\perp}$. Then $\mathbf{u}\perp\mathbf{w}$ and $\norm{\mathbf{v}}^2=\norm{\mathbf{u}}^2+\norm{\mathbf{w}}^2\geq\norm{\mathbf{u}}^2=\norm{P_{\mathcal{U}}\mathbf{v}}^2$.
\item This is the explicit $\mathcal{U}$-component constructed in the proof of Theorem \ref{thm:orthogonal-direct-sum-decomposition}.
\end{enumerate}
\hlt{Structural Commentary}: Orthogonal projection keeps the $\mathcal{U}$-component and discards the $\mathcal{U}^{\perp}$-component.\\
\hlt{Key Insight}: $P_{\mathcal{U}}$ is the identity on $\mathcal{U}$ and zero on $\mathcal{U}^{\perp}$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:orthogonal-projection-minimizes-distance} \hlt{Orthogonal Projection Minimizes Distance}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of $\mathcal{V}$. If $\mathbf{v}\in\mathcal{V}$ and $\mathbf{u}\in\mathcal{U}$, then $\norm{\mathbf{v}-P_{\mathcal{U}}\mathbf{v}}\leq\norm{\mathbf{v}-\mathbf{u}}$.\\
Equality holds if and only if $\mathbf{u}=P_{\mathcal{U}}\mathbf{v}$.
\begin{proof}
Since $\mathbf{v}-P_{\mathcal{U}}\mathbf{v}\in\mathcal{U}^{\perp}$ and $P_{\mathcal{U}}\mathbf{v}-\mathbf{u}\in\mathcal{U}$, the two vectors are orthogonal.\\
Thus $\norm{\mathbf{v}-\mathbf{u}}^2 =\norm{(\mathbf{v}-P_{\mathcal{U}}\mathbf{v})+(P_{\mathcal{U}}\mathbf{v}-\mathbf{u})}^2 =\norm{\mathbf{v}-P_{\mathcal{U}}\mathbf{v}}^2+\norm{P_{\mathcal{U}}\mathbf{v}-\mathbf{u}}^2$.\\
Taking square roots gives the inequality. Equality holds exactly when $\norm{P_{\mathcal{U}}\mathbf{v}-\mathbf{u}}=0$, i.e. when $\mathbf{u}=P_{\mathcal{U}}\mathbf{v}$.\\
\hlt{Structural Commentary}: Projection error is orthogonal to every in-subspace correction.\\
\hlt{Key Insight}: Moving away from the projection adds a perpendicular error term and can only increase distance.
\end{proof}
\end{theorem}

\begin{principle} \label{pri:least-squares-projection-principle} \hlt{Least Squares as Orthogonal Projection}\\
In an inner-product space, the best approximation to $\mathbf{v}$ from a finite-dimensional subspace $\mathcal{U}$ is $P_{\mathcal{U}}\mathbf{v}$. The residual $\mathbf{v}-P_{\mathcal{U}}\mathbf{v}$ is orthogonal to every allowable correction inside $\mathcal{U}$.
\end{principle}

\begin{definition} \label{def:distance-to-subspace} \hlt{Distance to a Subspace}\\
Let $\mathcal{U}$ be a nonempty subset of an inner-product space $\mathcal{V}$. The \hlt{distance} from $\mathbf{v}\in\mathcal{V}$ to $\mathcal{U}$ is $d(\mathbf{v},\mathcal{U})\defeq \inf_{\mathbf{u}\in\mathcal{U}}\norm{\mathbf{v}-\mathbf{u}}$.\\
If $\mathcal{U}$ is finite-dimensional, Theorem \ref{thm:orthogonal-projection-minimizes-distance} gives $d(\mathbf{v},\mathcal{U})=\norm{\mathbf{v}-P_{\mathcal{U}}\mathbf{v}}$.
\end{definition}

\begin{theorem} \label{thm:normal-equations-for-projection} \hlt{Normal Equations for Projection}\\
Let $\mathcal{U}$ be a finite-dimensional subspace of an inner-product space $\mathcal{V}$, and suppose $\mathcal{U}=\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$.\\
For $\mathbf{v}\in\mathcal{V}$, the projection has the form $P_{\mathcal{U}}\mathbf{v}=\sum\limits_{j=1}^{n}c_j\mathbf{u}_j$, where $(c_1,\ldots,c_n)$ solves the normal equations
\begin{align}
\begin{bmatrix}
\innerprod{\mathbf{u}_1,\mathbf{u}_1} & \innerprod{\mathbf{u}_2,\mathbf{u}_1} & \cdots & \innerprod{\mathbf{u}_n,\mathbf{u}_1}\\
\innerprod{\mathbf{u}_1,\mathbf{u}_2} & \innerprod{\mathbf{u}_2,\mathbf{u}_2} & \cdots & \innerprod{\mathbf{u}_n,\mathbf{u}_2}\\
\vdots & \vdots & \ddots & \vdots\\
\innerprod{\mathbf{u}_1,\mathbf{u}_n} & \innerprod{\mathbf{u}_2,\mathbf{u}_n} & \cdots & \innerprod{\mathbf{u}_n,\mathbf{u}_n}
\end{bmatrix}
\begin{bmatrix}
c_1\\ c_2\\ \vdots\\ c_n
\end{bmatrix}
=
\begin{bmatrix}
\innerprod{\mathbf{v},\mathbf{u}_1}\\
\innerprod{\mathbf{v},\mathbf{u}_2}\\
\vdots\\
\innerprod{\mathbf{v},\mathbf{u}_n}
\end{bmatrix}.
\nonumber
\end{align}
This system is consistent. If $\mathbf{u}_1,\ldots,\mathbf{u}_n$ are linearly independent, then the solution is unique.
\begin{proof}
Since $P_{\mathcal{U}}\mathbf{v}\in\mathcal{U}$, write $P_{\mathcal{U}}\mathbf{v}=\sum_{j=1}^{n}c_j\mathbf{u}_j$.\\
The residual $\mathbf{v}-P_{\mathcal{U}}\mathbf{v}$ lies in $\mathcal{U}^{\perp}$, so for each $i$, $0=\innerprod{\mathbf{v}-\sum\limits_{j=1}^{n}c_j\mathbf{u}_j,\mathbf{u}_i} =\innerprod{\mathbf{v},\mathbf{u}_i}-\sum\limits_{j=1}^{n}c_j\innerprod{\mathbf{u}_j,\mathbf{u}_i}$.\\
These equations are exactly the displayed system, so consistency follows from the existence of $P_{\mathcal{U}}\mathbf{v}$. If $\mathbf{u}_1,\ldots,\mathbf{u}_n$ are linearly independent, they form a basis of $\mathcal{U}$, so the coordinates of $P_{\mathcal{U}}\mathbf{v}$ in this basis are unique.\\
\hlt{Structural Commentary}: Normal equations express residual orthogonality.\\
\hlt{Key Insight}: The projection error is perpendicular to $\mathcal{U}$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:matrix-least-squares-normal-equations} \hlt{Matrix Least Squares and Normal Equations}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{y}\in\mathbb{F}^m$, where $A^*$ denotes the conjugate transpose of $A$.\\
A vector $\mathbf{x}_0\in\mathbb{F}^n$ satisfies $\norm{\mathbf{y}-A\mathbf{x}_0}_2=\min_{\mathbf{x}\in\mathbb{F}^n}\norm{\mathbf{y}-A\mathbf{x}}_2$ if and only if $A^*A\mathbf{x}_0=A^*\mathbf{y}$.\\
The normal equations are always consistent, and their solution is unique if and only if $\operatorname{rank}(A)=n$.
\begin{proof}
Let $\mathcal{U}=\operatorname{col}(A)$, and write $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]$. Since $\{A\mathbf{x}:\mathbf{x}\in\mathbb{F}^n\}=\mathcal{U}$, the vector $\mathbf{x}_0$ is least-squares optimal exactly when $A\mathbf{x}_0=P_{\mathcal{U}}\mathbf{y}$, or equivalently when $\mathbf{y}-A\mathbf{x}_0\in\mathcal{U}^{\perp}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathbf{x}_0$ is least-squares optimal, then $\mathbf{y}-A\mathbf{x}_0\in\mathcal{U}^{\perp}$.\\
Hence $\innerprod{\mathbf{y}-A\mathbf{x}_0,\mathbf{a}_j}=0$, $j=1,\ldots,n$, which is $A^*(\mathbf{y}-A\mathbf{x}_0)=\mathbf{0}$, so $A^*A\mathbf{x}_0=A^*\mathbf{y}$.
\item[$(\Leftarrow)$] If $A^*A\mathbf{x}_0=A^*\mathbf{y}$, then $A^*(\mathbf{y}-A\mathbf{x}_0)=\mathbf{0}$, so $\mathbf{y}-A\mathbf{x}_0\in\mathcal{U}^{\perp}$.\\
Hence $A\mathbf{x}_0=P_{\mathcal{U}}\mathbf{y}$, so $\mathbf{x}_0$ is least-squares optimal.
\end{enumerate}
Consistency follows by choosing $\mathbf{x}_0$ such that $A\mathbf{x}_0=P_{\mathcal{U}}\mathbf{y}$.\\
If $\mathbf{x}_0,\mathbf{x}_1$ solve the normal equations, then $\norm{A(\mathbf{x}_0-\mathbf{x}_1)}_2^2 =\innerprod{\mathbf{x}_0-\mathbf{x}_1,A^*A(\mathbf{x}_0-\mathbf{x}_1)}=0$, so $\mathbf{x}_0-\mathbf{x}_1\in\ker(A)$.\\
Thus uniqueness is equivalent to $\ker(A)=\{\mathbf{0}\}$, equivalently $\operatorname{rank}(A)=n$.\\
\hlt{Structural Commentary}: Least squares is orthogonal projection onto the column space of $A$.\\
\hlt{Key Insight}: The normal equations say exactly that the residual is orthogonal to every column of $A$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:full-column-rank-projection-formula} \hlt{Projection Formula for Full Column Rank}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ have rank $n$.\\
Then $A^*A$ is invertible and the orthogonal projection onto $\operatorname{col}(A)$ is $P_{\operatorname{col}(A)}=A(A^*A)^{-1}A^*$.
\begin{proof}
If $A^*A\mathbf{x}=\mathbf{0}$, then $\norm{A\mathbf{x}}_2^2=\innerprod{\mathbf{x},A^*A\mathbf{x}}=0$, so $A\mathbf{x}=\mathbf{0}$.\\
Since $\operatorname{rank}(A)=n$, we get $\mathbf{x}=\mathbf{0}$. Hence $A^*A$ is invertible. By Theorem \ref{thm:matrix-least-squares-normal-equations}, the least-squares solution is $\mathbf{x}_0=(A^*A)^{-1}A^*\mathbf{y}$. The fitted vector is $P_{\operatorname{col}(A)}\mathbf{y}=A\mathbf{x}_0=A(A^*A)^{-1}A^*\mathbf{y}$.\\
Since this holds for every $\mathbf{y}$, the displayed operator formula follows.\\
\hlt{Structural Commentary}: Full column rank makes the normal-equation coordinates unique.\\
\hlt{Key Insight}: Solve for coefficients, then map back by $A$.
\end{proof}
\end{corollary}

\begin{definition} \label{def:minimum-norm-solution} \hlt{Minimum Norm Solution}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $\mathbf{y}\in\operatorname{col}(A)$.\\
A solution $\mathbf{s}$ of $A\mathbf{x}=\mathbf{y}$ is a \hlt{minimum norm solution} if $\norm{\mathbf{s}}_2\leq\norm{\mathbf{x}}_2$ for every solution $\mathbf{x}$ of $A\mathbf{x}=\mathbf{y}$.
\end{definition}

\begin{theorem} \label{thm:minimum-norm-solution-consistent-system} \hlt{Minimum Norm Solution of a Consistent Linear System}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $\mathbf{y}\in\mathbb{F}^m$, and suppose $A\mathbf{x}=\mathbf{y}$ is consistent.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item There is a unique minimum norm solution $\mathbf{s}\in\mathbb{F}^n$, and $\mathbf{s}\in\operatorname{col}(A^*)$.
\item $\mathbf{s}$ is the only solution of $A\mathbf{x}=\mathbf{y}$ that lies in $\operatorname{col}(A^*)$.
\item The system $AA^*\mathbf{u}=\mathbf{y}$ is consistent. If $AA^*\mathbf{u}_0=\mathbf{y}$, then $\mathbf{s}=A^*\mathbf{u}_0$.
\end{enumerate}
\begin{proof}
First note that $\operatorname{col}(A^*)=(\ker A)^{\perp}$. Indeed, if $\mathbf{z}=A^*\mathbf{u}$ and $\mathbf{x}\in\ker(A)$, then $\innerprod{\mathbf{x},\mathbf{z}}=\innerprod{A\mathbf{x},\mathbf{u}}=0$, so $\mathbf{z}$ is orthogonal to every vector in $\ker(A)$. Thus $\operatorname{col}(A^*)\subseteq(\ker A)^{\perp}$. Also,
\begin{align}
\dim(\operatorname{col}(A^*))=\operatorname{rank}(A^*)=\operatorname{rank}(A)
=n-\dim(\ker(A))
=\dim((\ker A)^{\perp}), \nonumber
\end{align}
so the two subspaces are equal. Hence $\mathbb{F}^n=\operatorname{col}(A^*)\oplus\ker(A)$ orthogonally.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose a solution $\mathbf{x}_0$ of $A\mathbf{x}=\mathbf{y}$ and write $\mathbf{x}_0=\mathbf{s}+\mathbf{v}$, with $\mathbf{s}\in\operatorname{col}(A^*)$ and $\mathbf{v}\in\ker(A)$. Then $A\mathbf{s}=A\mathbf{x}_0=\mathbf{y}$. Every solution has the form $\mathbf{s}+\mathbf{w}$ with $\mathbf{w}\in\ker(A)$.\\
Since $\mathbf{s}\perp\mathbf{w}$, $\norm{\mathbf{s}+\mathbf{w}}_2^2=\norm{\mathbf{s}}_2^2+\norm{\mathbf{w}}_2^2\geq\norm{\mathbf{s}}_2^2$, with equality only when $\mathbf{w}=\mathbf{0}$.
\item If $\mathbf{s}'\in\operatorname{col}(A^*)$ is another solution, then $\mathbf{s}'-\mathbf{s}\in\ker(A)\cap\operatorname{col}(A^*)=\{\mathbf{0}\}$, so $\mathbf{s}'=\mathbf{s}$.
\item Since $\mathbf{s}\in\operatorname{col}(A^*)$, write $\mathbf{s}=A^*\mathbf{u}_0$. Then $AA^*\mathbf{u}_0=A\mathbf{s}=\mathbf{y}$.\\
Conversely, if $AA^*\mathbf{u}_0=\mathbf{y}$, then $A^*\mathbf{u}_0$ is a solution in $\operatorname{col}(A^*)$, so $A^*\mathbf{u}_0=\mathbf{s}$ by (ii).
\end{enumerate}
\hlt{Structural Commentary}: Solutions are $\mathbf{s}+\ker(A)$; $\operatorname{col}(A^*)$ meets this set orthogonally.\\
\hlt{Key Insight}: The minimum norm solution has no nullspace component.
\end{proof}
\end{theorem}
