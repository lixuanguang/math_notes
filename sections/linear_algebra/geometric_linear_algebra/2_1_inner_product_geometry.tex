\subsubsection{Inner Product Geometry}

Throughout this subsection, $\mathbb{F}$ denotes either $\mathbb{R}$ or $\mathbb{C}$. Inner products turn linear spaces into geometric spaces: they define norms, orthogonality, orthogonal complements, orthonormal bases, Gram--Schmidt, Bessel and Parseval identities, Riesz representation, and Gram matrices as records of geometry.

\begin{definition} \label{def:inner-product-preliminary-terminology} \hlt{Preliminary Terminology}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\mathbf{x},\mathbf{y}\in\mathbb{R}^n$, the \hlt{dot product} is $\mathbf{x}\cdot\mathbf{y}\defeq x_1y_1+\cdots+x_ny_n$.
\item For $z=a+bi\in\mathbb{C}$, the \hlt{complex conjugate} is $\overline{z}\defeq a-bi$, and the \hlt{modulus} is $\abs{z}\defeq\sqrt{a^2+b^2}$.
\item A map $f:\mathcal{V}\to\mathbb{F}$ is \hlt{conjugate linear} if $f(\mathbf{u}+\mathbf{v})=f(\mathbf{u})+f(\mathbf{v})$ and $f(\lambda\mathbf{v})=\overline{\lambda}f(\mathbf{v})$.
\item A function $B:\mathcal{V}\times\mathcal{V}\to\mathbb{F}$ is \hlt{sesquilinear} if it is linear in one slot and conjugate linear in the other.
\end{enumerate}
\end{definition}

\begin{motivation} \label{mot:inner-product-from-dot-product} \hlt{From Dot Products to Inner Products}\\
For $\mathbf{x}=(x_1,\ldots,x_n)\in\mathbb{R}^n$, the Euclidean length is $\norm{\mathbf{x}}=\sqrt{x_1^2+\cdots+x_n^2}$.\\
The expression under the square root is the dot product of $\mathbf{x}$ with itself. Over $\mathbb{C}$, conjugates are needed to keep squared length real and nonnegative: $\norm{\mathbf{z}}^2=\abs{z_1}^2+\cdots+\abs{z_n}^2 =z_1\overline{z_1}+\cdots+z_n\overline{z_n}$.\\
Thus the standard inner product on $\mathbb{F}^n$ is $\innerprod{\mathbf{x},\mathbf{y}} \defeq x_1\overline{y_1}+\cdots+x_n\overline{y_n}$.
\end{motivation}

\begin{definition} \label{def:inner-product} \hlt{Inner Product}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. An \hlt{inner product} on $\mathcal{V}$ is a function that assigns to each pair $\mathbf{u},\mathbf{v}\in\mathcal{V}$ a scalar $\innerprod{\mathbf{u},\mathbf{v}}\in\mathbb{F}$ satisfying, for all $\mathbf{u},\mathbf{v},\mathbf{w}\in\mathcal{V}$ and all $\lambda\in\mathbb{F}$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{positivity}: $\innerprod{\mathbf{v},\mathbf{v}}\geq 0$;
\item \hlt{definiteness}: $\innerprod{\mathbf{v},\mathbf{v}}=0$ if and only if $\mathbf{v}=\mathbf{0}$;
\item \hlt{additivity in the first slot}: $\innerprod{\mathbf{u}+\mathbf{v},\mathbf{w}}=\innerprod{\mathbf{u},\mathbf{w}}+\innerprod{\mathbf{v},\mathbf{w}}$;
\item \hlt{homogeneity in the first slot}: $\innerprod{\lambda\mathbf{u},\mathbf{v}}=\lambda\innerprod{\mathbf{u},\mathbf{v}}$;
\item \hlt{conjugate symmetry}: $\innerprod{\mathbf{u},\mathbf{v}}=\overline{\innerprod{\mathbf{v},\mathbf{u}}}$.
\end{enumerate}
In the positivity axiom, $\innerprod{\mathbf{v},\mathbf{v}}\geq 0$ means that $\innerprod{\mathbf{v},\mathbf{v}}$ is a real nonnegative scalar.
\end{definition}

\begin{definition} \label{def:inner-product-space} \hlt{Inner-Product Space}\\
An \hlt{inner-product space} is a vector space $\mathcal{V}$ together with an inner product on $\mathcal{V}$.
\end{definition}

\begin{example} \label{ex:standard-inner-product-on-fn} \hlt{Standard Inner Product on $\mathbb{F}^n$}\\
For $\mathbf{x},\mathbf{y}\in\mathbb{F}^n$, the standard inner product is $\innerprod{\mathbf{x},\mathbf{y}}=\sum\limits_{j=1}^{n}x_j\overline{y_j}$. When $\mathbb{F}=\mathbb{R}$, this is the usual dot product.
\end{example}

\begin{example} \label{ex:function-and-matrix-inner-products} \hlt{Function and Matrix Inner Products}\\
If $\mathcal{V}=C_{\mathbb{F}}[a,b]$, then $\innerprod{f,g}=\int_a^b f(t)\overline{g(t)}\,dt$ is an inner product.\\
If $\mathcal{V}=\operatorname{Mat}(m,n,\mathbb{F})$, then the \hlt{Frobenius inner product} is $\innerprod[F]{A,B} \defeq \sum\limits_{i=1}^{m}\sum\limits_{j=1}^{n} a_{ij}\overline{b_{ij}}$.
\end{example}

\begin{proposition} \label{prop:basic-inner-product-identities} \hlt{Basic Inner-Product Identities}\\
Let $\mathcal{V}$ be an inner-product space. Then for all $\mathbf{u},\mathbf{v},\mathbf{w}\in\mathcal{V}$ and $\lambda\in\mathbb{F}$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item for fixed $\mathbf{u}$, the map $\mathbf{v}\mapsto \innerprod{\mathbf{v},\mathbf{u}}$ is linear;
\item $\innerprod{\mathbf{0},\mathbf{u}}=0$ and $\innerprod{\mathbf{u},\mathbf{0}}=0$;
\item $\innerprod{\mathbf{u},\mathbf{v}+\mathbf{w}}=\innerprod{\mathbf{u},\mathbf{v}}+\innerprod{\mathbf{u},\mathbf{w}}$;
\item $\innerprod{\mathbf{u},\lambda\mathbf{v}}=\overline{\lambda}\innerprod{\mathbf{u},\mathbf{v}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item This is exactly additivity and homogeneity in the first slot.
\item A linear map sends $\mathbf{0}$ to $0$, so $\innerprod{\mathbf{0},\mathbf{u}}=0$. Conjugate symmetry gives $\innerprod{\mathbf{u},\mathbf{0}}=\overline{\innerprod{\mathbf{0},\mathbf{u}}}=0$.
\item By conjugate symmetry and additivity, $\innerprod{\mathbf{u},\mathbf{v}+\mathbf{w}} =\overline{\innerprod{\mathbf{v}+\mathbf{w},\mathbf{u}}} =\overline{\innerprod{\mathbf{v},\mathbf{u}}+\innerprod{\mathbf{w},\mathbf{u}}} =\innerprod{\mathbf{u},\mathbf{v}}+\innerprod{\mathbf{u},\mathbf{w}}$.
\item By conjugate symmetry and homogeneity, $\innerprod{\mathbf{u},\lambda\mathbf{v}} =\overline{\innerprod{\lambda\mathbf{v},\mathbf{u}}} =\overline{\lambda\innerprod{\mathbf{v},\mathbf{u}}} =\overline{\lambda}\innerprod{\mathbf{u},\mathbf{v}}$.
\end{enumerate}
\hlt{Structural Commentary}: First-slot linearity and conjugate symmetry force conjugate linearity in second slot.\\
\hlt{Key Insight}: Once one slot is linear, the other slot is forced to be conjugate linear.
\end{proof}
\end{proposition}

\begin{definition} \label{def:norm-induced-by-inner-product} \hlt{Norm Induced by an Inner Product}\\
For $\mathbf{v}\in\mathcal{V}$, the \hlt{norm} of $\mathbf{v}$ is $\norm{\mathbf{v}}\defeq\sqrt{\innerprod{\mathbf{v},\mathbf{v}}}$.
\end{definition}

\begin{example} \label{ex:norms-induced-by-inner-products} \hlt{Norms from Standard Inner Products}\\
In $\mathbb{F}^n$, the norm from the standard inner product is the \hlt{Euclidean norm} $\norm{(x_1,\ldots,x_n)}_2=\sqrt{\abs{x_1}^2+\cdots+\abs{x_n}^2}$.\\
In $C_{\mathbb{F}}[a,b]$, with the integral inner product, the derived norm is the \hlt{$L^2$ norm} $\norm{f}_{2}=\sqrt{\int_a^b\abs{f(t)}^2\,dt}$.\\
In $\operatorname{Mat}(m,n,\mathbb{F})$, with the Frobenius inner product, the \hlt{Frobenius norm} is $\norm{A}_{F}=\sqrt{\innerprod[F]{A,A}}=\sqrt{\sum_{i=1}^{m}\sum_{j=1}^{n}\abs{a_{ij}}^2}$, $A=[a_{ij}]$. The Frobenius norm is also called the \hlt{Schur norm} or \hlt{Hilbert-Schmidt norm}.
\end{example}

\begin{proposition} \label{prop:derived-norm-basic-properties}\label{prop:basic-norm-identities} \hlt{Basic Properties of the Derived Norm}\\
Let $\mathcal{V}$ be an inner-product space with derived norm $\norm{\cdot}$. For all $\mathbf{v}\in\mathcal{V}$ and $\lambda\in\mathbb{F}$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\norm{\mathbf{v}}$ is real and nonnegative;
\item $\norm{\mathbf{v}}=0$ if and only if $\mathbf{v}=\mathbf{0}$;
\item $\norm{\lambda\mathbf{v}}=\abs{\lambda}\norm{\mathbf{v}}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By positivity, $\innerprod{\mathbf{v},\mathbf{v}}$ is real and nonnegative, so its square root is real and nonnegative.
\item By definition, $\norm{\mathbf{v}}=0$ if and only if $\innerprod{\mathbf{v},\mathbf{v}}=0$, which holds if and only if $\mathbf{v}=\mathbf{0}$ by definiteness.
\item Compute $\norm{\lambda\mathbf{v}}^2 =\innerprod{\lambda\mathbf{v},\lambda\mathbf{v}} =\lambda\overline{\lambda}\innerprod{\mathbf{v},\mathbf{v}} =\abs{\lambda}^2\norm{\mathbf{v}}^2$. Taking nonnegative square roots, $\norm{\lambda\mathbf{v}}=\abs{\lambda}\norm{\mathbf{v}}$.
\end{enumerate}
\hlt{Structural Commentary}: Positivity gives length; sesquilinearity gives scaling.\\
\hlt{Key Insight}: Squared length is easier to manipulate than length itself.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:cauchy-schwarz-inequality} \hlt{Cauchy-Schwarz Inequality}\\
For all $\mathbf{u},\mathbf{v}\in\mathcal{V}$, $\abs{\innerprod{\mathbf{u},\mathbf{v}}} \leq \norm{\mathbf{u}}\norm{\mathbf{v}}$.
Equality holds if and only if one of $\mathbf{u},\mathbf{v}$ is a scalar multiple of the other.
\begin{proof}
If $\mathbf{v}=\mathbf{0}$, both sides are $0$. Assume $\mathbf{v}\neq\mathbf{0}$.\\
For any $\lambda\in\mathbb{F}$, positivity gives $0\leq \norm{\mathbf{u}-\lambda\mathbf{v}}^2 =\norm{\mathbf{u}}^2-\lambda\innerprod{\mathbf{v},\mathbf{u}}-\overline{\lambda}\innerprod{\mathbf{u},\mathbf{v}}+\abs{\lambda}^2\norm{\mathbf{v}}^2$.\\
Choose $\lambda=\frac{\innerprod{\mathbf{u},\mathbf{v}}}{\norm{\mathbf{v}}^2}$. Since $\innerprod{\mathbf{v},\mathbf{u}}=\overline{\innerprod{\mathbf{u},\mathbf{v}}}$, the inequality becomes $0\leq \norm{\mathbf{u}}^2-\frac{\abs{\innerprod{\mathbf{u},\mathbf{v}}}^2}{\norm{\mathbf{v}}^2}$.\\
Multiplying by $\norm{\mathbf{v}}^2$ and taking square roots proves the inequality. Equality holds exactly when $\mathbf{u}-\lambda\mathbf{v}=\mathbf{0}$ for this choice of $\lambda$, so $\mathbf{u}$ is a scalar multiple of $\mathbf{v}$. The case $\mathbf{v}=\mathbf{0}$ is also covered because $\mathbf{v}=0\mathbf{u}$.\\
\hlt{Structural Commentary}: Positivity of squared length controls the mixed inner-product term.\\
\hlt{Key Insight}: The best scalar comparison between two vectors bounds their inner product.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:triangle-inequality-inner-product} \hlt{Triangle Inequality}\\
For all $\mathbf{u},\mathbf{v}\in\mathcal{V}$, $\norm{\mathbf{u}+\mathbf{v}} \leq \norm{\mathbf{u}}+\norm{\mathbf{v}}$.\\
Equality holds if and only if one of $\mathbf{u},\mathbf{v}$ is a nonnegative scalar multiple of the other.
\begin{proof}
We compute
\begin{align}
\norm{\mathbf{u}+\mathbf{v}}^2 &=\innerprod{\mathbf{u}+\mathbf{v},\mathbf{u}+\mathbf{v}} =\norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2+\innerprod{\mathbf{u},\mathbf{v}}+\innerprod{\mathbf{v},\mathbf{u}} =\norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2+2\operatorname{Re}\innerprod{\mathbf{u},\mathbf{v}} \nonumber\\
&\leq \norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2+2\abs{\innerprod{\mathbf{u},\mathbf{v}}} \leq \norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2+2\norm{\mathbf{u}}\norm{\mathbf{v}} =\bigl(\norm{\mathbf{u}}+\norm{\mathbf{v}}\bigr)^2.
\nonumber
\end{align}
Taking square roots gives the inequality. If either vector is $\mathbf{0}$, the equality condition is immediate. Otherwise, equality holds exactly when $\innerprod{\mathbf{u},\mathbf{v}}=\abs{\innerprod{\mathbf{u},\mathbf{v}}}$ and equality holds in Cauchy-Schwarz. Thus $\mathbf{u}=a\mathbf{v}$ for some $a\in\mathbb{F}$ and $a\norm{\mathbf{v}}^2=\innerprod{\mathbf{u},\mathbf{v}}$ is real nonnegative, so $a\geq 0$. Conversely, if $\mathbf{u}=a\mathbf{v}$ with $a\geq 0$, then $\norm{\mathbf{u}+\mathbf{v}}=(a+1)\norm{\mathbf{v}}=\norm{\mathbf{u}}+\norm{\mathbf{v}}$.\\
\hlt{Structural Commentary}: Square the norm and bound the cross term by Cauchy-Schwarz.\\
\hlt{Key Insight}: Equality means the two vectors point in the same direction.
\end{proof}
\end{theorem}

\begin{definition} \label{def:norm-and-normed-vector-space} \hlt{Norm and Normed Vector Space}\\
Let $\mathcal{V}$ be a vector space over $\mathbb{F}$. A \hlt{norm} on $\mathcal{V}$ is a function $\norm{\cdot}:\mathcal{V}\to[0,\infty)$ such that, $\forall \mathbf{u},\mathbf{v}\in\mathcal{V}$ and $\lambda\in\mathbb{F}$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\norm{\mathbf{u}}\geq0$;
\item $\norm{\mathbf{u}}=0$ if and only if $\mathbf{u}=\mathbf{0}$;
\item $\norm{\lambda\mathbf{u}}=\abs{\lambda}\norm{\mathbf{u}}$;
\item $\norm{\mathbf{u}+\mathbf{v}}\leq\norm{\mathbf{u}}+\norm{\mathbf{v}}$.
\end{enumerate}
A \hlt{normed vector space} is a vector space equipped with a norm. Its \hlt{unit ball} is $\{\mathbf{v}\in\mathcal{V}:\norm{\mathbf{v}}\leq1\}$.
\end{definition}

\begin{theorem} \label{thm:parallelogram-identity-inner-product} \hlt{Parallelogram Identity}\\
For all $\mathbf{u},\mathbf{v}\in\mathcal{V}$, $\norm{\mathbf{u}+\mathbf{v}}^2+\norm{\mathbf{u}-\mathbf{v}}^2 = 2\bigl(\norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2\bigr)$.
\begin{proof}
Expanding both squared norms gives
\begin{align}
\norm{\mathbf{u}+\mathbf{v}}^2 =\norm{\mathbf{u}}^2+\innerprod{\mathbf{u},\mathbf{v}} +\innerprod{\mathbf{v},\mathbf{u}}+\norm{\mathbf{v}}^2, \quad \norm{\mathbf{u}-\mathbf{v}}^2 =\norm{\mathbf{u}}^2-\innerprod{\mathbf{u},\mathbf{v}} -\innerprod{\mathbf{v},\mathbf{u}}+\norm{\mathbf{v}}^2. \nonumber
\end{align}
Adding the two identities cancels the cross terms and yields the result.\\
\hlt{Structural Commentary}: Opposite cross terms cancel when the two diagonal norms are added.\\
\hlt{Key Insight}: Inner-product norms remember parallelogram geometry through cancellation.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:coordinate-norms-on-fn} \hlt{Coordinate Norms on $\mathbb{F}^n$}\\
For $\mathbf{x}=(x_1,\ldots,x_n)\in\mathbb{F}^n$, $\norm{\mathbf{x}}_1=\sum\limits_{j=1}^{n}\abs{x_j}$, $\norm{\mathbf{x}}_2=\left(\sum\limits_{j=1}^{n}\abs{x_j}^2\right)^{1/2}$, $\norm{\mathbf{x}}_{\infty}=\max_{1\leq j\leq n}\abs{x_j}$ are norms on $\mathbb{F}^n$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\norm{\cdot}_1$, nonnegativity, definiteness, and homogeneity follow from the corresponding facts for modulus. The triangle inequality follows from $\norm{\mathbf{x}+\mathbf{y}}_1=\sum\limits_{j=1}^{n}\abs{x_j+y_j}\leq\sum_{j=1}^{n}\bigl(\abs{x_j}+\abs{y_j}\bigr)=\norm{\mathbf{x}}_1+\norm{\mathbf{y}}_1$.
\item For $\norm{\cdot}_2$, the norm is induced by the standard inner product, so the axioms follow from Proposition \ref{prop:derived-norm-basic-properties} and Theorem \ref{thm:triangle-inequality-inner-product}.
\item For $\norm{\cdot}_{\infty}$, nonnegativity, definiteness, and homogeneity are immediate from the maximum of finitely many moduli. If $k$ is chosen so that $\norm{\mathbf{x}+\mathbf{y}}_{\infty}=\abs{x_k+y_k}$, then $\norm{\mathbf{x}+\mathbf{y}}_{\infty}\leq\abs{x_k}+\abs{y_k}\leq\norm{\mathbf{x}}_{\infty}+\norm{\mathbf{y}}_{\infty}$.
\end{enumerate}
\hlt{Structural Commentary}: The $\ell_1$ norm sums coordinate magnitudes, $\ell_{\infty}$ keeps only the largest coordinate magnitude, and $\ell_2$ comes from the standard inner product.\\
\hlt{Key Insight}: All three obey the same abstract axioms, even though their unit balls have different shapes.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:l1-linfty-not-inner-product-norms} \hlt{$\ell_1$ and $\ell_{\infty}$ Are Not Inner-Product Norms}\\
If $n\geq2$, then the norms $\norm{\cdot}_1$ and $\norm{\cdot}_{\infty}$ on $\mathbb{F}^n$ are not induced by any inner product.
\begin{proof}
Let $\mathbf{e}_1,\mathbf{e}_2$ be the first two standard coordinate vectors.\\
For $\norm{\cdot}_1$, $\norm{\mathbf{e}_1+\mathbf{e}_2}_1^2+\norm{\mathbf{e}_1-\mathbf{e}_2}_1^2=4+4=8$, but $2\norm{\mathbf{e}_1}_1^2+2\norm{\mathbf{e}_2}_1^2=4$.\\
For $\norm{\cdot}_{\infty}$, $\norm{\mathbf{e}_1+\mathbf{e}_2}_{\infty}^2+\norm{\mathbf{e}_1-\mathbf{e}_2}_{\infty}^2=1+1=2$ but $2\norm{\mathbf{e}_1}_{\infty}^2+2\norm{\mathbf{e}_2}_{\infty}^2=4$.\\
Both norms violate Theorem \ref{thm:parallelogram-identity-inner-product}, which every inner-product norm must satisfy.\\
\hlt{Structural Commentary}: The parallelogram identity detects inner-product norms.\\
\hlt{Key Insight}: Different unit-ball geometry can rule out hidden inner-product structure.
\end{proof}
\end{corollary}


\begin{theorem} \label{thm:polarization-identities} \hlt{Polarization Identities}\\
Let $\mathcal{V}$ be an inner-product space and let $\mathbf{u},\mathbf{v}\in\mathcal{V}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item If $\mathbb{F}=\mathbb{R}$, then $\innerprod{\mathbf{u},\mathbf{v}} =\frac{1}{4}\left(\norm{\mathbf{u}+\mathbf{v}}^2-\norm{\mathbf{u}-\mathbf{v}}^2\right)$.
\item If $\mathbb{F}=\mathbb{C}$, then $\innerprod{\mathbf{u},\mathbf{v}} =\frac{1}{4}\left(\norm{\mathbf{u}+\mathbf{v}}^2-\norm{\mathbf{u}-\mathbf{v}}^2 +i\norm{\mathbf{u}+i\mathbf{v}}^2-i\norm{\mathbf{u}-i\mathbf{v}}^2\right)$.
\end{enumerate}
\begin{proof}
Let $z\defeq\innerprod{\mathbf{u},\mathbf{v}}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathbb{F}=\mathbb{R}$, then expanding squared norms gives $\norm{\mathbf{u}+\mathbf{v}}^2-\norm{\mathbf{u}-\mathbf{v}}^2=4\operatorname{Re} z=4z$, proving (i).
\item If $\mathbb{F}=\mathbb{C}$, the first expansion gives $\norm{\mathbf{u}+\mathbf{v}}^2-\norm{\mathbf{u}-\mathbf{v}}^2=4\operatorname{Re} z$. The same expansion with $i\mathbf{v}$ gives $\norm{\mathbf{u}+i\mathbf{v}}^2-\norm{\mathbf{u}-i\mathbf{v}}^2=4\operatorname{Im} z$. Combining the real and imaginary parts gives (ii).
\end{enumerate}
\hlt{Structural Commentary}: Two norm differences extract the real and imaginary parts.\\
\hlt{Key Insight}: The norm determines the inner product because it determines both parts of every cross term.
\end{proof}
\end{theorem}

\begin{principle} \label{pri:inner-product-geometry} \hlt{Geometry from Algebra}\\
Most geometric statements in an inner-product space come from expanding $\norm{\mathbf{u}\pm\mathbf{v}}^2$ and then interpreting the cross terms $\innerprod{\mathbf{u},\mathbf{v}}$ and $\innerprod{\mathbf{v},\mathbf{u}}$.
\end{principle}

\begin{definition} \label{def:unit-vector-and-normalization} \hlt{Unit Vector and Normalization}\\
A vector $\mathbf{u}\in\mathcal{V}$ is a \hlt{unit vector} if $\norm{\mathbf{u}}=1$. If $\mathbf{v}\neq\mathbf{0}$, then $\frac{\mathbf{v}}{\norm{\mathbf{v}}}$ is called the \hlt{normalization} of $\mathbf{v}$.
\end{definition}

\begin{proposition} \label{prop:normalization-gives-unit-vector} \hlt{Normalization Gives a Unit Vector}\\
If $\mathbf{v}\in\mathcal{V}$ and $\mathbf{v}\neq\mathbf{0}$, then $\frac{\mathbf{v}}{\norm{\mathbf{v}}}$ is a unit vector in the same direction as $\mathbf{v}$.
\begin{proof}
By Proposition \ref{prop:derived-norm-basic-properties}, $\norm{\mathbf{v}}>0$ and $\norm{\frac{\mathbf{v}}{\norm{\mathbf{v}}}} =\frac{1}{\norm{\mathbf{v}}}\norm{\mathbf{v}}=1$.\\
The vector $\frac{\mathbf{v}}{\norm{\mathbf{v}}}$ is a positive scalar multiple of $\mathbf{v}$, so it lies in the same one-dimensional direction.
\hlt{Structural Commentary}: Normalization changes scale but preserves direction.\\
\hlt{Key Insight}: Divide by the length to make the length equal to one.
\end{proof}
\end{proposition}

\begin{definition} \label{def:orthogonal-vectors} \hlt{Orthogonal Vectors}\\
Vectors $\mathbf{u},\mathbf{v}\in\mathcal{V}$ are \hlt{orthogonal}, written $\mathbf{u}\perp\mathbf{v}$, if $\innerprod{\mathbf{u},\mathbf{v}}=0$.
\end{definition}

\begin{proposition} \label{prop:orthogonality-and-zero} \hlt{Orthogonality and the Zero Vector}\\
Let $\mathcal{V}$ be an inner-product space.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathbf{u}\perp\mathbf{v}$ if and only if $\mathbf{v}\perp\mathbf{u}$.
\item $\mathbf{0}$ is orthogonal to every vector in $\mathcal{V}$.
\item If $\mathbf{v}$ is orthogonal to itself, then $\mathbf{v}=\mathbf{0}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\innerprod{\mathbf{u},\mathbf{v}}=0$, then conjugate symmetry gives $\innerprod{\mathbf{v},\mathbf{u}}=\overline{0}=0$, and the converse is identical.
\item This follows from Proposition \ref{prop:basic-inner-product-identities}, which gives $\innerprod{\mathbf{0},\mathbf{u}}=0$ and $\innerprod{\mathbf{u},\mathbf{0}}=0$.
\item If $\mathbf{v}\perp\mathbf{v}$, then $\innerprod{\mathbf{v},\mathbf{v}}=0$, hence $\mathbf{v}=\mathbf{0}$ by definiteness.
\end{enumerate}
\hlt{Structural Commentary}: Conjugate symmetry makes orthogonality symmetric, and definiteness isolates $\mathbf{0}$.\\
\hlt{Key Insight}: Conjugate symmetry preserves the equation ``equals zero.''
\end{proof}
\end{proposition}

\begin{definition} \label{def:orthogonal-complement} \hlt{Orthogonal Complement}\\
If $\mathcal{U}\subseteq\mathcal{V}$, then the \hlt{orthogonal complement} of $\mathcal{U}$ is $\mathcal{U}^{\perp}\defeq\{\mathbf{v}\in\mathcal{V}:\innerprod{\mathbf{v},\mathbf{u}}=0\text{ for every }\mathbf{u}\in\mathcal{U}\}$.
\end{definition}

\begin{proposition} \label{prop:basic-properties-orthogonal-complement} \hlt{Basic Properties of Orthogonal Complements}\\
For subsets $\mathcal{U},\mathcal{W}\subseteq\mathcal{V}$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{U}^{\perp}$ is a subspace of $\mathcal{V}$;
\item $\{\mathbf{0}\}^{\perp}=\mathcal{V}$;
\item $\mathcal{V}^{\perp}=\{\mathbf{0}\}$;
\item $\mathcal{U}\cap\mathcal{U}^{\perp}\subseteq\{\mathbf{0}\}$;
\item if $\mathcal{U}\subseteq\mathcal{W}$, then $\mathcal{W}^{\perp}\subseteq\mathcal{U}^{\perp}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item We have $\mathbf{0}\in\mathcal{U}^{\perp}$. If $\mathbf{v},\mathbf{w}\in\mathcal{U}^{\perp}$ and $a\in\mathbb{F}$, then for every $\mathbf{u}\in\mathcal{U}$, $\innerprod{a\mathbf{v}+\mathbf{w},\mathbf{u}}=a\innerprod{\mathbf{v},\mathbf{u}}+\innerprod{\mathbf{w},\mathbf{u}}=0$.
\item For every $\mathbf{v}\in\mathcal{V}$, $\innerprod{\mathbf{v},\mathbf{0}}=0$, so $\mathbf{v}\in\{\mathbf{0}\}^{\perp}$.
\item If $\mathbf{v}\in\mathcal{V}^{\perp}$, then $\innerprod{\mathbf{v},\mathbf{v}}=0$, so $\mathbf{v}=\mathbf{0}$.
\item If $\mathbf{v}\in\mathcal{U}\cap\mathcal{U}^{\perp}$, then $\innerprod{\mathbf{v},\mathbf{v}}=0$, so $\mathbf{v}=\mathbf{0}$.
\item If $\mathbf{v}\in\mathcal{W}^{\perp}$, then $\mathbf{v}$ is orthogonal to every vector in $\mathcal{W}$, hence to every vector in $\mathcal{U}$.
\end{enumerate}
\hlt{Structural Commentary}: Orthogonal complements are solution spaces of homogeneous inner-product equations. Larger input sets impose more equations, so their complements become smaller.\\
\hlt{Key Insight}: Orthogonality constraints are linear constraints.
\end{proof}
\end{proposition}

\begin{definition} \label{def:orthonormal-list} \hlt{Orthonormal List}\\
A list $(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ of vectors in an inner-product space $\mathcal{V}$ is \hlt{orthonormal} if
\begin{align}
\innerprod{\mathbf{e}_j,\mathbf{e}_k}=
\begin{cases}
1, & j=k,\\
0, & j\neq k.
\end{cases}
\nonumber
\end{align}
Equivalently, every $\mathbf{e}_j$ has norm $1$ and distinct vectors in the list are orthogonal.
\end{definition}

\begin{proposition} \label{prop:orthonormal-linear-combination-norm} \hlt{Norm of an Orthonormal Linear Combination}\\
If $(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ is an orthonormal list in $\mathcal{V}$, then $\norm{a_1\mathbf{e}_1+\cdots+a_m\mathbf{e}_m}^2=\abs{a_1}^2+\cdots+\abs{a_m}^2$ for all $a_1,\ldots,a_m\in\mathbb{F}$.
\begin{proof}
Using orthonormality, $\norm{\sum\limits_{j=1}^{m}a_j\mathbf{e}_j}^2 =\innerprod{\sum\limits_{j=1}^{m}a_j\mathbf{e}_j,\sum\limits_{k=1}^{m}a_k\mathbf{e}_k} =\sum\limits_{j=1}^{m}\sum\limits_{k=1}^{m}a_j\overline{a_k}\innerprod{\mathbf{e}_j,\mathbf{e}_k} =\sum\limits_{j=1}^{m}\abs{a_j}^2$.\\
\hlt{Structural Commentary}: Orthonormality deletes every cross term in the squared-norm expansion.\\
\hlt{Key Insight}: In orthonormal coordinates, length is computed coefficient-by-coefficient.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:orthonormal-list-linearly-independent} \hlt{Orthonormal Lists Are Linearly Independent}\\
Every orthonormal list of vectors in an inner-product space is linearly independent.
\begin{proof}
If $a_1\mathbf{e}_1+\cdots+a_m\mathbf{e}_m=\mathbf{0}$, then Proposition \ref{prop:orthonormal-linear-combination-norm} gives $0=\norm{\mathbf{0}}^2=\norm{a_1\mathbf{e}_1+\cdots+a_m\mathbf{e}_m}^2=\sum\limits_{j=1}^{m}\abs{a_j}^2$.\\
Hence each $a_j=0$, so the list is linearly independent.\\
\hlt{Structural Commentary}: The norm identity converts a linear relation into a sum of nonnegative scalars.\\
\hlt{Key Insight}: A zero orthonormal linear combination can only have zero coefficients.
\end{proof}
\end{corollary}

\begin{definition} \label{def:orthonormal-basis} \hlt{Orthonormal Basis}\\
An \hlt{orthonormal basis} of $\mathcal{V}$ is a basis of $\mathcal{V}$ that is an orthonormal list.
\end{definition}

\begin{proposition} \label{prop:right-length-orthonormal-list-is-basis} \hlt{Right-Length Orthonormal Lists Are Bases}\\
If $\mathcal{V}$ is finite-dimensional and $(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ is an orthonormal list in $\mathcal{V}$ with $n=\dim(\mathcal{V})$, then $(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ is an orthonormal basis of $\mathcal{V}$.
\begin{proof}
By Corollary \ref{cor:orthonormal-list-linearly-independent}, the list is linearly independent.\\
Since its length is $\dim(\mathcal{V})$, Proposition \ref{prop:dimension-bounds-and-basis-criteria} implies that it is a basis.\\
\hlt{Structural Commentary}: Orthonormality gives independence; the correct length forces spanning.\\
\hlt{Key Insight}: In finite dimensions, $n$ independent vectors in an $n$-dimensional space already form a basis.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:gram-schmidt-process} \hlt{Gram--Schmidt Process}\\
Let $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ be a linearly independent list in an inner-product space $\mathcal{V}$.\\
Define $\mathbf{u}_1=\mathbf{v}_1$, $\mathbf{e}_1=\frac{\mathbf{u}_1}{\norm{\mathbf{u}_1}}$ and, for $2\leq j\leq m$, $\mathbf{u}_j=\mathbf{v}_j-\sum\limits_{k=1}^{j-1}\innerprod{\mathbf{v}_j,\mathbf{e}_k}\mathbf{e}_k$, $\mathbf{e}_j=\frac{\mathbf{u}_j}{\norm{\mathbf{u}_j}}$.\\
Then $(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ is orthonormal and $\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)=\operatorname{span}(\mathbf{e}_1,\ldots,\mathbf{e}_j)$.
\begin{proof}
Proceed by induction. The case $j=1$ is normalization of the nonzero vector $\mathbf{v}_1$. Assume the result holds through $j-1$. Since $\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_{j-1})=\operatorname{span}(\mathbf{e}_1,\ldots,\mathbf{e}_{j-1})$ and the $\mathbf{v}$-list is linearly independent, $\mathbf{u}_j\neq\mathbf{0}$.\\
For $k<j$, $\innerprod{\mathbf{u}_j,\mathbf{e}_k} =\innerprod{\mathbf{v}_j,\mathbf{e}_k} -\sum\limits_{i=1}^{j-1}\innerprod{\mathbf{v}_j,\mathbf{e}_i}\innerprod{\mathbf{e}_i,\mathbf{e}_k} =\innerprod{\mathbf{v}_j,\mathbf{e}_k}-\innerprod{\mathbf{v}_j,\mathbf{e}_k}=0$.\\
Thus $\mathbf{e}_j$ is a unit vector orthogonal to $\mathbf{e}_1,\ldots,\mathbf{e}_{j-1}$.\\
Also $\mathbf{u}_j$ lies in $\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_j)$, while $\mathbf{v}_j=\mathbf{u}_j+\sum\limits_{k=1}^{j-1}\innerprod{\mathbf{v}_j,\mathbf{e}_k}\mathbf{e}_k$ lies in $\operatorname{span}(\mathbf{e}_1,\ldots,\mathbf{e}_j)$. \\
Combining this with the induction hypothesis gives equality of the two spans at step $j$.\\
\hlt{Structural Commentary}: Subtract the previous-span component, then normalize.\\
\hlt{Key Insight}: Gram--Schmidt changes the basis vectors but preserves every initial span.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:gram-schmidt-preserves-orthogonal-lists} \hlt{Gram--Schmidt on Orthogonal Lists}\\
If $(\mathbf{v}_1,\ldots,\mathbf{v}_m)$ is a nonzero mutually orthogonal list, then Gram--Schmidt produces $\mathbf{e}_j=\frac{\mathbf{v}_j}{\norm{\mathbf{v}_j}}$, $j=1,\ldots,m$.\\
In particular, if the original list is orthonormal, Gram--Schmidt leaves it unchanged.
\begin{proof}
Assume $\mathbf{e}_i=\mathbf{v}_i/\norm{\mathbf{v}_i}$ for $i<j$. Since $\mathbf{v}_j\perp\mathbf{v}_i$, we have $\innerprod{\mathbf{v}_j,\mathbf{e}_i}=0$ for every $i<j$.\\
Hence the Gram--Schmidt remainder at step $j$ is $\mathbf{u}_j=\mathbf{v}_j$, so $\mathbf{e}_j=\mathbf{v}_j/\norm{\mathbf{v}_j}$.\\
If the original list is orthonormal, then $\norm{\mathbf{v}_j}=1$ for every $j$.\\
\hlt{Structural Commentary}: Orthogonal input has no previous-span component to subtract.\\
\hlt{Key Insight}: Gram--Schmidt only removes nonorthogonal overlap.
\end{proof}
\end{lemma}

\begin{corollary} \label{cor:finite-dimensional-inner-product-space-has-orthonormal-basis} \hlt{Existence of Orthonormal Bases}\\
Every finite-dimensional inner-product space has an orthonormal basis.
\begin{proof}
Choose any basis of $\mathcal{V}$ and apply Theorem \ref{thm:gram-schmidt-process}.\\
The resulting orthonormal list has the same span as the original basis, hence spans $\mathcal{V}$.\\
\hlt{Structural Commentary}: Gram--Schmidt orthonormalizes a basis without changing its span.\\
\hlt{Key Insight}: Finite-dimensional inner-product spaces always admit geometry-friendly coordinates.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:orthonormal-list-extends-to-basis} \hlt{Extending an Orthonormal List}\\
Every orthonormal list in a finite-dimensional inner-product space can be extended to an orthonormal basis.
\begin{proof}
Let $(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ be orthonormal. It is linearly independent by Corollary \ref{cor:orthonormal-list-linearly-independent}, so extend it to a basis $(\mathbf{e}_1,\ldots,\mathbf{e}_m,\mathbf{v}_{m+1},\ldots,\mathbf{v}_n)$ by Theorem \ref{thm:extending-independent-lists}. Applying Gram--Schmidt to this basis leaves the first $m$ vectors unchanged by Lemma \ref{lem:gram-schmidt-preserves-orthogonal-lists}, and orthonormalizes the rest.\\
\hlt{Structural Commentary}: Extend to a basis, then orthonormalize the new part.\\
\hlt{Key Insight}: Existing orthonormal directions are already cleaned up, so Gram--Schmidt does not move them.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:orthogonal-direct-sum-decomposition} \hlt{Direct Sum with the Orthogonal Complement}\\
If $\mathcal{U}$ is a finite-dimensional subspace of $\mathcal{V}$, then $\mathcal{V}=\mathcal{U}\oplus\mathcal{U}^{\perp}$.
\begin{proof}
Let $(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ be an orthonormal basis of $\mathcal{U}$. For $\mathbf{v}\in\mathcal{V}$, define $\mathbf{u}=\sum\limits_{j=1}^{m}\innerprod{\mathbf{v},\mathbf{e}_j}\mathbf{e}_j$ and set $\mathbf{w}=\mathbf{v}-\mathbf{u}$. Clearly $\mathbf{u}\in\mathcal{U}$. For each $k$, $\innerprod{\mathbf{w},\mathbf{e}_k} =\innerprod{\mathbf{v},\mathbf{e}_k} -\sum\limits_{j=1}^{m}\innerprod{\mathbf{v},\mathbf{e}_j}\innerprod{\mathbf{e}_j,\mathbf{e}_k} =0$.\\
Thus $\mathbf{w}$ is orthogonal to every vector in $\mathcal{U}$, so $\mathbf{w}\in\mathcal{U}^{\perp}$ and $\mathbf{v}=\mathbf{u}+\mathbf{w}$. Hence $\mathcal{V}=\mathcal{U}+\mathcal{U}^{\perp}$. The intersection is $\{\mathbf{0}\}$ by Proposition \ref{prop:basic-properties-orthogonal-complement}, so the sum is direct.\\
\hlt{Structural Commentary}: Each vector splits into a $\mathcal{U}$-component and an orthogonal residual.\\
\hlt{Key Insight}: Projection plus residual gives the canonical direct-sum decomposition.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:dimension-of-orthogonal-complement} \hlt{Dimension of the Orthogonal Complement}\\
If $\mathcal{V}$ is finite-dimensional and $\mathcal{U}$ is a subspace of $\mathcal{V}$, then $\dim(\mathcal{U}^{\perp})=\dim(\mathcal{V})-\dim(\mathcal{U})$.
\begin{proof}
Since $\mathcal{V}$ is finite-dimensional, $\mathcal{U}$ is finite-dimensional by Corollary \ref{cor:finite-dimensional-spaces-and-subspaces-have-bases}. Theorem \ref{thm:orthogonal-direct-sum-decomposition} gives $\mathcal{V}=\mathcal{U}\oplus\mathcal{U}^{\perp}$. Thus Corollary \ref{cor:dimension-direct-sum} gives $\dim(\mathcal{V})=\dim(\mathcal{U})+\dim(\mathcal{U}^{\perp})$.\\
\hlt{Structural Commentary}: Dimensions add across the orthogonal split.\\
\hlt{Key Insight}: Orthogonal complements contain exactly the dimensions not used by $\mathcal{U}$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:double-orthogonal-complement} \hlt{Double Orthogonal Complement}\\
If $\mathcal{U}$ is a finite-dimensional subspace of $\mathcal{V}$, then $\mathcal{U}=(\mathcal{U}^{\perp})^{\perp}$.
\begin{proof}
If $\mathbf{u}\in\mathcal{U}$, then $\mathbf{u}$ is orthogonal to every vector in $\mathcal{U}^{\perp}$, so $\mathcal{U}\subseteq(\mathcal{U}^{\perp})^{\perp}$. Conversely, let $\mathbf{v}\in(\mathcal{U}^{\perp})^{\perp}$. By Theorem \ref{thm:orthogonal-direct-sum-decomposition}, write $\mathbf{v}=\mathbf{u}+\mathbf{w}$ with $\mathbf{u}\in\mathcal{U}$ and $\mathbf{w}\in\mathcal{U}^{\perp}$. Since $\mathcal{U}\subseteq(\mathcal{U}^{\perp})^{\perp}$, we have $\mathbf{w}=\mathbf{v}-\mathbf{u}\in(\mathcal{U}^{\perp})^{\perp}$. Hence $\mathbf{w}\in\mathcal{U}^{\perp}\cap(\mathcal{U}^{\perp})^{\perp}$, so $\mathbf{w}=\mathbf{0}$ and $\mathbf{v}=\mathbf{u}\in\mathcal{U}$.\\
\hlt{Structural Commentary}: The orthogonal split leaves no extra direction beyond $\mathcal{U}$.\\
\hlt{Key Insight}: Taking the orthogonal complement twice recovers the original finite-dimensional subspace.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:orthonormal-basis-coordinates} \hlt{Coordinates in an Orthonormal Basis}\\
Let $\mathcal{V}$ be finite-dimensional and let $\beta=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ be an orthonormal basis of $\mathcal{V}$.\\
Then for every $\mathbf{v}\in\mathcal{V}$, $\mathbf{v}=\sum\limits_{j=1}^{n}\innerprod{\mathbf{v},\mathbf{e}_j}\mathbf{e}_j$, $\norm{\mathbf{v}}^2=\sum\limits_{j=1}^{n}\abs{\innerprod{\mathbf{v},\mathbf{e}_j}}^2$ and
\begin{align}
[\mathbf{v}]_{\beta}=
\begin{bmatrix}
\innerprod{\mathbf{v},\mathbf{e}_1}\\
\vdots\\
\innerprod{\mathbf{v},\mathbf{e}_n}
\end{bmatrix}.
\nonumber
\end{align}
\begin{proof}
Since $\beta$ is a basis, write $\mathbf{v}=\sum\limits_{j=1}^{n}a_j\mathbf{e}_j$. Taking inner product with $\mathbf{e}_k$ gives $\innerprod{\mathbf{v},\mathbf{e}_k}=\sum\limits_{j=1}^{n}a_j\innerprod{\mathbf{e}_j,\mathbf{e}_k}=a_k$.\\
Thus the displayed expansion and coordinate column follow. The norm identity follows from Proposition \ref{prop:orthonormal-linear-combination-norm}.\\
\hlt{Structural Commentary}: An orthonormal basis makes coordinate extraction an inner-product computation.\\
\hlt{Key Insight}: Pairing with $\mathbf{e}_k$ isolates the $k$th coordinate.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:cross-terms} \hlt{Cross Terms}\\
In an expansion such as $\norm{\mathbf{u}+\mathbf{v}}^2$, the terms involving both vectors, $\innerprod{\mathbf{u},\mathbf{v}}$ and $\innerprod{\mathbf{v},\mathbf{u}}$, are called \hlt{cross terms}. The terms $\norm{\mathbf{u}}^2$ and $\norm{\mathbf{v}}^2$ are the pure squared-length terms.
\end{remark}

\begin{theorem} \label{thm:pythagorean-inner-product} \hlt{Pythagorean Theorem}\\
If $\mathbf{u},\mathbf{v}\in\mathcal{V}$ and $\mathbf{u}\perp\mathbf{v}$, then $\norm{\mathbf{u}+\mathbf{v}}^2=\norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2$.
\begin{proof}
Since $\mathbf{u}\perp\mathbf{v}$, Proposition \ref{prop:orthogonality-and-zero} gives $\innerprod{\mathbf{v},\mathbf{u}}=0$. Thus
\begin{align}
\norm{\mathbf{u}+\mathbf{v}}^2 =\innerprod{\mathbf{u}+\mathbf{v},\mathbf{u}+\mathbf{v}}=\innerprod{\mathbf{u},\mathbf{u}}+\innerprod{\mathbf{u},\mathbf{v}}
  +\innerprod{\mathbf{v},\mathbf{u}}+\innerprod{\mathbf{v},\mathbf{v}} =\norm{\mathbf{u}}^2+\norm{\mathbf{v}}^2.
\nonumber
\end{align}
\hlt{Structural Commentary}: Orthogonality cancels the cross terms in the squared-norm expansion.\\
\hlt{Key Insight}: Perpendicular components add in squared length, not in length.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:bessel-inequality-finite-orthonormal-list} \hlt{Bessel's Inequality}\\
Let $(\mathbf{e}_1,\ldots,\mathbf{e}_m)$ be an orthonormal list in an inner-product space $\mathcal{V}$.\\
Then, for every $\mathbf{v}\in\mathcal{V}$, $\sum\limits_{j=1}^{m}\abs{\innerprod{\mathbf{v},\mathbf{e}_j}}^2\leq\norm{\mathbf{v}}^2$.
\begin{proof}
Set $\mathbf{u}=\sum_{j=1}^{m}\innerprod{\mathbf{v},\mathbf{e}_j}\mathbf{e}_j$. For each $k$, orthonormality gives $\innerprod{\mathbf{v}-\mathbf{u},\mathbf{e}_k} =\innerprod{\mathbf{v},\mathbf{e}_k} -\sum\limits_{j=1}^{m}\innerprod{\mathbf{v},\mathbf{e}_j}\innerprod{\mathbf{e}_j,\mathbf{e}_k}=0$.\\
Thus $\mathbf{v}-\mathbf{u}$ is orthogonal to $\mathbf{u}$. By Pythagorean Theorem \ref{thm:pythagorean-inner-product} and Proposition \ref{prop:orthonormal-linear-combination-norm},
\begin{align}
\norm{\mathbf{v}}^2=\norm{\mathbf{u}}^2+\norm{\mathbf{v}-\mathbf{u}}^2
\geq \norm{\mathbf{u}}^2
=\sum_{j=1}^{m}\abs{\innerprod{\mathbf{v},\mathbf{e}_j}}^2.
\nonumber
\end{align}
\hlt{Structural Commentary}: The chosen directions give an orthogonal part of $\mathbf{v}$; the residual is perpendicular.\\
\hlt{Key Insight}: Partial orthonormal coordinates cannot contain more energy than the whole vector.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:orthonormal-sequence-coefficients-vanish} \hlt{Coefficients Against an Orthonormal Sequence Vanish}\\
Let $(\mathbf{e}_1,\mathbf{e}_2,\ldots)$ be an orthonormal sequence in an inner-product space $\mathcal{V}$, meaning every finite sublist is orthonormal. Then, for every $\mathbf{v}\in\mathcal{V}$, $\lim\limits_{n\to\infty}\innerprod{\mathbf{v},\mathbf{e}_n}=0$.
\begin{proof}
Fix $\epsilon>0$. By Theorem \ref{thm:bessel-inequality-finite-orthonormal-list}, every finite set of indices satisfying $\abs{\innerprod{\mathbf{v},\mathbf{e}_j}}\geq\epsilon$ has cardinality at most $\norm{\mathbf{v}}^2/\epsilon^2$. Hence only finitely many $j$ have $\abs{\innerprod{\mathbf{v},\mathbf{e}_j}}\geq\epsilon$. This holds for every $\epsilon>0$, so $\innerprod{\mathbf{v},\mathbf{e}_n}\to0$.\\
\hlt{Structural Commentary}: Bessel bounds the number of large orthonormal coefficients.\\
\hlt{Key Insight}: Infinitely many large coefficients would force infinite energy.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:parseval-identity-finite-dimensional} \hlt{Parseval's Identity}\\
Let $\beta=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ be an orthonormal basis of an inner-product space $\mathcal{V}$.\\
Then, for all $\mathbf{v},\mathbf{w}\in\mathcal{V}$, $\innerprod{\mathbf{v},\mathbf{w}} =\sum\limits_{j=1}^{n}\innerprod{\mathbf{v},\mathbf{e}_j}\overline{\innerprod{\mathbf{w},\mathbf{e}_j}} =\innerprod[\mathbb{F}^n]{[\mathbf{v}]_{\beta},[\mathbf{w}]_{\beta}}$.
\begin{proof}
By Theorem \ref{thm:orthonormal-basis-coordinates}, $\mathbf{v}=\sum_{j=1}^{n}\innerprod{\mathbf{v},\mathbf{e}_j}\mathbf{e}_j$ and $\mathbf{w}=\sum_{k=1}^{n}\innerprod{\mathbf{w},\mathbf{e}_k}\mathbf{e}_k$.\\
Hence $\innerprod{\mathbf{v},\mathbf{w}} =\sum\limits_{j=1}^{n}\sum\limits_{k=1}^{n}\innerprod{\mathbf{v},\mathbf{e}_j}\overline{\innerprod{\mathbf{w},\mathbf{e}_k}}\innerprod{\mathbf{e}_j,\mathbf{e}_k} =\sum\limits_{j=1}^{n}\innerprod{\mathbf{v},\mathbf{e}_j}\overline{\innerprod{\mathbf{w},\mathbf{e}_j}}$.\\
The last expression is exactly the standard inner product of the coordinate columns in $\mathbb{F}^n$.\\
\hlt{Structural Commentary}: Orthonormal coordinates make inner products coordinatewise.\\
\hlt{Key Insight}: Orthonormal coordinates preserve angles and lengths, not just linear combinations.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:matrix-entries-from-orthonormal-codomain} \hlt{Matrix Entries from an Orthonormal Codomain Basis}\\
Let $\mathcal{V}$ and $\mathcal{W}$ be finite-dimensional vector spaces over $\mathbb{F}$, with $\mathcal{W}$ an inner-product space. Let $\beta=(\mathbf{v}_1,\ldots,\mathbf{v}_n)$ be a basis of $\mathcal{V}$ and let $\gamma=(\mathbf{w}_1,\ldots,\mathbf{w}_m)$ be an orthonormal basis of $\mathcal{W}$. If $T\in\LL(\mathcal{V},\mathcal{W})$, then
\begin{align}
[T]_{\gamma\leftarrow\beta}
=\begin{bmatrix}
\innerprod{T\mathbf{v}_1,\mathbf{w}_1} & \cdots & \innerprod{T\mathbf{v}_n,\mathbf{w}_1}\\
\vdots & \ddots & \vdots\\
\innerprod{T\mathbf{v}_1,\mathbf{w}_m} & \cdots & \innerprod{T\mathbf{v}_n,\mathbf{w}_m}
\end{bmatrix}.
\nonumber
\end{align}
Conversely, for each $A\in\operatorname{Mat}(m,n,\mathbb{F})$, there is a unique $T\in\LL(\mathcal{V},\mathcal{W})$ such that $[T]_{\gamma\leftarrow\beta}=A$.
\begin{proof}
The $j$th column of $[T]_{\gamma\leftarrow\beta}$ is $[T\mathbf{v}_j]_{\gamma}$. Since $\gamma$ is orthonormal, Theorem \ref{thm:orthonormal-basis-coordinates} gives
\begin{align}
[T\mathbf{v}_j]_{\gamma}
=\begin{bmatrix}
\innerprod{T\mathbf{v}_j,\mathbf{w}_1}\\
\vdots\\
\innerprod{T\mathbf{v}_j,\mathbf{w}_m}
\end{bmatrix}.
\nonumber
\end{align}
This proves the displayed matrix formula. The converse is exactly the surjectivity and injectivity of the coordinate encoding in Theorem \ref{thm:linear-maps-and-matrices-same-data}.\\
\hlt{Structural Commentary}: Columns are basis-image coordinates, read by inner products.\\
\hlt{Key Insight}: Each entry measures $T\mathbf{v}_j$ against one output direction.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:inner-product-through-operator-matrix} \hlt{Inner Products Through an Operator Matrix}\\
Let $\beta=(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ be an orthonormal basis of a finite-dimensional inner-product space $\mathcal{V}$, and let $T\in\LL(\mathcal{V})$. Then, for all $\mathbf{v},\mathbf{w}\in\mathcal{V}$, $\innerprod{T\mathbf{v},\mathbf{w}} =\innerprod[\mathbb{F}^n]{[T]_{\beta\leftarrow\beta}[\mathbf{v}]_{\beta},[\mathbf{w}]_{\beta}}$.
\begin{proof}
By Proposition \ref{prop:matrix-representation-action}, $[T\mathbf{v}]_{\beta}=[T]_{\beta\leftarrow\beta}[\mathbf{v}]_{\beta}$. Applying Theorem \ref{thm:parseval-identity-finite-dimensional} to $T\mathbf{v}$ and $\mathbf{w}$ gives the result.\\
\hlt{Structural Commentary}: Orthonormal coordinates turn $\innerprod{T\mathbf{v},\mathbf{w}}$ into matrix computation.\\
\hlt{Key Insight}: Orthonormal coordinates make operator geometry computable by matrices.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:riesz-representation-finite-dimensional} \hlt{Riesz Representation Theorem}\\
Let $\mathcal{V}$ be a finite-dimensional inner-product space and let $\varphi:\mathcal{V}\to\mathbb{F}$ be a linear functional. Then there is a unique vector $\mathbf{u}\in\mathcal{V}$ such that $\varphi(\mathbf{v})=\innerprod{\mathbf{v},\mathbf{u}}$ for every $\mathbf{v}\in\mathcal{V}$.
\begin{proof}
Let $(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ be an orthonormal basis of $\mathcal{V}$.\\
By Theorem \ref{thm:orthonormal-basis-coordinates}, $\varphi(\mathbf{v}) =\varphi\left(\sum\limits_{j=1}^{n}\innerprod{\mathbf{v},\mathbf{e}_j}\mathbf{e}_j\right) =\sum\limits_{j=1}^{n}\innerprod{\mathbf{v},\mathbf{e}_j}\varphi(\mathbf{e}_j)$. Set $\mathbf{u}=\sum\limits_{j=1}^{n}\overline{\varphi(\mathbf{e}_j)}\,\mathbf{e}_j$.\\
Then orthonormality gives $\innerprod{\mathbf{v},\mathbf{u}}=\sum_{j=1}^{n}\innerprod{\mathbf{v},\mathbf{e}_j}\varphi(\mathbf{e}_j)=\varphi(\mathbf{v})$.
For uniqueness, suppose $\innerprod{\mathbf{v},\mathbf{u}_1}=\innerprod{\mathbf{v},\mathbf{u}_2}$ for every $\mathbf{v}\in\mathcal{V}$. Taking $\mathbf{v}=\mathbf{u}_1-\mathbf{u}_2$ gives $\norm{\mathbf{u}_1-\mathbf{u}_2}^2=0$, hence $\mathbf{u}_1=\mathbf{u}_2$.\\
\hlt{Structural Commentary}: An orthonormal basis packages a functional as one representing vector.\\
\hlt{Key Insight}: Finite-dimensional functionals are inner products with fixed vectors.
\end{proof}
\end{theorem}

\begin{definition} \label{def:riesz-vector} \hlt{Riesz Vector}\\
Let $\mathcal{V}$ be finite-dimensional and let $\varphi:\mathcal{V}\to\mathbb{F}$ be a linear functional. The unique vector $\mathbf{u}\in\mathcal{V}$ satisfying $\varphi(\mathbf{v})=\innerprod{\mathbf{v},\mathbf{u}}$ for every $\mathbf{v}\in\mathcal{V}$ is the \hlt{Riesz vector} representing $\varphi$.
\end{definition}

The vector $\sum_{j=1}^{n}\overline{\varphi(\mathbf{e}_j)}\mathbf{e}_j$ seems to depend on the orthonormal basis, but Theorem \ref{thm:riesz-representation-finite-dimensional} shows that the resulting representing vector is intrinsic.
\end{perspective}

\begin{example} \label{ex:frobenius-trace-riesz-vector} \hlt{Trace Functional and the Frobenius Inner Product}\\
On $\operatorname{Mat}(n,n,\mathbb{F})$ with the Frobenius inner product, the Riesz vector of the trace functional $A\mapsto\operatorname{tr}(A)$ is $I_n$, since $\innerprod[F]{A,I_n}=\sum\limits_{j=1}^{n}a_{jj}=\operatorname{tr}(A)$.
\end{example}

\begin{definition} \label{def:gram-matrix-and-determinant} \hlt{Gram Matrix}\\
For vectors $\mathbf{u}_1,\ldots,\mathbf{u}_n$ in an inner-product space, the \hlt{Gram matrix} is
\begin{align}
\operatorname{Gram}(\mathbf{u}_1,\ldots,\mathbf{u}_n)
\defeq
\begin{bmatrix}
\innerprod{\mathbf{u}_1,\mathbf{u}_1} & \innerprod{\mathbf{u}_2,\mathbf{u}_1} & \cdots & \innerprod{\mathbf{u}_n,\mathbf{u}_1}\\

The \hlt{Gram determinant} is $\operatorname{GramDet}(\mathbf{u}_1,\ldots,\mathbf{u}_n)\defeq\det \operatorname{Gram}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$.
