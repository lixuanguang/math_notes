\subsubsection{Advanced Matrix Analysis}

This subsection collects matrix identities that organise larger linear systems by blocks, tensor products, vectorisation, and rank bookkeeping.

\begin{definition} \label{def:kronecker-product} \hlt{Kronecker Product}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$ and $B\in\operatorname{Mat}(p,q,\mathbb{F})$. The \hlt{Kronecker product} of $A$ and $B$ is the block matrix
\begin{align}
A\otimes B
\defeq
\begin{bmatrix}
a_{11}B & a_{12}B & \cdots & a_{1n}B\\
a_{21}B & a_{22}B & \cdots & a_{2n}B\\
\vdots & \vdots & \ddots & \vdots\\
a_{m1}B & a_{m2}B & \cdots & a_{mn}B
\end{bmatrix}
\in\operatorname{Mat}(mp,nq,\mathbb{F}).
\nonumber
\end{align}
The Kronecker product is also called the tensor product of matrices.
\end{definition}

\begin{proposition} \label{prop:basic-kronecker-identities} \hlt{Basic Kronecker Identities}\\
Whenever the expressions are defined,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $c(A\otimes B)=(cA)\otimes B=A\otimes(cB)$;
\item $(A+B)\otimes C=A\otimes C+B\otimes C$ and $A\otimes(B+C)=A\otimes B+A\otimes C$;
\item $(A\otimes B)\otimes C=A\otimes(B\otimes C)$;
\item $(A\otimes B)^t=A^t\otimes B^t$;
\item $I_m\otimes I_n=I_{mn}$.
\end{enumerate}
If $\mathbb{F}=\mathbb{C}$, then also $\overline{A\otimes B}=\overline{A}\otimes\overline{B}$ and $(A\otimes B)^*=A^*\otimes B^*$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The $(i,j)$ block of $c(A\otimes B)$ is $c(a_{ij}B)=(ca_{ij})B=a_{ij}(cB)$, which is $(i,j)$ block of $(cA)\otimes B$ and $A\otimes(cB)$.
\item The $(i,j)$ block of $(A+B)\otimes C$ is $(a_{ij}+b_{ij})C=a_{ij}C+b_{ij}C$, so $(A+B)\otimes C=A\otimes C+B\otimes C$.\\
The proof of $A\otimes(B+C)=A\otimes B+A\otimes C$ is identical inside each block.
\item If $B=[b_{rs}]$, then both $(A\otimes B)\otimes C$ and $A\otimes(B\otimes C)$ have blocks $a_{ij}b_{rs}C$ in the same relative positions.
\item The $(j,i)$ block of $(A\otimes B)^t$ is $(a_{ij}B)^t=a_{ij}B^t$, which is the $(j,i)$ block of $A^t\otimes B^t$.
\item The block matrix $I_m\otimes I_n$ has $I_n$ on its diagonal blocks and zero elsewhere, hence equals $I_{mn}$.
\end{enumerate}
If $\mathbb{F}=\mathbb{C}$, conjugation is checked blockwise, adjoint identity follows from conjugation together with part (iv).\\
\hlt{Structural Commentary}: Kronecker products replace scalar entries by whole scaled blocks. Standard algebraic rules survive because each block is still controlled by scalar distributivity and transposition.\\
\hlt{Key Insight}: Check identities block-by-block; each block reduces to an ordinary scalar-matrix identity.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:kronecker-mixed-product-property} \hlt{Mixed Product Property}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $B\in\operatorname{Mat}(p,q,\mathbb{F})$, $C\in\operatorname{Mat}(n,r,\mathbb{F})$, $D\in\operatorname{Mat}(q,s,\mathbb{F})$.  $(A\otimes B)(C\otimes D)=(AC)\otimes(BD)$.
\begin{proof}
Write $A=[a_{ik}]$ and $C=[c_{kj}]$.\\
The $(i,j)$ block of $(A\otimes B)(C\otimes D)$ is $\sum\limits_{k=1}^{n}(a_{ik}B)(c_{kj}D) =\left(\sum\limits_{k=1}^{n}a_{ik}c_{kj}\right)BD =(AC)_{ij}BD$, which is exactly the $(i,j)$ block of $(AC)\otimes(BD)$.\\
\hlt{Structural Commentary}: The outer indices multiply as $AC$, while the inner block action multiplies as $BD$.\\
\hlt{Key Insight}: Kronecker multiplication separates into ordinary multiplication in each factor.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:inverse-of-kronecker-product} \hlt{Inverse of a Kronecker Product}\\
If $A\in\operatorname{Mat}(m,m,\mathbb{F})$ and $B\in\operatorname{Mat}(n,n,\mathbb{F})$ are invertible, then $A\otimes B$ is invertible and $(A\otimes B)^{-1}=A^{-1}\otimes B^{-1}$.
\begin{proof}
By Theorem \ref{thm:kronecker-mixed-product-property}, $(A\otimes B)(A^{-1}\otimes B^{-1})=I_m\otimes I_n=I_{mn}$ and the reverse product is identical.\\
\hlt{Structural Commentary}: Inverting a Kronecker product inverts each independent factor.\\
\hlt{Key Insight}: The mixed product property turns the inverse check into two ordinary inverse checks.
\end{proof}
\end{corollary}

\begin{definition} \label{def:vectorization-map} \hlt{Vectorisation}\\
Let $X=[\mathbf{x}_1\ \mathbf{x}_2\ \cdots\ \mathbf{x}_n]\in\operatorname{Mat}(m,n,\mathbb{F})$. The \hlt{vectorisation} of $X$ is
\begin{align}
\operatorname{vec}(X)\defeq
\begin{bmatrix}
\mathbf{x}_1\\
\mathbf{x}_2\\
\vdots\\
\mathbf{x}_n
\end{bmatrix}
\in\mathbb{F}^{mn}.
\nonumber
\end{align}
Thus $\operatorname{vec}$ stacks the columns of $X$ vertically.
\end{definition}

\begin{theorem} \label{thm:vectorization-of-triple-product} \hlt{Vectorisation of a Triple Product}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $X\in\operatorname{Mat}(n,p,\mathbb{F})$, and $B\in\operatorname{Mat}(p,q,\mathbb{F})$. Then $\operatorname{vec}(AXB)=(B^t\otimes A)\operatorname{vec}(X)$.
\begin{proof}
Write $B=[\mathbf{b}_1\ \cdots\ \mathbf{b}_q]$ and $X=[\mathbf{x}_1\ \cdots\ \mathbf{x}_p]$.\\
The $k$th column of $AXB$ is $AX\mathbf{b}_k=A\sum\limits_{i=1}^{p}b_{ik}\mathbf{x}_i =\sum\limits_{i=1}^{p}b_{ik}A\mathbf{x}_i =(\mathbf{b}_k^t\otimes A)\operatorname{vec}(X)$.\\
Stacking these $q$ columns gives $(B^t\otimes A)\operatorname{vec}(X)$.\\
\hlt{Structural Commentary}: Vectorisation rewrites two-sided multiplication as one matrix action.\\
\hlt{Key Insight}: Left multiplication contributes $A$; right multiplication contributes $B^t$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:hadamard-product} \hlt{Hadamard Product}\\
Let $A=[a_{ij}],B=[b_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$.\\
The \hlt{Hadamard product}, or \hlt{entrywise product}, of $A$ and $B$ is $A\circ B\defeq [a_{ij}b_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$.
\end{definition}

\begin{proposition} \label{prop:basic-hadamard-identities} \hlt{Basic Hadamard Identities}\\
For matrices of compatible sizes:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\circ B=B\circ A$;
\item $A\circ(\gamma B+C)=\gamma(A\circ B)+A\circ C$;
\item $(A\circ B)^*=A^*\circ B^*$;
\item if $E$ is the all-ones matrix, then $A\circ E=A$;
\item if $D=\operatorname{diag}(d_1,\ldots,d_n)$, then $A\circ D=\operatorname{diag}(a_{11}d_1,\ldots,a_{nn}d_n)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(A\circ B)_{ij}=a_{ij}b_{ij}=b_{ij}a_{ij}=(B\circ A)_{ij}$.
\item $\bigl(A\circ(\gamma B+C)\bigr)_{ij}=a_{ij}(\gamma b_{ij}+c_{ij})=\gamma a_{ij}b_{ij}+a_{ij}c_{ij}$.
\item $\bigl((A\circ B)^*\bigr)_{ij}=\overline{a_{ji}b_{ji}}=\overline{a_{ji}}\,\overline{b_{ji}}=(A^*\circ B^*)_{ij}$.
\item $(A\circ E)_{ij}=a_{ij}$.
\item Since $D$ has zero off-diagonal entries, $A\circ D$ is diagonal with entries $a_{ii}d_i$.
\end{enumerate}
\hlt{Structural Commentary}: Hadamard multiplication is entrywise, with no row-column summation.\\
\hlt{Key Insight}: Prove Hadamard identities entry by entry.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:schur-product-theorem} \hlt{Schur Product Theorem}\\
If $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$ are positive semidefinite, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, then $A\circ B$ is positive semidefinite.
\begin{proof}
By Theorem \ref{thm:positive-semidefinite-matrix-equivalences}, regard $A$ and $B$ as Gram matrices: $a_{ij}=\innerprod{\mathbf{u}_j,\mathbf{u}_i}$, $b_{ij}=\innerprod{\mathbf{v}_j,\mathbf{v}_i}$ for suitable vectors $\mathbf{u}_1,\ldots,\mathbf{u}_n$ and $\mathbf{v}_1,\ldots,\mathbf{v}_n$. Then $(A\circ B)_{ij} =\innerprod{\mathbf{u}_j,\mathbf{u}_i}\innerprod{\mathbf{v}_j,\mathbf{v}_i} =\innerprod{\mathbf{u}_j\otimes\mathbf{v}_j,\mathbf{u}_i\otimes\mathbf{v}_i}$.\\
Thus $A\circ B$ is the Gram matrix of the vectors $\mathbf{u}_j\otimes\mathbf{v}_j$, hence is positive semidefinite by Proposition \ref{prop:gram-matrix-positive-semidefinite}.\\
\hlt{Structural Commentary}: The Schur product is the Gram matrix of paired tensor vectors.\\
\hlt{Key Insight}: Pair the $j$th vector from each Gram representation and take tensor products.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:hadamard-positive-definite-criterion} \hlt{Positive Definite Hadamard Products}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$ be positive semidefinite, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A$ is positive definite and every diagonal entry of $B$ is positive, then $A\circ B$ is positive definite.
\item If $A$ and $B$ are positive definite, then $A\circ B$ is positive definite.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\lambda_{\min}>0$ be the smallest eigenvalue of $A$. Then $A-\lambda_{\min}I_n\succeq0$, so $(A-\lambda_{\min}I_n)\circ B\succeq0$ by Theorem \ref{thm:schur-product-theorem}. Thus, for $\mathbf{x}\neq\mathbf{0}$, $\mathbf{x}^*(A\circ B)\mathbf{x} =\mathbf{x}^*((A-\lambda_{\min}I_n)\circ B)\mathbf{x} \lambda_{\min}\sum\limits_{j=1}^{n}b_{jj}\abs{x_j}^2>0$.
\item If $B\succ0$, then $b_{jj}=\mathbf{e}_j^*B\mathbf{e}_j>0$ for every $j$, so (i) applies.
\end{enumerate}
\hlt{Structural Commentary}: Subtract the spectral floor of $A$; the diagonal of $B$ gives strictness.\\
\hlt{Key Insight}: Use $I_n\circ B=\operatorname{diag}(b_{11},\ldots,b_{nn})$.
\end{proof}
\end{corollary}

\begin{method} \label{meth:schur-product-positivity} \hlt{Schur Product Positivity Method}\\
To prove an entrywise product is positive semidefinite:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item factor or interpret each matrix as a Gram matrix;
\item pair the corresponding vectors;
\item take tensor products of the paired vectors;
\item identify the Hadamard product as the Gram matrix of those tensor products.
\end{enumerate}
\end{method}

\begin{theorem} \label{thm:sylvester-linear-matrix-equation} \hlt{Sylvester Linear Matrix Equation}\\
Let $A\in\operatorname{Mat}(m,m,\mathbb{C})$ and $B\in\operatorname{Mat}(n,n,\mathbb{C})$. If $\operatorname{spec}(A)\cap\operatorname{spec}(B)=\emptyset$, then for every $C\in\operatorname{Mat}(m,n,\mathbb{C})$ the equation $AX-XB=C$ has a unique solution $X\in\operatorname{Mat}(m,n,\mathbb{C})$.
\begin{proof}
Let $L(X)=AX-XB$. Since $L$ has equal finite-dimensional domain and codomain, it suffices to show $\ker L=\{0\}$. If $AX=XB$, then $p(A)X=Xp(B)$ for every polynomial $p$. \\
With $p=p_B$, Cayley--Hamilton gives $p_B(A)X=Xp_B(B)=0$. For every $\lambda\in\operatorname{spec}(A)$, disjointness gives $p_B(\lambda)\neq0$; hence $p_B(A)$ is invertible by Corollary \ref{cor:polynomial-in-matrix-invertible-spectrum-test}. Thus $X=0$, so $L$ is bijective.\\
\hlt{Structural Commentary}: Polynomial calculus pass through homogeneous solutions; disjoint spectra invert $p_B(A)$.\\
\hlt{Key Insight}: Spectral separation turns an intertwining relation into the zero relation.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:block-upper-triangular-remove-disjoint-coupling} \hlt{Removing a Spectrally Separated Off-Diagonal Block}\\
Let $M=\begin{bmatrix}A&C\\0&B\end{bmatrix}$ and $\operatorname{spec}(A)\cap\operatorname{spec}(B)=\emptyset$ Then $M$ is similar to $A\oplus B$.
\begin{proof}
By Theorem \ref{thm:sylvester-linear-matrix-equation}, choose $X$ such that $AX-XB=-C$. Then
\begin{align}
\begin{bmatrix}I&-X\\0&I\end{bmatrix}
\begin{bmatrix}A&C\\0&B\end{bmatrix}
\begin{bmatrix}I&X\\0&I\end{bmatrix}
=
\begin{bmatrix}A&AX+C-XB\\0&B\end{bmatrix}
=
\begin{bmatrix}A&0\\0&B\end{bmatrix}.
\nonumber
\end{align}
\hlt{Structural Commentary}: A block shear cancels the off-diagonal block via Sylvester's equation.\\
\hlt{Key Insight}: Spectrally separated diagonal blocks cannot carry essential upper-right coupling.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:schur-block-form-by-eigenvalue-groups} \hlt{Schur Form Grouped by Eigenvalues}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have distinct eigenvalues $\lambda_1,\ldots,\lambda_d$ with algebraic multiplicities $n_1,\ldots,n_d$.\\
Then $A$ is unitarily similar to a block upper-triangular matrix
\begin{align}
T=
\begin{bmatrix}
T_{11}&T_{12}&\cdots&T_{1d}\\
0&T_{22}&\cdots&T_{2d}\\
\vdots&\vdots&\ddots&\vdots\\
0&0&\cdots&T_{dd}
\end{bmatrix},
\qquad
T_{ii}\in\operatorname{Mat}(n_i,n_i,\mathbb{C}),
\nonumber
\end{align}
where each $T_{ii}$ is upper triangular with all diagonal entries equal to $\lambda_i$. Moreover, $A$ is similar to $T_{11}\oplus\cdots\oplus T_{dd}$.
\begin{proof}
Order the Schur diagonal as $\lambda_1,\ldots,\lambda_1,\lambda_2,\ldots,\lambda_2,\ldots,\lambda_d,\ldots,\lambda_d$.\\
Theorem \ref{thm:matrix-schur-triangularization-ordered} gives the displayed block Schur form. Since $\operatorname{spec}(T_{11})$ is disjoint from the trailing block spectrum, Corollary \ref{cor:block-upper-triangular-remove-disjoint-coupling} splits off $T_{11}$. Iterating splits off every diagonal block.\\
\hlt{Structural Commentary}: Grouped Schur form isolates spectral blocks; Sylvester shears split disjoint groups.\\
\hlt{Key Insight}: Cross-group upper blocks are coordinate artefacts.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:minimal-polynomial-direct-sum-disjoint-spectra} \hlt{Minimal Polynomial of a Spectrally Disjoint Direct Sum}\\
Let $A=A_1\oplus\cdots\oplus A_d$, where the spectra of $A_1,\ldots,A_d$ are pairwise disjoint. Then $\mu_A=\mu_{A_1}\mu_{A_2}\cdots\mu_{A_d}$.
\begin{proof}
Let $f=\prod\limits_i\mu_{A_i}$. Since $f(A_i)=0$ for all $i$, $f(A)=0$ and $\mu_A\mid f$. Conversely, $\mu_A(A_i)=0$, so $\mu_{A_i}\mid\mu_A$ for each $i$. Disjoint spectra make the $\mu_{A_i}$ pairwise relatively prime, hence $f\mid\mu_A$. Thus $\mu_A=f$.\\
\hlt{Structural Commentary}: Direct sums are annihilated blockwise; disjoint spectra make the block factors coprime.\\
\hlt{Key Insight}: Each spectral component contributes its own minimal-polynomial factor.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:commuting-with-disjoint-blocks-has-zero-off-diagonal-blocks} \hlt{Commuting with Disjoint Blocks}\\
Let $A=A_{11}\oplus\cdots\oplus A_{dd}$, where the spectra of $A_{11},\ldots,A_{dd}$ are pairwise disjoint.\\
Partition $B=[B_{ij}]$ conformally with $A$. If $AB=BA$, then $B_{ij}=0$ for all $i\neq j$.
\begin{proof}
Equating $(i,j)$ blocks in $AB=BA$ gives $A_{ii}B_{ij}-B_{ij}A_{jj}=0$.\\
For $i\neq j$, the spectra of $A_{ii}$ and $A_{jj}$ are disjoint, so Theorem \ref{thm:sylvester-linear-matrix-equation} with $C=0$ gives $B_{ij}=0$.\\
\hlt{Structural Commentary}: A commuting matrix cannot mix direct-sum blocks with disjoint spectra.\\
\hlt{Key Insight}: The off-diagonal block solves a homogeneous Sylvester equation, whose only solution is zero.
\end{proof}
\end{corollary}

\begin{definition} \label{def:nonderogatory-matrix} \hlt{Nonderogatory Matrix}\\
A matrix $A\in\operatorname{Mat}(n,n,\mathbb{C})$ is \hlt{nonderogatory} if its Jordan form has exactly one Jordan block for each distinct eigenvalue. Equivalently, $\dim E_{\lambda}(A)=1$ for every $\lambda\in\operatorname{spec}(A)$.
\end{definition}

\begin{lemma} \label{lem:single-jordan-block-commutant-polynomials} \hlt{Commutant of One Jordan Block}\\
If $J=J_m(\lambda)$ and $XJ=JX$, then $X=p(J)$ for some polynomial $p$ with $\deg p<m$.
\begin{proof}
It suffices to treat $N=J_m(0)$, since $J_m(\lambda)=\lambda I_m+N$.\\
The vector $\mathbf{e}_m$ is cyclic for $N$: $(\mathbf{e}_m,N\mathbf{e}_m,\ldots,N^{m-1}\mathbf{e}_m)$ is the standard basis in reverse order.\\
Choose the unique polynomial $p$ with $\deg p<m$ such that $p(N)\mathbf{e}_m=X\mathbf{e}_m$.\\
For every $r=0,\ldots,m-1$, $p(N)N^r\mathbf{e}_m=N^rp(N)\mathbf{e}_m=N^rX\mathbf{e}_m=XN^r\mathbf{e}_m$.\\
Thus $p(N)$ and $X$ agree on a basis, so $X=p(N)=q(J)$ for $q(z)=p(z-\lambda)$.\\
\hlt{Structural Commentary}: A single Jordan block is generated by one cyclic vector.\\
\hlt{Key Insight}: The cyclic vector determines the commutant.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:nonderogatory-commutant-polynomials} \hlt{Nonderogatory Commutants Are Polynomials}\\
If $A\in\operatorname{Mat}(n,n,\mathbb{C})$ is nonderogatory, then $\{A\}'=\{p(A):p\in\mathcal{P}(\mathbb{C})\}$.
\begin{proof}
The inclusion $\{p(A):p\in\mathcal{P}(\mathbb{C})\}\subseteq\{A\}'$ is immediate.\\
For the reverse inclusion, replace $A$ by its Jordan form $J=J_{m_1}(\lambda_1)\oplus\cdots\oplus J_{m_d}(\lambda_d)$, where the $\lambda_i$ are distinct. If $XJ=JX$ and $X=[X_{ij}]$ is partitioned conformally with $J$, Corollary \ref{cor:commuting-with-disjoint-blocks-has-zero-off-diagonal-blocks} gives $X_{ij}=0$ for $i\neq j$. Each diagonal block $X_{ii}$ commutes with $J_{m_i}(\lambda_i)$, so Lemma \ref{lem:single-jordan-block-commutant-polynomials} gives $X_{ii}=p_i(J_{m_i}(\lambda_i))$ for some $\deg p_i<m_i$.\\
Because the polynomials $(z-\lambda_i)^{m_i}$ are pairwise coprime, Hermite interpolation gives a polynomial $p$ such that $p\equiv p_i\pmod{(z-\lambda_i)^{m_i}}$ for every $i$. Then $X=p(J)$. Conjugating back to $A$ proves the claim.\\
\hlt{Structural Commentary}: Disjoint eigenvalue blocks force block diagonality; one Jordan block per eigenvalue makes each diagonal commutant polynomial.\\
\hlt{Key Insight}: Hermite interpolation glues the separate block polynomials into one polynomial in $A$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:simultaneous-schur-triangularization-commuting-family} \hlt{Simultaneous Schur Triangularisation}\\
Let $\emptyset\neq\mathcal{F}\subseteq\operatorname{Mat}(n,n,\mathbb{C})$ be a commuting family.\\
Then there is a unitary $U\in\operatorname{Mat}(n,n,\mathbb{C})$ such that $U^*AU$ is upper triangular for every $A\in\mathcal{F}$.\\
If every matrix in $\mathcal{F}$ is real and has only real eigenvalues, then $U$ can be chosen real orthogonal.
\begin{proof}
Induct on $n$. Choose common unit eigenvector $\mathbf{x}$, extend to orthonormal basis, let $V=[\mathbf{x}\ V_2]$.\\
For every $A\in\mathcal{F}$, $V^*AV=
\begin{bmatrix}
\lambda_A&*\\
0&A'
\end{bmatrix}$. The compressed blocks still commute, so induction triangularises them simultaneously; adjoining $\mathbf{x}$ gives the result. In the real case, use the same induction with real orthonormal extensions. A real common eigenvector exists because either all matrices are scalar, or a nonscalar $A_0\in\mathcal{F}$ has a proper real eigenspace invariant under all of $\mathcal{F}$.\\
\hlt{Structural Commentary}: A common eigenline creates the first block; commutation descends to the compression.\\
\hlt{Key Insight}: Simultaneous triangularisation is Schur induction with a common eigenvector at every step.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:commuting-sum-product-eigenvalues} \hlt{Eigenvalues of Sums and Products of Commuting Matrices}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{C})$ and $AB=BA$. Their eigenvalues can be ordered as $\lambda_1,\ldots,\lambda_n$ and $\mu_1,\ldots,\mu_n$ so that $A+B$ has eigenvalues $\lambda_i+\mu_i$ and $AB$ has eigenvalues $\lambda_i\mu_i$, counted with multiplicity.
\begin{proof}
Choose simultaneous Schur forms $T=U^*AU$ and $R=U^*BU$. Set $\lambda_i=t_{ii}$ and $\mu_i=r_{ii}$. Then $T+R$ and $TR$ are upper triangular with diagonals $\lambda_i+\mu_i$ and $\lambda_i\mu_i$, and are similar to $A+B$ and $AB$.\\
\hlt{Structural Commentary}: Simultaneous Schur form aligns both spectra on one diagonal.\\
\hlt{Key Insight}: In common triangular coordinates, diagonals add and multiply.
\end{proof}
\end{corollary}

\begin{lemma} \label{lem:adjoints-of-finitely-many-normal-matrices-polynomial} \hlt{Adjoints of Normal Matrices as One Polynomial}\\
Let $A_1,\ldots,A_k$ be normal complex matrices. Then $\exists$ a polynomial $p\in\mathcal{P}(\mathbb{C})$ s.t. $p(A_j)=A_j^* \ \forall j=1,\ldots,k$.
\begin{proof}
Let $\Omega=\bigcup\limits_{j=1}^{k}\operatorname{spec}(A_j)$. Since $\Omega$ is finite, choose a Lagrange interpolation polynomial $p$ satisfying $p(\lambda)=\overline{\lambda}$ for every $\lambda\in\Omega$. For each $A_j$, Corollary \ref{cor:normal-spectral-resolution-functional-calculus} gives $p(A_j)=A_j^*$.\\
\hlt{Structural Commentary}: One polynomial interpolates conjugation on all spectra.\\
\hlt{Key Insight}: For normal matrices, $A^*$ is $p(A)$ with $p(\lambda)=\overline{\lambda}$.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:fuglede-putnam-normal-intertwining} \hlt{Fuglede--Putnam Theorem}\\
Let $A\in\operatorname{Mat}(m,m,\mathbb{C})$ and $B\in\operatorname{Mat}(n,n,\mathbb{C})$ be normal, and let $X\in\operatorname{Mat}(m,n,\mathbb{C})$.\\
Then $AX=XB$ $\Longleftrightarrow$ $A^*X=XB^*$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Assume $AX=XB$. Then $A^rX=XB^r$ for every $r\geq0$, so $p(A)X=Xp(B)$ for every polynomial $p$. By Lemma \ref{lem:adjoints-of-finitely-many-normal-matrices-polynomial}, choose $p$ with $p(A)=A^*$ and $p(B)=B^*$. Thus $A^*X=XB^*$.
\item[$(\Leftarrow)$] Assume $A^*X=XB^*$. Apply the same argument to the normal matrices $A^*$ and $B^*$, choosing $q$ with $q(A^*)=A$ and $q(B^*)=B$. Hence $AX=XB$.
\end{enumerate}
\hlt{Structural Commentary}: An intertwining relation passes through every polynomial in the two normal matrices.\\
\hlt{Key Insight}: Normality makes adjoints polynomial, so intertwining transfers to adjoints.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:brauer-rank-one-eigenvalue-shift} \hlt{Brauer Rank-One Eigenvalue Shift}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$, let $A\mathbf{x}=\lambda\mathbf{x}$ with $\mathbf{x}\neq\mathbf{0}$, and let $\mathbf{y}\in\mathbb{C}^n$. If $\lambda,\lambda_2,\ldots,\lambda_n$ are the eigenvalues of $A$ with multiplicity, then the eigenvalues of $A+\mathbf{x}\mathbf{y}^*$ are $\lambda+\mathbf{y}^*\mathbf{x},\lambda_2,\ldots,\lambda_n$.\\
Moreover, $\mathbf{x}$ is an eigenvector of $A+\mathbf{x}\mathbf{y}^*$ corresponding to $\lambda+\mathbf{y}^*\mathbf{x}$.
\begin{proof}
Put $\mathbf{u}=\mathbf{x}/\norm{\mathbf{x}}_2$. By Schur triangularisation, choose a unitary $U=[\mathbf{u}\ U_2]$ such that $U^*AU=\begin{bmatrix}\lambda&*\\0&T'\end{bmatrix}$, where the diagonal entries of $T'$ are $\lambda_2,\ldots,\lambda_n$. Also $U^*(\mathbf{x}\mathbf{y}^*)U
=
\begin{bmatrix}
\mathbf{y}^*\mathbf{x}&*\\
0&0
\end{bmatrix}$.
Therefore $U^*(A+\mathbf{x}\mathbf{y}^*)U$ is upper triangular with diagonal entries $\lambda+\mathbf{y}^*\mathbf{x},\lambda_2,\ldots,\lambda_n$. Finally, $(A+\mathbf{x}\mathbf{y}^*)\mathbf{x}=A\mathbf{x}+\mathbf{x}(\mathbf{y}^*\mathbf{x})=(\lambda+\mathbf{y}^*\mathbf{x})\mathbf{x}$.
\hlt{Structural Commentary}: A rank-one eigenvector update changes only the first Schur diagonal entry.\\
\hlt{Key Insight}: Align the update with the first coordinate; only one diagonal term shifts.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:damped-rank-one-eigenvalue-reset} \hlt{Damped Rank-One Eigenvalue Reset}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ have eigenpair $(\lambda,\mathbf{x})$, let $\mathbf{y}^*\mathbf{x}=1$, and let $\tau\in\mathbb{C}$. If $\lambda,\lambda_2,\ldots,\lambda_n$ are the eigenvalues of $A$, then $A_{\tau}\defeq \tau A+(1-\tau)\lambda\mathbf{x}\mathbf{y}^*$ has eigenvalues $\lambda,\tau\lambda_2,\ldots,\tau\lambda_n$.
\begin{proof}
The eigenvalues of $\tau A$ are $\tau\lambda,\tau\lambda_2,\ldots,\tau\lambda_n$. Since $A_{\tau}=\tau A+\mathbf{x}((1-\overline{\tau})\overline{\lambda}\mathbf{y})^*$, Theorem \ref{thm:brauer-rank-one-eigenvalue-shift} shifts only $\tau\lambda$ by $(1-\tau)\lambda\mathbf{y}^*\mathbf{x}=(1-\tau)\lambda$. Thus $\tau\lambda$ becomes $\lambda$, while the other eigenvalues stay $\tau\lambda_i$.\\
\hlt{Structural Commentary}: Damping scales the spectrum; Brauer's update restores one eigenvalue.\\
\hlt{Key Insight}: Brauer's theorem changes one spectral coordinate without moving the rest.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:schur-complement-lower-right-determinant} \hlt{Schur Complement of the Lower-Right Block}\\
Let $M=
\begin{bmatrix}
A & B\\
C & D
\end{bmatrix}$, $A\in\operatorname{Mat}(r,r,\mathbb{F})$, $D\in\operatorname{Mat}(s,s,\mathbb{F})$.\\
If $D$ is invertible, then the \hlt{Schur complement of $D$ in $M$} is $M/D\defeq A-BD^{-1}C$, and $\det(M)=\det(D)\det(M/D)$.
\begin{proof}
Block Gaussian elimination gives $\begin{bmatrix}
I_r & -BD^{-1}\\
0 & I_s
\end{bmatrix}
\begin{bmatrix}
A & B\\
C & D
\end{bmatrix}
=
\begin{bmatrix}
A-BD^{-1}C & 0\\
C & D
\end{bmatrix}$.\\
The elimination matrix has determinant $1$, and the matrix on the right is lower block-triangular, so Theorem \ref{thm:block-upper-triangular-determinant} applied after transposition gives $\det(M)=\det(A-BD^{-1}C)\det(D)$.\\
\hlt{Structural Commentary}: This is the lower-right version of Schur complementation. Eliminating the upper-right block isolates the determinant contribution of $D$.\\
\hlt{Key Insight}: Eliminate by blocks, then read the determinant from the diagonal blocks.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:sylvester-determinant-identity} \hlt{Sylvester Determinant Identity}\\
Let $A\in\operatorname{Mat}(n,m,\mathbb{F})$ and $B\in\operatorname{Mat}(m,n,\mathbb{F})$. Then $\det(I_n+AB)=\det(I_m+BA)$.
\begin{proof}
Let $M=
\begin{bmatrix}
I_n & -A\\
B & I_m
\end{bmatrix}$.\\
Using Proposition \ref{prop:schur-complement-lower-right-determinant} with lower-right block $I_m$ gives $\det(M)=\det(I_n+AB)$. Using Proposition \ref{prop:schur-complement-determinant-formula} with upper-left block $I_n$ gives $\det(M)=\det(I_m+BA)$. Hence the two determinants are equal.\\
\hlt{Structural Commentary}: Two Schur complements of one block matrix give Sylvester's identity.\\
\hlt{Key Insight}: Compare the same determinant after eliminating different identity blocks.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:ab-ba-nonzero-eigenvalues} \hlt{Nonzero Eigenvalues of $AB$ and $BA$}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{C})$, $B\in\operatorname{Mat}(n,m,\mathbb{C})$, and suppose $n\geq m$. Then $p_{BA}(z)=z^{n-m}p_{AB}(z)$.\\
Consequently, for $\lambda\neq0$, $m_{AB}(\lambda)=m_{BA}(\lambda)$. If $m_{AB}(0)=k$, then $m_{BA}(0)=k+n-m$.
\begin{proof}
For $z\neq0$, Sylvester's determinant identity gives
\begin{align}
\det(zI_n-BA) =z^n\det(I_n-z^{-1}BA) =z^n\det(I_m-z^{-1}AB) =z^{n-m}\det(zI_m-AB).
\nonumber
\end{align}
Both sides are polynomials in $z$, so the identity holds for all $z$. The claimed multiplicities follow by factoring $p_{AB}$ and comparing with $p_{BA}=z^{n-m}p_{AB}$.\\
\hlt{Structural Commentary}: Sylvester's determinant identity moves the rectangular product across the determinant.\\
\hlt{Key Insight}: Passing from $AB$ to $BA$ only adds the necessary extra zero roots when the sizes differ.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:ab-ba-same-nonzero-jordan-blocks} \hlt{Nonzero Jordan Blocks of $AB$ and $BA$}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{C})$. For every $\lambda\neq0$, the Jordan blocks of $AB$ and $BA$ at $\lambda$ coincide.
\begin{proof} Set $X=\begin{bmatrix}AB&A\\0&0\end{bmatrix}$ and $Y=\begin{bmatrix}0&A\\0&BA\end{bmatrix}$. Then $\begin{bmatrix}I_n&0\\B&I_n\end{bmatrix}
X
\begin{bmatrix}I_n&0\\-B&I_n\end{bmatrix}
=Y$, so $X$ and $Y$ are similar.\\
For $\lambda\neq0$ and $p\geq1$, the extra diagonal block in each shifted power is $(-\lambda)^pI_n$, hence invertible. Thus block row and column operations give
\begin{align}
\operatorname{rank}\bigl((X-\lambda I_{2n})^p\bigr)
&=n+\operatorname{rank}\bigl((AB-\lambda I_n)^p\bigr), \nonumber\\
\operatorname{rank}\bigl((Y-\lambda I_{2n})^p\bigr)
&=n+\operatorname{rank}\bigl((BA-\lambda I_n)^p\bigr).
\nonumber
\end{align}
Since $X$ and $Y$ are similar, these left sides agree for all $p$.\\
Hence the shifted rank drops of $AB$ and $BA$ at $\lambda$ agree, so Theorem \ref{thm:jordan-block-sizes-from-rank-drops} gives the same Jordan blocks.\\
\hlt{Structural Commentary}: Similar block embeddings make the nonzero shifted ranks match.\\
\hlt{Key Insight}: At $\lambda\neq0$, the extra block is invertible and splits off.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:ab-ba-similar-rank-power-criterion} \hlt{Similarity Criterion for $AB$ and $BA$}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{C})$. $AB$ is similar to $BA$ if and only if $\operatorname{rank}\bigl((AB)^p\bigr)=\operatorname{rank}\bigl((BA)^p\bigr)$, $p=1,\ldots,n$.
\begin{proof}
If $AB$ and $BA$ are similar, then their powers are similar and have equal rank.\\
Conversely, Theorem \ref{thm:ab-ba-same-nonzero-jordan-blocks} gives equality of all nonzero Jordan blocks. The displayed rank equalities, together with the trivial equality at $p=0$, give the same Weyr characteristic at $\lambda=0$ by Theorem \ref{thm:jordan-block-sizes-from-rank-drops}. Hence the nilpotent Jordan blocks also agree, so the two Jordan forms are identical up to block order.\\
\hlt{Structural Commentary}: Nonzero blocks agree automatically; only the nilpotent blocks must be tested.\\
\hlt{Key Insight}: Powers of $AB$ and $BA$ measure exactly the zero-Jordan-block data.
\end{proof}
\end{corollary}

\begin{method} \label{meth:checking-ab-ba-similarity} \hlt{Checking Whether $AB$ and $BA$ Are Similar}\\
For $A,B\in\operatorname{Mat}(n,n,\mathbb{C})$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The nonzero Jordan blocks of $AB$ and $BA$ already match.
\item Compute $\operatorname{rank}\bigl((AB)^p\bigr)$ and $\operatorname{rank}\bigl((BA)^p\bigr)$ for $p=1,\ldots,n$.
\item If all ranks agree, then $AB\sim BA$; if some rank differs, then only the nilpotent Jordan structure differs.
\end{enumerate}
\end{method}

\begin{corollary} \label{cor:hermitian-products-and-conjugate-products} \hlt{Hermitian Products and Conjugate Products}\\
Let $A,H,K\in\operatorname{Mat}(n,n,\mathbb{C})$, and suppose $H$ and $K$ are Hermitian.\\
Then $HK$ is similar to $KH$, and $A\overline{A}$ is similar to $\overline{A}A$.
\begin{proof}
For each $p=1,\ldots,n$, $\operatorname{rank}\bigl((HK)^p\bigr) =\operatorname{rank}\bigl(((HK)^p)^*\bigr) = \operatorname{rank}\bigl((KH)^p\bigr)$.\\
Also, $\operatorname{rank}\bigl((A\overline{A})^p\bigr) =\operatorname{rank}\bigl(\overline{(A\overline{A})^p}\bigr) =\operatorname{rank}\bigl((\overline{A}A)^p\bigr)$. Apply Corollary \ref{cor:ab-ba-similar-rank-power-criterion} in both cases.\\
\hlt{Structural Commentary}: Adjoint and conjugation preserve rank and reverse product order.\\
\hlt{Key Insight}: The rank-power test turns order reversal into similarity.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:conjugate-similarity-hermitian-product-equivalences} \hlt{Conjugate Similarity and Hermitian Products}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ is similar to $\overline{A}$.
\item $A$ is similar to $A^*$.
\item $A$ is similar to $A^*$ via a Hermitian similarity matrix.
\item $A=HK=LM$, where $H,K,L,M$ are Hermitian, and $H,M$ are invertible.
\item $A=HK$ for some Hermitian matrices $H,K$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow(ii)$ Theorem \ref{thm:matrix-similar-to-transpose-and-symmetric-factorizations} applied to $\overline{A}$ gives $\overline{A}\sim(\overline{A})^t=A^*$. Thus $A\sim A^*$.
\item[$(ii)$]$\Rightarrow(iii)$ Choose invertible $S$ such that $A=SA^*S^{-1}$, so $AS=SA^*$. For $\theta\in\mathbb{R}$, put $S_{\theta}=e^{i\theta}S$.\\
Then $AS_{\theta}=S_{\theta}A^*$ and, by taking adjoints, $AS_{\theta}^*=S_{\theta}^*A^*$. Hence $A(S_{\theta}+S_{\theta}^*)=(S_{\theta}+S_{\theta}^*)A^*$.\\
The matrix $H_{\theta}=S_{\theta}+S_{\theta}^*$ is Hermitian. Also $S_{\theta}^{-1}H_{\theta}=I_n+e^{-2i\theta}S^{-1}S^*$. Only finitely many values of $-e^{2i\theta}$ can be eigenvalues of $S^{-1}S^*$, so choose $\theta$ for which $H_{\theta}$ is invertible. Then $A=H_{\theta}A^*H_{\theta}^{-1}$.
\item[$(iii)$]$\Rightarrow(iv)$ Let $R$ be invertible Hermitian and satisfy $A=RA^*R^{-1}$. Then $AR=RA^*$ and $R^{-1}A=A^*R^{-1}$.\\
Thus $A^*R^{-1}$ and $RA^*$ are Hermitian, and $A=R(A^*R^{-1})=(RA^*)R^{-1}$.\\
This gives the two asserted Hermitian factorizations, with the first factor in the first product and the second factor in the second product invertible.
\item[$(iv)$]$\Rightarrow(v)$ Immediate.
\item[$(v)$]$\Rightarrow(i)$ If $A=HK$ with $H,K$ Hermitian, then Corollary \ref{cor:hermitian-products-and-conjugate-products} gives $HK\sim KH$.\\
By Theorem \ref{thm:matrix-similar-to-transpose-and-symmetric-factorizations}, $KH\sim (KH)^t=H^tK^t=\overline{H}\,\overline{K}=\overline{A}$. Hence $A\sim\overline{A}$.
\end{enumerate}
\hlt{Structural Commentary}: The equivalences move between conjugation, adjoints, and Hermitian products by combining transpose similarity with a Hermitianized similarity matrix.\\
\hlt{Key Insight}: A phase-rotated average $e^{i\theta}S+e^{-i\theta}S^*$ turns an arbitrary similarity to $A^*$ into a Hermitian one.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:block-centrosymmetric-commutation} \hlt{Block Centrosymmetry and Commutation}\\
Let $M=
\begin{bmatrix}
A & B\\
C & D
\end{bmatrix}\in\operatorname{Mat}(2n,2n,\mathbb{F})$,
$L_{2n}=
\begin{bmatrix}
0 & I_n\\
I_n & 0
\end{bmatrix}$.\\
Then $M$ has block centrosymmetric form $A=D$ and $B=C$ if and only if $L_{2n}M=ML_{2n}$.
\begin{proof}
Block multiplication gives $L_{2n}M=
\begin{bmatrix}
C & D\\
A & B
\end{bmatrix}$ and 
$ML_{2n}=
\begin{bmatrix}
B & A\\
D & C
\end{bmatrix}$.\\
The two matrices are equal exactly when $A=D$ and $B=C$.\\
\hlt{Structural Commentary}: The matrix $L_{2n}$ swaps the two block coordinates. Commuting with that swap means $M$ treats the two block positions symmetrically.\\
\hlt{Key Insight}: Compute both block products; equality forces matching opposite blocks.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:frobenius-rank-inequality} \hlt{Frobenius Rank Inequality}\\
Let $A\in\operatorname{Mat}(m,k,\mathbb{F})$, $B\in\operatorname{Mat}(k,p,\mathbb{F})$, and $C\in\operatorname{Mat}(p,n,\mathbb{F})$. $\operatorname{rank}(AB)+\operatorname{rank}(BC) \leq \operatorname{rank}(B)+\operatorname{rank}(ABC)$.
\begin{proof}
Let $R=\operatorname{col}(B)$ and $S=\operatorname{col}(BC)$. Then $S\subseteq R$. The induced map $T_A$ maps $R$ onto $\operatorname{col}(AB)$ and maps $S$ onto $\operatorname{col}(ABC)$. Rank-nullity on the restrictions $T_A|_R$ and $T_A|_S$ gives
\begin{align}
\operatorname{rank}(AB)&=\dim R-\dim(R\cap\ker A), \nonumber\\
\operatorname{rank}(ABC)&=\dim S-\dim(S\cap\ker A). \nonumber
\end{align}
Since $S\cap\ker A\subseteq R\cap\ker A$, we have $\dim(S\cap\ker A)\leq\dim(R\cap\ker A)$.\\
Substituting $\dim R=\operatorname{rank}(B)$ and $\dim S=\operatorname{rank}(BC)$ gives the stated inequality.\\
\hlt{Structural Commentary}: Frobenius tracks $A$ on nested column spaces.\\
\hlt{Key Insight}: The smaller column space cannot lose more dimensions to $\ker A$ than the larger one.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:frobenius-rank-inequality-special-cases} \hlt{Rank Inequalities from Frobenius}\\
The following are special cases of Theorem \ref{thm:frobenius-rank-inequality}:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
$\operatorname{rank}(AB)\leq\min(\operatorname{rank}(A),\operatorname{rank}(B))$;
\item
$\operatorname{rank}(A)+\operatorname{rank}(B)-k\leq\operatorname{rank}(AB)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
Apply Frobenius to generic factors $X,Y,Z$. Taking $Z=0$ gives $\operatorname{rank}(XY)\leq\operatorname{rank}(Y)$, and taking $X=0$ gives $\operatorname{rank}(YZ)\leq\operatorname{rank}(Y)$. Renaming the factors gives $\operatorname{rank}(AB)\leq\operatorname{rank}(B)$ and $\operatorname{rank}(AB)\leq\operatorname{rank}(A)$.
\item
Taking the middle matrix to be $I_k$ gives $\operatorname{rank}(A)+\operatorname{rank}(B)\leq k+\operatorname{rank}(AB)$, which is Sylvester's inequality.
\end{enumerate}
\hlt{Structural Commentary}: Frobenius is the three-factor parent inequality. The usual upper and lower product-rank bounds appear by making one factor zero or identity.\\
\hlt{Key Insight}: Zero factors isolate upper bounds; an identity middle factor yields the Sylvester lower bound.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:sylvester-law-of-nullity} \hlt{Sylvester's Law of Nullity}\\
For a matrix $X$, let $\operatorname{nullity}(X)=\dim(\ker X)$; this is the dimension of its nullspace.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A\in\operatorname{Mat}(m,k,\mathbb{F})$ and $B\in\operatorname{Mat}(k,n,\mathbb{F})$, then $\operatorname{nullity}(AB)\leq\operatorname{nullity}(A)+\operatorname{nullity}(B)$
\item
If $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$, then $\max(\operatorname{nullity}(A),\operatorname{nullity}(B))\leq\operatorname{nullity}(AB)$
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
By rank-nullity and Theorem \ref{thm:sylvester-rank-inequality},
\begin{align*}
\operatorname{nullity}(AB) &=n-\operatorname{rank}(AB) \leq n-\operatorname{rank}(A)-\operatorname{rank}(B)+k \\
&=(k-\operatorname{rank}(A))+(n-\operatorname{rank}(B)) =\operatorname{nullity}(A)+\operatorname{nullity}(B).
\end{align*}
\item For square $A,B$, $\operatorname{rank}(AB)\leq\operatorname{rank}(A)$, $\operatorname{rank}(AB)\leq\operatorname{rank}(B)$. Subtract from $n$ gives desired lower bounds.
\end{enumerate}
\hlt{Structural Commentary}: Nullity is rank loss from the domain.\\
\hlt{Key Insight}: Use rank-nullity to translate rank bounds into kernel bounds.
\end{proof}
\end{proposition}

\begin{definition} \label{def:increasing-hermitian-eigenvalue-ordering} \hlt{Increasing Hermitian Eigenvalue Order}\\
For a Hermitian matrix $A\in\operatorname{Mat}(n,n,\mathbb{C})$, write $\lambda_1(A)\leq\lambda_2(A)\leq\cdots\leq\lambda_n(A)$ for its eigenvalues counted with algebraic multiplicity. If only one Hermitian matrix is under discussion, we write $\lambda_i$ instead of $\lambda_i(A)$.
\end{definition}

\begin{theorem} \label{thm:rayleigh-quotient-spectral-interval} \hlt{Rayleigh Quotient on a Spectral Interval}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ be Hermitian with increasing eigenvalues $\lambda_1\leq\cdots\leq\lambda_n$ and corresponding orthonormal eigenvectors $\mathbf{u}_1,\ldots,\mathbf{u}_n$. If $1\leq p\leq q\leq n$ and $\mathbf{x}\in\operatorname{span}(\mathbf{u}_p,\ldots,\mathbf{u}_q)$ is a unit vector, then $\lambda_p\leq \mathbf{x}^*A\mathbf{x}\leq \lambda_q$.
\begin{proof}
Write $\mathbf{x}=\sum_{i=p}^{q}c_i\mathbf{u}_i$. Orthonormality gives $\sum_{i=p}^{q}\abs{c_i}^2=1$, and $\mathbf{x}^*A\mathbf{x} =\sum\limits_{i=p}^{q}\lambda_i\abs{c_i}^2$.\\
This is a convex combination of $\lambda_p,\ldots,\lambda_q$, so it lies between the smallest and largest of them.\\
\hlt{Structural Commentary}: In an eigenbasis, the Rayleigh quotient is an eigenvalue average.\\
\hlt{Key Insight}: Spectral coordinates turn a quadratic form into weighted scalar data.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:rayleigh-extrema-and-diagonal-entry-bounds} \hlt{Rayleigh Extrema and Diagonal Entry Bounds}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ be Hermitian.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathcal{U}_{p,q}=\operatorname{span}(\mathbf{u}_p,\ldots,\mathbf{u}_q)$, then $\min\limits_{\substack{\mathbf{x}\in\mathcal{U}_{p,q}\\ \norm{\mathbf{x}}_2=1}}\mathbf{x}^*A\mathbf{x}=\lambda_p(A)$, and $\max\limits_{\substack{\mathbf{x}\in\mathcal{U}_{p,q}\\ \norm{\mathbf{x}}_2=1}}\mathbf{x}^*A\mathbf{x}=\lambda_q(A).$
\item For every unit vector $\mathbf{x}$, $\lambda_1(A)\leq \mathbf{x}^*A\mathbf{x}\leq\lambda_n(A)$.
\item If $A=[a_{ij}]$, then $\lambda_1(A)\leq a_{ii}\leq\lambda_n(A)$ for $i=1,\ldots,n$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Theorem \ref{thm:rayleigh-quotient-spectral-interval} gives the inequalities on $\mathcal{U}_{p,q}$, with equality at $\mathbf{u}_p$ and $\mathbf{u}_q$.
\item This is (i) with $p=1$ and $q=n$.
\item Apply (ii) to $\mathbf{x}=\mathbf{e}_i$, since $\mathbf{e}_i^*A\mathbf{e}_i=a_{ii}$.
\end{enumerate}
\hlt{Structural Commentary}: Rayleigh extrema on extreme eigendirections; diagonal entries are coordinate quotients.\\
\hlt{Key Insight}: The whole numerical range of a Hermitian matrix sits inside its spectral interval.
\end{proof}
\end{corollary}

\begin{method} \label{meth:rayleigh-eigenbasis-method} \hlt{Rayleigh Eigenbasis Method}\\
To bound a Hermitian quadratic form:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item diagonalise $A$ with an orthonormal eigenbasis;
\item expand the unit vector in that basis;
\item rewrite $\mathbf{x}^*A\mathbf{x}$ as $\sum_i\lambda_i\abs{c_i}^2$;
\item use the eigenvalue range allowed by the support of $\mathbf{x}$.
\end{enumerate}
\end{method}

\begin{theorem} \label{thm:rank-one-psd-eigenvalue-interlacing} \hlt{Rank-One Positive Perturbation Interlacing}\\
Let $A,E\in\operatorname{Mat}(n,n,\mathbb{C})$ be Hermitian, with $E\succeq0$ and $\operatorname{rank}(E)\leq1$.\\
Then $\lambda_i(A)\leq\lambda_i(A+E)\leq\lambda_{i+1}(A)\quad(1\leq i<n)$, $\lambda_n(A)\leq\lambda_n(A+E)$\\
Equivalently, $\lambda_i(A-E)\leq\lambda_i(A)\leq\lambda_{i+1}(A-E)\quad(1\leq i<n)$, $\lambda_n(A-E)\leq\lambda_n(A)$.
\begin{proof}
Let $\mathbf{u}_j$ and $\mathbf{v}_j$ be orthonormal eigenvectors of $A$ and $A+E$, respectively, in increasing eigenvalue order.\\
For the lower bound, set $\mathcal{U}=\operatorname{span}(\mathbf{u}_i,\ldots,\mathbf{u}_n)$, $\mathcal{V}=\operatorname{span}(\mathbf{v}_1,\ldots,\mathbf{v}_i)$. Since $\dim\mathcal{U}+\dim\mathcal{V}=n+1$, choose a unit $\mathbf{x}\in\mathcal{U}\cap\mathcal{V}$. Then $\lambda_i(A)\leq \mathbf{x}^*A\mathbf{x}\leq \mathbf{x}^*(A+E)\mathbf{x}\leq\lambda_i(A+E)$ by Theorem \ref{thm:rayleigh-quotient-spectral-interval} and $E\succeq0$.\\
For the upper bound, set $\mathcal{U}=\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_{i+1})$, $\mathcal{V}=\operatorname{span}(\mathbf{v}_i,\ldots,\mathbf{v}_n)$, $\mathcal{W}=\ker(E)$. Here $\dim(\mathcal{U}\cap\mathcal{V})\geq2$ and $\dim\mathcal{W}\geq n-1$, so choose a unit $\mathbf{x}\in\mathcal{U}\cap\mathcal{V}\cap\mathcal{W}$. Thus $\lambda_i(A+E)\leq \mathbf{x}^*(A+E)\mathbf{x} =\mathbf{x}^*A\mathbf{x} \leq \lambda_{i+1}(A)$.\\
The statements for $A-E$ follow by applying the proved result to $A-E$ and $E$.\\
\hlt{Structural Commentary}: Interlacing comes from test vectors in overlapping spectral subspaces.\\
\hlt{Key Insight}: Rank one leaves a large kernel, so the upper-bound vector can avoid the perturbation.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:rank-two-one-positive-one-negative-interlacing} \hlt{Rank-Two Indefinite Interlacing}\\
Let $A,H\in\operatorname{Mat}(n,n,\mathbb{C})$ be Hermitian, $n\geq2$. Suppose $\operatorname{rank}(H)\leq2$ and $H$ has at most one positive and one negative eigenvalue. Then
\begin{align}
\lambda_1(A+H)\leq\lambda_2(A),\qquad
\lambda_{i-1}(A)\leq\lambda_i(A+H)\leq\lambda_{i+1}(A)\quad(2\leq i<n),
\qquad
\lambda_{n-1}(A)\leq\lambda_n(A+H).
\nonumber
\end{align}
\begin{proof}
Write $H=E_+-E_-$ with $E_\pm\succeq0$ and $\operatorname{rank}(E_\pm)\leq1$. Set $B=A-E_-$; then $A=B+E_-$ and $A+H=B+E_+$. For $2\leq i<n$, rank-one interlacing gives $\lambda_{i-1}(A)\leq\lambda_i(B)\leq\lambda_i(A+H)\leq\lambda_{i+1}(B)\leq\lambda_{i+1}(A)$.\\
The endpoints follow similarly: $\lambda_1(A+H)\leq\lambda_2(B)\leq\lambda_2(A)$, and $\lambda_{n-1}(A)\leq\lambda_n(B)\leq\lambda_n(A+H)$.\\
\hlt{Structural Commentary}: Such an $H$ is a difference of two rank-one PSD perturbations.\\
\hlt{Key Insight}: Interlace once with $E_-$, then once with $E_+$.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:cauchy-principal-submatrix-interlacing} \hlt{Cauchy Interlacing: One Principal Deletion}\\
Let $A=
\begin{bmatrix}
B & \mathbf{y}\\
\mathbf{y}^* & c
\end{bmatrix}
\in\operatorname{Mat}(n+1,n+1,\mathbb{C})$ be Hermitian, with $B\in\operatorname{Mat}(n,n,\mathbb{C})$.\\
Then, for $1\leq i\leq n$, $\lambda_i(A)\leq\lambda_i(B)\leq\lambda_{i+1}(A)$.
\begin{proof}
Let $\mathbf{u}_j$ and $\mathbf{v}_j$ be orthonormal eigenvectors of $A$ and $B$, respectively, in increasing eigenvalue order. Identify $\mathbb{C}^n$ with $\{(\mathbf{z},0)\}\subseteq\mathbb{C}^{n+1}$, so $(\mathbf{z},0)^*A(\mathbf{z},0)=\mathbf{z}^*B\mathbf{z}$.\\
For the lower bound, choose a unit $\mathbf{x}=(\mathbf{z},0)\in \operatorname{span}((\mathbf{v}_1,0),\ldots,(\mathbf{v}_i,0)) \cap \operatorname{span}(\mathbf{u}_i,\ldots,\mathbf{u}_{n+1})$, possible because the dimensions sum to $n+2$. Then $\lambda_i(A)\leq \mathbf{x}^*A\mathbf{x}=\mathbf{z}^*B\mathbf{z}\leq\lambda_i(B)$.\\
For the upper bound, choose a unit $\mathbf{x}=(\mathbf{z},0)\in \operatorname{span}((\mathbf{v}_i,0),\ldots,(\mathbf{v}_n,0)) \cap \operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_{i+1})$. The same dimension count applies, and $\lambda_i(B)\leq \mathbf{z}^*B\mathbf{z}=\mathbf{x}^*A\mathbf{x}\leq\lambda_{i+1}(A)$.\\
\hlt{Structural Commentary}: A principal submatrix is a coordinate slice of the larger quadratic form.\\
\hlt{Key Insight}: Deleting one coordinate costs one dimension, so eigenvalues shift by at most one index.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:singular-values-interlace-after-deleting-row-or-column} \hlt{Singular Values Interlace After Deleting a Row or Column}\\
Let $B$ be obtained from $A\in\operatorname{Mat}(n,n,\mathbb{C})$ by deleting one row or column.\\
With singular values ordered decreasingly, $sigma_i(A)\geq\sigma_i(B)\geq\sigma_{i+1}(A)\quad(1\leq i<n)$.
\begin{proof}
For a row deletion, permute rows so that $PA=\begin{bmatrix}B\\ \mathbf{y}^*\end{bmatrix}$. Then $PA$ has the same singular values as $A$, and $BB^*$ is a principal submatrix of $(PA)(PA)^*$.\\
Cauchy interlacing applied to these Gram matrices gives the result after taking square roots and reversing the increasing eigenvalue order. A column deletion follows by applying the row case to $A^*$.\\
\hlt{Structural Commentary}: Row deletion makes $BB^*$ a principal submatrix of a permuted $AA^*$.\\
\hlt{Key Insight}: Singular-value interlacing is Cauchy interlacing applied to Gram matrices.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:cauchy-interlacing-general-principal-submatrix} \hlt{Cauchy Interlacing for Principal Submatrices}\\
Let $B\in\operatorname{Mat}(n,n,\mathbb{C})$ be a principal submatrix of Hermitian $A\in\operatorname{Mat}(n+m,n+m,\mathbb{C})$.\\
Then, for $1\leq i\leq n$, $\lambda_i(A)\leq\lambda_i(B)\leq\lambda_{i+m}(A)$.
\begin{proof}
After a permutation similarity, $B$ is leading.\\
Delete the other $m$ coordinates one at a time and apply Theorem \ref{thm:cauchy-principal-submatrix-interlacing} at each step.\\
\hlt{Structural Commentary}: Principal submatrices come from repeated coordinate deletion.\\
\hlt{Key Insight}: Each deletion shifts eigenvalue indices by at most one.
\end{proof}
\end{corollary}

\begin{definition} \label{def:principal-and-leading-principal-minors} \hlt{Principal and Leading Principal Minors}\\
For $A\in\operatorname{Mat}(n,n,\mathbb{C})$, a \hlt{principal minor} is the determinant of a principal submatrix of $A$.\\
A \hlt{leading principal minor} is the determinant of the leading $k\times k$ principal submatrix for some $k=1,\ldots,n$.
\end{definition}

\begin{theorem} \label{thm:sylvester-positive-definite-criterion-leading-minors} \hlt{Sylvester's Criterion}\\
A Hermitian matrix $A\in\operatorname{Mat}(n,n,\mathbb{C})$ is positive definite if and only if all leading principal minors are positive.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $A\succ0$, then all leading principal submatrix is PSD by Corollary \ref{cor:positive-semidefinite-principal-block-column-inclusion}, so its determinant is positive.
\item[$(\Leftarrow)$] Induct on $n$. The case $n=1$ is immediate. Write $A=
\begin{bmatrix}
B & \mathbf{y}\\
\mathbf{y}^* & a
\end{bmatrix}$, $B\in\operatorname{Mat}(n-1,n-1,\mathbb{C})$\\
The leading principal minors of $B$ are positive, so $B\succ0$.\\
Cauchy interlacing gives $\lambda_1(A)\leq\lambda_1(B)\leq\lambda_2(A)\leq\cdots\leq\lambda_{n-1}(B)\leq\lambda_n(A)$.\\
Thus $\lambda_2(A),\ldots,\lambda_n(A)>0$. Since $\det A=\prod_j\lambda_j(A)>0$, also $\lambda_1(A)>0$; hence $A\succ0$.
\end{enumerate}
\hlt{Structural Commentary}: Leading minors pass positivity from the leading block to almost all eigenvalues.\\
\hlt{Key Insight}: The determinant fixes the last eigenvalue's sign.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:schur-majorization-eigenvalues-by-diagonal} \hlt{Schur Majorization of the Diagonal}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ be Hermitian, with increasing eigenvalues $\lambda_i(A)$ and increasing diagonal entries $d_i$.\\
Then $\sum\limits_{i=1}^{k}\lambda_i(A)\leq \sum\limits_{i=1}^{k}d_i$, $(1\leq k\leq n)$ with equality for $k=n$.
\begin{proof}
Permute coordinates so that $a_{ii}=d_i$, and let $A_k$ be the leading $k\times k$ principal submatrix.\\
Cauchy interlacing gives $\lambda_i(A)\leq\lambda_i(A_k)$ for $1\leq i\leq k$, hence $\sum\limits_{i=1}^{k}\lambda_i(A)\leq\sum\limits_{i=1}^{k}\lambda_i(A_k)=\operatorname{tr}(A_k)=\sum\limits_{i=1}^{k}d_i$.\\
For $k=n$, this is $\operatorname{tr}(A)=\operatorname{tr}(A)$.\\
\hlt{Structural Commentary}: Ordered diagonal sums are traces of leading principal blocks.\\
\hlt{Key Insight}: Diagonal majorization is Cauchy interlacing plus trace.
\end{proof}
\end{theorem}

\begin{method} \label{meth:eigenvalue-interlacing-method} \hlt{Eigenvalue Interlacing Method}\\
To prove or use an interlacing estimate:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item order Hermitian eigenvalues increasingly;
\item identify dimension lost: rank-one perturbation, one deleted coordinate, $m$-coordinate principal submatrix;
\item choose test vectors by intersecting spectral subspaces with the unaffected subspace;
\item convert the Rayleigh quotient bounds into eigenvalue inequalities.
\end{enumerate}
\end{method}

\begin{theorem} \label{thm:weyl-eigenvalue-inequalities-hermitian-sums} \hlt{Weyl Eigenvalue Inequalities}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{C})$ be Hermitian. For $1\leq i\leq n$,
\begin{align*}
\lambda_i(A+B) &\leq \lambda_{i+j}(A)+\lambda_{n-j}(B),
& 0 &\leq j\leq n-i,\\
\lambda_{i-j+1}(A)+\lambda_j(B) &\leq \lambda_i(A+B),
& 1 &\leq j\leq i.
\end{align*}
\begin{proof}
Let $\mathbf{u}_k,\mathbf{v}_k,\mathbf{w}_k$ be increasing eigenbases for $A,A+B,B$.\\
Fix $i$ and $0\leq j\leq n-i$, and set $\mathcal{U}=\operatorname{span}(\mathbf{u}_1,\ldots,\mathbf{u}_{i+j})$, $\mathcal{V}=\operatorname{span}(\mathbf{v}_i,\ldots,\mathbf{v}_n)$, $\mathcal{W}=\operatorname{span}(\mathbf{w}_1,\ldots,\mathbf{w}_{n-j})$.\\
Their dimensions sum to $2n+1$, so choose a unit $\mathbf{x}\in\mathcal{U}\cap\mathcal{V}\cap\mathcal{W}$.\\
Then $\lambda_i(A+B)\leq \mathbf{x}^*(A+B)\mathbf{x} =\mathbf{x}^*A\mathbf{x}+\mathbf{x}^*B\mathbf{x} \leq \lambda_{i+j}(A)+\lambda_{n-j}(B)$.\\
The lower bound follows by applying this upper bound to $-A$ and $-B$ and using $\lambda_k(-C)=-\lambda_{n-k+1}(C)$.\\
\hlt{Structural Commentary}: Weyl bounds come from intersecting the spectral flags of $A$, $B$, and $A+B$.\\
\hlt{Key Insight}: One shared test vector controls all three Rayleigh quotients.
\end{proof}
\end{theorem}

\begin{method} \label{meth:weyl-inequality-method} \hlt{Weyl Bound Method}\\
To bound eigenvalues of $A+B$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item choose the target index $i$;
\item use the upper Weyl family with $j=0,\ldots,n-i$ to get upper bounds;
\item use the lower Weyl family with $j=1,\ldots,i$ to get lower bounds;
\item for positive semidefinite or low-rank $B$, insert the known zero or nonnegative eigenvalues to recover monotonicity or interlacing.
\end{enumerate}
\end{method}

\begin{definition} \label{def:star-congruence-and-hermitian-inertia} \hlt{$*$-Congruence and Hermitian Inertia}\\
Two matrices $A,B\in\operatorname{Mat}(n,n,\mathbb{C})$ are \hlt{$*$-congruent} if $A=SBS^*$ for some invertible $S\in\operatorname{Mat}(n,n,\mathbb{C})$.\\
For Hermitian $A$, its \hlt{inertia} is $\operatorname{in}(A)=(n_+(A),n_-(A),n_0(A))$, where these are the numbers of positive, negative, and zero eigenvalues of $A$, counted with multiplicity.
\end{definition}

\begin{lemma} \label{lem:positive-subspace-dimension-inertia} \hlt{Positive Subspaces and Inertia}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ be Hermitian. The maximal dimensions of subspaces on which $\mathbf{x}^*A\mathbf{x}$ is positive or negative for every nonzero $\mathbf{x}$ are $n_+(A)$ and $n_-(A)$, respectively.
\begin{proof}
Let $P_+$ and $P_{\leq0}$ be the positive and nonpositive spectral subspaces.\\
Then $\dim P_+=n_+(A)$ and $P_+$ is positive for $A$. If $\dim\mathcal{U}>n_+(A)$, then $\mathcal{U}\cap P_{\leq0}$ contains a nonzero vector, so $\mathcal{U}$ cannot be positive. Apply this to $-A$ for the negative case.\\
\hlt{Structural Commentary}: The positive spectral subspace is maximal; the nonpositive part blocks enlargement.\\
\hlt{Key Insight}: Inertia counts maximal sign-definite dimensions.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:sylvester-inertia-theorem-hermitian} \hlt{Sylvester's Inertia Theorem}\\
Hermitian matrices $A,B$ of the same size are $*$-congruent if and only if $n_+(A)=n_+(B)$ and $n_-(A)=n_-(B)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $A=SBS^*$ with $S$ invertible. Then $\mathbf{x}^*A\mathbf{x}=(S^*\mathbf{x})^*B(S^*\mathbf{x})$.\\
Thus $S^*$ transports positive and negative subspaces, so Lemma \ref{lem:positive-subspace-dimension-inertia} gives $n_\pm(A)=n_\pm(B)$.
\item[$(\Leftarrow)$] Let the common inertia be $(p,q,r)$ and set $D=I_p\oplus(-I_q)\oplus0_r$.\\
By the spectral theorem, each of $A$ and $B$ is $*$-congruent to $D$, hence they are $*$-congruent to each other.
\end{enumerate}
\hlt{Structural Commentary}: $*$-congruence is a coordinate change for Hermitian quadratic forms.\\
\hlt{Key Insight}: Spectral decomposition reduces the form to signed squares and zero directions.
\end{proof}
\end{theorem}

\begin{definition} \label{def:phase-inertia-matrix} \hlt{Phase Inertia Matrix}\\
A \hlt{phase inertia matrix} is a diagonal complex matrix whose nonzero diagonal entries have modulus $1$.
\end{definition}

\begin{proposition} \label{prop:normal-matrix-phase-inertia-form} \hlt{Phase Inertia Form for Normal Matrices}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ be normal. Then $A$ is $*$-congruent to a phase inertia matrix $D$, whose entries record the spectral rays of $A$.
\begin{proof}
Write $A=U\operatorname{diag}(\lambda_j)U^*$. Write $\lambda_j=r_jd_j$, where $r_j>0$ and $|d_j|=1$ if $\lambda_j\neq0$, while $r_j=1$ and $d_j=0$ if $\lambda_j=0$. Set $D=\operatorname{diag}(d_j)$, $S=U\operatorname{diag}(\sqrt{r_j})$. Then $S$ is invertible and $A=SDS^*$.\\
The entries $d_j=e^{i\theta}$ count exactly the eigenvalues on the ray $\{\rho e^{i\theta}:\rho>0\}$.\\
\hlt{Structural Commentary}: Normal diagonalization separates modulus from phase; $*$-congruence absorbs moduli.\\
\hlt{Key Insight}: Only spectral phases survive.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:hermitian-inertia-as-real-phase-inertia} \hlt{Hermitian Inertia as Real Phase Inertia}\\
For Hermitian matrices, the only nonzero spectral rays are the positive and negative real axes. Sylvester's inertia theorem is therefore the real-ray version of phase inertia.
\end{perspective}
