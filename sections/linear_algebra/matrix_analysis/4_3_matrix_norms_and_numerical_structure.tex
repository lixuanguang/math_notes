\subsubsection{Matrix Norms and Numerical Structure}

Numerical linear algebra often replaces a matrix by a unitarily similar one with many forced zero entries. Upper Hessenberg form is the standard preliminary reduction before QR-type eigenvalue algorithms.

\begin{definition} \label{def:gersgorin-disks} \hlt{Gersgorin Disks}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$. The \hlt{deleted absolute row sum} in row $k$ is $R'_k(A)\defeq\sum\limits_{j\neq k}\abs{a_{kj}}$.\\
The \hlt{$k$th Gersgorin disk} of $A$ is $G_k(A)\defeq\{z\in\mathbb{C}:\abs{z-a_{kk}}\leq R'_k(A)\}$, where $G(A)\defeq\bigcup\limits_{k=1}^{n}G_k(A)$
\end{definition}

\begin{theorem} \label{thm:gersgorin-disk-theorem} \hlt{Gersgorin Disk Theorem}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $\operatorname{spec}(A)\subseteq G(A)=\bigcup\limits_{k=1}^{n}G_k(A)$.
\begin{proof}
Let $\lambda\in\operatorname{spec}(A)$ and choose an eigenvector $\mathbf{x}=[x_1\ \cdots\ x_n]^t$.\\
Pick $k$ such that $\abs{x_k}=\max_j\abs{x_j}$; then $x_k\neq0$. The $k$th row of $A\mathbf{x}=\lambda\mathbf{x}$ gives $(\lambda-a_{kk})x_k=\sum\limits_{j\neq k}a_{kj}x_j$.\\
Divide by $x_k$ and use $\abs{x_j/x_k}\leq1$: $\abs{\lambda-a_{kk}} =\abs{\sum\limits_{j\neq k}a_{kj}\frac{x_j}{x_k}} \leq\sum\limits_{j\neq k}\abs{a_{kj}}\abs{\frac{x_j}{x_k}} \leq R'_k(A)$.\\
Thus $\lambda\in G_k(A)\subseteq G(A)$.\\
\hlt{Structural Commentary}: Normalize by a largest component; all row ratios have modulus at most one.\\
\hlt{Key Insight}: Row $k$ traps $\lambda$ in the disk centered at $a_{kk}$ with radius $R'_k(A)$.
\end{proof}
\end{theorem}

\begin{method} \label{meth:gersgorin-disk-localisation} \hlt{Gersgorin Localisation Method}\\
For $A=[a_{ij}]$, Gersgorin localisation gives a disk enclosure for the spectrum.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For each row $k$, place a disk at the diagonal entry $a_{kk}$ with radius $R'_k(A)=\sum\limits_{j\neq k}\abs{a_{kj}}$.\\
The row enclosure is $G(A)=\bigcup_{k=1}^{n}G_k(A)$.
\item Theorem \ref{thm:gersgorin-disk-theorem} guarantees that every eigenvalue of $A$ lies in $G(A)$.
\item For a sharper enclosure, repeat the same construction for $A^t$. The centers are still $a_{kk}$, while the radii becomes deleted column sums of $A$. Then use $\operatorname{spec}(A)\subseteq G(A)\cap G(A^t)$.
This is the intersection of two unions of disks; an eigenvalue need not lie in the overlap of each individual row-column disk pair.
\end{enumerate}
\end{method}

\begin{corollary} \label{cor:gersgorin-row-column-refinement} \hlt{Row-Column Gersgorin Refinement}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $\operatorname{spec}(A)\subseteq G(A)\cap G(A^t)$.\\
If $R_k(A)=\sum\limits_{j=1}^{n}\abs{a_{kj}}$ and $R_{\max}(A)=\max_{1\leq k\leq n}R_k(A)$, then $\operatorname{spec}(A)\subseteq\{z\in\mathbb{C}:\abs{z}\leq \min(R_{\max}(A),R_{\max}(A^t))\}$.
\begin{proof}
Theorem \ref{thm:gersgorin-disk-theorem} gives $\operatorname{spec}(A)\subseteq G(A)$.\\
Proposition \ref{prop:transpose-preserves-spectrum-and-geometric-multiplicity} gives $\operatorname{spec}(A)=\operatorname{spec}(A^t)$, so applying Gersgorin to $A^t$ gives $\operatorname{spec}(A)\subseteq G(A^t)$.\\
Also $G_k(A)\subseteq\{z:\abs{z}\leq \abs{a_{kk}}+R'_k(A)=R_k(A)\}$, and similarly for $A^t$.\\
\hlt{Structural Commentary}: Row disks and column disks are both valid because $A$ and $A^t$ have the same spectrum.\\
\hlt{Key Insight}: Intersecting independent enclosures usually sharpens the eigenvalue search region.
\end{proof}
\end{corollary}

\begin{definition} \label{def:diagonal-dominance} \hlt{Diagonal Dominance}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$. Matrix $A$ is \hlt{diagonally dominant} if $\abs{a_{kk}}\geq R'_k(A)$ for every $k$, and \hlt{strictly diagonally dominant} if $\abs{a_{kk}}>R'_k(A)$ for every $k$.
\end{definition}

\begin{corollary} \label{cor:strict-diagonal-dominance-invertible} \hlt{Strict Diagonal Dominance Implies Invertibility}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A$ is strictly diagonally dominant, then $A$ is invertible.
\item If $A$ has real spectrum and nonnegative diagonal entries, then diagonal dominance gives $\operatorname{spec}(A)\subseteq[0,\infty)$; strict diagonal dominance gives $\operatorname{spec}(A)\subseteq(0,\infty)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Strict dominance gives $0\notin G_k(A)$ for every $k$, since $\abs{a_{kk}}>R'_k(A)$.\\
Thus $0\notin\operatorname{spec}(A)$ by Gersgorin, so $A$ is invertible by Corollary \ref{cor:matrix-invertible-iff-zero-not-spectrum}.
\item If $a_{kk}\geq0$ and $a_{kk}\geq R'_k(A)$, then each $G_k(A)$ lies in the closed right half-plane. In the strict case, each disk lies in the open right half-plane. Gersgorin and the real-spectrum hypothesis give the claim.
\end{enumerate}
\hlt{Structural Commentary}: Gersgorin converts diagonal dominance into disk placement.\\
\hlt{Key Insight}: The disks control whether eigenvalues can reach $0$ or the left half-plane.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:weak-diagonal-dominance-invertible-with-one-strict-row} \hlt{Weak Diagonal Dominance Criterion}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{C})$ be diagonally dominant with no zero entries.\\
If $\abs{a_{kk}}>R'_k(A)$ for some $k$, then $A$ is invertible.
\begin{proof}
Assume $A\mathbf{x}=0$ with $\mathbf{x}\neq\mathbf{0}$, and choose $k$ with $\abs{x_k}=\norm{\mathbf{x}}_{\infty}$.\\
Row $k$ gives $\abs{a_{kk}}\norm{\mathbf{x}}_{\infty} =\abs{-\sum\limits_{j\neq k}a_{kj}x_j} \leq\sum\limits_{j\neq k}\abs{a_{kj}}\abs{x_j} \leq R'_k(A)\norm{\mathbf{x}}_{\infty} \leq \abs{a_{kk}}\norm{\mathbf{x}}_{\infty}$.\\
Hence all inequalities are equalities. Since no $a_{kj}$ vanishes, $\abs{x_j}=\norm{\mathbf{x}}_{\infty}$ for every $j$. Applying the same chain to any row $i$ gives $\abs{a_{ii}}=R'_i(A)$ for every $i$, contradicting the strict row. Thus $\ker(A)=\{\mathbf{0}\}$, so $A$ is invertible.\\
\hlt{Structural Commentary}: A zero vector in the kernel would force equality in every Gersgorin row.\\
\hlt{Key Insight}: One strict row breaks that equality pattern.
\end{proof}
\end{theorem}

\begin{definition} \label{def:upper-hessenberg-matrix} \hlt{Upper Hessenberg Matrix}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{F})$. Then $A$ is \hlt{upper Hessenberg} if
\begin{align}
a_{ij}=0\qquad\text{whenever } i>j+1.
\nonumber
\end{align}
Equivalently, all entries below the first subdiagonal are zero.
\end{definition}

\begin{theorem} \label{thm:unitary-upper-hessenberg-reduction} \hlt{Unitary Upper Hessenberg Reduction}\\
For every $A\in\operatorname{Mat}(n,n,\mathbb{F})$, there exists a unitary matrix $U$ such that $U^*AU$ is upper Hessenberg. Moreover, the subdiagonal entries of $U^*AU$ can be chosen real and nonnegative.
\begin{proof}
Construct the unitary similarity column by column. At step $k=1,\ldots,n-2$, assume entries below the first subdiagonal are already zero in columns $1,\ldots,k-1$. Look at the tail of column $k$ in rows $k+1,\ldots,n$. By Corollary \ref{cor:householder-unitary-annihilator}, choose a unitary $V_k\in\operatorname{Mat}(n-k,n-k,\mathbb{F})$ such that $V_k^*$ sends this tail to a vector whose only nonzero entry is its first entry, chosen real and nonnegative.\\
Set $U_k=I_k\oplus V_k$ and replace the current matrix $A_{k-1}$ by $A_k=U_k^*A_{k-1}U_k$. Because $U_k$ fixes the first $k$ coordinate vectors, right multiplication by $U_k$ does not change the first $k$ columns before the left action is applied. The left action affects only rows $k+1,\ldots,n$, so previously created zeros in columns $1,\ldots,k-1$ stay zero, while column $k$ gains zeros below row $k+1$.\\
After $n-2$ steps, $A_{n-2}$ is upper Hessenberg with real nonnegative subdiagonal entries. Since each step is a unitary similarity, $A_{n-2}=U^*AU$ for the product $U=U_1U_2\cdots U_{n-2}$.\\
\hlt{Structural Commentary}: Householder similarities annihilate one column tail at a time while fixing the earlier coordinate flag.\\
\hlt{Key Insight}: Acting only on the trailing coordinates preserves previous zeros and creates the next Hessenberg column.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:hessenberg-before-qr} \hlt{Why Hessenberg Form Matters}\\
Upper Hessenberg matrices are almost triangular but still unitarily similar to the original matrix. This preserves spectral data while making iterative eigenvalue algorithms cheaper because only one subdiagonal must be tracked.
\end{perspective}
