\subsection{Matrix Classes and Invariants}

% **Role:** quantitative matrix behavior and computation.
%
% Trace, determinant, rank, spectral radius, PSD/PD matrices, idempotents, nilpotents.

Matrix classes package operator structure in coordinates. Trace, determinant, rank, positivity, idempotents, nilpotents, spectral radius, and structured products give computable invariants.

\begin{definition} \label{def:matrix-trace-and-determinant} \hlt{Trace and Determinant of a Matrix}\\
For $A=[A_{j,k}]\in\operatorname{Mat}(n,n,\mathbb{F})$, the \hlt{trace} of $A$ is $\operatorname{tr}(A)\defeq A_{1,1}+\cdots+A_{n,n}$.\\
Let $\mathfrak{S}_n$ denote the set of permutations of $\{1,\ldots,n\}$. The \hlt{sign} of $\sigma\in\mathfrak{S}_n$ is $\sign(\sigma)=1$ if the number of pairs $j<k$ with $\sigma(j)>\sigma(k)$ is even, and $\sign(\sigma)=-1$ otherwise.\\
The \hlt{determinant} of $A$ is $\det A \defeq \sum\limits_{\sigma\in\mathfrak{S}_n} \sign(\sigma)A_{\sigma(1),1}\cdots A_{\sigma(n),n}$.
\end{definition}

\begin{definition} \label{def:gram-determinant} \hlt{Gram Determinant}\\
Vectors $\mathbf{u}_1,\ldots,\mathbf{u}_n$ in inner-product space has \hlt{Gram determinant} $\operatorname{GramDet}(\mathbf{u}_1,\ldots,\mathbf{u}_n) \defeq \det \operatorname{Gram}(\mathbf{u}_1,\ldots,\mathbf{u}_n)$.
\end{definition}

\begin{example} \label{ex:frobenius-trace-riesz-vector} \hlt{Trace Functional and the Frobenius Inner Product}\\
On $\operatorname{Mat}(n,n,\mathbb{F})$ with the Frobenius inner product from Example \ref{ex:function-and-matrix-inner-products}, the Riesz vector of the trace functional $A\mapsto\operatorname{tr}(A)$ is $I_n$, since $\innerprod[F]{A,I_n}=\sum\limits_{j=1}^{n}a_{jj}=\operatorname{tr}(A)$.
\end{example}

\begin{proposition} \label{prop:trace-cyclicity-two-factors} \hlt{Trace Is Cyclic for Two Factors}\\
If $A\in\operatorname{Mat}(m,n,\mathbb{F})$ and $B\in\operatorname{Mat}(n,m,\mathbb{F})$, then $\operatorname{tr}(AB)=\operatorname{tr}(BA)$.
\begin{proof}
Expand both traces by entries: $\operatorname{tr}(AB) =\sum\limits_{j=1}^{m}\sum\limits_{k=1}^{n}A_{j,k}B_{k,j} =\sum\limits_{k=1}^{n}\sum\limits_{j=1}^{m}B_{k,j}A_{j,k} =\operatorname{tr}(BA)$.\\
\hlt{Structural Commentary}: Cyclicity is just reindexing the same double sum.\\
\hlt{Key Insight}: The two double sums contain exactly the same scalar terms.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:determinant-basic-laws} \hlt{Basic Determinant Laws}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\det I_n=1$.
\item $\det(A)$ is linear as a function of each column of $A$ when the other columns are fixed.
\item Swapping two columns negates the determinant; equal columns force $\det A=0$.
\item If $A$ is triangular, then $\det A=A_{1,1}\cdots A_{n,n}$.
\item $\det(AB)=\det(A)\det(B)$.
\item $A$ is invertible if and only if $\det A\neq0$. In that case $\det(A^{-1})=(\det A)^{-1}$.
\item $\det(A^t)=\det(A)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\det I_n=\sum\limits_{\sigma\in\mathfrak{S}_n}\sign(\sigma)(I_n)_{\sigma(1),1}\cdots (I_n)_{\sigma(n),n}=1$ as product is nonzero only when $\sigma(k)=k \ \forall k$.
\item For fixed $k$, $\det(A_{\cdot,1},\ldots,\mathbf{x},\ldots,A_{\cdot,n}) =\sum\limits_{\sigma\in\mathfrak{S}_n} \sign(\sigma)x_{\sigma(k)}\prod\limits_{\ell\neq k}A_{\sigma(\ell),\ell}$, which is linear in $\mathbf{x}$.
\item If $B_{\cdot,k}=A_{\cdot,\tau(k)}$ for transposition $\tau$, then $\det B=\sum\limits_{\rho\in\mathfrak{S}_n}\sign(\rho\tau)\prod\limits_{k=1}^n A_{\rho(k),k} =-\det A$.\\
If the swapped columns are equal, then $B=A$, so $\det A=-\det A$ and $\det A=0$.
\item If $A$ is upper triangular and $\prod_{k=1}^n A_{\sigma(k),k}\neq0$, then $\sigma(k)\leq k$ for all $k$, hence $\sigma(k)=k$ for all $k$.\\
Thus $\det A=A_{1,1}\cdots A_{n,n}$. If $A$ is lower triangular, nonzero products require $\sigma(k)\geq k$ for all $k$, again forcing $\sigma(k)=k$ for all $k$, so the same diagonal product formula holds.
\item If $F$ is alternating and multilinear in columns, then expanding each column in the standard basis gives\\
$F(A_{\cdot,1},\ldots,A_{\cdot,n}) =\sum\limits_{i_1,\ldots,i_n}A_{i_1,1}\cdots A_{i_n,n}F(\mathbf{e}_{i_1},\ldots,\mathbf{e}_{i_n}) =\sum\limits_{\sigma\in\mathfrak{S}_n}\sign(\sigma)F(I_n)A_{\sigma(1),1}\cdots A_{\sigma(n),n}$.\\
The terms with repeated $i_j$ vanish by alternation, and the surviving ordered bases differ from $(\mathbf{e}_1,\ldots,\mathbf{e}_n)$ by sign of permutation. For $H(B)=\det(AB)$, map $H$ is alternating multilinear and $H(I_n)=\det A$.\\
Hence $\det(AB)=H(B)=H(I_n)\det B=\det(A)\det(B)$.
\item If $A$ is invertible, then $1=\det I_n=\det(AA^{-1})=\det(A)\det(A^{-1})$.\\
If $A_{\cdot,k}=\sum\limits_{\ell\neq k}c_\ell A_{\cdot,\ell}$, then $\det A=\sum\limits_{\ell\neq k}c_\ell\det(A_{\cdot,1},\ldots,A_{\cdot,\ell},\ldots,A_{\cdot,\ell},\ldots,A_{\cdot,n})=0$.\\
Thus $\det A\neq0$ iff the columns of $A$ are independent, iff $A$ is invertible.
\item Using the permutation definition and replacing $\sigma$ by $\sigma^{-1}$,\\
$\det(A^t) =\sum\limits_{\sigma\in\mathfrak{S}_n}\sign(\sigma)A_{1,\sigma(1)}\cdots A_{n,\sigma(n)} =\sum\limits_{\tau\in\mathfrak{S}_n}\sign(\tau)A_{\tau(1),1}\cdots A_{\tau(n),n} =\det(A)$; $\sign(\tau^{-1})=\sign(\tau)$.
\end{enumerate}
\hlt{Structural Commentary}: The determinant is the normalized alternating multilinear measure of columns.\\
\hlt{Key Insight}: Multilinearity expands columns; alternation kills repeated directions.
\end{proof}
\end{theorem}

\begin{method} \label{meth:determinant-by-structure} \hlt{Determinant by Structure}\\
To compute or simplify a determinant:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item first look for triangular or block triangular form;
\item use column linearity to split complicated columns only when it reduces structure;
\item use column swaps to move useful pivots into position, tracking a factor of $-1$ for each swap;
\item detect zero determinants from dependent columns or repeated columns;
\item for products or changes of basis, use multiplicativity instead of expanding permutations.
\end{enumerate}
\end{method}

\begin{lemma} \label{lem:determinant-identity-direct-sum} \hlt{Identity Blocks Do Not Change Determinant}\\
Let $A\in\operatorname{Mat}(m,m,\mathbb{F})$ and let $n\geq0$. Then $\det(I_n\oplus A)=\det(A)=\det(A\oplus I_n)$.
\begin{proof}
In the permutation formula for $I_n\oplus A$, a nonzero term must satisfy $\sigma(k)=k$ for $1\leq k\leq n$. Remaining choices are exactly permutations of $\{n+1,\ldots,n+m\}$, after relabeling, gives $\det(A)$. Hence $\det(I_n\oplus A)=\det(A)$.\\
The same argument applied to the last $n$ coordinates gives $\det(A\oplus I_n)=\det(A)$.\\
The case $n=0$ is the empty identity block convention.\\
\hlt{Structural Commentary}: Identity blocks add directions without changing volume.\\
\hlt{Key Insight}: In the determinant expansion, the identity block forces the matching coordinates to be fixed.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:block-upper-triangular-determinant} \hlt{Determinant of an Upper Block-Triangular Matrix}\\
Let $A\in\operatorname{Mat}(r,r,\mathbb{F})$, $B\in\operatorname{Mat}(r,s,\mathbb{F})$, and $D\in\operatorname{Mat}(s,s,\mathbb{F})$. Then $\det
\begin{bmatrix}
A & B\\
0 & D
\end{bmatrix}
=\det(A)\det(D)$.
\begin{proof}
Factor $\begin{bmatrix}
A & B\\
0 & D
\end{bmatrix}
=
\begin{bmatrix}
I_r & 0\\
0 & D
\end{bmatrix}
\begin{bmatrix}
I_r & B\\
0 & I_s
\end{bmatrix}
\begin{bmatrix}
A & 0\\
0 & I_s
\end{bmatrix}$.\\
The first and last factors have determinants $\det(D)$ and $\det(A)$ by Lemma \ref{lem:determinant-identity-direct-sum}.\\
The middle factor is upper triangular with every diagonal entry equal to $1$, so its determinant is $1$.\\
Multiplicativity of determinant gives the formula.\\
\hlt{Structural Commentary}: Block upper triangular matrices separate determinant contribution into diagonal blocks. The off-diagonal block does not affect the determinant.\\
\hlt{Key Insight}: Factor the matrix into two block diagonal pieces and one unit upper triangular piece.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:schur-complement-determinant-formula} \hlt{Schur Complement Determinant Formula}\\
Let $M=
\begin{bmatrix}
A & B\\
C & D
\end{bmatrix}$, $A\in\operatorname{Mat}(r,r,\mathbb{F})$, $D\in\operatorname{Mat}(s,s,\mathbb{F})$.\\
If $A$ is invertible, then the \hlt{Schur complement of $A$ in $M$} is $M/A\defeq D-CA^{-1}B$, and $\det(M)=\det(A)\det(M/A)$.
\begin{proof}
Block Gaussian elimination gives $\begin{bmatrix}
I_r & 0\\
-CA^{-1} & I_s
\end{bmatrix}
\begin{bmatrix}
A & B\\
C & D
\end{bmatrix}
=
\begin{bmatrix}
A & B\\
0 & D-CA^{-1}B
\end{bmatrix}$.\\
The elimination matrix is lower block-triangular with identity diagonal blocks, so its determinant is $1$. Take determinants, apply Theorem \ref{thm:block-upper-triangular-determinant} to right-hand block triangular matrix, $\det(M)=\det(A)\det(D-CA^{-1}B)$.\\
\hlt{Structural Commentary}: Schur complementation is block Gaussian elimination.\\
\hlt{Key Insight}: Eliminating $C$ splits the determinant into $\det(A)$ times the remaining block determinant.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:schur-complement-lower-right-determinant} \hlt{Schur Complement of the Lower-Right Block}\\
Let $M=
\begin{bmatrix}
A & B\\
C & D
\end{bmatrix}$, $A\in\operatorname{Mat}(r,r,\mathbb{F})$, $D\in\operatorname{Mat}(s,s,\mathbb{F})$.\\
If $D$ is invertible, then the \hlt{Schur complement of $D$ in $M$} is $M/D\defeq A-BD^{-1}C$, and $\det(M)=\det(D)\det(M/D)$.
\begin{proof}
Block Gaussian elimination gives $\begin{bmatrix}
I_r & -BD^{-1}\\
0 & I_s
\end{bmatrix}
\begin{bmatrix}
A & B\\
C & D
\end{bmatrix}
=
\begin{bmatrix}
A-BD^{-1}C & 0\\
C & D
\end{bmatrix}$.\\
The elimination matrix has determinant $1$. The transpose of the matrix on the right is upper block-triangular, and $\det(X^t)=\det(X)$ by Theorem \ref{thm:determinant-basic-laws}. Therefore Theorem \ref{thm:block-upper-triangular-determinant} gives $\det(M)=\det(A-BD^{-1}C)\det(D)$.\\
\hlt{Structural Commentary}: This is the lower-right version of Schur complementation. Eliminating the upper-right block isolates the determinant contribution of $D$.\\
\hlt{Key Insight}: Eliminate by blocks, then read the determinant from the diagonal blocks.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:sylvester-determinant-identity} \hlt{Sylvester Determinant Identity}\\
Let $A\in\operatorname{Mat}(n,m,\mathbb{F})$ and $B\in\operatorname{Mat}(m,n,\mathbb{F})$. Then $\det(I_n+AB)=\det(I_m+BA)$.
\begin{proof}
Let $M=
\begin{bmatrix}
I_n & -A\\
B & I_m
\end{bmatrix}$.\\
Using Proposition \ref{prop:schur-complement-lower-right-determinant} with lower-right block $I_m$ gives $\det(M)=\det(I_n+AB)$. Using Proposition \ref{prop:schur-complement-determinant-formula} with upper-left block $I_n$ gives $\det(M)=\det(I_m+BA)$. Hence the two determinants are equal.\\
\hlt{Structural Commentary}: Two Schur complements of one block matrix give Sylvester's identity.\\
\hlt{Key Insight}: Compare the same determinant after eliminating different identity blocks.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:cauchy-binet-minors} \hlt{Cauchy--Binet Formula}\\
Let $A\in\operatorname{Mat}(k,n,\mathbb{F})$, $B\in\operatorname{Mat}(n,k,\mathbb{F})$, and $k\leq n$. For $J=\{j_1<\cdots<j_k\}\subseteq\{1,\ldots,n\}$, let $A^J$ use the columns of $A$ in $J$ and $B_J$ use the rows of $B$ in $J$.\\
Then $\det(AB)=\sum\limits_J \det(A^J)\det(B_J)$, where $J$ ranges over all $k$-subsets.
\begin{proof}
Put $S(A,B)\defeq \sum\limits_J \det(A^J)\det(B_J)$. For an elementary $n\times n$ matrix $E$, determinant multilinearity and row/column operation rules give $S(A,B)=S(AE,E^{-1}B)$, and $AB=(AE)(E^{-1}B)$.\\
If $\operatorname{rank}A<k$, both sides vanish. Otherwise reduce $A$ by elementary column operations to $A'=[C\ 0]$ with $C$ invertible, and transform $B$ oppositely to $B'$.\\
Then $AB=A'B'$ and $\det(A'B')=\det(C)\det(B'_{J_0})=S(A',B')$, $J_0=\{1,\ldots,k\}$, as every other $k$-minor of $A'$ is zero. Hence $\det(AB)=S(A,B)$.\\
\hlt{Structural Commentary}: Cauchy--Binet is determinant multiplicativity for rectangular middle dimensions.\\
\hlt{Key Insight}: The middle space contributes one $k$-coordinate shadow at a time.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:bordered-determinant-formulas} \hlt{Bordered Determinant Formulas}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$, let $\mathbf{x},\mathbf{y}\in\mathbb{F}^n$, and let $c\in\mathbb{F}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $c\neq0$, then $\det
\begin{bmatrix}
c & \mathbf{x}^t\\
\mathbf{y} & A
\end{bmatrix}
=
c\det\left(A-\frac{1}{c}\mathbf{y}\mathbf{x}^t\right)$.
\item If $A$ is invertible, then $\det
\begin{bmatrix}
A & \mathbf{x}\\
\mathbf{y}^t & c
\end{bmatrix}
=
\left(c-\mathbf{y}^tA^{-1}\mathbf{x}\right)\det(A)$
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item This is Proposition \ref{prop:schur-complement-determinant-formula} with the scalar block $c$ in the upper-left corner.
\item This is Proposition \ref{prop:schur-complement-determinant-formula} with $A$ as the upper-left block and $D=c$.
\end{enumerate}
\hlt{Structural Commentary}: Bordered determinants are Schur complements with scalar borders.\\
\hlt{Key Insight}: Treat the border as a $1\times1$ block and apply the Schur complement formula.
\end{proof}
\end{proposition}

\begin{definition} \label{def:minors-cofactors-and-adjugate} \hlt{Minors, Cofactors, and Adjugate}\\
For $A\in\operatorname{Mat}(n,n,\mathbb{F})$, let $M_{ij}$ delete row $i$ and column $j$ from $A$; set $\det(\emptyset_{0\times0})=1$. Define $C_{ij}\defeq(-1)^{i+j}\det(M_{ij})$.\\
Then $C=[C_{ij}]$ is the \hlt{cofactor matrix}, and $\operatorname{adj}(A)\defeq C^t$ is the \hlt{adjugate}.
\end{definition}

\begin{lemma} \label{lem:cofactor-expansion} \hlt{Cofactor Expansion}\\
For $A=[a_{ij}]\in\operatorname{Mat}(n,n,\mathbb{F})$, $\det A=\sum\limits_{k=1}^{n}a_{ik}C_{ik}$ and $\det A=\sum\limits_{k=1}^{n}a_{kj}C_{kj}$.
\begin{proof}
Using $\det A=\sum\limits_{\sigma\in S_n}\operatorname{sgn}(\sigma)\prod\limits_{r=1}^n a_{r,\sigma(r)}$, group terms by $\sigma(i)=k$.\\
For fixed $k$, the remaining permutation sum equals $(-1)^{i+k}\det(M_{ik})=C_{ik}$, giving the row formula. Applying this to $A^t$ and using Theorem \ref{thm:determinant-basic-laws} gives the column formula.\\
\hlt{Structural Commentary}: Cofactor expansion is the big determinant formula collected around one row or column.\\
\hlt{Key Insight}: Fix one entry; the rest is its minor.
\end{proof}
\end{lemma}

\begin{remark} \label{rem:bordered-determinant-adjugate-form} \hlt{Adjugate Form of the Bordered Formula}\\
Proposition \ref{prop:bordered-determinant-formulas} gives the adjugate form $\det
\begin{bmatrix}
A & \mathbf{x}\\
\mathbf{y}^t & c
\end{bmatrix}
=
c\det(A)-\mathbf{y}^t\operatorname{adj}(A)\mathbf{x}$.
No invertibility of $A$ is required.
\end{remark}

\begin{theorem} \label{thm:adjugate-identity-and-cramers-rule} \hlt{Adjugate Identity and Cramer's Rule}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ and let $\mathbf{b}\in\mathbb{F}^n$. Then $A\operatorname{adj}(A)=\operatorname{adj}(A)A=(\det A)I_n$. If $\det A\neq0$, then, for $A\mathbf{x}=\mathbf{b}$, $A^{-1}=\frac{1}{\det A}\operatorname{adj}(A)$ and $x_j=\frac{\det A_j(\mathbf{b})}{\det A}$ for $1\leq j\leq n$, where $A_j(\mathbf{b})$ replaces column $j$ of $A$ by $\mathbf{b}$.
\begin{proof}
Since $(\operatorname{adj}A)_{kj}=C_{jk}$, $(A\operatorname{adj}A)_{ij}=\sum\limits_k a_{ik}C_{jk}$.\\
For $i=j$ this is $\det A$ by Lemma \ref{lem:cofactor-expansion}; for $i\neq j$ it is the cofactor expansion of a matrix with rows $i$ and $j$ equal, hence $0$. Thus $A\operatorname{adj}(A)=(\det A)I_n$. The same argument applied to $A^t$ gives $\operatorname{adj}(A)A=(\det A)I_n$. If $\det A\neq0$, division gives the inverse formula. Finally, $(\operatorname{adj}A\,\mathbf{b})_j=\sum_i C_{ij}b_i=\det A_j(\mathbf{b})$, by expansion down column $j$, so $x=A^{-1}\mathbf{b}$ gives Cramer's rule.\\
\hlt{Structural Commentary}: Cofactors assemble all one-column replacements of $A$ into a single inverse formula.\\
\hlt{Key Insight}: Off-diagonal products vanish because they are determinants with two equal rows or columns.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:similarity-preserves-trace-and-determinant} \hlt{Trace and Determinant Are Similarity Invariants}\\
If $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$ are similar, then $\operatorname{tr}(A)=\operatorname{tr}(B)$, $\det(A)=\det(B)$.
\begin{proof}
Write $B=PAP^{-1}$ with $P$ invertible.\\
Proposition \ref{prop:trace-cyclicity-two-factors} gives $\operatorname{tr}(B)=\operatorname{tr}\bigl(P(AP^{-1})\bigr) =\operatorname{tr}\bigl((AP^{-1})P\bigr) =\operatorname{tr}(A)$.\\
Theorem \ref{thm:determinant-basic-laws} gives $\det(B)=\det(P)\det(A)\det(P^{-1})=\det(A)$.\\
\hlt{Structural Commentary}: Trace and determinant ignore the coordinate system used to write an operator.\\
\hlt{Key Insight}: Conjugating by $P$ inserts $P$ and $P^{-1}$, which cancel inside trace and determinant.
\end{proof}
\end{theorem}

\begin{principle} \label{princ:trace-determinant-coordinate-free} \hlt{Trace and Determinant Are Coordinate-Free}\\
Trace and determinant are first computed from a matrix, but their real structural meaning is operator-level: changing basis conjugates the matrix, and conjugation preserves both quantities.
\end{principle}

\begin{theorem} \label{thm:operator-matrix-trace-determinant-basis-independent} \hlt{Trace and Determinant of an Operator Are Basis-Independent}\\
Let $\mathcal{V}$ be finite-dimensional, let $T\in\LL(\mathcal{V})$, and let $\beta,\gamma$ be bases of $\mathcal{V}$.\\
Then $\operatorname{tr}\bigl([T]_{\beta\leftarrow\beta}\bigr) =\operatorname{tr}\bigl([T]_{\gamma\leftarrow\gamma}\bigr)$, $\det\bigl([T]_{\beta\leftarrow\beta}\bigr) =\det\bigl([T]_{\gamma\leftarrow\gamma}\bigr)$.
\begin{proof}
By Theorem \ref{thm:change-of-basis-formula}, the two matrices are similar: $[T]_{\gamma\leftarrow\gamma} =C_{\gamma\leftarrow\beta}[T]_{\beta\leftarrow\beta}C_{\beta\leftarrow\gamma}$.\\
Theorem \ref{thm:similarity-preserves-trace-and-determinant} now gives equality of traces and determinants.\\
\hlt{Structural Commentary}: Basis changes conjugate matrices; trace and determinant remain operator invariants.\\
\hlt{Key Insight}: Similar matrices have the same trace and determinant.
\end{proof}
\end{theorem}

\begin{definition} \label{def:operator-trace-and-determinant} \hlt{Trace and Determinant of an Operator}\\
Let $\mathcal{V}$ be finite-dim, let $T\in\LL(\mathcal{V})$. Choose any basis $\beta$ of $\mathcal{V}$. Define $\operatorname{tr}T\defeq\operatorname{tr}\bigl([T]_{\beta\leftarrow\beta}\bigr)$, $\det T\defeq\det\bigl([T]_{\beta\leftarrow\beta}\bigr)$.\\
Theorem \ref{thm:operator-matrix-trace-determinant-basis-independent} shows these definitions do not depend on chosen basis.
\end{definition}

\begin{proposition} \label{prop:isometry-determinant-absolute-one} \hlt{Unitary Matrices and Isometries Have Determinant of Modulus One}\\
Let $U\in\operatorname{Mat}(n,n,\mathbb{F})$ be unitary. Then $\abs{\det U}=1$. More generally, if $S\in\LL(\mathcal{V})$ is an isometry on a finite-dimensional inner-product space over $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, then $\abs{\det S}=1$.
\begin{proof}
The permutation definition gives $\det(U^*)=\overline{\det U}$. By Theorem \ref{thm:determinant-basic-laws}, taking determinants in $U^*U=I_n$ gives $1=\det(U^*)\det(U)=\overline{\det U}\det U=\abs{\det U}^2$. Hence $\abs{\det U}=1$.\\
For an isometry $S$, choose an orthonormal basis and let $U$ be the matrix of $S$ in that basis. Then $U$ is unitary by Theorem \ref{thm:finite-dimensional-isometry-iff-adjoint-product-identity}, and $\det S=\det U$.\\
\hlt{Structural Commentary}: In orthonormal coordinates, isometries are unitary or orthogonal, so the determinant calculation is the matrix one.\\
\hlt{Key Insight}: Determinants turn $U^*U=I$ into $\abs{\det U}^2=1$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:operator-trace-determinant-characteristic-coefficients} \hlt{Trace and Determinant from $q_T$}\\
Let $\mathcal{V}\neq\{\mathbf{0}\}$ be finite-dim over $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, let $T\in\LL(\mathcal{V})$, $q_T(z)=z^n+c_{n-1}z^{n-1}+\cdots+c_1z+c_0$, $n=\dim\mathcal{V}$.\\
Then $\operatorname{tr}T=-c_{n-1}$ and $\det T=(-1)^nc_0$. If $\mathbb{F}=\mathbb{C}$ and $\lambda_1,\ldots,\lambda_n$ list the eigenvalues of $T$ with algebraic multiplicity, then $\operatorname{tr}T=\lambda_1+\cdots+\lambda_n$, $\det T=\lambda_1\cdots\lambda_n$.
\begin{proof}
Choose a basis $\beta$ and set $A=[T]_{\beta\leftarrow\beta}$.\\
Then $q_T=p_A$, and by Definition \ref{def:operator-trace-and-determinant}, $\operatorname{tr}T=\operatorname{tr}A$ and $\det T=\det A$.\\
Theorem \ref{thm:matrix-characteristic-polynomial-basic-coefficients} gives $\operatorname{tr}T=-c_{n-1}$ and $\det T=(-1)^nc_0$.\\
If $\mathbb{F}=\mathbb{C}$, then comparing $q_T(z)=(z-\lambda_1)\cdots(z-\lambda_n)$ with the displayed form of $q_T$ gives the formulas.\\
\hlt{Structural Commentary}: Characteristic polynomial stores trace and determinant in first and last coefficients.\\
\hlt{Key Insight}: Compare coefficients of $\det(zI-T)$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:trace-determinant-from-eigenvalues} \hlt{Trace and Determinant from Eigenvalues}\\
Let $\lambda_1,\ldots,\lambda_n$ list the eigenvalues of $A\in\operatorname{Mat}(n,n,\mathbb{C})$ according to $m_A$. Then $\operatorname{tr}A=\sum\limits_{j=1}^{n}\lambda_j$ and $\det A=\prod\limits_{j=1}^{n}\lambda_j$.
\begin{proof}
By Fundamental Theorem of Algebra, $p_A(z)=\prod\limits_{j=1}^{n}(z-\lambda_j) =z^n-\left(\sum\limits_{j=1}^{n}\lambda_j\right)z^{n-1}+\cdots+(-1)^n\prod\limits_{j=1}^{n}\lambda_j$.\\
Theorem \ref{thm:matrix-characteristic-polynomial-basic-coefficients} gives $p_A(z)=z^n-(\operatorname{tr}A)z^{n-1}+\cdots+(-1)^n\det A$. Uniqueness of coefficients gives the identities.\\
\hlt{Structural Commentary}: Coefficients of $p_A$ encode symmetric functions of the eigenvalues.\\
\hlt{Key Insight}: Trace is the first root sum and determinant is the full root product.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:transpose-conjugate-adjoint-characteristic-polynomial} \hlt{Transpose, Conjugate, and Adjoint Spectra}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $p_{A^t}(z)=p_A(z)$, $p_{\overline{A}}(z)=\overline{p_A(\overline{z})}$, and $p_{A^*}(z)=\overline{p_A(\overline{z})}$.\\
Consequently, if $\lambda\in\operatorname{spec}(A)$, then $m_{A^t}(\lambda)=m_A(\lambda)$ and $m_{\overline{A}}(\overline{\lambda})=m_{A^*}(\overline{\lambda})=m_A(\lambda)$.
\begin{proof}
Use determinant identities:
\begin{align}
p_{A^t}(z)&=\det(zI_n-A^t)=\det((zI_n-A)^t)=p_A(z), \nonumber\\
p_{\overline{A}}(z)&=\det(zI_n-\overline{A}) =\det(\overline{\overline{z}I_n-A}) =\overline{\det(\overline{z}I_n-A)} =\overline{p_A(\overline{z})}.
\nonumber
\end{align}
Since $A^*=\overline{A}^{\,t}$, the first identity applied to $\overline{A}$ gives the formula for $A^*$.\\
Multiplicity is preserved because conjugating a polynomial sends a factor $(z-\lambda)^k$ to $(z-\overline{\lambda})^k$.\\
\hlt{Structural Commentary}: Transpose preserves $p_A$; conjugation conjugates its roots.\\
\hlt{Key Insight}: The adjoint combines conjugation with transposition, so its spectrum is conjugated.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:transpose-preserves-spectrum-and-geometric-multiplicity} \hlt{Transpose Preserves Spectrum and Geometric Multiplicity}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ and $\lambda\in\mathbb{F}$. Then $\lambda\in\operatorname{spec}(A)$ $\Longleftrightarrow$ $\lambda\in\operatorname{spec}(A^t)$, and $\dim E_{\lambda}(A)=\dim E_{\lambda}(A^t)$.
\begin{proof}
Since $(A-\lambda I_n)^t=A^t-\lambda I_n$, Theorem \ref{thm:row-rank-equals-column-rank} gives $\operatorname{rank}(A-\lambda I_n)=\operatorname{rank}(A^t-\lambda I_n)$.\\
Matrix rank-nullity then gives $\dim E_{\lambda}(A) =n-\operatorname{rank}(A-\lambda I_n) =n-\operatorname{rank}(A^t-\lambda I_n) =\dim E_{\lambda}(A^t)$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\lambda\in\operatorname{spec}(A)$, then $E_{\lambda}(A)\neq\{\mathbf{0}\}$, so $E_{\lambda}(A^t)\neq\{\mathbf{0}\}$ and $\lambda\in\operatorname{spec}(A^t)$.
\item[$(\Leftarrow)$] The same argument with $A$ and $A^t$ interchanged gives the converse.
\end{enumerate}
\hlt{Structural Commentary}: Transposition preserves rank, so shifted kernels have the same dimension.\\
\hlt{Key Insight}: Spectrum and geometric multiplicity are kernel facts for $A-\lambda I_n$.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:square-matrix-algebraic-invertibility-dictionary} \hlt{Square Matrix Algebraic Invertibility Dictionary}\\
Let $A=[\mathbf{a}_1\ \cdots\ \mathbf{a}_n]\in\operatorname{Mat}(n,n,\mathbb{F})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ is invertible;
\item $\mathbf{a}_1,\ldots,\mathbf{a}_n$ form a basis of $\mathbb{F}^n$;
\item $\mathbf{a}_1,\ldots,\mathbf{a}_n$ are linearly independent;
\item $\mathbf{a}_1,\ldots,\mathbf{a}_n$ span $\mathbb{F}^n$;
\item for every $\mathbf{b}\in\mathbb{F}^n$, the system $A\mathbf{x}=\mathbf{b}$ has a unique solution;
\item the homogeneous system $A\mathbf{x}=\mathbf{0}$ has only the zero solution.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ By Theorem \ref{thm:invertible-matrix-columns-basis}, the columns of an invertible matrix form a basis.
\item[$(ii)$]$\Rightarrow (iii)$ A basis is linearly independent.
\item[$(iii)$]$\Rightarrow (iv)$ In $\mathbb{F}^n$, $n$ linearly independent vectors form a basis, hence span.
\item[$(iv)$]$\Rightarrow (v)$ In $\mathbb{F}^n$, $n$ spanning vectors form a basis, so every $\mathbf{b}$ has unique coordinates $\mathbf{x}$ in $A\mathbf{x}=\mathbf{b}$.
\item[$(v)$]$\Rightarrow (vi)$ Apply (v) to $\mathbf{b}=\mathbf{0}$; since $\mathbf{x}=\mathbf{0}$ is a solution, it is the only one.
\item[$(vi)$]$\Rightarrow (i)$ Condition (vi) says the columns are independent.\\
Hence they form a basis of $\mathbb{F}^n$, and Corollary \ref{cor:basis-columns-form-invertible-matrix} gives invertibility.
\end{enumerate}
\hlt{Structural Commentary}: Square invertibility is column basishood.\\
\hlt{Key Insight}: Invertible columns give unique coordinates.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:matrix-nonsingularity-rank-determinant-spectrum-dictionary} \hlt{Nonsingularity by Rank, Determinant, and Spectrum}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ is invertible;
\item $\operatorname{rank}(A)=n$;
\item $\det A\neq0$;
\item $0\notin\operatorname{spec}(A)$.
\end{enumerate}
Over $\mathbb{C}$, this is also equivalent to all eigenvalues of $A$ being nonzero.
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ This is Corollary \ref{cor:square-matrix-invertible-iff-full-rank}.
\item[$(ii)$]$\Rightarrow (iii)$ Full rank gives invertibility by Corollary \ref{cor:square-matrix-invertible-iff-full-rank}, hence $\det A\neq0$ by Theorem \ref{thm:determinant-basic-laws}.
\item[$(iii)$]$\Rightarrow (iv)$ If $\det A\neq0$, then $A$ is invertible by Theorem \ref{thm:determinant-basic-laws}; hence $0\notin\operatorname{spec}(A)$ by Corollary \ref{cor:matrix-invertible-iff-zero-not-spectrum}.
\item[$(iv)$]$\Rightarrow (i)$ This is Corollary \ref{cor:matrix-invertible-iff-zero-not-spectrum}.
\end{enumerate}
Over $\mathbb{C}$, Theorem \ref{thm:matrix-characteristic-polynomial-basic-coefficients} identifies $\lambda_1,\ldots,\lambda_n$ as the roots of $p_A$, so (iv) is equivalent to $\lambda_j\neq0$ for all $j$.\\
\hlt{Structural Commentary}: Rank, determinant, and spectrum detect singularity.\\
\hlt{Key Insight}: Singularity can be seen as rank loss, zero determinant, or the zero eigenvalue.
\end{proof}
\end{corollary}

\begin{method} \label{meth:trace-determinant-invariant-computation} \hlt{Trace--Determinant Invariant Method}\\
To get eigenvalue data without solving for eigenvalues:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item choose a basis in which $[T]_{\beta\leftarrow\beta}$ is easiest to compute;
\item compute $\operatorname{tr}T$ and $\det T$ from this matrix;
\item if the eigenvalues are $\lambda_1,\ldots,\lambda_n$, conclude $\lambda_1+\cdots+\lambda_n=\operatorname{tr}T$, and $\lambda_1\cdots\lambda_n=\det T$;
\item for equations $ST-TS=A$, first test $\operatorname{tr}A=0$; otherwise no solution exists.
\end{enumerate}
\end{method}

\begin{theorem} \label{thm:full-rank-products-preserve-rank} \hlt{Full-Rank Products Preserve Rank}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$. If $X\in\operatorname{Mat}(p,m,\mathbb{F})$ has full column rank and $Y\in\operatorname{Mat}(n,q,\mathbb{F})$ has full row rank, then $\operatorname{rank}(A)=\operatorname{rank}(XAY)$. This holds when $X$ and $Y$ are square invertible matrices of appropriate sizes.
\begin{proof}
Because $X$ has full column rank, the induced map $L_X:\mathbb{F}^m\to\mathbb{F}^p$, $L_X(\mathbf{u})=X\mathbf{u}$, has kernel $\{\mathbf{0}\}$.\\
Thus $\mathbf{z}\in\ker(XA) \iff X(A\mathbf{z})=\mathbf{0} \iff A\mathbf{z}=\mathbf{0} \iff \mathbf{z}\in\ker(A)$.\\
Since $XA$ and $A$ have the same number of columns, Corollary \ref{cor:matrix-rank-nullity} gives $\operatorname{rank}(XA)=\operatorname{rank}(A)$.\\
Because $Y$ has full row rank, $Y^t$ has full column rank. Applying the left-multiplication result to $A^t$ and $Y^t$, then using Theorem \ref{thm:row-rank-equals-column-rank}, gives $\operatorname{rank}(AY)=\operatorname{rank}((AY)^t) =\operatorname{rank}(Y^tA^t) =\operatorname{rank}(A^t) =\operatorname{rank}(A)$.\\
Applying the left-multiplication result once more to $AY$ gives $\operatorname{rank}(XAY)=\operatorname{rank}(AY)=\operatorname{rank}(A)$.\\
\hlt{Structural Commentary}: Multiplication by a full-column-rank matrix on the left loses no output information, while full row rank on the right loses no input information.\\
\hlt{Key Insight}: Rank changes only when multiplication introduces a new kernel or collapses the row side.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:rank-product-and-concatenation-bounds} \hlt{Rank Bounds for Products and Concatenations}\\
Let $A\in\operatorname{Mat}(m,k,\mathbb{F})$, $B\in\operatorname{Mat}(k,n,\mathbb{F})$, and $C\in\operatorname{Mat}(m,p,\mathbb{F})$. Then $\operatorname{rank}(AB)\leq \min(\operatorname{rank}(A),\operatorname{rank}(B))$ and $\operatorname{rank}(AB)=\operatorname{rank}(A)$ if and only if $\operatorname{col}(AB)=\operatorname{col}(A)$.\\
Also, $\max(\operatorname{rank}(A),\operatorname{rank}(C))\leq\operatorname{rank}\!\begin{bmatrix}A & C\end{bmatrix}$ and $\operatorname{rank}\!\begin{bmatrix}A & C\end{bmatrix}=\operatorname{rank}(A)$ if and only if $\operatorname{col}(A)=\operatorname{col}(A)+\operatorname{col}(C)$.
\begin{proof}
Each column of $AB$ is a linear combination of the columns of $A$, so $\operatorname{col}(AB)\subseteq\operatorname{col}(A)$. Therefore $\operatorname{rank}(AB)\leq\operatorname{rank}(A)$; since both spaces are finite-dimensional, equality holds exactly when $\operatorname{col}(AB)=\operatorname{col}(A)$. Applying this same argument to $(AB)^t=B^tA^t$ and using Theorem \ref{thm:row-rank-equals-column-rank} gives $\operatorname{rank}(AB)\leq\operatorname{rank}(B)$.\\
For the block matrix, its columns are precisely the columns of $A$ and $C$, hence $\operatorname{col}\!\begin{bmatrix}A & C\end{bmatrix} =\operatorname{col}(A)+\operatorname{col}(C)$.\\
This space contains both $\operatorname{col}(A)$ and $\operatorname{col}(C)$, giving the lower bound. Equality with $\operatorname{rank}(A)$ occurs exactly when the containing subspace has the same dimension as $\operatorname{col}(A)$, equivalently when $\operatorname{col}(A)+\operatorname{col}(C)=\operatorname{col}(A)$.\\
\hlt{Structural Commentary}: Products can only reduce accessible column directions, while horizontal concatenation can only add column directions.\\
\hlt{Key Insight}: Rank tracks whether a matrix operation preserves, loses, or adds column-space dimension.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:sylvester-rank-inequality} \hlt{Sylvester Rank Inequality}\\
Let $A\in\operatorname{Mat}(m,k,\mathbb{F})$ and $B\in\operatorname{Mat}(k,n,\mathbb{F})$. Then $\operatorname{rank}(A)+\operatorname{rank}(B)-k\leq\operatorname{rank}(AB)$.
\begin{proof}
Let $R=\operatorname{col}(B)\subseteq\mathbb{F}^k$. The columns of $AB$ are $A$ applied to the columns of $B$, so $\operatorname{col}(AB)=A(R)$.\\
Apply rank-nullity to the restriction $T_A|_R:R\to\mathbb{F}^m$. Its range is $A(R)=\operatorname{col}(AB)$ and its kernel is $R\cap\ker(A)$.\\
Hence $\operatorname{rank}(AB) =\dim(A(R)) =\dim(R)-\dim(R\cap\ker(A)) \geq \operatorname{rank}(B)-\dim(\ker(A))$.\\
Since $A$ has $k$ columns, Corollary \ref{cor:matrix-rank-nullity} gives $\dim(\ker(A))=k-\operatorname{rank}(A)$. Substitution gives desired inequality.\\
\hlt{Structural Commentary}: Multiplication by $A$ can lose only the part of $\operatorname{col}(B)$ lying in $\ker(A)$. Sylvester's inequality is this loss estimate in dimensions.\\
\hlt{Key Insight}: The shared middle dimension $k$ bounds how much rank can disappear in the product.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:frobenius-rank-inequality} \hlt{Frobenius Rank Inequality}\\
Let $A\in\operatorname{Mat}(m,k,\mathbb{F})$, $B\in\operatorname{Mat}(k,p,\mathbb{F})$, and $C\in\operatorname{Mat}(p,n,\mathbb{F})$.\\
Then $\operatorname{rank}(AB)+\operatorname{rank}(BC) \leq \operatorname{rank}(B)+\operatorname{rank}(ABC)$.
\begin{proof}
Let $R=\operatorname{col}(B)$ and $S=\operatorname{col}(BC)$. Since every column of $BC$ is a linear combination of columns of $B$, we have $S\subseteq R$. The induced map $T_A$ maps $R$ onto $\operatorname{col}(AB)$ and maps $S$ onto $\operatorname{col}(ABC)$.\\
Rank-nullity on the restrictions $T_A|_R$ and $T_A|_S$ gives
\begin{align}
\operatorname{rank}(AB)&=\dim R-\dim(R\cap\ker(A)), \quad \operatorname{rank}(ABC)=\dim S-\dim(S\cap\ker(A)). \nonumber
\end{align}
Since $S\cap\ker(A)\subseteq R\cap\ker(A)$, we have $\dim(S\cap\ker(A))\leq\dim(R\cap\ker(A))$.\\
Thus $\dim R-\dim(R\cap\ker(A))+\dim S \leq \dim R+\dim S-\dim(S\cap\ker(A))$.\\
Substituting $\dim R=\operatorname{rank}(B)$ and $\dim S=\operatorname{rank}(BC)$ gives the stated inequality.\\
\hlt{Structural Commentary}: Frobenius tracks $A$ on nested column spaces.\\
\hlt{Key Insight}: The smaller column space cannot lose more dimensions to $\ker(A)$ than the larger one.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:frobenius-rank-inequality-special-cases} \hlt{Rank Inequalities from Frobenius}\\
The following are special cases of Theorem \ref{thm:frobenius-rank-inequality}:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
$\operatorname{rank}(AB)\leq\min(\operatorname{rank}(A),\operatorname{rank}(B))$;
\item
$\operatorname{rank}(A)+\operatorname{rank}(B)-k\leq\operatorname{rank}(AB)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
Apply Frobenius to conformable factors $X,Y,Z$. Taking $Z=0$ gives $\operatorname{rank}(XY)\leq\operatorname{rank}(Y)$, because $\operatorname{rank}(Y0)=\operatorname{rank}(XY0)=0$. Taking $X=0$ gives $\operatorname{rank}(YZ)\leq\operatorname{rank}(Y)$.\\
Renaming the factors gives $\operatorname{rank}(AB)\leq\operatorname{rank}(B)$ and $\operatorname{rank}(AB)\leq\operatorname{rank}(A)$.
\item
Taking the middle matrix to be $I_k$ gives $\operatorname{rank}(A)+\operatorname{rank}(B)\leq k+\operatorname{rank}(AB)$, which is Sylvester's inequality.
\end{enumerate}
\hlt{Structural Commentary}: Frobenius is the three-factor parent inequality. The usual upper and lower product-rank bounds appear by making one factor zero or identity.\\
\hlt{Key Insight}: Zero factors isolate upper bounds; an identity middle factor yields the Sylvester lower bound.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:sylvester-law-of-nullity} \hlt{Sylvester's Law of Nullity}\\
For a matrix $X$, let $\operatorname{nullity}(X)=\dim(\ker(X))$; this is the dimension of its nullspace.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A\in\operatorname{Mat}(m,k,\mathbb{F})$ and $B\in\operatorname{Mat}(k,n,\mathbb{F})$, then $\operatorname{nullity}(AB)\leq\operatorname{nullity}(A)+\operatorname{nullity}(B)$.
\item
If $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$, then $\max(\operatorname{nullity}(A),\operatorname{nullity}(B))\leq\operatorname{nullity}(AB)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item
By rank-nullity and Theorem \ref{thm:sylvester-rank-inequality},
\begin{align*}
\operatorname{nullity}(AB) &=n-\operatorname{rank}(AB) \leq n-\operatorname{rank}(A)-\operatorname{rank}(B)+k \\
&=(k-\operatorname{rank}(A))+(n-\operatorname{rank}(B)) =\operatorname{nullity}(A)+\operatorname{nullity}(B).
\end{align*}
\item For square $A,B$, $\operatorname{rank}(AB)\leq\operatorname{rank}(A)$, $\operatorname{rank}(AB)\leq\operatorname{rank}(B)$. Subtract from $n$ gives desired lower bounds.
\end{enumerate}
\hlt{Structural Commentary}: Nullity is rank loss from the domain.\\
\hlt{Key Insight}: Use rank-nullity to translate rank bounds into kernel bounds.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:rank-triangle-inequality} \hlt{Rank Triangle Inequality}\\
If $A,B\in\operatorname{Mat}(m,n,\mathbb{F})$, then $\abs{\operatorname{rank}(A)-\operatorname{rank}(B)} \leq \operatorname{rank}(A+B) \leq \operatorname{rank}(A)+\operatorname{rank}(B)$.
\begin{proof}
Since $\operatorname{col}(A+B)\subseteq\operatorname{col}(A)+\operatorname{col}(B)$, Theorem \ref{thm:dimension-formula-for-sum} gives $\operatorname{rank}(A+B)\leq\dim(\operatorname{col}(A)+\operatorname{col}(B))\leq\operatorname{rank}(A)+\operatorname{rank}(B)$.
Applying this upper bound to $A=(A+B)-B$ gives $\operatorname{rank}(A)\leq\operatorname{rank}(A+B)+\operatorname{rank}(B)$, so $\operatorname{rank}(A)-\operatorname{rank}(B)\leq\operatorname{rank}(A+B)$. Applying it to $B=(A+B)-A$ gives $\operatorname{rank}(B)-\operatorname{rank}(A)\leq\operatorname{rank}(A+B)$.\\
Together these are exactly $\abs{\operatorname{rank}(A)-\operatorname{rank}(B)}\leq\operatorname{rank}(A+B)$.\\
\hlt{Structural Commentary}: The rank of a sum is controlled by the sum of the two column spaces. The reverse bounds come by solving for either summand.\\
\hlt{Key Insight}: Rank behaves like a dimension-valued size under addition.
\end{proof}
\end{theorem}

\begin{definition} \hlt{Commutator}\\
For $S,T\in\LL(\mathcal{V})$, the \hlt{commutator} of $S$ and $T$ is $[S,T]\defeq ST-TS$.\\
For $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$, the \hlt{matrix commutator} is $[A,B]\defeq AB-BA$.
\end{definition}

\begin{proposition} \label{prop:commutators-have-zero-trace} \hlt{Trace Obstruction for Commutators}\\
If $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$, then $\operatorname{tr}(AB-BA)=0$. Hence any matrix similar to a commutator has trace zero.
\begin{proof}
By linearity of trace and Proposition \ref{prop:trace-cyclicity-two-factors}, $\operatorname{tr}(AB-BA)=\operatorname{tr}(AB)-\operatorname{tr}(BA)=0$.\\
If $C$ is similar to $AB-BA$, then Theorem \ref{thm:similarity-preserves-trace-and-determinant} gives $\operatorname{tr}(C)=\operatorname{tr}(AB-BA)=0$.\\
\hlt{Structural Commentary}: Trace kills commutators because it is cyclic.\\
\hlt{Key Insight}: $AB$ and $BA$ have the same trace, so their difference has trace zero.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:scalar-matrix-dependent-orbits} \hlt{Scalar Matrices and One-Step Orbits}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ with $n\geq2$.\\
Then $(\mathbf{x},A\mathbf{x})$ is linearly dependent for every $\mathbf{x}\in\mathbb{F}^n$ if and only if $A$ is a scalar matrix.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $A=cI_n$. Then $A\mathbf{x}=c\mathbf{x}$ for every $\mathbf{x}$, so $(\mathbf{x},A\mathbf{x})$ is linearly dependent.
\item[$(\Leftarrow)$] Suppose $(\mathbf{x},A\mathbf{x})$ is linearly dependent for every $\mathbf{x}$.\\
For each standard basis vector $\mathbf{e}_i$, $\exists a_i\in\mathbb{F}$ such that $A\mathbf{e}_i=a_i\mathbf{e}_i$. For $i\geq2$, dependence of $\mathbf{e}_1+\mathbf{e}_i$ and $A(\mathbf{e}_1+\mathbf{e}_i)=a_1\mathbf{e}_1+a_i\mathbf{e}_i$ means $a_1\mathbf{e}_1+a_i\mathbf{e}_i=b(\mathbf{e}_1+\mathbf{e}_i)$ for some $b\in\mathbb{F}$. Comparing the independent vectors $\mathbf{e}_1,\mathbf{e}_i$ gives $a_1=b=a_i$. Hence every $a_i$ equals $a_1$, so $A=a_1I_n$.
\end{enumerate}
\hlt{Structural Commentary}: A scalar matrix preserves every one-dimensional coordinate direction and every diagonal line between two coordinates. These diagonal lines force all coordinate scalars to agree.\\
\hlt{Key Insight}: If every vector is an eigenvector, then there can be only one eigenvalue.
\end{proof}
\end{lemma}

\begin{lemma} \label{lem:nonscalar-similar-zero-entry} \hlt{Nonscalar Matrices Have a Zero Diagonal Entry After Similarity}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ with $n\geq2$.\\
If $A$ is not a scalar matrix, then $A$ is similar over $\mathbb{F}$ to a matrix whose $(1,1)$ entry is $0$.
\begin{proof}
By Lemma \ref{lem:scalar-matrix-dependent-orbits}, there is $\mathbf{x}\in\mathbb{F}^n$ such that $(\mathbf{x},A\mathbf{x})$ is linearly independent. Extend $(\mathbf{x},A\mathbf{x})$ to a basis $\beta$ of $\mathbb{F}^n$. In the basis $\beta$, the first column of the matrix of $A$ is the coordinate vector of $A\mathbf{x}$, namely $(0,1,0,\ldots,0)^t$. Thus the $(1,1)$ entry is $0$, and this matrix is similar to $A$.\\
\hlt{Structural Commentary}: Nonscalarity produces a vector whose first iterate points in a genuinely new direction. Using these two vectors as the start of a basis forces a zero in the first diagonal position.\\
\hlt{Key Insight}: Put $\mathbf{x}$ first and $A\mathbf{x}$ second in the basis.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:similar-to-constant-diagonal} \hlt{Similarity to a Constant-Diagonal Matrix}\\
Let $n\geq1$, let $A\in\operatorname{Mat}(n,n,\mathbb{F})$, and assume $k\cdot1_{\mathbb{F}}\neq0$ for every $1\leq k\leq n$. Then $A$ is similar over $\mathbb{F}$ to a matrix whose diagonal entries are all equal to $\frac{1}{n}\operatorname{tr}(A)$.
\begin{proof}
Set $\tau=\frac{1}{n}\operatorname{tr}(A)$ and $B=A-\tau I_n$. Then $\operatorname{tr}(B)=0$. It is enough to show that every trace-zero matrix is similar to a matrix with zero diagonal, because Theorem \ref{thm:similarity-preserves-scalar-shifts} then adds $\tau I_n$ back to the diagonal.\\
To prove the trace-zero claim by induction on $n$. If $n=1$, a trace-zero matrix is $[0]$. Let $n\geq2$ and assume the claim holds for smaller sizes. Let $\operatorname{tr}(B)=0$. If $B$ is scalar, then $B=cI_n$ and $0=\operatorname{tr}(B)=nc$.\\
Since $n\cdot1_{\mathbb{F}}\neq0$, we get $c=0$, so $B$ already has zero diagonal.\\
If $B$ is not scalar, Lemma \ref{lem:nonscalar-similar-zero-entry} makes $B$ similar to $\begin{bmatrix}
0 & *\\
* & B_1
\end{bmatrix}$, and $\operatorname{tr}(B_1)=0$.\\
By induction, choose an invertible $S_1$ such that $S_1^{-1}B_1S_1$ has zero diagonal. Conjugating the displayed block matrix by $1\oplus S_1$ leaves the first diagonal entry equal to $0$ and replaces the lower-right block by $S_1^{-1}B_1S_1$.\\
Hence the resulting similar matrix has every diagonal entry equal to $0$.\\
\hlt{Structural Commentary}: The proof removes one zero diagonal entry at a time.\\
\hlt{Key Insight}: Similarity creates the first zero; trace zero handles the smaller block.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:shoda-trace-zero-commutators} \hlt{Shoda's Theorem in the Distinct-Diagonal Case}\\
Let $n\geq1$ and assume $k\cdot1_{\mathbb{F}}\neq0$ for every $1\leq k\leq n$. For $A\in\operatorname{Mat}(n,n,\mathbb{F})$, the matrix $A$ is a commutator in $\operatorname{Mat}(n,n,\mathbb{F})$ if and only if $\operatorname{tr}(A)=0$. In particular, no nonzero scalar matrix is a commutator.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Proposition \ref{prop:commutators-have-zero-trace} gives $\operatorname{tr}(A)=0$.\\
Thus $cI_n$ can be a commutator only when $nc=\operatorname{tr}(cI_n)=0$, hence only when $c=0$.
\item[$(\Leftarrow)$] Suppose $\operatorname{tr}(A)=0$. By Theorem \ref{thm:similar-to-constant-diagonal}, there is an invertible $S$ such that $B=S^{-1}AS=[b_{ij}]$ has zero diagonal. Let $X=\operatorname{diag}(1,2,\ldots,n)$, viewing the integers as scalars in $\mathbb{F}$, and define $Y=[y_{ij}]$ by
\begin{align}
y_{ij}=
\begin{cases}
(i-j)^{-1}b_{ij}, & i\neq j,\\
0, & i=j.
\end{cases}
\nonumber
\end{align}
The hypothesis ensures that $i-j\neq0$ in $\mathbb{F}$ whenever $i\neq j$. Thus $XY-YX=[(i-j)y_{ij}]=B$.\\
Hence $A=(SXS^{-1})(SYS^{-1})-(SYS^{-1})(SXS^{-1})$, so $A$ is a commutator.
\end{enumerate}
\hlt{Structural Commentary}: Trace is the obstruction; similarity exposes zero diagonal form.\\
\hlt{Key Insight}: Cyclicity forces trace zero; distinct diagonals realize it.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:shoda-field-hypothesis} \hlt{Field Hypothesis in the Diagonal Proof}\\
The proof above uses that the integers $1,\ldots,n$ are distinct and that each smaller block size remains nonzero in $\mathbb{F}$. In small characteristic, this diagonal proof must be replaced by a more delicate argument.
\end{remark}

\begin{corollary} \label{cor:identity-is-not-operator-commutator} \hlt{The Identity Is Not a Commutator}\\
Assume $\dim\mathcal{V}\cdot1_{\mathbb{F}}\neq0$. There do not exist $S,T\in\LL(\mathcal{V})$ such that $ST-TS=I_{\mathcal{V}}$.
\begin{proof}
If $ST-TS=I_{\mathcal{V}}$, choose a basis and let $A=[S]$ and $B=[T]$ in that basis. Then $AB-BA=I_n$.\\
Taking traces and using Proposition \ref{prop:trace-cyclicity-two-factors}, $0=\operatorname{tr}(AB)-\operatorname{tr}(BA)=\operatorname{tr}(I_n)=n\cdot1_{\mathbb{F}}$, contradicting the hypothesis.\\
\hlt{Structural Commentary}: Commutators have zero trace, while the identity has dimension as its trace.\\
\hlt{Key Insight}: Trace detects a scalar obstruction invisible from multiplication alone.
\end{proof}
\end{corollary}

\begin{definition} \label{def:idempotent-and-nilpotent-matrices} \hlt{Idempotent and Nilpotent Matrices}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ is \hlt{idempotent} if $A^2=A$.
\item $A$ is \hlt{nilpotent} if $A^k=0$ for some positive integer $k$.
\end{enumerate}
An idempotent matrix is an algebraic projection: it fixes its range and kills its kernel.
\end{definition}

\begin{proposition} \label{prop:idempotent-matrix-basic-invariants} \hlt{Basic Invariants of Idempotents}\\
If $P\in\operatorname{Mat}(n,n,\mathbb{F})$ satisfies $P^2=P$, then $\mathbb{F}^n=\operatorname{range}(P)\oplus\ker(P)$, $P$ is diagonalizable, and its only possible eigenvalues are $0$ and $1$. Moreover $\operatorname{tr}(P)=\operatorname{rank}(P)$.\\
If $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, then $P$ is the orthogonal projection onto $\operatorname{range}(P)$ if and only if $P=P^*$.
\begin{proof}
For every $\mathbf{x}$, write $\mathbf{x}=P\mathbf{x}+(\mathbf{x}-P\mathbf{x})$. The first term lies in $\operatorname{range}(P)$, while $P(\mathbf{x}-P\mathbf{x})=P\mathbf{x}-P^2\mathbf{x}=0$, so the second term lies in $\ker(P)$. The intersection is zero because $\mathbf{x}=P\mathbf{y}$ and $P\mathbf{x}=0$ imply $\mathbf{x}=P^2\mathbf{y}=P\mathbf{x}=0$.\\
In a basis adapted to $\operatorname{range}(P)\oplus\ker(P)$, the matrix of $P$ is $I_r\oplus0$, where $r=\operatorname{rank}(P)$.\\
Hence $P$ is diagonalizable, its eigenvalues lie in $\{0,1\}$, and $\operatorname{tr}(P)=r$.\\
For the final claim, self-adjoint idempotents are exactly orthogonal projections by Theorem \ref{thm:orthogonal-projection-self-adjoint-idempotent}.\\
\hlt{Structural Commentary}: Idempotents split the space into fixed directions and killed directions.\\
\hlt{Key Insight}: In adapted coordinates, every idempotent is just $I_r\oplus0$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:nilpotent-matrix-basic-invariants} \hlt{Basic Invariants of Nilpotents}\\
For $N\in\operatorname{Mat}(n,n,\mathbb{F})$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $N$ is nilpotent;
\item the characteristic polynomial of $N$ is $t^n$;
\item $N^n=0$.
\end{enumerate}
In this case, $\operatorname{tr}(N)=0$, $\det(N)=0$ for $n\geq1$, and the spectral radius of $N$ over $\mathbb{C}$ is $0$.
\begin{proof}
If $N^k=0$, then every eigenvalue $\lambda$ of $N$ over an algebraic closure satisfies $\lambda^k=0$, hence $\lambda=0$. Thus the characteristic polynomial is $t^n$. If the characteristic polynomial is $t^n$, Cayley--Hamilton gives $N^n=0$. The implication $N^n=0\Rightarrow$ nilpotent is immediate. The trace, determinant, and spectral-radius statements follow from the eigenvalue data and from Definitions \ref{def:matrix-trace-and-determinant} and \ref{def:spectral-radius-and-convergent-matrix}.\\
\hlt{Structural Commentary}: Nilpotent matrices have no spectral mass away from zero.\\
\hlt{Key Insight}: Cayley--Hamilton turns the characteristic polynomial $t^n$ into the equation $N^n=0$.
\end{proof}
\end{proposition}

\begin{definition} \label{def:positive-semidefinite-matrix} \hlt{Positive Semidefinite and Positive Definite Matrices}\\
Let $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$ and let $A\in\operatorname{Mat}(n,n,\mathbb{F})$.\\
The matrix $A$ is \hlt{positive semidefinite}, written $A\succeq0$, if $A=A^*$ and $\mathbf{x}^*A\mathbf{x}\geq0$ for every $\mathbf{x}\in\mathbb{F}^n$.\\
It is \hlt{positive definite}, written $A\succ0$, if $A=A^*$ and $\mathbf{x}^*A\mathbf{x}>0$ for every nonzero $\mathbf{x}\in\mathbb{F}^n$.\\
Negative semidefinite and negative definite mean $-A\succeq0$ and $-A\succ0$, respectively.
\end{definition}

\begin{proposition} \label{prop:rectangular-gram-products-are-psd} \hlt{Rectangular Gram Products Are Positive Semidefinite}\\
Let $R\in\operatorname{Mat}(m,n,\mathbb{F})$, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$. Then $R^*R\succeq0$, $RR^*\succeq0$, and $(R^*R)_{j,j}=\norm{R_{\cdot,j}}_2^2$; $(RR^*)_{i,i}=\norm{R_{i,\cdot}}_2^2$.\\
Full column rank gives $R^*R\succ0$; full row rank gives $RR^*\succ0$.
\begin{proof}
The matrices are self-adjoint by Proposition \ref{prop:basic-adjoint-identities}.\\
For $\mathbf{x}\in\mathbb{F}^n$ and $\mathbf{y}\in\mathbb{F}^m$, $\mathbf{x}^*R^*R\mathbf{x}=\norm{R\mathbf{x}}_2^2\geq0$ and $\mathbf{y}^*RR^*\mathbf{y}=\norm{R^*\mathbf{y}}_2^2\geq0$. Hence both products are positive semidefinite. Taking $\mathbf{x}=\mathbf{e}_j$ and $\mathbf{y}=\mathbf{e}_i$ gives the diagonal formulas. If $R$ has full column rank, then $R\mathbf{x}\neq0$ for $\mathbf{x}\neq0$, so $R^*R\succ0$. If $R$ has full row rank, apply the same argument to $R^*$.\\
\hlt{Structural Commentary}: Multiplying a matrix by its adjoint produces a Gram matrix of columns or rows.\\
\hlt{Key Insight}: The quadratic form is a squared norm, so no cancellation can make it negative.
\end{proof}
\end{proposition}

For edges $e=u\to v$ on vertices $1,\ldots,n$, $B\in\operatorname{Mat}(m,n,\mathbb{R})$ has row $e$ equal to $-\mathbf{e}_u^t+\mathbf{e}_v^t$.
\end{definition}

\begin{perspective} \label{per:incidence-least-squares-ranking} \hlt{Incidence Data and Least Squares}\\
For an incidence matrix $B$ as in Example \ref{ex:directed-incidence-matrix-four-node}, $B\mathbf{x}=\mathbf{b}$ seeks node potentials matching edge differences. If loop consistency fails, least squares projects $\mathbf{b}$ onto $\operatorname{col}(B)$, where $B^*(B\mathbf{x}_0-\mathbf{b})=\mathbf{0}$ (Theorem \ref{thm:matrix-least-squares-normal-equations} on $B$). The constant-vector kernel means potentials are unique only after fixing one node or choosing minimum norm.
\end{perspective}

\begin{proposition} \label{prop:incidence-laplacian-positive-semidefinite} \hlt{Incidence Laplacians Are Positive Semidefinite}\\
Let $B$ be a directed incidence matrix and set $L=B^tB$. Then $L\succeq0$ and $\mathbf{x}^tL\mathbf{x}=\sum\limits_{e=(u\to v)}(x_v-x_u)^2$, with $\ker(L)$ equal to the vectors constant on each connected component.
\begin{proof}
Since $L=B^tB$, Proposition \ref{prop:rectangular-gram-products-are-psd} gives $L\succeq0$.\\
For $e=(u\to v)$, $(B\mathbf{x})_e=x_v-x_u$, so $\mathbf{x}^tL\mathbf{x}=\norm{B\mathbf{x}}_2^2=\sum\limits_{e=(u\to v)}(x_v-x_u)^2$. Thus $\mathbf{x}\in\ker(L)$ if and only if $B\mathbf{x}=\mathbf{0}$, equivalently $x_u=x_v$ on every edge, which means $\mathbf{x}$ is constant on each connected component.\\
\hlt{Structural Commentary}: Laplacians sum squared edge differences.\\
\hlt{Key Insight}: A graph Laplacian is a Gram product, so positivity is automatic.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:positive-semidefinite-matrix-equivalences} \hlt{Equivalent Forms of Positive Semidefiniteness}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\succeq0$.
\item $A$ is Hermitian and all eigenvalues of $A$ are nonnegative.
\item $A=B^*B$ for some $B\in\operatorname{Mat}(m,n,\mathbb{F})$.
\item $A$ has a positive semidefinite square root.
\end{enumerate}
In this case the positive semidefinite square root is unique.\\
If $A=U\Lambda U^*$ with $\Lambda=\operatorname{diag}(\lambda_1,\ldots,\lambda_n)$ and $\lambda_i\geq0$, then $A^{1/2}=U\operatorname{diag}(\sqrt{\lambda_1},\ldots,\sqrt{\lambda_n})U^*$.
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ This is the matrix form of Theorem \ref{thm:positive-operator-characterization}.
\item[$(ii)$]$\Rightarrow (iv)$ By the spectral theorem, write $A=U\Lambda U^*$, where $U$ is unitary and $\Lambda=\operatorname{diag}(\lambda_1,\ldots,\lambda_n)$ with $\lambda_i\geq0$. Then $A^{1/2}=U\operatorname{diag}(\sqrt{\lambda_1},\ldots,\sqrt{\lambda_n})U^*$ is positive semidefinite and satisfies $(A^{1/2})^2=A$.
\item[$(iv)$]$\Rightarrow (iii)$ If $C\succeq0$ and $C^2=A$, then $C=C^*$ and $A=C^*C$, so (iii) holds with $m=n$.
\item[$(iii)$]$\Rightarrow (i)$ If $A=B^*B$, then $A^*=A$ and $\mathbf{x}^*A\mathbf{x}=\mathbf{x}^*B^*B\mathbf{x}=\norm{B\mathbf{x}}_2^2\geq0$, so $A\succeq0$.
\end{enumerate}
The positive semidefinite square root is unique by Proposition \ref{prop:positive-square-root-unique}.\\
\hlt{Structural Commentary}: Positive semidefinite matrices are Gram-type matrices with nonnegative spectrum.\\
\hlt{Key Insight}: Positivity is best checked either on eigenvalues or by finding a factor $B$ with $A=B^*B$.
\end{proof}
\end{theorem}

\begin{method} \label{meth:psd-spectral-factorization} \hlt{PSD Spectral Factorisation Method}\\
To analyse a Hermitian matrix $A$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item diagonalise $A=U\Lambda U^*$;
\item check the signs of the diagonal entries of $\Lambda$;
\item if $\Lambda\geq0$, set $A^{1/2}=U\Lambda^{1/2}U^*$ and use $A=(A^{1/2})^*A^{1/2}$;
\item translate kernel, rank, and zero-quadratic-form questions into the factor $A^{1/2}$.
\end{enumerate}
\end{method}

\begin{theorem} \label{thm:positive-definite-matrix-equivalences} \hlt{Equivalent Forms of Positive Definiteness}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\succ0$.
\item $A$ is Hermitian and all eigenvalues of $A$ are positive.
\item $A\succeq0$ and $A$ is invertible.
\item $A=B^*B$ for some invertible $B\in\operatorname{Mat}(n,n,\mathbb{F})$.
\item $A$ is invertible and $A^{-1}\succ0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*).]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ By definition $A$ is Hermitian. If $(\lambda,\mathbf{v})$ is an eigenpair with $\mathbf{v}\neq\mathbf{0}$, then $\lambda\norm{\mathbf{v}}_2^2=\mathbf{v}^*A\mathbf{v}>0$, so every eigenvalue is positive.
\item[$(ii)$]$\Rightarrow (iii)$ Positive eigenvalues imply $A\succeq0$ by Theorem \ref{thm:positive-semidefinite-matrix-equivalences}; they also exclude $0$, so $A$ is invertible.
\item[$(iii)$]$\Rightarrow (iv)$ By Theorem \ref{thm:positive-semidefinite-matrix-equivalences}, $A$ has a positive semidefinite square root $A^{1/2}$.\\
Since $A$ is invertible, $A^{1/2}$ is invertible, and $A=(A^{1/2})^*A^{1/2}$.
\item[$(iv)$]$\Rightarrow (i)$ If $A=B^*B$ and $B$ is invertible, then for every nonzero $\mathbf{x}$, $\mathbf{x}^*A\mathbf{x}=\norm{B\mathbf{x}}_2^2>0$.
\item[$(ii)$]$\Leftrightarrow (v)$ Hermitian-ness is preserved by inversion, and eigenvalues of $A^{-1}$ are reciprocal eigenvalues of $A$. Thus positive eigenvalues of $A$ are equivalent to positive eigenvalues of $A^{-1}$.
\end{enumerate}
\hlt{Structural Commentary}: Positive definite matrices are the invertible part of the positive semidefinite cone.\\
\hlt{Key Insight}: Positive definite means every eigenvalue is strictly on the positive side of zero.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:trace-determinant-positive-semidefinite} \hlt{Trace and Determinant of Positive Semidefinite Matrices}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ be positive semidefinite, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$. Then $\operatorname{tr}A\geq0$, with equality if and only if $A=0$, and $\det A\geq0$, with strict inequality if and only if $A$ is positive definite.
\begin{proof}
List the eigenvalues of $A$ as $\lambda_1,\ldots,\lambda_n$. By Theorem \ref{thm:positive-semidefinite-matrix-equivalences}, each $\lambda_j\geq0$.\\
Hence $\operatorname{tr}A=\sum\limits_{j=1}^{n}\lambda_j\geq0$ and $\det A=\prod\limits_{j=1}^{n}\lambda_j\geq0$.\\
The trace is zero exactly when every $\lambda_j=0$, which for a Hermitian matrix means $A=0$. The determinant is positive exactly when every $\lambda_j>0$, which is positive definiteness by Theorem \ref{thm:positive-definite-matrix-equivalences}.\\
\hlt{Structural Commentary}: Trace and determinant aggregate the nonnegative eigenvalues by sum and product.\\
\hlt{Key Insight}: Equality is controlled by which eigenvalues vanish.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:gram-factor-kernel-rank-column-data} \hlt{Kernel, Rank, and Columns of $B^*B$}\\
Let $B\in\operatorname{Mat}(m,n,\mathbb{F})$, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, and set $A=B^*B$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\succeq0$;
\item $\ker(A)=\ker(B)$;
\item $\operatorname{rank}(A)=\operatorname{rank}(B)$;
\item $\operatorname{col}(A)=\operatorname{col}(B^*)$;
\item $A=0$ if and only if $B=0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Positive semidefiniteness follows from Theorem \ref{thm:positive-semidefinite-matrix-equivalences}.
\item If $B\mathbf{x}=0$, then $A\mathbf{x}=0$. Conversely, if $A\mathbf{x}=0$, then $0=\mathbf{x}^*A\mathbf{x}=\norm{B\mathbf{x}}_2^2$ so $B\mathbf{x}=0$.
\item Since $A$ and $B$ both have $n$ columns, rank-nullity and (ii) give $\operatorname{rank}(A)=\operatorname{rank}(B)$.
\item Since $A=A^*$, Proposition \ref{prop:adjoint-kernel-range-orthogonality} and (ii) give $\operatorname{col}(A)=\ker(A)^{\perp}=\ker(B)^{\perp}=\operatorname{col}(B^*)$.
\item By (iii), $\operatorname{rank}(A)=0$ if and only if $\operatorname{rank}(B)=0$.\\
A matrix has rank zero exactly when it is the zero matrix, so $A=0$ if and only if $B=0$.
\end{enumerate}
\hlt{Structural Commentary}: The Gram product $B^*B$ preserves the kernel of $B$.\\
\hlt{Key Insight}: The identity $\mathbf{x}^*B^*B\mathbf{x}=\norm{B\mathbf{x}}_2^2$ is the whole mechanism.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:positive-semidefinite-principal-block-column-inclusion} \hlt{Principal Blocks and Column Inclusion}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{F})$ be PSD, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, and partition $A=
\begin{bmatrix}
A_{11} & A_{12}\\
A_{12}^* & A_{22}
\end{bmatrix}$, $A_{11}\in\operatorname{Mat}(k,k,\mathbb{F})$.\\
Then $A_{11}\succeq0$, $\operatorname{col}(A_{12})\subseteq\operatorname{col}(A_{11})$, and $\operatorname{rank}(A_{12})\leq\operatorname{rank}(A_{11})$. If $A\succ0$, then $A_{11}\succ0$.
\begin{proof}
Write $A=B^*B$ and partition $B=[B_1\ B_2]$ conformally. Then $A_{11}=B_1^*B_1$ and $A_{12}=B_1^*B_2$.\\
Hence $A_{11}\succeq0$ and $\operatorname{col}(A_{12})=\operatorname{col}(B_1^*B_2)\subseteq\operatorname{col}(B_1^*)=\operatorname{col}(B_1^*B_1)=\operatorname{col}(A_{11})$, equality $\operatorname{col}(B_1^*)=\operatorname{col}(B_1^*B_1)$ follows from Theorem \ref{thm:gram-factor-kernel-rank-column-data}. The rank inequality follows by taking dimensions\\
If $A\succ0$ and $\mathbf{x}\neq0$ in $\mathbb{F}^k$, then $\mathbf{x}^*A_{11}\mathbf{x}=
\begin{bmatrix}\mathbf{x}\\0\end{bmatrix}^*
A
\begin{bmatrix}\mathbf{x}\\0\end{bmatrix}>0$, so $A_{11}\succ0$.\\
\hlt{Structural Commentary}: In a positive block matrix, cross-block columns lie in principal block's column space.\\
\hlt{Key Insight}: Factor first; then column inclusion is immediate from $A_{12}=B_1^*B_2$.
\end{proof}
\end{corollary}

\begin{definition} \label{def:principal-and-leading-principal-minors} \hlt{Principal and Leading Principal Minors}\\
For $A\in\operatorname{Mat}(n,n,\mathbb{C})$, a \hlt{principal minor} is the determinant of a principal submatrix of $A$.\\
A \hlt{leading principal minor} is the determinant of the leading $k\times k$ principal submatrix for some $k=1,\ldots,n$.
\end{definition}

\begin{theorem} \label{thm:sylvester-positive-definite-criterion-leading-minors} \hlt{Sylvester's Criterion}\\
A Hermitian matrix $A\in\operatorname{Mat}(n,n,\mathbb{C})$ is positive definite if and only if all leading principal minors are positive.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $A\succ0$, then each leading principal submatrix is positive definite by Corollary \ref{cor:positive-semidefinite-principal-block-column-inclusion}. Its eigenvalues are positive, so its determinant is positive.
\item[$(\Leftarrow)$] Induct on $n$. If $n=1$, the single leading principal minor is the only diagonal entry, so positivity of that minor is exactly positive definiteness. Assume $n\geq2$ and write $A=
\begin{bmatrix}
B & \mathbf{y}\\
\mathbf{y}^* & a
\end{bmatrix}$, $B\in\operatorname{Mat}(n-1,n-1,\mathbb{C})$\\
The leading principal minors of $B$ are positive, so $B\succ0$.\\
Since $\det(B)>0$ and $\det(A)>0$, Proposition \ref{prop:schur-complement-determinant-formula} gives $a-\mathbf{y}^*B^{-1}\mathbf{y}=\frac{\det(A)}{\det(B)}>0$.\\
For $\mathbf{u}\in\mathbb{C}^{n-1}$ and $t\in\mathbb{C}$, $\begin{bmatrix}\mathbf{u}\\ t\end{bmatrix}^*
A
\begin{bmatrix}\mathbf{u}\\ t\end{bmatrix} =(\mathbf{u}+B^{-1}\mathbf{y}t)^*B(\mathbf{u}+B^{-1}\mathbf{y}t) +\left(a-\mathbf{y}^*B^{-1}\mathbf{y}\right)\abs{t}^2$.\\
First term is nonnegative as $B\succ0$, and second coefficient is positive. If displayed vector is nonzero, then either $t\neq0$ and second term is positive, or $t=0$ and then $\mathbf{u}\neq0$, making first term positive. Thus $A\succ0$.
\end{enumerate}
\hlt{Structural Commentary}: Leading minors make each leading block positive and final Schur complement positive.\\
\hlt{Key Insight}: Complete the square with the leading block.
\end{proof}
\end{theorem}

\begin{method} \label{meth:two-by-two-quadratic-form-test} \hlt{$2\times2$ Quadratic Form Test}\\
Let $A=\begin{bmatrix}a&b\\ b&c\end{bmatrix}$ be real symmetric, $\Delta=ac-b^2$, and $q(x,y)=\begin{bmatrix}x&y\end{bmatrix}A\begin{bmatrix}x\\y\end{bmatrix}=ax^2+2bxy+cy^2$.\\
For $a\neq0$, $q(x,y)=a\left(x+\frac{b}{a}y\right)^2+\frac{\Delta}{a}y^2$.\\
If $a=0$, swap variables when $c\neq0$; if $a=c=0$, use $q=2bxy$. Thus:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $q\succ0$ if and only if $a>0$ and $\Delta>0$;
\item $q\prec0$ if and only if $a<0$ and $\Delta>0$;
\item $q$ is indefinite if and only if $\Delta<0$;
\item if $\Delta=0$, then $q\succeq0$ iff $a,c\geq0$, and $q\preceq0$ iff $a,c\leq0$.
\end{enumerate}
This is the $2\times2$ form of Sylvester's criterion and inertia.\\
\hlt{Structural Commentary}: Completing the square is the $2\times2$ shadow of the $LDL^*$ and Schur-complement tests.\\
\hlt{Key Insight}: $\Delta$ and the leading sign classify the form.
\end{method}

\begin{corollary} \label{cor:positive-semidefinite-zero-quadratic-form} \hlt{Zero Quadratic Form Detects the Kernel}\\
If $A\in\operatorname{Mat}(n,n,\mathbb{F})$ is positive semidefinite, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, then $\mathbf{x}^*A\mathbf{x}=0$ if and only if $A\mathbf{x}=0$.
\begin{proof}
Write $A=B^*B$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $\mathbf{x}^*A\mathbf{x}=0$, then $\norm{B\mathbf{x}}_2^2=0$, so $B\mathbf{x}=0$ and hence $A\mathbf{x}=B^*B\mathbf{x}=0$.
\item[$(\Leftarrow)$] If $A\mathbf{x}=0$, then $\mathbf{x}^*A\mathbf{x}=0$.
\end{enumerate}
\hlt{Structural Commentary}: A zero PSD quadratic value has no hidden cancellation.\\
\hlt{Key Insight}: Zero length-square means the factored vector itself is zero.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:positive-semidefinite-stability-rules} \hlt{Stability Rules for Positive Semidefinite Matrices}\\
Let $A,B\succeq0$ over $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$. Whenever the expressions are defined:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $aA+bB\succeq0$ for $a,b\geq0$; if $a,b>0$ and one of $A,B$ is positive definite, then $aA+bB\succ0$.
\item $S^*AS\succeq0$; moreover, $S^*AS\succ0$ if and only if $S\mathbf{x}\notin\ker(A)$ for all $\mathbf{x}\neq\mathbf{0}$.
\item $A\oplus B\succeq0$, with $A\oplus B\succ0$ if and only if $A\succ0$ and $B\succ0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathbf{x}^*(aA+bB)\mathbf{x}=a\mathbf{x}^*A\mathbf{x}+b\mathbf{x}^*B\mathbf{x}\geq0$. If, for example, $A\succ0$ and $a>0$, then for every nonzero $\mathbf{x}$ the first summand is positive and the second is nonnegative, so $aA+bB\succ0$; the case $B\succ0$ is identical.
\item $\mathbf{x}^*S^*AS\mathbf{x}=(S\mathbf{x})^*A(S\mathbf{x})\geq0$. By Corollary \ref{cor:positive-semidefinite-zero-quadratic-form}, equality occurs exactly when $S\mathbf{x}\in\ker(A)$.
\item $(\mathbf{x}\oplus\mathbf{y})^*(A\oplus B)(\mathbf{x}\oplus\mathbf{y})=\mathbf{x}^*A\mathbf{x}+\mathbf{y}^*B\mathbf{y}\geq0$. This expression is positive for every nonzero $\mathbf{x}\oplus\mathbf{y}$ exactly when each summand is positive on its own nonzero vectors, namely when $A\succ0$ and $B\succ0$.
\end{enumerate}
\hlt{Structural Commentary}: The PSD cone is stable under sums, pullbacks, and direct sums.\\
\hlt{Key Insight}: Test the quadratic form after each construction.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:two-by-two-positive-semidefinite-square-root} \hlt{Explicit Square Root for a $2\times2$ Positive Semidefinite Matrix}\\
Let $A\in\operatorname{Mat}(2,2,\mathbb{F})$, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, be positive semidefinite and nonzero. Set $\tau=\bigl(\operatorname{tr}A+2\sqrt{\det A}\bigr)^{1/2}$.\\
Then $\tau>0$ and $A^{1/2}=\frac{1}{\tau}\bigl(A+\sqrt{\det A}\,I_2\bigr)$.
\begin{proof}
Let $C=\frac{1}{\tau}\bigl(A+\sqrt{\det A}\,I_2\bigr)$. By Proposition \ref{prop:trace-determinant-positive-semidefinite}, $\det A\geq0$ and $\operatorname{tr}A>0$, hence $\tau>0$; by Theorem \ref{thm:positive-semidefinite-stability-rules}, $C\succeq0$. Cayley--Hamilton gives $A^2-(\operatorname{tr}A)A+(\det A)I_2=0$, so
\begin{align}
C^2 =\frac{1}{\tau^2}\bigl(A^2+2\sqrt{\det A}\,A+(\det A)I_2\bigr) =\frac{1}{\tau^2}\bigl((\operatorname{tr}A)A+2\sqrt{\det A}\,A\bigr)=A.
\nonumber
\end{align}
Thus $C=A^{1/2}$ by uniqueness of the positive semidefinite square root.\\
\hlt{Structural Commentary}: In dimension $2$, Cayley--Hamilton makes $A^{1/2}$ linear in $A$.\\
\hlt{Key Insight}: Square the linear candidate and use Cayley--Hamilton.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:simultaneous-congruence-one-positive-definite-form} \hlt{Simultaneous Congruence with One Positive Definite Form}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$ be Hermitian, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, and let $A\succ0$.\\
Then, for some invertible $S$ and real diagonal $\Lambda$, $A=SS^*$ and $B=S\Lambda S^*$.\\
The diagonal entries of $\Lambda$ are the eigenvalues of $A^{-1}B$.
\begin{proof}
Let $C=A^{-1/2}BA^{-1/2}=U\Lambda U^*$ be spectral decomposition, $S=A^{1/2}U$. Then $SS^*=A$ and $S\Lambda S^*=B$.\\
Also $A^{-1}B=A^{-1/2}CA^{1/2}$, so $A^{-1}B$ is similar to $C$ and has eigenvalues given by the diagonal entries of $\Lambda$.\\
\hlt{Structural Commentary}: The positive definite form $A$ normalizes $B$ to a Hermitian matrix.\\
\hlt{Key Insight}: Normalize, diagonalise, then transform back.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:positive-definite-hermitian-product-spectrum} \hlt{Products with One Positive Definite Factor}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, and let $A\succ0$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $B$ is Hermitian, then $AB$ is diagonalizable with real spectrum.
\item If $B\succeq0$, then $AB$ is diagonalizable with nonnegative real spectrum; if $B\neq0$, one eigenvalue is positive.
\item If $B\succ0$, then $AB$ is diagonalizable with positive real spectrum.
\end{enumerate}
\begin{proof}
Let $C=A^{1/2}BA^{1/2}$. Since $A^{-1/2}(AB)A^{1/2}=C$, the matrices $AB$ and $C$ are similar.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $B$ is Hermitian, then $C$ is Hermitian, hence diagonalizable with real spectrum.
\item If $B\succeq0$, then $C\succeq0$ by Theorem \ref{thm:positive-semidefinite-stability-rules}, so its spectrum is nonnegative.\\
If $B\neq0$, then $C\neq0$, hence some eigenvalue is positive.
\item If $B\succ0$, then $C\succ0$ by Theorem \ref{thm:positive-semidefinite-stability-rules}, so all eigenvalues are positive.
\end{enumerate}
\hlt{Structural Commentary}: A positive definite factor makes $AB$ similar to a Hermitian congruence.\\
\hlt{Key Insight}: Use $AB\sim A^{1/2}BA^{1/2}$.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:real-quadratic-form-principal-axes} \hlt{Principal Axes for Real Quadratic Forms}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{R})$ be symmetric. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item there are a real orthogonal $Q$ and real $\lambda_1,\ldots,\lambda_n$ such that, with $\mathbf{y}=Q^t\mathbf{x}$, $\mathbf{x}^tA\mathbf{x}=\sum\limits_{j=1}^{n}\lambda_jy_j^2$.
\item $A\succeq0$ if and only if every $\lambda_j\geq0$;
\item $A\succ0$ if and only if every $\lambda_j>0$;
\item $A$ is indefinite if and only if some $\lambda_j>0$ and some $\lambda_k<0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By the real spectral theorem, write $A=Q\operatorname{diag}(\lambda_1,\ldots,\lambda_n)Q^t$.\\
Then $\mathbf{x}^tA\mathbf{x} =\mathbf{x}^tQ\operatorname{diag}(\lambda_j)Q^t\mathbf{x} =\mathbf{y}^t\operatorname{diag}(\lambda_j)\mathbf{y} =\sum\limits_{j=1}^{n}\lambda_jy_j^2$.
\item The displayed sum is nonnegative for all $\mathbf{y}$ exactly when each coefficient $\lambda_j$ is nonnegative.
\item The displayed sum is positive for all nonzero $\mathbf{y}$ exactly when each coefficient $\lambda_j$ is positive.
\item The diagonal form takes both signs exactly when its coefficients include both signs.
\end{enumerate}
\hlt{Structural Commentary}: Principal axes turn the form into signed squares.\\
\hlt{Key Insight}: Eigenvalue signs classify the quadratic form.
\end{proof}
\end{theorem}

\begin{method} \label{meth:quadratic-form-principal-axes} \hlt{Quadratic Form Principal-Axes Method}\\
To analyse a real quadratic form:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item write it as $\mathbf{x}^tA\mathbf{x}$ with $A=A^t$; split each cross coefficient equally between $a_{ij}$ and $a_{ji}$;
\item diagonalise $A=Q\Lambda Q^t$;
\item change variables by $\mathbf{y}=Q^t\mathbf{x}$;
\item classify the form from the signs of the eigenvalues in $\Lambda$.
\end{enumerate}
\end{method}

\begin{definition} \label{def:star-congruence-and-hermitian-inertia} \hlt{$*$-Congruence and Hermitian Inertia}\\
Two matrices $A,B\in\operatorname{Mat}(n,n,\mathbb{C})$ are \hlt{$*$-congruent} if $A=SBS^*$ for some invertible $S\in\operatorname{Mat}(n,n,\mathbb{C})$.\\
For Hermitian $A$, its \hlt{inertia} is $\operatorname{in}(A)=(n_+(A),n_-(A),n_0(A))$, where these are the numbers of positive, negative, and zero eigenvalues of $A$, counted with multiplicity.
\end{definition}

\begin{perspective} \label{per:ordinary-congruence-bilinear-form-bridge} \hlt{Congruence versus Similarity}\\
Similarity $B=P^{-1}AP$ is change of basis for an operator. Congruence $N=P^tMP$ is change of basis for a bilinear form, with the same basis change in both slots. Over $\mathbb{R}$ this is the real case of $*$-congruence. Hence inertia classifies forms, not similarity classes of operators.
\end{perspective}

\begin{lemma} \label{lem:positive-subspace-dimension-inertia} \hlt{Positive Subspaces and Inertia}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ be Hermitian. The maximal dimensions of subspaces on which $\mathbf{x}^*A\mathbf{x}$ is positive or negative for every nonzero $\mathbf{x}$ are $n_+(A)$ and $n_-(A)$, respectively.
\begin{proof}
Let $P_+$, $P_{\leq0}$ be positive, nonpositive spectral subspaces. Then $\dim P_+=n_+(A)$, $P_+$ is positive for $A$.\\
If $\dim\mathcal{U}>n_+(A)$, then $\dim\mathcal{U}+\dim P_{\leq0}>n_+(A)+n_-(A)+n_0(A)=n$, so $\mathcal{U}\cap P_{\leq0}$ contains a nonzero vector.\\
On this vector quadratic form is nonpositive, so $\mathcal{U}$ cannot be positive. Apply this to $-A$ for negative case.\\
\hlt{Structural Commentary}: The positive spectral subspace is maximal; the nonpositive part blocks enlargement.\\
\hlt{Key Insight}: Inertia counts maximal sign-definite dimensions.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:sylvester-inertia-theorem-hermitian} \hlt{Sylvester's Inertia Theorem}\\
Hermitian matrices $A,B$ of the same size are $*$-congruent if and only if $n_+(A)=n_+(B)$ and $n_-(A)=n_-(B)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose $A=SBS^*$ with $S$ invertible. Then $\mathbf{x}^*A\mathbf{x}=(S^*\mathbf{x})^*B(S^*\mathbf{x})$.\\
Thus $S^*$ transports positive and negative subspaces, so Lemma \ref{lem:positive-subspace-dimension-inertia} gives $n_\pm(A)=n_\pm(B)$.
\item[$(\Leftarrow)$] Let the common inertia be $(p,q,r)$ and set $D=I_p\oplus(-I_q)\oplus0_r$.\\
By the spectral theorem, each of $A$ and $B$ is $*$-congruent to $D$, hence they are $*$-congruent to each other.
\end{enumerate}
\hlt{Structural Commentary}: $*$-congruence is a coordinate change for Hermitian quadratic forms.\\
\hlt{Key Insight}: Spectral decomposition reduces the form to signed squares and zero directions.
\end{proof}
\end{theorem}

\begin{definition} \label{def:phase-inertia-matrix} \hlt{Phase Inertia Matrix}\\
A \hlt{phase inertia matrix} is a diagonal complex matrix whose nonzero diagonal entries have modulus $1$.
\end{definition}

\begin{proposition} \label{prop:normal-matrix-phase-inertia-form} \hlt{Phase Inertia Form for Normal Matrices}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$ be normal.\\
Then $A$ is $*$-congruent to a phase inertia matrix $D$, whose entries record the spectral rays of $A$.
\begin{proof}
Write $A=U\operatorname{diag}(\lambda_j)U^*$. Write $\lambda_j=r_jd_j$, where $r_j>0$ and $|d_j|=1$ if $\lambda_j\neq0$, while $r_j=1$ and $d_j=0$ if $\lambda_j=0$. Set $D=\operatorname{diag}(d_j)$, $S=U\operatorname{diag}(\sqrt{r_j})$. Then $S$ is invertible and $A=SDS^*$.\\
The entries $d_j=e^{i\theta}$ count exactly the eigenvalues on the ray $\{\rho e^{i\theta}:\rho>0\}$.\\
\hlt{Structural Commentary}: Normal diagonalization separates modulus from phase; $*$-congruence absorbs moduli.\\
\hlt{Key Insight}: Only spectral phases survive.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:hermitian-inertia-as-real-phase-inertia} \hlt{Hermitian Inertia as Real Phase Inertia}\\
For Hermitian matrices, the only nonzero spectral rays are the positive and negative real axes.\\
Sylvester's inertia theorem is therefore the real-ray version of phase inertia.
\end{perspective}

\begin{definition} \label{def:kronecker-product} \hlt{Kronecker Product}\\
Let $A=[a_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$ and $B\in\operatorname{Mat}(p,q,\mathbb{F})$. The \hlt{Kronecker product} of $A$ and $B$ is the block matrix
\begin{align}
A\otimes B
\defeq
\begin{bmatrix}
a_{11}B & a_{12}B & \cdots & a_{1n}B\\
a_{21}B & a_{22}B & \cdots & a_{2n}B\\
\vdots & \vdots & \ddots & \vdots\\
a_{m1}B & a_{m2}B & \cdots & a_{mn}B
\end{bmatrix}
\in\operatorname{Mat}(mp,nq,\mathbb{F}).
\nonumber
\end{align}
The Kronecker product is also called the tensor product of matrices.
\end{definition}

\begin{proposition} \label{prop:basic-kronecker-identities} \hlt{Basic Kronecker Identities}\\
Whenever the expressions are defined,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $c(A\otimes B)=(cA)\otimes B=A\otimes(cB)$;
\item $(A+B)\otimes C=A\otimes C+B\otimes C$ and $A\otimes(B+C)=A\otimes B+A\otimes C$;
\item $(A\otimes B)\otimes C=A\otimes(B\otimes C)$;
\item $(A\otimes B)^t=A^t\otimes B^t$;
\item $I_m\otimes I_n=I_{mn}$.
\end{enumerate}
If $\mathbb{F}=\mathbb{C}$, then also $\overline{A\otimes B}=\overline{A}\otimes\overline{B}$ and $(A\otimes B)^*=A^*\otimes B^*$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The $(i,j)$ block of $c(A\otimes B)$ is $c(a_{ij}B)=(ca_{ij})B=a_{ij}(cB)$, which is $(i,j)$ block of $(cA)\otimes B$ and $A\otimes(cB)$.
\item The $(i,j)$ block of $(A+B)\otimes C$ is $(a_{ij}+b_{ij})C=a_{ij}C+b_{ij}C$, so $(A+B)\otimes C=A\otimes C+B\otimes C$.\\
The proof of $A\otimes(B+C)=A\otimes B+A\otimes C$ is identical inside each block.
\item If $B=[b_{rs}]$, then both $(A\otimes B)\otimes C$ and $A\otimes(B\otimes C)$ have blocks $a_{ij}b_{rs}C$ in the same relative positions.
\item The $(j,i)$ block of $(A\otimes B)^t$ is $(a_{ij}B)^t=a_{ij}B^t$, which is the $(j,i)$ block of $A^t\otimes B^t$.
\item The block matrix $I_m\otimes I_n$ has $I_n$ on its diagonal blocks and zero elsewhere, hence equals $I_{mn}$.
\end{enumerate}
If $\mathbb{F}=\mathbb{C}$, conjugation is checked blockwise, adjoint identity follows from conjugation together with part (iv).\\
\hlt{Structural Commentary}: Kronecker products replace scalar entries by whole scaled blocks. Standard algebraic rules survive because each block is still controlled by scalar distributivity and transposition.\\
\hlt{Key Insight}: Check identities block-by-block; each block reduces to an ordinary scalar-matrix identity.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:kronecker-mixed-product-property} \hlt{Mixed Product Property}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $B\in\operatorname{Mat}(p,q,\mathbb{F})$, $C\in\operatorname{Mat}(n,r,\mathbb{F})$, $D\in\operatorname{Mat}(q,s,\mathbb{F})$.  $(A\otimes B)(C\otimes D)=(AC)\otimes(BD)$.
\begin{proof}
Write $A=[a_{ik}]$ and $C=[c_{kj}]$. The $(i,j)$ block of $(A\otimes B)(C\otimes D)$ is\\
$\sum\limits_{k=1}^{n}(a_{ik}B)(c_{kj}D) =\left(\sum\limits_{k=1}^{n}a_{ik}c_{kj}\right)BD =(AC)_{ij}BD$, which is exactly the $(i,j)$ block of $(AC)\otimes(BD)$.\\
\hlt{Structural Commentary}: The outer indices multiply as $AC$, while the inner block action multiplies as $BD$.\\
\hlt{Key Insight}: Kronecker multiplication separates into ordinary multiplication in each factor.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:inverse-of-kronecker-product} \hlt{Inverse of a Kronecker Product}\\
If $A\in\operatorname{Mat}(m,m,\mathbb{F})$ and $B\in\operatorname{Mat}(n,n,\mathbb{F})$ are invertible, then $A\otimes B$ is invertible and $(A\otimes B)^{-1}=A^{-1}\otimes B^{-1}$.
\begin{proof}
By Theorem \ref{thm:kronecker-mixed-product-property}, $(A\otimes B)(A^{-1}\otimes B^{-1})=I_m\otimes I_n=I_{mn}$ and the reverse product is identical.\\
\hlt{Structural Commentary}: Inverting a Kronecker product inverts each independent factor.\\
\hlt{Key Insight}: The mixed product property turns the inverse check into two ordinary inverse checks.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:kronecker-product-and-sum-diagonalizable-spectra} \hlt{Kronecker Product and Sum Spectra in the Diagonalizable Case}\\
Let $A=P\Lambda P^{-1}$, $B=QMQ^{-1}$, where $\Lambda=\operatorname{diag}(\lambda_1,\ldots,\lambda_n)$ and $M=\operatorname{diag}(\mu_1,\ldots,\mu_q)$.\\
Then $A\otimes B\sim\Lambda\otimes M$; and $A\otimes I_q+I_n\otimes B\sim \Lambda\otimes I_q+I_n\otimes M$.\\
Their diagonal entries are respectively $\lambda_i\mu_j$ and $\lambda_i+\mu_j$ for $1\leq i\leq n$, $1\leq j\leq q$.
\begin{proof}
By Theorem \ref{thm:kronecker-mixed-product-property} and $(P\otimes Q)^{-1}=P^{-1}\otimes Q^{-1}$, $A\otimes B =(P\otimes Q)(\Lambda\otimes M)(P^{-1}\otimes Q^{-1})$.\\
Similarly, $A\otimes I_q+I_n\otimes B =(P\otimes Q)(\Lambda\otimes I_q+I_n\otimes M)(P^{-1}\otimes Q^{-1})$.\\
The middle matrices are diagonal with the stated $(i,j)$-indexed entries.\\
\hlt{Structural Commentary}: Eigenvectors tensor together, and Kronecker sums add the two scalar eigenvalues.\\
\hlt{Key Insight}: Diagonalize each factor first; the Kronecker operation becomes entrywise on eigenvalue pairs.
\end{proof}
\end{proposition}

\begin{definition} \label{def:vectorization-map} \hlt{Vectorisation}\\
Let $X=[\mathbf{x}_1\ \mathbf{x}_2\ \cdots\ \mathbf{x}_n]\in\operatorname{Mat}(m,n,\mathbb{F})$. The \hlt{vectorisation} of $X$ is
\begin{align}
\operatorname{vec}(X)\defeq
\begin{bmatrix}
\mathbf{x}_1\\
\mathbf{x}_2\\
\vdots\\
\mathbf{x}_n
\end{bmatrix}
\in\mathbb{F}^{mn}.
\nonumber
\end{align}
Thus $\operatorname{vec}$ stacks the columns of $X$ vertically.
\end{definition}

\begin{theorem} \label{thm:vectorization-of-triple-product} \hlt{Vectorisation of a Triple Product}\\
Let $A\in\operatorname{Mat}(m,n,\mathbb{F})$, $X\in\operatorname{Mat}(n,p,\mathbb{F})$, and $B\in\operatorname{Mat}(p,q,\mathbb{F})$. Then $\operatorname{vec}(AXB)=(B^t\otimes A)\operatorname{vec}(X)$.
\begin{proof}
Write $B=[\mathbf{b}_1\ \cdots\ \mathbf{b}_q]$ and $X=[\mathbf{x}_1\ \cdots\ \mathbf{x}_p]$.\\
The $k$th column of $AXB$ is $AX\mathbf{b}_k=A\sum\limits_{i=1}^{p}b_{ik}\mathbf{x}_i =\sum\limits_{i=1}^{p}b_{ik}A\mathbf{x}_i =(\mathbf{b}_k^t\otimes A)\operatorname{vec}(X)$.\\
Stacking these $q$ columns gives $(B^t\otimes A)\operatorname{vec}(X)$.\\
\hlt{Structural Commentary}: Vectorisation rewrites two-sided multiplication as one matrix action.\\
\hlt{Key Insight}: Left multiplication contributes $A$; right multiplication contributes $B^t$.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:vectorized-minimal-polynomial-linear-system} \hlt{Vectorising a Minimal-Polynomial Test}\\
For $A\in\operatorname{Mat}(n,n,\mathbb{F})$ and a monic candidate $p_m(z)=a_0+a_1z+\cdots+a_{m-1}z^{m-1}+z^m$,\\
the matrix equation $a_0I+a_1A+\cdots+a_{m-1}A^{m-1}=-A^m$ can be written as the ordinary linear system
\begin{align}
\begin{bmatrix}
\operatorname{vec}(I)&\operatorname{vec}(A)&\cdots&\operatorname{vec}(A^{m-1})
\end{bmatrix}
\begin{bmatrix}
a_0\\a_1\\ \vdots\\ a_{m-1}
\end{bmatrix}
=-\operatorname{vec}(A^m) \nonumber
\end{align}
\hlt{Structural Commentary}: Vectorisation turns matrix identities into linear systems.\\
\hlt{Key Insight}: Polynomial membership becomes a coefficient system in vectorized powers.
\end{perspective}

\begin{definition} \label{def:hadamard-product} \hlt{Hadamard Product}\\
Let $A=[a_{ij}],B=[b_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$.\\
The \hlt{Hadamard product}, or \hlt{entrywise product}, of $A$ and $B$ is $A\circ B\defeq [a_{ij}b_{ij}]\in\operatorname{Mat}(m,n,\mathbb{F})$.
\end{definition}

\begin{proposition} \label{prop:basic-hadamard-identities} \hlt{Basic Hadamard Identities}\\
For matrices of compatible sizes:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\circ B=B\circ A$;
\item $A\circ(\gamma B+C)=\gamma(A\circ B)+A\circ C$;
\item $(A\circ B)^*=A^*\circ B^*$;
\item if $E$ is the all-ones matrix, then $A\circ E=A$;
\item if $D=\operatorname{diag}(d_1,\ldots,d_n)$, then $A\circ D=\operatorname{diag}(a_{11}d_1,\ldots,a_{nn}d_n)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(A\circ B)_{ij}=a_{ij}b_{ij}=b_{ij}a_{ij}=(B\circ A)_{ij}$.
\item $\bigl(A\circ(\gamma B+C)\bigr)_{ij}=a_{ij}(\gamma b_{ij}+c_{ij})=\gamma a_{ij}b_{ij}+a_{ij}c_{ij}$.
\item $\bigl((A\circ B)^*\bigr)_{ij}=\overline{a_{ji}b_{ji}}=\overline{a_{ji}}\,\overline{b_{ji}}=(A^*\circ B^*)_{ij}$.
\item $(A\circ E)_{ij}=a_{ij}$.
\item Since $D$ has zero off-diagonal entries, $A\circ D$ is diagonal with entries $a_{ii}d_i$.
\end{enumerate}
\hlt{Structural Commentary}: Hadamard multiplication is entrywise, with no row-column summation.\\
\hlt{Key Insight}: Prove Hadamard identities entry by entry.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:schur-product-theorem} \hlt{Schur Product Theorem}\\
If $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$ are positive semidefinite, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$, then $A\circ B$ is positive semidefinite.
\begin{proof}
By Theorem \ref{thm:positive-semidefinite-matrix-equivalences}, regard $A$ and $B$ as Gram matrices: $a_{ij}=\innerprod{\mathbf{u}_j,\mathbf{u}_i}$, $b_{ij}=\innerprod{\mathbf{v}_j,\mathbf{v}_i}$ for suitable vectors $\mathbf{u}_1,\ldots,\mathbf{u}_n$ and $\mathbf{v}_1,\ldots,\mathbf{v}_n$. Then $(A\circ B)_{ij} =\innerprod{\mathbf{u}_j,\mathbf{u}_i}\innerprod{\mathbf{v}_j,\mathbf{v}_i} =\innerprod{\mathbf{u}_j\otimes\mathbf{v}_j,\mathbf{u}_i\otimes\mathbf{v}_i}$.\\
Thus $A\circ B$ is the Gram matrix of the vectors $\mathbf{u}_j\otimes\mathbf{v}_j$, hence is positive semidefinite by Proposition \ref{prop:gram-matrix-positive-semidefinite}.\\
\hlt{Structural Commentary}: The Schur product is the Gram matrix of paired tensor vectors.\\
\hlt{Key Insight}: Pair the $j$th vector from each Gram representation and take tensor products.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:hadamard-positive-definite-criterion} \hlt{Positive Definite Hadamard Products}\\
Let $A,B\in\operatorname{Mat}(n,n,\mathbb{F})$ be positive semidefinite, where $\mathbb{F}\in\{\mathbb{R},\mathbb{C}\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A$ is positive definite and every diagonal entry of $B$ is positive, then $A\circ B$ is positive definite.
\item If $A$ and $B$ are positive definite, then $A\circ B$ is positive definite.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $\lambda_{\min}>0$ be the smallest eigenvalue of $A$. Then $A-\lambda_{\min}I_n\succeq0$, so $(A-\lambda_{\min}I_n)\circ B\succeq0$ by Theorem \ref{thm:schur-product-theorem}. Thus, for $\mathbf{x}\neq\mathbf{0}$, $\mathbf{x}^*(A\circ B)\mathbf{x} =\mathbf{x}^*((A-\lambda_{\min}I_n)\circ B)\mathbf{x} \lambda_{\min}\sum\limits_{j=1}^{n}b_{jj}\abs{x_j}^2>0$.
\item If $B\succ0$, then $b_{jj}=\mathbf{e}_j^*B\mathbf{e}_j>0$ for every $j$, so (i) applies.
\end{enumerate}
\hlt{Structural Commentary}: Subtract the spectral floor of $A$; the diagonal of $B$ gives strictness.\\
\hlt{Key Insight}: Use $I_n\circ B=\operatorname{diag}(b_{11},\ldots,b_{nn})$.
\end{proof}
\end{corollary}

\begin{method} \label{meth:schur-product-positivity} \hlt{Schur Product Positivity Method}\\
To prove an entrywise product is positive semidefinite:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item factor or interpret each matrix as a Gram matrix;
\item pair the corresponding vectors;
\item take tensor products of the paired vectors;
\item identify the Hadamard product as the Gram matrix of those tensor products.
\end{enumerate}
\end{method}

\begin{definition} \label{def:spectral-radius-and-convergent-matrix} \hlt{Spectral Radius and Convergent Matrix}\\
For $A\in\operatorname{Mat}(n,n,\mathbb{C})$, the \hlt{spectral radius} is $\rho(A)\defeq\max\{\abs{\lambda}:\lambda\in\operatorname{spec}(A)\}$.\\
We say $A^p\to0$ if every entry of $A^p$ converges to $0$ as $p\to\infty$.
\end{definition}

\begin{theorem} \label{thm:matrix-powers-converge-to-zero-spectral-radius} \hlt{Matrix Powers Converge to Zero}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. Then $A^p\to0$ if and only if $\rho(A)<1$.
\begin{proof}
Choose a Jordan form $A=SJS^{-1}$.\\
Since $S$ and $S^{-1}$ are fixed, $A^p\to0$ if and only if every Jordan block power in $J^p$ converges to $0$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Each block power tends to $0$, so Proposition \ref{prop:powers-of-jordan-block} gives $\abs{\lambda}<1$ for every eigenvalue. Thus $\rho(A)<1$.
\item[$(\Leftarrow)$] If $\rho(A)<1$, then every eigenvalue satisfies $\abs{\lambda}<1$. Proposition \ref{prop:powers-of-jordan-block} makes each block power tend to $0$.
\end{enumerate}
\hlt{Structural Commentary}: Similarity reduces matrix-power convergence to Jordan block-power convergence.\\
\hlt{Key Insight}: The largest eigenvalue modulus controls the asymptotic size of all blocks.
\end{proof}
\end{theorem}

\begin{definition} \label{def:power-bounded-matrix} \hlt{Power Bounded Matrix}\\
A matrix $A\in\operatorname{Mat}(n,n,\mathbb{C})$ is \hlt{power bounded} if for each entry position $(i,j)$, the sequence of entries of $A^p$ in that position is bounded for $p=1,2,\ldots$.
\end{definition}

\begin{theorem} \label{thm:matrix-power-bounded-jordan-criterion} \hlt{Power Bounded Jordan Criterion}\\
Let $A\in\operatorname{Mat}(n,n,\mathbb{C})$. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A$ is power bounded.
\item $\rho(A)\leq1$, and every eigenvalue $\lambda$ with $\abs{\lambda}=1$ has only $1\times1$ Jordan blocks.
\item $\rho(A)\leq1$, and $\dim E_{\lambda}(A)=m_A(\lambda)$ for every $\lambda\in\operatorname{spec}(A)$ with $\abs{\lambda}=1$.
\end{enumerate}
\begin{proof}
Work in Jordan form.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow(ii)$: If $\abs{\lambda}>1$, the diagonal entries $\lambda^p$ are unbounded.\\
If $\abs{\lambda}=1$ and $J_k(\lambda)$ has $k\geq2$, then the first superdiagonal contains $p\lambda^{p-1}$, also unbounded.
\item[$(ii)$]$\Rightarrow(i)$: Blocks with $\abs{\lambda}<1$ converge to $0$ by Proposition \ref{prop:powers-of-jordan-block}; $1\times1$ blocks with $\abs{\lambda}=1$ are bounded. Finite direct sums and fixed similarities preserve boundedness.
\item[$(ii)$]$\Leftrightarrow(iii)$: For fixed $\lambda$, $\dim E_{\lambda}(A)$ counts Jordan blocks, while $m_A(\lambda)$ counts their total size.\\
Equality holds exactly when every block has size $1$.
\end{enumerate}
\hlt{Structural Commentary}: Unit-circle eigenvalues are harmless only when their nilpotent parts vanish.\\
\hlt{Key Insight}: Nontrivial Jordan blocks on the unit circle create polynomial growth.
\end{proof}
\end{theorem}
