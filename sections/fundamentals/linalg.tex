\subsection{Linear Algebra}

Linear algebra is the branch of mathematics that studies vectors, matrices, and linear transformations. It provides the foundation for solving systems of equations and analysing multidimensional data. Its applications span across machine learning, computer graphics, physics, finance, and engineering.\\
References from \cite{garcia2017second} and \cite{axler2024linear}.

\subsubsection{Vector Spaces}

\begin{definition} \hlt{Vector Space}\\
Let $\mathbb{F} = \mathbb{R}$ or $\mathbb{C}$. Then $\mathcal{V}$ is a vector space over a field $\mathbb{F}$ if the following are satisfied:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Identities: $\exists \ \mathbf{0} \in \mathcal{V}$ such that $\mathbf{0} + \mathbf{u} = \mathbf{u} \ \forall \ \mathbf{u} \in \mathcal{V}$, and $\exists \ \mathbf{1} \in \mathcal{V}$ such that $\mathbf{1} \mathbf{u} = \mathbf{u} \ \forall \ \mathbf{u} \in \mathcal{V}$
\item Commutativity: $\mathbf{u} + \mathbf{v} = \mathbf{v} + \mathbf{u} \ \forall \ \mathbf{u}, \mathbf{v} \in \mathcal{V}$
\item Associativity: $\mathbf{u} + (\mathbf{v} + \mathbf{w}) = (\mathbf{u} + \mathbf{v}) + \mathbf{w}$ and $(ab) \mathbf{u} = a(b \mathbf{u}) \ \forall \ a, b \in \mathbb{F}$ and $\forall \ \mathbf{u}, \mathbf{v}, \mathbf{w} \in \mathcal{V}$
\item Distributive: $a(\mathbf{u} + \mathbf{v}) = a \mathbf{u} + a \mathbf{v}$ and $(a + b)\mathbf{u} = a \mathbf{u} + b \mathbf{u} \ \forall \ a, b \in \mathbb{F}$ and $\forall \ \mathbf{u}, \mathbf{v} \in \mathcal{V}$
\item Additive Inverse: $\forall \ \mathbf{u} \in \mathcal{V} \ \exists$ unique vector $-\mathbf{u} \in \mathcal{V}$ such that $\mathbf{u} + (- \mathbf{u}) = \mathbf{0}$
\item Multiplicative Inverse: $\forall \ \mathbf{u} \in \mathcal{V}$ with $\mathbf{u} \neq \mathbf{0} \ \exists$ unique $\mathbf{w} \in \mathcal{V}$ such that $\mathbf{u} \mathbf{w} = 1$
\end{enumerate}
\end{definition}

\begin{remark} \hlt{Properties of Vector Spaces}\\
Let $\mathcal{V}$ be a vector space over a field $\mathbb{F}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item A vector space has a unique additive identity
\item Every element in a vector space has a unique additive inverse
\item $\mathbf{0} \mathbf{v} = \mathbf{v}$ for every $\mathbf{v} \in \mathcal{V}$
\item $a\mathbf{0} = \mathbf{0}$ for every $a \in \mathbb{F}$
\item $(-1)\mathbf{v} = -\mathbf{v}$ for every $\mathbf{v} \in \mathcal{V}$
\end{enumerate}
\end{remark}

\begin{definition} \hlt{Subspace}\\
A subspace of an $\mathbb{F}$-vector space $\mathcal{V}$ is a subset $\mathcal{U}$ of $\mathcal{V}$ that is an $\mathbb{F}$-vector space with the same vector addition and scalar multiplication operations as in $\mathcal{V}$. To check $\mathcal{U}$ satisfies the following:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Additive Identity: $\mathbf{0} \in \mathcal{U}$
\item Closure Under Addition: $\mathbf{u}, \mathbf{v} \in \mathcal{U}$ implies $\mathbf{u} + \mathbf{v} \in \mathcal{U}$
\item Closed Under Scalar Multiplication: $a \in \mathbb{F}$ and $\mathbf{u} \in \mathcal{U}$ implies $a \mathbf{u} \in \mathcal{U}$.
\end{enumerate}
\end{definition}

\begin{definition} \hlt{Sum of Subspaces}\\
Suppose $\mathcal{U}_1, \ldots, \mathcal{U}_m$ are subspaces of $\mathcal{V}$. The sum of $\mathcal{U}_1, \ldots, \mathcal{U}_m$ as denoted by $\mathcal{U}_1 + \cdots + \mathcal{U}_m$ is defined to be set of all possible sums of elements of $\mathcal{U}_1, \ldots, \mathcal{U}_m$:
\begin{align*}
\mathcal{U}_1 + \cdots + \mathcal{U}_m \coloneqq \{ \mathbf{u}_1 + \cdots + \mathbf{u}_m : \mathbf{u}_1 \in \mathcal{U}_1, \ldots, \mathbf{u}_m \in \mathcal{U}_m \} \nonumber
\end{align*}
\end{definition}

\begin{definition} \hlt{Direct Sum of Subspaces}\\
If $\mathcal{U}_1, \ldots, \mathcal{U}_m$ are subspaces of $\mathcal{V}$ such that $\mathcal{V} = \mathcal{U}_1 + \cdots + \mathcal{U}_m$, then every element of $\mathcal{V}$ may be written as
\begin{align*}
\mathbf{u}_1 + \cdots + \mathbf{u}_m \nonumber
\end{align*}
where each $\mathbf{u}_j \in \mathcal{U}_j$. Then $\mathcal{V}$ is the direct sum of subspaces $\mathcal{U}_1, \ldots, \mathcal{U}_m$, written as $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$
\end{definition}

\begin{definition} \hlt{Linear Combination}\\
Let $\mathcal{U}$ be non-empty subset of $\mathbb{F}$-vector space $\mathcal{V}$. A linear combination of elements of $\mathcal{U}$ is expressed as
\begin{align}
c_1 \mathbf{v}_1 + c_2 \mathbf{v}_2 + \cdots + c_r \mathbf{v}_r \nonumber
\end{align}
where $r$ is a positive integer, and $\mathbf{v}_1, \mathbf{v}_2, \ldots, \mathbf{v}_r \in \mathcal{U}$, $c_1, c_2, \ldots, c_r \in \mathbb{F}$.\\
A linear combination is \hlt{trivial} if $c_1 = c_2 = \cdots = c_r = 0$.
\end{definition}

\begin{definition} \hlt{Span}\\
Let $\mathcal{U}$ be subset of $\mathbb{F}$-vector space $\mathcal{V}$. If $\mathcal{U} \neq \emptyset$, then $\text{span}(\mathcal{U})$ is the set of linear combinations of elements of $\mathcal{U}$.\\
Define $\text{span}(\emptyset) = \{\mathbf{0}\}$. If $\beta = \mathbf{v}_1, \mathbf{v}_2, \ldots, \mathbf{v}_r$ is a list of $r \geq 1$ vectors in $\mathcal{V}$, define $\text{span}(\beta) = \text{span}\{ \mathbf{v}_1, \mathbf{v}_2, \ldots, \mathbf{v}_r \}$.
\end{definition}

\begin{lemma} \label{lem:lineardependencelemma} \hlt{Linear Dependence Lemma}\\
If $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly dependent in $\mathcal{V}$ and $\mathbf{v}_1 \neq 0$, then $\exists \ j \in \{2, \ldots, m \}$ such that the following hold:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathbf{v}_j \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{j-1})$
\item if $\mathbf{v}_j$ is removed, the span of remaining list equals $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_m)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Let $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ be linearly dependent with $\mathbf{v}_1 \neq \mathbf{0}$. Then $\exists$ scalars $a_1, \ldots, a_m$ not all zero, s.t. 
\begin{align}
a_1 \mathbf{v}_1 + \cdots + a_m \mathbf{v}_m = \mathbf{0} \nonumber
\end{align}
Since $\mathbf{v}_1 \neq \mathbf{0}$, not all of $a_2, \ldots, a_m$ can be zero. Otherwise $a_1 \mathbf{v_1} = \mathbf{0}$, implying $a_1 = 0$, a contradiction. Hence $\exists$ some index $j \in \{ 2, \ldots, m \}$ with $a_j \neq 0$.\\
Choose $j$ be largest index. Then $a_{j+1} = \cdots = a_m = 0$, so the dependence relation reduces to
\begin{align}
a_1 \mathbf{v}_1 + \cdots + a_j \mathbf{v}_j = \mathbf{0} \nonumber
\end{align}
Solving for $\mathbf{v}_j$, we get
\begin{align}
\mathbf{v}_j = - \frac{a_1}{a_j}\mathbf{v}_1 - \cdots - \frac{a_{j-1}}{a_j} \mathbf{v}_{j-1} \nonumber
\end{align}
So $\mathbf{v}_j \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{j-1})$.
\item Remove $\mathbf{v}_j$ from the list. Since $\mathbf{v}_j$ is already a linear combination of earlier vectors, any linear combination involving $\mathbf{v}_j$ can be rewritten with only remaining vectors. Hence removing $\mathbf{v}_j$ does not change the span. Hence the span of remaining list equals $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_m)$.
\end{enumerate}
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:finitedimlenofvec} In a finite-dimensional vector space, length of every linearly independent list of vectors is less than or equal to length of every spanning list of vectors.
\begin{proof} Let $(\mathbf{u}_1, \ldots, \mathbf{u}_m)$ be linearly independent in $\mathcal{V}$, and let $(\mathbf{w}_1, \ldots, \mathbf{w}_n)$ span $\mathcal{V}$. To prove $m \leq n$.\\
Begin with spanning list $\mathcal{B}_0 = (\mathbf{w}_1, \ldots, \mathbf{w}_n)$. Construct for each $k = 0, 1, \ldots, m$ a spanning list $\mathcal{B}_k$ of length $n$ which contains $\mathbf{u}_1, \ldots, \mathbf{u}_k$.\\
For $k = 0$, this is true by assumption. For $0 \leq k < m$ and suppose $\mathcal{B}_k$ is a spanning list of length $n$ containing $\mathbf{u}_1, \ldots, \mathbf{u}_k$. Since $\mathcal{B}_k$ spans $\mathcal{V}$, the enlarged list $(\mathbf{u}_{k+1}, \mathcal{B}_k)$ is linearly dependent. Also $\mathbf{u}_{k+1} \neq \mathbf{0}$, because $(\mathbf{u}_1, \ldots, \mathbf{u}_m)$ is linearly independent. By Lemma \ref{lem:lineardependencelemma}, some vector in $\mathcal{B}_k$ can be removed from $(\mathbf{u}_{k+1}, \mathcal{B}_k)$ without changing the span.\\
Therefore the removed vector must be one of the original $\mathbf{w}$-vectors still present in $\mathcal{B}_k$. After removing it, obtain a new spanning list $\mathcal{B}_{k+1}$ of length $n$ containing $\mathbf{u}_1, \ldots, \mathbf{u}_k, \mathbf{u}_{k+1}$.\\
By induction, after $m$ steps there is a spanning list $\mathcal{B}_m$ of length $n$ containing $\mathbf{u}_1, \ldots, \mathbf{u}_m$. Since $\mathcal{B}_m$ has only $n$ vectors but contains $m$ distinct linearly independent vectors, we must have $m \leq n$. Thus the length of every linearly independent list is less than or equal to the length of every spanning list.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:everysubspacefindimisfindim} Every subspace of finite-dimensional vector space is finite dimensional.
\begin{proof}
Let $\mathcal{V}$ be finite-dimensional and $\mathcal{U} \subset \mathcal{V}$ a subspace. If $\mathcal{U} = \{\mathbf{0}\}$, the result is trivial.\\
Assume $\mathcal{U} \neq \{ \mathbf{0} \}$. Construct a linearly independent list in $\mathcal{U}$ as follows. Choose $\mathbf{v}_1 \in \mathcal{U}$, $\mathbf{v}_1 \neq \mathbf{0}$. Suppose chosen linearly independent vectors $\mathbf{v}_1, \ldots, \mathbf{v}_{k} \in \mathcal{U}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item If $\mathcal{U} = \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$, stop
\item Otherwise, choose $\mathbf{v}_{k+1} \in \mathcal{U}$ such that $\mathbf{v}_{k+1} \notin \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$
\end{enumerate}
Then $(\mathbf{v}_1, \ldots, \mathbf{v}_{k+1})$ remains linearly independent. Continuing, obtain a linearly linearly independent list in $\mathcal{U}$.\\
Now since $\mathcal{V}$ is finite-dimensional, $\exists$ a spanning list of length $n$. By Theorem \ref{thm:finitedimlenofvec}, every linearly independent list in $\mathcal{V}$ has length at most $n$, and construction cannot continue indefinitely. After finitely many steps (i.e., $m$), process stops. At termination, have $\mathcal{U} = \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_m)$, otherwise the list could be extended further.\\
Thus $\mathcal{U}$ is spanned by finitely many vectors, hence is finite-dimensional.
\end{proof}
\end{proposition}

\begin{definition} A \hlt{basis} of $\mathcal{V}$ is a list of vectors in $\mathcal{V}$ that is linearly independent and spans $\mathcal{V}$.
\end{definition}

\begin{proposition} A list $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ of vectors in $\mathcal{V}$ is a basis of $\mathcal{V}$ if and only if every $\mathbf{v} \in \mathcal{V}$ can be written uniquely in form $\mathbf{v} = a_1 \mathbf{v}_1 + \cdots + a_n \mathbf{v}_n $, where $a_1, \ldots, a_n \in \mathbb{F}$.
\begin{proof}
$(\Rightarrow)$ Suppose $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is basis of $\mathcal{V}$. Then it spans $\mathcal{V}$, so for every $\mathbf{v} \in \mathcal{V} \ \exists$ scalars $a_1, \ldots, a_n$ such that $\mathbf{v} = a_1 \mathbf{v}_1 + \cdots + a_n \mathbf{v}_n$. To prove uniqueness, suppose also $\mathbf{v} = b_1 \mathbf{v}_1 + \cdots + b_n \mathbf{v}_n$. Subtracting gives
\begin{align}
0 = (a_1 - b_1) \mathbf{v}_1 + \cdots + (a_n - b_n) \mathbf{v}_n \nonumber
\end{align}
Since the list is linearly independent, all coefficients must vanish, $a_j - b_j = 0$ for all $j$. Hence $a_j = b_j$ for all $j$, so the representation is unique.\\
$(\Leftarrow)$ Suppose every $\mathbf{v} \in \mathcal{V}$ can be written uniquely in the form $\mathbf{v} = a_1 \mathbf{v}_1 + \cdots + a_n \mathbf{v}_n$. First, the list spans $\mathcal{V}$, since by assumption every $\mathbf{v} \in \mathcal{V}$ admits such a representation..\\
Next, prove linearly independent. Suppose $0 = a_1 \mathbf{v}_1 + \cdots + a_n \mathbf{v}_n$. But also $0 = 0 \cdot \mathbf{v}_1 + \cdots + 0 \cdot \mathbf{v}_n$. By uniqueness of representation of zero vector, we must have $a_1 = \cdots = a_n = 0$.\\
Thus the list is linearly independent and therefore a basis of $\mathcal{V}$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:spanlistreduce} Every spanning list in a vector space can be reduced to a basis of vector space.
\begin{proof} Let $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ be spanning list for $\mathcal{V}$. Construct a basis by removing redundant vectors.\\
Initialise $\mathcal{B} = (\mathbf{v}_1, \ldots, \mathbf{v}_n)$. Process the vectors in order. For each index $j = 1, \ldots, n$, do the following:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item If current vector $\mathbf{v}_j$ lies in span of preceding vectors already retained in $\mathcal{B}$, remove $\mathbf{v}_j$ from $\mathcal{B}$.
\item Otherwise, keep $\mathbf{v}_j$ in $\mathcal{B}$.
\end{enumerate}
At each step, a vector is removed only if it lies in span of previously retained vectors. By Lemma \ref{lem:lineardependencelemma}, removing such a vector does not change the span. Since original list spans $\mathcal{V}$, final list $\mathcal{B}$ also spans $\mathcal{V}$. By construction, no vector in $\mathcal{B}$ lies in span of those preceding it, equivalent to linear independence (by Lemma \ref{lem:lineardependencelemma}).\\
Hence $\mathcal{B}$ is a basis of $\mathcal{V}$.
\end{proof}
\end{theorem}

\begin{corollary} Every finite-dimensional vector space has a basis.
\begin{proof} Let $\mathcal{V}$ be a finite-dimensional vector space. By definition, $\exists$ a finite list of vectors $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ that spans $\mathcal{V}$. By Theorem \ref{thm:spanlistreduce}, any spanning list can be reduced to a basis by removing redundant vectors. Applying this result to $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ obtain a sublist that is both linearly independent and still spans $\mathcal{V}$. Hence this sublist is a basis of $\mathcal{V}$, therefore $\mathcal{V}$ has a basis.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:linerlyindeplistoffimdimspaceextend} Every linearly independent list of vectors in a finite-dimensional vector space can be extended to a basis of the vector space.
\begin{proof} Let $\mathcal{V}$ be finite dimensional and $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ be linearly independent list in $\mathcal{V}$. Since $\mathcal{V}$ is finite dimensional, choose a finite spanning list $(\mathbf{w}_1, \ldots, \mathbf{w}_n)$ for $\mathcal{V}$.\\
Start with $\mathcal{B} = (\mathbf{v}_1, \ldots, \mathbf{v}_m)$. Now inspect the vectors $\mathbf{w}_1, \ldots, \mathbf{w}_n$ one at a time. For each $j = 1, \ldots, n$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item If $\mathbf{w}_j \in \text{span}(\mathcal{B})$, leave $\mathcal{B}$ unchanged.
\item If $\mathbf{w}_j \notin \text{span}(\mathcal{B})$, append $\mathbf{w}_j$ to $\mathcal{B}$.
\end{enumerate}
At every step, $\mathcal{B}$ remains linearly independent. Indeed, appending $\mathbf{w}_j$ only occurs when $\mathbf{w}_j \notin \text{span}(\mathcal{B})$, and adding a vector outside the span of a linearly independent list preserves linear independence.\\
After all $\mathbf{w}_j$'s have been inspected, the final list $\mathcal{B}$ contains $\mathbf{v}_1, \ldots, \mathbf{v}_m$. Moreover, every $\mathbf{w}_j$ lies in $\text{span}(\mathcal{B})$; either $\mathbf{w}_j$ was added to $\mathcal{B}$, or it was already in span of $\mathcal{B}$ when considered. Therefore, 
\begin{align}
\text{span}(\mathbf{w}_1, \ldots, \mathbf{w}_n) \subseteq \text{span}(\mathcal{B}) \nonumber
\end{align}
Since $(\mathbf{w}_1, \ldots, \mathbf{w}_n)$ spans $\mathcal{V}$, thus $\mathcal{B}$ spans $\mathcal{V}$. Thus $\mathcal{B}$ is both linearly independent and spanning, so $\mathcal{B}$ is a basis of $\mathcal{V}$. Hence every linearly independent list in a finite-dimensional vector space can be extended to a basis.
\end{proof}
\end{theorem}

This theorem can be applied to show that every subspace of a finite-dimensional vector space can be paired with another subspace to form a direct sum of the whole space.

\begin{proposition} Suppose $\mathcal{V}$ is finite dimensional and $\mathcal{U}$ is a subspace of $\mathcal{V}$. Then there is a subspace $\mathcal{W}$ of $\mathcal{V}$ such that $\mathcal{V} = \mathcal{U} \oplus \mathcal{W}$.
\begin{proof} Let $\mathcal{V}$ be finite dimensional, and let $\mathcal{U}$ be subspace of $\mathcal{V}$, By Proposition \ref{prop:everysubspacefindimisfindim}, $\mathcal{U}$ is finite dimensional. Choose a basis $(\mathbf{u}_1, \ldots, \mathbf{u}_m)$ of $\mathcal{U}$. Since this list is linearly independent in $\mathcal{V}$, Theorem \ref{thm:linerlyindeplistoffimdimspaceextend} allows us to extend it to a basis of $\mathcal{V}$, $(\mathbf{u}_1, \ldots \mathbf{u}_m, \mathbf{v}_1, \ldots, \mathbf{v}_m)$.\\
Define $\mathcal{W} = \text{span}(\mathbf{w}_1, \ldots, \mathbf{w}_n)$. Claim that $\mathcal{V} = \mathcal{U} \oplus \mathcal{W}$.\\
First, since $(\mathbf{u}_1, \ldots \mathbf{u}_m, \mathbf{v}_1, \ldots, \mathbf{v}_m)$ spans $\mathcal{V}$, every $\mathbf{v} \in \mathcal{V}$ can be written as 
\begin{align}
\mathbf{v} = a_1 \mathbf{u}_1 + \cdots + a_m \mathbf{u}_m + b_1 \mathbf{w}_1 + \cdots + b_n \mathbf{w}_n = \mathbf{u} + \mathbf{w} \nonumber
\end{align}
Let $\mathbf{u} = a_1 \mathbf{u}_1 + \cdots + a_m \mathbf{u}_m \in \mathcal{U}$ and $\mathbf{w} = b_1 \mathbf{w}_1 + \cdots + b_n \mathbf{w}_n \in \mathcal{W}$. Since $\mathbf{v} \in \mathcal{U}$, $\exists$ scalars $a_1, \ldots, a_m$ such that
\begin{align}
\mathbf{v} = b_1 \mathbf{w} + \cdots + b_n \mathbf{w}_n \nonumber
\end{align}
Subtracting,
\begin{align}
a_1 \mathbf{u}_1 + \cdots + a_m \mathbf{u}_m - b_1 \mathbf{w}_1 - \cdots - b_n \mathbf{w}_n = \mathbf{0} \nonumber
\end{align}
Because $(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{w}_1, \ldots, \mathbf{w}_n)$ is linearly independent, all coefficients are zero.\\
In particular, $a_1 = \cdots = a_m = 0$, so $\mathbf{v} = \mathbf{0}$. Therefore $\mathcal{U} \cap \mathcal{W} = \{ \mathbf{0} \}$.\\
Thus $\mathcal{V} = \mathcal{U} + \mathcal{W}$ and $\mathcal{U} \cap \mathcal{W} = \{ \mathbf{0} \}$, so $\mathcal{V} = \mathcal{U} \oplus \mathcal{W}$.
\end{proof}
\end{proposition}

\begin{theorem} Any two bases of a finite-dimensional vector space have the same length.
\begin{proof}
Let $\mathcal{V}$ be finite dimensional, and $\mathcal{B}_1, \mathcal{B}_2$ be two bases of $\mathcal{V}$. Since $\mathcal{B}_1$ is a basis, it is linearly independent. Since $\mathcal{B}_2$ is a basis, it spans $\mathcal{V}$. Therefore, $\abs{\mathcal{B}_1} \leq \abs{\mathcal{B}_2}$.\\
By same reasoning, now using that $\mathcal{B}_2$ is linearly independent and $\mathcal{B}_1$ spans $\mathcal{V}$, $\abs{\mathcal{B}_2} \leq \abs{\mathcal{B}_1}$. Hence $\abs{\mathcal{B}_1} = \abs{\mathcal{B}_2}$.
\end{proof}
\end{theorem}

\begin{definition} The \hlt{dimension} of a finite-dimensional vector space is defined to be the length of any basis of the vector space, denoted as $\text{dim}(\mathcal{V})$.
\end{definition}

\begin{proposition} If $\mathcal{V}$ is finite dimensional and $\mathcal{U}$ is a subspace of $\mathcal{V}$, then $\text{dim}(\mathcal{U}) \leq \text{dim}(\mathcal{V})$.
\begin{proof} Let $\mathcal{V}$ be finite-dimensional and $\mathcal{U} \subseteq \mathcal{V}$ a subspace. Let $(\mathbf{u}_1, \ldots, \mathbf{u}_m)$ be a basis of $\mathcal{U}$. Then this list is linearly independent in $\mathcal{V}$.\\
By Theorem \ref{thm:linerlyindeplistoffimdimspaceextend}, it can be extended to a basis of $\mathcal{V}$, say $(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{v}_1, \ldots, \mathbf{v}_k)$. Thus the dimension of $\mathcal{U}$ is $m$, and the dimension of $\mathcal{V}$ is $m+k$. Hence $\text{dim}(\mathcal{U}) = m \leq m+k = \text{dim}(\mathcal{V})$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:fimdimspanlistisabasis} If $\mathcal{V}$ is finite dimensional, then every spanning list of vectors in $\mathcal{V}$ with length $\text{dim}(\mathcal{V})$ is a basis of $\mathcal{V}$.
\begin{proof} Let $\text{dim}(\mathcal{V}) = n$ and let $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ be a spanning list of $\mathcal{V}$.\\
By Theorem \ref{thm:spanlistreduce}, this list can be reduced to a basis by removing vectors that are linear combinations of preceding ones. Thus there exists a sublist of $(\mathbf{v}_1, \ldots \mathbf{v}_n)$ that forms a basis of $\mathcal{V}$. Every basis of $\mathcal{V}$ has exactly $n$ vectors. Since the original list already has length $n$, no vector can be removed in this reduction process; otherwise would obtain a basis with fewer than $n$ vectors, a contradiction.\\
Therefore original list $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is already linearly independent. Since it also spans $\mathcal{V}$, it is a basis of $\mathcal{V}$.
\end{proof}
\end{proposition}

\begin{proposition} If $\mathcal{V}$ is finite dimensional, then every linearly independent list of vectors in $\mathcal{V}$ with length $\text{dim}(\mathcal{V})$ is a basis of $\mathcal{V}$.
\begin{proof} Let $\text{dim}(\mathcal{V}) = n$ and $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ be linearly independent in $\mathcal{V}$. By Theorem \ref{thm:linerlyindeplistoffimdimspaceextend}, this list can extended to a basis of $\mathcal{V}$. Thus $\exists$ vectors $\mathbf{w}_1, \ldots, \mathbf{w}_k$ such that $(\mathbf{v}_1, \ldots, \mathbf{v}_n, \mathbf{w}_1, \ldots, \mathbf{w}_k)$ is a basis of $\mathcal{V}$.\\
But every basis of $\mathcal{V}$ has exactly $n$ vectors. Therefore $n + k = n$, which implies $k=0$. Hence no additional vectors are needed, and the original list $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is already a basis of $\mathcal{V}$.
\end{proof}
\end{proposition}

The next theorem gives formula for dimension of sum of two subspaces of a finite-dimensional vector space.

\begin{theorem} If $\mathcal{U}_1$ and $\mathcal{U}_2$ are subspaces of a finite-dimensional vector space, then
\begin{align}
\text{dim}(\mathcal{U}_1 + \mathcal{U}_2) = \text{dim}(\mathcal{U}_1) + \text{dim}(\mathcal{U}_2) - \text{dim}(\mathcal{U}_1 \cap \mathcal{U}_2) \nonumber
\end{align}
\begin{proof} Let $\mathcal{U}_1, \mathcal{U}_2 \subset \mathcal{V}$, suppose $\mathcal{V}$ is finite dim. Let $(\mathbf{u}_1, \ldots, \mathbf{u}_m)$ be basis of $\mathcal{U}_1 \cap \mathcal{U}_2$, then $\text{dim}(\mathcal{U}_1 \cap \mathcal{U}_2) = m$.\\
Extend this to a basis of $\mathcal{U}_1$, $(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{v}_1, \ldots, \mathbf{v}_j)$, $\text{dim}(\mathcal{U}_1) = m + j$.\\
Similarly, extend it to a basis of $\mathcal{U}_2$, $(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{w}_1, \ldots, \mathbf{w}_k)$, $\text{dim}(\mathcal{U}_2) = m + k$.\\
We claim that $\mathcal{B} = (\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{v}_1, \ldots, \mathbf{v}_j, \mathbf{w}_1, \ldots, \mathbf{w}_k)$ is a basis of $\mathcal{U}_1 + \mathcal{U}_2$.\\
\hlt{Spanning}: Every vector in $\mathcal{U}_1$ is a linear combination of $\mathbf{u}_i$'s and $\mathbf{w}_i$'s. Hence any vector in $\mathcal{U}_1 + \mathcal{U}_2$ is a linear combination of vectors in $\mathcal{B}$. There $\mathcal{B}$ spans $\mathcal{U}_1 + \mathcal{U}_2$.\\
\hlt{Linear Independence}. Suppose
\begin{align}
a_1 \mathbf{u}_1 + \cdots + a_m \mathbf{u}_m + b_1 \mathbf{v}_1 + \cdots + b_j \mathbf{v}_j + c_1 \mathbf{w}_1 + \cdots + c_k \mathbf{w}_k = \mathbf{0} \nonumber
\end{align}
where $a_i, b_i, c_i \in \mathbb{F}$. Rearrange,
\begin{align}
c_1 \mathbf{w}_1 + \cdots + c_k \mathbf{w}_k = - a_1 \mathbf{u}_1 - \cdots - a_m \mathbf{u}_m - b_1 \mathbf{v}_1 - \cdots - b_j \mathbf{v}_j \nonumber
\end{align}
The RHS is in $\mathcal{U}_1$, and LHS is in $\mathcal{U}_2$, hence this vector lies in $\mathcal{U}_1 \cap \mathcal{U}_2$. Therefore $\exists$ scalars $d_1, \ldots, d_m$ such that
\begin{align}
c_1 \mathbf{w}_1 + \cdots + c_k \mathbf{w}_k = d_1 \mathbf{u}_1 + \cdots + d_m \mathbf{u}_m \nonumber
\end{align}
Substituting back gives
\begin{align}
\mathbf{0} = (a_1 + d_1) \mathbf{u}_1 + \cdots + (a_m + d_m) \mathbf{u}_m + b_1 \mathbf{v}_1 + \cdots + b_j \mathbf{v}_j \nonumber
\end{align}
Note that $c_1 \mathbf{w}_1 + \cdots + c_k \mathbf{w}_k = d_1 \mathbf{u}_1 + \cdots + d_m \mathbf{u}_m$, the LHS is in $\text{span}(\mathbf{u}_1, \ldots, \mathbf{u}_m)$. Since $(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{w}_1, \ldots, \mathbf{w}_m))$ is a basis of $\mathcal{U}_2$, these spans intersect trivially, so both sides must be zero. Hence
\begin{align}
c_1 = \cdots = c_k = 0, \ \ \ d_1 = \cdots = d_m = 0 \nonumber
\end{align}
and therefore $a_1 = \cdots = a_m = 0$. Thus all coefficients vanish, proving learning independence.\\
Since $\mathcal{B}$ is a basis of $\mathcal{U}_1 + \mathcal{U}_2$, we obtain
\begin{align}
\text{dim}(\mathcal{U}_1 + \mathcal{U}_2) = m + j + k = (m+j) + (m+k) - m = \text{dim}(\mathcal{U}_1) + \text{dim}(\mathcal{U}_2) - \text{dim}(\mathcal{U}_1 \cap \mathcal{U}_2) \nonumber
\end{align}
\end{proof}
\end{theorem}

Note that dimension meshes well with direct sums, which is analogous to disjoint union.

\begin{proposition} Suppose $\mathcal{V}$ is finite dimensional and $\mathcal{U}_1, \ldots, \mathcal{U}_m$ are subspaces of $\mathcal{V}$ such that
\begin{align}
\mathcal{V} = \mathcal{U}_1 + \cdots + \mathcal{U}_m, \ \ \ \ \text{dim}(\mathcal{V}) = \text{dim}(\mathcal{U}_1) + \cdots + \text{dim}(\mathcal{U}_m) \nonumber
\end{align}
Then $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$.
\begin{proof} Let $\mathcal{V}$ be finite dimensional, and suppose $\mathcal{V} = \mathcal{U}_1 + \cdots + \mathcal{U}_m$, $\text{dim}(\mathcal{V}) = \text{dim}(\mathcal{U}_1) + \cdots + \text{dim}(\mathcal{U}_m)$.\\
For each $j$, let $\mathcal{B}_j = (\mathbf{u}_{j1}, \ldots, \mathbf{u}_{d_j})$ be a basis of $\mathcal{U}_j$, where $d_j = \text{dim}(\mathcal{U}_j)$.\\
Form the concentrated list $\mathcal{B} = \mathcal{B}_1 \cup \cdots \cup \mathcal{B}_m$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{B}$ spans $\mathcal{V}$. Since $\mathcal{V} = \mathcal{U}_1 + \cdots + \mathcal{U}_m$, and $\mathbf{v} \in \mathcal{V}$ can be written as
\begin{align}
\mathbf{v} = \mathbf{u}_1 + \cdots + \mathbf{u}_m, \ \ \mathbf{u}_j \in \mathcal{U}_j \nonumber
\end{align}
Each $\mathbf{u}_j$ is a linear combination of vectors in $\mathcal{B}_j$, hence $\mathbf{v}$ is a linear combination of vectors in $\mathcal{B}$.
\item $\mathcal{B}$ is a basis. The length of $\mathcal{B}$ is $\abs{\mathcal{B}} = d_1 + \cdots + d_m = \text{dim}(\mathcal{V})$. Since $\mathcal{B}$ is a spanning list of length $\text{dim}(\mathcal{V})$, Proposition \ref{prop:fimdimspanlistisabasis} implies $\mathcal{B}$ is a basis of $\mathcal{V}$. In particular, it is linearly independent.
\item Directness of the Sum. Suppose $\mathbf{u}_1 + \cdots + \mathbf{u}_m = \mathbf{0}$, $\mathbf{u}_j \in \mathcal{U}_j$, write each $\mathbf{u}_j$ as a linear combination of basis vectors in $\mathcal{B}_j$. Then we obtain a linear combination of vectors in $\mathcal{B}$ equal to $\mathbf{0}$.\\
Since $\mathcal{B}$ is linearly independent, all coefficients must equal zero. Hence each $\mathbf{u}_j = \mathbf{0}$. Thus
\begin{align}
\mathcal{U}_1 \cap (\mathcal{U}_2 + \cdots + \mathcal{U}_m) = \{ \mathbf{0} \} \nonumber
\end{align}
and similarly for all components, so the sum is direct. Therefore $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$.
\end{enumerate}
\end{proof}
\end{proposition}

\subsubsection{Linear Maps}

\begin{definition} A \hlt{Linear Map} from $\mathcal{V}$ to $\mathcal{W}$ is a function $T: \mathcal{V} \rightarrow \mathcal{W}$ with following properties:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{Additivity}: $T(\mathbf{u} + \mathbf{v}) = T\mathbf{u} + T\mathbf{v}$ for all $\mathbf{u}, \mathbf{v} \in \mathcal{V}$
\item \hlt{Homogeneity}: $T(a\mathbf{v}) = a(T\mathbf{v})$ for all $a \in \mathbb{F}$ and all $\mathbf{v} \in \mathcal{V}$.
\end{enumerate}
The set of all linear maps from $\mathcal{V}$ to $\mathcal{W}$ is denoted $\mathcal{L}(\mathcal{V}, \mathcal{W})$.\\
The linear map is \hlt{injective} if whenever $\mathbf{u}, \mathbf{v} \in \mathcal{V}$ and $T\mathbf{u} = T\mathbf{v}$, that $\mathbf{u} = \mathbf{v}$; and \hlt{surjective} if its range equals $\mathcal{W}$.
\end{definition}

\begin{example} Examples of linear maps include
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Zero and Identitiy Map: $0 \in \mathcal{L}(\mathcal{V}, \mathcal{W})$ is defined by $0 \mathbf{v} = 0$. $I \in \mathcal{L}(\mathcal{V}, \mathcal{V})$ is defined by $I \mathbf{v} = \mathbf{v}$
\item Differentiation: $T \in \mathcal{L}(\mathcal{P}(\mathbf{R}), \mathcal{P}(\mathbf{R}))$ is defined by $T \mathbf{p} = \mathbf{p}'$
\item Integration: $T \in \mathcal{L}(\mathcal{P}(\mathbf{R}), \mathbf{R})$ is defined by $T \mathbf{p} = \int_0^1 \mathbf{p}(\mathbf{x}) \ d \mathbf{x}$
\item Backward Shift: Let $\mathbb{F}^{\infty}$ be vector space of all sequences of elements of $\mathbb{F}$. Let $T \in \mathcal{L}(\mathbb{F}^{\infty}, \mathbb{F}^{\infty})$. The backward shift map is then $T(x_1, x_2, x_3, \ldots) = (x_2, x_3, \ldots)$
\item From $\mathbb{F}^{n}$ to $\mathbb{F}^{m}$: $T \in \mathcal{L}(\mathbb{F}^{n}, \mathbb{F}^{m})$ defined by $T(x_1, \ldots, x_n) = (a_{1,1} x_1 + \cdots + a_{1,n} x_n, \ldots, a_{m,1} x_1 + \cdots + a_{m,n} x_n)$, where $a_{j,k} \in \mathbb{F}$, $j=1, \ldots, m$ and $k=1, \ldots, n$.
\end{enumerate}
\end{example}

\begin{remark} \hlt{Properties of Product of Linear Maps}\\
Let $T \in \mathcal{L}(\mathcal{U}, \mathcal{V})$, and $S \in \mathcal{L}(\mathcal{V}, \mathcal{W})$. The \hlt{product} of $S$ and $T$ is $ST$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Associativity: $(T_1 T_2)T_3 = T_1 (T_2 T_3)$
\item Identity: $TI = T$ (identity map on $\mathcal{V}$) and $IT = T$ (identity map on $\mathcal{W}$)
\item Distributive: $(S_1 + S_2)T = S_1 T + S_2 T$ and $S(T_1 + T_2) = ST_1 + ST_2$ whenever $T, T_1, T_2 \in \mathcal{L}(\mathcal{U}, \mathcal{V})$ and $S, S_1, S_2 \in \mathcal{L}(\mathcal{V}, \mathcal{W})$
\end{enumerate}
Multiplication of linear maps is not commutative.
\end{remark}

\begin{definition} For $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$, the \hlt{null space} of $T$, denoted $\text{null}(T)$, is the subset of $\mathcal{V}$ consisting of those vectors that maps $T$ to $0$, $\text{null}(T) = \{ \mathbf{v} \in \mathcal{V}: T \mathbf{v} = \mathbf{0} \}$.
\end{definition}

\begin{proposition} \label{prop:nullspaceissubspace} If $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$, then $\text{null}(T)$ is a subspace of $\mathcal{V}$.
\begin{proof} Let $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$. Because $T$ is linear, $T(\mathbf{0}) = \mathbf{0}$, hence $\mathbf{0} \in \text{null}(T)$.\\
Now let $\mathbf{u}, \mathbf{v} \in \text{null}(T)$, and let $\alpha, \beta \in \mathbb{F}$. Then $T(\alpha \mathbf{u} + \beta \mathbf{v}) = \alpha T\mathbf{u} + \beta T\mathbf{v} = \mathbf{0} + \mathbf{0} = \mathbf{0}$, so $\alpha \mathbf{u} + \beta \mathbf{v} \in \text{null}(T)$.\\
Therefore $\text{null}(T)$ is closed under linear combinations and contains $\mathbf{0}$, so it is a subspace of $\mathcal{V}$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:tinjectiveiifnullteqzero} Let $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$. Then $T$ is injective if and only if $\text{null}(T) = \{ \mathbf{0} \}$.
\begin{proof} Let $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$.\\
($\Rightarrow$) Suppose $T$ is injective. If $\mathbf{v} \in \text{null}(T)$, then $T(\mathbf{v}) = \mathbf{0} = T(\mathbf{0})$. Since $T$ is injective, $\mathbf{v} = \mathbf{0}$, so $\text{null}(T) = \{ \mathbf{0} \}$.\\
($\Leftarrow$) Suppose $\text{null}(T) = \{ \mathbf{0} \}$. If $T(\mathbf{u}) = T(\mathbf{v})$, then $T(\mathbf{u} - \mathbf{v}) = T (\mathbf{u}) - T (\mathbf{v}) = \mathbf{0}$, so $\mathbf{u} - \mathbf{v} \in \text{null}(T)$. Thus $\mathbf{u} - \mathbf{v} = \mathbf{}0$, hence $\mathbf{u} = \mathbf{v}$. Therefore $T$ is injective.
\end{proof}
\end{proposition}

Infectivity is equivalent to saying the only vector mapped to 0 is the zero vector.

\begin{definition} For $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$, the \hlt{range} of $T$, denoted $\text{range}(T)$, is subset of $\mathcal{W}$ consisting of vectors that are of from $T \mathbf{v}$ for some $\mathbf{v} \in \mathcal{V}$, where $\text{range}(T) = \{ T \mathbf{v} : \mathbf{v} \in \mathcal{V} \}$.
\end{definition}

\begin{proposition} If $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$, then $\text{range}(T)$ is a subspace of $\mathcal{W}$.
\begin{proof} Let $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$. Because $T(\mathbf{0}) = \mathbf{0}$, we have $\mathbf{0} \in \text{range}(T)$. Now let $\mathbf{w}_1, \mathbf{w}_2 \in \text{range}(T)$ and $\alpha, \beta \in \mathbb{F}$.\\
Then $\exists \ \mathbf{v}_1, \mathbf{v}_2 \in \mathcal{V}$ s.t. $T (\mathbf{v}_1) = \mathbf{w}_1$ and $T (\mathbf{v}_2) = \mathbf{w}_2$. Hence $\alpha \mathbf{w}_1 + \beta \mathbf{w}_2 = \alpha T(\mathbf{v}_1) + \beta T (\mathbf{v}_2) = T(\alpha \mathbf{v}_1 + \beta \mathbf{v}_2)$, so $\alpha \mathbf{w}_1 + \beta \mathbf{w}_2 \in \text{range}(T)$. Therefore $\text{range}(T)$ is closed under addition and scalar multiplication and contains $\mathbf{0}$, hence is a subspace of $\mathcal{W}$.
\end{proof}
\end{proposition}

The range is the image of a linear map, and linearity ensures that images of linear combinations are linear combinations of images. This directly induces subspace structure, making separate closure checks unnecessary.

\begin{theorem} \label{thm:ranknullitythm} \hlt{Rank Nullity Theorem} If $\mathcal{V}$ is finite-dimensional and $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$, then $\text{range}(T)$ is a finite dimensional subspace of $\mathcal{W}$, and $\text{dim}(\mathcal{V}) = \text{dim}(\text{null}(T)) + \text{dim}(\text{range}(T))$.
\begin{proof} Let $(\mathbf{u}_1, \ldots, \mathbf{u}_m)$ be a basis of $\text{null}(T)$. Extend this list to a basis $(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{w}_1, \ldots, \mathbf{w}_n)$ of $\mathcal{V}$. Then $\text{dim}(\mathcal{V}) = m + n$. Claim that $(T \mathbf{w}_1, \ldots, T \mathbf{w}_n)$ is a basis of $\text{range}(T)$.\\
\hlt{Spanning}: Let $\mathbf{v} \in \mathcal{V}$. Write $\mathbf{v} = a_1 \mathbf{u}_1 + \cdots + a_m \mathbf{u}_m + b_1 \mathbf{w}_1 + \cdots + b_n \mathbf{w}_n$. Applying $T$ and using $T(\mathbf{u}_j) = \mathbf{0}$, obtain $T(\mathbf{v}) = b_1 T \mathbf{w}_1 + \cdots + b_n T \mathbf{w}_n$. Thus every vector in $\text{range}(T)$ is a linear combination of $T\mathbf{w}_1, \ldots, T\mathbf{w}_n$, so these vectors span $\text{range}(T)$.\\
\hlt{Linear Independence}: Suppose $c_1 T \mathbf{w}_1 + \cdots + c_n T \mathbf{w}_1 = \mathbf{0}$.\\
Then $T(c_1 \mathbf{w}_1 + \cdots + c_n \mathbf{w}_n) = \mathbf{0}$, so $c_1 \mathbf{w}_1 + \cdots + c_n \mathbf{w}_n \in \text{null}(T)$. Hence there exists scalars $d_1, \ldots, d_m$ such that $c_1 \mathbf{w}_1 + \cdots + c_n \mathbf{w}_n = d_1 \mathbf{u}_1 + \cdots + d_m \mathbf{u}_m$. Rewriting, $d_1 \mathbf{u}_1 + \cdots + d_m \mathbf{u}_m - c_1 \mathbf{w}_1 - \cdots - c_n \mathbf{w}_n = \mathbf{0}$.\\
Because $(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{w}_1, \ldots, \mathbf{w}_n)$ is a basis of $\mathcal{V}$, all coefficients must be $0$. Hence $c_1 = \cdots = c_n = 0$, so the vectors $T \mathbf{w}_1, \ldots, T \mathbf{w}_n$ are linearly independent. Thus $(T\mathbf{w}_1, \ldots, T \mathbf{w}_n)$ is a basis of $\text{range}(T)$, and so $\text{dim}(\text{range}(T)) = n$. Therefore $\text{dim}(\mathcal{V}) = m + n = \text{dim}(\text{null}(T)) + \text{dim}(\text{range}(T))$.
\end{proof}
\end{theorem}

This decomposes $\mathcal{V}$ into directions annihilated by $T$ (nullspace), and those survive under $T$ (by $\mathbf{w}_j$).

\begin{corollary} If $\mathcal{V}$ and $\mathcal{W}$ are finite-dimensional vector spaces such that $\text{dim}(\mathcal{V}) > \text{dim}(\mathcal{W})$, then no linear map from $\mathcal{V}$ to $\mathcal{W}$ is injective.
\begin{proof} Let $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$. By Rank-Nullity Theorem \ref{thm:ranknullitythm}, $\text{dim}(\mathcal{V}) = \text{dim}(\text{null}(T)) + \text{dim}(\text{range}(T))$.\\
Because $\text{range}(T)$ is a subspace of $\mathcal{W}$, we have $\text{dim}(\text{range}(T)) \leq \text{dim}(\mathcal{W})$. Thus
\begin{align}
\text{dim}(\text{null}(T)) = \text{dim}(\mathcal{V}) - \text{dim}(\text{range}(T)) \geq \text{dim}(\mathcal{V}) - \text{dim}(\mathcal{W}) \nonumber
\end{align}
Since $\text{dim}(\mathcal{V}) > \text{dim}(\mathcal{W})$, it follows that $\text{dim}(\text{null}(T)) > 0$. Hence $\text{null}(T)$ contains a nonzero vector, so $T$ is not injective by Proposition \ref{prop:tinjectiveiifnullteqzero}.
\end{proof}
\end{corollary}

You cannot fit a higher-dimensional space injectively into a lower-dimensional one. Rank–nullity quantifies this: the deficit in dimensions must appear as a nontrivial nullspace.

\begin{corollary} If $\mathcal{V}$ and $\mathcal{W}$ are finite-dimensional vector spaces such that $\text{dim}(\mathcal{V}) < \text{dim}(\mathcal{W})$, then no linear map from $\mathcal{V}$ to $\mathcal{W}$ is surjective.
\begin{proof} Let $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$. By Rank-Nullity Theorem \ref{thm:ranknullitythm}, $\text{dim}(\text{range}(T)) = \text{dim}(\mathcal{V}) - \text{dim}(\text{null}(T))$.\\
Because $\text{dim}(\text{null}(T)) \geq 0$, it follows that $\text{dim}(\text{range}(T)) \leq \text{dim}(\mathcal{V}) < \text{dim}(\mathcal{W})$.\\
Thus $\text{range}(T)$ is a proper subspace of $\mathcal{W}$, so $\text{range}(T) \neq \mathcal{W}$. Hence $T$ is not surjective.
\end{proof}
\end{corollary}

A linear map cannot cover a higher-dimensional space if its domain has fewer degrees of freedom. Rank–nullity makes this precise by bounding the dimension of the range by $\text{dim}(\mathcal{V})$.

\begin{definition} A $m \times n$ \hlt{matrix} $\mathcal{M}$ has $m$ rows and $n$ columns,
\begin{align}
\begin{bmatrix}
a_{1,1} & \cdots & a_{1,n} \\
\vdots & \ddots & \vdots \\
a_{m,1} & \cdots & a_{m,n}
\end{bmatrix} \nonumber
\end{align}
\end{definition}

\begin{proposition} Suppose $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$, and $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$, and $(\mathbf{w}_1, \ldots, \mathbf{w}_m)$ is a basis of $\mathcal{W}$. Then $\mathcal{M}(T\mathbf{v}) = \mathcal{M}(T)\mathcal{M}(\mathbf{v})$ for every $\mathbf{v} \in \mathcal{V}$.
\begin{proof} Let
\begin{align}
\mathcal{M}(\mathbf{v}) =
\begin{bmatrix}
\mathbf{b}_1 \\
\vdots
\mathbf{b}_n
\end{bmatrix} \nonumber
\end{align}
so that $\mathbf{v} = \mathbf{b}_1 \mathbf{v}_1 + \cdots + \mathbf{b}_n \mathbf{v}_n$. By linearity of $T$, $T\mathbf{v} = \mathbf{b}_1 T \mathbf{v}_1 +\cdots + \mathbf{b}_n T \mathbf{v}_n$.\\
By definition of $\mathcal{M}(T)$, the $k$-th column of $\mathcal{M}(T)$ is $\mathcal{M}(T \mathbf{v}_k)$, therefore $\mathcal{M}(T\mathbf{v}) = \mathbf{b}_1 \mathcal{M}(T \mathbf{v}_1) +\cdots + \mathbf{b}_n \mathcal{M}(T \mathbf{v}_n)$.\\
But multiplying $\mathcal{M}(T)$ by $\mathcal{M}(\mathbf{v})$ forms exactly this same linear combination of the columns of $\mathcal{M}(T)$.\\
Hence $\mathcal{M}(T \mathbf{v}) = \mathcal{M}(T) \mathcal{M}(\mathbf{v})$.
\end{proof}
\end{proposition}

\begin{definition} A linear map $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$ is \hlt{invertible} if $\exists$ linear map $S \in \mathcal{L}(\mathcal{W}, \mathcal{V})$ such that $ST$ equals identity map on $\mathcal{V}$ and $TS$ equals identity map on $\mathcal{W}$.\\
A linear map $S \in \mathcal{L}(\mathcal{W}, \mathcal{V})$ satisfying $ST = I$ and $TS=I$ is the \hlt{inverse} of $T$.
\end{definition}

\begin{proposition} A linear map is invertible if and only if it is injective and surjective.
\begin{proof} ($\Rightarrow$) Suppose $T$ is invertible with inverse $T^{-1}$. If $T (\mathbf{u}) = T (\mathbf{v})$, then applying $T^{-1}$ gives
\begin{align}
\mathbf{u} = T^{-1}(T (\mathbf{u})) = T^{-1}(T(\mathbf{v})) = \mathbf{v} \nonumber
\end{align}
so $T$ is injective. For any $\mathbf{w} \in \mathcal{W}$, we have $\mathbf{w} = T(T^{-1}(\mathbf{w}))$, so $\mathbf{w} \in \text{range}(T)$, hence $T$ is surjective.\\
($\Leftarrow$) Suppose $T$ is injective and surjective. Define $S: \mathcal{W} \rightarrow \mathcal{V}$ as follows: for each $\mathbf{w} \in \mathcal{W}$, let $S(\mathbf{w})$ be the unique vector $\mathbf{v} \in \mathcal{V}$ such that $T(\mathbf{v}) = \mathbf{w}$. (Surjectivity ensures existence; injectivity ensures uniqueness.)\\
Then for all $\mathbf{w} \in \mathcal{W}$, $T(S(\mathbf{w})) = \mathbf{w}$, so $T \circ S = I_{\mathcal{W}}$.\\
For $\mathbf{v} \in \mathcal{V}$, let $\mathbf{w} = T(\mathbf{v})$, then $S(T(\mathbf{v})) = S(\mathbf{w}) = \mathbf{v}$, so $S \circ T = I_{\mathcal{V}}$.\\
To show that $S$ is linear. Let $\mathbf{w}_1, \mathbf{w}_2 \in \mathcal{W}$, and $\alpha \in \mathbb{F}$, then $T(S(\mathbf{w}_1 + \mathbf{w}_2)) = \mathbf{w}_1 + \mathbf{w}_2 = T(S(\mathbf{w}_1) + S(\mathbf{w}_2))$. By injectivity of $T$, this implies $S(\mathbf{w}_1 + \mathbf{w}_2) = S(\mathbf{w}_1) + S(\mathbf{w}_2)$.\\
Similarly, $T(S(\alpha \mathbf{w}_1)) = \alpha \mathbf{w}_1 = T(\alpha S(\mathbf{w}_1))$ so $S(\alpha \mathbf{w}_1) = \alpha S(\mathbf{w}_1)$. Thus $S$ is linear, hence $S \in \mathcal{L}(\mathcal{W}, \mathcal{V})$, and $S$ is the inverse of $T$. Therefore $T$ is invertible.
\end{proof}
\end{proposition}

Invertibility is equivalent to having a well-defined undo operation. Surjectivity guarantees you can undo every output (existence), and injectivity guarantees the undoing is unique; together giving a bona fide inverse.

\begin{definition} Two vector spaces are \hlt{isomorphic} if there is an invertible linear map from one vector space onto the other. Two isomorphic spaces have the same properties.
\end{definition}

\begin{theorem} \label{thm:isomorphicfindimvectorspace} Two finite-dimensional vector spaces are isomorphic if and only if they have the same $\text{dim}$.
\begin{proof} ($\Rightarrow$) Suppose $\mathcal{V} \cong \mathcal{W}$. Then there exists an invertible linear map $T: \mathcal{V} \rightarrow \mathcal{W}$. Hence $\text{null}(T) = \{ \mathbf{0} \}$ and $\text{range}(T) = \mathcal{W}$. By Rank-Nullity Theorem \ref{thm:ranknullitythm},
\begin{align}
\text{dim}(\mathcal{V}) &= \text{dim}(\text{null}(T)) + \text{dim}(\text{range}(T)) = 0 + \text{dim}(\mathcal{W}) = \text{dim}(\mathcal{W}) \nonumber
\end{align}
($\Leftarrow$) Suppose $\text{dim}(\mathcal{V}) = \text{dim}(\mathcal{W}) = n$. Let $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ be a basis of $\mathcal{V}$ and $(\mathbf{w}_1, \ldots, \mathbf{w}_n)$ be basis of $\mathcal{W}$.\\
Define $T: \mathcal{V} \rightarrow \mathcal{W}$ by $ T\left( \sum\limits_{i=1}^n a_i \mathbf{v}_i \right) = \sum\limits_{i=1}^n a_i \mathbf{w}_i$. This is well-defined because each vector in $\mathcal{V}$ has a unique representation in basis $(\mathbf{v}_i)$, and it is linear because it preserves linear combinations.\\
To show surjectivity, let $\mathbf{w} \in \mathcal{W}$. Then $\mathbf{w} = \sum\limits_{i=1}^n a_i \mathbf{w}_i = T \left( \sum\limits_{i=1}^n a_i \mathbf{v}_i \right)$, so $\mathbf{w} \in \text{range}(T)$.\\
To show injectivity, suppose $T \left( \sum\limits_{i=1}^n a_i \mathbf{v}_i \right) = 0$. Then $\sum\limits_{i=1}^n a_i \mathbf{w}_i = \mathbf{0}$. Because $(\mathbf{w}_i, \ldots, \mathbf{w}_n)$ is linear independent, all $a_i = 0$. Hence the original vector is $\mathbf{0}$, so $\text{null}(T) = \{ \mathbf{0} \}$, and $T$ is injective.\\
Thus $T$ is bijective and linear, hence an isomorphism. Therefore $\mathcal{V} \cong \mathcal{W}$.
\end{proof}
\end{theorem}

Finite-dimensional vector spaces are completely classified by dimension: once dimensions match, you can construct an isomorphism by mapping one basis to another and extending linearly.

\begin{theorem} \label{thm:invertiblelinmap} Suppose $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$ and $(\mathbf{w}_1, \cdots, \mathbf{w}_m)$ is a basis of $\mathcal{W}$. Then $\mathcal{M}$ is an invertible linear map between $\mathcal{L}(\mathcal{V}, \mathcal{W}) $ and $\text{Mat}(m,n,\mathbb{F})$,
\begin{proof} Let $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ be a basis of $\mathcal{V}$, and $(\mathbf{w}_1, \ldots, \mathbf{w}_m)$ be a basis of $\mathcal{W}$. Define $\mathcal{M}: \mathcal{L}(\mathcal{V}, \mathcal{W}) \rightarrow \text{Mat}(m, n, \mathbb{F})$ by sending $T$ to its matrix relative to these basis.\\
First, $\mathcal{M}$ is linear as the matrix entries depend linearly on $T$.\\
To show $\mathcal{M}$ is invertible, it suffices to show it is bijective.\\
\hlt{Injectivity}: Suppose $\mathcal{M}(T) = 0$. Then every column of $\mathcal{M}(T)$ is $\mathbf{0}$, so for each $k = 1, \ldots, n$, $T \mathbf{v}_k = \mathbf{0}$.\\
Because $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$, this implies $T = 0$. Hence $\mathcal{M}$ is injective.\\
\hlt{Surjectivity}: Let $A = [a_{j,k}] \in \text{Mat}(m,n,\mathbb{F})$. Define $T: \mathcal{V} \rightarrow \mathcal{W}$ by specifying its action on basis $T \mathbf{v}_k = \sum\limits_{j=1}^m a_{j,k} \mathbf{w}_j$ for $k = 1, \ldots, n$, and extend linearly to all of $\mathcal{V}$. Then $T$ is linear, and by construction the $k$-th column of $\mathcal{M}(T)$ is the coordinate vector of $T \mathbf{v}_k$, which is exactly the $k$-t column of $A$. Hence $\mathcal{M}(T) = A$.\\
Thus $\mathcal{M}$ is surjective. Therefore $\mathcal{M}$ is bijective and hence an isomorphism.
\end{proof}
\end{theorem}

A linear map is completely determined by its action on a basis. The matrix $\mathcal{M}(T)$ records exactly this data, and every matrix corresponds to a unique such assignment. This establishes a one-to-one correspondence between $\mathcal{L}(\mathcal{V}, \mathcal{W})$ and $\text{Mat}(m,n,\mathbb{F})$.

\begin{proposition} If $\mathcal{V}$ and $\mathcal{W}$ are finite-dimensional, then $\mathcal{L}(\mathcal{V}, \mathcal{W})$ is finite-dimensional, and
\begin{align}
\text{dim}\mathcal{L}(\mathcal{V}, \mathcal{W}) = (\text{dim}(\mathcal{V}))(\text{dim}(\mathcal{W})) \nonumber
\end{align}
\begin{proof} Let $\text{dim}(\mathcal{V}) = n$ and $\text{dim}(\mathcal{W}) = m$. By Theorem \ref{thm:invertiblelinmap}, $\mathcal{L}(\mathcal{V}, \mathcal{W})$ is isomorphic to $\text{Mat}(m,n,\mathbb{F})$.\\
The space $\text{Mat}(m,n,\mathbb{F})$ has dimension $mn$, because a basis is given by the matrices $E_{j,k}$ that have a $1$ in the $(i,j)$-entry and $0$ elsewhere for $1 \leq j \leq m, 1 \leq k \leq n$.\\
Because isomorphic vector spaces have the same dimension, hence $\text{dim}\mathcal{L}(\mathcal{V}, \mathcal{W}) = mn = (\text{dim}(\mathcal{V}))(\text{dim}(\mathcal{W}))$.
\end{proof}
\end{proposition}

The result is a direct consequence of identifying linear maps with matrices. Once this identification is made, counting degrees of freedom becomes counting matrix entries, giving $mn$.

\begin{definition} A linear map from a vector space to itself is an \hlt{operator}, denoted $\mathcal{L}(\mathcal{V})$.
\end{definition}

\begin{theorem} Suppose $\mathcal{V}$ is finite-dimensional. If $T \in \mathcal{L}(\mathcal{V})$, then the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $T$ is invertible.
\item $T$ is injective.
\item $T$ is surjective.
\end{enumerate}
\begin{proof} Suppose $\mathcal{V}$ is finite-dimensional and $T \in \mathcal{L}(\mathcal{V})$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(i)$]$\Rightarrow (ii)$ If $T$ is invertible, then it has an inverse, so it is injective.
\item[$(ii)$]$\Rightarrow (iii)$ Suppose $T$ is injective, then $\text{null}(T) = \{\mathbf{0} \}$, so $\text{dim}(\text{null}(T)) = 0$. By Rank-Nullity Theorem \ref{thm:ranknullitythm}, $\text{dim}(\text{range}(T)) = \text{dim}(\mathcal{V}) - \text{dim}(\text{null}(T)) = \text{dim}(\mathcal{V})$. Because $\text{range}(T)$ is a subspace of $\mathcal{V}$ with same dimension as $\mathcal{V}$, it follows that $\text{range}(T) = \mathcal{V}$, so $T$ is surjective.
\item[$(iii)$]$\Rightarrow (i)$ Suppose $T$ is surjective, then $\text{range}(T) = \mathcal{V}$, so $\text{dim}(\text{range}(T)) = \text{dim}(\mathcal{V})$.\\
By Rank-Nullity Theorem \ref{thm:ranknullitythm}, $\text{dim}(\mathcal{V}) = \text{dim}(\text{null}(T)) + \text{dim}(\text{range}(T)) = \text{dim}(\text{null}(T)) +\text{dim}(\mathcal{V})$, which implies $\text{dim}(\text{null}(T)) = 0$. Thus $\text{null}(T) = \{ \mathbf{0} \}$, so $T$ is injective.\\
Therefore $T$ is both injective and surjective, hence invertible.
\end{enumerate}
\end{proof}
\end{theorem}

In finite dimensions, injectivity and surjectivity are not independent: rank–nullity forces a trade-off between kernel and range dimensions. If one is maximal, the other must be as well, collapsing the distinction between injective, surjective, and invertible.

\subsubsection{Eigenvalues and Eigenvectors}

\begin{definition} \hlt{Addition and Multiplication Modulo $n$}\\
Let $1 < n \in \Z$. For $a,b \in \Z$ with $0 \leq a, b < n$, choose arbitrary $A,B \in \Z$ s.t. $A \equiv a \Mod{n}$, $B \equiv b \Mod{n}$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{Addition Modulo $n$}: $a +_{n} b \defeq (a+b) \Mod{n}$
\item \hlt{Multiplication Modulo $n$}: $a \cdot_{n} b \defeq (a \cdot b) \Mod{n}$
\end{enumerate}
The set $\{0, 1\, \ldots, n-1 \}$ equipped with these operations is denoted $\Z / n \Z$.
\end{definition}

\begin{theorem} \label{thm:primefieldthmforfinitefields} \hlt{The Prime Fields Theorem for Finite Fields}\\
Let $1 < n \in \Z$, then $\Z / n \Z$ is a field if and only if $n$ is prime.
\begin{proof} Let $n > 1$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $n$ is not prime, then $\Z / n \Z$ is not a field.\\
If $n$ is composite, write $n = rs$ with $1 < r, s < n$. Then $\Z / n \Z$, $[r][s] = [rs] = [0]$ while $[r] \neq 0$ and $[s] \neq 0$. Thus $\exists$ nonzero elements whose product is $0$, i.e., nonzero elements in nullspace of multiplication by $[r]$.\\
Equivalently, consider the linear map $\mathcal{M}_{[r]}: \Z / n \Z \rightarrow \Z / n \Z$, $\mathcal{M}_{[r]}([x]) = [r][x]$. Then $[s] \neq 0$ lies in $\text{ker}(\mathcal{M}_{[r]})$, so the nullspace is nontrivial. Hence $\mathcal{M}_{[r]}$ is not injective, so $[r]$ cannot have a multiplicative inverse. Therefore the ring is not a field.
\item[$(\Leftarrow)$] If $n = p$ is prime, then $\Z / p \Z$ is a field.\\
Fix a nonzero element $[x] \in \Z / p \Z$. Consider the linear map $\mathcal{M}_{[x]}: \Z / p \Z \rightarrow \Z / p \Z$, $\mathcal{M}_{[x]}([z]) = [x][z]$.\\
To show $\text{ker}(\mathcal{M}_{[x]}) = \{ 0 \}$. Indeed, if $[x][z] = [0]$, then $p \ \vert \ xz$. Since $p$ is prime and $x \not\equiv 0 (\text{mod} \ p)$, it follows that $p \ \vert \ z$, hence $[z] = 0$. Thus the nullspace is trivial, so $\mathcal{M}_{[x]}$ is injective. Because $\Z / p \Z$ is finite, injectivity implies surjectivity. Hence $\exists$ $[z]$ such that $[x][z] = [1]$, so $[x]$ has a multiplicative inverse.\\
Since every nonzero element is invertible, $\Z / p \Z$ is a field.
\end{enumerate}
\end{proof}
\end{theorem}

The entire theorem reduces to:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Composite $n$: multiplication map has nontrivial nullspace $\rightarrow$ zero-divisors $\rightarrow$ not invertible.
\item Prime $p$: multiplication by nonzero element has trivial nullspace $\rightarrow$ injective $\rightarrow$ bijective $\rightarrow$ invertible.
\end{enumerate}
The argument is parallel to standard results about linear operators:
\begin{align}
\text{ker}(T) = \{ 0 \} \Leftrightarrow T \text{ injective } \Leftrightarrow T \text{ invertible (finite-dimensional case)} \nonumber
\end{align}

\begin{definition} Let $T \in \mathcal{L}(\mathcal{V})$, and $\exists$ direct sum decomposition $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$ where each $\mathcal{U}_j$ is proper subspace of $\mathcal{V}$. The \hlt{restriction} of $T$ to smaller domain $\mathcal{U}_j$ is denoted $T|_{\mathcal{U}_j}$
\end{definition}

\begin{definition} For $T \in \mathcal{L}(\mathcal{V})$ and $\mathcal{U}$ a subspace of $\mathcal{V}$, $\mathcal{U}$ is invariant under $T$ if $\mathbf{u} \in \mathcal{U}$ implies $T \mathbf{u} \in \mathcal{U}$ ($\mathcal{U}$ is invariant under $T$ if $T|_{\mathcal{U}}$ is an operator under $\mathcal{U}$.
\end{definition}

\begin{definition} A scalar $\lambda \in \mathbb{F}$ is an \hlt{eigenvalue} of $T \in \mathcal{L}(\mathcal{V})$ if $\exists$ a nonzero vector $\mathbf{u} \in \mathcal{V}$ such that $T \mathbf{u} = \lambda \mathbf{u}$.\\
A vector $\mathbf{u} \in \mathcal{V}$ is an \hlt{eigenvector} of $T$ if $T \mathbf{u} = \lambda \mathbf{u}$.
\end{definition}

Note that the set of eigenvectors of $T$ corresponding to $\lambda$ equals $\text{null}(T - \lambda I)$.

\begin{theorem} \label{thm:disctingeigenvectorsindependence} Let $T \in \mathcal{L}(\mathcal{V})$, and $\lambda_1, \ldots, \lambda_m$ distinct eigenvalues of $T$, $\mathbf{v}_1, \ldots, \mathbf{v}_m$ are corresponding nonzero eigenvectors, then $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly independent.
\begin{proof} Let $T \in \mathcal{L}(\mathcal{V})$, let $\lambda_1, \ldots, \lambda_m$ be distinct eigenvalues with corresponding nonzero eigenvectors $\mathbf{v}_1, \ldots, \mathbf{v}_m$.\\
Assume for contradiction that $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly dependent. By linear dependence lemma \ref{lem:lineardependencelemma}, $\exists$ smallest index $k \geq 2$ such that $\mathbf{v}_k \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$. Thus $\exists$ scalars $a_1, \ldots, a_{k-1}$ such that $\mathbf{v}_k = a_1 \mathbf{v}_1 + \cdots + a_{k-1} \mathbf{v}_{k-1}$.\\
By minimality of $k$, the list $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$  is linearly independent.\\
Applying $T$ to both sides, $T(\mathbf{v}_k) = a_1 T(\mathbf{v}_1) + \cdots + a_{k-1} T(\mathbf{v}_{k-1})$. Using the eigenvector relations $T(\mathbf{v}_i) = \lambda_i \mathbf{v}_i$,
\begin{align}
\lambda_k \mathbf{v}_k = a_1 \lambda_1 \mathbf{v}_1 + \cdots + a_{k-1} \lambda_{k-1} \mathbf{v}_{k-1} \nonumber
\end{align}
Multiply the original equation by $\lambda_k$,
\begin{align}
\lambda_k \mathbf{v}_k = a_1 \lambda_k \mathbf{v}_1 + \cdots + a_{k-1} \lambda_k \mathbf{v}_{k-1} \nonumber
\end{align}
Subtracting the two equations gives
\begin{align}
0 = \sum\limits_{i=1}^{k-1} a_i (\lambda_k - \lambda_i) \mathbf{v}_i \nonumber
\end{align}
Because $(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$ is linearly independent, each coefficient must be zero, $a_i (\lambda_k - \lambda_i) = 0$ for all $i < k$.\\
Since the eigenvalues are distinct ($\lambda_k - \lambda_i \neq 0$), hence $a_i = 0$ for all $i$.\\
Thus $\mathbf{v}_k = \mathbf{0}$, contradicting that $\mathbf{v}_k$ is an eigenvector. Therefore, $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly independent.
\end{proof}
\end{theorem}

\begin{corollary} Each operator on $\mathcal{V}$ has at most $\text{dim}(\mathcal{V})$ distinct eigenvalues.
\begin{proof} Let $T \in \mathcal{L}(\mathcal{V})$, and $\lambda_1, \ldots, \lambda_m$ be distinct eigenvalues of $T$. For each $i$, choose a corresponding nonzero eigenvector $\mathbf{v}_i \in \mathcal{V}$. By Theorem \ref{thm:disctingeigenvectorsindependence}, the list $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly independent.\\
Let $(\mathbf{w}_1, \ldots, \mathbf{w}_n)$ be a basis of $\mathcal{V}$, then this list spans $\mathcal{V}$ and has length $n = \text{dim}(\mathcal{V})$. Applying Theorem \ref{thm:finitedimlenofvec} to the linearly independent list $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ and the spanning list $(\mathbf{w}_1, \ldots, \mathbf{w}_n)$, we obtain $m \leq n = \text{dim}(\mathcal{V})$.\\
Hence $T$ has at most $\text{dim}(\mathcal{V})$ distinct eigenvalues.
\end{proof}
\end{corollary}

\begin{theorem} Every operator on a finite-dimensional, nonzero complex vector space has an eigenvalue.
\begin{proof} Let $\mathcal{V}$ be a complex vector space with $\text{dim}(\mathcal{V})=n>0$, and let $T \in \mathcal{L}(\mathcal{V})$. Choose $\mathbf{v} \in \mathcal{V}$ with $\mathbf{v} \neq \mathbf{0}$. Consider the list $(\mathbf{v}, T\mathbf{v}, T^2 \mathbf{v}, \ldots, T^n \mathbf{v})$, a list of $n+1$ vectors in a $n$-dimensional space, hence it is linearly dependent. Therefore $\exists$ scalars $a_0, \ldots, a_m \in \C$, not all zero, such that 
\begin{align}
0 = a_0 \mathbf{v} + a_1 T \mathbf{v} + \cdots + a_n T^n \mathbf{v} \nonumber
\end{align}
Let $m$ be largest index such that $a_m \neq 0$. Then $m \geq 1$, and defining the polynomial
\begin{align}
p(z) = a_0 + a_1 z + \cdots + a_m z^m \nonumber
\end{align}
we have $p(T) \mathbf{v} = 0$. As $\C$ is algebraically closed, the polynomial $p$ factors as $p(z) = c(z - \lambda_1) \cdots (z - \lambda_m)$, where $c \neq 0$ and $\lambda_1, \ldots, \lambda_m \in \C$. Substituting $T$ for $\mathbf{z}$, we obtain $p(T) = c(T - \lambda_1 I) \cdots (T - \lambda_m I)$ and hence
\begin{align}
c(T - \lambda_1 I) \cdots (T - \lambda_m I) \mathbf{v} = \mathbf{0} \nonumber
\end{align}
Because $\mathbf{v} \neq \mathbf{0}$, this implies the operator $(T - \lambda_1 I) \cdots (T - \lambda_m I)$ is not injective. Therefore at least one factor $T - \lambda_j I$ is not injective. Hence $\text{ker}(T - \lambda_i I) \neq \{ 0 \}$, so there exists a nonzero vector $\mathbf{u}$ such that $T (\mathbf{u}) = \lambda_j \mathbf{u}$.\\
Thus $\lambda_j$ is an eigenvalue of $T$.
\end{proof}
\end{theorem}

\begin{definition} \hlt{Upper triangular matrices} are of the form
\begin{align}
\begin{bmatrix}
\lambda_1 & & * \\
& \ddots & \\
0 & & \lambda_n
\end{bmatrix} \nonumber
\end{align}
\end{definition}

Next proposition shows useful connection between upper-triangular matrices and invariant subspaces.

\begin{proposition} \label{prop:invariantsubspaceequivalence} Suppose $T \in \mathcal{L}(\mathcal{V})$, and $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$. Following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item The matrix of $T$ with respect to $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is upper triangular.
\item $T \mathbf{v}_k \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ for each $k = 1, \ldots, n$
\item $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is invariant under $T$ for each $k=1, \ldots, n$
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[]
\item[$(i)$]$\Rightarrow (ii)$ If the matrix $T$ is upper triangular, then for each $k$, the coordinate representation of $T \mathbf{v}_k$ has nonzero entries below the $k$-th position. Hence $T \mathbf{v}_k \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$.
\item[$(ii)$]$\Rightarrow (iii)$ Fix $k$, and let $\mathbf{u} \in \text{span}(\mathbf{v_1}, \ldots, \mathbf{v}_k)$. Then $\mathbf{u} = a_1 \mathbf{v}_1 + \cdots + a_k \mathbf{v}_k$.\\
By linearity, $T\mathbf{u} = a_1 T \mathbf{v}_1 + \cdots + a_k T \mathbf{v}_k$. By $(ii)$, each $T\mathbf{v}_j \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$, hence $T \mathbf{u}$ lies in the same span. Therefore, $T(\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)) \subseteq \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$, so the subspace is invariant.
\item[$(iii)$]$\Rightarrow (ii)$ Fix $k$. Since $\mathbf{v}_k \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$ and this subspace is invariant under $T$, $T\mathbf{v}_k \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$.
\item[$(ii)$]$\Rightarrow (i)$ For each $k$, $T \mathbf{v}_k \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$. Then the coordinate vector of $T \mathbf{v}_k$ relative to the basis $(\mathbf{v}_1, \ldots \mathbf{v}_n)$ has zero coefficients on $\mathbf{v}_{k+1}, \ldots, \mathbf{v}_n$. Therefore, the $k$-t column of the matrix of $T$ has zeroes below the diagonal. Since this holds for every $k$, the matrix of $T$ is upper triangular.
\end{enumerate}
\end{proof}
\end{proposition}

An operator is upper triangular if and only if it preserves a nested sequence of subspaces (a flag):
\begin{align}
\{ 0 \} \subset \text{span}(\mathbf(v)_1) \subset \text{span}(\mathbf(v)_1, \mathbf(v)_2) \subset \cdots \subset \text{span}(\mathbf(v)_1, \ldots, \mathbf(v)_n) = \mathcal{V} \nonumber
\end{align}

\begin{theorem} 
\label{thm:complexvecspaceuppertrigmatrix} Suppose $\mathcal{V}$ is a complex vector space and $T \in \mathcal{L}(\mathcal{V})$, then $T$ has an upper-triangular matrix with respect to some basis of $\mathcal{V}$.
\begin{proof} Note that result holds if $\text{dim}(\mathcal{V}) = 1$.\\
Suppose $\text{dim}(\mathcal{V}) > 1$ and the result holds for all complex vector spaces with dimension less than $\text{dim}(\mathcal{V})$. Let $\lambda$ be any eigenvalue of $T$, and let $\mathcal{U} = \text{range}(T - \lambda I)$. As $T - \lambda I$ is not surjective, $\text{dim}(\mathcal{U}) < \text{dim}(\mathcal{V})$, hence $\mathcal{U}$ is invariant under $T$. To show this, let $\mathbf{u} \in \mathcal{U}$, then
\begin{align}
T \mathbf{u} = (T - \lambda I)\mathbf{u} + \lambda \mathbf{u} \nonumber
\end{align}
$(T - \lambda I) \mathbf{u} \in \mathcal{U}$ by definition, and $\lambda \mathbf{u} \in \mathcal{U}$. Hence, equation shows $T \mathbf{u} \in \mathcal{U}$, and $\mathcal{U}$ is invariant under $T$.\\
Hence $T | _{\mathcal{U}}$ is an operator on $\mathcal{U}$. By induction, $\exists$ basis $(\mathbf{u}_1, \ldots \mathbf{u}_m)$ of $\mathcal{U}$ with respect to which $T|_{\mathcal{U}}$ has an upper-triangular matrix. Hence for each $j$,
\begin{align}
T \mathbf{u}_j = (T|_{\mathcal{U}})(\mathbf{u}_j) \in \text{span}(\mathbf{u}_1, \ldots, \mathbf{u}_j) \nonumber
\end{align}
Extend $(\mathbf{u}_1, \ldots, \mathbf{u}_m)$ to basis $(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{v}_1, \ldots, \mathbf{v}_n)$ of $\mathcal{V}$. For each $k$,
\begin{align}
T \mathbf{v}_k = (T - \lambda I)\mathbf{v}_k + \lambda \mathbf{v}_k \nonumber
\end{align}
Definition of $\mathcal{U}$ shows that $(T - \lambda I)\mathbf{v}_k \in \mathcal{U} = \text{span}(\mathbf{u}_1, \ldots, \mathbf{u}_m)$. Hence,
\begin{align}
T \mathbf{v}_k \in \text{span}(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{v}_1, \ldots, \mathbf{v}_k) \nonumber
\end{align}
Hence $T$ has an upper triangular matrix with respect to basis $(\mathbf{u}_1, \ldots, \mathbf{u}_m, \mathbf{v}_1, \ldots, \mathbf{v}_k)$.
\end{proof}
\end{theorem}

\begin{proposition} Suppose $T \in \mathcal{L}(\mathcal{V})$ has an upper-triangular matrix with respect to some basis of $\mathcal{V}$. Then $T$ is invertible if and only if all entries of the diagonal of upper-triangular matrix are non-zero.
\begin{proof} Let $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ be basis of $\mathcal{V}$ with respect to $T$ which has upper-triangular matrix
\begin{align}
\mathcal{M}(T,(\mathbf{v}_1, \ldots, \mathbf{v}_n)) =
\begin{bmatrix}
\lambda_1 & & * \\
& \lambda_2 & & \\
& & \ddots & \\
& & & \lambda_n
\end{bmatrix} \nonumber
\end{align} 
($\Rightarrow$) If $\lambda_1 = 0$, then $T \mathbf{v}_1 = \mathbf{0}$, hence $T$ is not invertible\\
Suppose $1 < k \leq n$ and $\lambda_k = 0$, then $T$ maps each of the vectors $\mathbf{v}_1, \ldots, \mathbf{v}_{k-1}$ into $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$. As $\lambda_k = 0$, the matrix representation implies $T \mathbf{v}_k \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$. Hence, define a linear map
\begin{align}
S: \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k) \rightarrow \text{span}(\mathbf{v}_1, \mathbf{v}_{k-1}) \nonumber
\end{align}
by $S \mathbf{v} = T \mathbf{v}$ for $\mathbf{v} \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$. Then $S$ is $T$ restricted to $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$.\\
Note $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$ has dimension $k$ and $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$ has dimension $k-1$ (as $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is linearly independent). As $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$ has larger dimension than $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$, no linear map from $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_k)$ to $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$ is injective, hence there exists nonzero vector $\mathbf{v} \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k})$ such that $S \mathbf{v} = \mathbf{0}$. Hence $T \mathbf{v} = \mathbf{0}$, and hence $T$ is not invertible.\\
($\Leftarrow$) Suppose $T$ is not invertible. Thus $T$ is not injective, and hence there exists a nonzero vector $\mathbf{v} \in \mathcal{V}$ such that $T \mathbf{v} = \mathbf{0}$. As $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$, then $\mathbf{v} = a_1 \mathbf{v}_1 + \cdots + a_k \mathbf{v}_k$ where $a_1, \ldots, a_k \in \mathbb{F}$ and $a_k \neq 0$ (represent $\mathbf{v}$ as linear combination of $(\mathbf{v}_1, \ldots, \mathbf{v}_k)$ then choose $k$ to be largest index with nonzero coefficient).
\begin{align}
\mathbf{0} &= T \mathbf{v} = T(a_1 \mathbf{v}_1 + \cdots + a_k \mathbf{v}_k) \nonumber \\
&= (a_1 T \mathbf{v}_1 + \cdots + a_{k-1} T \mathbf{v}_{k-1}) + a_k T \mathbf{v}_k \nonumber
\end{align}
The last term in parentheses is in $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$ (due to upper-triangular form).\\
Hence this shows $a_k T \mathbf{v}_k \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$. Multiply by $1/a_k$, which is allowed as $a_k \neq 0$, conclude that $T \mathbf{v}_k \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{k-1})$. Thus when $T \mathbf{v}_k$ is written as linear combination of the basis $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$, the coefficient of $\mathbf{v}_k$ will be $0$, hence $\lambda_k$ must be $0$, completing the proof.
\end{proof}
\end{proposition}

\begin{proposition} Suppose $T \in \mathcal{L}(\mathcal{V})$ has upper-triangular matrix with respect to some basis of $\mathcal{V}$, then eigenvalues of $T$ consists of precisely the entries of the diagonal of the upper-triangular matrix.
\begin{proof} Suppose $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$ with respect to $T$ which $T$ has an upper-triangular matrix
\begin{align}
\mathcal{M}(T,(\mathbf{v}_1, \ldots, \mathbf{v}_n)) =
\begin{bmatrix}
\lambda_1 & & & * \\
& \lambda_2 & & \\
& & \ddots & \\
0 & & & \lambda_n
\end{bmatrix} \nonumber
\end{align} 
Let $\lambda \in \mathbb{F}$, then
\begin{align}
\mathcal{M}(T - \lambda I,(\mathbf{v}_1, \ldots, \mathbf{v}_n)) =
\begin{bmatrix}
\lambda_1 - \lambda & & & * \\
& \lambda_2 - \lambda & & \\
& & \ddots & \\
0 & & & \lambda_n - \lambda
\end{bmatrix} \nonumber
\end{align} 
Hence $T - \lambda I$ is not invertible if and only if $\lambda$ equals one of the $\lambda_j$.\\
Hence $\lambda$ is an eigenvalue of $T$ if and only if $\lambda$ equals one of the $\lambda_j$.
\end{proof}
\end{proposition}

\begin{definition} A \hlt{diagonal matrix} is a square matrix that is $0$ everywhere except possibly along diagonal.
\end{definition}

Operator $T \in \mathcal{L}(\mathcal{V})$ has diagonal matrix $\text{diag}(\lambda_1, \ldots, \lambda_n)$ w.r.t. basis $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ of $\mathcal{V}$ if and only if
\begin{align}
T \mathbf{v}_1 &= \lambda_1 \mathbf{v}_1 \nonumber \\
&\vdots \nonumber \\
T \mathbf{v}_n &= \lambda_n \mathbf{v}_n \nonumber
\end{align}
This follows from definition of matrix of operator with respect to a basis,\\

If an operator has a diagonal matrix w.r.t. some basis, the entires along diagonal are precisely eigenvalues.

\begin{proposition} If $T \in \mathcal{L}(\mathcal{V})$ has $\text{dim}(\mathcal{V})$ distinct eigenvalues, then $T$ has a diagonal matrix with respect to some basis of $\mathcal{V}$.
\begin{proof} Suppose $T \in \mathcal{L}(\mathcal{V})$ has $\text{dim}(\mathcal{V})$ distinct eigenvalues $\lambda_1, \ldots, \lambda_{\text{dim}(\mathcal{V})}$. For each $j$, let $\mathbf{v} \in \mathcal{V}$ be nonzero eigenvector corresponding to the eigenvalue $\lambda_j$. As nonzero eigenvectors corresponding to distinct eigenvalues are linearly independent, $(\mathbf{v}_1, \ldots, \mathbf{v}_{\text{dim}(\mathcal{V})})$ is linearly independent. A linearly independent list of $\text{dim}(\mathcal{V})$ vectors in $\mathcal{V}$ is a basis of $\mathcal{V}$, hence $(\mathbf{v}_1, \ldots, \mathbf{v}_{\text{dim}(\mathcal{V})})$ is a basis of $\mathcal{V}$. With respect to this basis, $T$ is a diagonal matrix.
\end{proof}
\end{proposition}

\begin{proposition} Suppose $T \in \mathcal{L}(\mathcal{V})$. Let $\lambda_1, \ldots, \lambda_m$ denote distinct eigenvalues of $T$. Then the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $T$ has a diagonal matrix with respect to some basis of $\mathcal{V}$;
\item $\mathcal{V}$ has a basis consisting of eigenvectors of $T$;
\item $\exists$ one-dimensional subspaces $\mathcal{U}_1, \ldots, \mathcal{U}_n$ of $\mathcal{V}$, each invariant under $T$, such that
\begin{align}
\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_n \nonumber
\end{align}
\item $\mathcal{V} = \text{null}(T - \lambda_1 I) \oplus \cdots \oplus \text{null}(T - \lambda_m I)$
\item $\text{dim}(\mathcal{V}) = \text{dim}(\text{null}(T - \lambda_1 I)) + \cdots + \text{dim}(\text{null}(T - \lambda_m I))$
\end{enumerate}
\begin{proof} Already shown earlier that $(i)$ and $(ii)$ are equivalent. \\
$(ii) \Rightarrow (iii)$. Suppose $(ii)$ holds, thus $\mathcal{V}$ has a basis $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ consisting of eigenvectors of $T$. For each $j$, let $\mathcal{U}_j = \text{span}(\mathbf{v}_j)$, and hence each $\mathcal{U}_j$ is a one-dimensional subspace of $\mathcal{V}$ that is invariant under $T$ (as each $\mathbf{v}_j$ is an eigenvector of $T$). As $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$, each vector in $\mathcal{V}$ can be written uniquely as a linearly combination of $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$. Hence each vector in $\mathcal{V}$ can be written uniquely as a sum $\mathbf{u}_1 + \cdots + \mathbf{u}_n$, $\mathbf{u}_j \in \mathcal{U}_j$. Thus $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_n$, and $(ii)$ implies $(iii)$.\\
$(iii) \Rightarrow (ii)$. Suppose $(iii)$ holds, hence there are one-dimensional subspaces $\mathcal{U}_1, \ldots, \mathcal{U}_n$ of $\mathcal{V}$, each invariant under $T$, such that $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_n$. For each $j$, let $\mathbf{v}_j$ be nonzero vector in $\mathcal{U}_j$. Then each $\mathbf{v}_j$ is an eigenvector of $T$. As each vector in $\mathcal{V}$ can be written uniquely as $\mathbf{u}_1 + \cdots + \mathbf{u}_n, \mathbf{u}_j \in \mathcal{U}_j$, $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is a basis of $\mathcal{V}$.\\
Hence this shows so far $(i) \Leftrightarrow (ii) \Leftrightarrow (iii)$.\\
$(ii) \Rightarrow (iv)$. Suppose $(ii)$ holds, then $\mathcal{V}$ has a basis consisting of eigenvectors of $T$, and hence every vector in $\mathcal{V}$ is a linear combination of eigenvectors of $T$. Thus 
\begin{align}
\mathcal{V} = \text{null}(T - \lambda_1 I) + \cdots + \text{null}(T - \lambda_m I) \nonumber
\end{align}
Suppose $\mathbf{0} = \mathbf{u}_1 + \cdots + \mathbf{u}_m, \mathbf{u}_j \in \text{null}(T - \lambda_j I)$. As nonzero eigenvectors corresponding to distinct eigenvalues are linearly independent, this implies each $\mathbf{u}_j = \mathbf{0}$. This implies the sum is a direct sum.\\
$(iv) \Rightarrow (v)$. Suppose $(iv)$ holds. By definition, each $\text{null}(T - \lambda_j I)$ is a subspace of $\mathcal{V}$. By definition of direct sum, each $\mathbf{V} \in \mathcal{V}$ can be uniquely expressed as $\mathbf{v} = \mathbf{u}_1 + \cdots + \mathbf{u}_m, \mathbf{u}_j \in \text{null}(T - \lambda_j I)$.\\
The only way for each $\mathbf{u}_1 + \cdots + \mathbf{u}_m = \mathbf{0}$ with each $\mathbf{u}_j \in \text{null}(T - \lambda_j I)$ is if each $\mathbf{u}_j = \mathbf{0}$. Hence these properties together imply $(iv) \Rightarrow (v)$.\\
$(v) \Rightarrow (ii)$. Suppose $(v)$ holds. Choose a basis $\mathbf{v}_j \in \text{null}(T - \lambda_j I)$, and form a list $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ of eigenvectors of $T$, where $n = \text{dim}(\mathcal{V})$. Suppose $a_1 \mathbf{v}_1 + \cdots + a_n \mathbf{v}_n = \mathbf{0}$, where $a_1, \ldots, a_n \in \mathbb{F}$. For each $j = 1, \ldots, m$, let $\mathbf{u}_j$ be sum of all terms $a_k \mathbf{v}_k$ such that $\mathbf{v}_k \in \text{null}(T - \lambda_j I)$. Thus each $\mathbf{u}_j$ is an eigenvector of $T$ with eigenvalue $\lambda_j$, and $\mathbf{u}_1 + \cdots + \mathbf{u}_m = \mathbf{0}$. As nonzero eigenvectors corresponding to distinct eigenvalues are linearly independent, this implies each $\mathbf{u}_j = \mathbf{0}$. As each $\mathbf{u}_j$ is sum of $a_k \mathbf{v}_k$, where $\mathbf{v}_k$ were chosen to be a basis of $\text{null}(T - \lambda_j I)$, this implies all $a_k = 0$. Hence $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is linearly independent and hence is a basis of $\mathcal{V}$.
\end{proof}
\end{proposition}

Next theorem shows that an invariant subspace of dimension $1$ or $2$ always exists.

\begin{theorem} \label{thm:invariantsubspaceexistence} Every operator on a finite-dimensional, nonzero, real vector space has an invariant subspace of dimension $1$ or $2$.
\begin{proof} Let $\mathcal{V}$ be a real vector space, $\text{dim}(\mathcal{V}) = n > 0$, and $T \in \mathcal{L}(\mathcal{V})$. Choose $\mathbf{v} \in \mathcal{V}$ with $\mathbf{v} \neq \mathbf{0}$, then
\begin{align}
(\mathbf{v}, T\mathbf{v}, T^2 \mathbf{v}, \ldots, T^n \mathbf{v}) \nonumber
\end{align}
cannot be linearly independent as $\mathcal{V}$ has dimension $n$, and there are $n+1$ vectors.\\
Hence $\exists \ a_0, \ldots a_n \in \R$ not all $0$, such that
\begin{align}
\mathbf{0} = a_0 \mathbf{v} + a_1 T \mathbf{v} + \cdots + a_n T^n \mathbf{v} \nonumber
\end{align}
Let $a$ be coefficients of a polynomial, then the factored form is
\begin{align}
a_0 + a_1 x + \cdots + a_n x^n = c(x - \lambda_1) \cdots (x - \lambda_m)(x^2 \alpha_1 x + \beta_1) \cdots (x^2 + \alpha_M x + \beta_M) \nonumber
\end{align}
where $c \neq 0 \in \R$, $\lambda_j, \alpha_j, \beta_j \in \R$, $m + M \geq 1$, and equation holds for all $x \in \R$. Then
\begin{align}
\mathbf{0} &= a_0 \mathbf{v} + a_1 T \mathbf{v} + \cdots + a_n T^n \mathbf{v} \nonumber \\
&= (a_0 I + a_1 T + \cdots + a_n T^n)\mathbf{v} \nonumber \\
&= c(T - \lambda_1 I)\cdots (T - \lambda_m I)(T^2 + \alpha_1 T + \beta_1 I) \cdots (T^2 + \alpha_M T + \beta_M I)\mathbf{v} \nonumber
\end{align}
This means $T - \lambda_j I$ is not injective for at least one $j$, or $(T^2 + \alpha_j T + \beta_j I)$ is not injective for at least one $j$. If $T - \lambda_j I$ is not injective for at least one $j$, then $T$ has an eigenvalue, and hence a one-dimensional subspace.\\
If $(T^2 + \alpha_j T + \beta_j T)$ is not injective for some $j$, then there exists nonzero vector $\mathbf{u} \in \mathcal{V}$ such that
\begin{align}
T^2 \mathbf{u} + \alpha_j T \mathbf{u} + \beta_j \mathbf{u} = \mathbf{0} \nonumber
\end{align}
Consider typical element of $\text{span}(\mathbf{u}, T\mathbf{u})$ of form $a \mathbf{u} + b T \mathbf{u}$, where $a, b \in \R$, then
\begin{align}
T(a \mathbf{u} + bT \mathbf{u}) &= a T \mathbf{u} + b T^2 \mathbf{u} \nonumber \\
&= a T \mathbf{u} - b \alpha_j T \mathbf{u} - b \beta_j \mathbf{u} \nonumber
\end{align}
last equality comes from solving the equality for $T^2 \mathbf{u}$. This shows $T(a \mathbf{u} + b T \mathbf{u}) \in \text{span}(\mathbf{u}, T \mathbf{u})$. hence $\text{span}(\mathbf{u}, T \mathbf{u})$ is invariant under $T$.
\end{proof}
\end{theorem}

\begin{definition} Let $\mathcal{U}$, $\mathcal{W}$ be subspaces of $\mathcal{V}$, where $\mathcal{V} = \mathcal{U} \oplus \mathcal{W}$.\\
Let $\mathbf{u} \in \mathcal{U}, \mathbf{w} \in \mathcal{W}, \mathbf{v} \in \mathcal{V}$, and $\mathbf{v} = \mathbf{u} + \mathbf{w}$. Define \hlt{projection} $P_{\mathcal{U}, \mathcal{W}} \in \mathcal{L}(\mathcal{V})$ by $P_{\mathcal{U}, \mathcal{W}} \mathbf{v} = \mathbf{u}$. Note that
\begin{align}
P_{\mathcal{W}, \mathcal{U}} \mathbf{v} &= \mathbf{w}, \ \ \ \mathbf{v} = P_{\mathcal{U}, \mathcal{W}} \mathbf{v} + P_{\mathcal{W}, \mathcal{U}} \mathbf{v} \nonumber \\
\text{range} (P_{\mathcal{U}, \mathcal{W}}) &= \mathcal{U}, \ \ \ \text{null}(P_{\mathcal{U}, \mathcal{W}}) = \mathcal{W} \nonumber
\end{align}
\end{definition}

\begin{theorem} Every operator on an odd-dimensional real vector space has an eigenvalue.
\begin{proof} Suppose $\mathcal{V}$ is a real vector space with odd dimension.\\
Note that result holds for $\text{dim}(\mathcal{V}) = 1$.\\
Suppose $\text{dim}(\mathcal{V})$ is an odd number greater than $1$. Suppose $T \in \mathcal{L}(\mathcal{V})$, to prove $T$ has an eigenvalue. By Theorem \ref{thm:invariantsubspaceexistence}, there is a two-dimensional subspace $\mathcal{U}$ of $\mathcal{V}$ that is invariant under $T$. Let $\mathcal{W}$ be any subspace of $\mathcal{V}$ such that $\mathcal{V} = \mathcal{U} \oplus \mathcal{W}$, then Theorem \ref{thm:invariantsubspaceexistence} guarantees that such a $\mathcal{W}$ exists.\\
As $\mathcal{W}$ has dimension $2$ less than $\text{dim}(\mathcal{V})$, apply induction hypothesis to $T|_{\mathcal{W}}$. However, $\mathcal{W}$ might not be invariant under $T$, which means $T_{\mathcal{W}}$ might not be an operator on $\mathcal{W}$.\\
Define $S \in \mathcal{L}(\mathcal{W})$ by $S\mathbf{w} = P_{\mathcal{W}, \mathcal{U}} (T \mathbf{w})$ for $\mathbf{w} \in \mathcal{W}$. By induction, $S$ has an eigenvalue $\lambda$.\\
Let $\mathbf{w} \in \mathcal{W}$ be nonzero eigenvector for $S$ corresponding to the eigenvalue $\lambda$, thus $(S - \lambda I)\mathbf{w} = 0$. To look for an eigenvector of $T$ in $\mathcal{U} + \text{span}(\mathbf{w})$. Consider typical vector $\mathbf{u} + a \mathbf{w}$ in $\mathcal{U} = \text{span}(\mathbf{w})$, where $\mathbf{u} \in \mathcal{U}$ and $a \in \R$.
\begin{align}
(T - \lambda I)(\mathbf{u} + a \mathbf{w}) &= T \mathbf{u} - \lambda \mathbf{u} + a (T \mathbf{w} - \lambda \mathbf{w}) \nonumber \\
&= T \mathbf{u} - \lambda \mathbf{u} + a (P_{\mathcal{U}, \mathcal{W}}(T \mathbf{w}) + P_{\mathcal{W}, \mathcal{U}}(T \mathbf{w}) - \lambda \mathbf{w}) \nonumber \\
&= T \mathbf{u} - \lambda \mathbf{u} + a (P_{\mathcal{U}, \mathcal{W}}(T \mathbf{w}) + S \mathbf{w} - \lambda \mathbf{w}) \nonumber \\
&= T \mathbf{u} - \lambda \mathbf{u} + a P_{\mathcal{U}, \mathcal{W}}(T \mathbf{w}) \nonumber
\end{align}
Note $T \mathbf{u} \in \mathcal{U}$ (as $\mathcal{U}$ is invariant under $T$), $\lambda \mathbf{u} \in \mathcal{U}$ (as $\mathbf{u} \in \mathcal{U}$), and $a P_{\mathcal{U}, \mathcal{W}}(T \mathbf{w}) \in \mathcal{U}$ (by definition). Hence $T - \lambda I$ maps $\mathcal{U}+ \text{span}(\mathbf{w})$ into  $\mathcal{U}$. As $\mathcal{U}+ \text{span}(\mathbf{w})$ has larger dimension than $\mathcal{U}$ this means $(T - \lambda I)|_{\mathcal{U}+ \text{span}(\mathbf{w})}$ is not injective, and there exists a nonzero vector $\mathbf{v} \in \mathcal{U}+ \text{span}(\mathbf{w})$ such that $(T - \lambda I) \mathbf{v} = \mathbf{0}$. Hence $T$ has an eigenvalue as desired.
\end{proof}
\end{theorem}

\subsubsection{Inner-Product Spaces}

\begin{definition} The \hlt{norm} is defined as $\norm{\mathbf{z}} = \sqrt{\abs{z_1}^2 + \cdots + \abs{z_n}^2} = \sqrt{\innerprod{\mathbf{z}, \mathbf{z}}}$ for $\mathbf{z} = (z_1, \ldots, z_n) \in \C^n$.\\
The norm should be think of as the \hlt{inner product} of $\mathbf{z}$ with itself, $\norm{z}^2 = z_1 \overline{z_1} + \cdots + z_n \overline{z_n}$.\\
If $\mathbf{p} \in \mathcal{P}_m(\mathbb{F})$ (polynomial space0, then $\norm{\mathbf{p}} = \sqrt{\int_0^1 \abs{\mathbf{p}(x)}^2 \ dx}$
\end{definition}

\begin{remark} An inner product on $\mathcal{V}$ that is a function that takes each ordered pair $(\mathbf{u}, \mathbf{v})$ of elements in $\mathcal{V}$ to a number $\innerprod{\mathbf{u}, \mathbf{v}} \in \mathbb{F}$ has following properties:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Positivity: $\innerprod{\mathbf{v}, \mathbf{v}} \geq 0 \ \forall \ \mathbf{v} \in \mathcal{V}$
\item Definiteness: $\innerprod{\mathbf{v}, \mathbf{v}} = 0$ if and only if $\mathbf{v} = \mathbf{0}$
\item Additivity in First Slot: $\innerprod{\mathbf{u} + \mathbf{v}, \mathbf{w}} = \innerprod{\mathbf{u}, \mathbf{w}} + \innerprod{\mathbf{v}, \mathbf{w}}$ for all $\mathbf{u}, \mathbf{v}, \mathbf{w} \in \mathcal{V}$
\item Homogeneity in First Slot: $\innerprod{a\mathbf{v}, \mathbf{w}} = a \innerprod{\mathbf{v}, \mathbf{w}}$ for all $a \in \mathbb{F}$ and $\mathbf{v}, \mathbf{w} \in \mathcal{V}$
\item Conjugate Symmetry: $\innerprod{\mathbf{v}, \mathbf{w}} = \overline{\innerprod{\mathbf{w}, \mathbf{v}}}$ for all $\mathbf{v}, \mathbf{w} \in \mathcal{V}$
\end{enumerate}
\end{remark}

\begin{definition} An \hlt{inner-product space} is a vector space $\mathcal{V}$ with an inner product on $\mathcal{V}$.
\end{definition}

\begin{definition} Two vectors $\mathbf{u}$, $\mathbf{v}$ are \hlt{orthogonal} if $\innerprod{\mathbf{u}, \mathbf{v}} = 0$.
\end{definition}

\begin{theorem} \label{thm:pythagorean} \hlt{Pythagorean Theorem}\\
If $\mathbf{u}, \mathbf{v}$ are orthogonal vectors in $\mathcal{V}$, then $\norm{\mathbf{u} + \mathbf{v}}^2 = \norm{\mathbf{u}}^2 + \norm{\mathbf{v}}^2$.
\begin{proof} Suppose $\mathbf{u}, \mathbf{v}$ are orthogonal vectors in $\mathcal{V}$, then
\begin{align}
\norm{\mathbf{u} + \mathbf{v}}^2 &= \innerprod{\mathbf{u} + \mathbf{v}, \mathbf{u} + \mathbf{v}} \nonumber \\
&= \norm{\mathbf{u}}^2 + \norm{\mathbf{u}}^2 + \innerprod{\mathbf{u}, \mathbf{v}} + \innerprod{\mathbf{v}, \mathbf{u}} \nonumber \\
&= \norm{\mathbf{u}}^2 + \norm{\mathbf{v}}^2 \nonumber
\end{align}
\end{proof}
\end{theorem}

\begin{definition} To compute the \hlt{orthogonal decomposition} of vector $\mathbf{u}$, let $a \in \mathbb{F}$, then
\begin{align}
\mathbf{u} = a \mathbf{v} + (\mathbf{u} - a \mathbf{v}) \nonumber
\end{align}
To choose $a$ so that $\mathbf{v}$ is orthogonal to $(\mathbf{u} - a \mathbf{v})$, which is
\begin{align}
0 = \innerprod{\mathbf{u} - a \mathbf{v}, \mathbf{v}} = \innerprod{\mathbf{u}, \mathbf{v}} - a \norm{\mathbf{v}}^2 \nonumber
\end{align}
Need to choose $a$ to be $\innerprod{\mathbf{u}, \mathbf{v}}/\norm{\mathbf{v}}^2$, where $\mathbf{v} \neq \mathbf{0}$, then
\begin{align}
\mathbf{u} = \frac{\innerprod{\mathbf{u}, \mathbf{v}}}{\norm{\mathbf{v}}^2} \mathbf{v} + \left( \mathbf{u} - \frac{\innerprod{\mathbf{u}, \mathbf{v}}}{\norm{\mathbf{u}}^2}\mathbf{v} \right) \nonumber
\end{align}
\end{definition}

\begin{theorem} \label{thm:cauchyschwarzineq} \hlt{Cauchy-Schwarz Inequality}\\
If $\mathbf{u}, \mathbf{v} \in \mathcal{V}$, then $\abs{\innerprod{\mathbf{u}, \mathbf{v}}} \leq \norm{\mathbf{u}} \norm{\mathbf{u}}$.\\
This inequality is an equality if and only if one of $\mathbf{u}, \mathbf{v}$ is a scalar multiple of the other.
\begin{proof} Let $\mathbf{u}, \mathbf{v} \in \mathcal{V}$. If $\mathbf{v} = \mathbf{0}$, then inequality holds.\\
Assume $\mathbf{u} \neq \mathbf{0}$. Consider orthogonal decomposition
\begin{align}
\mathbf{u} = \frac{\innerprod{\mathbf{u}, \mathbf{v}}}{\norm{\mathbf{v}}^2} \mathbf{v} + \mathbf{w} \nonumber
\end{align}
where $\mathbf{w}$ is orthogonal to $\mathbf{v}$. By Pythagorean Theorem \ref{thm:pythagorean},
\begin{align}
\norm{\mathbf{u}}^2 &= \norm{\frac{\innerprod{\mathbf{u}, \mathbf{v}}}{\norm{\mathbf{v}}^2} \mathbf{v}}^2 + \norm{\mathbf{w}}^2 \nonumber \\
&= \frac{\abs{\innerprod{\mathbf{u}, \mathbf{v}}}^2}{\norm{\mathbf{v}}^2} + \norm{\mathbf{w}}^2 \nonumber \\
&\geq \frac{\abs{\innerprod{\mathbf{u}, \mathbf{v}}}^2}{\norm{\mathbf{v}}^2} \nonumber
\end{align}
Multiplying both sides by $\norm{\mathbf{v}}^2$ then taking square roots give the desired inequality.\\
Note that this is an equality if and only if $\mathbf{w} = \mathbf{0}$, but $\mathbf{w} = \mathbf{0}$ if and only if $\mathbf{u}$ is a multiple of $\mathbf{v}$.
\end{proof}
\end{theorem}

The next result is on triangle inequality, which states the length of any side of a triangle is less than the sum of length of the other two sides.

\begin{theorem} \label{thm:triangleineq} \hlt{Triangle Inequality}\\
If $\mathbf{u}, \mathbf{v} \in \mathcal{V}$, then $\norm{\mathbf{u} + \mathbf{v}} \leq \norm{\mathbf{u}} + \norm{\mathbf{v}}$.\\
This inequality is an equality if and only if one of $\mathbf{u}, \mathbf{v}$ is a nonnegative multiple of the other.
\begin{proof} Let $\mathbf{u}, \mathbf{v} \in \mathcal{V}$, then
\begin{align}
\norm{\mathbf{u} + \mathbf{v}}^2 &= \innerprod{\mathbf{u} + \mathbf{v}, \mathbf{u} + \mathbf{v}} \nonumber \\
&= \innerprod{\mathbf{u}, \mathbf{u}} + \innerprod{\mathbf{v}, \mathbf{v}} + \innerprod{\mathbf{u}, \mathbf{v}} + \innerprod{\mathbf{v}, \mathbf{u}} \nonumber \\
&= \innerprod{\mathbf{u}, \mathbf{u}} + \innerprod{\mathbf{v}, \mathbf{v}} + \innerprod{\mathbf{u}, \mathbf{v}} + \overline{\innerprod{\mathbf{u}, \mathbf{v}}} \nonumber \\
&= \norm{\mathbf{u}}^2 + \norm{\mathbf{u}}^2 + 2 \text{Re}\innerprod{\mathbf{u}, \mathbf{v}} \nonumber \\
&\leq \norm{\mathbf{u}}^2 + \norm{\mathbf{v}}^2 + 2 \abs{\innerprod{\mathbf{u}, \mathbf{v}}} \nonumber \\
&\leq \norm{\mathbf{u}}^2 + \norm{\mathbf{v}}^2 + 2 \norm{\mathbf{u}} \norm{\mathbf{v}} \ \ \ \ \ \text{by Cauchy-Schwarz Inequality Theorem \ref{thm:cauchyschwarzineq}} \nonumber \\
&= (\norm{\mathbf{u}} + \norm{\mathbf{v}})^2 \nonumber
\end{align}
Taking square roots on both sides gives the triangle inequality.\\
Note this is an equality if and only if $\innerprod{\mathbf{u}, \mathbf{v}} = \norm{\mathbf{u}} \norm{\mathbf{v}}$.
\end{proof}
\end{theorem}

The next result is on parallelogram inequality, which states that in any parallelogram, the sum of squares of length of diagonals equals the sum of lengths of the four sides.

\begin{theorem} \label{thm:parallelogramineq} \hlt{Parallelogram Inequality}\\
If $\mathbf{u}, \mathbf{v} \in \mathcal{V}$, then $\norm{\mathbf{u} + \mathbf{v}}^2 + \norm{\mathbf{u} - \mathbf{v}}^2 = 2 (\norm{\mathbf{u}}^2 + \norm{\mathbf{v}}^2)$.
\begin{proof} Let $\mathbf{u}, \mathbf{v} \in \mathcal{V}$, then
\begin{align}
\norm{\mathbf{u} + \mathbf{v}}^2 + \norm{\mathbf{u} - \mathbf{v}}^2 &= \innerprod{\mathbf{u} + \mathbf{v}, \mathbf{u} + \mathbf{v}} + \innerprod{\mathbf{u} - \mathbf{v}, \mathbf{u} - \mathbf{v}} \nonumber \\
&= 2\norm{\mathbf{u}}^2 + 2\norm{\mathbf{v}}^2 + \innerprod{\mathbf{u}, \mathbf{v}} + \innerprod{\mathbf{v}, \mathbf{u}} - \innerprod{\mathbf{u}, \mathbf{v}} - \innerprod{\mathbf{v}, \mathbf{u}} \nonumber \\
&= 2 (\norm{\mathbf{u}}^2 + \norm{\mathbf{v}}^2) \nonumber
\end{align}
\end{proof}
\end{theorem} 

\begin{definition} A list of vectors $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ in $\mathcal{V}$ is \hlt{orthonormal} if 
\begin{align}
\innerprod{\mathbf{e}_j, \mathbf{e}_k} =
\begin{cases}
\innerprod{\mathbf{e}_j, \mathbf{e}_k} = 0 &\text{if } j \neq k \\
\innerprod{\mathbf{e}_j, \mathbf{e}_k} = 1 &\text{if } j = k
\end{cases} \nonumber
\end{align}
\end{definition}

\begin{proposition} \label{prop:sumoforthonormal} If $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ is an orthonormal list of vectors in $\mathcal{V}$, then
\begin{align}
\norm{a_1 \mathbf{e}_1 + \cdots + a_m \mathbf{e}_m}^2 = \abs{a_1}^2 + \cdots + \abs{a_m}^2 \nonumber
\end{align}
for all $a_1, \ldots, a_m \in \mathbb{F}$.
\begin{proof} As each $\mathbf{e}_j$ has norm $1$, by Pythagorean Theorem \ref{thm:pythagorean}, this holds.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:orthonormallistisindep} Every orthonormal list of vectors is linearly independent.
\begin{proof} Suppose $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ is an orthonormal list of vectors in $\mathcal{V}$ and $a_1, \ldots, a_m \in \mathbb{F}$ are such that
\begin{align}
a_1 \mathbf{e}_1 + \cdots + a_m \mathbf{e}_m = \mathbf{0} \nonumber
\end{align}
then $\abs{a_1}^2 + \cdots + \abs{a_m}^2 = 0$ by Proposition \ref{prop:sumoforthonormal}, which means all $a_j$ are $0$ as desired.
\end{proof}
\end{corollary}

\begin{definition} An \hlt{orthonormal basis} of $\mathcal{V}$ is an orthonormal list of vectors in $\mathcal{V}$ that is also a basis.
\end{definition}

\begin{theorem} \label{thm:orthonormalbasisinnerprod} Suppose $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ is an orthonormal basis of $\mathcal{V}$, then
\begin{align}
\mathbf{v} &= \innerprod{\mathbf{v}, \mathbf{e}_1}\mathbf{e}_1 + \cdots + \innerprod{\mathbf{v}, \mathbf{e}_n}\mathbf{e}_n \nonumber \\
\norm{\mathbf{v}}^2 &= \abs{\innerprod{\mathbf{v}, \mathbf{e}_1}}^2 + \cdots + \abs{\innerprod{\mathbf{v}, \mathbf{e}_n}}^2 \nonumber
\end{align}
for every $\mathbf{e} \in \mathcal{V}$.
\begin{proof} Let $\mathbf{v} \in \mathcal{V}$. As $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ is a basis of $\mathcal{V}$, there exists scalars $a_1, \ldots, a_n$ such that
\begin{align}
\mathbf{v} = a_1 \mathbf{e}_1 + \cdots + a_n \mathbf{e}_n \nonumber
\end{align}
Take inner product of both sides of equation with $\mathbf{e}_j$ to get $\innerprod{\mathbf{v}, \mathbf{e}_j} = a_j$, hence proving.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:gramschmidtprocedure} If $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is a linearly independent list of vectors in $\mathcal{V}$, then there exists an orthonormal list $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ of vectors in $\mathcal{V}$ such that
\begin{align}
\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_j) = \text{span}(\mathbf{e}_1, \ldots, \mathbf{e}_j) \nonumber
\end{align}
for $j = 1, \ldots, m$.
\begin{proof} Suppose $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly independent list of vectors in $\mathcal{V}$. To construct $\mathbf{e}$, let $\mathbf{e} = \mathbf{v}_1/\norm{\mathbf{v}_1}$, which satisfies condition for $j=1$.\\
Suppose $j > 1$, and orthonormal list $(\mathbf{e}_1, \ldots, \mathbf{e}_{j-1})$ has been chosen so that
\begin{align}
\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{j-1}) = \text{span}(\mathbf{e}_1, \ldots, \mathbf{e}_{j-1}) \nonumber
\end{align}
Let
\begin{align}
\mathbf{e}_j = \frac{\mathbf{v}_j - \innerprod{\mathbf{v}_j, \mathbf{e}_1}\mathbf{e}_1 - \cdots - \innerprod{\mathbf{v}_j, \mathbf{e}_{j-1}}\mathbf{e}_{j-1}}{\norm{\mathbf{v}_j - \innerprod{\mathbf{v}_j, \mathbf{e}_1}\mathbf{e}_1 - \cdots - \innerprod{\mathbf{v}_j, \mathbf{e}_{j-1}}\mathbf{e}_{j-1}}} \nonumber
\end{align}
Note that $\mathbf{v}_j \notin \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{j-1})$ (as $(\mathbf{v}_1, \ldots, \mathbf{v}_m)$ is linearly independent), hence $\mathbf{v}_j \notin \text{span}(\mathbf{e}_1, \ldots, \mathbf{e}_{j-1})$. Hence above equation is not dividing by $\mathbf{0}$, and $\mathbf{e}_j$ is well defined. Note $\norm{\mathbf{e}_j} = 1$. Let $1 \leq k < j$, then
\begin{align}
\innerprod{\mathbf{e}_j, \mathbf{e}_k} &= \innerprod{\frac{\mathbf{v}_j - \innerprod{\mathbf{v}_j, \mathbf{e}_1}\mathbf{e}_1 - \cdots - \innerprod{\mathbf{v}_j, \mathbf{e}_{j-1}}\mathbf{e}_{j-1}}{\norm{\mathbf{v}_j - \innerprod{\mathbf{v}_j, \mathbf{e}_1}\mathbf{e}_1 - \cdots - \innerprod{\mathbf{v}_j, \mathbf{e}_{j-1}}\mathbf{e}_{j-1}}}, \mathbf{e}_{k}} \nonumber \\
&= \frac{\innerprod{\mathbf{v}_j, \mathbf{e}_k} - \innerprod{\mathbf{v}_j, \mathbf{e}_k}}{\norm{\mathbf{v}_j - \innerprod{\mathbf{v}_j, \mathbf{e}_1}\mathbf{e}_1 - \cdots - \innerprod{\mathbf{v}_j, \mathbf{e}_{j-1}}\mathbf{e}_{j-1}}} \nonumber \\
&= 0 \nonumber
\end{align}
Hence $(\mathbf{e}_1, \ldots, \mathbf{e}_j)$ is an orthonormal list.\\
Note that $\mathbf{v}_j \in \text{span}(\mathbf{e}_1, \ldots, \mathbf{e}_j)$, and hence
\begin{align}
\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_j) \subset \text{span}(\mathbf{e}_1, \ldots, \mathbf{e}_j) \nonumber
\end{align}
Both lists are linearly independent, hence both spaces have dimension $j$, thus must be equal.
\end{proof}
\end{theorem}

\begin{corollary} Every finite-dimensional inner-product space has an orthonormal basis.
\begin{proof} Choose a basis of $\mathcal{V}$. Apply Gram-Schmidt \ref{thm:gramschmidtprocedure} to it to produce an orthonormal list. This orthonormal list is linearly independent by Corollary \ref{cor:orthonormallistisindep} and its span equals $\mathcal{V}$, hence it is an orthonormal basis of $\mathcal{V}$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:linindepextendorthonorm} Every orthonormal list of vectors in $\mathcal{V}$ can be extended to an orthonormal basis of $\mathcal{V}$.
\begin{proof} Suppose $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ is an orthonormal list of vectors in $\mathcal{V}$. Then $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ is linearly independent by Corollary \ref{cor:orthonormallistisindep}, hence it can be extended to a basis $(\mathbf{e}_1, \ldots, \mathbf{e}_m, \mathbf{v}_1, \ldots, \mathbf{v}_n)$ of $\mathcal{V}$. Apply Gram-Schmidt procedure \ref{thm:gramschmidtprocedure} to $(\mathbf{e}_1, \ldots, \mathbf{e}_m, \mathbf{v}_1, \ldots, \mathbf{v}_n)$ of $\mathcal{V}$ to produce an orthonormal list
\begin{align}
(\mathbf{e}_1, \ldots, \mathbf{e}_m, \mathbf{f}_1, \ldots, \mathbf{f}_n) \nonumber
\end{align}
Note Gram-Schmidt leaves first $m$ vectors unchanged as they are already orthonormal. Note that above is an orthonormal basis of $\mathcal{V}$ as it is linearly independent, and its span equals $\mathcal{V}$. Hence this gives the extension of $(\mathbf{e}_1, \ldots, \mathbf{e}_m,)$ to an orthonormal basis of $\mathcal{V}$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:uppertriangmatorthonormalbasis} Suppose $T \in \mathcal{L}(\mathcal{V})$. If $T$ has an upper-triangular matrix with respect to some basis of $\mathcal{V}$, then $T$ has an upper-triangular matrix with respect to some orthonormal basis of $\mathcal{V}$.
\begin{proof} Suppose $T$ has an upper-triangular matrix with respect to some basis $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ of $\mathcal{V}$. Thus $\text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_j)$ is invariant under $T$ for each $j=1, \ldots, n$ by Proposition \ref{prop:invariantsubspaceequivalence}.\\
Apply Gram-Schmidt procedure \ref{thm:gramschmidtprocedure} to $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ to produce an orthonormal basis $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ of $\mathcal{V}$. Note that $\text{span}(\mathbf{e}_1, \ldots, \mathbf{e}_j) = \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_j)$ for each $j$, hence $\text{span}(\mathbf{e}_1, \ldots, \mathbf{e}_j)$ is invariant under $T$ for each $j = 1, \ldots, n$. Hence by Proposition \ref{prop:invariantsubspaceequivalence}, $T$ has an upper-triangular matrix with respect to some orthonormal basis $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:complexvecspaceuppertrigmatorthonbasis} Suppose $\mathcal{V}$ is a complex vector space and $T \in \mathcal{L}(\mathcal{V})$, then $T$ has an upper-triangular matrix with respect to some orthonormal basis of $\mathcal{V}$.
\begin{proof} This follows from Theorem \ref{thm:complexvecspaceuppertrigmatrix} and Corollary \ref{cor:uppertriangmatorthonormalbasis}.
\end{proof}
\end{corollary}

\begin{definition} If $\mathcal{U}$ is a subset of $\mathcal{V}$, then the \hlt{orthogonal complement} of $\mathcal{U}$, denoted $\mathcal{U}^{\perp}$, is the set of all vectors in $\mathcal{V}$ that are orthogonal to every vector in $\mathcal{U}$:
\begin{align}
\mathcal{U}^{\perp} = \{ \mathbf{v} \in \mathcal{V}: \innerprod{\mathbf{v}, \mathbf{u}} = 0 \ \ \forall \ \mathbf{u} \in \mathcal{U} \} \nonumber
\end{align}
Note that $\mathcal{U}^{\perp} \subset \mathcal{V}$, $\mathcal{V}^{\perp} = \{ \mathbf{0} \}$, $\{ \mathbf{0} \}^{\perp} = \mathcal{V}$. Also note that if $\mathcal{U}_1 \subset \mathcal{U}_2$, then $\mathcal{U}^{\perp}_1 \subset \mathcal{U}^{\perp}_2$.
\end{definition}

\begin{theorem} \label{thm:subspaceperpproperty} If $\mathcal{U}$ is a subspace of $\mathcal{V}$, then $\mathcal{V} = \mathcal{U} \oplus \mathcal{U}^{\perp}$.
\begin{proof} Suppose $\mathcal{U}$ is a subspace of $\mathcal{V}$.\\
First show that $\mathcal{V} = \mathcal{U} + \mathcal{U}^{\perp}$. Suppose $\mathbf{v} \in \mathcal{V}$, let $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ be an orthonormal basis of $\mathcal{U}$. Note that
\begin{align}
\mathbf{v} &= \innerprod{\mathbf{v}, \mathbf{e}_1}\mathbf{e}_1 + \cdots + \innerprod{\mathbf{v}, \mathbf{e}_m}\mathbf{e}_m + \mathbf{v} - \innerprod{\mathbf{v}, \mathbf{e}_1}\mathbf{e}_1 - \cdots - \innerprod{\mathbf{v}, \mathbf{e}_m}\mathbf{e}_m \nonumber \\
&= \mathbf{u} + \mathbf{w} \nonumber
\end{align}
Clearly $\mathbf{u} \in \mathcal{U}$. As $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ is an orthonormal list, for each $j$, note that
\begin{align}
\innerprod{\mathbf{w}, \mathbf{e}_j} = \innerprod{v, \mathbf{e}_j} - \innerprod{\mathbf{v}, \mathbf{e}_j} = 0 \nonumber
\end{align}
Hence $\mathbf{w}$ is orthogonal to every vector in $\text{span}(\mathbf{e}_1, \ldots, \mathbf{e}_m)$. Hence $\mathbf{w} \in \mathcal{U}^{\perp}$, hence $\mathbf{v} = \mathbf{u} + \mathbf{w}$, where $\mathbf{u} \in \mathcal{U}$ and $\mathbf{w} \in \mathcal{U}^{\perp}$, completing the proof for equation $\mathcal{V} = \mathcal{U} + \mathcal{U}^{\perp}$.\\
If $\mathbf{v} \in \mathcal{U} \cap \mathcal{U}^{\perp}$, then $\mathbf{v}$ is orthogonal to every vector in $\mathcal{U}$, which implies $(\mathbf{v}, \mathbf{v}) = 0$, which implies $\mathbf{v} = \mathbf{0}$. Hence $\mathcal{U} \cap \mathcal{U}^{\perp} = \{ \mathbf{0} \}$. Note that $\mathcal{V} = \mathcal{U} + \mathcal{U}^{\perp}$ and above equation implies $\mathcal{V} = \mathcal{U} \oplus \mathcal{U}^{\perp}$.
\end{proof}
\end{theorem}

\begin{corollary} If $\mathcal{U} \subset \mathcal{V}$, then $\mathcal{U} = (\mathcal{U}^{\perp})^{\perp}$.
\begin{proof} Suppose $\mathcal{U} \subset \mathcal{V}$.\\
First to show $\mathcal{U} \subset (\mathcal{U}^{\perp})^{\perp}$. Let $\mathbf{u} \in \mathcal{U}$. Then $\innerprod{\mathbf{u}, \mathbf{v}} = 0$ for every $\mathbf{v} \in \mathcal{U}^{\perp}$ by definition of $\mathcal{U}^{\perp}$. As $\mathbf{u}$ is orthogonal to every vector in $\mathcal{U}^{\perp}$, then we have $\mathbf{u} \in (\mathcal{U}^{\perp})^{\perp}$, completing the proof.\\
Next to show $(\mathcal{U}^{\perp})^{\perp} \subset \mathcal{U}$, suppose $\mathbf{v} \in (\mathcal{U}^{\perp})^{\perp}$. By Theorem \ref{thm:subspaceperpproperty}, $\mathbf{v} = \mathbf{u} + \mathbf{w}$, where $\mathbf{u} \in \mathcal{U}$ and $\mathbf{w} \in \mathcal{U}^{\perp}$. Then $\mathbf{v} - \mathbf{u} = \mathbf{w} \in \mathcal{U}^{\perp}$. As $\mathbf{v} \in (\mathcal{U}^{\perp})^{\perp}$ and $\mathbf{u} \in (\mathcal{U}^{\perp})^{\perp}$, then $\mathbf{v} - \mathbf{u} \in (\mathcal{U}^{\perp})^{\perp}$, Thus $\mathbf{v} - \mathbf{u} \in \mathcal{U}^{\perp} \cap (\mathcal{U}^{\perp})^{\perp}$, which implies $\mathbf{v} - \mathbf{u}$ is orthogonal to itself, which implies $\mathbf{v} - \mathbf{u} = \mathbf{0}$, hence $\mathbf{v} = \mathbf{u} \in \mathcal{U}$. Hence $(\mathcal{U}^{\perp})^{\perp} \subset \mathcal{U}$.
\end{proof}
\end{corollary}

\begin{definition} Let $\mathcal{U} \subset \mathcal{V}$. Decomposition $\mathcal{V} = \mathcal{U} \oplus \mathcal{U}^{\perp}$ means each vector $\mathbf{v} \in \mathbf{V}$ can be written as $\mathbf{v} = \mathbf{u} + \mathbf{w}$, where $\mathbf{u} \in \mathcal{U}$ and $\mathbf{w} \in \mathcal{U}^{\perp}$.\\
The \hlt{orthogonal projection} of $\mathcal{V}$ onto $\mathcal{U}$ is an operator on $\mathcal{V}$, denoted $P_{\mathcal{U}} = P_{\mathcal{U}, \mathcal{U}^{\perp}}$.\\
Note that for $\mathbf{v} \in \mathcal{V}$, denote $P_{\mathcal{U}}\mathbf{v}$ to be vector $\mathbf{u}$ in decomposition $\mathbf{v} = \mathbf{u} + \mathbf{w}$.
\end{definition}

\begin{remark} \hlt{Properties of Orthogonal Projection} of $P_{\mathcal{U}} \in \mathcal{L}(\mathcal{V})$:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\text{range}P_{\mathcal{U}} = \mathcal{U}$
\item $\text{null}P_{\mathcal{U}} = \mathcal{U}^{\perp}$
\item $\mathbf{v} - P_{\mathcal{U}}\mathbf{v} \in \mathcal{U}^{\perp}$ for every $\mathbf{v} \in \mathcal{V}$
\item $P_{\mathcal{U}}^2 = P_{\mathcal{U}}$
\item $\norm{P_{\mathcal{U}} \mathbf{v}} \leq \norm{\mathbf{v}}$ for every $\mathbf{v} \in \mathcal{V}$
\item If $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ is an orthonormal basis of $\mathcal{U}$, then $P_{\mathcal{U}} \mathbf{v} = \innerprod{\mathbf{v}, \mathbf{e}_1}\mathbf{e}_1 + \cdots + \innerprod{\mathbf{v}, \mathbf{e}_m}\mathbf{v}_m$ for every $\mathbf{v} \in \mathcal{V}$
\end{enumerate}
\end{remark}

\begin{proposition} Suppose $\mathcal{U} \subset \mathcal{V}$ and $\mathbf{v} \in \mathcal{V}$. Then
\begin{align}
\norm{\mathbf{v} - P_{\mathcal{U}} \mathbf{v}} \leq \norm{\mathbf{v} - \mathbf{u}} \nonumber
\end{align}
for every $\mathbf{v} \in \mathcal{U}$. Furthermore, if $\mathbf{v} \in \mathcal{U}$ and the inequality above is an equality, then $\mathbf{v} = P_{\mathcal{U}} \mathbf{v}$.
\begin{proof} Suppose $\mathbf{u} \in \mathcal{U}$, then
\begin{align}
\norm{\mathbf{v} - P_{\mathcal{U}}\mathbf{v}} &\leq \norm{\mathbf{v} - P_{\mathcal{U}}\mathbf{v}}^2 + \norm{P_{\mathcal{U}}\mathbf{v} - \mathbf{u}}^2 \nonumber \\
&= \norm{(\mathbf{v} - P_{\mathcal{U}}\mathbf{v}) + (P_{\mathcal{U}}\mathbf{v} - \mathbf{u})}^2 \ \ \ \text{by Pythagorean Theorem \ref{thm:pythagorean}} \nonumber \\
&= \norm{\mathbf{v} - \mathbf{u}}^2 \nonumber
\end{align}
Note that for equality case, this happens if and only if $\norm{P_{\mathcal{U}}\mathbf{v} - \mathbf{u}} = 0$, hence $\mathbf{u} = P_{\mathcal{U}}\mathbf{v}$.
\end{proof}
\end{proposition}

\begin{definition} A \hlt{linear functional} on $\mathcal{V}$ is a linear map from $\mathcal{V}$ to the scalars $\mathbb{F}$.
\end{definition}

\begin{theorem} Suppose $\varphi$ is a linear functional on $\mathcal{V}$. Then there is a unique vector $\mathbf{v} \in \mathcal{V}$ such that
\begin{align}
\varphi(\mathbf{u}) = \innerprod{\mathbf{u}, \mathbf{v}} \nonumber
\end{align}
for every $\mathbf{u} \in \mathcal{V}$.
\begin{proof} First, to show $\exists$ a vector $\mathbf{v} \in \mathcal{V}$ such that $\varphi(\mathbf{u}) = \innerprod{\mathbf{u}, \mathbf{v}}$ for every $\mathbf{u} \in \mathcal{V}$.\\
Let $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ be an orthonormal basis of $\mathcal{V}$, then
\begin{align}
\varphi{\mathbf{u}} &= \varphi(\innerprod{\mathbf{u}, \mathbf{e}_1}\mathbf{e}_1 + \cdots + \innerprod{\mathbf{u}, \mathbf{e}_n}\mathbf{e}_n) \ \ \ \text{by Theorem \ref{thm:orthonormalbasisinnerprod}} \nonumber \\
&= \innerprod{\mathbf{u}, \mathbf{e}_1}\varphi(\mathbf{e}_1) + \cdots + \innerprod{\mathbf{u}, \mathbf{e}_n}\varphi(\mathbf{e}_n) \nonumber \\
&= \innerprod{\mathbf{u}, \overline{\varphi(\mathbf{e}_1)}\mathbf{e}_1 + \cdots + \overline{\varphi(\mathbf{e}_n)}\mathbf{e}_n} \nonumber
\end{align}
for every $\mathbf{u} \in \mathcal{V}$. Setting $\mathbf{v} = \overline{\varphi(\mathbf{e}_1)}\mathbf{e}_1 + \cdots + \overline{\varphi(\mathbf{e}_n)}\mathbf{e}_n$, then $\varphi(\mathbf{u}) = \innerprod{\mathbf{u}, \mathbf{v}}$ for every $\mathbf{v} \in \mathcal{V}$ as desired.\\
Next, to prove that only one vector $\mathbf{v} \in \mathcal{V}$ has the desired behaviour. Suppose $\mathbf{v}_1, \mathbf{v}_2 \in \mathcal{V}$, such that
\begin{align}
\varphi(\mathbf{u}) = \innerprod{\mathbf{u}, \mathbf{v}_1} = \innerprod{\mathbf{u}, \mathbf{v}_2} \nonumber
\end{align}
for every $\mathbf{u} \in \mathcal{V}$. Taking $\mathbf{u} = \mathbf{v}_1 - \mathbf{v}_2$ shows $\mathbf{v}_1 - \mathbf{v}_2 = \mathbf{0}$, hence $\mathbf{v}_1 = \mathbf{v}_2$, proving uniqueness.
\end{proof}
\end{theorem}

\begin{definition} Let $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$, where $\mathcal{W}, \mathcal{V}$ are finite-dimensional nonzero, inner-product space over $\mathbb{F}$. Fix $\mathbf{w} \in \mathcal{W}$, and consider the linear functional $f_{\mathbf{w}}: \mathcal{V} \rightarrow \mathbb{F}$ given by $f_{\mathbf{w}}(\mathbf{v}) = \innerprod{T\mathbf{v}, \mathbf{w}}$. There is a unique vector $\mathbf{y} \in \mathcal{V}$ such that $f_{\mathbf{w}}(\mathbf{v}) = \innerprod{\mathbf{v}, \mathbf{y}} \ \forall \ \mathbf{v} \in \mathcal{V}$, and let this unique vector $\mathbf{y}$ be $T^* \mathbf{w}$. Hence
\begin{align}
\innerprod{T\mathbf{v}, \mathbf{w}} = \innerprod{\mathbf{v}, T^* \mathbf{w}} \nonumber
\end{align}
for all $\mathbf{v} \in \mathcal{V}$. Doing this for every $\mathbf{w} \in \mathcal{W}$ gives a map $T^*: \mathcal{W} \rightarrow \mathcal{V}$, which is the \hlt{adjoint} of $T$.
\end{definition}

\begin{remark} \hlt{Property of Adjoints}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item Additivity: $(S + T)^* = S^* + T^*$ for all $S, T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$
\item Conjugate Homogeneity: $(aT)^* = \overline{a}T^*$ for all $a \in \mathbb{F}$ and $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$
\item Adjoint of Adjoint: $(T^*)^* = T$ for all $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$
\item Identity: $I^* = I$, where $I$ is the identity operator on $\mathcal{V}$
\item Products: $(ST^*) = T^* S^*$ for all $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$ and $S \in \mathcal{L}(\mathcal{W}, \mathcal{U})$ ($\mathcal{U}$ is inner-product space over $\mathbb{F}$)
\end{enumerate}
\end{remark}

Next result shows relationship between null space and range of a linear map and its adjoint.

\begin{proposition} Suppose $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$, then
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\text{null}(T^*) = (\text{range}(T))^{\perp}$
\item $\text{range}(T^*) = (\text{null}()T)^{\perp}$
\item $\text{null}(T) = (\text{range}(T^*))^{\perp}$
\item $\text{range}(T) = (\text{null}(T^*))^{\perp}$
\end{enumerate}
\begin{proof} For (i), let $\mathbf{w} \in \mathcal{W}$, then
\begin{align}
\mathbf{w} \in \text{null}(T^*) \Leftrightarrow T^* \mathbf{w} = \mathbf{0} \Leftrightarrow \innerprod{\mathbf{v}, T^* \mathbf{w}} = \mathbf{0} \ \forall \ \mathbf{v} \in \mathcal{V} \Leftrightarrow \innerprod{T \mathbf{v}, \mathbf{w}} = \mathbf{0} \ \forall \ \mathbf{v} \in \mathcal{V} \Leftrightarrow \mathbf{w} \in \text{range(T)}^{\perp} \nonumber
\end{align}
Take orthogonal complement of both sides of $(i)$ to get $(iv)$.\\
Replace $T$ with $T^*$ in $(i)$ and $(iv)$ to get $(iii)$ and $(ii)$.
\end{proof}
\end{proposition}

\begin{definition} The \hlt{conjugate transpose} of an $m \times n$ matrix is obtained by interchanging rows and columns, then taking the complex conjugate of each entry.
\end{definition}

\begin{proposition} Suppose $T \in \mathcal{L}(\mathcal{V}, \mathcal{W})$. If $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ is an orthonormal basis of $\mathcal{V}$ and $(\mathbf{f}_1, \ldots, \mathbf{f}_m)$ is an orthonormal basis of $\mathcal{W}$, then
\begin{align}
\mathcal{M}(T^*, (\mathbf{f}_1, \ldots, \mathbf{f}_m), (\mathbf{e}_1, \ldots, \mathbf{e}_n)) \nonumber
\end{align}
is the conjugate transpose of
\begin{align}
\mathcal{M}(T, (\mathbf{e}_1, \ldots, \mathbf{e}_n), (\mathbf{f}_1, \ldots, \mathbf{f}_m)) \nonumber
\end{align}
\begin{proof} Let $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ be orthonormal basis of $\mathcal{V}$ and $(\mathbf{f}_1, \ldots, \mathbf{f}_m)$ be orthonormal basis of $\mathcal{W}$. The $k$ column of $\mathcal{M}(T)$ is obtained by writing $T \mathbf{e}_k$ as a linear combination of $\mathbf{f}_j$, and the scalars used in this linear combination becomes the $k$ column of $\mathcal{M}(T)$. As $(\mathbf{f}_1, \ldots, \mathbf{f}_m)$ is an orthonormal basis of $\mathcal{W}$, $T \mathbf{e}_k$ can be written as a linear combination of the $\mathbf{f}_j$.
\begin{align}
T \mathbf{e}_k = \innerprod{T \mathbf{e}_k, \mathbf{f}_1} \mathbf{f}_1 + \cdots + \innerprod{T \mathbf{e}_k, \mathbf{f}_m} \mathbf{f}_m \nonumber
\end{align}
Hence the entry in row $j$ column $k$ of $\mathcal{M}(T)$ is $\innerprod{T \mathbf{e}_k, \mathbf{f}_j}$. Replace $T$ with $T^*$ and interchanging the roles played by $\mathbf{e}_j$ and $\mathbf{f}_j$, then the entry in row $j$ column $k$ of $\mathcal{K}(T^*)$ is $\innerprod{T^* \mathbf{f}_k, \mathbf{e}_j} = \innerprod{\mathbf{f}_k, T \mathbf{e}_j} = \overline{\innerprod{T \mathbf{e}_j, \mathbf{f}_k}}$, which equals the complex conjugate of entry in row $k$ column $j$ of $\mathcal{M}(T)$. Hence $\mathcal{M}(T^*)$ equals conjugate transpose of $\mathcal{M}(T)$.
\end{proof}
\end{proposition}

\subsubsection{Operators on Inner-Product Spaces}

\begin{definition} An operator $T \in \mathcal{L}(\mathcal{V})$ is \hlt{self-adjoint} or \hlt{Hermitian} if $T = T^*$.
\end{definition}

A self-adjoint operator is analogous to a real number. It plays similar role to complex conjugation on $\C$.

\begin{proposition} Every eigenvalue of a self-adjoint operator is real.
\begin{proof} Let $T \in \mathcal{L}(\mathcal{V})$. Let $\lambda$ be an eigenvalue of $T$, $\mathbf{v}$ be nonzero vector in $\mathcal{V}$ such that $T\mathbf{v} = \lambda \mathbf{v}$. Then
\begin{align}
\lambda \norm{\mathbf{v}}^2 &= \innerprod{\lambda \mathbf{v}, \mathbf{v}} = \innerprod{T \mathbf{v}, \mathbf{v}} = \innerprod{\mathbf{v}, T\mathbf{v}} = \innerprod{\mathbf{v}, \lambda, \mathbf{v}} = \overline{\lambda} \norm{\mathbf{v}}^2 \nonumber
\end{align}
Note that $\lambda = \overline{\lambda}$, hence $\lambda$ is real.	
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:complexinnerprodzero} If $\mathcal{V}$ is a complex inner-product space and $T$ is an operator on $\mathcal{V}$ such that $\innerprod{T \mathbf{v}, \mathbf{v}} = 0$ for all $\mathbf{v} \in \mathcal{V}$, then $T = 0$.
\begin{proof} Suppose $\mathcal{V}$ is a complex inner-product space and $T \in \mathcal{L}(\mathcal{V})$, then
\begin{align}
\innerprod{T \mathbf{u}, \mathbf{w}} = \frac{\innerprod{T(\mathbf{u} + \mathbf{w}), \mathbf{u} + \mathbf{w}} - \innerprod{T(\mathbf{u} - \mathbf{w}), \mathbf{u} - \mathbf{w}}}{4} + \frac{\innerprod{T(\mathbf{u} + i\mathbf{w}), \mathbf{u} + i \mathbf{w}} - \innerprod{T(\mathbf{u} - i\mathbf{w}), \mathbf{u} - i \mathbf{w}}}{4}i \nonumber 
\end{align}
for all $\mathbf{u}, \mathbf{w} \in \mathcal{V}$. Note each term on right side is of form $\innerprod{T \mathbf{v}, \mathbf{v}}$ for appropriate $\mathbf{v} \in \mathcal{V}$. If $\innerprod{T \mathbf{v}, \mathbf{v}} = 0$ for all $\mathbf{v} \in \mathcal{V}$, then equation above implies $\innerprod{T \mathbf{u}, \mathbf{w}} = 0$ for all $\mathbf{u}, \mathbf{w} \in \mathcal{V}$. This implies $T = 0$ (take $\mathbf{w} = T \mathbf{u}$).
\end{proof}
\end{proposition}

\begin{corollary} Let $\mathcal{V}$ be complex inner-product space and let $T \in \mathcal{L}(\mathcal{V})$. Then $T$ is self-adjoint if and only if $\innerprod{T \mathbf{v}, \mathbf{v}} \in \R$ for every $\mathbf{v} \in \mathcal{V}$.
\begin{proof} Let $\mathbf{v} \in \mathcal{V}$. Then
\begin{align}
\innerprod{T \mathbf{v}, \mathbf{v}} - \overline{\innerprod{T \mathbf{v}, \mathbf{v}}} = \innerprod{T \mathbf{v}, \mathbf{v}} - \innerprod{\mathbf{v}, T \mathbf{v}} = \innerprod{T \mathbf{v}, \mathbf{v}} - \innerprod{T^* \mathbf{v}, \mathbf{v}} = \innerprod{(T - T^*)\mathbf{v}, \mathbf{v}} \nonumber
\end{align}
If $\innerprod{T \mathbf{v}, \mathbf{v}} \in \R$ for every $\mathbf{v} \in \mathcal{V}$, then the left-side of equation equals $0$, hence $\innerprod{(T - T^*)\mathbf{v}, \mathbf{v}} = 0$ for every $\mathbf{v} \in \mathcal{V}$. This implies $T - T^* = 0$ by Proposition \ref{prop:complexinnerprodzero}, hence $T$ is self-adjoint.\\
Conversely, if $T$ is self-adjoint, then the right-side of the equation above equals $0$, hence $\innerprod{T \mathbf{v}, \mathbf{v}} = \overline{\innerprod{T \mathbf{v}, \mathbf{v}}}$ for every $\mathbf{v} \in \mathcal{V}$. This implies $\innerprod{T \mathbf{v}, \mathbf{v}} \in \R$ for every $\mathbf{v} \in \mathcal{V}$ as desired.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:selfadjointopt0} If $T$ is a self-adjoint operator on $\mathcal{V}$ such that $\innerprod{T \mathbf{v}, \mathbf{v}} = 0$ for all $\mathbf{v} \in \mathcal{V}$, then $T = 0$.
\begin{proof} Let $\mathcal{V}$ be a real inner-product space, and $T$ a self-adjoint operator on $\mathcal{V}$. For $\mathbf{u}, \mathbf{v} \in \mathcal{V}$, we have
\begin{align}
\innerprod{T \mathbf{u}, \mathbf{v}} = \frac{\innerprod{T(\mathbf{u} + \mathbf{w}), \mathbf{u} + \mathbf{w}} - \innerprod{T(\mathbf{u} - \mathbf{w}), \mathbf{u} - \mathbf{w}}}{4} \nonumber
\end{align}
Note that
\begin{align}
\innerprod{T \mathbf{w}, \mathbf{u}} &= \innerprod{\mathbf{w}, T \mathbf{u}} \ \ \ \text{as $T$ is self-adjoint} \nonumber \\
&= \innerprod{T \mathbf{u}, \mathbf{w}} \ \ \ \text{as this is real inner-product space} \nonumber
\end{align}
If $\innerprod{T \mathbf{v}, \mathbf{v}} = 0$ for all $\mathbf{v} \in \mathcal{V}$, then first equation implies $\innerprod{T \mathbf{u}, \mathbf{w}} = 0$ for all $\mathbf{u}, \mathbf{w} \in \mathcal{V}$.\\
This implies $T = 0$ (taking $\mathbf{w} = T \mathbf{u}$).
\end{proof}
\end{proposition}

\begin{definition} An operator on an inner-product space is \hlt{normal} if it commutes with its adjoint, i.e., $T \in \mathcal{L}(\mathcal{V})$ is normal if $T T^* = T^* T$. Every self-adjoint operator is normal.
\end{definition}

The next proposition implies $\text{null}(T) = \text{null}(T^*)$ for every normal operator $T$.

\begin{proposition} \label{prop:nullTeqnullTselfadjt} An operator $T \in \mathcal{L}(\mathcal{V})$ is normal if and only if $\norm{T \mathbf{v}} = \norm{T^* \mathbf{v}}$ for all $\mathbf{v} \in \mathcal{V}$.
\begin{proof} Let $T \in \mathcal{L}(\mathcal{V})$. Note that
\begin{align}
T \text{ is normal} &\Leftrightarrow T^*T - TT^* = 0 \nonumber \\
&\Leftrightarrow \innerprod{(T^* T - TT^*)\mathbf{v}, \mathbf{v}} = 0 \ \ \text{for all } \mathbf{v} \in \mathcal{V} \ \ \text{(by Proposition \ref{prop:selfadjointopt0})} \nonumber \\
&\Leftrightarrow \innerprod{T^*T \mathbf{v}, \mathbf{v}} = \innerprod{TT^*\mathbf{v}, \mathbf{v}} \ \ \text{for all } \mathbf{v} \in \mathcal{V} \nonumber \\
&\Leftrightarrow \norm{T \mathbf{v}}^2 = \norm{T^* \mathbf{v}}^2 \ \ \text{for all } \mathbf{v} \in \mathcal{V} \nonumber
\end{align}
hence proving the proposition.
\end{proof}
\end{proposition}

The next corollary implies that a normal operator and its adjoint have the same eigenvectors.

\begin{corollary} \label{cor:sameeigenadjtandnormal} Suppose $T \in \mathcal{L}(\mathcal{V})$ is normal. If $\mathbf{v} \in \mathcal{V}$ is an eigenvector of $T$ with eigenvalue $\lambda \in \mathbb{F}$, then $\mathbf{v}$ is also an eigenvector of $T^*$ with eigenvalue $\overline{\lambda}$.
\begin{proof} Suppose $\mathbf{v} \in \mathcal{V}$ is an eigenvector of $T$ with eigenvalue $\lambda$, hence $(T - \lambda I)\mathbf{v} = \mathbf{0}$. As $T$ is normal, so is $T - \lambda I$. By Proposition \ref{prop:nullTeqnullTselfadjt},
\begin{align}
0 = \norm{(T - \lambda I)\mathbf{v}} = \norm{(T - \lambda I)^* \mathbf{v}} = \norm{(T^* - \overline{\lambda}I)\mathbf{v}} \nonumber
\end{align}
hence $\mathbf{v}$ is an eigenvector of $T^*$ with eigenvalue $\overline{\lambda}$ as desired.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:normaleigenvectororthogonal} If  $T \in \mathcal{L}(\mathcal{V})$ is normal, eigenvectors of $T$ with unique eigenvalues are orthogonal.
\begin{proof} Suppose $T \in \mathcal{L}(\mathcal{V})$ is normal, and $\alpha, \beta$ are distinct eigenvalues of $T$, with corresponding eigenvectors $\mathbf{u}, \mathbf{v}$. Thus $T \mathbf{u} = \alpha \mathbf{u}$ and $T \mathbf{v} = \beta \mathbf{v}$. By Corollary \ref{cor:sameeigenadjtandnormal}, $T^* \mathbf{v} = \overline{\beta}\mathbf{v}$, hence
\begin{align}
(\alpha - \beta)\innerprod{\mathbf{u} - \mathbf{v}} &= \innerprod{\alpha \mathbf{u}, \mathbf{v}} - \innerprod{\mathbf{u}, \overline{\beta}\mathbf{v}} \nonumber \\
&= \innerprod{T \mathbf{u}, \mathbf{v}} - \innerprod{\mathbf{u}, T^* \mathbf{v}} = 0 \nonumber
\end{align}
As $\alpha \neq \beta$, equation implies $\innerprod{\mathbf{u}, \mathbf{v}} = 0$, hence $\mathbf{u}$ and $\mathbf{v}$ are orthogonal as desired.
\end{proof}
\end{corollary}

The complex spectral theorem implies that every self-adjoint operator on a finite dimensional complex inner-product space has a diagonal matrix with respect to some orthonormal basis.

\begin{theorem} \label{thm:complexspectralthm} \hlt{Complex Spectral Theorem}\\
Suppose $\mathcal{V}$ is a complex inner-product space and $T \in \mathcal{L}(\mathcal{V})$. Then $\mathcal{V}$ has an orthonormal basis consisting of eigenvectors of $T$ if and only if $T$ is normal.
\begin{proof} $(\Rightarrow)$ First, suppose $\mathcal{V}$ has an orthonormal basis consisting of eigenvectors of $T$, and $T$ has a diagonal matrix with respect to this basis. The matrix of $T^*$ is from taking the conjugate transpose of $T$, hence $T^*$ also has a diagonal matrix. Any two diagonal matrices commute, hence $T$ commutes with $T^*$, and $T$ is normal.\\
$(\Leftarrow)$ Suppose $T$ is normal. There is an orthonormal basis $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ of $\mathcal{V}$ with respect to which $T$ has an upper-triangular matrix by Corollary \ref{cor:complexvecspaceuppertrigmatorthonbasis}, hence
\begin{align}
\mathcal{M}(T, (\mathbf{e}_1, \ldots, \mathbf{e}_n)) &=
\begin{bmatrix}
a_{1,1} & \cdots & a_{1,n} \\
\vdots & \ddots & \vdots \\
0 & \cdots & a_{n,n}
\end{bmatrix} \nonumber
\end{align}
Note that from the matrix, $\norm{T \mathbf{e}_1}^2 = \abs{a_{1,1}}^2$, and $\norm{T^* \mathbf{e}_1}^2 = \abs{a_{1,1}}^2 + \abs{a_{1,2}}^2 + \cdots + \abs{a_{1,n}}^2$.\\
As $T$ is normal, $\norm{T \mathbf{e}_1} = \norm{T^* \mathbf{e}_1}$ by Proposition \ref{prop:nullTeqnullTselfadjt}, hence above equations imply all entires in first row of matrix except possibly first entry $a_{1,1}$ is equal $0$.\\
From matrix, note that $\norm{T \mathbf{e}_{2}}^2 = \abs{a_{2,2}}^2$ (as $a_{1,2}=0$ is shown above), $\norm{T^* \mathbf{e}_2}^2 = \abs{a_{2,2}}^2 + \abs{a_{2,3}}^2 + \cdots + \abs{a_{2,n}}^2$. As $T$ is normal, $\norm{T \mathbf{e}_2} = \norm{T^* \mathbf{e}_2}$, hence above equations imply all entries in second row of matrix except possibly the diagonal entry $a_{2,2}$ is equal $0$.\\
Doing this for all non-diagonal entries show that these equal $0$ as desired.
\end{proof}
\end{theorem}

Note that for $\alpha, \beta \in \R$, and $\alpha^2 < 4 \beta$, let $x \in \R$, then $x^2 + \alpha x + \beta = \left(x + \frac{\alpha}{2} \right)^2 + \left(\beta - \frac{\alpha^2}{4} \right) > 0$, hence $x^2 + \alpha x + \beta$ is an invertible real number. Replacing $x$ with a self-adjoint operator leads to the lemma below.

\begin{lemma} \label{lem:sefladjinvertible} Suppose $T \in \mathcal{L}(\mathcal{V})$ is self-adjoint. If $\alpha, \beta \in \R$ are such that $\alpha^2 < 4 \beta$, then
\begin{align}
T^2 + \alpha T + \beta I \nonumber
\end{align}
is invertible.
\begin{proof} Suppose $\alpha, \beta \in \R$ are such that $\alpha^2 < 4 \beta$. Let $\mathbf{v}$ be nonzero vector in $\mathcal{V}$, then
\begin{align}
\innerprod{(T^2 + \alpha T + \beta I)\mathbf{v}, \mathbf{v}} &= \innerprod{T^2 \mathbf{v}, \mathbf{v}} + \alpha \innerprod{T\mathbf{v}, \mathbf{v}} + \beta \innerprod{\mathbf{v}, \mathbf{v}} \nonumber \\
&= \innerprod{T \mathbf{v}, T \mathbf{v}} + \alpha \innerprod{T \mathbf{v}, \mathbf{v}} + \beta \norm{\mathbf{v}}^2 \nonumber \\
&\geq \norm{T \mathbf{v}}^2 - \abs{\alpha}\norm{T \mathbf{v}} \norm{\mathbf{v}} + \beta \norm{\mathbf{v}}^2 \ \ \ \ \text{by Cauchy-Schwarz Inequality} \nonumber \\
&= \left( \norm{T \mathbf{v}} - \frac{\abs{\alpha} \norm{\mathbf{v}}}{2} \right)^2 + \left(\beta - \frac{\alpha^2}{4} \right) \norm{\mathbf{v}}^2 > 0 \nonumber
\end{align}
Note last inequality implies $(T^2 + \alpha T + \beta I)\mathbf{v} \neq 0$, hence $T^2 + \alpha T + \beta I$ is injective, which implies it is invertible.
\end{proof}
\end{lemma}

\begin{lemma} Suppose $T \in \mathcal{L}(\mathcal{V})$ is self-adjoint, then $T$ has an eigenvalue.
\begin{proof} Assume $\mathcal{V}$ is a real inner-product space. Let $\text{dim}(\mathcal{V}) = n$, $\mathbf{v} \in \mathcal{V}$ with $\mathbf{v} \neq \mathbf{0}$, then
\begin{align}
(\mathbf{v}, T\mathbf{v}, T^2 \mathbf{v}, \ldots, T^n \mathbf{v}) \nonumber
\end{align}
is not linearly independent as $\text{dim}(\mathcal{V}) = n$ and there are $n+1$ vectors. Hence $\exists \ a_0, \ldots, a_n \in \R$ not all $0$ s.t.
\begin{align}
0 = a_0 \mathbf{v} + a_1 T \mathbf{v} + \cdots + a_n T^n \mathbf{v} \nonumber
\end{align}
Let $a$ be coefficients of a polynomial, written in factored form as
\begin{align}
a_0 + a_1 x + \cdots + a_n x^n = c(x^2 + \alpha_1 x + \beta_1) \cdots (x^2 + \alpha_M x + \beta_M)(x - \lambda_1) \cdots (x - \lambda_m) \nonumber
\end{align}
where $c \neq 0 \in \R, \alpha_j, \beta_j, \lambda_j \in \R$, each $\alpha_j^2 < 4 \beta_j$, $m + M \geq 1$, and equation holds for all $x \in \R$. Then
\begin{align}
0 &= a_0 \mathbf{v} + a_1 T \mathbf{v} + \cdots + a_n T^n \mathbf{v} \nonumber \\
&= (a_0 I + a_1 T + \cdots + a_n T^n)\mathbf{v} \nonumber \\
&= c(T^2 + \alpha_1 T + \beta_1 I) \cdots (T^2 + \alpha_M T + \beta_M I)(T - \lambda_1 I)\cdots (T - \lambda_m I)\mathbf{v} \nonumber
\end{align}
Note each $T^2 + \alpha_j T + \beta_j I$ is invertible as $T$ is self-adjoint and each $\alpha_j^2 < 4 \beta_j$ as by Lemma \ref{lem:sefladjinvertible}.\\
Also $c \neq 0$, hence equation implies
\begin{align}
0 = (T - \lambda_1 T) \cdots (T - \lambda_m I)\mathbf{v} \nonumber
\end{align}
Hence $T - \lambda_j I$ is not injective for at least one $j$, thus $T$ has an eigenvalue.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:realspectralthm} \hlt{Real Spectral Theorem}\\
Suppose $\mathcal{V}$ is a real inner-product space and $T \in \mathcal{L}(\mathcal{V})$, then $\mathcal{V}$ has an orthonormal basis consisting of eigenvectors of $T$ if and only if $T$ is self-adjoint.
\begin{proof} $(\Rightarrow)$ Suppose $\mathcal{V}$ has an orthonormal basis consisting of eigenvectors of $T$. With respect to this basis, $T$ has a diagonal matrix which equals its conjugate transpose, $T = T^*$, hence $T$ is self-adjoint.\\
$(\Leftarrow)$ Suppose $T$ is self-adjoint. Note $\mathcal{V}$ has an orthonormal basis clearly if $\text{dim}(\mathcal{V}) = 1$.\\
Assume $\text{dim}(\mathcal{V}) > 1$, and $\lambda$ be any eigenvalue of $T$, let $\mathbf{u} \in \mathcal{V}$ denote corresponding eigenvector with $\norm{\mathbf{u}} = 1$. Let $\mathcal{U}$ be one-dimensional subspace of $\mathcal{V}$ consisting of all scalar multiples of $\mathbf{u}$. Note $\mathbf{v} \in \mathcal{V}$ is in $\mathcal{U}^{\perp}$ if and only if $\innerprod{\mathbf{u}, \mathbf{v}} = 0$. Suppose $\mathbf{v} \in \mathcal{U}^{\perp}$, then $T$ is self-adjoint, and
\begin{align}
\innerprod{\mathbf{u}, T \mathbf{v}} = \innerprod{T \mathbf{u}, \mathbf{v}} = \innerprod{\lambda \mathbf{u}, \mathbf{v}} = \lambda \innerprod{\mathbf{u}, \mathbf{v}} = 0 \nonumber
\end{align} 
hence $T \mathbf{v} \in \mathcal{U}^{\perp}$. Thus $T \mathbf{v} \in \mathcal{U}^{\perp}$ whenever $\mathbf{v} \in \mathcal{U}^{\perp}$, i.e., $\mathcal{U}^{\perp}$ is invariant under $T$.  Hence define an operator $S \in \mathcal{L}(\mathcal{U}^{\perp})$ by $S = T|_{\mathcal{U}^{\perp}}$. If $\mathbf{v}, \mathbf{w} \in \mathcal{U}^{\perp}$, then
\begin{align}
\innerprod{S \mathbf{v}, \mathbf{w}} = \innerprod{T \mathbf{v}, \mathbf{w}} = \innerprod{\mathbf{v}, T \mathbf{w}} = \innerprod{\mathbf{v}, S \mathbf{w}} \nonumber
\end{align}
which shows $S$ is self-adjoint. By induction hypothesis, there is an orthonormal basis of $\mathcal{U}^{\perp}$ consisting of eigenvectors of $S$. As $S \mathbf{v} = T \mathbf{v}$ for every $\mathbf{v} \in \mathcal{U}^{\perp}$, clearly every eigenvector of $S$ is an eigenvector of $T$. Hence adjoining $\mathbf{u}$ to an orthonormal basis of $\mathcal{U}^{\perp}$ consisting of eigenvectors of $S$ gives an orthonormal basis of $\mathcal{V}$ consisting of eigenvectors of $T$, as desired.
\end{proof}
\end{theorem}

For $T \in \mathcal{L}(\mathcal{V})$ self-adjoint, next corollary provides nicest decomposition of $\mathcal{V}$ into subspaces invariant in $T$.

\begin{corollary} \label{cor:selfadjeigenortho} Suppose $T \in \mathcal{L}(\mathcal{V})$ is self-adjoint, let $\lambda_1, \ldots, \lambda_m$ be distinct eigenvalues of $T$, then
\begin{align}
\mathcal{V} = \text{null}(T - \lambda_1 I) \oplus \cdots \oplus \text{null}(T - \lambda_m I) \nonumber
\end{align}
Each vector in each $\text{null}(T - \lambda_j I)$ is orthogonal to all vectors in other subspaces of this decomposition.
\begin{proof} The Spectral Theorems \ref{thm:complexspectralthm} and \ref{thm:realspectralthm} implies $\mathcal{V}$ has a basis consisting of eigenvectors of $T$. The desired decomposition of $\mathcal{V}$ follows from Proposition \ref{prop:invariantsubspaceequivalence}.\\
The orthogonality statement follows from Corollary \ref{cor:normaleigenvectororthogonal}.
\end{proof}
\end{corollary}

\begin{lemma} \label{lem:realinnerprodspaceequiv} Suppose $\mathcal{V}$ is a real inner-product space, $\text{dim}(\mathcal{V})$ and $T \in \mathcal{L}(\mathcal{V})$, these are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $T$ is normal but not self-adjoint
\item the matrix of $T$ with respect to every orthonormal basis of $\mathcal{V}$ has the form
\begin{align}
\begin{bmatrix}
a & -b \\
b & a
\end{bmatrix} \nonumber
\end{align}
with $b \neq 0$.
\item the matrix of $T$ with respect to some orthonormal basis of $\mathcal{V}$ has the form
\begin{align}
\begin{bmatrix}
a & -b \\
b & a
\end{bmatrix} \nonumber
\end{align}
with $b > 0$.
\end{enumerate}
\begin{proof} $(i) \Rightarrow (ii)$. Suppose $(i)$ holds. Let $(\mathbf{e}_1, \mathbf{e}_2)$ be an orthonormal basis of $\mathcal{V}$. Let
\begin{align}
\mathcal{M}(T, (\mathbf{e}_1, \mathbf{e}_2)) =
\begin{bmatrix}
a & c \\
b & d
\end{bmatrix} \nonumber
\end{align}
Then $\norm{T \mathbf{e}_1}^2 = a^2 + b^2$ and $\norm{T^* \mathbf{e}_1}^2 = a^2 + c^2$. As $T$ is normal, $\norm{T \mathbf{e}_1} = \norm{T^* \mathbf{e}_1}$, then these equations imply $b^2 = c^2$, hence $c = b$ or $c = -b$. Note that $c \neq b$ (otherwise $T$ will be self-adjoint), hence $c = -b$, thus
\begin{align}
\mathcal{M}(T, (\mathbf{e}_1, \mathbf{e}_2)) =
\begin{bmatrix}
a & -b \\
b & d
\end{bmatrix} \nonumber
\end{align}
Hence $T^*$ is the transpose of the matrix above. As $T$ is normal, $TT^* = T^*T$, hence these two matrices must be equal, and $bd = ab$. Also, $b \neq 0$ otherwise $T$ would be self-adjoint, hence $d=a$.\\
$(ii) \Rightarrow (iii)$. Suppose $(ii)$ holds. Let $(\mathbf{e}_1, \mathbf{e}_2)$ be any orthonormal basis of $\mathcal{V}$. The matrix of $T$ with respect to this basis has form given in $(ii)$, with $b \neq 0$. If $b>0$, then $(iii)$ holds, hence $(ii)$ implies $(iii)$. If $b<0$, then
\begin{align}
\mathcal{M}(T, (\mathbf{e}_1, - \mathbf{e}_2)) =
\begin{bmatrix}
a & b \\
-b & a
\end{bmatrix} \nonumber
\end{align}
where $-b > 0$, hence $(ii)$ implies $(iii)$.\\
$(iii) \Rightarrow (i)$. Suppose $(iii)$ holds, so matrix of $T$ with respect to some orthonormal basis has form given in $(iii)$ with $b > 0$. Clearly matrix of $T$ is not equal to its transpose as $b \neq 0$, hence $T$ is not self-adjoint. Note that $TT^* = T^* T$, and hence $T$ is normal. Thus $(iii)$ implies $(i)$.
\end{proof}
\end{lemma}

The next proposition play key role in characterisation of normal operators on real inner-product space.

\begin{proposition} Suppose $T \in \mathcal{L}(\mathcal{V})$ is normal and $\mathcal{U}$ is a subspace of $\mathcal{V}$ that is invariant under $T$. Then
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{U}^{\perp}$ is invariant under $T$
\item $\mathcal{U}$ is invariant under $T^*$
\item $(T|_{\mathcal{U}})^* = (T^*)_{\mathcal{U}}$
\item $T|_{\mathcal{U}}$ is a normal operator on $\mathcal{U}$
\item $T_{\mathcal{U}^{\perp}}$ is a normal operator on $\mathcal{U}^{\perp}$
\end{enumerate}
\begin{proof} $(i)$. Let $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ be an orthonormal basis of $\mathcal{U}$. Extend to an orthonormal basis $(\mathbf{e}_1, \ldots, \mathbf{e}_m, \mathbf{f}_1, \ldots, \mathbf{f}_n)$ of $\mathcal{V}$ (by Corollary \ref{cor:linindepextendorthonorm}). As $\mathcal{U}$ is invariant under $T$, each $T \mathbf{e}_j$ is a linear combination of $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$. Hence matrix of $T$ with respect to basis $(\mathbf{e}_1, \ldots, \mathbf{e}_m, \mathbf{f}_1, \ldots, \mathbf{f}_n)$ is of the form
\begin{align}
\mathcal{M}(T, (\mathbf{e}_1, \ldots, \mathbf{e}_m, \mathbf{f}_1, \ldots, \mathbf{f}_n)) =
\begin{bmatrix}
A & B \\
0 & C
\end{bmatrix} \nonumber
\end{align}
where $A$ is a $m \times m$ matrix, $0$ is a $m \times n$ matrix of all $0$s, $B$ is a $m \times n$ matrix, $C$ is a $n \times n$ matrix.\\
For each $j \in \{1, \ldots, m\}$, $\norm{T \mathbf{e}_j}^2$ equals sum of squares of absolute values of entries in $j$th column of $A$, hence $\sum\limits_{j=1}^m \norm{T \mathbf{e}_j}^2$ is sum of squares of absolute values of entries of $A$.\\
For each $j \in \{1, \ldots, m\}$, $\norm{T^* \mathbf{e}_j}^2$ equals sum of squares of absolute values of entires in $j$th row of $A$ and $B$, hence $\sum\limits_{j=1}^m \norm{T^* \mathbf{e}_j}^2$ is sum of squares of the absolute values of entries of $A$ and $B$.\\
As $T$ is normal, $\norm{T \mathbf{e}_j} = \norm{T^* \mathbf{e}_j}$ for each $j$, hence $\sum\limits_{j=1}^m \norm{T \mathbf{e}_j}^2 = \sum\limits_{j=1}^m \norm{T^* \mathbf{e}_j}^2$ This implies the sum of squares of absolute values of entries of $B$ must equal $0$, hence $B$ is matrix consisting of all $0$s, hence
\begin{align}
\mathcal{M}(T, (\mathbf{e}_1, \ldots, \mathbf{e}_m, \mathbf{f}_1, \ldots, \mathbf{f}_n)) =
\begin{bmatrix}
A & 0 \\
0 & C
\end{bmatrix} \nonumber
\end{align}
This representation shows that $T \mathbf{f}_k$ is the span of $(\mathbf{f}_1, \ldots, \mathbf{f}_n)$ for each $k$. As $(\mathbf{f}_1, \ldots, \mathbf{f}_n)$ is a basis of $\mathcal{U}^{\perp}$, this implies $T \mathbf{v} \in \mathcal{U}^{\perp}$ whenever $\mathbf{v} \in \mathcal{U}^{\perp}$. In other words, $\mathcal{U}^{\perp}$. is invariant under $T$.\\
$(ii)$. Note that $\mathcal{T^*}$ has block of $0$ is lower left, hence each $T^* \mathbf{e}_j$ can be written as linear combination of $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$, thus $\mathcal{U}$ is invariant under $T^*$.\\
$(iii)$. Let $S = T|_{\mathcal{U}}$. Fix $\mathbf{v} \in \mathcal{U}$, then
\begin{align}
\innerprod{S \mathbf{u}, \mathbf{v}} = \innerprod{T \mathbf{u}, \mathbf{v}} = \innerprod{\mathbf{u}, T^* \mathbf{v}} \nonumber
\end{align}
for all $\mathbf{u} \in \mathcal{U}$. As $T^* \mathbf{v} \in \mathcal{U}$ by $(ii)$, equation above shows $S^* \mathbf{v} = T^* \mathbf{v}$. Hence $(T|_{\mathcal{U}})^* = (T^*)_{\mathcal{U}}$.\\
$(iv)$. Note $T$ commutes with $T^*$ as $T$ is normal, and $(T|_{\mathcal{U}})^* = (T^*)_{\mathcal{U}}$ by $(iii)$, hence $T|_{\mathcal{U}}$ commutes with its adjoint and hence is normal.\\
$(v)$. Note that in $(iv)$, it is shown that restriction of $T$ to any invariant subspace is normal. However, $\mathcal{U}^{\perp}$ is invariant under $T$ by $(i)$, hence $T|_{\mathcal{U}}$ is normal.
\end{proof}
\end{proposition}

\begin{definition} A \hlt{block diagonal matrix} is a square matrix of form
\begin{align}
\begin{bmatrix}
A_1 & & 0 \\
& \ddots & & \\
0 & & A_m
\end{bmatrix} \nonumber
\end{align}
where $A_1, \ldots, A_m$ are square matrices along the diagonal, and all other entries of matrix equal $0$.
\end{definition}

\begin{theorem} Suppose $\mathcal{V}$ is a real inner-product space and $T \in \mathcal{L}(\mathcal{V})$. Then $T$ is normal if and only if there is an orthonormal basis of $\mathcal{V}$ with respect to which $T$ has a block diagonal matrix where each block is a $1 \times 1$ or $2 \times 2$ matrix of form
\begin{align}
\begin{bmatrix}
a & -b \\
b & a
\end{bmatrix} \nonumber
\end{align}
where $b > 0$.
\begin{proof} $(\Rightarrow)$ Suppose $\exists$ orthonormal basis of $\mathcal{V}$ such that $T$ is a block-diagonal matrix where each block is $1 \times 1$ or $2 \times 2$ matrix of form shown earlier. Note $T T^* = T^* T$, hence $T$ is normal.\\
$(\Leftarrow)$ Suppose $T$ is normal. Note result is trivial if $\text{dim}(\mathcal{V}) = 1$. If $\text{dim}(\mathcal{V}) = 2$, if $T$ is self-adjoint, result holds by Real Spectral Theorem \ref{thm:realspectralthm}; if not, then result holds by Lemma \ref{lem:realinnerprodspaceequiv}.\\
Assume $\text{dim}(\mathcal{V}) > 2$, and desired result holds on vector spaces of smaller dimension. Let $\mathcal{U} \subset \mathcal{V}$, $\text{dim}(\mathcal{U})=1$ that is invariant under $T$ if the subspace exists (if $T$ has nonzero eigenvector, let $\mathcal{U}$ be span of this eigenvector). If no such subspace exists, let $\mathcal{U} \subset \mathcal{V}$, $\text{dim}(\mathcal{U})=2$ that is invariant under $T$.\\
If $\text{dim}(\mathcal{U})=1$, choose vector in $\mathcal{U}$ with norm $1$, and this vector will b an orthonormal basis of $\mathcal{U}$, and $T|_{\mathcal{U}}$ is $1 \times 1$ matrix. If $\text{dim}(\mathcal{U})=2$, then $T_{\mathcal{U}}$ is normal but not self adjoint (else $T|_{\mathcal{U}}$ have nonzero eigenvector), hence choose an orthonormal basis of $\mathcal{U}$ with respect to which the matrix of $T|_{\mathcal{U}}$ has form in statement.\\
Now $\mathcal{U}^{\perp}$ is invariant under $T$ and $T|_{\mathcal{U}^{\perp}}$ is a normal operator on $\mathcal{U}^{\perp}$. By induction, there is an orthonormal basis of $\mathcal{U}^{\perp}$ with respect to which matrix of $T|_{\mathcal{U}^\perp}$ has the desired form. Adjoining this basis to basis of $\mathcal{U}$ gives an orthonormal basis of $\mathcal{V}$ with respect to which matrix of $T$ has the desireed form.
\end{proof}
\end{theorem}

\begin{definition} An operator $T \in \mathcal{L}(\mathcal{V})$ is \hlt{positive semi-definite} if $T$ is self-adjoint and $\innerprod{T \mathbf{v}, \mathbf{v}} \geq 0 \ \forall \mathbf{v} \in \mathcal{V}$.
\end{definition}

\begin{definition} An operator $S$ is a \hlt{square root} of an operator $T$ if $S^2 = T$.
\end{definition}

The next theorem characterises positive operators that correspond to characterisations of nonnegative numbers among $\C$.

\begin{theorem} \label{thm:positiveopcharacterisation} Let $T \in \mathcal{L}(\mathcal{V})$, then the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $T$ is positive
\item $T$ is self-adjoint and all eigenvalues of $T$ are nonnegative
\item $T$ has a positive square root
\item $T$ has a self-adjoint square root
\item there exists an operator $S \in \mathcal{L}(\mathcal{V})$ such that $T = S^* S$
\end{enumerate}
\begin{proof} $(i) \Rightarrow (ii)$. Suppose $(i)$ holds, so $T$ is positive. By definition of positive operator, $T$ is self-adjoint. Suppose $\lambda$ is an eigenvalue of $T$. Let $\mathbf{v}$ be a nonzero eigenvector of $T$ corresponding to $\lambda$, then
\begin{align}
0 \leq \innerprod{T \mathbf{v}, \mathbf{v}} = \innerprod{\lambda \mathbf{v}, \mathbf{v}} = \lambda \innerprod{\mathbf{v}, \mathbf{v}} \nonumber
\end{align}
hence $\lambda$ is a nonnegative number.\\
$(ii) \Rightarrow (iii)$. Suppose $(ii)$ holds, so $T$ is self-adjoint and all the eigenvalues of $T$ are nonnegative. By Spectral Theorem \ref{thm:realspectralthm} and \ref{thm:complexspectralthm}, there is an orthonormal basis $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ of $\mathcal{V}$ consisting of eigenvectors of $T$. Let $\lambda_1, \ldots, \lambda_n$ be eigenvalues of $T$ corresponding to $\mathbf{e}_1, \ldots, \mathbf{e}_n$ so that each $\lambda_j$ is a nonnegative number. Let $S \in \mathcal{L}(\mathcal{V})$ such that $S \mathbf{e}_j = \sqrt{\lambda_j}\mathbf{e}_j$ for $j = 1, \ldots, n$, then $S$ is a positive operator. Also, $S^2 \mathbf{e}_j = \lambda_j \mathbf{e}_j = T \mathbf{e}_j$ for each $j$, which implies $S^2 = T$. Hence $S$ is a positive square root of $T$, and $(iii)$ holds.\\
$(iii) \Rightarrow (iv)$. Note that by definition, every positive operator is self-adjoint hence this holds.\\
$(iv) \Rightarrow (v)$. Let $S \in \mathcal{L}(\mathcal{V})$, $T = S^* S$, then $T^* = (S^* S)^* = S^* (S^*)^* = S^* S = T$, hence $T$ is self-adjoint. Then
\begin{align}
\innerprod{T \mathbf{v}, \mathbf{v}} = \innerprod{S^* S \mathbf{v}, \mathbf{v}} = \innerprod{S \mathbf{v}, S \mathbf{v}} \geq 0 \nonumber
\end{align}
for every $\mathbf{v} \in \mathcal{V}$, hence $T$ is positive.
\end{proof}
\end{theorem}

A positive operator can have infinitely many square roots, but only one can be positive.

\begin{proposition} Every positive operator on $\mathcal{V}$ has a unique positive square root.
\begin{proof} Suppose $T \in \mathcal{L}(\mathcal{V})$ is positive. Let $\lambda_1, \ldots, \lambda_m$ denote distinct eigenvalues of $T$. As $T$ is positive, all these are nonnegative by Theorem \ref{thm:positiveopcharacterisation}. As $T$ is self-adjoint, by Corollary \ref{cor:selfadjeigenortho},
\begin{align}
\mathcal{V} = \text{null}(T - \lambda_1 I) \oplus \cdots \oplus \text{null}(T - \lambda_m I) \nonumber
\end{align}
Suppose $S \in \mathcal{L}(\mathcal{V})$ is a positive square root of $T$, and $\alpha$ is an eigenvalue of $S$. If $\mathbf{v} \in \text{null}(S - \alpha I)$, then $S \mathbf{v} = \alpha \mathbf{v}$, implying $T \mathbf{v} = S^2 \mathbf{v} = \alpha^2 \mathbf{v}$ so that $\mathbf{v} \in \text{null}(T - \alpha^2 I)$. Thus $\alpha^2$ is an eigenvalue of $T$, which means $\alpha^2$ must equal some $\lambda_j$. Hence $\alpha = \sqrt{\lambda_j}$ for some $j$. Also,
\begin{align}
\text{null}(S - \sqrt{\lambda_j}I) \subset \text{null}(T - \lambda_j I) \nonumber
\end{align}
From above, the only possible eigenvalues for $S$ are $\sqrt{\lambda_1}, \ldots, \sqrt{\lambda_m}$. As $S$ is self-adjoint, this implies
\begin{align}
\mathcal{V} = \text{null}(S - \sqrt{\lambda_1}I) \oplus \cdots \oplus \text{null}(S - \sqrt{\lambda_m}I) \nonumber
\end{align}
Hence these implies $\text{null}(S - \sqrt{\lambda_j}I) = \text{null}(T - \lambda_j I)$ for each $j$. On $\text{null}(T - \lambda_j I)$, the operator $S$ is just multiplication by $\sqrt{\lambda_j}$. Hence $S$ is a uniquely determined positive square root of $T$.
\end{proof}
\end{proposition}

\begin{definition} An operator $S \in \mathcal{L}(\mathcal{V})$ is an \hlt{isometry} if $\norm{S \mathbf{v}} = \norm{\mathbf{v}}$ for all $\mathbf{v} \in \mathcal{V}$.\\
An isometry on a real inner-product space is an \hlt{orthogonal operator}.\\
An isometry on a complex inner-product space is an \hlt{unitary operator}.\\
If $S \in \mathcal{L}(\mathcal{V})$ is an isometry, then $S$ is injective, hence every isometry is invertible. 
\end{definition}

Next theorem provides several conditions that are equivalent to being an isometry.

\begin{theorem} \label{thm:characterisationofisometry} Suppose $S \in \mathcal{L}(\mathcal{V})$, then the following are equivalent:
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $S$ is an isometry
\item $\innerprod{S \mathbf{u}, S \mathbf{v}} = \innerprod{\mathbf{u}, \mathbf{v}}$ for all $\mathbf{u}, \mathbf{v} \in \mathcal{V}$ (inner product is preserved)
\item $S^* S = I$
\item $(S \mathbf{e}_1, \ldots, S \mathbf{e}_n)$ is orthonormal whenever $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ is an orthonormal list of vectors in $\mathcal{V}$
\item $\exists$ an orthonormal basis $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ of $\mathcal{V}$ such that $(S \mathbf{e}_1, \ldots, S \mathbf{e}_n)$ is orthonormal
\item $S^*$ is an isometry
\item $\innerprod{S^* \mathbf{u}, S^* \mathbf{v}} = \innerprod{\mathbf{u}, \mathbf{v}}$ for all $\mathbf{u}, \mathbf{v} \in \mathcal{V}$
\item $S S^* = I$
\item $(S^* \mathbf{e}_1, \ldots, S^* \mathbf{e}_n)$ is orthonormal whenever $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ is an orthonormal list of vectors in $\mathcal{V}$
\item $\exists$ an orthonormal basis $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ of $\mathcal{V}$ such that $(S^* \mathbf{e}_1, \ldots, S^* \mathbf{e}_n)$ is orthonormal
\end{enumerate}
\begin{proof} $(i)$. Suppose $(i)$ holds, and $\mathcal{V}$ is a real-inner product space, then for every $\mathbf{u}, \mathbf{v} \in \mathcal{V}$, we have
\begin{align}
\innerprod{S \mathbf{u}, S \mathbf{v}} &= (\norm{S \mathbf{u} + S \mathbf{v}}^2 - \norm{S \mathbf{u} - S \mathbf{v}}^2)/4 \nonumber \\
&= (\norm{S(\mathbf{u} + \mathbf{v})}^2 - \norm{S(\mathbf{u} - \mathbf{v})}^2)/4 \nonumber \\
&= (\norm{\mathbf{u} + \mathbf{v}}^2 - \norm{\mathbf{u} - \mathbf{v}}^2)/4 \nonumber \\
&= \innerprod{\mathbf{u}, \mathbf{v}} \nonumber
\end{align}
$(ii) \Rightarrow (iii)$. Suppose $(ii)$ holds, then
\begin{align}
\innerprod{(S^* S - I)\mathbf{u}, \mathbf{v}} = \innerprod{S \mathbf{u}, S \mathbf{v}} - \innerprod{\mathbf{u}, \mathbf{v}} = 0\nonumber
\end{align}
for every $\mathbf{u}, \mathbf{v} \in \mathcal{V}$. Let $\mathbf{v} = (S^* S - I)\mathbf{u}$, then $S^* S - I = 0$, hence $S^* S = I$.\\
$(iii) \Rightarrow (iv)$. Suppose $(iii)$ holds, and $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ is an orthonormal list of vectors in $\mathcal{V}$, then
\begin{align}
\innerprod{S \mathbf{e}_j, S \mathbf{e}_k} = \innerprod{S^* S \mathbf{e}_j, \mathbf{e}_k} = \innerprod{\mathbf{e}_j, \mathbf{e}_k} \nonumber
\end{align}
hence $(S \mathbf{e}_1, \ldots, S \mathbf{e}_n)$ is orthonormal.\\
$(iv) \Rightarrow (v)$ is quite obvious by implication.\\
$(v) \Rightarrow (i)$. Suppose $(v)$ holds, let $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ be an orthonormal basis of $\mathcal{V}$ such that $(S \mathbf{e}_1, \ldots, S \mathbf{e}_n)$ is orthonormal. If $\mathbf{v} \in \mathcal{V}$, then
\begin{align}
\norm{S \mathbf{v}}^2 &= \norm{S(\innerprod{\mathbf{v}, \mathbf{e}_1}\mathbf{e}_1 + \cdots + \innerprod{\mathbf{v}, \mathbf{e}_n}\mathbf{e}_n)}^2 \nonumber \\
&= \norm{\innerprod{\mathbf{v}, \mathbf{e}_1}S\mathbf{e}_1 + \cdots + \innerprod{\mathbf{v, \mathbf{e}_n}}S \mathbf{e}_n}^2 \nonumber \\
&= \abs{\innerprod{\mathbf{v}, \mathbf{e}_1}}^2 + \cdots + \abs{\innerprod{\mathbf{v}, \mathbf{e}_n}}^2 = \norm{\mathbf{v}}^2 \nonumber
\end{align}
Taking square roots, $S$ is shown to be an isometry.\\
Replacing $S$ with $S^*$, $(vi)$ to $(x)$ are all equivalent to each other.\\
To connect the two groups of conditions, show $(iii) \rightarrow (viii)$. In general, $S$ does not commute with $S^*$. However, $S^* S = I$ if and only if $SS^* = I$. Hence this completes the proof.
\end{proof}
\end{theorem}

\begin{theorem} Suppose $\mathcal{V}$ is a complex inner-product space and $S \in \mathcal{L}(\mathcal{V})$, then $S$ is an isometry if and only if there is an orthonormal basis of $\mathcal{V}$ consisting of eigenvectors of $S$ all of whose corresponding eigenvalues have absolute value $1$.
\begin{proof} $(\Rightarrow)$. Note that $\lambda I$ is an isometry whenever $\lambda \in \mathbb{F}$ satisfies $\abs{\lambda}=1$.\\
Let $\lambda_1, \ldots, \lambda_n$ be scalars with absolute value $1$ and $S \in \mathcal{L}(\mathcal{V})$ satisfies $S(\mathbf{e}_j) = \lambda_j \mathbf{e}_j$ for some orthonormal basis $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ of $\mathcal{V}$. Suppose $\mathbf{v} \in \mathcal{V}$, then
\begin{align}
\mathbf{v} &= \innerprod{\mathbf{v}, \mathbf{e}_1}\mathbf{e}_1 + \cdots + \innerprod{\mathbf{v}, \mathbf{e}_n} \mathbf{e}_n \nonumber \\
\norm{\mathbf{v}}^2 &= \abs{\innerprod{\mathbf{v}, \mathbf{e}_1}}^2 + \cdots + \abs{\innerprod{\mathbf{v}, \mathbf{e}_n}}^2 \nonumber
\end{align}
Applying $S$ to both sides gives
\begin{align}
S \mathbf{v} &= \innerprod{\mathbf{v}, \mathbf{e}_1} S \mathbf{e}_1 + \cdots + \innerprod{\mathbf{v}, \mathbf{e}_n} S \mathbf{e}_n \nonumber \\
&= \lambda_1 \innerprod{\mathbf{v}, \mathbf{e}_1} \mathbf{e}_1 + \cdots + \lambda_n \innerprod{\mathbf{v}, \mathbf{e}_n} \mathbf{e}_n \nonumber \\
\norm{S \mathbf{v}}^2 &= \abs{\innerprod{\mathbf{v}, \mathbf{e}_1}}^2 + \cdots + \abs{\innerprod{\mathbf{v}, \mathbf{e}_n}}^2 \nonumber
\end{align}
Hence this shows $\norm{\mathbf{v}} = \norm{S \mathbf{v}}$, thus $S$ is an isometry.\\
$(\Leftarrow)$. Suppose $S$ is an isometry. By Complex Spectral Theorem \ref{thm:complexspectralthm}, there is an orthonormal basis $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ of $\mathcal{V}$ consisting of eigenvectors of $S$. For $j \in \{1, \ldots, n \}$, let $\lambda_j$ be eigenvalue corresponding to $\mathbf{e}_j$, then
\begin{align}
\abs{\lambda_j} = \norm{\lambda_j \mathbf{e}_j} = \norm{S \mathbf{e}_j} = \norm{\mathbf{e}_j} = 1 \nonumber
\end{align}
Hence each eigenvalue of $S$ has absolute value $1$.
\end{proof}
\end{theorem}

The next theorem implies that an isometry on an odd-dimensional real inner-product space must have $1$ or $-1$ as an eigenvalue.

\begin{theorem} Suppose $\mathcal{V}$ is a real inner-product space with $S \in \mathcal{L}(\mathcal{V})$. Then $S$ is an isometry if and only if there is an orthonormal basis of $\mathcal{V}$ with respect to which $S$ has a block diagonal matrix where each block on the diagonal is a $1 \times 1$ matrix containing $1$ or $-1$, or a $2 \times 2$ matrix of form
\begin{align}
\begin{bmatrix}
\sin \theta & - \sin \theta \\
\sin \theta & \cos \theta
\end{bmatrix} \nonumber
\end{align}
with $\theta \in (0, \pi)$.
\begin{proof} $(\Rightarrow)$. Suppose $S$ is an isometry. As $S$ is normal, $\exists$ an orthonormal basis of $V$ such that with respect to this basis $S$ has a block diagonal matrix, where each block is a $1 \times 1$ matrix or $2 \times 2$ matrix of form
\begin{align}
\begin{bmatrix}
a & -b \\
b & a
\end{bmatrix} \nonumber
\end{align}
with $b > 0$.\\
If $\lambda$ is an entry in a $1 \times 1$ along the diagonal of matrix of $S$ with respect to basis above, then $\exists$ a basis vector $\mathbf{e}_j$ such that $S \mathbf{e}_j = \lambda \mathbf{e}_j$. As $S$ is an isometry, this implies $\abs{\lambda} = 1$, hence $\lambda = 1$ or $-1$.\\
Consider $2 \times 2$ matrix from earlier along the diagonal of the matrix of $S$. $\exists$ basis vectors $\mathbf{e}_j, \mathbf{e}_{j+1}$ such that
\begin{align}
S \mathbf{e}_j = a \mathbf{e}_j + b \mathbf{e}_{j+1} \nonumber
\end{align}
Hence $1 = \norm{\mathbf{e}_j}^2 = \norm{S \mathbf{e}_j}^2 = a^2 + b^2$. Along with condition $b>0$, this implies $\exists$ number $\theta \in (0, \pi)$ such that $a = \cos \theta$ and $b = \sin \theta$, hence the matrix has the required form.\\
$(\Leftarrow)$. Suppose $\exists$ an orthonormal basis of $\mathcal{V}$ with respect to which matrix $S$ has form in theorem. Then there is a direct sum decomposition $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$ where each $\mathcal{U}_j$ is a subspace of $\mathcal{V}$ of dimension $1$ or $2$. Furthermore any two vectors belonging to distinct $\mathcal{U}_j$ are orthogonal, and each $S|_{\mathcal{U}_j}$ is an isometry mapping $\mathcal{U}_j$ to $\mathcal{U}_j$. If $\mathbf{v} \in \mathcal{V}$, then $\mathbf{v} = \mathbf{u}_1 + \cdots + \mathbf{u}_m$ where each $\mathbf{u}_j \in \mathcal{U}_j$. Applying $S$ to above equation and taking norms gives
\begin{align}
\norm{S \mathbf{v}}^2 &= \norm{S \mathbf{u}_1 + \cdots + S \mathbf{u}_m}^2 \nonumber \\
&= \norm{S \mathbf{u}_1}^2 + \cdots + \norm{S \mathbf{u}_m}^2 \nonumber \\
&= \norm{\mathbf{u}_1}^2 + \cdots + \norm{\mathbf{u}_m}^2 = \norm{\mathbf{v}}^2 \nonumber
\end{align}
Hence $S$ is an isometry as desired.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:polardecomposition} \hlt{Polar Decomposition}\\
If $T \in \mathcal{L}(\mathcal{V})$, then there exists an isometry $S \in \mathcal{L}(\mathcal{V})$ such that $T = S \sqrt{T^* T}$.
\begin{proof} Suppose $T \in \mathcal{L}(\mathcal{V})$. If $\mathbf{v} \in \mathcal{V}$, then
\begin{align}
\norm{T \mathbf{v}}^2 = \innerprod{T\mathbf{v}, T\mathbf{v}} = \innerprod{T^* T \mathbf{v}, \mathbf{v}} = \innerprod{\sqrt{T^* T} \sqrt{T^* T} \mathbf{v}, \mathbf{v}} = \innerprod{\sqrt{T^* T} \mathbf{v}, \sqrt{T^* T} \mathbf{v}} = \norm{\sqrt{T^* T} \mathbf{v}}^2 \nonumber
\end{align}
Hence $\norm{T \mathbf{v}} = \norm{\sqrt{T^* T} \mathbf{v}}$ for all $\mathbf{v} \in \mathcal{V}$.\\
Define linear map $S_1: \text{range}(\sqrt{T^* T}) \rightarrow \text{range}(T)$ by $S_1(\sqrt{T^* T} \mathbf{v}) = T \mathbf{v}$. To extend $S_1$ to an isometry $S \in \mathcal{L}(\mathcal{V})$ such that $T = S \sqrt{T^* T}$.\\
First to check $S_1$ is well defined. Suppose $\mathbf{v}_1, \mathbf{v}_2 \in \mathcal{V}$ such that $\sqrt{T^* T} \mathbf{v}_1 = \sqrt{T^* T}\mathbf{v}_2$, to show $T \mathbf{v}_1 = T \mathbf{v}_2$, 
\begin{align}
\norm{T \mathbf{v}_1 - T \mathbf{v}_2} &= \norm{T(\mathbf{v}_1 - \mathbf{v}_2)} = \norm{\sqrt{T^* T}(\mathbf{v}_1 - \mathbf{v}_2)} = \norm{\sqrt{T^* T}\mathbf{v}_1 - \sqrt{T^*T}\mathbf{v}_2} = 0 \nonumber
\end{align}
As $T \mathbf{v}_1 = T \mathbf{v}_2$, $S_1$ is well defined. Next, note that 
\begin{align}
S_1(\sqrt{T^* T}\mathbf{v}_1 + \sqrt{T^*T}\mathbf{v}_2) = S_1(\sqrt{T^*T}(\mathbf{v}_1 + \mathbf{v}_2)) = T(\mathbf{v}_1 + \mathbf{v}_2) = T\mathbf{v}_1 + T \mathbf{v}_2 = S_1(\sqrt{T^*T}\mathbf{v}_1) + S_1(\sqrt{T^*T}\mathbf{v}_2) \nonumber
\end{align}
Hence $S_1$ is additive. In addition, for any scalar $\alpha in \mathbb{F}$,
\begin{align}
S_1 (\alpha \sqrt{T^* T}\mathbf{v}_1) = S_1(\sqrt{T^* T} (\alpha \mathbf{v}_1)) = T(\alpha \mathbf{v}) = \alpha T \mathbf{v} = \alpha S_1 (\sqrt{T^* T}\mathbf{v}_1) \nonumber
\end{align}
hence scalar multiplication is preserved. Since $S_1$ satisfies additivity and homogeneity, it is a linear map.\\
By definition, $S_1$ maps $\text{range}(\sqrt{T^* T})$ onto $\text{range}(T)$, hence this imply $\norm{S_1 \mathbf{u}} = \norm{\mathbf{u}}$ for all $\mathbf{u} \in \text{range}(\sqrt{T^* T})$, hence $S_1$ is injective. Thus $\text{dim}(\text{range}(\sqrt{T^* T})) = \text{dim}(\text{range}(T))$, and thus $\text{dim}(\text{range}(\sqrt{T^* T}))^{\perp} = \text{dim}(\text{range}(T))^{\perp}$. The orthonormal bases $(\mathbf{e}_1, \ldots, \mathbf{e}_m)$ of $(\text{range}(\sqrt{T^* T}))^{\perp}$ and $(\mathbf{f}_1, \ldots, \mathbf{f}_m)$ of $(\text{range}(T))^{\perp}$ can be chosen, and these two bases have the same length. Define linear map $S_2: (\text{range}(\sqrt{T^* T}))^{\perp} \rightarrow (\text{range}(T))^{\perp}$ by
\begin{align}
S_2 (a_1 \mathbf{e}_1 + \cdots + a_m \mathbf{e}_m) = a_1 \mathbf{f}_1 + \cdots + a_m \mathbf{f}_1 \nonumber
\end{align}
Obviously $\norm{S_2 \mathbf{w}} = \norm{\mathbf{w}}$ for all $\mathbf{w} \in (\text{range}(\sqrt{T^* T}))^{\perp}$.\\
Let $S$ be the operator on $\mathcal{V}$ that equals $S_1$ on $\text{range}(\sqrt{T^*T})$ and equals $S_2$ on $(\text{range}(\sqrt{T^*T}))^{\perp}$. Note each $\mathbf{v} \in \mathcal{V}$ can be written uniquely in the form $\mathbf{v} = \mathbf{u} + \mathbf{w}$ where $\mathbf{u} \in \text{range}(\sqrt{T^*T})$ and $\mathbf{w} \in (\text{range}(\sqrt{T^*T}))^{\perp}$. Then define $S \mathbf{v}$ by $S \mathbf{v} = S_1 \mathbf{u} + S_2 \mathbf{w}$. For each $\mathbf{v} \in \mathcal{V}$, then $S(\sqrt{T^*T} \mathbf{v}) = S_1(\sqrt{T^*T}\mathbf{v}) = T\mathbf{v}$, hence $T = S\sqrt{T^*T}$.\\
To show $S$ is an isometry, note that
\begin{align}
\norm{S \mathbf{v}}^2 = \norm{S_1 \mathbf{u} + S_2 \mathbf{w}}^2 = \norm{S_1 \mathbf{u}}^2 + \norm{S_2 \mathbf{w}}^2 = \norm{\mathbf{u}}^2 + \norm{\mathbf{w}}^2 = \norm{\mathbf{v}}^2 \nonumber
\end{align}
where the second equality above holds as $S_1 \mathbf{u} \in \text{range}(T)$ and $S_2 \mathbf{u} \in (\text{range}(T))^{\perp}$.
\end{proof}
\end{theorem}

Polar decomposition states that each operator on $\mathcal{V}$ is the product of an isometry and a positive operator.

\begin{definition} Suppose $T \in \mathcal{L}(\mathcal{V})$. The \hlt{singular values} of $T$ are eigenvalues of $\sqrt{T^*T}$, with each eigenvalue $\lambda$ repeated $\text{dim}(\text{null}(\sqrt{T^*T} - \lambda I))$ times, all nonnegative as these are eigenvalues of positive operator $\sqrt{T^*T}$.
\end{definition}

\begin{theorem} \label{thm:singularvaluedecomposition} \hlt{Singular Value Decomposition}\\
If $T \in \mathcal{L}(\mathcal{V})$ has singular values $s_1, \ldots, s_n$, then $\exists$ orthonormal bases $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ and $(\mathbf{f}_1, \ldots, \mathbf{f}_n)$ of $\mathcal{V}$ s.t.
\begin{align}
T \mathbf{v} = s_1 \innerprod{\mathbf{v}, \mathbf{e}_1} \mathbf{f}_1 + \cdots + s_n \innerprod{\mathbf{v}, \mathbf{e}_n} \mathbf{f}_n \nonumber
\end{align}
for every $\mathbf{v} \in \mathcal{V}$.
\begin{proof} By Spectral Theorem \ref{thm:complexspectralthm} applied to $\sqrt{T^*T}$, $\exists$ an orthonormal bases $(\mathbf{e}_1, \ldots, \mathbf{e}_n)$ of $\mathcal{V}$ such that $\sqrt{T^*T} \mathbf{e}_j = s_j \mathbf{e}_j$ for $j = 1, \ldots, n$. Then $\mathbf{v} = \innerprod{\mathbf{v}, \mathbf{e}_1} \mathbf{e}_1 + \cdots + \innerprod{\mathbf{v}, \mathbf{e}_n} \mathbf{e}_n$ for every $\mathbf{v} \in \mathcal{V}$. Apply $\sqrt{T^*T}$ to both sides of this equation to get $\sqrt{T^*T}\mathbf{v} = s_1 \innerprod{\mathbf{v}, \mathbf{e}_1} \mathbf{e}_1 + \cdots + s_n \innerprod{\mathbf{v}, \mathbf{e}_n} \mathbf{e}_n$ for every $\mathbf{v} \in \mathcal{V}$.\\
By Polar Decomposition Theorem \ref{thm:polardecomposition}, $\exists$ an isometry $S \in \mathcal{L}(\mathcal{V})$ such that $T = S \sqrt{T^*T}$. Apply $S$ to both sides of the equation to get $T \mathbf{v} = s_1 \innerprod{\mathbf{v}, \mathbf{e}_1} S \mathbf{e}_1 + \cdots + s_n \innerprod{\mathbf{v}, \mathbf{e}_n} S \mathbf{e}_n$ for every $\mathbf{v} \in \mathcal{V}$.\\
For each $j$, let $\mathbf{f}_j = S \mathbf{e}_j$. As $S$ is an isometry, $(\mathbf{f}_1, \ldots, \mathbf{f}_n)$ is an orthonormal basis of $\mathcal{V}$. Then
\begin{align}
T \mathbf{v} = s_1 \innerprod{\mathbf{v}, \mathbf{e}_1} \mathbf{f}_1 + \cdots + s_n \innerprod{\mathbf{v}, \mathbf{e}_n} \mathbf{f}_n \nonumber
\end{align}
for every $\mathbf{v} \in \mathcal{V}$, completing the proof.
\end{proof}
\end{theorem}

\subsubsection{Operators on Complex Vector Spaces}

\begin{definition} Given $T \in \mathcal{L}(\mathcal{V})$, and $\lambda$ an eigenvalue of $T$, a vector $\mathbf{v} \in \mathcal{V}$ is a \hlt{generalised eigenvector} of $T$ corresponding to $\lambda$ if $(T - \lambda I)^j \mathbf{v} = \mathbf{0}$ for some positive integer $j$.
\end{definition}

Suppose $T \in \mathcal{L}(\mathcal{V})$ and $k \geq 0$, $k \in \Z$. If $T^k \mathbf{v} = \mathbf{0}$, then $T^{k+1}\mathbf{v} = T(T^k \mathbf{v}) = T(\mathbf{0}) = \mathbf{0}$, hence,
\begin{align}
\{\mathbf{0} \} = \text{null}(T^0) \subset \text{null}(T^1) \subset \cdots \subset \text{null}(T^k) \subset \cdots \nonumber
\end{align}

Next proposition states that once two consecutive terms in this sequence of subspaces are equal, then later terms in the sequence are equal.

\begin{proposition} \label{prop:equalnullspace} If $T \in \mathcal{L}(\mathcal{V})$ and $m \geq 0$, $m \in \Z$ such that $\text{null}(T^m) = \text{null}(T^{m+1})$, then
\begin{align}
\text{null}(T^0) \subset \text{null}(T^1) \subset \cdots \subset \text{null}(T^m) = \text{null}(T^{m+1}) = \text{null}(T^{m+2}) = \cdots \nonumber
\end{align}
\begin{proof} Suppose $T \in \mathcal{L}(\mathcal{V})$ and $m \geq 0$, $m \in \Z$ such that $\text{null}(T^m) = \text{null}(T^{m+1})$. Let $k$ be a positive integer.\\
Note that $\text{null}(T^{m+k}) \subset \text{null}(T^{m+k+1})$. To prove inclusion in other direction, suppose $\mathbf{v} \in \text{null}(T^{m+k+1})$, then
\begin{align}
\mathbf{0} &= T^{m+k+1} \mathbf{v} = T^{m+1} (T^k \mathbf{v}) \nonumber \\
T^k \mathbf{v} \in \text{null}(T^{m+1}) = \text{null}(T^m) \nonumber
\end{align}
Hence $\mathbf{0} = T^m(T^k \mathbf{v}) = T^{m+k} \mathbf{v}$, thus $\mathbf{v} \in \text{null}(T^{m+k})$. Hence this implies $\text{null}(T^{m+k+1}) \subset \text{null}(T^{m+k})$.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:equalnullspacedimv} If $T \in \mathcal{L}(\mathcal{V})$, then
\begin{align}
\text{null}(T^{\text{dim}(\mathcal{V})}) = \text{null}(T^{\text{dim}(\mathcal{V} + 1)}) = \text{null}(T^{\text{dim}(\mathcal{V}+2)}) = \cdots \nonumber
\end{align}
\begin{proof} Suppose $T \in \mathcal{L}(\mathcal{V})$, and $\text{null}(T^{\text{dim}(\mathcal{V})}) \neq \text{null}(T^{\text{dim}(\mathcal{V}+1)})$, then by Proposition \ref{prop:equalnullspace},
\begin{align}
\{ \mathbf{0} \} = \text{null}(T^0) \subsetneq \text{null}(T^1) \subsetneq \cdots \subsetneq \text{null}(T^{\text{dim}(\mathcal{V})}) \subsetneq \text{null}(T^{\text{dim}(\mathcal{V}+1)}) \nonumber
\end{align}
By strict inclusion, dimension must increase by at least $1$, hence $\text{dim}(\text{null}(T^{\text{dim}(\mathcal{V}+1)})) \geq \text{dim}(\mathcal{V} = 1)$, a contraction as a subspace of $\mathcal{V}$ cannot have a larger dimension than $\text{dim}(\mathcal{V})$.
\end{proof}
\end{proposition}

The next corollary implies that the generalised eigenvectors of $T \in \mathcal{L}(\mathcal{V})$ corresponding to an eigenvalue $\lambda$ is a subspace of $\mathcal{V}$.

\begin{corollary} \label{cor:geneigennullpowerdim} Suppose $T \in \mathcal{L}(\mathcal{V})$ and $\lambda$ is an eigenvalue of $T$, then the set of generalised eigenvectors of $T$ corresponding to $\lambda$ equals $\text{null}(T - \lambda I)^{\text{dim}(\mathcal{V})}$.
\begin{proof} If $\mathbf{v} \in \text{null} (T - \lambda I)^{\text{dim}(\mathcal{V})}$, then $\mathbf{v}$ is a generalised eigenvector of $T$ corresponding to $\lambda$.\\
Conversely, suppose $\mathbf{v} \in \mathcal{V}$ is a generalised eigenvector of $T$ corresponding to $\lambda$, thus there exists $j \geq 0 \in \Z$ such that $\mathbf{v} \in \text{null}(T - \lambda I)^j$. By Proposition \ref{prop:equalnullspace} and Proposition \ref{prop:equalnullspacedimv}, replacing $T$ with $T - \lambda I$, then $\mathbf{v} \in \text{null}(T - \lambda I)^{\text{dim}(\mathcal{V})}$ as desired.
\end{proof}
\end{corollary}

\begin{definition} An operator is \hlt{nilpotent} if some power of it equals $0$.
\end{definition}

\begin{corollary} Suppose $N \in \mathcal{L}(\mathcal{V})$ is nilpotent, then $N^{\text{dim}(\mathcal{V})} = 0$.
\begin{proof} As $N$ is nilpotent, every eigenvector in $\mathcal{V}$ is a generalised eigenvector corresponding to eigenvalue $0$. Hence from Corollary \ref{cor:geneigennullpowerdim}, $\text{null}(N^{\text{dim}(\mathcal{V})}) = \mathcal{V}$ as desired.
\end{proof}
\end{corollary}

Suppose $T \in \mathcal{L}(\mathcal{V})$ and $k \geq 0, k \in \Z$. If $\mathbf{w} \in \text{range}(T^{k+1})$, then $\exists$ $\mathbf{v} \in \mathcal{V}$ with $\mathbf{w} = T^{k+1} \mathbf{v} = T^k (T \mathbf{v}) \in \text{range}(T^k)$, hence $\text{range}(T^{k+1}) \subset \text{range}(T^k)$, thus
\begin{align}
\mathcal{V} = \text{range}(T^0) \supset \text{range}(T^1) \supset \cdots \supset \text{range}(T^k) \subset \text{range}(T^{k+1}) \subset \cdots \nonumber
\end{align} 

Next proportion shows that inclusions become equalities once power reaches dimension $\mathcal{V}$.

\begin{proposition} If $T \in \mathcal{L}(\mathcal{V})$, then $\text{range}(T^{\text{dim}(\mathcal{V})}) = \text{range}(T^{\text{dim}(\mathcal{V}+1)}) = \text{range}(T^{\text{dim}(\mathcal{V})+2}) = \cdots$
\begin{proof} Suppose $m > \text{dim}(\mathcal{V})$, then
\begin{align}
\text{dim}(\text{range}(T^m)) &= \text{dim}(\mathcal{V}) - \text{dim}(\text{null}(T^m)) = \text{dim}(\mathcal{V}) - \text{dim}(\text{null}(T^{\text{dim}(\mathcal{V})})) = \text{dim}(\text{range}(T^{\text{dim}(\mathcal{V}})) \nonumber
\end{align}
by Proposition \ref{prop:equalnullspacedimv}. As $\text{range}(T^{\text{dim}(\mathcal{V})}) \subset \text{range}(T^m)$, and $\text{dim}(\text{range}(T^{\text{dim}(\mathcal{V}})) = \text{dim}(\text{range}(T^m))$ from above, this implies $\text{range}(T^{\text{dim}(\mathcal{V}}) = \text{range}(T^m)$ as desired.
\end{proof}
\end{proposition}

Next theorem implies that the number of times an eigenvalue is repeated on diagonal of upper-triangular matrix of $T$ is independent of the choice of basis.

\begin{theorem} \label{thm:eigenvaluerepeatisindepbasis} Let $T \in \mathcal{L}(\mathcal{V})$ and $\lambda \in \mathbb{F}$, then for every basis of $\mathcal{V}$ with respect to which $T$ has an upper-triangular matrix, $\lambda$ appears on diagonal of matrix of $T$ precisely $\dim(\text{null}(T - \lambda I))^{\dim(\mathcal{V})}$ times.
\begin{proof} If $T$ is upper-triangular with some basis $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ and diagonal $\lambda_1, \ldots, \lambda_n$, then the number of times a given scalar $\lambda_j$ appears on diagonal equals $\text{dim}(\text{null}(T - \lambda I)^n)$. As replacing $T$ by $T - \lambda I$ shifts diagonal by $-\lambda$, it suffes to prove the case $\lambda = 0$. Let $n = \text{dim}(\mathcal{V})$.\\
Assume $n > 1$, and statement is true in dimension $n-1$. Pick basis $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ which makes $T$ upper-triangular.
\begin{align}
\mathcal{M}(T) =
\begin{bmatrix}
\lambda_1 & & & * \\
& \ddots & & \\
& & \lambda_{n-1} & \\
0 & & & \lambda_n
\end{bmatrix} \nonumber
\end{align}
Let $\mathcal{U} = \text{span}(\mathbf{v}_1, \ldots \mathbf{v}_{n-1})$. By Proposition \ref{prop:invariantsubspaceequivalence}, $\mathcal{U}$ is invariant under $T$, and
\begin{align}
\mathcal{M}(T|_{\mathcal{U}}) =
\begin{bmatrix}
\lambda_1 & & * \\
& \ddots & \\
0 & & \lambda_{n-1}
\end{bmatrix} \nonumber
\end{align}
By induction hypothesis applied to $T|_{\mathcal{U}}$ (space dimension $n-1$), the number of zeros on diagonal is $\text{dim}(\text{null}(T|_{\mathcal{U}}))^{n-1}$. The nullspaces in power sequence stabilises by dimension $n-1$, hence $\text{null}(T|_{\mathcal{U}})^{n-1} = \text{null}(T|_{\mathcal{U}})^n$, thus $0$ appears on diagonal exactly $\text{null}(T|_{\mathcal{U}})^n$ times. To finish proof, relate $\text{null}(T^n)$ to $\text{null} (K|_{\mathcal{U}})^n$ times.\\
For case $\lambda_n \neq 0$, claim $\text{null}(T^n) \subset \mathcal{U}$. Note that powers of upper-triangular matrix are still upper triangular with diagonal $\lambda_1^n, \ldots \lambda_n^n$. In particular
\begin{align}
\mathcal{M}(T^n) = \mathcal(T)^n \Rightarrow T^n \mathbf{v}_n = \mathbf{u} + \lambda_n^n \mathbf{v}_n \nonumber
\end{align}
for some $\mathbf{u} \in \mathcal{U}$. Take any $\mathbf{v} \in \text{null}(T^n)$ and write $\mathbf{v} = \tilde{\mathbf{u}} + a \mathbf{v}_n$, where $\tilde{\mathbf{u}} \in \mathcal{U}$ and $a \in \mathbb{F}$, then
\begin{align}
\mathbf{0} = T^n \mathbf{v} = T^n \tilde{\mathbf{u}} + a T^n \mathbf{v}_n = T^n \tilde{\mathbf{u}} + a \mathbf{u} + a \lambda_n^n \mathbf{v}_n \nonumber
\end{align}
Note that $T^n \tilde{\mathbf{u}}, a \mathbf{u} \in \mathcal{U}$ and $\mathbf{v}_n \neq \mathcal{U}$, which implies $a \lambda_n^n = 0$. However $\lambda_n \neq 0$, so $a=0$, hence $\mathbf{v} = \tilde{\mathbf{u}} \in \mathcal{U}$. Thus $\text{null}(T^n) = \text{null}(K|_{\mathcal{U}})^n$, hence total number of zeros on diagonal of $\mathcal{M}(T)$ equals $\mathcal{M}(T|_{\mathcal{U}})$, namely $\text{dim}(\text{null}(T^n))$.\\
For case $\lambda_n =0 $, to get $\text{dim}(\text{null}(T^n)) = \text{dim}(\text{null}(K|_{\mathcal{U}}))^n + 1$. By dimension for sum of subspaces,
\begin{align}
\text{dim}(\text{null}(T^n)) &= \text{dim}z\mathcal{U} \cap \text{null}(T^n)) + \text{dim}(\mathcal{U} + \text{null}(T^n)) - \text{dim}(\mathcal{U}) \nonumber
\end{align}
Note that $\mathcal{U} \cap \text{null}(T^n) = \text{null}(K|_{\mathcal{U}})^n$, hence result will follow if $\mathcal{U} + \text{null}(T^n) = \mathcal{V}$, i.e., there is a vector in $\text{null}(T^n)$ not in $\mathcal{U}$. The vector might be of form $\mathbf{u} - \mathbf{v}_n$ with $\mathbf{u} \in \mathcal{U}$. Want $T^n (\mathbf{u} - \mathbf{v}_n) = \mathbf{0}$, i.e., $T^n \mathbf{u} = T^n \mathbf{v}_n$.\\
This is solvable if $T^n \mathbf{v}_n \in \text{range}(T|_{\mathcal{U}})^n$. When $\lambda_n = 0$, the last diagonal entry is $0$, hence the last coordinate of $T \mathbf{v}_n$ is $0$, hence $T \mathbf{v}_n \in \mathcal{U}$, therefore
\begin{align}
T^n \mathbf{v}_n = T^{n-1}(T \mathbf{v}_n) \in \text{range}(T|_{\mathcal{U}})^{n-1} \nonumber
\end{align}
As range stabilises, thus $ \text{range}(T|_{\mathcal{U}})^{n-1} = \text{range}(T|_{\mathcal{U}})^n$, hence there exists $\mathbf{u} \in \mathcal{U}$ with $T^n \mathbf{u} = T^n \mathbf{v}_n$, making $\mathbf{u} - \mathbf{v}_n \in \text{null}(T^n)$. Since $\mathbf{v}_n \notin \mathcal{U}$, this vector is not in $\mathcal{U}$. As $\mathcal{U} + \text{null}(T^n) = \mathcal{V}$, and with first equation in this case, the $\lambda_n = 0$ case is done.\\
In either case the number of zeroes on diagonal of $\mathcal{M}(T)$ equals $\text{dim}(\text{null}(T^n))$. Replacing $T$ with $T - \lambda I$ gives the general statement: the number of times $\lambda$ appears on diagonal equals $\text{dim}(\text{null}(T - \lambda I)^n)$.
\end{proof}
\end{theorem}

\begin{definition} Suppose $T \in \mathcal{L}(\mathcal{V})$. The \hlt{multiplicity} of eigenvalue $\lambda$ in $T$ is defined to be dimension of subspace of generalised eigenvectors corresponding to $\lambda$, which is $\text{dim}(\text{null}(T - \lambda I)^{\text{dim}(\mathcal{V})})$.
\end{definition}

\begin{proposition} \label{prop:complexvectorspacesummultiplicities} If $\mathcal{V}$ is a complex vector space and $T \in \mathcal{L}(\mathcal{V})$, then the sum of multiplicities of all eigenvalues of $T$ equals $\text{dim}(\mathcal{V})$.
\begin{proof}
Over $\C$, every complex operator $T$ has at least one eigenvalue, ensuring that $T$ can be represented in upper-triangular form. There exists a basis of $\mathcal{V}$ such that the matrix of $T$ is upper-triangular
\begin{align}
\begin{bmatrix}
\lambda_1 & &  * \\
& \ddots & \\
0 & & \lambda_n
\end{bmatrix} \nonumber
\end{align}
where $n = \text{dim}(\mathcal{V})$. The eigenvalues of $T$ are exactly diagonal entries $\lambda_1, \ldots, \lambda_n$, where each eigenvalue may repeat. By Theorem \ref{thm:eigenvaluerepeatisindepbasis}, the multiplicity of $\lambda$ is number of times $\lambda$ appears on diagonal, hence listing all diagonal entries counting repeats is listing all eigenvalues with their multiplicities.\\
The diagonal of triangular matrix has length $n = \text{dim}(\mathcal{V})$, hence proving this.
\end{proof}	
\end{proposition}

\begin{definition} Let $\mathcal{V}$ be complex vector space, and $T \in \mathcal{L}(\mathcal{V})$. Let $\lambda_1, \ldots, \lambda_m$ be distinct eigenvalues of $T$, $d_j$ be multiplicity of $\lambda_j$ as an eigenvalue of $T$. The \hlt{characteristic polynomial} of $T$ is $(z - \lambda_1)^{d_1} \cdots (z - \lambda_m)^{d_m}$.
\end{definition}

\begin{theorem} \label{thm:cayleyhamiltonthm} \hlt{Cayley-Hamilton Theorem}\\
Suppose $\mathcal{V}$ is complex vector space and $T \in \mathcal{L}(\mathcal{V})$, and $q$ is characteristic polynomial of $T$. Then $q(T) = 0$.
\begin{proof} Suppose $(\mathbf{v}_1, \ldots, \mathbf{v}_n)$ is basis of $\mathcal{V}$ with respect to which matrix of $T$ has upper-triangular form
\begin{align}
\mathcal{M}(T) =
\begin{bmatrix}
\lambda_1 & & * \\
& \ddots & \\
0 & & \lambda_n
\end{bmatrix} \nonumber
\end{align}
To show $q(T)\mathbf{v}_j = 0$ for $j = 1, \ldots, n$. As $q(x) = (x - \lambda_1) \cdots (x - \lambda_n)$, this reduces to showing
\begin{align}
(T - \lambda_1 I)(T - \lambda_2 I) \cdots (T - \lambda_j I) \mathbf{v}_j = 0 \nonumber
\end{align}
for each $j$. For base case $j=1$, since $T\mathbf{v}_1 = \lambda_1 \mathbf{v}_1$, then $(T - \lambda_1 I)\mathbf{v}_1 = 0$.\\
Suppose claim holds for all $\mathbf{v}_1, \ldots, \mathbf{v}_{j-1}$. As $T$ is upper-triangular in this basis, then
\begin{align}
(T - \lambda_j I)\mathbf{v}_j \in \text{span}(\mathbf{v}_1, \ldots, \mathbf{v}_{j-1}) \nonumber
\end{align}
By induction hypothesis, $(T - \lambda_1 I) \cdots (T - \lambda_{j-1}I)$ applied $(T - \lambda_j I)\mathbf{v}_j$ gives $0$, hence completing the proof.
\end{proof}
\end{theorem}

Next proposition shows that the null space of any polynomial of $T \in \mathcal{L}(\mathcal{V})$ is invariant under $T$.

\begin{proposition} \label{prop:nullpolyinvariantundert} If $T \in \mathcal{L}(\mathcal{V})$ and $p \in \mathcal{P}(\mathbb{F})$, then $\text{null}(p(T))$ is invariant under $T$.
\begin{proof} Let $\mathbf{v} \in \text{null}(p(T))$, then $p(T)\mathbf{v} = 0$, thus
\begin{align}
(p(T))(T\mathbf{v}) = T(p(T)\mathbf{v}) = T(0) = 0 \nonumber
\end{align}
hence $T \mathbf{v} \in \text{null}(p(T))$, thus $\text{null}(p(T))$ is invariant under $T$ as desired.
\end{proof}
\end{proposition}

Next theorem shows every operator on complex vector space can be composed of nilpotent operator plus scalar multiple of the identity.

\begin{theorem} \label{thm:majorstructurethm} \hlt{Major Structure Theorem / Primary Decomposition Theorem}\\
Suppose $\mathcal{V}$ is a complex vector space and $T \in \mathcal{L}(\mathcal{V})$. Let $\lambda_1, \ldots, \lambda_m$ be distinct eigenvalues of $T$, and let $\mathcal{U}_1, \ldots, \mathcal{U}_m$ be corresponding subspaces of generalised eigenvalues. Then
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m$
\item each $\mathcal{U}_j$ is invariant under $T$
\item each $(T - \lambda_j I)|_{\mathcal{U}_j}$ is nilpotent
\end{enumerate}
\begin{proof} 
$(i)$ The multiplicity of $\lambda_j$ is $\text{dim}(\mathcal{U}_j)$. The sum of these multiplicities equal $\text{dim}(\mathcal{U}_j)$, hence
\begin{align}
\text{dim}(\mathcal{V}) = \text{dim}(\mathcal{U}_1) + \cdots + \text{dim}(\mathcal{U}_m) \nonumber
\end{align}
Let $\mathcal{U} = \mathcal{U}_1 + \cdots + \mathcal{U}_m$, clearly $\mathcal{U}$ is invariant under $T$. Define $S \in \mathcal{L}(\mathcal{U})$ by $S = T_{\mathcal{U}}$. Note that $S$ has same eigenvalues with same multiplicities as $T$, as all generalised eigenvectors of $T$ are in $\mathcal{U}$ (domain of $S$). Applying Proposition \ref{prop:complexvectorspacesummultiplicities}, then
\begin{align}
\text{dim}(\mathcal{U}) = \text{dim}(\mathcal{U}_1) + \cdots + \text{dim}(\mathcal{U}_m) \nonumber
\end{align}
Hence $\text{dim}(\mathcal{U}) = \text{dim}(\mathcal{V})$. As $\mathcal{U}$ is a subspace of $\mathcal{V}$, this implies $\mathcal{U} = \mathcal{V}$, i.e., $\mathcal{V} = \mathcal{U}_1 + \cdots + \mathcal{U}_m$.\\
$(ii)$. Note that $\mathcal{U}_j = \text{null}(T - \lambda_j I)^{\text{dim}(\mathcal{V})}$ for each $j$ (by \ref{cor:geneigennullpowerdim}). By \ref{prop:nullpolyinvariantundert} (with $p(z) = (z - \lambda_j)^{\text{dim}(\mathcal{V})}$).\\
$(iii)$. This is obvious from the definitions.
\end{proof}
\end{theorem}

The following corollary shows that on a complex vector space there are enough generalised eigenvectors to form a basis for the domain of an operator.

\begin{corollary} Suppose $\mathcal{V}$ is a complex vector space and $T \in \mathcal{L}(\mathcal{V})$. Then there is a basis of $\mathcal{V}$ consisting of generalised eigenvectors of $T$.
\begin{proof} Choose a basis for each $\mathcal{U}_j$ in Theorem \ref{thm:majorstructurethm}. Put all these bases together to form a basis of $\mathcal{V}$ consisting of generalised eigenvectors of $T$.
\end{proof}
\end{corollary}

\begin{lemma} \label{lem:nilpotentopstrictuptrigmatrix} Suppose $N$ is nilpotent operator on $\mathcal{V}$, then $\exists$ a basis of $\mathcal{V}$ w.r.t. which matrix of $N$ has form
\begin{align}
\begin{bmatrix}
0 & & *\\
& \ddots & \\
0 & & 0
\end{bmatrix} \nonumber
\end{align}
where all entries on and below diagonal are $0$s.
\begin{proof} Since $N$ is nilpotent, $\exists$ some integer $m \geq 1$ such that $N^m = 0$. Hence $\exists$ ascending chain of subspaces
\begin{align}
\{ 0 \} \subseteq \text{null}(N)	 \subseteq \text{null}(N^2) \subseteq \cdots \subseteq \text{null}(N^m) = \mathcal{V} \nonumber	
\end{align}
Build a basis of $\mathcal{V}$ inductively along this chain. First choose a basis of $\text{null}(N)$, then extend this to a basis of $\text{null}(N^2)$, then to $\text{null}(N^3)$ and so on. After finitely many steps, a basis of $\mathcal{V}$ is constructed that is adapted to the chain of subspaces
\begin{align}
\text{null}(N) \subseteq \text{null}(N^2) \subseteq \cdots \subseteq \text{null}(N^m) \nonumber
\end{align}
Examine matrix of $N$ with respect to this basis.\\
The first set of basis vectors (in $\text{null}(N)$) are sent to $0$ under $N$, hence the corresponding columns of the matrix are zero columns. The next set of basis vectors (those in $\text{null}(N^2)$ but not in $\text{null}(N)$) are mapped by $N$ into $\text{null}(N)$, hence the image of these vectors are linear combinations of the earlier basis vectors. Thus the nonzero entries in these columns occur only in rows corresponding to earlier basis vectors (i.e., above the diagonal). Similarly, the next set of basis vectors (in $\text{null}(N^3)$ but not in $\text{null}(N^2)$) are mapped to $\text{null}(N^2)$, hence their images are linear combinations of basis vectors chosen earlier, hence nonzero entries in their columns lie above the diagonal. Continuing inductively, for each $k$, the basis vectors from $\text{null}(N^k)$ are mapped by $N$ into $\text{null}(N^{k-1})$, hence each column has nonzero entries only in rows corresponding to earlier basis vectors.\\
Hence, all diagonal entries of $N$ are zero (since $N$ is nilpotent, no vector is mapped to a nonzero multiple of itself). All entries below the diagonal are zero (since images of later basis vectors lie in the span of earlier ones). All possible nonzero entries occur strictly above diagonal. Hence matrix of $N$ is strictly upper triangular.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:triangularformthmforlinopoverc} \hlt{Triangular Form Theorem for Linear Operators over $\C$}\\
Suppose $\mathcal{V}$ is a complex vector space and $T \in \mathcal{L}(\mathcal{V})$. Let $\lambda_1, \ldots, \lambda_m$ be distinct eigenvalues of $T$. Then there exists a basis of $\mathcal{V}$ with respect to which the matrix of $T$ is block diagonal of form
\begin{align}
\begin{bmatrix}
A_1 & & 0 \\
 & \ddots & \\
0 & & A_m
\end{bmatrix} \nonumber
\end{align}
where each $A_j$ is an upper triangular matrix of form
\begin{align}
A_j = \begin{bmatrix}
\lambda_j & & * \\
& \ddots & \\
0 & & \lambda_j
\end{bmatrix} \nonumber
\end{align}
\begin{proof} For $j=1, \ldots, m$, let $\mathcal{U})_j$ be subspace of generalised eigenvectors of $T$ corresponding to $\lambda_j$. By Major Structure Theorem \ref{thm:majorstructurethm}(c), $(T - \lambda_j I)|_{\mathcal{U}_j}$ is nilpotent. For each $j$, choose basis of $\mathcal{U}_j$ such that matrix of $(T - \lambda_j I)|_{\mathcal{U}_j}$ with respect to this basis is as seen in Lemma \ref{lem:nilpotentopstrictuptrigmatrix}. Hence the matrix of $T|_{\mathcal{U}_j}$ will be as seen in $A_j$ above. Putting together the basis for $\mathcal{U}_j$ gives a basis for $\mathcal{V}$, by Major Structure Theorem \ref{thm:majorstructurethm}(a). The matrix of $T$ with respect to this basis has the desired form.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:nilpotentaddidenhassqrroot} Suppose $N \in \mathcal{L}(\mathcal{V})$ is nilpotent, then $I+N$ has a square root.
\begin{proof} Let $m \geq 1$ with $N^m = 0$. Seek square rot of $I+N$ in form
\begin{align}
S = I + a_1 N + a_2 N^2 + \cdots + a_{m-1} N^{m-1} \nonumber
\end{align}
where $a_j \in \C$ are to be chosen (ignore higher powers as $N^m = 0$.\\
Let $a_0 = 1$. Using $N^i N^j = N^{i+j}$, expand
\begin{align}
S^2 = \sum\limits_{i=0}^{m-1} \sum\limits_{j=0}^{m-1} a_i a_j N^{i+j} = \sum\limits_{k=0}^{m-1} \left( \sum\limits_{r=0}^k a_r a_{k-r} \right)N^k \nonumber
\end{align}
as all terms with exponent $\geq m$ vanish. For each $k$, let $c_k = \sum\limits_{r=0}^k$ be coefficient of $N^k$ in $S^2$. To have
\begin{align}
c_0 = 1, \ \ \ c_1 = 1, \ \ \ c_k = 0 \ (2 \leq k \leq m-1) \nonumber
\end{align}
Note that $c_0 = a_0^2 = 1$ automatically. For $k \geq 1$, split off the terms with $r=0$, $k$:
\begin{align}
c_k = a_0 a_k + a_k a_0 + \sum\limits_{r=1}^{k-1} a_r a_{k-r} = 2a_k + \sum\limits_{r=1}^{k-1} a_r a_{k-r} \nonumber
\end{align}
Hence to solve triangular system
\begin{align}
2a_1 = 1, \ \ \ \ 2a_k + \sum\limits_{r=1}^{k-1} a_r a_{k-r} = 0 \ (2 \leq k \leq m-1) \nonumber
\end{align}
Recursively define $a_1 = \frac{1}{2}, a_k = - \frac{1}{2} \sum\limits_{r=1}^{k-1} a_r a_{k-r}$. As the right-hand side for $a_k$ involves only $a_1, \ldots, a_{k-1}$, this uniquely determines $a_k$ for all $k$. With these choices, then $c_1 = 1$, $c_k = 0$ for $2 \leq k \leq m-1$, therefore $S^2 = 1+N$. Hence $S$ is a square root of $1 + N$.
\end{proof}
\end{lemma}

\begin{theorem} Let $\mathcal{V}$ be a complex vector space and $T \in \mathcal{L}(\mathcal{V})$. If $T$ is invertible, then $T$ has a square root.
\begin{proof} Let $\lambda_1, \ldots, \lambda_m$ be distinct eigenvalues of $T$, and $\mathcal{U}_j$ be the generalised eigenspace corresponding to $\lambda_j$. By Major Structure Theorem \ref{thm:majorstructurethm}, for each $j$, there exists a nilpotent operator $N_j \in \mathcal{L}(\mathcal{U}_j)$ such that $T|_{\mathcal{U}_j} = \lambda_j I + N_j$. As $T$ is invertible, $0$ is not an eigenvalue, hence each $\lambda_j \neq 0$, thus
\begin{align}
T|_{\mathcal{U}_j} = \lambda_j \left(I + \frac{N_j}{\lambda_j} \right) \nonumber
\end{align}
Since $N_j$ is nilpotent, so is $N_j/\lambda_j$, hence by Lemma \ref{lem:nilpotentaddidenhassqrroot}, there exists operator $R_j \in \mathcal{L}(\mathcal{U}_j)$ with
\begin{align}
R^2 = I + \frac{N_j}{\lambda_j} \nonumber
\end{align}
on $\mathcal{U}_j$. Choose $\alpha_j \in \C$ with $\alpha_j^2 = \lambda_j$ (possible as the field is $\C$). Define $S_j = \alpha_j R_j \in \mathcal{L}(\mathcal{U}_j)$, then
\begin{align}
S_j^2 = \alpha_j^2 R_j^2 = \lambda_j \left( I + \frac{N_j}{\lambda_j} \right) = T|_{\mathcal{U}_j} \nonumber
\end{align}
Using direct sum decomposition $\mathcal{V} = \oplus_j \mathcal{U}_j$, each $\mathbf{v} \in \mathcal{V}$ can be written as $\mathbf{v} = \mathbf{u}_1 + \cdots + \mathbf{u}_m$ with $\mathbf{u} \in \mathcal{U}_j$. Define $S \in \mathcal{L}(\mathcal{V})$ by 
\begin{align}
S(\mathbf{v}) = S(\mathbf{u}_1 + \cdots + \mathbf{u}_m) = S_1 \mathbf{u}_1 + \cdots + S_m \mathbf{u}_m \nonumber
\end{align}
This is well-defined (the decomposition is unique) and linear (blockwise linearity).\\
As each $S_j (\mathcal{U}_j) \subseteq \mathcal{U}_j$, then $\mathbf{v} = \mathbf{u}_1 + \cdots + \mathbf{u}_m$,
\begin{align}
S^2 \mathbf{v} = S(S_1 \mathbf{u}_1 + \cdots + S_m \mathbf{u}_m) = S_1^2 \mathbf{u}_1 + \cdots + S_m^2 \mathbf{u}_m = T|_{\mathcal{U}_1} \mathbf{u}_1 + \cdots + T|_{\mathcal{U}_m} \mathbf{u}_m = T(\mathbf{u}_1 + \cdots + \mathbf{u}_m) = T \mathbf{v} \nonumber
\end{align}
Hence $S^2 = T$, and thus $S$ is a square root of $T$.
\end{proof}
\end{theorem}

\begin{definition} A \hlt{monic polynomial} is a non-zero univariate polynomial in which the leading coefficient equals $1$, i.e., $x^n + c_{n-1} x^{n-1} + \cdots + c_2 x^2 + c_1 x + x_0$ with $n \geq 0$.
\end{definition}

\begin{definition} A \hlt{minimal polynomial} of $T$ is monic polynomial $\mathbf{p} \in \mathcal{P}(\mathbb{F})$ of smallest degree s.t. $\mathbf{p}(T) = 0$.
\end{definition}

\begin{theorem} \label{thm:minimalpolydivides} Let $T \in \mathcal{L}(\mathcal{V})$ and let $\mathbf{q} \in \mathcal{P}(\mathbb{F})$, then $\mathbf{q}(T) = 0$ if and only if minimal polynomial of $T$ divides $\mathbf{q}$.
\begin{proof} $(\Leftarrow)$ Let $\mathbf{q}$ be minimal polynomial of $T$. Suppose $\mathbf{p}$ divides $\mathbf{q}$, hence there exists polynomial $\mathbf{s} \in \mathcal{P}(\mathbb{F})$ such that $\mathbf{q} = \mathbf{s}\mathbf{p}$. Then
\begin{align}
\mathbf{q}(T) = \mathbf{s}(T) \mathbf{p}(T) = \mathbf{s}(T)0 = 0 \nonumber
\end{align}
$(\Rightarrow)$ Suppose $\mathbf{q}(T) = 0$. By division algorithm, there exists polynomials $\mathbf{s}, \mathbf{r} \in \mathcal{P}(\mathbb{F})$ such that $\mathbf{q} = \mathbf{s} \mathbf{p} + \mathbf{r}$ and $\text{deg}(\mathbf{r}) < \text{deg}(\mathbf{p})$, then
\begin{align}
0 = \mathbf{q}(T) = \mathbf{s}(T) \mathbf{p}(T) + \mathbf{r}(T) = \mathbf{r}(T) \nonumber 
\end{align}
As $\mathbf{p}$ is minimal polynomial of $T$ and $\text{deg}(\mathbf{r}) < \text{deg}(\mathbf{p})$, the equation above implies $\mathbf{r} = 0$. Hence equation becomes $\mathbf{q} = \mathbf{s}\mathbf{p}$, hence $\mathbf{p}$ divides $\mathbf{q}$ as desired.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:minimalpolyiseigenvalues} Let $T \in \mathcal{L}(\mathcal{V})$, then the roots of minimal polynomial of $T$ are precisely eigenvalues of $T$.
\begin{proof} $(\Rightarrow)$ Let
\begin{align}
\mathbf{p} = a_0 + a_1 z + a_2 z^2 + \cdots + a_{m-1} z^{m-1} + z^m \nonumber
\end{align}
be minimal polynomial of $T$. Suppose $\lambda \in \mathbb{F}$ is a root of $\mathbf{p}$, then the polynomial of $T$ is $\mathbf{p}(z) = (z - \lambda)\mathbf{q}(z)$, where $\mathbf{q}$ is a monic polynomial with coefficients in $\mathbb{F}$. As $\mathbf{p}(T) = 0$, then
\begin{align}
0 = (T - \lambda I)(\mathbf{q}(T) \mathbf{v}) \nonumber
\end{align}
for all $\mathbf{v} \in \mathcal{V}$. As $\text{deg}(\mathbf{q}) < \text{deg}(\mathbf{q})$, there must exist at least one vector $\mathbf{v} \in \mathcal{V}$ such that $\mathbf{q}(T) \mathbf{v} \neq 0$. This implies $\lambda$ is an eigenvalue of $T$ as desired.\\
$(\Leftarrow)$ Suppose $\lambda \in \mathbb{F}$ is an eigenvalue of $T$. Let $\mathbf{v}$ be nonzero vector in $\mathcal{V}$ such that $T \mathbf{v} = \lambda \mathbf{v}$, Repeated application of $T$ to both sides of equation show $T^j \mathbf{v} = \lambda^j \mathbf{v}$ for every nonnegative integer $j$. Hence
\begin{align}
0 = \mathbf{p}(T) \mathbf{v} &= (a_0 + a_1 T + a_2 T^2 + \cdots + a_{m-1} T^{m-1} + T^m) \mathbf{v} \nonumber \\
&= (a_0 + a_1 \lambda + a_2 \lambda^2 + \cdots + a_{m-1} \lambda^{m-1} + \lambda^m) \mathbf{v} \nonumber \\
&= \mathbf{p}(\lambda) \mathbf{v} \nonumber
\end{align}
As $\mathbf{v} \neq \mathbf{0}$, this implies $\mathbf{p}(\lambda) = 0$ as desired.
\end{proof}
\end{theorem}

\begin{remark} \hlt{Computing Minimal Polynomial}\\
Given operator $T \in \mathcal{L}(\mathcal{V})$, consider $(\mathcal{M}(I), \mathcal{M}(T), \mathcal{M}(T)^2, \ldots, \mathcal{M}(T)^m)$ for $m = 1, 2, \ldots$ until the list is linearly dependent. Then find the scalars $a_0, a_1, \ldots, a_{m-1} \in \mathbb{F}$ such that
\begin{align}
a_0 \mathcal{M}(I) + a_1 \mathcal{M}(T) + a_2 \mathcal{M}(T)^2 + \cdots + a_{m-1} \mathcal{M}(T)^{m-1} + \mathcal{M}(T)^m = 0 \nonumber
\end{align}
The scalars will be the coefficients of minimal polynomial of $T$.
\end{remark}

\begin{lemma} Let $N \in \mathcal{L}(\mathcal{V})$ be nilpotent. Then $\exists$ vectors $\mathbf{v}_1, \ldots, \mathbf{v}_k \in \mathcal{V}$ such that
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $(\mathbf{v}_1, N\mathbf{v}_1, \ldots, N^{m(\mathbf{v}_1)}\mathbf{v}_1, \ldots, \mathbf{v}_k, N\mathbf{v}_k, \ldots, N^{m(\mathbf{v}_k)}\mathbf{v}_k)$ is a basis of $\mathcal{V}$.
\item $(N^{m(\mathbf{v}_1)}\mathbf{v}_1, \ldots, N^{m(\mathbf{v}_k)}\mathbf{v}_k)$ is a basis of $\text{null}(N)$.
\end{enumerate}
\begin{proof} Proceed by induction on $\text{dim}(\mathcal{V})$. If $\text{dim}(\mathcal{V}) = 0$, the result is trivial.\\
Assume lemma holds for all nilpotent operators on smaller dim, let $N$ be a nilpotent operator on $\mathcal{V}$.\\
Since $N$ is not injective, $\text{dim}(\text{range}(N)) < \text{dim}(\mathcal{V})$. By induction hypothesis applied to $N|_{\text{range}(N)}$, $\exists$ vectors $\mathbf{u}_1, \ldots, \mathbf{u}_j \in \text{range}(N)$ such that
\begin{align}
(\mathbf{u}_1, N \mathbf{u}_1, \ldots, N^{m(\mathbf{u}_1)}\mathbf{u}_1, \ldots, \mathbf{u}_j, N\mathbf{u}_j, \ldots, N^{m(\mathbf{u}_j)}\mathbf{u}_j) \nonumber
\end{align}
is a basis of $\text{range}(N)$, and
\begin{align}
(N^{m(\mathbf{u}_1)}\mathbf{u}_1, \ldots, N^{m(\mathbf{u}_j)}\mathbf{u}_j) \nonumber
\end{align}
is a basis of $\text{null}(N) \cap \text{range}(N)$.\\
As each $\mathbf{u}_r \in \text{range}(N)$, choose $\mathbf{v}_r \in \mathcal{V}$ such that $N\mathbf{v}_r = \mathbf{u}_r$. Define $m(\mathbf{v}_r) = m(\mathbf{u}_r)+1$.\\
Let $\mathcal{W}$ be any subspace satisfying
\begin{align}
\text{null}(N) = (\text{null}(N) \cap \text{range}(N)) \oplus \mathcal{W} \nonumber
\end{align}
Choose basis $\mathbf{v}_{j+1}, \ldots, \mathbf{v}_k$ of $\mathcal{W}$. Since each $\mathbf{v}_s \in \text{null}(N)$, then $m(\mathbf{v}_s) = 0$ for $s>j$. Consider any relation
\begin{align}
0 = \sum\limits_{r=1}^{k} \sum\limits_{s=0}^{m(\mathbf{v}_r)} a_{r,s} N^s \mathbf{v}_r \nonumber
\end{align}
Apply $N$ to both sides of equation. Using $N\mathbf{v}_r = \mathbf{u}_r$ for $r \leq j$ and $N \mathbf{v}_s = 0$ for $s>j$, then
\begin{align}
0 = \sum\limits_{r=1}^{j} \sum\limits_{s=0}^{m(\mathbf{v}_r)} a_{r,s} N^s \mathbf{u}_r \nonumber
\end{align}
By inductive assumption, $(\mathbf{u}_1, N \mathbf{u}_1, \ldots, N^{m(\mathbf{u}_1)}\mathbf{u}_1, \ldots)$ is linearly independent.\\
Hence $a_{r,s} = 0$ for $1 \leq r \leq j, 0 \leq s \leq m(\mathbf{v}_r) - 1$. Equation then reduces to
\begin{align}
0 = a_{1, m(\mathbf{v}_1)} N^{m(\mathbf{v}_1)} \mathbf{v}_1 + \cdots + a_{j, m(\mathbf{v}_j)} N^{m(\mathbf{v}_j)} \mathbf{v}_j + a_{j+1, 0} \mathbf{v}_{j+1} + \cdots + a_{k,0} \mathbf{v}_k \nonumber
\end{align}
The first part lies in $\text{null}(N) \cap \text{range}(N)$, and the second lies in $\mathcal{W}$.\\
Since sum is direct, both parts must vanish, and
\begin{align}
a_{1, m(\mathbf{v}_1)} = \cdots = a_{j, m(\mathbf{v}_j)} = a_{j+1, 0} = \cdots = a_{k,0} = 0 \nonumber
\end{align}
Hence all coefficients are zero, and the list in $(i)$ is linearly independent. The number of listed vector equals
\begin{align}
\sum\limits_{r=1}^k (m(\mathbf{v}_r) + 1) = \text{dim}(\text{null}(N)) + \text{dim}(\text{range}(N)) = \text{dim}(\mathcal{V}) \nonumber
\end{align}
Hence the list in $(i)$ is a basis of $\mathcal{V}$. Finally,
\begin{align}
(N^{m(\mathbf{v}_1)}\mathbf{v}_1, \ldots, N^{m(\mathbf{v}_k)}\mathbf{v}_k) = (N^{m(\mathbf{u}_1)}\mathbf{u}_1, \ldots, N^{m(\mathbf{u}_j)}\mathbf{u}_j, \mathbf{v}_{j+1}, \ldots, \mathbf{v}_k) \nonumber
\end{align}
forms a basis of $\text{null}(N)$ by inductive hypothesis and direct-sum decomposition of $\text{null}(n)$.\\
Hence both $(i)$ and $(ii)$ hold.
\end{proof}
\end{lemma}

\begin{definition} Suppose $T \in \mathcal{L}(\mathcal{V})$. A basis of $\mathcal{V}$ is a \hlt{Jordan Basis} for $T$ if, with respect to this basis $T$ has a block diagonal matrix
\begin{align}
\begin{bmatrix}
A_1 & \cdots & 0 \\
\vdots & \ddots & \vdots \\
0 & \cdots & A_m
\end{bmatrix} \nonumber
\end{align}
where each $A_j$ is an upper-triangular matrix of the form
\begin{align}
A_j = 
\begin{bmatrix}
\lambda_j & 1 & & 0 \\
& \ddots & \ddots & \\
& & \ddots & 1 \\
0 & & & \lambda_j
\end{bmatrix} \nonumber
\end{align}
Each diagonal of $A_j$ is an eigenvalue of $\lambda_j$ of $T$, and line above diagonal is $1$, while rest of entries are $0$.
\end{definition}

\begin{theorem} Let $\mathcal{V}$ be a complex vector space, and $T \in \mathcal{L}(\mathcal{V})$. Then there exists a basis of $\mathcal{V}$ with respect to which the matrix of $T$ is in Jordan form.
\begin{proof} First, treat case where $T$ is nilpotent. Let $N \in \mathcal{L}(\mathcal{V})$ be nilpotent, so $N^m = 0$ for some $m$.\\
For each $\mathbf{v} \in \mathcal{V}$ with minimal $k$ such that $N^k \mathbf{v} = \mathbf{0}$, the sequence $\mathbf{v}, N \mathbf{v}, \ldots, N^{k-1} \mathbf{v}$ is linearly independent. Reversing the order gives the basis $N^{k-1}\mathbf{v}, \ldots, N \mathbf{v}, \mathbf{v}$ on which $N$ acts by sending each vector to the previous one, and the first to be $\mathbf{0}$. Hence relative to this basis, $N$ has a block of form
\begin{align}
\begin{bmatrix}
0 & 1 & & 0 \\
& \ddots & \ddots & \\
& & \ddots & 1 \\
0 & & & 0
\end{bmatrix} \nonumber
\end{align}
and $\mathcal{V}$ admits a direct sum decomposition into subspaces spanned by such chains. Thus $N$ has a block-diagonal matrix whose blocks are Jordan blocks with eigenvalue value $0$.\\
Now let $T \in \mathcal{L}(\mathcal{V})$ be arbitrary. Let $\lambda_1, \ldots, \lambda_m$ be distinct eigenvalues of $T$, and let $\mathcal{U}_j$ be generalised eigenspace corresponding to $\lambda_j$. Then
\begin{align}
\mathcal{V} = \mathcal{U}_1 \oplus \cdots \oplus \mathcal{U}_m \nonumber
\end{align}
and each restriction $(T - \lambda_j I)|_{\mathcal{U}_j}$ is nilpotent. By the nilpotent case, there exists a Jordan basis of $\mathcal{U}_j$ for $(T - \lambda_j I)|_{\mathcal{U}_j}$; equivalently, $T_{\mathcal{U}_j}$ has Jordan blocks with eigenvalue $\lambda_j$. Taking union of these bases yields a basis of $\mathcal{V}$ in which $T$ is Jordan form.
\end{proof}
\end{theorem}
