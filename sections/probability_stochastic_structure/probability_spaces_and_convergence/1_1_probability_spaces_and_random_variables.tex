\subsection{Probability Spaces and Random Variables}
% Probability spaces, events, random variables, distributions, pushforward measures, independence.

\begin{definition} \label{def:distribution-function-and-density} \hlt{Distribution Functions and Densities}\\
A \hlt{distribution function} is an increasing right-continuous $F:\R\to\R$ with $F(-\infty)=0$ and $F(\infty)=1$.\\
A \hlt{density} for $F$ is an integrable $g\geq0$ with $\int_{-\infty}^{\infty}g=1$ and $F(t)=\int_{-\infty}^{t}g(x)\,dx$, in which case continuous $g$ gives $F'(t)=g(t)$ by Theorem \ref{thm:integral-function-continuity-and-derivative}.
\end{definition}
