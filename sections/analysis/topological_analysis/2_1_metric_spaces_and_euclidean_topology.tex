\subsubsection{Metric Spaces and Euclidean Topology}

Metric topology replaces coordinate formulas by a distance function. Neighborhoods then define open and closed sets, limit points, closure, and relative openness through small balls.

\begin{definition} \label{def:metric-space} \hlt{Metric Space}\\
A \hlt{metric space} is a set $X$ equipped with a function $d:X\times X\to\R$ such that, for all $p,q,r\in X$,
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item $d(p,q)>0$ if $p\neq q$, and $d(p,p)=0$;
\item $d(p,q)=d(q,p)$;
\item $d(p,q)\leq d(p,r)+d(r,q)$.
\end{enumerate}
The function $d$ is called a \hlt{distance function} or \hlt{metric}.
\end{definition}

\begin{example} \label{ex:euclidean-metric-and-subspaces} \hlt{Euclidean Metric and Subspaces}\\
The Euclidean space $\R^k$ is a metric space with $d(\mathbf{x},\mathbf{y})\defeq\norm{\mathbf{x}-\mathbf{y}}_2,\qquad \mathbf{x},\mathbf{y}\in\R^k$.\\
Every subset $Y$ of a metric space $X$ is a metric space by restricting the same distance function to points of $Y$.
\end{example}

\begin{definition} \label{def:euclidean-intervals-cells-balls-convex} \hlt{Euclidean Intervals, Cells, Balls, and Convexity}\\
In $\R$, $(a,b)\defeq\{x\in\R:a<x<b\}$, $[a,b]\defeq\{x\in\R:a\leq x\leq b\}$, with $[a,b)$ and $(a,b]$ as mixed inequalities.\\
If $a_i<b_i$ for $i=1,\ldots,k$, the \hlt{$k$-cell} $\{\mathbf{x}=(x_1,\ldots,x_k)\in\R^k:a_i\leq x_i\leq b_i,\ 1\leq i\leq k\}$ is a coordinate box.\\
For $\mathbf{x}\in\R^k$, $r>0$, \hlt{open ball} and \hlt{closed ball} is $B_r(\mathbf{x})\defeq\{\mathbf{y}\in\R^k:\norm{\mathbf{y}-\mathbf{x}}_2<r\}$, $\overline{B}_r(\mathbf{x})\defeq\{\mathbf{y}\in\R^k:\norm{\mathbf{y}-\mathbf{x}}_2$.\\
A set $E\subseteq\R^k$ is \hlt{convex} if $\lambda\mathbf{x}+(1-\lambda)\mathbf{y}\in E$ whenever $\mathbf{x},\mathbf{y}\in E$ and $0<\lambda<1$.
\end{definition}

\begin{proposition} \label{prop:balls-and-cells-convex} \hlt{Balls and Cells Are Convex}\\
Open balls, closed balls, and $k$-cells in $\R^k$ are convex.
\begin{proof}
Let $\mathbf{y},\mathbf{z}\in B_r(\mathbf{x})$ and $0<\lambda<1$. Then
\begin{align}
\norm{\lambda\mathbf{y}+(1-\lambda)\mathbf{z}-\mathbf{x}}_2
&=\norm{\lambda(\mathbf{y}-\mathbf{x})+(1-\lambda)(\mathbf{z}-\mathbf{x})}_2 \nonumber\\
&\leq \lambda\norm{\mathbf{y}-\mathbf{x}}_2+(1-\lambda)\norm{\mathbf{z}-\mathbf{x}}_2<r. \nonumber
\end{align}
The same proof with $\leq r$ gives closed balls. For a $k$-cell, each coordinate of $\lambda\mathbf{y}+(1-\lambda)\mathbf{z}$ lies between the corresponding coordinates of $\mathbf{y}$ and $\mathbf{z}$, hence remains in the same interval $[a_i,b_i]$.\\
\hlt{Structural Commentary}: Convexity is checked by following the line segment between two points.\\
\hlt{Key Insight}: Balls use the triangle inequality; cells use coordinate inequalities.
\end{proof}
\end{proposition}

\begin{definition} \label{def:metric-topology-core-definitions}\label{def:metric-topology-core-definitions-source-todo} \hlt{Metric Topology Terminology}\\
Let $X$ be a metric space, and let all points and sets below be understood as points of $X$ and subsets of $X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item A \hlt{neighborhood} of $p$ with radius $r>0$ is $B_r(p)\defeq\{q\in X:d(p,q)<r\}$.
\item A point $p$ is a \hlt{limit point} of $E$ if every neighborhood of $p$ contains a point $q\neq p$ with $q\in E$.
\item If $p\in E$ and $p$ is not a limit point of $E$, then $p$ is an \hlt{isolated point} of $E$.
\item A set $E$ is \hlt{closed} if every limit point of $E$ is a point of $E$.
\item A point $p$ is an \hlt{interior point} of $E$ if some neighborhood of $p$ is contained in $E$.
\item A set $E$ is \hlt{open} if every point of $E$ is an interior point of $E$.
\item The \hlt{complement} of $E$ is $E^c\defeq X\setminus E$.
\item A set $E$ is \hlt{dense} in $X$ if every point of $X$ is either a limit point of $E$, a point of $E$, or both.
\end{enumerate}

\end{definition}

\begin{theorem} \label{thm:every-neighborhood-open} \hlt{Every Neighborhood Is Open}\\
Every neighborhood in a metric space is an open set.
\begin{proof}
Let $E=B_r(p)$, and choose $q\in E$. Put $h\defeq r-d(p,q)>0$. If $s\in B_h(q)$, then
\begin{align}
d(p,s)\leq d(p,q)+d(q,s)<d(p,q)+h=r. \nonumber
\end{align}
Thus $B_h(q)\subseteq E$, so $q$ is an interior point of $E$. Since $q$ was arbitrary, $E$ is open.\\
\hlt{Structural Commentary}: Around any point inside a ball, the leftover distance to the boundary gives a smaller ball.\\
\hlt{Key Insight}: The triangle inequality turns radius slack into openness.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:limit-point-neighborhoods-infinitely-many} \hlt{Limit Points Force Infinitely Many Nearby Points}\\
If $p$ is a limit point of $E$, then every neighborhood of $p$ contains infinitely many points of $E$.
\begin{proof}
Suppose some neighborhood $N$ of $p$ contains only finitely many points of $E$ distinct from $p$, say $q_1,\ldots,q_n$. If there are none, then $N$ itself contradicts that $p$ is a limit point. Otherwise put
\begin{align}
\rho\defeq\min_{1\leq m\leq n}d(p,q_m)>0. \nonumber
\end{align}
The neighborhood $B_\rho(p)$ contains no point $q\in E$ with $q\neq p$, contradicting the definition of limit point.\\
\hlt{Structural Commentary}: A finite list of nearby exceptions has a smallest positive distance from $p$.\\
\hlt{Key Insight}: Limit points cannot be witnessed by only finitely many neighboring points.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:finite-point-set-no-limit-points} \hlt{Finite Point Sets Have No Limit Points}\\
A finite point set in a metric space has no limit points.
\begin{proof}
If a finite set had a limit point, Theorem \ref{thm:limit-point-neighborhoods-infinitely-many} would force every neighborhood of that point to contain infinitely many points of the finite set.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:metric-de-morgan-laws} \hlt{Arbitrary De Morgan Laws}\\
Let $X$ be a set. For any collection $\{E_\alpha\}$ of subsets of $X$,
\begin{align}
\left(\bigcup_\alpha E_\alpha\right)^c=\bigcap_\alpha E_\alpha^c,\qquad
\left(\bigcap_\alpha E_\alpha\right)^c=\bigcup_\alpha E_\alpha^c. \nonumber
\end{align}
\begin{proof}
The first identity follows directly from membership:
\begin{align}
x\in\left(\bigcup_\alpha E_\alpha\right)^c
\Longleftrightarrow x\notin E_\alpha\text{ for every }\alpha
\Longleftrightarrow x\in\bigcap_\alpha E_\alpha^c. \nonumber
\end{align}
The second follows by applying the first identity to the complements $E_\alpha^c$, or by the same membership argument.\\
\hlt{Structural Commentary}: Complements exchange unions and intersections.\\
\hlt{Key Insight}: Negating ``for some'' gives ``for every'', and conversely.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:open-iff-complement-closed} \hlt{Open Sets and Closed Complements}\\
A set $E$ is open if and only if its complement $E^c$ is closed.
\begin{proof}
Suppose first that $E^c$ is closed. If $x\in E$, then $x\notin E^c$, so $x$ is not a limit point of $E^c$. Hence some neighborhood $N$ of $x$ has $N\cap E^c=\emptyset$, and therefore $N\subseteq E$. Thus $E$ is open.

Conversely, suppose $E$ is open and let $x$ be a limit point of $E^c$. If $x\in E$, then some neighborhood of $x$ is contained in $E$, contradicting that every neighborhood of $x$ meets $E^c$ away from $x$. Hence $x\in E^c$, so $E^c$ is closed.\\
\hlt{Structural Commentary}: Openness of a set is equivalent to closure of the outside.\\
\hlt{Key Insight}: Interior points of $E$ cannot be limit points of $E^c$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:closed-iff-complement-open} \hlt{Closed Sets and Open Complements}\\
A set $F$ is closed if and only if $F^c$ is open.
\begin{proof}
Apply Theorem \ref{thm:open-iff-complement-closed} to $E=F^c$.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:metric-open-closed-set-calculus}\label{prop:metric-topology-basic-calculus-source-todo} \hlt{Basic Topological Calculus}\\
In a metric space:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item any union of open sets is open;
\item any intersection of closed sets is closed;
\item any finite intersection of open sets is open;
\item any finite union of closed sets is closed.
\end{enumerate}
\begin{proof}
For (i), if $x\in\bigcup_\alpha G_\alpha$, then $x\in G_\alpha$ for some $\alpha$. Since $G_\alpha$ is open, some neighborhood of $x$ lies in $G_\alpha$, hence in the union.

For (ii), take complements and use Theorem \ref{thm:metric-de-morgan-laws}, Theorem \ref{thm:open-iff-complement-closed}, and (i):
\begin{align}
\left(\bigcap_\alpha F_\alpha\right)^c=\bigcup_\alpha F_\alpha^c. \nonumber
\end{align}
For (iii), let $H=\bigcap_{i=1}^nG_i$ and choose $x\in H$. For each $i$, choose $r_i>0$ with $B_{r_i}(x)\subseteq G_i$. Put $r=\min(r_1,\ldots,r_n)$. Then $B_r(x)\subseteq H$, so $H$ is open.

Finally, (iv) follows by taking complements of (iii):
\begin{align}
\left(\bigcup_{i=1}^nF_i\right)^c=\bigcap_{i=1}^nF_i^c. \nonumber
\end{align}
\hlt{Structural Commentary}: Arbitrary unions are easy for open sets; finite intersections require a minimum radius.\\
\hlt{Key Insight}: The finite minimum is the reason infinite intersections of open sets need not be open.
\end{proof}
\end{theorem}

\begin{definition} \label{def:metric-closure} \hlt{Closure}\\
If $E\subseteq X$ and $E'$ denotes the set of all limit points of $E$ in $X$, then the \hlt{closure} of $E$ is
\begin{align}
\overline{E}\defeq E\cup E'. \nonumber
\end{align}
\end{definition}

\begin{theorem} \label{thm:closure-smallest-closed-containing-set} \hlt{Closure Is the Smallest Closed Superset}\\
If $X$ is a metric space and $E\subseteq X$, then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\overline{E}$ is closed;
\item $\overline{E}=E$ if and only if $E$ is closed;
\item $\overline{E}\subseteq F$ for every closed set $F\subseteq X$ such that $E\subseteq F$.
\end{enumerate}
Thus $\overline{E}$ is the smallest closed subset of $X$ that contains $E$.
\begin{proof}
For (i), suppose $p\notin\overline{E}$. Then $p\notin E$ and $p$ is not a limit point of $E$, so some neighborhood $B_r(p)$ is disjoint from $E$. If $q\in B_{r/2}(p)$, then $B_{r-d(p,q)}(q)\subseteq B_r(p)$, so $q$ is not a limit point of $E$ and $q\notin E$. Hence $B_{r/2}(p)\subseteq(\overline{E})^c$, so $(\overline{E})^c$ is open and $\overline{E}$ is closed by Corollary \ref{cor:closed-iff-complement-open}.

For (ii), if $\overline{E}=E$, then (i) says $E$ is closed. If $E$ is closed, then every limit point of $E$ lies in $E$, so $E'\subseteq E$ and $\overline{E}=E$.

For (iii), if $F$ is closed and $E\subseteq F$, then every limit point of $E$ is a limit point of $F$, hence lies in $F$. Thus $E\cup E'\subseteq F$.\\
\hlt{Structural Commentary}: Closure adds exactly the limit points forced by $E$.\\
\hlt{Key Insight}: Closed supersets must contain all limit points, so they must contain $\overline{E}$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:supremum-belongs-to-closure} \hlt{Suprema Belong to Closures}\\
Let $E$ be a nonempty set of real numbers that is bounded above, and let $y=\sup E$. Then $y\in\overline{E}$. Hence $y\in E$ if $E$ is closed.
\begin{proof}
If $y\in E$, then $y\in\overline{E}$. Assume $y\notin E$. For every $h>0$, the number $y-h$ is not an upper bound of $E$, so there exists $x\in E$ such that $y-h<x\leq y$. Since $y\notin E$, this gives $x\neq y$ and $\abs{x-y}<h$. Thus every neighborhood of $y$ contains a point of $E$ different from $y$, so $y$ is a limit point of $E$ and $y\in\overline{E}$. If $E$ is closed, Theorem \ref{thm:closure-smallest-closed-containing-set}(ii) gives $\overline{E}=E$.\\
\hlt{Structural Commentary}: The least-upper-bound property produces points of $E$ arbitrarily close below the supremum.\\
\hlt{Key Insight}: A supremum is either attained or approached.
\end{proof}
\end{theorem}

\begin{definition} \label{def:relative-open-subset} \hlt{Relative Openness}\\
Suppose $E\subseteq Y\subseteq X$, where $X$ is a metric space. The set $E$ is \hlt{open relative to $Y$} if for each $p\in E$ there is an $r>0$ such that $q\in E$ whenever $q\in Y$ and $d(p,q)<r$.
\end{definition}

\begin{theorem} \label{thm:relative-open-intersection-with-open-set} \hlt{Relative Opens Are Intersections}\\
Suppose $Y\subseteq X$. A subset $E$ of $Y$ is open relative to $Y$ if and only if
\begin{align}
E=Y\cap G \nonumber
\end{align}
for some open subset $G$ of $X$.
\begin{proof}
Suppose $E$ is open relative to $Y$. For each $p\in E$, choose $r_p>0$ such that $q\in Y$ and $d(p,q)<r_p$ imply $q\in E$. Let $V_p=B_{r_p}(p)$ in $X$, and set
\begin{align}
G\defeq\bigcup_{p\in E}V_p. \nonumber
\end{align}
By Theorem \ref{thm:every-neighborhood-open} and Theorem \ref{thm:metric-open-closed-set-calculus}, $G$ is open in $X$. Since $p\in V_p$ for each $p\in E$, we have $E\subseteq G\cap Y$. Conversely, if $q\in G\cap Y$, then $q\in V_p$ for some $p\in E$, and the choice of $V_p$ gives $q\in E$. Thus $E=G\cap Y$.

Conversely, suppose $G$ is open in $X$ and $E=G\cap Y$. If $p\in E$, choose $r>0$ with $B_r(p)\subseteq G$. Then $B_r(p)\cap Y\subseteq G\cap Y=E$, so $E$ is open relative to $Y$.\\
\hlt{Structural Commentary}: Relative openness is ordinary openness viewed through the ambient subset $Y$.\\
\hlt{Key Insight}: Intersect open sets in $X$ with $Y$ to get exactly the open sets of $Y$.
\end{proof}
\end{theorem}

\begin{method} \label{meth:metric-topology-proof-patterns}\label{meth:metric-topology-proof-patterns-source-todo} \hlt{Metric Proof Patterns}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item To prove openness, start with an arbitrary point and manufacture a radius.
\item To prove closedness, either keep all limit points or prove the complement is open.
\item To prove a closure claim, show both that the candidate is closed and that every closed superset must contain it.
\end{enumerate}
\end{method}
