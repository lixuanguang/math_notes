This subsection treats continuity as structure preservation: function limits describe local behavior, continuity matches the limit to the function value, and continuous maps preserve compactness and connectedness through inverse images. The sequence machinery from Definition \ref{def:sequence-convergence-metric}, compactness from Definition \ref{def:open-cover-and-compactness}, and connectedness from Definition \ref{def:separated-and-connected-sets} are used repeatedly below.

\begin{definition} \label{def:function-limit-metric}\label{def:continuity-core-definitions-source-todo} \hlt{Limit of a Function Between Metric Spaces}\\
Let $X,Y$ be metric spaces, $E\subseteq X$, $f:E\to Y$, and let $p$ be a limit point of $E$. For $q\in Y$, $\lim_{x\to p} f(x)=q$ means that, for every $\varepsilon>0$, there exists $\delta>0$ such that $x\in E$, $0<d_X(x,p)<\delta$ $\Longrightarrow$ $d_Y(f(x),q)<\varepsilon$.\\
The point $p$ need not belong to $E$, and $f(p)$ is irrelevant when it does.\\
\hlt{Key Insight}: Domain $E$ controls approach: $J\setminus\{a\}$ gives two-sided limit,  $(a,b)$ or $(c,a)$ gives one-sided limits.
\end{definition}

\begin{theorem} \label{thm:function-limit-sequential-characterization} \hlt{Sequential Characterization of Function Limits}\\
Let $X,Y$ be metric spaces, $E\subseteq X$, $f:E\to Y$, and let $p$ be a limit point of $E$. For $q\in Y$, $\lim_{x\to p}f(x)=q$ if and only if $f(p_n)\to q$ for every sequence $(p_n)\subseteq E\setminus\{p\}$ with $p_n\to p$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $p_n\to p$ with $p_n\in E\setminus\{p\}$, then eventually $0<d_X(p_n,p)<\delta$; hence $d_Y(f(p_n),q)<\varepsilon$, so $f(p_n)\to q$.
\item[$(\Leftarrow)$] If the limit fails, then for some $\varepsilon>0$ and each $n$ there is $x_n\in E$ with $0<d_X(x_n,p)<1/n$ and $d_Y(f(x_n),q)\geq\varepsilon$. Then $x_n\to p$, $x_n\neq p$, but $f(x_n)\not\to q$.
\end{enumerate}
\hlt{Structural Commentary}: Function limits can be tested by all sequences approaching the input point.\\
\hlt{Key Insight}: The negation of an epsilon--delta limit manufactures a bad sequence.
\end{proof}
\end{theorem}

\begin{principle} \label{pri:continuity-challenge-response} \hlt{Challenge--Response Order}\\
Limits and continuity are controlled by quantifier order. In Definition \ref{def:function-limit-metric}, the $\varepsilon$-challenge at the range point must be answered by a $\delta$-restriction near the domain point. Uniform continuity, Definition \ref{def:uniform-continuity}, strengthens the same pattern by requiring one $\delta$ to work for all relevant domain points at once.
\end{principle}

\begin{corollary} \label{cor:function-limits-are-unique} \hlt{Uniqueness of Function Limits}\\
If a function has a limit at $p$, then that limit is unique.
\begin{proof}
Let $q$ and $r$ be limits of $f$ at $p$. Choose $p_n\in E\setminus\{p\}$ with $d_X(p_n,p)<1/n$. Then $p_n\to p$, so Theorem \ref{thm:function-limit-sequential-characterization} gives $f(p_n)\to q$ and $f(p_n)\to r$. By Theorem \ref{thm:metric-sequence-basic-properties}, $q=r$.\\
\hlt{Structural Commentary}: Limit uniqueness is inherited from metric sequence uniqueness.\\
\hlt{Key Insight}: One approaching sequence cannot have two different limits.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:function-limit-algebra} \hlt{Algebra of Function Limits}\\
Let $X$ be a metric space, $E\subseteq X$, $p$ be limit point of $E$.\\
For $f,g:E\to\C$, assume $\lim_{x\to p}f(x)=A$ and $\lim_{x\to p}g(x)=B$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\lim\limits_{x\to p}(f+g)(x)=A+B$;
\item $\lim\limits_{x\to p}(fg)(x)=AB$;
\item if $B\neq0$ and $f/g$ is defined near $p$, then $\lim\limits_{x\to p}(f/g)(x)=A/B$.
\end{enumerate}
For $f,g:E\to\R^k$ with limits $a,b$, the corresponding limits are $f+g\to a+b$, $\lambda f\to\lambda a$, and $f\cdot g\to a\cdot b$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For any $p_n\in E\setminus\{p\}$ with $p_n\to p$, Theorem \ref{thm:function-limit-sequential-characterization} gives $f(p_n)\to A$ and $g(p_n)\to B$.\\
Hence $(f+g)(p_n)\to A+B$ by Theorem \ref{thm:complex-sequence-limit-algebra}, so Theorem \ref{thm:function-limit-sequential-characterization} gives (i).
\item The same sequence gives $(fg)(p_n)\to AB$ by Theorem \ref{thm:complex-sequence-limit-algebra}; hence (ii).
\item If $B\neq0$ and $f/g$ is defined near $p$, then every such sequence is eventually in the domain of $f/g$, and $(f/g)(p_n)\to A/B$ by Theorem \ref{thm:complex-sequence-limit-algebra}; hence (iii).
\end{enumerate}
The vector laws follow by the same sequential characterization and Theorem \ref{thm:rk-sequence-coordinate-convergence}.\\
\hlt{Structural Commentary}: Limit algebra for functions is pulled back to sequence algebra.\\
\hlt{Key Insight}: Sequential characterization lets previous sequence theorems do the work.
\end{proof}
\end{theorem}

\begin{definition} \label{def:continuity-at-a-point-and-on-a-set} \hlt{Continuity at a Point and on a Set}\\
For metric spaces $X,Y$, $E\subseteq X$, $p\in E$, and $f:E\to Y$, the function $f$ is \hlt{continuous at $p$} if $\forall \varepsilon>0\ \exists \delta>0$, $x\in E$, $d_X(x,p)<\delta$ $\Longrightarrow$ $d_Y(f(x),f(p))<\varepsilon$.\\
It is \hlt{continuous on $E$} if continuous at every point of $E$; isolated points are automatic.
\end{definition}

\begin{theorem} \label{thm:continuity-as-limit-at-limit-point} \hlt{Continuity as Equality with the Function Limit}\\
Let $X,Y$ be metric spaces, $E\subseteq X$, $p\in E$, and $f:E\to Y$.\\
If $p$ is a limit point of $E$, then $f$ is continuous at $p$ if and only if $\lim_{x\to p}f(x)=f(p)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $0<d_X(x,p)<\delta$, continuity at $p$ gives $d_Y(f(x),f(p))<\varepsilon$, hence $\lim_{x\to p}f(x)=f(p)$.
\item[$(\Leftarrow)$] If $\lim_{x\to p}f(x)=f(p)$, the same $\delta$ proves continuity for $x\neq p$; for $x=p$, $d_Y(f(p),f(p))=0$.
\end{enumerate}
\hlt{Structural Commentary}: Continuity is the compatibility of the value at $p$ with nearby values.\\
\hlt{Key Insight}: Limits ignore the point; continuity ties the point back in.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:sequential-continuity-real-line} \hlt{The Sequence-First View}\\
For functions on subsets of $\R$, continuity at limit points is equivalently sequential: $x_n\to p$ in the domain implies $f(x_n)\to f(p)$; isolated points are automatic by Theorems \ref{thm:function-limit-sequential-characterization} and \ref{thm:continuity-as-limit-at-limit-point}.
\end{remark}

\begin{theorem} \label{thm:continuous-composition} \hlt{Composition of Continuous Functions}\\
Let $E\subseteq X$ and let $f:E\to Y$, $g:f(E)\to Z$ be maps between metric spaces.\\
If $f$ is continuous at $p\in E$ and $g$ is continuous at $f(p)$, then $g\circ f$ is continuous at $p$.
\begin{proof}
Given $\varepsilon>0$, choose $\eta>0$ for $g$ at $f(p)$, then choose $\delta>0$ for $f$ at $p$ with tolerance $\eta$.\\
If $x\in E$ and $d_X(x,p)<\delta$, then $d_Y(f(x),f(p))<\eta$, hence $d_Z(g(f(x)),g(f(p)))<\varepsilon$.\\
\hlt{Structural Commentary}: The output tolerance for $g$ becomes the input tolerance for $f$.\\
\hlt{Key Insight}: Continuity composes by passing epsilon through an intermediate eta.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:continuity-open-preimages}\label{thm:continuity-structure-results-source-todo} \hlt{Continuity via Open Preimages}\\
For metric spaces $X,Y$, map $f:X\to Y$ is continuous if and only if $f^{-1}(V)$ is open in $X \forall$ open $V\subseteq Y$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $p\in f^{-1}(V)$ and $V$ is open, choose $\varepsilon>0$ with $B_\varepsilon(f(p))\subseteq V$.\\
Continuity gives $\delta>0$ with $B_\delta(p)\subseteq f^{-1}(V)$, so $f^{-1}(V)$ is open.
\item[$(\Leftarrow)$] For $p\in X$ and $\varepsilon>0$, the ball $V=B_\varepsilon(f(p))$ is open by Theorem \ref{thm:every-neighborhood-open}.\\
Since $p\in f^{-1}(V)$ and $f^{-1}(V)$ is open, some $B_\delta(p)\subseteq f^{-1}(V)$; this is continuity at $p$.
\end{enumerate}
\hlt{Structural Commentary}: Continuity is exactly preservation of openness under inverse image.\\
\hlt{Key Insight}: Images are not Boolean-exact, but preimages are; continuity belongs to preimages.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:continuity-closed-preimages} \hlt{Continuity via Closed Preimages}\\
For metric spaces $X,Y$, map $f:X\to Y$ is continuous if and only if $f^{-1}(C)$ is closed in $X \forall$ closed $C\subseteq Y$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $C$ is closed, then $Y\setminus C$ is open. By Theorem \ref{thm:continuity-open-preimages}, $f^{-1}(Y\setminus C)$ is open; by Proposition \ref{prop:image-preimage-calculus}, this is $X\setminus f^{-1}(C)$, so $f^{-1}(C)$ is closed.
\item[$(\Leftarrow)$] If $V$ is open, then $Y\setminus V$ is closed. Hence $f^{-1}(Y\setminus V)$ is closed, so $f^{-1}(V)=X\setminus f^{-1}(Y\setminus V)$ is open. Theorem \ref{thm:continuity-open-preimages} gives continuity.
\end{enumerate}
\hlt{Structural Commentary}: Closed-set continuity is the complement form of open-set continuity.\\
\hlt{Key Insight}: $f^{-1}(Y\setminus C)=X\setminus f^{-1}(C)$ transfers open and closed formulations.
\end{proof}
\end{corollary}

\begin{definition} \label{def:homeomorphism} \hlt{Homeomorphism}\\
A \hlt{homeomorphism} from $X$ to $Y$ is a bijection $f:X\to Y$ such that both $f$ and $f^{-1}$ are continuous.\\
If such a map exists, $X$ and $Y$ are \hlt{homeomorphic}, written $X\simeq Y$.
\end{definition}

\begin{proposition} \label{prop:homeomorphisms-preserve-topology} \hlt{Homeomorphisms Preserve Topology}\\
If $f:X\to Y$ is a homeomorphism, then $U\mapsto f(U)$ bijects open subsets of $X$ with open subsets of $Y$.
\begin{proof}
If $U\subseteq X$ is open, then $f(U)=(f^{-1})^{-1}(U)$ is open in $Y$ since $f^{-1}$ is continuous. If $V\subseteq Y$ is open, then $f^{-1}(V)$ is open in $X$ since $f$ is continuous. These two assignments are inverse because $f$ is bijective.\\
\hlt{Structural Commentary}: A homeomorphism preserves open sets in both directions.\\
\hlt{Key Insight}: Topological sameness is continuity plus continuous reversibility.
\end{proof}
\end{proposition}

\begin{example} \label{ex:circle-param-continuous-bijection-not-homeomorphism} \hlt{Continuous Bijection Need Not Be a Homeomorphism}\\
Let $f:[0,2\pi)\to S^1$, and $f(t)=(\cos t,\sin t)$. This continuous bijection is not a homeomorphism: points of $S^1$ approaching $(1,0)$ from below have preimages tending to $2\pi$, not $0$.
\end{example}

\begin{remark} \label{rem:continuity-domain-only} \hlt{Continuity Only Uses the Domain Metric}\\
For $f:E\to Y$ with $E\subseteq X$, continuity on $E$ depends only on the restricted metric on $E$, not on $X\setminus E$.
\end{remark}

\begin{theorem} \label{thm:continuous-algebra-and-components} \hlt{Continuous Algebra and Components}\\
Let $X$ be a metric space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $f,g:X\to\C$ are continuous, so are $f+g$, $fg$, and $f/g$ where $g\neq0$.
\item For $f=(f_1,\ldots,f_k):X\to\R^k$, $f$ is continuous if and only if each $f_j$ is continuous.
\item If $f,g:X\to\R^k$ are continuous, so are $f+g$ and $f\cdot g$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item At isolated points continuity is automatic; at limit points, use Theorems \ref{thm:function-limit-algebra} and \ref{thm:continuity-as-limit-at-limit-point}.
\item By Theorem \ref{thm:rk-sequence-coordinate-convergence}, convergence in $\R^k$ is exactly coordinatewise convergence; apply this to $f(x_n)\to f(x)$.
\item Apply (ii) to the coordinate functions of $f+g$ and $f\cdot g$, using the scalar algebra from (i).
\end{enumerate}
\hlt{Structural Commentary}: Continuity is stable under the algebraic operations already stable for limits.\\
\hlt{Key Insight}: Componentwise control is equivalent to Euclidean control in finite dimensions.
\end{proof}
\end{theorem}

\begin{example} \label{ex:coordinate-polynomial-rational-norm-continuity} \hlt{Coordinate, Polynomial, Rational, and Norm Maps}\\
Coordinate functions on $\R^k$ are continuous; hence polynomials are continuous, and rational functions are continuous where defined. Also $\mathbf{x}\mapsto\norm{\mathbf{x}}_2$ is continuous since $\bigl|\norm{\mathbf{x}}_2-\norm{\mathbf{y}}_2\bigr|\leq\norm{\mathbf{x}-\mathbf{y}}_2$.\\
Thus $p\mapsto\norm{f(p)}_2$ is continuous whenever $f:X\to\R^k$ is, by Theorem \ref{thm:continuous-composition}.
\end{example}

\begin{definition} \label{def:bounded-mapping-into-rk} \hlt{Bounded Mapping}\\
A mapping $f:E\to\R^k$ is \hlt{bounded} if there exists $M\in\R$ such that $\norm{f(x)}_2\leq M$ for every $x\in E$.
\end{definition}

\begin{theorem} \label{thm:continuous-image-compact} \hlt{Continuous Images of Compact Sets Are Compact}\\
If $f:X\to Y$ is continuous and $X$ is compact, then $f(X)$ is compact.
\begin{proof}
Let $\{V_\alpha\}$ cover $f(X)$. Then $\{f^{-1}(V_\alpha)\}$ is an open cover of $X$ by Theorem \ref{thm:continuity-open-preimages}.\\
Choose $\alpha_1,\ldots,\alpha_n$ with $X\subseteq\bigcup_{i=1}^n f^{-1}(V_{\alpha_i})$; then $f(X)\subseteq\bigcup_{i=1}^n V_{\alpha_i}$.\\
\hlt{Structural Commentary}: Pull back, take a finite subcover, push forward.\\
\hlt{Key Insight}: Compactness is preserved by continuous images because inverse images preserve openness.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:continuous-rk-image-compact-closed-bounded} \hlt{Continuous Images into $\R^k$ Are Closed and Bounded}\\
If $f:X\to\R^k$ is continuous and $X$ is compact, then $f(X)$ is closed and bounded. In particular, $f$ is bounded.
\begin{proof}
By Theorem \ref{thm:continuous-image-compact}, $f(X)$ is compact.\\
By Theorem \ref{thm:heine-borel-rk}, compact subsets of $\R^k$ are closed and bounded; hence $f$ is bounded.\\
\hlt{Structural Commentary}: Compactness in the domain becomes closed-bounded geometry in Euclidean range.\\
\hlt{Key Insight}: Continuous functions on compact spaces cannot escape to infinity.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:extreme-value-theorem-compact} \hlt{Extreme Value Theorem}\\
Let $f:X\to\R$ be continuous, let $X$ be compact.\\
If $M\defeq\sup_{p\in X}f(p)$, $m\defeq\inf_{p\in X}f(p)$, then $\exists$ $p,q\in X$ such that $f(p)=M$ and $f(q)=m$.
\begin{proof}
By Corollary \ref{cor:continuous-rk-image-compact-closed-bounded}, $f(X)$ is closed and bounded in $\R$. Thus $\sup f(X)$ and $\inf f(X)$ exist and belong to $f(X)$ by Theorems \ref{thm:supremum-belongs-to-closure} and \ref{thm:lub-property-gives-glb-property}. Hence $M=f(p)$ and $m=f(q)$ for some $p,q\in X$.\\
\hlt{Structural Commentary}: Compactness turns suprema and infima into attained values.\\
\hlt{Key Insight}: Closed bounded images contain their endpoints.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:compact-bijection-has-continuous-inverse} \hlt{Compact Bijective Maps Have Continuous Inverses}\\
If $f:X\to Y$ is a continuous bijection from compact $X$ to a metric space $Y$, then $f^{-1}:Y\to X$ is continuous.
\begin{proof}
It suffices, by Theorem \ref{thm:continuity-open-preimages}, to show that $f(V)$ is open in $Y$ for every open $V\subseteq X$. If $V$ is open, then $V^c$ is compact by Theorem \ref{thm:closed-subsets-of-compact-sets-compact}, so $f(V^c)$ is compact, hence closed by Theorems \ref{thm:continuous-image-compact} and \ref{thm:compact-subsets-of-metric-spaces-closed}.\\
Since $f$ is bijective, $f(V)=Y\setminus f(V^c)$ is open.\\
\hlt{Structural Commentary}: Compactness turns closed complements into closed images.\\
\hlt{Key Insight}: A continuous bijection from compact domain has no room for inverse discontinuity.
\end{proof}
\end{theorem}

\begin{definition} \label{def:uniform-continuity} \hlt{Uniform Continuity}\\
A mapping $f:X\to Y$ between metric spaces is \hlt{uniformly continuous} on $X$ if $\forall$ $\varepsilon>0$ $\exists$ $\delta>0$ such that $d_X(p,q)<\delta$ $\Longrightarrow$ $d_Y(f(p),f(q))<\varepsilon$ $\forall p,q\in X$. The same $\delta$ must work for every point of $X$.\\
\hlt{Structural Commentary}: Uniform continuity is setwise: for each $\varepsilon$, one $\delta$ must work for all pairs in the set.
\end{definition}

\begin{theorem} \label{thm:compact-continuity-uniform-continuity} \hlt{Continuity on Compact Sets Is Uniform}\\
If $f:X\to Y$ is continuous and $X$ is compact, then $f$ is uniformly continuous.
\begin{proof}
Let $\varepsilon>0$. For each $a\in X$, choose $r_a>0$ such that $d_X(x,a)<r_a$ implies $d_Y(f(x),f(a))<\varepsilon/2$.\\
Compactness gives a finite cover $X\subseteq\bigcup_{i=1}^n B_{r_{a_i}/2}(a_i)$. Set $\delta\defeq\frac12\min_i r_{a_i}>0$. If $d_X(p,q)<\delta$ and $p\in B_{r_{a_i}/2}(a_i)$, then $d_X(q,a_i)<r_{a_i}$ also; hence $f(p)$ and $f(q)$ are each within $\varepsilon/2$ of $f(a_i)$. Thus $d_Y(f(p),f(q))<\varepsilon$.\\
\hlt{Structural Commentary}: Compactness converts pointwise radii into finitely many radii with positive minimum.\\
\hlt{Key Insight}: Uniform continuity is continuity with the point-dependence removed by a finite subcover.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:compactness-hypotheses-are-essential} \hlt{Compactness Hypotheses Are Essential}\\
Let $E$ be a noncompact subset of $\R$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item there exists a continuous function on $E$ which is not bounded;
\item there exists a continuous bounded function on $E$ which has no maximum;
\item if $E$ is bounded, then there exists a continuous function on $E$ which is not uniformly continuous.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $E$ is unbounded, $f(x)=x$ works. If $E$ is bounded, Theorem \ref{thm:heine-borel-rk} gives $x_0\in\overline{E}\setminus E$, and $f(x)=1/(x-x_0)$ is continuous on $E$ and unbounded.
\item If $E$ is unbounded, $h(x)=x^2/(1+x^2)$ has $\sup_E h=1$ but $h<1$. If $E$ is bounded, with $x_0$ as above, $g(x)=1/(1+(x-x_0)^2)$ is bounded by $1$ and has supremum $1$ but no maximum.
\item If $E$ is bounded, let $f(x)=1/(x-x_0)$. Given $\delta>0$, let $x\in E$ with $\abs{x-x_0}<\delta/2$, then $y\in E$ with $\abs{y-x_0}<\min\{\delta/2,1/(\abs{f(x)}+1)\}$. Then $\abs{x-y}<\delta$, $\abs{f(x)-f(y)}>1$, so $f$ is not uniformly continuous.
\end{enumerate}
\hlt{Structural Commentary}: Compactness is essential in the preceding theorems.\\
\hlt{Key Insight}: Noncompact sets either escape to infinity or miss an accumulation point.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:continuous-image-connected} \hlt{Continuous Images of Connected Sets Are Connected}\\
If $f:X\to Y$ is continuous and $E\subseteq X$ is connected, then $f(E)$ is connected.
\begin{proof}
If $f(E)=A\cup B$ with $A,B$ nonempty separated, set $G\defeq E\cap f^{-1}(A)$ and $H\defeq E\cap f^{-1}(B)$. Then $E=G\cup H$ and $G,H$ are nonempty. Since $G\subseteq f^{-1}(\overline{A})$ and $f^{-1}(\overline{A})$ is closed, $\overline{G}\subseteq f^{-1}(\overline{A})$; hence $\overline{G}\cap H=\emptyset$. Similarly $G\cap\overline{H}=\emptyset$, so $G,H$ separate $E$, a contradiction.\\
\hlt{Structural Commentary}: A separation of the image pulls back to a separation of the domain.\\
\hlt{Key Insight}: Continuous maps cannot create a gap in the image of a connected set.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:intermediate-value-theorem} \hlt{Intermediate Value Theorem}\\
If $f:[a,b]\to\R$ is continuous and $c$ lies strictly between $f(a)$ and $f(b)$, then $f(x)=c$ for some $x\in(a,b)$.
\begin{proof}
The interval $[a,b]$ is connected by Theorem \ref{thm:connected-subsets-of-real-line-interval-property}, so $f([a,b])$ is connected by Theorem \ref{thm:continuous-image-connected}.\\
Connected subsets of $\R$ contain every value between any two of their points; hence $c=f(x)$ for some $x\in[a,b]$. Since $c\neq f(a),f(b)$, $x\in(a,b)$.\\
\hlt{Structural Commentary}: The intermediate value theorem is connectedness of intervals transported by continuity.\\
\hlt{Key Insight}: Continuous real functions on intervals cannot jump over a value.
\end{proof}
\end{theorem}

\begin{definition} \label{def:path-connected-set} \hlt{Path-Connected Set}\\
A set $E$ in a metric space is \hlt{path-connected} if for every $p,q\in E$ there is a continuous map $\gamma:[a,b]\to E$ with $\gamma(a)=p$ and $\gamma(b)=q$. Such a map $\gamma$ is a \hlt{path} from $p$ to $q$.
\end{definition}

\begin{theorem} \label{thm:path-connected-implies-connected} \hlt{Path-Connected Sets Are Connected}\\
Every path-connected set is connected.
\begin{proof}
Suppose $E=A\cup B$ with $A,B$ nonempty separated.\\
Choose $p\in A$, $q\in B$, and a path $\gamma:[a,b]\to E$ from $p$ to $q$. Since $[a,b]$ is connected, so is $\gamma([a,b])$.\\
But $\gamma([a,b])$ meets both $A$ and $B$, which separate it, a contradiction.\\
\hlt{Structural Commentary}: A path transports connectedness from an interval to the set.\\
\hlt{Key Insight}: To connect two points by a path, the path image cannot be split into two separated pieces.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:convex-sets-path-connected} \hlt{Convex Sets Are Path-Connected}\\
Every convex subset of $\R^k$ is path-connected.
\begin{proof}
For $\mathbf{x},\mathbf{y}\in E$, set $\gamma(t)=(1-t)\mathbf{x}+t\mathbf{y}$ on $[0,1]$. Convexity gives $\gamma([0,1])\subseteq E$, with $\gamma(0)=\mathbf{x}$ and $\gamma(1)=\mathbf{y}$.\\
Since the coordinates of $\gamma$ are affine in $t$, $\gamma$ is continuous. Hence $E$ is path-connected.\\
\hlt{Structural Commentary}: Convexity supplies the whole straight path between any two points.\\
\hlt{Key Insight}: In Euclidean space, line segments are continuous paths.
\end{proof}
\end{corollary}

\begin{definition} \label{def:one-sided-limits-and-discontinuities} \hlt{One-Sided Limits and Types of Discontinuity}\\
Let $f:(a,b)\to\R$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $a\leq x<b$, $f(x+)=q$ means $f(t_n)\to q$ whenever $t_n\in(x,b)$ and $t_n\to x$.
\item For $a<x\leq b$, $f(x-)=q$ means the analogous limit through $(a,x)$.
\item For $x\in(a,b)$, $\lim_{t\to x}f(t)$ exists if and only if $f(x+)=f(x-)$.
\end{enumerate}
A discontinuity with both one-sided limits is of the \hlt{first kind}, or \hlt{simple}; otherwise it is of the \hlt{second kind}\\
Simple discontinuities are jumps, $f(x+)\neq f(x-)$, or removable value mismatches.
\end{definition}

\begin{example} \label{ex:basic-discontinuity-types} \hlt{Basic Discontinuity Types}\\
Dirichlet's function $\mathbf{1}_{\Q}$ has a discontinuity of the second kind at every point.\\
The function $f(x)=x$ on $\Q$ and $f(x)=0$ off $\Q$ is continuous only at $0$; elsewhere its discontinuities are of the second kind. A jump in a piecewise linear function is a simple discontinuity.\\
Use Corollary \ref{cor:rationals-and-irrationals-dense}: every interval contains rational and irrational points.
\end{example}

\begin{definition} \label{def:monotone-real-function} \hlt{Monotone Functions}\\
Let $f$ be real-valued on $(a,b)$. The function is \hlt{monotonically increasing} if $a<x<y<b$ implies $f(x)\leq f(y)$. Reversing the inequality gives \hlt{monotonically decreasing}. A \hlt{monotone} function is one of these two types.
\end{definition}

\begin{theorem} \label{thm:monotone-functions-one-sided-limits} \hlt{One-Sided Limits of Monotone Functions}\\
If $f$ is increasing on $(a,b)$, then for every $x\in(a,b)$, $\sup\limits_{a<t<x}f(t)=f(x-)\leq f(x)\leq f(x+)=\inf\limits_{x<t<b}f(t)$.\\
Moreover, $a<x<y<b$ implies $f(x+)\leq f(y-)$. For decreasing $f$, reverse the inequalities.
\begin{proof}
Let $A\defeq\sup_{a<t<x}f(t)$.\\
For every $\varepsilon>0$, choose $s<x$ with $A-\varepsilon<f(s)\leq A$; then $s<t<x$ implies $A-\varepsilon<f(t)\leq A$, so $f(x-)=A$.\\
Let $B\defeq\inf_{x<t<b}f(t)$. For every $\varepsilon>0$, choose $s>x$ with $B\leq f(s)<B+\varepsilon$; then $x<t<s$ implies $B\leq f(t)<B+\varepsilon$, so $f(x+)=B$. If $x<y$, then $\inf_{x<t<y}f(t)\leq\sup_{x<t<y}f(t)$ gives $f(x+)\leq f(y-)$.\\
For decreasing $f$, apply the increasing case to $-f$ and multiply the resulting inequalities by $-1$.\\
\hlt{Structural Commentary}: Monotonicity converts one-sided limits into suprema and infima.\\
\hlt{Key Insight}: A monotone function has jumps, but not oscillatory one-sided failure.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:monotone-no-second-kind-discontinuities} \hlt{Monotone Functions Have No Second-Kind Discontinuities}\\
A monotone real function on an interval has no discontinuities of the second kind.
\begin{proof}
At every interior point, Theorem \ref{thm:monotone-functions-one-sided-limits} gives both one-sided limits.\\
Hence any discontinuity is of the first kind by Definition \ref{def:one-sided-limits-and-discontinuities}.\\
\hlt{Structural Commentary}: Monotonicity rules out oscillatory discontinuities.\\
\hlt{Key Insight}: All monotone discontinuities are jumps or removable value mismatches.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:monotone-discontinuities-countable} \hlt{Monotone Discontinuities Are Countable}\\
If $f$ is monotone on $(a,b)$, then the set of points of $(a,b)$ at which $f$ is discontinuous is at most countable.
\begin{proof}
Assume $f$ is increasing, and let $E$ be its discontinuity set.\\
For each $x\in E$, Theorem \ref{thm:monotone-functions-one-sided-limits} gives $f(x-)<f(x+)$; choose $r(x)\in\Q$ with $f(x-)<r(x)<f(x+)$.\\
If $x_1<x_2$, then $f(x_1+)\leq f(x_2-)$, so $r(x_1)\neq r(x_2)$. Thus $E$ injects into $\Q$, hence is countable by Corollary \ref{cor:rationals-countable}. If $f$ is decreasing, apply the increasing case to $-f$, which has the same discontinuity set.\\
\hlt{Structural Commentary}: Each jump contains a rational marker, and monotonicity keeps the markers distinct.\\
\hlt{Key Insight}: Countability enters by injecting discontinuities into $\Q$.
\end{proof}
\end{theorem}

\begin{remark} \label{rem:monotone-discontinuities-need-not-be-isolated} \hlt{Monotone Discontinuities Need Not Be Isolated}\\
For countable $E=\{x_n:n\in\N\}\subset(a,b)$ and positive summable $(c_n)$, $f(x)=\sum\limits_{x_n<x}c_n$ is increasing, has jumps of size $c_n$ at $x_n$, and is continuous off $E$. Thus monotone discontinuities may be dense.
\end{remark}

\begin{definition} \label{def:extended-real-function-limits} \hlt{Extended Limits and Limits at Infinity}\\
Neighborhoods of $+\infty$ are intervals $(c,+\infty)$; neighborhoods of $-\infty$ are $(-\infty,c)$.\\
For $f:E\to\R$ and extended real $A,x$, $f(t)\to A$ as $t\to x$ means: every neighborhood $U$ of $A$ contains $f(t)$ for all punctured $t\in E$ sufficiently near $x$, i.e. $\exists V\ni x$ such that $t\in V\cap E,\ t\neq x$ $\Longrightarrow$ $f(t)\in U$.\\
For real $A,x$, this agrees with Definition \ref{def:function-limit-metric}.\\
\hlt{Structural Commentary}: Extended limits use same neighborhood test, $\pm\infty$ allowed as input or output limits.
\end{definition}

\begin{theorem} \label{thm:extended-limit-algebra} \hlt{Algebra of Extended Limits}\\
Let $f,g:E\to\R$ and $f(t)\to A$, $g(t)\to B$ as $t\to x$, with $A,B,x$ extended real. Then, whenever RHS is defined:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item if $f(t)\to A'$ as $t\to x$, then $A'=A$;
\item $(f+g)(t)\to A+B$;
\item $(fg)(t)\to AB$;
\item $(f/g)(t)\to A/B$;
\end{enumerate}
The forms $\infty-\infty$, $0\cdot\infty$, $\infty/\infty$, and $A/0$ are undefined.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A'\neq A$, choose disjoint neighborhoods of $A$ and $A'$.\\
Definition \ref{def:extended-real-function-limits} would eventually place $f(t)$ in both, a contradiction.
\item For any neighborhood $W$ of $A+B$, definedness gives neighborhoods $U$ of $A$ and $V$ of $B$ such that $u+v\in W$ for $u\in U$, $v\in V$. Apply the two limits in a common punctured neighborhood of $x$.
\item The same neighborhood argument applies to multiplication, excluding $0\cdot\infty$.
\item Apply it to division; since $B\neq0$, choose the neighborhood of $B$ away from $0$.
\end{enumerate}
\hlt{Structural Commentary}: Extended limit algebra is ordinary limit algebra in neighborhood form.\\
\hlt{Key Insight}: The undefined forms are exactly where neighborhood control breaks.
\end{proof}
\end{theorem}

\begin{method} \label{meth:continuity-proof-patterns-source-todo}\label{meth:continuity-proof-patterns} \hlt{Continuity Proof Patterns}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{$\varepsilon$--$\delta$ at $p$}: prove $\forall\varepsilon>0\,\exists\delta>0$ such that $x\in E$ and $d_X(x,p)<\delta$ imply $d_Y(f(x),f(p))<\varepsilon$.
\item \hlt{Limit at $p$}: prove same estimate with $0<d_X(x,p)<\delta$, target $q$; equivalently test all $x_n\to p$, $x_n\neq p$.
\item \hlt{Topological continuity}: prove $V\subseteq Y$ open $\Rightarrow f^{-1}(V)$ open, or $C\subseteq Y$ closed $\Rightarrow f^{-1}(C)$ closed.
\item \hlt{Compactness}: from open cover $\{V_\alpha\}$ of $f(X)$, pulled-back cover $\{f^{-1}(V_\alpha)\}$ of $X$ extract finite subcover.
\item \hlt{Uniform continuity}: choose local radii $r_p$, cover compact $X$ by $B_{r_p/2}(p)$, and set $\delta=\frac12\min_i r_{p_i}$.
\end{enumerate}
\hlt{Key Insight}: Continuity is local control; compactness turns local controls into finite control.
\end{method}
