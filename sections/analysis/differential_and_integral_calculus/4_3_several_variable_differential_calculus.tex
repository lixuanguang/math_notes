\subsubsection{Several-Variable Differential Calculus}

Several-variable differentiation treats the derivative at a point as the linear map that best approximates the function there.

\begin{definition} \label{def:several-variable-differentiation-core-source-todo}\label{def:total-derivative-rn-rm} \hlt{Total Derivative}\\
Let $E\subseteq\R^n$ be open, let $\f:E\to\R^m$, and let $\x\in E$.\\
If $\exists A\in\LL(\R^n,\R^m)$ s.t. $\lim\limits_{\h\to\ze}\frac{\norm{\f(\x+\h)-\f(\x)-A\h}_2}{\norm{\h}_2}=0$, then $\f$ is \hlt{differentiable at $\x$}, and $A$ denoted $D\f(\x)$ or $\f'(\x)$.\\
Equivalently, $\f(\x+\h)-\f(\x)=D\f(\x)\h+\mathbf{r}(\h)$, and $\frac{\norm{\mathbf{r}(\h)}_2}{\norm{\h}_2}\to0$.\\
If this holds for every $\x\in E$, then $\f$ is differentiable on $E$.
\end{definition}

\begin{theorem} \label{thm:total-derivative-unique-and-continuous}\label{thm:several-variable-differentiation-main-source-todo} \hlt{Basic Consequences of Total Differentiability}\\
If $\f:E\to\R^m$ is differentiable at $\x\in E$, then $D\f(\x)$ is unique and $\f$ is continuous at $\x$.
\begin{proof}
If $A_1,A_2$ both satisfy Definition \ref{def:total-derivative-rn-rm}, set $B=A_1-A_2$. Then $\norm{B\h}_2/\norm{\h}_2\to0$ as $\h\to\ze$.\\
For fixed $\h\neq\ze$, applying this to $t\h$ and using linearity makes the quotient independent of $t$, so $B\h=\ze$.\\
Hence $B=0$. For continuity, write $\f(\x+\h)-\f(\x)=D\f(\x)\h+\mathbf{r}(\h)$.\\
The operator-norm estimate gives $\norm{D\f(\x)\h}_2\leq\normop{D\f(\x)}\norm{\h}_2$, and remainder is $o(\norm{\h}_2)$, so $\f(\x+\h)\to\f(\x)$.\\
\hlt{Structural Commentary}: The derivative is a rigid first-order linear part.\\
\hlt{Key Insight}: A remainder smaller than $\norm{\h}$ cannot hide a second linear map.
\end{proof}
\end{theorem}

\begin{example} \label{ex:derivative-of-linear-map} \hlt{Derivative of a Linear Map}\\
If $A\in\LL(\R^n,\R^m)$ and $\f(\x)=A\x$, then $D\f(\x)=A$ for every $\x\in\R^n$.\\
The remainder in Definition \ref{def:total-derivative-rn-rm} is identically zero.
\end{example}

\begin{theorem} \label{thm:chain-rule-several-variables} \hlt{Chain Rule}\\
If $\f:E\subseteq\R^n\to\R^m$ is differentiable at $\x_0$, $\g:G\subseteq\R^m\to\R^k$ is differentiable at $\f(\x_0)$, and $\f(E)\subseteq G$, then $\g\circ\f$ is differentiable at $\x_0$ and $D(\g\circ\f)(\x_0)=D\g(\f(\x_0))\,D\f(\x_0)$.
\begin{proof}
Let $A=D\f(\x_0)$, $B=D\g(\f(\x_0))$, $\mathbf{u}(\h)=\f(\x_0+\h)-\f(\x_0)-A\h$, and $\mathbf{v}(\mathbf{k})=\g(\f(\x_0)+\mathbf{k})-\g(\f(\x_0))-B\mathbf{k}$.\\
For $\mathbf{k}=\f(\x_0+\h)-\f(\x_0)$, differentiability of $\f$ gives $\mathbf{k}\to\ze$ and $(\g\circ\f)(\x_0+\h)-(\g\circ\f)(\x_0)-BA\h =B\mathbf{u}(\h)+\mathbf{v}(\mathbf{k})$, where $\norm{\mathbf{u}(\h)}_2=o(\norm{\h}_2)$ and $\norm{\mathbf{v}(\mathbf{k})}_2=o(\norm{\mathbf{k}}_2)=o(\norm{\h}_2)$. Divide by $\norm{\h}_2$.\\
\hlt{Structural Commentary}: The first-order parts compose as linear maps.\\
\hlt{Key Insight}: The output increment of $\f$ is small enough to feed into the remainder estimate for $\g$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:partials-gradient-directional-derivatives} \hlt{Partial, Gradient, and Directional Derivatives}\\
Let $E\subseteq\R^n$ be open, $\f=(f_1,\ldots,f_m):E\to\R^m$, and let $\mathbf{e}_1,\ldots,\mathbf{e}_n$ be the standard basis of $\R^n$.\\
If indicated limits exist, \hlt{partial derivative} is $(D_jf_i)(\x)\defeq\lim_{t\to0}\frac{f_i(\x+t\mathbf{e}_j)-f_i(\x)}{t}$ for $1\leq i\leq m,\ 1\leq j\leq n$.\\
\hlt{Directional derivative} is $(D_{\mathbf{u}}f)(\x)\defeq\lim_{t\to0}\frac{f(\x+t\mathbf{u})-f(\x)}{t}$ for $\norm{\mathbf{u}}_2=1$.\\
If $m=1$ and all partials exist, the \hlt{gradient} is $(\nabla f)(\x)\defeq \sum\limits_{j=1}^{n}(D_jf)(\x)\mathbf{e}_j$.
\end{definition}

\begin{theorem} \label{thm:total-derivative-matrix-partials} \hlt{Matrix of the Total Derivative}\\
If $\f:E\to\R^m$ is differentiable at $\x$, then every $(D_jf_i)(\x)$ exists and, in standard bases,
\begin{align}
[D\f(\x)] =
\begin{bmatrix}
(D_1f_1)(\x) & \cdots & (D_nf_1)(\x)\\
\vdots & & \vdots\\
(D_1f_m)(\x) & \cdots & (D_nf_m)(\x)
\end{bmatrix} \nonumber
\end{align}
If $m=1$ and $f$ is differentiable at $\x$, then $(D_{\mathbf{u}}f)(\x)=(\nabla f)(\x)\cdot\mathbf{u}$ for every unit vector $\mathbf{u}$.
\begin{proof}
Set $\h=t\mathbf{e}_j$ in Definition \ref{def:total-derivative-rn-rm}; the limit is $D\f(\x)\mathbf{e}_j$, so its components form column $j$ by Proposition \ref{prop:matrix-representation-action}. For $m=1$, set $\h=t\mathbf{u}$ and identify the linear functional with dot product against $\nabla f(\x)$.\\
\hlt{Structural Commentary}: Partials are coordinate columns of $D\f(\x)$.\\
\hlt{Key Insight}: Differentiability controls all directions at once; partial derivatives record its coordinate columns.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:bounded-derivative-convex-mean-value} \hlt{Mean-Value Inequality on Convex Sets}\\
Let $E\subseteq\R^n$ be open and convex.\\
If $\f:E\to\R^m$ is differentiable and $\normop{D\f}\leq M$ on $E$, then $\norm{\f(\b)-\f(\a)}_2\leq M\norm{\b-\a}_2$, and $\a,\b\in E$.\\
If $D\f=0$ on $E$, then $\f$ is constant.
\begin{proof}
Fix $\a,\b\in E$, set $\gamma(t)=(1-t)\a+t\b$, and let $\mathbf{g}=\f\circ\gamma$. Then $\gamma([0,1])\subseteq E$ and $\norm{\mathbf{g}'(t)}_2=\norm{D\f(\gamma(t))(\b-\a)}_2\leq M\norm{\b-\a}_2$. Apply Theorem \ref{thm:vector-valued-mean-value-inequality}; the constant case is $M=0$.\\
\hlt{Structural Commentary}: Restricting to a line segment reduces the estimate to one variable.\\
\hlt{Key Insight}: A uniform bound on the derivative bounds every secant displacement.
\end{proof}
\end{theorem}

\begin{definition} \label{def:c1-map} \hlt{$C^1$ Mapping}\\
Let $E\subseteq\R^n$ be open. A differentiable map $\f:E\to\R^m$ is \hlt{continuously differentiable}, or $C^1$, if $\x\mapsto D\f(\x)$ is continuous from $E$ into $\LL(\R^n,\R^m)$ in operator norm.
\end{definition}

\begin{theorem} \label{thm:c1-iff-continuous-partials} \hlt{$C^1$ Means Continuous Partials}\\
A map $\f:E\subseteq\R^n\to\R^m$ is $C^1$ iff all $D_jf_i$ exist and are continuous on $E$.
\begin{proof}
If $\f$ is $C^1$, Theorem \ref{thm:total-derivative-matrix-partials} makes each $D_jf_i$ a continuous matrix entry of $D\f$.\\
Conversely reduce componentwise to $m=1$. For $\h=\sum\limits_jh_j\mathbf{e}_j$, telescoping along coordinate increments and applying Corollary \ref{cor:mean-value-theorem} gives, by continuity of the partials near $\x$, $f(\x+\h)-f(\x)-\sum\limits_{j=1}^nh_j(D_jf)(\x)=o(\norm{\h}_2)$.\\
Hence $Df(\x)$ has row matrix $[(D_1f)(\x)\ \cdots\ (D_nf)(\x)]$, Definition \ref{def:operator-norm-rn-rm} turns this into continuity of $Df$.\\
\hlt{Structural Commentary}: Continuous partial derivatives assemble into a continuous total derivative.\\
\hlt{Key Insight}: Coordinate-by-coordinate increments work only because the partials vary continuously.
\end{proof}
\end{theorem}

\begin{definition} \label{def:jacobian-determinant} \hlt{Jacobian}\\
If $\f:E\subseteq\R^n\to\R^n$ is differentiable at $\x$, define $J_{\f}(\x)\defeq \det D\f(\x) =\frac{\partial(y_1,\ldots,y_n)}{\partial(x_1,\ldots,x_n)}$ when $f=(y_1,\ldots,y_n)$.\\
By Theorem \ref{thm:determinant-basic-laws}, $D\f(\x)$ is invertible if and only if $J_{\f}(\x)\neq0$.
\end{definition}

\begin{theorem} \label{thm:inverse-function-theorem-rn} \hlt{Inverse Function Theorem}\\
Let $\f:E\subseteq\R^n\to\R^n$ be $C^1$, let $\a\in E$, and suppose $D\f(\a)$ is invertible. With $\b=\f(\a)$, $\f$ restricts to a bijection $U\to V$ between open neighborhoods of $\a$ and $\b$; inverse $\g:V\to U$ is $C^1$, and $D\g(\y)=\bigl(D\f(\g(\y))\bigr)^{-1}$, $\y\in V$.
\begin{proof}
Put $A=D\f(\a)$. On a small ball, $\normop{D\f-A}$ is small compared with $\normop{A^{-1}}^{-1}$.\\
For fixed $\y$, solutions of $\f(\x)=\y$ are fixed points of $\varphi_{\y}(\x)=\x+A^{-1}(\y-\f(\x))$.\\
By Theorem \ref{thm:bounded-derivative-convex-mean-value}, $\varphi_{\y}$ is a contraction on a suitable complete closed ball; Theorem \ref{thm:contraction-principle} gives the local solution and uniqueness. If $\y=\f(\x)$ and $\y+\mathbf{k}=\f(\x+\h)$, the same estimates give $\norm{\h}_2=O(\norm{\mathbf{k}}_2)$ and\\
$\g(\y+\mathbf{k})-\g(\y)-\bigl(D\f(\x)\bigr)^{-1}\mathbf{k} =-\bigl(D\f(\x)\bigr)^{-1}\bigl[\f(\x+\h)-\f(\x)-D\f(\x)\h\bigr]$.\\
Dividing by $\norm{\mathbf{k}}_2$ gives $D\g(\y)=(D\f(\x))^{-1}$; continuity follows from $D\f$, $\g$, and Theorem \ref{thm:invertible-operators-open-and-inversion-continuous}.\\
\hlt{Structural Commentary}: Local inversion is a contraction built from $A^{-1}$.\\
\hlt{Key Insight}: Small derivative variation preserves local invertibility.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:local-diffeomorphism-open-map} \hlt{Local Diffeomorphisms Are Open}\\
If $\f:E\subseteq\R^n\to\R^n$ is $C^1$ and $D\f$ is everywhere invertible, then $\f$ is an open map (locally one-to-one).
\begin{proof}
Let $W\subseteq E$ be open and let $\y\in\f(W)$. Choose $\x\in W$ with $\f(\x)=\y$. Applying Theorem \ref{thm:inverse-function-theorem-rn} to $\f|_W$ at $\x$ gives open neighborhoods $U\subseteq W$ of $\x$ and $V$ of $\y$ with $\f(U)=V$. Hence $V\subseteq\f(W)$, so $\y$ is interior to $\f(W)$. Since $\y$ was arbitrary, $\f(W)$ is open. Local one-to-one-ness is the same theorem applied pointwise.\\
\hlt{Structural Commentary}: Openness is checked one image point at a time.\\
\hlt{Key Insight}: A local inverse turns nearby domain points into a full image neighborhood.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:implicit-function-theorem-rn-rm} \hlt{Implicit Function Theorem}\\
Let $\f:E\subseteq\R^{n+m}\to\R^n$ be $C^1$, $\f(\a,\b)=0$, $A=D\f(\a,\b)$. If $A_x$ (Definition \ref{def:implicit-derivative-blocks}) is invertible, then near $\b$, the equation $\f(\x,\y)=0$ has unique local solution $\x=\g(\y)$, where $\g$ is $C^1$, $\g(\b)=\a$, $D\g(\b)=-(A_x)^{-1}A_y$.
\begin{proof}
Set $\Phi(\x,\y)=(\f(\x,\y),\y)$. Then $D\Phi(\a,\b)(\h,\mathbf{k})=(A_x\h+A_y\mathbf{k},\mathbf{k})$, whose kernel is zero since $A_x$ is invertible; hence $D\Phi(\a,\b)$ is invertible by Theorem \ref{thm:finite-dimensional-operator-invertible-equivalences}. Theorem \ref{thm:inverse-function-theorem-rn} makes $\Phi$ locally invertible.\\
Points with $\f(\x,\y)=0$ are exactly those sent to $(0,\y)$, so $\Phi^{-1}(0,\y)=(\g(\y),\y)$ gives the unique local solution. Differentiating $\f(\g(\y),\y)=0$ at $\b$ gives $A_xD\g(\b)+A_y=0$.\\
\hlt{Structural Commentary}: Add $\y$ as a coordinate, then invert.\\
\hlt{Key Insight}: Invertibility of the $x$-block is precisely the linear ability to solve for $x$.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:constant-rank-theorem-rudin} \hlt{Rank Theorem}\\
Let $\mathbf{F}:E\subseteq\R^n\to\R^m$ be $C^1$ with constant rank $r$. At $\a\in E$, put $A=D\mathbf{F}(\a)$, $Y_1=\operatorname{range}(A)$, and let $P$ be the orthogonal projection onto $Y_1$. Locally, after a $C^1$ coordinate change $H$ with $C^1$ inverse, there is a $C^1$ map $\varphi:A(V)\subseteq Y_1\to\ker(P)$ such that $\mathbf{F}(H(\x))=A\x+\varphi(A\x)$.
\begin{proof}
If $r=0$, Theorem \ref{thm:bounded-derivative-convex-mean-value} makes $\mathbf{F}$ locally constant. For $r>0$, choose $S:Y_1\to\R^n$ with $AS=I_{Y_1}$ and set $G(\x)=\x+SP(\mathbf{F}(\x)-A\x)$. Since $DG(\a)=I$, Theorem \ref{thm:inverse-function-theorem-rn} gives a local inverse $H=G^{-1}$.\\
From $AG=P\mathbf{F}$ we get $P\mathbf{F}(H(\x))=A\x$. The remaining component $(I-P)\mathbf{F}(H(\x))$ is constant when $A\x$ is fixed, by the constant-rank hypothesis and the mean-value estimate, so it equals $\varphi(A\x)$.\\
\hlt{Structural Commentary}: Constant rank gives a local graph normal form.\\
\hlt{Key Insight}: Same rank, same local level-set geometry.
\end{proof}
\end{theorem}

\begin{method} \label{meth:several-variable-differentiation-proof-patterns-source-todo}\label{meth:several-variable-differentiation-proof-patterns} \hlt{Several-Variable Differentiation Proof Patterns}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Linear approximation}: write $\f(\x+\h)-\f(\x)=D\f(\x)\h+\mathbf{r}(\h)$ with $\norm{\mathbf{r}(\h)}_2=o(\norm{\h}_2)$.
\item \hlt{Norm estimates}: use $\norm{A\h}_2\leq\normop{A}\norm{\h}_2$ and the product estimate from Proposition \ref{prop:spectral-norm-basic-properties}.
\item \hlt{Line-segment reduction}: on convex sets, restrict to $\gamma(t)=(1-t)\a+t\b$ and apply one-variable results.
\item \hlt{Local inversion}: compare $D\f(\x)$ with an invertible $D\f(\a)$, build a contraction, and invoke Theorem \ref{thm:contraction-principle}.
\item \hlt{Coordinates}: read partial derivatives as the entries of the matrix of $D\f(\x)$, using Theorem \ref{thm:total-derivative-matrix-partials}.
\end{enumerate}
\hlt{Key Insight}: Several-variable calculus is one-variable limiting plus finite-dimensional linear algebra.
\end{method}
