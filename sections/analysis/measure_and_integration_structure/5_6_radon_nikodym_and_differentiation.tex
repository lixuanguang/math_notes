\subsection{Radon-Nikodym and Differentiation}

% **Role:** modern integration and almost-everywhere analysis.
%
% Signed measures, Hahn/Jordan decomposition, absolute continuity, Radon--Nikodym theorem, Lebesgue decomposition, Lebesgue differentiation theorem.

This subsection first decomposes finite signed set functions into honest measures, then compares measures by absolute continuity and singularity. The differentiation results later in the subsection are the local Euclidean counterpart of the Radon--Nikodym viewpoint.

\begin{definition} \label{def:positive-negative-null-sets-charge} \hlt{Positive, Negative, and Null Sets for a Charge}\\
Let $\lambda$ be a charge on $(X,\mathcal{M})$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{P}_\lambda\defeq\{P\in\mathcal{M}:\forall E\in\mathcal{M},\lambda(E\cap P)\geq0\}$, the \hlt{positive} sets.
\item $\mathcal{N}_\lambda\defeq\{N\in\mathcal{M}:\forall E\in\mathcal{M},\lambda(E\cap N)\leq0\}$, the \hlt{negative} sets.
\item $\mathcal{Z}_\lambda\defeq\{M\in\mathcal{M}:\forall E\in\mathcal{M},\lambda(E\cap M)=0\}$, the \hlt{null} sets.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:positive-negative-set-closure-charge} \hlt{Positive and Negative Set Closure}\\
For a charge $\lambda$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $A\in\mathcal{M}$ and $A\subseteq P\in\mathcal{P}_\lambda$ imply $A\in\mathcal{P}_\lambda$; $A\subseteq N\in\mathcal{N}_\lambda$ implies $A\in\mathcal{N}_\lambda$.
\item $\mathcal{P}_\lambda$ and $\mathcal{N}_\lambda$ are closed under finite unions.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $A\subseteq P\in\mathcal{P}_\lambda$, then $\lambda(E\cap A)=\lambda((E\cap A)\cap P)\geq0$ for all $E\in\mathcal{M}$, hence $A\in\mathcal{P}_\lambda$.\\
The negative case reverses the inequality.
\item If $P_1,P_2\in\mathcal{P}_\lambda$, then $\lambda(E\cap(P_1\cup P_2))=\lambda(E\cap P_1)+\lambda(E\cap P_1^c\cap P_2)\geq0$ $\forall E\in\mathcal{M}$, since $E\cap P_1^c\cap P_2\subseteq P_2$.\\
Thus $P_1\cup P_2\in\mathcal{P}_\lambda$; induction gives finite unions, negative case is identical with reversed inequalities.
\end{enumerate}
\hlt{Structural Commentary}: Sign sets are hereditary and finite-union stable.\\
\hlt{Key Insight}: Test sets can always be disjointized against the first positive set.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:hahn-decomposition-charge} \hlt{Hahn Decomposition Theorem}\\
Let $\lambda$ be a charge on $(X,\mathcal{M})$. Then there exist $P,N\in\mathcal{M}$ such that
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $P\cap N=\emptyset$ and $X=P\sqcup N$.
\item $P\in\mathcal{P}_\lambda$.
\item $N\in\mathcal{N}_\lambda$.
\end{enumerate}
Such a pair $(P,N)$ is a \hlt{Hahn decomposition} of $X$ for $\lambda$.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Put $\alpha=\sup\{\lambda(A):A\in\mathcal{P}_\lambda\}$. If $\alpha=+\infty$, choose $B_m\in\mathcal{P}_\lambda$, $\lambda(B_m)>m$, and set $C_m=\bigcup_{j=1}^{m}B_j$; then $C_m\in\mathcal{P}_\lambda$, $C_m\uparrow C$, and $\lambda(C_m)\geq\lambda(B_m)>m$, contradicting continuity from below and $\lambda(C)\in\R$.\\
Choose $A_n\in\mathcal{P}_\lambda$, $\lambda(A_n)\to\alpha$, replace $A_n$ by $\bigcup_{j=1}^{n}A_j$, and set $P=\bigcup_{n\geq1}A_n$, $N=X\setminus P$.\\
Hence $P,N\in\mathcal{M}$, $P\cap N=\emptyset$, and $X=P\sqcup N$.
\item For $E\in\mathcal{M}$, $E\cap A_n\uparrow E\cap P$ and $\lambda(E\cap A_n)\geq0$, so continuity from below gives $\lambda(E\cap P)\geq0$.\\
Thus $P\in\mathcal{P}_\lambda$, and $\lambda(P)=\alpha$.
\item Suppose $E\in\mathcal{M}$, $E\subseteq N$, and $\lambda(E)>0$. Then $E\notin\mathcal{P}_\lambda$, else $P\cup E\in\mathcal{P}_\lambda$ and $\lambda(P\cup E)>\alpha$.\\
Set $R_0=E$; while $R_{k-1}\notin\mathcal{P}_\lambda$, choose $n_k$ minimal with some $H\in\mathcal{M}$, $H\subseteq R_{k-1}$, $\lambda(H)\leq-1/n_k$, put $E_k=H$, and set $R_k=R_{k-1}\setminus E_k$. If $R_m\in\mathcal{P}_\lambda$, then $\lambda(R_m)=\lambda(E)-\sum_{k=1}^{m}\lambda(E_k)>0$, so $P\cup R_m\in\mathcal{P}_\lambda$ and $\lambda(P\cup R_m)>\alpha$; hence the construction is infinite.\\
For $F_m=\bigcup_{k=1}^{m}E_k$ and $F=\bigcup_{k\geq1}E_k$, $\lambda(F_m)=\sum_{k=1}^{m}\lambda(E_k)\leq-\sum_{k=1}^{m}1/n_k$; since $F_m\uparrow F$ and $\lambda(F)\in\R$, $\sum_k1/n_k<+\infty$ and $n_k\to+\infty$. If $G\in\mathcal{M}$ and $G\subseteq E\setminus F$, then $G\subseteq R_{k-1}$ for every $k$, so minimality gives $\lambda(G)>-1/(n_k-1)$ for all large $k$, hence $\lambda(G)\geq0$ and $E\setminus F\in\mathcal{P}_\lambda$. But $\lambda(E\setminus F)=\lambda(E)-\lambda(F)>0$, so $P\cup(E\setminus F)\in\mathcal{P}_\lambda$ and $\lambda(P\cup(E\setminus F))>\alpha$, contradiction. Thus $N\in\mathcal{N}_\lambda$.
\end{enumerate}
\hlt{Structural Commentary}: A charge splits $X$ into nonnegative-piece and nonpositive-piece regions.\\
\hlt{Key Insight}: Maximize positive mass; purify any leftover positive piece.
\end{proof}
\end{theorem}

\begin{lemma} \label{lem:hahn-positive-part-independent} \hlt{Hahn Parts Are Unique up to Null Sets}\\
Let $(P_i,N_i)$, $i=1,2$, be Hahn decompositions for a charge $\lambda$.\\
Then $\forall E\in\mathcal{M}$, $\lambda(E\cap P_1)=\lambda(E\cap P_2)$ and $\lambda(E\cap N_1)=\lambda(E\cap N_2)$.
\begin{proof}
Fix $E\in\mathcal{M}$. For $\{i,j\}=\{1,2\}$, $E\cap(P_i\setminus P_j)\subseteq P_i\cap N_j$, hence $\lambda(E\cap(P_i\setminus P_j))=0$.\\
Thus $\lambda(E\cap P_i)=\lambda(E\cap P_1\cap P_2)$ for $i=1,2$.\\
Similarly, $E\cap(N_i\setminus N_j)\subseteq N_i\cap P_j$, hence $\lambda(E\cap N_i)=\lambda(E\cap N_1\cap N_2)$ for $i=1,2$.\\
\hlt{Structural Commentary}: Hahn decompositions are not unique as sets, but the measures they define are unique.\\
\hlt{Key Insight}: Positive-and-negative overlap is necessarily null.
\end{proof}
\end{lemma}

\begin{definition} \label{def:jordan-positive-negative-total-variation} \hlt{Jordan Variations of a Charge}\\
Let $\lambda$ be a charge, $(P,N)$ a Hahn decomposition, and $E\in\mathcal{M}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\lambda^+(E)\defeq\lambda(E\cap P)$, the \hlt{positive variation}.
\item $\lambda^-(E)\defeq-\lambda(E\cap N)$, the \hlt{negative variation}.
\item $\abs{\lambda}(E)\defeq\lambda^+(E)+\lambda^-(E)$, the \hlt{total variation}.
\end{enumerate}
Lemma \ref{lem:hahn-positive-part-independent} makes these independent of $(P,N)$.
\end{definition}

\begin{theorem} \label{thm:jordan-decomposition-charge} \hlt{Jordan Decomposition Theorem}\\
For every charge $\lambda$, $\lambda=\lambda^+-\lambda^-$. If $\lambda=\mu-\nu$ for finite measures $\mu,\nu$, then $\lambda^+\leq\mu$ and $\lambda^-\leq\nu$.
\begin{proof}
Let $(P,N)$ be a Hahn decomposition and $E\in\mathcal{M}$. Then $\lambda(E)=\lambda(E\cap P)+\lambda(E\cap N)=\lambda^+(E)-\lambda^-(E)$. If $\lambda=\mu-\nu$, then $\lambda^+(E)=\mu(E\cap P)-\nu(E\cap P)\leq\mu(E)$ and $\lambda^-(E)=\nu(E\cap N)-\mu(E\cap N)\leq\nu(E)$.\\
\hlt{Structural Commentary}: Jordan decomposition is the minimal finite-measure difference.\\
\hlt{Key Insight}: Positive and negative Hahn pieces isolate the irreducible positive masses.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:variation-supremum-formulas-and-null-sets} \hlt{Variation by Subsets and Null Sets}\\
For a charge $\lambda$ and $E,M\in\mathcal{M}$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\lambda^+(E)=\sup\{\lambda(F):F\in\mathcal{M},F\subseteq E\}$ and $\lambda^-(E)=-\inf\{\lambda(F):F\in\mathcal{M},F\subseteq E\}$.
\item $M\in\mathcal{Z}_\lambda\Leftrightarrow\abs{\lambda}(M)=0$.
\item $\pi_\lambda=\lambda^+$ and $\nu_\lambda=\abs{\lambda}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item[$(i)$] Let $(P,N)$ be Hahn. If $F\in\mathcal{M}$ and $F\subseteq E$, then $\lambda(F)=\lambda^+(F)-\lambda^-(F)\leq\lambda^+(E)$, with equality at $F=E\cap P$; similarly $\lambda(F)\geq-\lambda^-(E)$, with equality at $F=E\cap N$.
\item[$(ii)$]$(\Rightarrow)$ If $M\in\mathcal{Z}_\lambda$, then $\lambda^+(M)=\lambda(M\cap P)=0$ and $\lambda^-(M)=-\lambda(M\cap N)=0$, so $\abs{\lambda}(M)=0$.
\item[$(ii)$]$(\Leftarrow)$ If $\abs{\lambda}(M)=0$, then $\lambda^\pm(G\cap M)=0$ for every $G\in\mathcal{M}$, hence $\lambda(G\cap M)=0$ and $M\in\mathcal{Z}_\lambda$.
\item[$(iii)$] The identity $\pi_\lambda=\lambda^+$ is (i). For finite partitions $(A_j)$ of $E$, $\sum_j\abs{\lambda(A_j)}\leq\sum_j\abs{\lambda}(A_j)=\abs{\lambda}(E)$, and $E=(E\cap P)\sqcup(E\cap N)$ attains equality; hence $\nu_\lambda=\abs{\lambda}$.
\end{enumerate}
\hlt{Structural Commentary}: Variation is intrinsic to $\lambda$.\\
\hlt{Key Insight}: The positive part is the largest charge obtainable inside a set.
\end{proof}
\end{corollary}

\begin{theorem} \label{thm:integral-charge-variation-density} \hlt{Variations of Integral Charges}\\
Let $f\in\mathscr{L}^1(\mu)$ and $\lambda(E)\defeq\int_E f\,d\mu$.\\
Then $\forall E\in\mathcal{M}$, $\lambda^+(E)=\int_E f^+\,d\mu$, $\lambda^-(E)=\int_E f^-\,d\mu$, and $\abs{\lambda}(E)=\int_E\abs{f}\,d\mu$.
\begin{proof} Set $P=\{f\geq0\}$ and $N=\{f<0\}$.\\
For $A\in\mathcal{M}$, $\lambda(A\cap P)=\int_{A\cap P}f\,d\mu\geq0$ and $\lambda(A\cap N)=\int_{A\cap N}f\,d\mu\leq0$, so $(P,N)$ is Hahn.\\
Thus $\lambda^+(E)=\lambda(E\cap P)=\int_E f^+\,d\mu$, $\lambda^-(E)=-\lambda(E\cap N)=\int_E f^-\,d\mu$, and $\abs{\lambda}(E)=\lambda^+(E)+\lambda^-(E)=\int_E\abs{f}\,d\mu$.\\
\hlt{Structural Commentary}: Measure variations match function parts.\\
\hlt{Key Insight}: Sign sets of the density are Hahn sets.
\end{proof}
\end{theorem}

\begin{definition} \label{def:absolute-continuity-and-singularity-measures} \hlt{Absolute Continuity and Singularity}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Absolute continuity for measures}: $\lambda\ll\mu\Leftrightarrow\forall E\in\mathcal{M},\mu(E)=0\Rightarrow\lambda(E)=0$.
\item \hlt{Absolute continuity for charges}: $\lambda\ll\mu\Leftrightarrow\abs{\lambda}\ll\mu$.
\item \hlt{Mutual singularity}: $\lambda\perp\mu\Leftrightarrow\exists A,B\in\mathcal{M}$ with $A\cap B=\emptyset$, $X=A\cup B$, $\lambda(A)=0$, and $\mu(B)=0$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:absolute-continuity-basic-calculus} \hlt{Absolute Continuity Calculus}\\
Let $\lambda,\mu,\nu$ be measures and $\eta$ a charge.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\lambda\ll\lambda$ and $\lambda\ll\mu\ll\nu\Rightarrow\lambda\ll\nu$.
\item $\ll$ is not symmetric.
\item $(\forall E\in\mathcal{M},\mu(E)=0\Rightarrow\eta(E)=0)\Rightarrow\eta^+,\eta^-,\abs{\eta}\ll\mu$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Reflexivity is immediate. If $\lambda\ll\mu\ll\nu$ and $\nu(E)=0$, then $\mu(E)=0$, hence $\lambda(E)=0$.
\item On $X=\{0,1\}$, $\mathcal{M}=2^X$, $\delta_0\ll\#$, but $\#\not\ll\delta_0$ since $\delta_0(\{1\})=0$ and $\#\{1\}=1$.
\item If $\mu(E)=0$, then $\mu(F)=0$ for every $F\in\mathcal{M}$, $F\subseteq E$, so $\eta(F)=0$.\\
Hence $E\in\mathcal{Z}_\eta$, and Corollary \ref{cor:variation-supremum-formulas-and-null-sets} gives $\abs{\eta}(E)=0$; therefore $\eta^\pm(E)\leq\abs{\eta}(E)=0$.
\end{enumerate}
\hlt{Structural Commentary}: Absolute continuity is null-set order; charges reduce to variation.\\
\hlt{Key Insight}: Null sets pass to measurable subsets, so all signed pieces vanish there.
\end{proof}
\end{proposition}

\begin{example} \label{ex:weighted-sum-dominates-countable-measures} \hlt{Dominating a Countable Family of Measures}\\
Let $(\mu_n)$ be measures with $\mu_n(X)\leq1$, and set $\lambda(E)\defeq\sum_{n\geq1}2^{-n}\mu_n(E)$.\\
Then $\lambda$ is finite, since $\lambda(X)\leq1$, and countably additive by Tonelli for nonnegative series.\\
Also $2^{-n}\mu_n(E)\leq\lambda(E)$, hence $\lambda(E)=0\Rightarrow\mu_n(E)=0$ and $\mu_n\ll\lambda$ for all $n$.\\
\hlt{Key Insight}: A summable positive mixture packages countably many null-set tests into one reference measure.
\end{example}

\begin{lemma} \label{lem:absolute-continuity-epsilon-delta-finite-source} \hlt{Epsilon--Delta Form of Absolute Continuity}\\
Let $\lambda$ be finite and $\mu$ a measure on $(X,\mathcal{M})$. Then $\lambda\ll\mu\Leftrightarrow\forall\varepsilon>0,\exists\delta>0,\forall E\in\mathcal{M},\mu(E)<\delta\Rightarrow\lambda(E)<\varepsilon$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Suppose the $\varepsilon$-$\delta$ condition fails. Then $\exists\varepsilon>0$ and $E_n\in\mathcal{M}$ with $\mu(E_n)<2^{-n}$ and $\lambda(E_n)\geq\varepsilon$.\\
Let $F_n=\bigcup_{k\geq n}E_k$ and $F=\bigcap_nF_n$. Then $\mu(F_n)\leq2^{-n+1}$, so $\mu(F)=0$.\\
Since $F_n\downarrow F$, $\lambda(F_n)\geq\varepsilon$, and $\lambda$ is finite, continuity from above gives $\lambda(F)\geq\varepsilon$, contradicting $\lambda\ll\mu$.
\item[$(\Leftarrow)$] If $\mu(E)=0$, then $\mu(E)<\delta$ for every $\delta>0$; hence $\lambda(E)<\varepsilon$ for every $\varepsilon>0$, so $\lambda(E)=0$.
\end{enumerate}
\hlt{Structural Commentary}: For finite target measure, absolute continuity is uniform smallness on small $\mu$-sets.\\
\hlt{Key Insight}: A failing sequence is converted into a decreasing null tail with positive $\lambda$-mass.
\end{proof}
\end{lemma}

\begin{example} \label{ex:absolute-continuity-delta-fails-infinite-target} \hlt{Finite Target Is Needed}\\
On $(\N,2^\N)$, let $\lambda=\#$ and $\mu(E)\defeq\sum_{n\in E}2^{-n}$. Since $\mu(E)=0\Leftrightarrow E=\emptyset$, $\lambda\ll\mu$.\\
But for $E_n=\{n\}$, $\mu(E_n)=2^{-n}\to0$ while $\lambda(E_n)=1$, so the $\varepsilon$-$\delta$ condition fails for $\varepsilon=1$.
\end{example}

\begin{example} \label{ex:mutual-absolute-continuity-with-infinite-measure} \hlt{Mutual Absolute Continuity Need Not Preserve Finiteness}\\
On $(\N,2^\N)$, set $\mu(E)\defeq\sum_{n\in E}2^{-n}$ and $\lambda(\emptyset)=0$, $\lambda(E)=+\infty$ for $E\neq\emptyset$.\\
Then $\mu(\N)=1$, while $\lambda$ is not $\sigma$-finite. Since both measures have null ideal $\{\emptyset\}$, $\lambda\ll\mu$ and $\mu\ll\lambda$.\\
\hlt{Key Insight}: Absolute continuity compares null sets; it does not compare finiteness, $\sigma$-finiteness, or size.
\end{example}

\begin{theorem} \label{thm:radon-nikodym-sigma-finite} \hlt{Radon--Nikodym Theorem}\\
Let $\lambda,\mu$ be $\sigma$-finite measures on $(X,\mathcal{M})$ with $\lambda\ll\mu$. Then there is a measurable $f:X\to[0,+\infty]$, unique $\mu$-a.e. and finite $\mu$-a.e., such that $\lambda(E)=\int_Ef\,d\mu$ for all $E\in\mathcal{M}$. Write $f=d\lambda/d\mu$.
\begin{proof}
Assume first $\lambda(X),\mu(X)<+\infty$.\\
For $c>0$, let $(P_k,N_k)$ be Hahn for $\lambda-kc\mu$, put $A_k=N_k\setminus\bigcup_{j<k}A_j$, and $B=X\setminus\bigcup_{k\geq1}A_k$.\\
Then $A_k\subseteq N_k$ and, for $k\geq2$, $A_k\subseteq P_{k-1}$; hence $E\subseteq A_k$ implies $(k-1)c\,\mu(E)\leq\lambda(E)\leq kc\,\mu(E)$.\\
Also $B\subseteq P_k$ for all $k$, so $\mu(B)=\lambda(B)=0$.\\
Define $f_c\defeq\sum_{k\geq1}(k-1)c\,\mathbf{1}_{A_k}$. Then $\int_E f_c\,d\mu\leq\lambda(E)\leq\int_E f_c\,d\mu+c\mu(X)$ for all $E\in\mathcal{M}$.\\
With $c_n=2^{-n}$ and $f_n=f_{c_n}$, $\norm{f_n-f_m}_1\leq(2^{-n}+2^{-m})\mu(X)\leq2^{-n+1}\mu(X)$ for $m\geq n$, so $f_n\to f$ in $L^1$ for some $f\geq0$. Letting $n\to\infty$ gives $\lambda(E)=\int_Ef\,d\mu$.\\
For $\sigma$-finite $\lambda,\mu$, choose disjoint $Y_r\in\mathcal{M}$ with $X=\bigsqcup_rY_r$ and $\lambda(Y_r),\mu(Y_r)<+\infty$.\\
Apply the finite case to $\lambda|_{Y_r},\mu|_{Y_r}$, get $f_r$, and set $f=\sum_r f_r\mathbf{1}_{Y_r}$. Then $\lambda(E)=\sum_r\lambda(E\cap Y_r)=\int_Ef\,d\mu$.\\
Since $\int_{Y_r}f\,d\mu=\lambda(Y_r)<+\infty$, $f<+\infty$ $\mu$-a.e. If $f,g$ both represent $\lambda$, then $f=g$ $\mu$-a.e. on each $Y_r$ by Proposition \ref{prop:l1-level-sets-null-and-indefinite-integrals}; hence $f=g$ $\mu$-a.e. on $X$.\\
\hlt{Structural Commentary}: Hahn decompositions of $\lambda-c\mu$ build lower step approximations to the desired density.\\
\hlt{Key Insight}: The density is recovered by measuring where $\lambda$ lies between consecutive multiples of $\mu$.
\end{proof}
\end{theorem}

\begin{example} \label{ex:radon-nikodym-needs-sigma-finite-source} \hlt{A Non-$\sigma$-Finite Failure}\\
Let $X=[0,1]$, $\mathcal{M}=\mathcal{B}([0,1])$, $\mu=\#$, and $\lambda=m$. Since $\mu(E)=0\Leftrightarrow E=\emptyset$, $\lambda\ll\mu$.\\
If $\lambda(E)=\int_E f\,d\mu$ for some $f\geq0$, then $0=\lambda(\{x\})=\int_{\{x\}}f\,d\mu=f(x)$ for every $x$, so $\lambda([0,1])=0$, contradiction.\\
Thus RN can fail when $\mu$ is not $\sigma$-finite.
\end{example}

\begin{example} \label{ex:countable-cocountable-radon-nikodym-failure} \hlt{Another Non-$\sigma$-Finite Failure}\\
Let $X$ be uncountable and $\mathcal{M}\defeq\{E\subseteq X:E\text{ is countable or }X\setminus E\text{ is countable}\}$.\\
Set $\mu(E)=\#E$ for finite $E$ and $\mu(E)=+\infty$ otherwise; set $\lambda(E)=0$ for countable $E$ and $\lambda(E)=+\infty$ otherwise. Since $\mu(E)=0\Leftrightarrow E=\emptyset$, $\lambda\ll\mu$. If $\lambda(E)=\int_Ef\,d\mu$ with $f\geq0$, then $0=\lambda(\{x\})=\int_{\{x\}}f\,d\mu=f(x)$ for all $x$, so $\lambda(X)=0$, contradiction.\\
\hlt{Structural Commentary}: This is the countable--cocountable version of Example \ref{ex:radon-nikodym-needs-sigma-finite-source}.\\
\hlt{Key Insight}: Non-$\sigma$-finite references can erase all singleton tests.
\end{example}

\begin{remark} \label{rem:rn-derivative-integrability-finiteness} \hlt{When the Radon--Nikodym Derivative Is Integrable}\\
In Theorem \ref{thm:radon-nikodym-sigma-finite}, $\frac{d\lambda}{d\mu}\in L^1(\mu)\Leftrightarrow\lambda(X)<+\infty$, since $\int_X(d\lambda/d\mu)\,d\mu=\lambda(X)$.
\end{remark}

\begin{theorem} \label{thm:radon-nikodym-extended-dominated-measure} \hlt{Extended-Valued Radon--Nikodym Theorem}\\
Let $\mu$ be $\sigma$-finite, let $\lambda$ be a measure on $(X,\mathcal{M})$, and assume $\lambda\ll\mu$.\\
Then $\exists$ measurable $f:X\to[0,+\infty]$ such that $\lambda(E)=\int_Ef\,d\mu$ for all $E\in\mathcal{M}$; any two such $f$ agree $\mu$-a.e.
\begin{proof}
Assume first $\mu(X)<+\infty$. Let $(P_n,N_n)$ be Hahn for $\lambda-n\mu$, and put $P=\bigcap_{n\geq1}P_n$, $N=\bigcup_{n\geq1}N_n$.\\
If $E\subseteq N_n$, then $\lambda(E)\leq n\mu(E)$; hence $\lambda|_N$ is $\sigma$-finite and Theorem \ref{thm:radon-nikodym-sigma-finite} gives $g:N\to[0,+\infty)$ with $\lambda(E\cap N)=\int_{E\cap N}g\,d\mu$. If $E\subseteq P$, then $\lambda(E)\geq n\mu(E)$ for all $n$, so $\mu(E)>0\Rightarrow\lambda(E)=+\infty$, while $\mu(E)=0\Rightarrow\lambda(E)=0$. Thus $f=g\mathbf{1}_N+\infty\cdot\mathbf{1}_P$ represents $\lambda$.\\
For general $\sigma$-finite $\mu$, choose $X=\bigsqcup_rY_r$ with $\mu(Y_r)<+\infty$, apply the finite-$\mu$ case to each restriction, and set $f=\sum_rf_r\mathbf{1}_{Y_r}$. Then $\lambda(E)=\sum_r\lambda(E\cap Y_r)=\int_Ef\,d\mu$. If $f,h$ represent $\lambda$, then on each $Y_r$, $\int_Af\,d\mu=\int_Ah\,d\mu$ for all $A\subseteq Y_r$; applying this to $\{f<h\}$ and $\{h<f\}$ gives $f=h$ $\mu$-a.e.\\
\hlt{Structural Commentary}: If $\lambda$ is not $\sigma$-finite, the density may have a genuinely infinite part.\\
\hlt{Key Insight}: Hahn decompositions isolate finite and infinite density regions.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:lebesgue-decomposition-measures} \hlt{Lebesgue Decomposition Theorem}\\
Let $\lambda,\mu$ be $\sigma$-finite measures on $(X,\mathcal{M})$. Then $\exists!$ measures $\lambda_s,\lambda_a$ with $\lambda=\lambda_s+\lambda_a$, $\lambda_s\perp\mu$, and $\lambda_a\ll\mu$.
\begin{proof}
Set $\rho=\lambda+\mu$. Choose $f=d\lambda/d\rho$ and $g=d\mu/d\rho$.\\
Let $A=\{g=0\}$, $B=\{g>0\}$, $\lambda_s(E)\defeq\lambda(E\cap A)$, and $\lambda_a(E)\defeq\lambda(E\cap B)$. Since $\mu(A)=\int_Ag\,d\rho=0$, $\lambda_s\perp\mu$.\\
If $\mu(E)=0$, then $g=0$ $\rho$-a.e. on $E$, so $\rho(E\cap B)=0$ and $\lambda_a(E)=0$; hence $\lambda_a\ll\mu$.\\
For uniqueness, let $(\tilde{\lambda}_s,\tilde{\lambda}_a)$ be another decomposition.\\
Since $\lambda_s\perp\mu$ and $\tilde{\lambda}_a\ll\mu$, also $\lambda_s\perp\tilde{\lambda}_a$; choose $X=C\cup D$ with $\lambda_s(C)=0$ and $\tilde{\lambda}_a(D)=0$.\\
Then $\lambda_s(E)=\lambda_s(E\cap D)\leq\lambda(E\cap D)=\tilde{\lambda}_s(E\cap D)\leq\tilde{\lambda}_s(E)$. By symmetry, $\lambda_s=\tilde{\lambda}_s$, hence $\lambda_a=\tilde{\lambda}_a$.\\
\hlt{Structural Commentary}: Every measure decomposes into a $\mu$-visible part and a part living on a $\mu$-null set.\\
\hlt{Key Insight}: Differentiate both measures with respect to their sum, then split where the $\mu$-density vanishes.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:radon-nikodym-change-of-measure-calculus} \hlt{Radon--Nikodym Calculus}\\
Let all measures below be $\sigma$-finite.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\lambda\ll\mu$, $g\geq0$ measurable $\Rightarrow\int_Xg\,d\lambda=\int_Xg(d\lambda/d\mu)\,d\mu$.
\item $\nu\ll\lambda\ll\mu\Rightarrow d\nu/d\mu=(d\nu/d\lambda)(d\lambda/d\mu)$ $\mu$-a.e.
\item $\lambda_1,\lambda_2\ll\mu\Rightarrow d(\lambda_1+\lambda_2)/d\mu=d\lambda_1/d\mu+d\lambda_2/d\mu$ $\mu$-a.e.
\item $\lambda\ll\mu\ll\lambda\Rightarrow d\lambda/d\mu=1/(d\mu/d\lambda)$ $\mu$-a.e.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $g=\mathbf{1}_E$, $\int_Xg\,d\lambda=\lambda(E)=\int_E(d\lambda/d\mu)\,d\mu=\int_Xg(d\lambda/d\mu)\,d\mu$.\\
By linearity the identity holds for nonnegative simple $g$; for measurable $g\geq0$, take simple $g_n\uparrow g$ and apply monotone convergence to both sides.
\item Let $h\defeq(d\nu/d\lambda)(d\lambda/d\mu)$.\\
For every $E\in\mathcal{M}$, (i) with $g=\mathbf{1}_E(d\nu/d\lambda)$ gives $\nu(E)=\int_E(d\nu/d\lambda)\,d\lambda=\int_Eh\,d\mu$.\\
Hence $h$ represents $\nu$ relative to $\mu$, so $h=d\nu/d\mu$ $\mu$-a.e.
\item Let $h\defeq d\lambda_1/d\mu+d\lambda_2/d\mu$. For every $E\in\mathcal{M}$, $\int_Eh\,d\mu=\lambda_1(E)+\lambda_2(E)=(\lambda_1+\lambda_2)(E)$.\\
Hence $h$ represents $\lambda_1+\lambda_2$ relative to $\mu$, so uniqueness gives the identity.
\item Apply (ii) to $\mu\ll\lambda\ll\mu$. Since $d\mu/d\mu=1$, $1=(d\mu/d\lambda)(d\lambda/d\mu)$ $\mu$-a.e.\\
Also $\mu\ll\lambda$ gives $d\mu/d\lambda>0$ $\mu$-a.e.; divide to obtain $d\lambda/d\mu=1/(d\mu/d\lambda)$ $\mu$-a.e.
\end{enumerate}
\hlt{Structural Commentary}: Radon--Nikodym derivatives obey chain and linearity rules.\\
\hlt{Key Insight}: Prove identities by integrating over arbitrary measurable sets, then invoke uniqueness.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:absolute-continuous-singular-zero-measure} \hlt{Absolutely Continuous and Singular Forces Zero}\\
For measures $\lambda,\mu$, $\lambda\ll\mu$ and $\lambda\perp\mu$ imply $\lambda=0$. For a charge $\eta$, $\abs{\eta}\perp\mu$ implies $\eta^+\perp\mu$ and $\eta^-\perp\mu$.
\begin{proof}
If $\lambda\perp\mu$, choose $X=A\cup B$ with $\lambda(A)=0$ and $\mu(B)=0$. Since $\lambda\ll\mu$, $\lambda(B)=0$, hence $\lambda(X)=0$.\\
If $\abs{\eta}\perp\mu$, choose $X=A\cup B$ with $\abs{\eta}(A)=0$ and $\mu(B)=0$. Since $\eta^\pm\leq\abs{\eta}$, $\eta^\pm(A)=0$, so $\eta^\pm\perp\mu$.\\
\hlt{Structural Commentary}: Absolute continuity and singular support are incompatible except at zero.\\
\hlt{Key Insight}: Visible and singular forces zero.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:radon-nikodym-for-charges} \hlt{Radon--Nikodym for Charges}\\
Let $\mu$ be $\sigma$-finite and let $\lambda$ be a charge with $\lambda\ll\mu$. Then $\exists f\in\mathscr{L}^1(\mu)$ such that, for all $E\in\mathcal{M}$, $\lambda(E)=\int_E f\,d\mu$, $\lambda^+(E)=\int_E f^+\,d\mu$, $\lambda^-(E)=\int_E f^-\,d\mu$, and $\abs{\lambda}(E)=\int_E\abs{f}\,d\mu$.
\begin{proof}
Since $\abs{\lambda}\ll\mu$, $\lambda^\pm\ll\mu$. By Radon--Nikodym, choose $f_\pm\geq0$ with $\lambda^\pm(E)=\int_Ef_\pm\,d\mu$. Since $\lambda^+\perp\lambda^-$, $f_+f_-=0$ $\mu$-a.e. Set $f=f_+-f_-$. Then $\int_X\abs{f}\,d\mu=\int_X(f_++f_-)\,d\mu=\abs{\lambda}(X)<+\infty$, so $f\in\mathscr{L}^1(\mu)$, $f^+=f_+$, $f^-=f_-$ $\mu$-a.e., and $\lambda(E)=\lambda^+(E)-\lambda^-(E)=\int_Ef\,d\mu$.\\
\hlt{Structural Commentary}: Signed absolute continuity is reduced to the two positive Jordan pieces.\\
\hlt{Key Insight}: Differentiate $\lambda^+$ and $\lambda^-$ separately, then subtract.
\end{proof}
\end{corollary}

\begin{example} \label{ex:stieltjes-basic-generators-rn-classification} \hlt{Radon--Nikodym Classification of Basic Stieltjes Generators}\\
Let $m$ be Lebesgue measure on $\mathcal{B}(\R)$, and let $\lambda_i$ be the Borel--Stieltjes measure generated by $g_i$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $g_1(x)=2x$: $\lambda_1=2m$, so $\lambda_1\ll m$, $d\lambda_1/dm=2$, $m\ll\lambda_1$, $dm/d\lambda_1=1/2$, and $\lambda_1(\R)=+\infty$.
\item $g_2(x)=\arctan x$: $\lambda_2(E)=\int_E(1+x^2)^{-1}\,dm$, so $d\lambda_2/dm=(1+x^2)^{-1}$, $\lambda_2(\R)=\pi$, $m\ll\lambda_2$, and $dm/d\lambda_2=1+x^2$.
\item $g_3=\mathbf{1}_{[0,+\infty)}$: $\lambda_3=\delta_0$,\\
so $\lambda_3\perp m$, $\lambda_3(\R)=1$, $\lambda_3\not\ll m$, and $m\not\ll\lambda_3$.
\item $g_4(x)=x\mathbf{1}_{[0,+\infty)}(x)$: $\lambda_4(E)=m(E\cap[0,+\infty))$, so $\lambda_4\ll m$, $d\lambda_4/dm=\mathbf{1}_{[0,+\infty)}$, $m\not\ll\lambda_4$, and $\lambda_4(\R)=+\infty$.
\end{enumerate}
Thus $\lambda_1,\lambda_2,\lambda_4\ll m$, $\lambda_3\perp m$, only $\lambda_2,\lambda_3$ are finite, and $m\ll\lambda_i$ exactly for $i=1,2$.\\
\hlt{Structural Commentary}: Stieltjes generators split naturally into density, atom, and restriction examples.\\
\hlt{Key Insight}: A positive density makes null sets equivalent; an atom lives on a Lebesgue-null set.
\end{example}

\begin{theorem} \label{thm:charges-total-variation-banach} \hlt{Charges Form a Banach Space}\\
Let $\operatorname{ca}(X,\mathcal{M})\defeq\{\lambda:\mathcal{M}\to\R:\lambda\text{ is a charge}\}$ and $\norm{\lambda}_{\mathrm{var}}\defeq\abs{\lambda}(X)$.\\
With pointwise operations, $(\operatorname{ca}(X,\mathcal{M}),\norm{\cdot}_{\mathrm{var}})$ is Banach.
\begin{proof}
Linearity follows from charge linearity.\\
Definiteness follows from $\abs{\lambda}(X)=0\Rightarrow\abs{\lambda}(E)=0\ \forall E$, homogeneity from $\abs{a\lambda}=|a|\abs{\lambda}$, and the triangle inequality from $\sum_j\abs{(\lambda+\eta)(A_j)}\leq\sum_j\abs{\lambda(A_j)}+\sum_j\abs{\eta(A_j)}$ over finite measurable partitions $(A_j)$.\\
Let $(\lambda_n)$ be variation-Cauchy. Since $\abs{\lambda_n(E)-\lambda_m(E)}\leq\norm{\lambda_n-\lambda_m}_{\mathrm{var}}$, define $\lambda(E)\defeq\lim_n\lambda_n(E)$; then $\lambda$ is finitely additive. For fixed $n$ and finite disjoint $(A_j)$, $\sum_j\abs{(\lambda-\lambda_n)(A_j)}\leq\limsup_m\norm{\lambda_m-\lambda_n}_{\mathrm{var}}$.\\
For disjoint $(E_k)$, set $E=\bigcup_kE_k$ and $R_K=\bigcup_{k>K}E_k$. Choose $n$ with the preceding limsup $<\varepsilon$.\\
Since $\lambda_n(R_K)\to0$, for large $K$, $\abs{\lambda(R_K)}\leq\abs{\lambda_n(R_K)}+\abs{(\lambda-\lambda_n)(R_K)}<2\varepsilon$. Hence $\lambda(R_K)\to0$, so $\lambda$ is countably additive. Taking suprema over finite partitions gives $\norm{\lambda-\lambda_n}_{\mathrm{var}}\leq\limsup_m\norm{\lambda_m-\lambda_n}_{\mathrm{var}}\to0$.\\
\hlt{Structural Commentary}: Total variation is the signed-measure analogue of an $L^1$ norm.\\
\hlt{Key Insight}: Variation-Cauchy controls every finite partition.
\end{proof}
\end{theorem}

\begin{definition} \label{def:lebesgue-averages-and-absolute-continuity} \hlt{Averages and Absolute Continuity}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Average}: for $f\in\mathscr{L}^1(E)$ and $A\subseteq E$, $m(A)>0$, set $[f:A]\defeq m(A)^{-1}\int_Af\,dm$.
\item \hlt{Density ratio}: for measurable $B$ and cube or ball $Q$, $m(Q)>0$, set $[B:Q]\defeq m(B\cap Q)/m(Q)$.
\item \hlt{Indefinite integral}: for $f\in\mathscr{L}^1([a,b])$, $F(x)=F(a)+\int_a^xf(t)\,dt$.
\item \hlt{Absolute continuity}: $F:[a,b]\to\R$ is absolutely continuous if $\forall\varepsilon>0,\exists\delta>0$,\\
finite disjoint $(a_j,b_j)\subseteq[a,b]$ satisfy $\sum_j(b_j-a_j)<\delta\Rightarrow\sum_j\abs{F(b_j)-F(a_j)}<\varepsilon$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:bounded-variation-one-variable} \hlt{Bounded Variation}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Total variation}: for $F:[a,b]\to\R$, $V_a^b(F)\defeq\sup\{\sum_{j=1}^N\abs{F(x_j)-F(x_{j-1})}:a=x_0<\cdots<x_N=b\}$.
\item \hlt{Bounded variation}: $F$ has bounded variation on $[a,b]$ if $V_a^b(F)<+\infty$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:vitali-coverings-and-density-points} \hlt{Vitali Coverings and Density Points}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $A\subseteq M$, $\mathcal{V}$ is a \hlt{Vitali covering} of $A$ if $\forall p\in A,\forall r>0,\exists V\in\mathcal{V}$ with $p\in V$ and $\diam(V)<r$.
\item For measurable $E\subseteq\R^n$, its \hlt{density-point set} is $\operatorname{dp}(E)\defeq\{p\in\R^n:\lim_{Q\downarrow p}[E:Q]=1\}$, with $Q$ cubes, equivalently balls, shrinking to $p$.
\end{enumerate}
\end{definition}

\begin{theorem} \label{thm:vitali-covering-lemma-balls-cubes} \hlt{Vitali Covering Lemma for Balls and Cubes}\\
Let $E\subseteq\R^n$, $m^\ast(E)<+\infty$, and let $\mathcal{V}$ be a Vitali cover of $E$ by closed balls or axis-parallel cubes.\\
Then $\exists$ pairwise disjoint $(V_j)\subseteq\mathcal{V}$ with $m^\ast(E\setminus\bigcup_{j\geq1}V_j)=0$.
\begin{proof}
Fix open $O\supseteq E$ with $m(O)<m^\ast(E)+\varepsilon$, and replace $\mathcal{V}$ by $\{V\in\mathcal{V}:V\subseteq O\}$.\\
Greedily choose a maximal disjoint $(V_j)$ by dyadic diameter scales.\\
For a fixed dilation $V\mapsto V^\ast$, if $W\cap V\neq\emptyset$ and $\diam W\leq2\diam V$, then $W\subseteq V^\ast$ and $m(V^\ast)\leq C_nm(V)$.\\
Let $R=E\setminus\bigcup_jV_j$. Fix $N$ and $p\in R$.\\
Since $V_1,\ldots,V_N$ are closed, choose $W\in\mathcal{V}$ with $p\in W$ and $W\cap\bigcup_{j\leq N}V_j=\emptyset$. Maximality gives some $j>N$ with $W\cap V_j\neq\emptyset$ and comparable diameter, hence $p\in W\subseteq V_j^\ast$. Thus $R\subseteq\bigcup_{j>N}V_j^\ast$.\\
Therefore $m^\ast(R)\leq C_n\sum_{j>N}m(V_j)\to0$, because $\sum_jm(V_j)=m(\bigsqcup_jV_j)\leq m(O)<+\infty$. Hence $m^\ast(R)=0$.\\
\hlt{Structural Commentary}: Vitali selection gives a disjoint cover modulo null sets.\\
\hlt{Key Insight}: $R\subseteq\bigcup_{j>N}V_j^\ast$ and disjoint tails vanish.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:lebesgue-density-theorem} \hlt{Lebesgue Density Theorem}\\
For measurable $E\subseteq\R^n$, $\lim_{Q\downarrow p}[E:Q]=\mathbf{1}_E(p)$ for a.e. $p$, where $Q$ ranges over cubes, equivalently balls.\\
Equivalently, $m(E\triangle\operatorname{dp}(E))=0$.
\begin{proof}
Assume $m(E)<+\infty$. For $a\in(0,1)$ set $E_a\defeq\{p\in E:\liminf_{Q\downarrow p}[E:Q]<a\}$.\\
Cubes $Q\ni p$ with $[E:Q]<a$ form a Vitali cover of $E_a$. For open $O\supseteq E_a$ with $m(O)<m(E_a)+\varepsilon$, Theorem \ref{thm:vitali-covering-lemma-balls-cubes} gives disjoint $(Q_j)\subseteq O$ covering $E_a$ modulo null sets.\\
Hence $m(E_a)\leq\sum_jm(E_a\cap Q_j)\leq\sum_jm(E\cap Q_j)<a\sum_jm(Q_j)\leq a\,m(O)<a(m(E_a)+\varepsilon)$.\\
Let $\varepsilon\downarrow0$; since $a<1$, $m(E_a)=0$. Thus $p\in E\setminus\bigcup_{a\in\Q\cap(0,1)}E_a$ has density $1$. Apply same argument to bounded pieces of $E^c$ to get density $0$ a.e. on $E^c$; then exhaust arbitrary $E$ by bounded pieces.\\
\hlt{Structural Commentary}: Measurable sets are locally $0$ or $1$ a.e.\\
\hlt{Key Insight}: Vitali turns bad density into $m(E_a)\le a\,m(E_a)$.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:density-points-commute-with-slices} \hlt{Density Points Commute with Slicing Almost Everywhere}\\
Let $E\subseteq\R^n\times\R^k$ be measurable. For a.e. $x\in\R^n$, $\operatorname{dp}(E_x)=\{y:(x,y)\in\operatorname{dp}(E)\}$ up to a null subset of $\R^k$.
\begin{proof}
Set $D=\operatorname{dp}(E)$ and $N=E\triangle D$.\\
By Theorem \ref{thm:lebesgue-density-theorem}, $m_{n+k}(N)=0$; hence Theorem \ref{thm:zero-slices-null-set-criterion} gives $m_k(N_x)=0$ for a.e. $x$.\\
Intersecting with the Fubini good set, $E_x,D_x$ are measurable and $m_k(E_x\triangle D_x)=m_k((E\triangle D)_x)=0$.\\
Applying Theorem \ref{thm:lebesgue-density-theorem} in $\R^k$ gives $m_k(E_x\triangle\operatorname{dp}(E_x))=0$.\\
Therefore $m_k(\operatorname{dp}(E_x)\triangle D_x)=0$; and $m_k(\operatorname{dp}(E_x)\triangle D_x)=0$.
This is the claimed equality modulo null sets.\\
\hlt{Structural Commentary}: Almost-everywhere local fullness is compatible with almost-everywhere slicing.\\
\hlt{Key Insight}: Null ambient errors have null slices for almost every base point.
\end{proof}
\end{corollary}

\begin{lemma} \label{lem:large-average-set-bound} \hlt{Large Averages Have Small Outer Measure}\\
For a cube $Q_0\subseteq\R^n$, $g\ge0$ in $\mathscr{L}^1(Q_0)$, and $\lambda>0$,\\
set $A_\lambda\defeq \left\{p\in Q_0:\forall r>0\ \exists\text{ cube }Q\subseteq Q_0,\ p\in Q,\ \diam Q<r,\ [g:Q]>\lambda\right\}$ for $m^\ast(A_\lambda)\leq \frac{1}{\lambda}\int_{Q_0} g$.
\begin{proof}
Let $\mathcal{V}_\lambda=\{Q\subseteq Q_0:Q\text{ is a cube and }[g:Q]>\lambda\}$. By definition, $\mathcal{V}_\lambda$ is a Vitali cover of $A_\lambda$; choose disjoint $(Q_j)\subseteq\mathcal{V}_\lambda$ with $m^\ast(A_\lambda\setminus\bigcup_jQ_j)=0$. Then $\lambda m^\ast(A_\lambda) \le \lambda\sum_jm(Q_j) <\sum_j\int_{Q_j}g \le\int_{Q_0}g$.\\
\hlt{Structural Commentary}: Weak type for local averages.\\
\hlt{Key Insight}: Disjoint Vitali cubes turn pointwise large averages into one integral bound.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:lebesgue-average-value-theorem} \hlt{Lebesgue Average Value Theorem}\\
For $f\in\mathscr{L}^1_{\mathrm{loc}}(\R^n)$, $\lim\limits_{Q\downarrow p}[f:Q]=f(p)$ for a.e. $p\in\R^n$ with $Q$ cubes, equivalently balls, shrinking to $p$.
\begin{proof}
Fix a cube $Q_0$; it suffices to prove the claim for a.e. $p\in Q_0^\circ$.\\
For $\alpha>0$, let $B_\alpha=\left\{p\in Q_0^\circ:\limsup\limits_{Q\downarrow p}|[f:Q]-f(p)|>\alpha\right\}$.\\
Choose $h\in C(Q_0)$ with $\|f-h\|_{L^1(Q_0)}<\eta$, and set $g=|f-h|$. Since $[h:Q]\to h(p)$, $B_\alpha\subseteq \{g>\alpha/3\}\cup A_{\alpha/3}(g)$.\\
By Markov and Lemma \ref{lem:large-average-set-bound}, $m^\ast(B_\alpha)\le \frac{3}{\alpha}\int_{Q_0}g+\frac{3}{\alpha}\int_{Q_0}g <\frac{6\eta}{\alpha}$.\\
Let $\eta\downarrow0$; then $m(B_\alpha)=0$. Count over $\alpha\in\Q_{>0}$ and exhaust $\R^n$ by cube interiors.\\
\hlt{Structural Commentary}: Local averages recover the original integrable function almost everywhere.\\
\hlt{Key Insight}: Continuous approximation handles the good part; Vitali's estimate makes the bad part small.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:lebesgue-antiderivative-theorem} \hlt{Lebesgue's Antiderivative Theorem}\\
For $f\in\mathscr{L}^1([a,b])$, define $F(x)=c+\int_a^x f(t)\,dt$. Then $F\in AC([a,b])$ and $F'(x)=f(x)$ for a.e. $x\in[a,b]$.
\begin{proof}
Given $\varepsilon>0$, choose $\delta>0$ so that $m(A)<\delta$ implies $\int_A|f|<\varepsilon$.\\
If $(\alpha_j,\beta_j)$ are disjoint and $\sum_j(\beta_j-\alpha_j)<\delta$, then $\sum_j|F(\beta_j)-F(\alpha_j)| \le \sum_j\int_{\alpha_j}^{\beta_j}|f| =\int_{\bigcup_j(\alpha_j,\beta_j)}|f| <\varepsilon$, so $F\in AC([a,b])$. For $x,x+h\in[a,b]$, let $I_h(x)=(\min\{x,x+h\},\max\{x,x+h\})$.\\
Then $\frac{F(x+h)-F(x)}{h}=[f:I_h(x)]$. Extending $f$ by $0$ outside $[a,b]$, Theorem \ref{thm:lebesgue-average-value-theorem} in one dimension gives $[f:I_h(x)]\to f(x)$ for a.e. $x$. Hence $F'=f$ a.e.\\
\hlt{Structural Commentary}: The indefinite Lebesgue integral is the correct antiderivative for integrable functions.\\
\hlt{Key Insight}: Differentiating the integral is the same as shrinking the averaging interval.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:lebesgue-last-theorem-and-consequences} \hlt{Lebesgue's Last Theorem and Consequences}\\
On $[a,b]$:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $F$ monotone $\Rightarrow F'$ exists a.e.;
\item $F$ Lipschitz $\Rightarrow F'$ exists a.e.;
\item $F\in BV([a,b])\Rightarrow F'$ exists a.e.;
\item $F\in AC([a,b])\Rightarrow F\in BV([a,b])$, hence $F'$ exists a.e.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Assume $F$ nondecreasing. Let $D^+F(x)=\limsup\limits_{h\downarrow0}\frac{F(x+h)-F(x)}h$ and $D_-F(x)=\liminf\limits_{h\downarrow0}\frac{F(x+h)-F(x)}h$.\\
For $\alpha<\beta$ in $\Q$, set $A_{\alpha,\beta}=\{D_-F<\alpha<\beta<D^+F\}$. The right-secant Vitali selection of Figure \ref{fig:lebesgue-right-secants} gives $m(A_{\alpha,\beta})=0$: otherwise selected intervals with slopes $<\alpha$ and $>\beta$ contradict monotonicity of total rise. Also $m^\ast(\{D^+F>M\})\le \frac{F(b)-F(a)}{M}\to0$, so $D^+F=D_-F<\infty$ a.e.; hence $F'$ exists a.e.
\item If $F$ is $L$-Lipschitz, set $G_+(x)=F(x)+Lx$ and $G_-(x)=Lx-F(x)$. Then $G_\pm$ are nondecreasing, by (i) $\exists$ null sets $N_\pm$ s.t. $G_\pm'$ exist off $N_\pm$. For $x\notin N_+\cup N_-$, $F'(x)=\frac{G_+'(x)-G_-'(x)}{2}$, so $F'$ exists a.e.
\item If $F\in BV([a,b])$, write its Jordan decomposition $F=F(a)+V^+-V^-$ with $V^\pm$ nondecreasing. By (i), $\exists$ null sets $N_\pm$ s.t. $(V^\pm)'$ exist off $N_\pm$. For $x\notin N_+\cup N_-$, $F'(x)=(V^+)'(x)-(V^-)'(x)$, so $F'$ exists a.e.
\item If $F\in AC([a,b])$, choose $\delta$ for $\varepsilon=1$, and partition $a=x_0<\cdots<x_N=b$ with $x_i-x_{i-1}<\delta$. Grouping any refinement by these subintervals gives $V_a^b(F)\le N$, so $F\in BV([a,b])$. By (iii), $\exists Z\subset[a,b]$ with $m(Z)=0$ s.t. $F'(x)$ exists for all $x\in[a,b]\setminus Z$.
\end{enumerate}
\hlt{Structural Commentary}: Monotonicity drives all cases.\\
\hlt{Key Insight}: Vitali controls secants; decompose to monotone.
\end{proof}
\end{theorem}
