\subsection{Sigma-Algebras and Measures}

% **Role:** modern integration and almost-everywhere analysis.
%
% Sigma-algebras, measures, outer measure, measurable sets, Borel/Lebesgue measure, null sets, completion, charges, regularity.

Measure theory first builds measurable sets, then measurable functions and integrals. Outer measure covers from outside; measurability splits test sets without losing measure.

\begin{perspective} \label{per:measure-integration-roadmap} \hlt{Why Abstract Measure Comes First}\\
Lebesgue integration is built so that, under hypotheses such as domination, $f_n\to f$ implies $\int f_n\,d\mu\to\int f\,d\mu$; see Theorem \ref{thm:dominated-convergence-lebesgue}. The abstract data separate into a measurable space $(X,\mathcal{M})$, a countably additive measure $\mu:\mathcal{M}\to[0,+\infty]$, and an integral defined by approximation from simple functions.\\
\hlt{Key Insight}: Countable additivity is the set-level structure behind the convergence theorems.
\end{perspective}

\begin{definition} \label{def:sigma-algebra-and-measurable-space} \hlt{$\sigma$-Algebras and Measurable Spaces}\\
Let $X$ be a set. A family $\mathcal{M}\subseteq\mathcal{P}(X)$ is a \hlt{$\sigma$-algebra} on $X$ if:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\emptyset,X\in\mathcal{M}$;
\item $E\in\mathcal{M}$ implies $X\setminus E\in\mathcal{M}$;
\item $E_n\in\mathcal{M}$ for $n\in\N$ implies $\bigcup\limits_{n=1}^{\infty}E_n\in\mathcal{M}$.
\end{enumerate}
The pair $(X,\mathcal{M})$ is a \hlt{measurable space}; the members of $\mathcal{M}$ are \hlt{measurable sets}.
\end{definition}

\begin{proposition} \label{prop:sigma-algebra-closure-and-generation} \hlt{Closure and Generation for $\sigma$-Algebras}\\
Let $\mathcal{M}\subseteq\mathcal{P}(X)$ be a $\sigma$-algebra.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(E_n)_{n\in\N}\subseteq\mathcal{M}\Rightarrow\bigcap\limits_{n=1}^{\infty}E_n\in\mathcal{M}$.
\item If $(\mathcal{M}_\alpha)_{\alpha\in A}$ is a nonempty family of $\sigma$-algebras on $X$, then $\bigcap\limits_{\alpha\in A}\mathcal{M}_\alpha$ is a $\sigma$-algebra on $X$.
\item For $\mathcal{A}\subseteq\mathcal{P}(X)$, let $\mathscr{S}_{\mathcal{A}}$ be the family of $\sigma$-algebras $\mathcal{N}$ on $X$ with $\mathcal{A}\subseteq\mathcal{N}$.\\
Then $\sigma(\mathcal{A})\defeq\bigcap_{\mathcal{N}\in\mathscr{S}_{\mathcal{A}}}\mathcal{N}$ is the smallest $\sigma$-algebra containing $\mathcal{A}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $X\setminus E_n\in\mathcal{M}$, Theorem \ref{thm:metric-de-morgan-laws} gives $\bigcap_nE_n=X\setminus\bigcup_n(X\setminus E_n)\in\mathcal{M}$.
\item Set $\mathcal{N}\defeq\bigcap_{\alpha\in A}\mathcal{M}_\alpha$. Then $\emptyset,X\in\mathcal{N}$, and $E\in\mathcal{N}$ implies $X\setminus E\in\mathcal{N}$.\\
Likewise $(E_n)\subseteq\mathcal{N}$ implies $\bigcup_nE_n\in\mathcal{N}$.
\item The indexing class in the displayed definition is nonempty since $\mathcal{P}(X)$ belongs to it. By (ii), its intersection is a $\sigma$-algebra; it contains $\mathcal{A}$ and is contained in every $\sigma$-algebra containing $\mathcal{A}$.
\end{enumerate}
\hlt{Structural Commentary}: Generation is intersection over all admissible $\sigma$-algebras.\\
\hlt{Key Insight}: Intersections preserve universal closure properties.
\end{proof}
\end{proposition}

\begin{example} \label{ex:basic-sigma-algebras-and-borel} \hlt{Basic $\sigma$-Algebras}\\
Standard models:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Extremes: $\{\emptyset,X\}\subseteq\mathcal{M}\subseteq\mathcal{P}(X)$ for every $\sigma$-algebra $\mathcal{M}$ on $X$.
\item For $E\subseteq X$, $\sigma(\{E\})=\{\emptyset,E,X\setminus E,X\}$, with repetitions omitted.
\item On $\N$, if $O\defeq\{2n-1:n\in\N\}$, then $\sigma(\{O\})=\{\emptyset,O,\N\setminus O,\N\}$.
\item If $X$ is uncountable, $\mathcal{C}_X\defeq\{E\subseteq X:E\text{ is countable or }X\setminus E\text{ is countable}\}$\\
is the countable--cocountable $\sigma$-algebra.
\end{enumerate}
\hlt{Key Insight}: A $\sigma$-algebra records exactly which distinctions the theory is allowed to observe.
\end{example}

\begin{definition} \label{def:borel-and-extended-borel-sigma-algebras} \hlt{Borel and Extended Borel $\sigma$-Algebras}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $n\in\N$, the \hlt{Borel $\sigma$-algebra} on $\R^n$ is $\mathcal{B}(\R^n)\defeq\sigma(\{U\subseteq\R^n:U\text{ is open}\})$.\\
On $\R$, it is also generated by open intervals, closed intervals, half-open intervals, or open half-rays $(a,+\infty)$.
\item The \hlt{extended Borel $\sigma$-algebra} on $\overline{\R}$ is $\overline{\mathcal{B}}\defeq\{E\cup F:E\in\mathcal{B}(\R),\ F\subseteq\{-\infty,+\infty\}\}$.
\end{enumerate}
\end{definition}

\begin{definition} \label{def:monotone-classes} \hlt{Monotone Classes}\\
A nonempty $\mathcal{C}\subseteq\mathcal{P}(X)$ is a \hlt{monotone class} iff $(E_n)\subseteq\mathcal{C},\ E_n\uparrow E\Rightarrow E\in\mathcal{C}$ and $(F_n)\subseteq\mathcal{C},\ F_n\downarrow F\Rightarrow F\in\mathcal{C}$.\\
For $\mathcal{A}\subseteq\mathcal{P}(X)$, define $\mathfrak{M}_{\mathcal{A}}\defeq \{\mathcal{C}\subseteq\mathcal{P}(X):\mathcal{C}\text{ is a monotone class},\ \mathcal{A}\subseteq\mathcal{C}\}$; $\operatorname{mon}(\mathcal{A})\defeq\bigcap\mathfrak{M}_{\mathcal{A}}$.\\
Then $\operatorname{mon}(\mathcal{A})$ is the least monotone class containing $\mathcal{A}$, and $\operatorname{mon}(\mathcal{A})\subseteq\sigma(\mathcal{A})$.\\
\hlt{Key Insight}: Monotone closure captures limits, not finite unions.
\end{definition}

\begin{definition} \label{def:algebra-field-and-measure-on-algebra} \hlt{Algebras of Sets and Premeasures}\\
Let $X$ be a set.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item An \hlt{algebra of sets} (or \hlt{field of sets}) on $X$ is $\mathcal{A}\subseteq\mathcal{P}(X)$ such that $\emptyset,X\in\mathcal{A}$; $E\in\mathcal{A}\Rightarrow X\setminus E\in\mathcal{A}$;\\
and $E_1,\ldots,E_n\in\mathcal{A}\Rightarrow\bigcup_{j=1}^{n}E_j\in\mathcal{A}$. Equivalently, $\mathcal{A}$ is a ring of sets containing $X$.
\item A \hlt{premeasure} (or \hlt{measure on an algebra}) on $\mathcal{A}$ is $\mu:\mathcal{A}\to[0,+\infty]$ such that $\mu(\emptyset)=0$ and, whenever $(E_n)_{n\in\N}\subseteq\mathcal{A}$ is pairwise disjoint with $\bigcup_{n=1}^{\infty}E_n\in\mathcal{A}$, then $\mu\left(\bigcup_{n=1}^{\infty}E_n\right)=\sum_{n=1}^{\infty}\mu(E_n)$.
\end{enumerate}
\hlt{Key Insight}: Premeasures are countably additive where the algebra permits unions.
\end{definition}

\begin{theorem} \label{thm:monotone-class-theorem-algebras} \hlt{Monotone Class Theorem for Algebras}\\
Let $\mathcal{A}\subseteq\mathcal{P}(X)$ be an algebra and set $\mathcal{C}\defeq\operatorname{mon}(\mathcal{A})$. Then $\mathcal{C}=\sigma(\mathcal{A})$.
\begin{proof} Since $\sigma(\mathcal{A})$ is monotone and contains $\mathcal{A}$, $\mathcal{C}\subseteq\sigma(\mathcal{A})$. It suffices to show that $\mathcal{C}$ is a $\sigma$-algebra.\\
For complements, set $\mathcal{D}\defeq\{E\in\mathcal{C}:E^c\in\mathcal{C}\}$. Then $E_n\uparrow E$ in $\mathcal{D}$ implies $E_n^c\downarrow E^c$, and $E_n\downarrow E$ implies $E_n^c\uparrow E^c$; hence $\mathcal{D}$ is monotone. Since $\mathcal{A}\subseteq\mathcal{D}$, minimality gives $\mathcal{C}\subseteq\mathcal{D}$.\\
For intersections, fix $B\in\mathcal{C}$ and set $\mathcal{D}_B\defeq\{E\in\mathcal{C}:E\cap B\in\mathcal{C}\}$. This is monotone because intersections preserve $\uparrow,\downarrow$ limits. If $B\in\mathcal{A}$, then $\mathcal{A}\subseteq\mathcal{D}_B$, so $\mathcal{C}\subseteq\mathcal{D}_B$; hence $E\cap B\in\mathcal{C}$ for $E\in\mathcal{C}$, $B\in\mathcal{A}$. Fixing $E\in\mathcal{C}$ now gives $\mathcal{A}\subseteq\mathcal{D}_E$, hence $\mathcal{C}\subseteq\mathcal{D}_E$. Thus $\mathcal{C}$ is closed under finite intersections.\\
Complements and finite intersections give finite unions. If $(E_n)\subseteq\mathcal{C}$, then $F_N\defeq\bigcup_{n=1}^{N}E_n\in\mathcal{C}$ and $F_N\uparrow\bigcup_nE_n$, so $\bigcup_nE_n\in\mathcal{C}$. Hence $\mathcal{C}$ is a $\sigma$-algebra containing $\mathcal{A}$, and $\sigma(\mathcal{A})\subseteq\mathcal{C}$.\\
\hlt{Structural Commentary}: Finite Boolean closure plus monotone limits gives $\sigma$-closure.\\
\hlt{Key Insight}: Prove closure properties by showing the sets that have the property form a monotone class.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:monotone-class-algebra-test} \hlt{Monotone-Class Test}\\
Let $\mathcal{A}\subseteq\mathcal{P}(X)$ be an algebra and let $\mathcal{C}\subseteq\mathcal{P}(X)$ be a monotone class. If $\mathcal{A}\subseteq\mathcal{C}$, then $\sigma(\mathcal{A})\subseteq\mathcal{C}$.
\begin{proof} Since $\mathcal{A}\subseteq\mathcal{C}$ and $\mathcal{C}$ is monotone, $\operatorname{mon}(\mathcal{A})\subseteq\mathcal{C}$. By Theorem \ref{thm:monotone-class-theorem-algebras}, $\operatorname{mon}(\mathcal{A})=\sigma(\mathcal{A})$.\\
\hlt{Structural Commentary}: Algebra checks plus monotone stability control $\sigma(\mathcal{A})$.\\
\hlt{Key Insight}: Prove on the algebra; extend by monotone closure.
\end{proof}
\end{corollary}

\begin{definition} \label{def:measure-measurability-core-source-todo}\label{def:ring-sigma-ring-set-function} \hlt{Rings, $\sigma$-Rings, and Set Functions}\\
Let $\mathcal{A}\subseteq\mathcal{P}(X)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{A}$ is a \hlt{ring of sets} iff $\emptyset\in\mathcal{A}$ and $A,B\in\mathcal{A}\Rightarrow A\cup B,\ A\setminus B\in\mathcal{A}$
\item $\mathcal{A}$ is a \hlt{$\sigma$-ring} iff it is a ring and $(A_n)\subseteq\mathcal{A}\Rightarrow\bigcup_{n=1}^{\infty}A_n\in\mathcal{A}$
\item A \hlt{set function} on $\mathcal{A}$ is $\phi:\mathcal{A}\to[-\infty,+\infty]$ with $\{+\infty,-\infty\}\nsubseteq\phi(\mathcal{A})$.
\item $\phi$ is \hlt{additive} iff $A\cap B=\emptyset\Rightarrow\phi(A\cup B)=\phi(A)+\phi(B)$
\item $\phi$ is \hlt{countably additive} iff $(A_n)\subseteq\mathcal{A}$ pairwise disjoint, $\bigcup_nA_n\in\mathcal{A}$ $\Rightarrow$ $\phi\left(\bigcup_{n=1}^{\infty}A_n\right)=\sum_{n=1}^{\infty}\phi(A_n)$.
\end{enumerate}
\hlt{Structural Commentary}: Rings support finite operations; $\sigma$-rings support countable unions.
\end{definition}

\begin{definition} \label{def:abstract-measure-on-measurable-space} \hlt{Measures on Measurable Spaces}\\
Let $(X,\mathcal{M})$ be a measurable space. A \hlt{measure} on $(X,\mathcal{M})$ is a map $\mu:\mathcal{M}\to[0,+\infty]$ such that:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu(\emptyset)=0$;
\item if $(E_n)\subseteq\mathcal{M}$ is pairwise disjoint, then $\mu\left(\bigcup_{n=1}^{\infty}E_n\right)=\sum_{n=1}^{\infty}\mu(E_n)$
\end{enumerate}
The triple $(X,\mathcal{M},\mu)$ is a \hlt{measure space}.\\
\hlt{Key Insight}: Measure is countably additive size on the chosen observable sets.
\end{definition}

\begin{definition} \label{def:abstract-measures-finite-sigma-finite} \hlt{Finite and $\sigma$-Finite Measures}\\
Let $(X,\mathcal{M},\mu)$ be a measure space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For nonnegative terms, $\sum_{n=1}^{\infty}a_n\defeq\lim_{N\to\infty}\sum_{n=1}^{N}a_n\in[0,+\infty]$.
\item $\mu$ is \hlt{finite} iff $\mu(E)<+\infty$ for every $E\in\mathcal{M}$.
\item $\mu$ is \hlt{$\sigma$-finite} iff there exist $E_n\in\mathcal{M}$ such that $X=\bigcup_{n=1}^{\infty}E_n$, $\mu(E_n)<+\infty$ for $n\in\N$
\end{enumerate}
\end{definition}

\begin{definition} \label{def:charges-finite-signed-measures} \hlt{Charges}\\
Let $(X,\mathcal{M})$ be a measurable space. A \hlt{charge} is a map $\lambda:\mathcal{M}\to\R$ with $\lambda(\emptyset)=0$ such that for every pairwise disjoint sequence $(E_n)\subseteq\mathcal{M}$, the series $\sum_{n=1}^{\infty}\lambda(E_n)$ converges in $\R$ and $\lambda(\bigcup_{n=1}^{\infty}E_n)=\sum_{n=1}^{\infty}\lambda(E_n)$.\\
\hlt{Key Insight}: Charges are real-valued signed measures with unconditional sums.
\end{definition}

\begin{example} \label{ex:first-abstract-measures} \hlt{First Abstract Measures}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The zero set function $\mu(E)=0$ on $\mathcal{M}$ is a finite measure.
\item If $X\neq\emptyset$ and $\mathcal{M}=2^X$, $\mu(\emptyset)=0$ and $\mu(E)=+\infty$ for $E\neq\emptyset$ give a measure, neither finite nor $\sigma$-finite.
\item For $p\in X$, the \hlt{unit mass} $\delta_p(E)\defeq\mathbf{1}_E(p)$ on $2^X$ is a finite measure.
\item On $2^\N$, \hlt{counting measure} is $\#(E)$ for finite $E$ and $+\infty$ otherwise; it is $\sigma$-finite.\\
More generally, $(a_n)\subseteq[0,+\infty]$ gives $\mu(E)\defeq\sum_{n\in E}a_n$, and every measure on $2^\N$ is uniquely of this form.
\item If $X$ is uncountable and $\mathcal{M}=2^X$, $\mu(E)=0$ for countable $E$ and $\mu(E)=+\infty$ otherwise defines a measure.
\item On $2^\N$, $\mu(E)=0$ for finite $E$ and $\mu(E)=+\infty$ otherwise is not a measure, since $\N=\bigcup_{n=1}^{\infty}\{n\}$ gives $+\infty\neq0$; the constant $+\infty$ rule fails at $\emptyset$.
\item Lebesgue measure on $\mathcal{B}(\R)$ is $\sigma$-finite and gives length $b-a$ to standard intervals; Borel--Stieltjes measures come from increasing right-continuous functions as in Theorem \ref{thm:borel-stieltjes-measure-generation}.
\end{enumerate}
\hlt{Structural Commentary}: Measures may be zero, atomic, counting-type, cardinality-based, or geometric.\\
\hlt{Key Insight}: Countable additivity turns disjoint decompositions into nonnegative series.
\end{example}

\begin{definition} \label{def:outer-measure-generated-by-premeasure} \hlt{Outer Measure Generated by a Premeasure}\\
Let $\mu$ be a premeasure on an algebra $\mathcal{A}\subseteq\mathcal{P}(X)$. The \hlt{outer measure generated by $\mu$} is the set function $\mu^\ast:\mathcal{P}(X)\to[0,+\infty]$ defined by $\mu^\ast(B)\defeq\inf\{\sum_{j=1}^{\infty}\mu(E_j):E_j\in\mathcal{A},\ B\subseteq\bigcup_{j=1}^{\infty}E_j\}$ for $B\subseteq X$.
\end{definition}

\begin{lemma} \label{lem:generated-outer-measure-basic-properties} \hlt{Generated Outer-Measure Properties}\\
Let $\mu$ be a premeasure on an algebra $\mathcal{A}\subseteq\mathcal{P}(X)$, and let $\mu^\ast$ be as in Definition \ref{def:outer-measure-generated-by-premeasure}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu^\ast(\emptyset)=0$.
\item $C\subseteq D\subseteq X\Rightarrow\mu^\ast(C)\leq\mu^\ast(D)$.
\item $(B_n)\subseteq\mathcal{P}(X)\Rightarrow\mu^\ast(\bigcup_{n=1}^{\infty}B_n)\leq\sum_{n=1}^{\infty}\mu^\ast(B_n)$.
\item $E\in\mathcal{A}\Rightarrow\mu^\ast(E)=\mu(E)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item The cover $(\emptyset)_j$ gives $\mu^\ast(\emptyset)\leq0$, while $\mu^\ast\geq0$ by definition.
\item Every algebra cover of $D$ covers $C$; taking infima gives monotonicity.
\item If $\sum_n\mu^\ast(B_n)=+\infty$, there is nothing to prove.\\
Otherwise, for $\varepsilon>0$ choose covers $(E_{n,k})_{k\geq1}\subseteq\mathcal{A}$ of $B_n$ with $\sum_{k=1}^{\infty}\mu(E_{n,k})\leq\mu^\ast(B_n)+\varepsilon2^{-n}$.\\
Then $(E_{n,k})_{n,k}$ covers $\bigcup_nB_n$, hence $\mu^\ast(\bigcup_nB_n)\leq\sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\mu(E_{n,k})\leq\sum_n\mu^\ast(B_n)+\varepsilon$; let $\varepsilon\downarrow0$.
\item For $E\in\mathcal{A}$, the cover $E,\emptyset,\emptyset,\ldots$ gives $\mu^\ast(E)\leq\mu(E)$.\\
Conversely, if $E\subseteq\bigcup_nF_n$ with $F_n\in\mathcal{A}$, set $G_n\defeq E\cap F_n\setminus\bigcup_{k<n}F_k$. Then $(G_n)$ is disjoint in $\mathcal{A}$, $\bigcup_nG_n=E$, and $\mu(E)=\sum_n\mu(G_n)\leq\sum_n\mu(F_n)$. Taking infima over covers gives $\mu(E)\leq\mu^\ast(E)$.
\end{enumerate}
\hlt{Structural Commentary}: Generated outer measures extend premeasures and are monotone and subadditive.\\
\hlt{Key Insight}: Intersect an arbitrary cover with $E$ and disjointize inside the algebra.
\end{proof}
\end{lemma}

\begin{proposition} \label{prop:single-set-envelope-vs-generated-outer-measure} \hlt{Single-Set Envelope Comparison}\\
Let $\mu$ be a premeasure on an algebra $\mathcal{A}\subseteq\mathcal{P}(X)$, and let $\mu^\ast$ be generated by $\mu$. For $B\subseteq X$, set $\mu^\sharp(B)\defeq\inf\{\mu(C):C\in\mathcal{A},\ B\subseteq C\}$. Then $\mu^\sharp|_{\mathcal{A}}=\mu$ and $\mu^\ast\leq\mu^\sharp$.
\begin{proof} If $E\in\mathcal{A}$, then $E$ is admissible, so $\mu^\sharp(E)\leq\mu(E)$. \\
Conversely, $E\subseteq C\in\mathcal{A}$ implies $\mu(E)\leq\mu(C)$; taking infima gives $\mu(E)\leq\mu^\sharp(E)$.\\
If $B\subseteq C\in\mathcal{A}$, then $C,\emptyset,\emptyset,\ldots$ covers $B$, so $\mu^\ast(B)\leq\mu(C)$. Taking infima over $C$ gives $\mu^\ast(B)\leq\mu^\sharp(B)$.\\
\hlt{Structural Commentary}: One-set envelopes dominate generated outer measures.\\
\hlt{Key Insight}: A single algebra superset is a countable algebra cover.
\end{proof}
\end{proposition}

\begin{example} \label{ex:single-set-envelope-not-outer-measure} \hlt{Single-Set Envelopes Need Not Be Outer Measures}\\
Let $X$ be uncountable, let $\mathcal{A}$ be the finite--cofinite algebra on $X$, and set $\mu(E)\defeq0$ for finite $E$, $\mu(E)\defeq+\infty$ for cofinite $E$. Then $\mu$ is a premeasure, $\mu^\ast(B)=0$ iff $B$ is countable, and $\mu^\sharp(B)=0$ iff $B$ is finite.\\
For any countably infinite $S\subseteq X$, $\mu^\sharp(S)=+\infty$ while $S=\bigcup_{x\in S}\{x\}$ and $\sum_{x\in S}\mu^\sharp(\{x\})=0$, QED.\\
\hlt{Structural Commentary}: Countable covers can be strictly smaller than one-set covers.\\
\hlt{Key Insight}: Outer measure is built from countable covers precisely to recover countable subadditivity.
\end{example}

\begin{theorem} \label{thm:generated-caratheodory-extension} \hlt{Generated Caratheodory Extension}\\
Let $\mu$ be a premeasure on an algebra $\mathcal{A}\subseteq\mathcal{P}(X)$, let $\mu^\ast$ be generated by $\mu$,\\
and set $\mathcal{C}_\mu\defeq\{E\subseteq X:\forall B\subseteq X, \mu^\ast(B)=\mu^\ast(B\cap E)+\mu^\ast(B\setminus E)\}$.\\
Then $\mathcal{A}\subseteq\mathcal{C}_\mu$, $\mathcal{C}_\mu$ is a complete $\sigma$-algebra, and $\mu^\ast|_{\mathcal{C}_\mu}$ is a measure extending $\mu$.
\begin{proof}
Lemma \ref{lem:generated-outer-measure-basic-properties} makes $\mu^\ast$ an outer measure; Theorem \ref{thm:caratheodory-measurable-sets-sigma-algebra} makes $\mathcal{C}_\mu$ a $\sigma$-algebra, makes $\mu^\ast|_{\mathcal{C}_\mu}$ a measure, and gives completeness through zero sets. Let $E\in\mathcal{A}$ and $B\subseteq X$.\\
Subadditivity gives $\mu^\ast(B)\leq\mu^\ast(B\cap E)+\mu^\ast(B\setminus E)$.\\
Conversely, if $B\subseteq\bigcup_nF_n$ with $F_n\in\mathcal{A}$, then $F_n\cap E,F_n\setminus E\in\mathcal{A}$ and $\mu(F_n)=\mu(F_n\cap E)+\mu(F_n\setminus E)$, hence $\mu^\ast(B\cap E)+\mu^\ast(B\setminus E)\leq\sum_n\mu(F_n)$. Infimizing over covers gives the reverse inequality, so $E\in\mathcal{C}_\mu$.\\
Thus $\mathcal{A}\subseteq\mathcal{C}_\mu$, and Lemma \ref{lem:generated-outer-measure-basic-properties} gives $\mu^\ast|_{\mathcal{A}}=\mu$.\\
\hlt{Structural Commentary}: Caratheodory equality is the additivity domain.\\
\hlt{Key Insight}: Every algebra set is a clean splitter for the outer measure it generated.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:caratheodory-splitters-finite-subadditive-size} \hlt{Caratheodory Splitters for Finitely Subadditive Size}\\
Let $\alpha:\mathcal{P}(X)\to[0,+\infty)$ satisfy $0\leq\alpha(E)\leq\alpha(E\cup F)\leq\alpha(E)+\alpha(F)$ for all $E,F\subseteq X$,\\
and set $\mathcal{S}\defeq\{E\subseteq X:(\forall A\subseteq X)\alpha(A)=\alpha(A\cap E)+\alpha(A\setminus E)\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mathcal{S}\neq\emptyset$, then $\mathcal{S}$ is an algebra on $X$.
\item If $E,F\in\mathcal{S}$ and $E\cap F=\emptyset$, then $\alpha(E\cup F)=\alpha(E)+\alpha(F)$.
\item Nonemptiness is necessary: $\alpha\equiv1$ gives $\mathcal{S}=\emptyset$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Choose $E_0\in\mathcal{S}$. Taking $A=\emptyset$ gives $\alpha(\emptyset)=0$, hence $\emptyset,X\in\mathcal{S}$. If $E\in\mathcal{S}$, the defining equality is symmetric in $E$ and $X\setminus E$, so $X\setminus E\in\mathcal{S}$. If $E,F\in\mathcal{S}$ and $A\subseteq X$, applying $F$ to $A$ and $A\setminus(E\cap F)$ and applying $E$ to $A\cap F$ gives $\alpha(A)=\alpha(A\cap E\cap F)+\alpha(A\setminus(E\cap F))$; hence $E\cap F\in\mathcal{S}$. Complements give finite unions.
\item Apply the defining equality for $E$ to $A=E\cup F$: $\alpha(E\cup F)=\alpha(E)+\alpha(F)$.
\item If $\alpha\equiv1$ and $E\in\mathcal{S}$, taking $A=\emptyset$ gives $1=1+1$, impossible.
\end{enumerate}
\hlt{Structural Commentary}: The Caratheodory splitting equation is already Boolean at the finite level.\\
\hlt{Key Insight}: Nonempty splitters force $\alpha(\emptyset)=0$; without that, the clean-splitting class may vanish.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:sigma-finite-generated-extension-uniqueness} \hlt{$\sigma$-Finite Generated Extensions Are Unique}\\
Let $\mu$ be a premeasure on an algebra $\mathcal{A}\subseteq\mathcal{P}(X)$, let $\mu^\ast$ be generated by $\mu$, and let $\mathcal{C}_\mu$ be as in Theorem \ref{thm:generated-caratheodory-extension}.\\
If $X=\bigcup_nF_n$ with $F_n\in\mathcal{A}$ and $\mu(F_n)<+\infty$, then $\mu^\ast|_{\mathcal{C}_\mu}$ is the unique measure $\nu$ on $\mathcal{C}_\mu$ satisfying $\nu|_{\mathcal{A}}=\mu$.
\begin{proof}
Existence is Theorem \ref{thm:generated-caratheodory-extension}. For uniqueness, let $\nu$ be a measure on $\mathcal{C}_\mu$ with $\nu|_{\mathcal{A}}=\mu$.\\
If $\mu(X)<+\infty$, then for $E\in\mathcal{C}_\mu$ and every algebra cover $E\subseteq\bigcup_nA_n$, $\nu(E)\leq\sum_n\mu(A_n)$, hence $\nu(E)\leq\mu^\ast(E)$.\\
Applying this to $X\setminus E$ and using $\nu(E)+\nu(X\setminus E)=\nu(X)=\mu(X)=\mu^\ast(E)+\mu^\ast(X\setminus E)$ forces $\nu(E)=\mu^\ast(E)$.\\
In general, set $H_n\defeq\bigcup_{j=1}^{n}F_j$. The finite case on $\mathcal{A}|_{H_n}\defeq\{A\cap H_n:A\in\mathcal{A}\}$ gives $\nu(E\cap H_n)=\mu^\ast(E\cap H_n)$; since $E\cap H_n\uparrow E$, continuity from below gives $\nu(E)=\mu^\ast(E)$.\\
\hlt{Structural Commentary}: Uniqueness reduces to finite-mass exhaustions.\\
\hlt{Key Insight}: Existence is Caratheodory; uniqueness needs a finite-mass exhaustion.
\end{proof}
\end{theorem}

\begin{corollary} \label{cor:sigma-finite-premeasure-unique-on-generated-sigma-algebra} \hlt{Uniqueness on the Generated $\sigma$-Algebra}\\
Let $\mu$ be a premeasure on an algebra $\mathcal{A}\subseteq\mathcal{P}(X)$. If $X=\bigcup_nF_n$ with $F_n\in\mathcal{A}$ and $\mu(F_n)<+\infty$, then any measures $\nu_1,\nu_2$ on $\sigma(\mathcal{A})$ with $\nu_1|_{\mathcal{A}}=\nu_2|_{\mathcal{A}}=\mu$ satisfy $\nu_1=\nu_2$.
\begin{proof}
For $E\in\sigma(\mathcal{A})$, repeat the proof of Theorem \ref{thm:sigma-finite-generated-extension-uniqueness} with $\nu_1,\nu_2$ in place of $\nu,\mu^\ast$; finite-mass comparison on $F_1\cup\cdots\cup F_n$ and continuity from below give $\nu_1(E)=\nu_2(E)$.\\
\hlt{Structural Commentary}: $\sigma$-finite algebra data determine the generated measure.\\
\hlt{Key Insight}: Finite algebra exhaustions force uniqueness.
\end{proof}
\end{corollary}

\begin{example} \label{ex:rational-half-open-algebra-sigma-finiteness-necessary} \hlt{Why $\sigma$-Finiteness Matters}\\
Let $X=\Q\cap(0,1]$, and let $\mathcal{A}$ be the algebra of finite unions of sets $(a,b]\cap X$, where $0\leq a\leq b\leq1$ and $a,b\in\Q$. Then every nonempty $A\in\mathcal{A}$ is infinite and $\sigma(\mathcal{A})=\mathcal{P}(X)$, since each singleton $\{r\}$ is a countable intersection of sets $(a_n,r]\cap X$ with $a_n\uparrow r$. If $\mu_1$ is counting measure and $\mu_2=2\mu_1$ on $\mathcal{P}(X)$, then $\mu_1|_{\mathcal{A}}=\mu_2|_{\mathcal{A}}$, but $\mu_1(\{r\})\neq\mu_2(\{r\})$. Thus uniqueness may fail without $\sigma$-finiteness.\\
\hlt{Key Insight}: If the starting algebra has no finite nonempty pieces, it cannot detect the later singleton scale.
\end{example}

\begin{proposition} \label{prop:half-open-real-line-length-premeasure} \hlt{Half-Open Interval Algebra on $\R$}\\
Let $\mathcal{F}$ be the finite-union algebra generated by $(a,b]$, $(-\infty,b]$, $(a,+\infty)$, and $\R$, $a,b\in\R$. Define $\ell((a,b])\defeq b-a$ and $\ell((-\infty,b])=\ell((a,+\infty))=\ell(\R)\defeq+\infty$, extended additively over finite disjoint unions. Then
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{F}$ is an algebra on $\R$ and $\sigma(\mathcal{F})=\mathcal{B}(\R)$.
\item $\ell$ is a premeasure on $\mathcal{F}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Generator complements lie in $\mathcal{F}$, hence $\mathcal{F}$ is an algebra.\\
Since $\mathcal{F}\subseteq\mathcal{B}(\R)$ and $(a,b)=\bigcup_{n>1/(b-a)}(a,b-1/n]$, $\sigma(\mathcal{F})=\mathcal{B}(\R)$.
\item Normal forms give well-defined finite additivity. For countable additivity, reduce to $(a,b]=\bigsqcup_j(a_j,b_j]$: finite subunions give $\sum_j(b_j-a_j)\leq b-a$, while compactness after enlarging to $(a_j-\varepsilon_j,b_j+\varepsilon_j)$ gives $b-a\leq\sum_j(b_j-a_j)+\varepsilon$. Let $\varepsilon\downarrow0$. Half-lines follow by truncation, finite disjoint unions componentwise.
\end{enumerate}
\hlt{Structural Commentary}: Lebesgue measure starts as length on a finite interval algebra, not on all subsets.\\
\hlt{Key Insight}: Compactness supplies the reverse inequality in countable additivity of interval length.
\end{proof}
\end{proposition}

\begin{remark} \label{rem:half-open-endpoints-make-an-algebra} \hlt{Why the Half-Open Convention Matters}\\
Finite unions generated by $(a,b)$, $(-\infty,b)$, $(a,+\infty)$, and $\R$ still generate $\mathcal{B}(\R)$, but are not an algebra, where\\
$(0,1)^c=(-\infty,0]\cup[1,+\infty)$. Half-open endpoints make complements remain in the starting family.
\end{remark}

\begin{definition} \label{def:borel-lebesgue-stieltjes-measures} \hlt{Borel--Stieltjes and Lebesgue--Stieltjes Measures}\\
Let $g:\R\to\R$ be increasing and right-continuous, set $g(\pm\infty)\defeq\lim_{x\to\pm\infty}g(x)$ in $\overline{\R}$, and let $\mathcal{F}$ be the algebra of finite unions generated by $(a,b]$, $(-\infty,b]$, $(a,+\infty)$, and $\R$. Define $\mu_g$ on generators by $\mu_g((a,b])\defeq g(b)-g(a)$, $\mu_g((-\infty,b])\defeq g(b)-g(-\infty)$, $\mu_g((a,+\infty))\defeq g(+\infty)-g(a)$, and $\mu_g(\R)\defeq g(+\infty)-g(-\infty)$; extend to finite disjoint unions by summation. Its unique extension to $\mathcal{B}(\R)$ is the \hlt{Borel--Stieltjes measure generated by $g$}; the completion of this Borel measure is the \hlt{Lebesgue--Stieltjes measure generated by $g$}.
\end{definition}

\begin{theorem} \label{thm:borel-stieltjes-measure-continuous-increasing}\label{thm:borel-stieltjes-measure-generation} \hlt{Borel--Stieltjes Measure Existence}\\
Let $g:\R\to\R$ be increasing and right-continuous, $g(\pm\infty)\defeq\lim_{x\to\pm\infty}g(x)$,\\
$\mathcal{I}\defeq\{(a,b]:a<b\}\cup\{(-\infty,b]:b\in\R\}\cup\{(a,+\infty):a\in\R\}\cup\{\R\}$, and let $\mathcal{F}$ be the finite-union algebra generated by $\mathcal{I}$. Define $G_g((a,b])\defeq g(b)-g(a)$, $G_g((-\infty,b])\defeq g(b)-g(-\infty)$, $G_g((a,+\infty))\defeq g(+\infty)-g(a)$, $G_g(\R)\defeq g(+\infty)-g(-\infty)$, and $\mu_g(\bigsqcup_{j=1}^{m}I_j)\defeq\sum_{j=1}^{m}G_g(I_j)$ for disjoint $I_j\in\mathcal{I}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu_g$ is a $\sigma$-finite premeasure on $\mathcal{F}$ and has a unique extension to $\mathcal{B}(\R)$, again denoted $\mu_g$.
\item If $g=\operatorname{id}_{\R}$, then $\mu_g$ is Lebesgue measure.
\item If $g$ is continuous, then $\mu_g(J)=g(b)-g(a)$ for $J\in\{(a,b),[a,b),(a,b],[a,b]\}$, $a<b$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Finite normal-form refinement makes $\mu_g$ well defined and finitely additive: adjacent $g$-increments telescope and monotonicity gives positivity.\\
If $(a,b]=\bigsqcup_{j\geq1}(a_j,b_j]$, finite additivity gives $\sum_{j=1}^{N}G_g((a_j,b_j])\leq G_g((a,b])$; conversely, covering $[a+\varepsilon,b]$ by finitely many $(a_j,b_j+\varepsilon_j)$ gives $g(b)-g(a+\varepsilon)\leq\sum_{j\in K}(g(b_j+\varepsilon_j)-g(a_j))$ for finite $K\subseteq\N$, and right-continuity lets $\varepsilon,\varepsilon_j\downarrow0$. Hence bounded generators are countably additive.\\
Half-lines follow by truncation and finite unions componentwise, so $\mu_g$ is premeasure. As $\R=\bigcup_{n\geq1}(-n,n]$, $\mu_g((-n,n])=g(n)-g(-n)<+\infty$, $\mu_g$ is $\sigma$-finite; Theorem \ref{thm:sigma-finite-generated-extension-uniqueness} gives uniqueness.
\item If $g=\operatorname{id}_{\R}$, then $G_g((a,b])=b-a$, so $\mu_g$ agrees with half-open length on $\mathcal{F}$.\\
Uniqueness identifies its Borel extension with Lebesgue measure.
\item If $g$ is continuous, then $(x-1/n,x]\downarrow\{x\}$ gives $\mu_g(\{x\})=\lim_n(g(x)-g(x-1/n))=0$.
\end{enumerate}
\hlt{Structural Commentary}: Stieltjes measure replaces ordinary length $b-a$ by the increment $g(b)-g(a)$.\\
\hlt{Key Insight}: Right-continuity aligns the half-open interval convention with countable covers.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:finite-borel-measure-distribution-function} \hlt{Finite Borel Measures Give Stieltjes Generators}\\
Let $\nu$ be finite on $\mathcal{B}(\R)$ and $g(x)\defeq\nu((-\infty,x])$. Then $g$ is increasing and right-continuous, $g(-\infty)=0$, $g(+\infty)=\nu(\R)$, $\nu((a,b])=g(b)-g(a)$ for $a\leq b$, and $\nu=\mu_g$ on $\mathcal{B}(\R)$.
\begin{proof}
For $a\leq b$, inclusion gives $g(a)\leq g(b)$ and $(-\infty,b]=(-\infty,a]\sqcup(a,b]$, hence $\nu((a,b])=g(b)-g(a)$.\\
If $x_n\downarrow x$, then $(-\infty,x_n]\downarrow(-\infty,x]$, so finite continuity from above gives $g(x_n)\downarrow g(x)$; thus $g$ is right-continuous.\\
Also $(-\infty,-n]\downarrow\emptyset$ and $(-\infty,n]\uparrow\R$, whence $g(-\infty)=0$ and $g(+\infty)=\nu(\R)$.\\
Thus $\nu$ and $\mu_g$ agree on the generators of $\mathcal{F}$; Theorem \ref{thm:borel-stieltjes-measure-generation} and uniqueness give $\nu=\mu_g$ on $\mathcal{B}(\R)$.\\
\hlt{Structural Commentary}: A finite real-line measure is encoded by its cumulative mass function.\\
\hlt{Key Insight}: Right-continuity of distribution functions is continuity from above for measures.
\end{proof}
\end{proposition}

\begin{example} \label{ex:borel-stieltjes-basic-generators} \hlt{Basic Borel--Stieltjes Generators}\\
The following generators illustrate the interval rule in Theorem \ref{thm:borel-stieltjes-measure-generation}.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $g(x)=2x$, then $\mu_g=2m$ on $\mathcal{B}(\R)$.
\item If $g(x)=\arctan x$, then $\mu_g((a,b])=\arctan b-\arctan a=\int_a^b(1+x^2)^{-1}\,dx$ and $\mu_g(\R)=\pi$.
\item If $g(x)=0$ for $x<0$ and $g(x)=1$ for $x\geq0$, then $\mu_g=\delta_0$.
\item If $g(x)=0$ for $x<0$ and $g(x)=x$ for $x\geq0$, then $\mu_g(E)=m(E\cap[0,+\infty))$ for Borel $E$.
\end{enumerate}
\end{example}

\begin{proposition} \label{prop:elementary-measure-constructions} \hlt{Elementary Measure Constructions}\\
Let $(X,\mathcal{M})$ be a measurable space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\mu$ is a measure on $\mathcal{M}$ and $A\in\mathcal{M}$, then $\mu_A(E)\defeq\mu(A\cap E)$ defines a measure $\mu_A$ on $\mathcal{M}$.
\item If $\mu_1,\ldots,\mu_m$ are measures on $\mathcal{M}$ and $a_1,\ldots,a_m\geq0$, then $\lambda\defeq\sum_{j=1}^{m}a_j\mu_j$ is a measure on $\mathcal{M}$.
\item If $(\mu_n)_{n\geq1}$ are measures on $\mathcal{M}$ with $\mu_n(X)=1$, then $\lambda\defeq\sum_{n=1}^{\infty}2^{-n}\mu_n$ is a measure on $\mathcal{M}$ and $\lambda(X)=1$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu_A(\emptyset)=0$. If $(E_k)$ is disjoint in $\mathcal{M}$, then $A\cap\bigsqcup_kE_k=\bigsqcup_k(A\cap E_k)$, whence $\mu_A(\bigsqcup_kE_k)=\sum_k\mu_A(E_k)$.
\item $\lambda(\emptyset)=0$. For disjoint $(E_k)$, finite summation gives $\lambda(\bigsqcup_kE_k)=\sum_{j=1}^{m}a_j\sum_k\mu_j(E_k)=\sum_k\lambda(E_k)$.
\item $\lambda(\emptyset)=0$. For disjoint $(E_k)$, nonnegative reordering gives $\lambda(\bigsqcup_kE_k)=\sum_n2^{-n}\sum_k\mu_n(E_k)=\sum_k\lambda(E_k)$.\\
Also $\lambda(X)=\sum_n2^{-n}\mu_n(X)=\sum_n2^{-n}=1$.
\end{enumerate}
\hlt{Structural Commentary}: New measures come from restriction, superposition, and countable convex mixtures.\\
\hlt{Key Insight}: Nonnegative series make the bookkeeping order-independent.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:additive-set-function-basic-laws} \hlt{Basic Laws for Additive Set Functions}\\
Let $\phi$ be additive on a ring $\mathcal{A}$. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\phi(\emptyset)=0$;
\item $\phi(A_1\cup\cdots\cup A_n)=\phi(A_1)+\cdots+\phi(A_n)$ for pairwise disjoint $A_i\in\mathcal{A}$;
\item $\phi(A\cup B)+\phi(A\cap B)=\phi(A)+\phi(B)$;
\item if $\phi$ is nonnegative and $A\subseteq B$, then $\phi(A)\leq\phi(B)$;
\item if $B\subseteq A$ and $\abs{\phi(B)}<+\infty$, then $\phi(A\setminus B)=\phi(A)-\phi(B)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Apply additivity to the empty disjoint union.
\item Induct on $n$, applying additivity to $(A_1\cup\cdots\cup A_{n-1})\cup A_n$.
\item By disjoint decompositions, $A\cup B=(A\setminus B)\cup(B\setminus A)\cup(A\cap B)$; $A=(A\setminus B)\cup(A\cap B)$; $B=(B\setminus A)\cup(A\cap B)$
\item If $A\subseteq B$, then $B=A\cup(B\setminus A)$ disjointly, so nonnegativity gives $\phi(A)\leq\phi(B)$.
\item If $B\subseteq A$, then $A=B\cup(A\setminus B)$ disjointly; since $\abs{\phi(B)}<+\infty$, subtract $\phi(B)$.
\end{enumerate}
\hlt{Structural Commentary}: Finite additivity plus disjoint decompositions controls all elementary identities.\\
\hlt{Key Insight}: Rewrite overlaps as disjoint pieces before applying $\phi$.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:continuity-from-below-set-functions} \hlt{Continuity from Below}\\
Suppose $\phi$ is countably additive on a ring $\mathcal{A}$.\\
If $A_1\subseteq A_2\subseteq A_3\subseteq\cdots$, and $A=\bigcup\limits_{n=1}^{\infty}A_n$ with all $A_n,A\in\mathcal{A}$, then $\phi(A_n)\to\phi(A)$.
\begin{proof}
Put $B_1=A_1$ and $B_n=A_n\setminus A_{n-1}$ for $n\geq2$. Then $B_n$ are pairwise disjoint, $A_n=B_1\cup\cdots\cup B_n$, and $A=\bigcup_nB_n$. Countable additivity turns $\phi(A_n)$ and $\phi(A)$ into partial sums and full sum of same series.\\
\hlt{Structural Commentary}: Increasing unions are disjoint tail decompositions in disguise.\\
\hlt{Key Insight}: Countable additivity converts set convergence into series convergence.
\end{proof}
\end{theorem}

\begin{definition} \label{def:measure-cells-intervals-volume-rp} \hlt{Cells, Intervals, and Volume in $\R^p$}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Cells in $\R$}. For $a\leq b$, a cell with endpoints $a,b$ is $I\in\{(a,b),[a,b],(a,b],[a,b)\}$, respectively open, closed, right-half-open, left-half-open; if $a=b$, only $[a,a]$ is nonempty.
\item \hlt{Length}. For a cell $I$ with endpoints $a\leq b$, $\ell(I)\defeq b-a$, independent of endpoint inclusion; $(-\infty,a)$, $(-\infty,a]$, $[b,+\infty)$, $(b,+\infty)$, and $\R$ are intervals, not cells, with length $+\infty$.
\item \hlt{Cells in $\R^p$}. For $p\geq1$, a cell is $I=\prod_{i=1}^{p}I_i$, each $I_i$ a cell in $\R$; its type is open, closed, or half-open, $\prod_i[a_i,b_i)$ is \hlt{southwest}, $\prod_i(a_i,b_i]$ is \hlt{northeast}, and a \hlt{cube} satisfies $\ell(I_1)=\cdots=\ell(I_p)$.
\item \hlt{Volume}. For $I=\prod_{i=1}^{p}I_i$, $\operatorname{vol}_p(I)\defeq\prod_{i=1}^{p}\ell(I_i)$ and $\operatorname{vol}_p(\emptyset)\defeq0$.\\
If endpoints are $a_i\leq b_i$, then $\operatorname{vol}_p(I)=\prod_{i=1}^{p}(b_i-a_i)$.
\item \hlt{Unbounded products}. For cells $I$, $\operatorname{vol}_p(I^\circ)=\operatorname{vol}_p(\overline I)=\operatorname{vol}_p(I)$.\\
For products $I=\prod_iI_i$ of possibly unbounded intervals, $\exists j:\ell(I_j)=0$ gives volume $0$, while $\ell(I_j)>0$ for all $j$ and $\exists j:\ell(I_j)=+\infty$ gives volume $+\infty$.
\end{enumerate}
\end{definition}

\begin{remark} \label{rem:cell-interval-topological-conventions} \hlt{Cells versus Intervals}\\
Cells are precisely bounded intervals with finite endpoints, so every cell is an interval but not conversely. Open cells are open in $\R^p$, closed cells are closed in $\R^p$, and nonempty half-open cells are neither.\\
In $\R$, $(-\infty,a)$ and $(b,+\infty)$ are open, $(-\infty,a]$ and $[b,+\infty)$ are closed, and $\R=(-\infty,+\infty)$ is clopen.\\
\hlt{Key Insight}: Topology likes open and closed endpoints; measure bookkeeping often likes half-open endpoints.
\end{remark}

\begin{proposition} \label{prop:half-open-cell-calculus-and-subdivision} \hlt{Half-Open Cell Calculus}\\
Fix $\sigma\in\{\hlt{southwest},\hlt{northeast}\}$, and let $\mathcal{H}_\sigma$ be the half-open cells in $\R^p$ with orientation $\sigma$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Any finite Boolean combination of $H_1,\ldots,H_n\in\mathcal{H}_\sigma$ is $\bigsqcup_{r=1}^{N}K_r$ with $K_r\in\mathcal{H}_\sigma$.
\item If $H\in\mathcal{H}_\sigma$, then $\R^p\setminus H=\bigsqcup_{r=1}^{\infty}K_r$ for some $K_r\in\mathcal{H}_\sigma$.
\item For $H\in\mathcal{H}_\sigma$ and $\delta>0$, $H=\bigsqcup_{r=1}^{N}K_r$ for some $K_r\in\mathcal{H}_\sigma$ with $\operatorname{diam}K_r<\delta$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For coordinate endpoints $E_j=\{c_{j,0}<\cdots<c_{j,m_j}\}$ of $H_1,\ldots,H_n$, set $[a,b)_\sigma=[a,b)$ if $\sigma=\hlt{southwest}$ and $[a,b)_\sigma=(a,b]$ if $\sigma=\hlt{northeast}$, $\mathcal{P}_j=\{[c_{j,q-1},c_{j,q})_\sigma:1\leq q\leq m_j\}$, and $\mathcal{G}=\{\prod_{j=1}^{p}P_j:P_j\in\mathcal{P}_j\}$; every finite Boolean combination is $\bigsqcup_{G\in\mathcal{G}'}G$, $\mathcal{G}'\subseteq\mathcal{G}$.
\item Write $H=\prod_{j=1}^{p}H_j$ and $D_s=(\prod_{j<s}H_j)\times(\R\setminus H_s)\times\R^{p-s}$; then $\R^p\setminus H=\bigsqcup_{s=1}^{p}D_s$. Decomposing each one-dimensional ray and $\R$ into countably many bounded $\sigma$-half-open intervals and taking products gives the required union; countability follows from Corollary \ref{cor:countable-cartesian-products}.
\item Choose $m$ with $2^{-m}(\sum_{j=1}^{p}\ell(H_j)^2)^{1/2}<\delta$. After $m$ bisections per coordinate, each subcell has\\
diameter $<\delta$; disjointness and volume preservation follow coordinatewise.
\end{enumerate}
\hlt{Structural Commentary}: Half-open cells behave like grid atoms: finite set operations become finite grid selection.\\
\hlt{Key Insight}: Half-open endpoints prevent overlap while still partitioning boxes cleanly.
\end{proof}
\end{proposition}

\begin{definition} \label{def:translation-of-euclidean-set} \hlt{Translation of a Set}\\
If $A\subseteq\R^p$ and $t\in\R^p$, the translate of $A$ by $t$ is $t+A\defeq\{t+a:a\in A\}$.
\end{definition}

\begin{proposition} \label{prop:cell-volume-translation-invariant} \hlt{Cell Volume Is Translation Invariant}\\
Let $I=\prod_{i=1}^{p}I_i$ be a cell in $\R^p$ and $t=(t_i)_{i=1}^{p}\in\R^p$. Then $t+I=\prod_{i=1}^{p}(t_i+I_i)$ is a cell with the same endpoint pattern as $I$, and $\operatorname{vol}_p(t+I)=\operatorname{vol}_p(I)$.
\begin{proof}
Let $a_i\leq b_i$ be the endpoints of $I_i$. Then $t_i+I_i$ has endpoints $t_i+a_i\leq t_i+b_i$, unchanged endpoint inclusion, and $\ell(t_i+I_i)=(t_i+b_i)-(t_i+a_i)=\ell(I_i)$. Hence $\operatorname{vol}_p(t+I)=\prod_{i=1}^{p}\ell(t_i+I_i)=\prod_{i=1}^{p}\ell(I_i)=\operatorname{vol}_p(I)$.\\
\hlt{Structural Commentary}: Translation moves location but not side lengths.\\
\hlt{Key Insight}: Coordinate volume depends only on side lengths.
\end{proof}
\end{proposition}

\begin{definition} \label{def:elementary-sets-and-regular-set-functions} \hlt{Elementary Sets and Regularity in $\R^p$}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Elementary sets}. $\mathcal{E}\defeq\{\bigcup_{r=1}^{n}I_r:n\in\N\}$, where each $I_r\subseteq\R^p$ is a bounded cell.
\item \hlt{Elementary volume}. If $A=\bigsqcup_{r=1}^{n}I_r$ with cells $I_r$, set $m(A)\defeq\sum_{r=1}^{n}\operatorname{vol}_p(I_r)$.
\item \hlt{Regularity}. A nonnegative additive $\mu:\mathcal{E}\to[0,+\infty]$ is regular iff $\forall A\in\mathcal{E}$, $\forall\varepsilon>0$,\\
$\exists F,G\in\mathcal{E}$ with $F\subseteq A\subseteq G$, $F$ closed, $G$ open, and $\mu(G)-\varepsilon\leq\mu(A)\leq\mu(F)+\varepsilon$.
\end{enumerate}
$\mathcal{E}$ is a ring; every $A\in\mathcal{E}$ admits a finite disjoint cell decomposition; $m$ is well defined, additive, and regular.
\end{definition}

\begin{perspective} \label{per:box-volume-and-linear-algebra} \hlt{Box Volume and Linear Algebra}\\
For a box $B=\prod_i(a_i,b_i)$ or $\prod_i[a_i,b_i]$, $m(B)=\prod_i(b_i-a_i)$, the diagonal case of determinant volume scaling.\\
Measure extends this product rule beyond boxes.\\
\hlt{Key Insight}: Lebesgue measure starts from determinant-style coordinate volume.
\end{perspective}

\begin{perspective} \label{per:outer-measure-overestimation} \hlt{Outer Measure as Overestimation}\\
Elementary $A=\bigsqcup_{r=1}^{n}I_r$ have $m(A)=\sum_{r=1}^{n}\operatorname{vol}_p(I_r)$.\\
General $E\subseteq\R^p$, e.g. $D=\{(x,y)\in\R^2:x^2+y^2\leq1\}$, need not be a finite disjoint cell union.\\
Replace decompositions by countable cell covers $E\subseteq\bigcup_{k=1}^{\infty}I_k$ and minimize $\sum_{k=1}^{\infty}\operatorname{vol}_p(I_k)$.\\
\hlt{Key Insight}: Outer covers size every $E\subseteq\R^p$ before measurability.
\end{perspective}

\begin{definition} \label{def:cell-cover-outer-measure} \hlt{Cell-Cover Outer Measure}\\
For $E\subseteq\R^p$, let $\mathcal{C}(E)$ be the countable cell covers $(I_k)_{k\geq1}$ with $E\subseteq\bigcup\limits_{k=1}^{\infty}I_k$. Define $m_p^\ast(E)\defeq\inf\limits_{(I_k)\in\mathcal{C}(E)}\sum\limits_{k=1}^{\infty}\operatorname{vol}_p(I_k)$.
Here $\mathcal{C}(E)\neq\emptyset$ since $E\subseteq\bigcup\limits_{j=1}^{\infty}[-j,j]^p$, and sum is nonnegative extended-real series, hence order-independent.
\end{definition}

\begin{proposition} \label{prop:outer-measure-cover-normalizations} \hlt{Equivalent Covering Families}\\
For $E\subseteq\R^p$, $\delta>0$, and fixed half-open orientation $\sigma$, let $\mathcal{C}_{\mathsf{cl}}(E)$, $\mathcal{C}_{\mathsf{op}}(E)$, $\mathcal{C}_{\sigma}(E)$, and $\mathcal{C}_{\delta}(E)$ be the subfamilies of $\mathcal{C}(E)$ with cells respectively closed, open, $\sigma$-half-open, and satisfying $\operatorname{diam}I_k<\delta$.\\
If $\alpha(\mathcal{A})\defeq\inf_{(I_k)\in\mathcal{A}}\sum_{k=1}^{\infty}\operatorname{vol}_p(I_k)$, then $m_p^\ast(E)=\alpha(\mathcal{C}_{\mathsf{cl}}(E))=\alpha(\mathcal{C}_{\mathsf{op}}(E))=\alpha(\mathcal{C}_{\sigma}(E))=\alpha(\mathcal{C}_{\delta}(E))$.
\begin{proof}
Let $\mathcal{C}_\ast(E)$ be any restricted class above. Since $\mathcal{C}_\ast(E)\subseteq\mathcal{C}(E)$, $\alpha(\mathcal{C}_\ast(E))\geq m_p^\ast(E)$.\\
Conversely, for $(I_k)\in\mathcal{C}(E)$ and $\varepsilon>0$, closures give $\overline{I_k}\supseteq I_k$ and $\operatorname{vol}_p(\overline{I_k})=\operatorname{vol}_p(I_k)$; open cells $J_k\supseteq I_k$ may be chosen with $\operatorname{vol}_p(J_k)\leq\operatorname{vol}_p(I_k)+\varepsilon2^{-k}$; each $J_k$ admits a countable disjoint $\sigma$-half-open decomposition with total volume $\operatorname{vol}_p(J_k)$; each half-open cell has a finite disjoint subdivision into cells of diameter $<\delta$ with unchanged total volume. Thus $\alpha(\mathcal{C}_\ast(E))\leq\sum_k\operatorname{vol}_p(I_k)+\varepsilon$; take infima over $(I_k)$ and let $\varepsilon\downarrow0$.\\
\hlt{Structural Commentary}: Near-optimal covers ignore endpoint convention.\\
\hlt{Key Insight}: Closures, small enlargements, and fine subdivisions preserve the infimum.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:cell-cover-outer-measure-basic-properties} \hlt{Basic Properties of Cell-Cover Outer Measure}\\
For $E,F,E_n\subseteq\R^p$,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Null empty set}. $0\leq m_p^\ast(E)\leq+\infty$ and $m_p^\ast(\emptyset)=0$.
\item \hlt{Monotonicity}. $E\subseteq F\Longrightarrow m_p^\ast(E)\leq m_p^\ast(F)$.
\item \hlt{Subadditivity}. $m_p^\ast(\bigcup_{n=1}^{\infty}E_n)\leq\sum_{n=1}^{\infty}m_p^\ast(E_n)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since all cover sums lie in $[0,+\infty]$, $0\leq m_p^\ast(E)\leq+\infty$; the cover $(\emptyset,\emptyset,\ldots)$ gives $m_p^\ast(\emptyset)=0$.
\item If $E\subseteq F$, then $\mathcal{C}(F)\subseteq\mathcal{C}(E)$, hence $\inf_{\mathcal{C}(E)}\leq\inf_{\mathcal{C}(F)}$.
\item If some $m_p^\ast(E_n)=+\infty$, the claim is trivial.\\
Otherwise, for $\varepsilon>0$, choose $(I_k^n)_{k\geq1}\in\mathcal{C}(E_n)$ with $\sum_k\operatorname{vol}_p(I_k^n)\leq m_p^\ast(E_n)+\varepsilon2^{-n}$.\\
Then $(I_k^n)_{(n,k)\in\N^2}$ countable, covers $\bigcup_nE_n$, and $m_p^\ast(\bigcup_nE_n)\leq\sum_{n,k}\operatorname{vol}_p(I_k^n)\leq\sum_nm_p^\ast(E_n)+\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Countable subadditivity is proved by merging near-optimal covers.\\
\hlt{Key Insight}: Nonnegative double sums may be scanned in one sequence.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:outer-measure-positive-distance-additivity} \hlt{Additivity at Positive Distance}\\
Let $A,B\subseteq\R^p$ be nonempty and disjoint, and suppose $\operatorname{dist}(A,B)\defeq\inf\{\norm{a-b}_2:a\in A,\ b\in B\}>0$.\\
Then $m_p^\ast(A\cup B)=m_p^\ast(A)+m_p^\ast(B)$.
\begin{proof}
The inequality $\leq$ follows from Theorem \ref{thm:cell-cover-outer-measure-basic-properties}. For $\geq$, set $\delta\defeq\operatorname{dist}(A,B)$. If $m_p^\ast(A\cup B)=+\infty$, done.\\
Otherwise, given $\varepsilon>0$, choose $(I_n)\in\mathcal{C}(A\cup B)$ with $\operatorname{diam}I_n<\delta$ and $\sum_n\operatorname{vol}_p(I_n)\leq m_p^\ast(A\cup B)+\varepsilon$.\\
Put $N_A\defeq\{n:I_n\cap A\neq\emptyset\}$ and $N_B\defeq\{n:I_n\cap B\neq\emptyset\}$. Since $N_A\cap N_B=\emptyset$, $(I_n)_{n\in N_A}$ covers $A$, $(I_n)_{n\in N_B}$ covers $B$, and $m_p^\ast(A)+m_p^\ast(B)\leq\sum_{n\in N_A}\operatorname{vol}_p(I_n)+\sum_{n\in N_B}\operatorname{vol}_p(I_n)\leq\sum_n\operatorname{vol}_p(I_n)\leq m_p^\ast(A\cup B)+\varepsilon$. Let $\varepsilon\downarrow0$.\\
\hlt{Structural Commentary}: Positive separation lets a sufficiently fine cover be sorted unambiguously.\\
\hlt{Key Insight}: Countable subadditivity becomes equality when no small cell can touch both pieces.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:outer-measure-agrees-with-cell-volume} \hlt{Outer Measure of a Cell}\\
If $I$ is a cell in $\R^p$, then $m_p^\ast(I)=\operatorname{vol}_p(I)$.
\begin{proof}
The cover $(I,\emptyset,\ldots)$ gives $m_p^\ast(I)\leq\operatorname{vol}_p(I)$. For $\geq$, the case $\operatorname{vol}_p(I)=0$ follows by nonnegativity.\\
Assume $\operatorname{vol}_p(I)>0$ and fix $\varepsilon>0$. Choose a closed cell $J\subseteq I$ with $\operatorname{vol}_p(I)-\varepsilon<\operatorname{vol}_p(J)$, taking $J=I$ when $I$ is closed. For any open-cell cover $(G_k)$ of $I$, compactness gives $J\subseteq\bigcup_{k=1}^{m}G_k$.\\
Let $\mathcal{Q}$ be the finite rectangular grid determined by the faces of $J,\overline{G_1},\ldots,\overline{G_m}$. Up to zero-volume boundary atoms, each $Q\in\mathcal{Q}$ with $Q\subseteq J$ lies in some $G_k$, hence $\operatorname{vol}_p(J)\leq\sum_{k=1}^{m}\operatorname{vol}_p(G_k)\leq\sum_{k=1}^{\infty}\operatorname{vol}_p(G_k)$.\\
Infimize over open-cell covers, use Proposition \ref{prop:outer-measure-cover-normalizations}, and let $\varepsilon\downarrow0$.\\
\hlt{Structural Commentary}: Compact inner cells reduce covers to finite grids.\\
\hlt{Key Insight}: Outer measure is calibrated so that cells keep their original volume.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:outer-measure-translation-invariance} \hlt{Translation Invariance of Outer Measure}\\
For $E\subseteq\R^p$ and $t\in\R^p$, $m_p^\ast(t+E)=m_p^\ast(E)$.
\begin{proof} The map $\mathcal{C}(E)\to\mathcal{C}(t+E)$, $(I_k)\mapsto(t+I_k)$, is bijective with inverse $(J_k)\mapsto(-t+J_k)$.\\
By Proposition \ref{prop:cell-volume-translation-invariant}, $\sum_k\operatorname{vol}_p(t+I_k)=\sum_k\operatorname{vol}_p(I_k)$, so the two infima are equal.\\
\hlt{Structural Commentary}: Translate covers forward and backward.\\
\hlt{Key Insight}: Translation invariance of cell volume lifts directly to translation invariance of outer measure.
\end{proof}
\end{theorem}

\begin{definition} \label{def:outer-measure-and-rudin-measurable-sets} \hlt{Outer Measure and $\mu$-Measurable Sets}\\
Let $\mu$ be additive, regular, nonnegative, and finite on $\mathcal{E}$.\\
For $E\subseteq\R^p$, define $\mu^\ast(E)\defeq\inf\sum\limits_{n=1}^{\infty}\mu(A_n)$, where infimum is over open elementary covers $E\subseteq\bigcup\limits_{n=1}^{\infty}A_n$.\\
For $A,B\subseteq\R^p$, set $A\triangle B\defeq(A\setminus B)\cup(B\setminus A)$ and $d_\mu(A,B)\defeq\mu^\ast(A\triangle B)$.\\
A set is \hlt{finitely $\mu$-measurable} if it is a $d_\mu$-limit of elementary sets.\\
It is \hlt{$\mu$-measurable} if it is a countable union of such sets; denote the classes by $\mathcal{M}_F(\mu)$ and $\mathcal{M}(\mu)$.
\end{definition}

\begin{definition} \label{def:tao-box-cover-outer-measure} \hlt{Box-Cover Outer Measure Model}\\
An \hlt{open box} in $\R^n$ is $B=\prod\limits_{i=1}^{n}(a_i,b_i)=\{(x_1,\ldots,x_n)\in\R^n:x_i\in(a_i,b_i) \ \forall i \}$, $b_i\geq a_i$, $\operatorname{vol}_n(B)\defeq\prod\limits_{i=1}^{n}(b_i-a_i)$.\\
For $\Omega\subseteq\R^n$, the \hlt{box-cover outer measure} is $m_n^\ast(\Omega)\defeq\inf\left\{\sum_{j\in J}\operatorname{vol}_n(B_j):(B_j)_{j\in J}\text{ covers }\Omega,\ J\text{ at most countable}\right\}$.
\end{definition}

Outer measure covers $A$ from outside; $d_\mu(A,B)=\mu^\ast(A\triangle B)$ measures closeness.\\
\hlt{Key Insight}: Countable subadditivity collects near-optimal covers.
\end{perspective}

\begin{method} \label{meth:lebesgue-measure-construction} \hlt{Lebesgue Measure Construction}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Cover from outside}\\
For $A\subseteq\R^n$, define $m_n^\ast(A)\defeq\inf\left\{\sum_{j\in J}\operatorname{vol}_n(B_j):(B_j)_{j\in J}\text{ covers }A,\ J\text{ at most countable}\right\}$.\\
This is defined for every set, but it is only countably subadditive in general.
\item \hlt{Keep the clean splitters}\\
Call $A$ measurable when every test set $X\subseteq\R^n$ splits additively: $m_n^\ast(X)=m_n^\ast(X\cap A)+m_n^\ast(X\cap A^c)$.\\
These measurable sets form a $\sigma$-algebra.
\item \hlt{Restrict the outer measure}\\
Define Lebesgue measure on measurable sets by $m(A)\defeq m_n^\ast(A)$.\\
On this restricted class, $m$ is countably additive.
\end{enumerate}
\hlt{Key Insight}: Outer measure sizes all sets; clean splitting restores additivity.
\end{method}

\begin{theorem} \label{thm:measure-measurability-main-source-todo}\label{thm:rudin-measure-extension} \hlt{Extension to a Measure}\\
Let $\mu$ be additive, regular, nonnegative, and finite on $\mathcal{E}$. Form $\mu^\ast$ from open elementary covers, set $d_\mu(A,B)=\mu^\ast(A\triangle B)$, and let $\mathcal{M}(\mu)$ be the countable unions of $d_\mu$-limits of elementary sets. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu^\ast(A)=\mu(A)$ for $A\in\mathcal{E}$;
\item $\mu^\ast$ is countably subadditive;
\item $\mathcal{M}(\mu)$ is a $\sigma$-ring;
\item $\mu^\ast$ is countably additive on $\mathcal{M}(\mu)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Regularity gives $\mu^\ast(A)\leq\mu(A)$; compact closed approximants inside $A$ give the reverse inequality.
\item Cover each $E_n$ within $2^{-n}\varepsilon$ of $\mu^\ast(E_n)$; the combined cover proves subadditivity.
\item If $A_k\to A$, $B_k\to B$ in $d_\mu$, $A_k,B_k\in\mathcal{E}$, then $d_\mu(A_k\diamond B_k,A\diamond B)\leq d_\mu(A_k,A)+d_\mu(B_k,B)\to0$, $\diamond\in\{\cup,\cap,\setminus\}$.\\
Thus $\mathcal{M}_F(\mu)$ is a ring. If $A=\bigcup_kA_k$ with $A_k\in\mathcal{M}_F(\mu)$, then $A\setminus B=\bigcup_k(A_k\setminus B)$, so $\mathcal{M}(\mu)$ is a $\sigma$-ring.
\item For $A=\bigsqcup\limits_n A_n$, $A_n\in\mathcal{M}_F(\mu)$, subadditivity gives $\leq$,\\
while finite additivity gives $\mu^\ast(A)\geq\mu^\ast\left(\bigsqcup\limits_{n=1}^{N}A_n\right)=\sum\limits_{n=1}^{N}\mu^\ast(A_n)$.\\
Let $N\to\infty$, then decompose general disjoint families in $\mathcal{M}(\mu)$ into finite-measurable pieces.
\end{enumerate}
\hlt{Structural Commentary}: Elementary sets are completed under the distance measured by symmetric difference.\\
\hlt{Key Insight}: Approximate by elementary sets, prove identities there, then pass through $d_\mu(A,B)$.
\end{proof}
\end{theorem}

\begin{definition} \label{def:extended-and-lebesgue-measure} \hlt{Extended and Lebesgue Measure}\\
The \hlt{extended measure} is $\mu^\ast|_{\mathcal{M}(\mu)}$, denoted again by $\mu$.\\
For the elementary volume $m$, it is \hlt{Lebesgue measure} on $\R^p$.
\end{definition}

\begin{definition} \label{def:abstract-outer-measure-clean-division} \hlt{Abstract Outer Measure and Clean Division}\\
An \hlt{abstract outer measure} on $M$ is a monotone, countably subadditive function $\omega:2^M\to[0,+\infty]$ with $\omega(\emptyset)=0$.\\
A set $E\subseteq M$ is \hlt{$\omega$-measurable} if it cleanly splits every $X\subseteq M$: $\omega(X)=\omega(X\cap E)+\omega(X\cap E^c)$.\\
For Lebesgue outer measure, this is Caratheodory measurability.
\end{definition}

\begin{lemma} \label{lem:caratheodory-finite-measure-test} \hlt{Finite-Measure Test for Caratheodory Measurability}\\
Let $\omega:2^M\to[0,+\infty]$ be an outer measure. Then $E\subseteq M$ is $\omega$-measurable iff, for all $X\subseteq M$,\\
$\omega(X)<+\infty\Longrightarrow \omega(X)\geq\omega(X\cap E)+\omega(X\cap E^c)$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $E$ is $\omega$-measurable, then $\omega(X)=\omega(X\cap E)+\omega(X\cap E^c)$ $\forall X\subseteq M$, hence finite-test inequality holds.
\item[$(\Leftarrow)$] For all $X\subseteq M$, $X=(X\cap E)\sqcup(X\cap E^c)$ gives $\omega(X)\leq\omega(X\cap E)+\omega(X\cap E^c)$. If $\omega(X)<+\infty$, the hypothesis gives $\geq$; if $\omega(X)=+\infty$, $\geq$ is automatic in $[0,+\infty]$. Thus equality holds for every $X$.
\end{enumerate}
\hlt{Structural Commentary}: To verify a clean split, only finite outer-measure test sets need real work.\\
\hlt{Key Insight}: Subadditivity supplies one inequality for free.
\end{proof}
\end{lemma}

\begin{definition} \label{def:outer-measure-zero-sets} \hlt{Zero Sets}\\
For outer measure $\omega:2^M\to[0,+\infty]$, \hlt{$\omega$-zero set} is $Z\subseteq M$ with $\omega(Z)=0$; $\mathcal{Z}_\omega\defeq\{Z\subseteq M:\omega(Z)=0\}$.
\end{definition}

\begin{theorem} \label{thm:caratheodory-measurable-sets-sigma-algebra} \hlt{Caratheodory Measurable Sets}\\
Let $\omega:2^M\to[0,+\infty]$ be an outer measure, and let $\mathcal{M}_\omega$ be the class of $\omega$-measurable subsets of $M$.\\
Then $\mathcal{M}_\omega$ is a $\sigma$-algebra, $\omega|_{\mathcal{M}_\omega}$ is countably additive, $\mathcal{Z}_\omega\subseteq\mathcal{M}_\omega$, and $E\in\mathcal{M}_\omega$, $Z\in\mathcal{Z}_\omega$\\
imply $E\cup Z,E\setminus Z,E\triangle Z\in\mathcal{M}_\omega$ and $\omega(E\cup Z)=\omega(E\setminus Z)=\omega(E\triangle Z)=\omega(E)$.
\begin{proof}
If $Z\in\mathcal{Z}_\omega$, then $\omega(X)\leq\omega(X\cap Z)+\omega(X\cap Z^c)\leq0+\omega(X)$, so $Z\in\mathcal{M}_\omega$. Complements are immediate.\\
If $E,F\in\mathcal{M}_\omega$, iterating the two clean cuts gives $\omega(X)=\sum_{\epsilon,\eta\in\{0,1\}}\omega(X\cap E^\epsilon\cap F^\eta)$, where $E^1=E$, $E^0=E^c$; hence all finite Boolean combinations of $E,F$ are measurable.\\
For disjoint $E_i\in\mathcal{M}_\omega$, set $E=\bigsqcup_iE_i$ and $F_N=(\bigcup_{i=1}^{N}E_i)^c$.\\
Finite splitting gives $\omega(X)=\sum_{i=1}^{N}\omega(X\cap E_i)+\omega(X\cap F_N)\geq\sum_{i=1}^{N}\omega(X\cap E_i)+\omega(X\cap E^c)$.\\
Letting $N\to\infty$ and using subadditivity for the reverse inequality yields $\omega(X)=\sum_{i=1}^{\infty}\omega(X\cap E_i)+\omega(X\cap E^c)$; hence $E\in\mathcal{M}_\omega$, and $X=E$ gives countable additivity.\\
General countable unions follow from $F_1=E_1$, $F_n=E_n\setminus\bigcup_{i<n}E_i$.\\
Finally, if $E\in\mathcal{M}_\omega$ and $Z\in\mathcal{Z}_\omega$, then $E=(E\setminus Z)\sqcup(E\cap Z)$ and $E\cup Z=E\sqcup(Z\setminus E)$ give $\omega(E\setminus Z)=\omega(E)$ and $\omega(E\cup Z)=\omega(E)$; also $E\triangle Z=(E\setminus Z)\sqcup(Z\setminus E)$ has measure $\omega(E)$.\\
\hlt{Structural Commentary}: Measurable sets preserve countable additivity.\\
\hlt{Key Insight}: Measurable cuts turn subadditivity into equality.
\end{proof}
\end{theorem}

\begin{definition} \label{def:lebesgue-sigma-algebra-caratheodory} \hlt{Lebesgue $\sigma$-Algebra via Caratheodory}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{L}^n\defeq\{E\subseteq\R^n:E$ is $m_n^\ast$-measurable$\}$ is the \hlt{Lebesgue $\sigma$-algebra}.
\item $m\defeq m_n^\ast|_{\mathcal{L}^n}$ is \hlt{Lebesgue measure}; for $E\in\mathcal{L}^n$, write $m(E)=m_n^\ast(E)$.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:cells-lebesgue-measurable-volume} \hlt{Cells Are Lebesgue Measurable}\\
For every coordinate cell $I\subseteq\R^n$, $I\in\mathcal{L}^n$ and $m(I)=\operatorname{vol}_n(I)$.
\begin{proof}
Since $I\in\mathcal{E}$, Theorem \ref{thm:rudin-measure-extension} gives $I\in\mathcal{L}^n$ and $m(I)=m_n^\ast(I)=\operatorname{vol}_n(I)$.\\
For the open-cell split, fix $A\subseteq\R^n$ with $m_n^\ast(A)<+\infty$.\\
Set $I_k\defeq\{x\in I:\operatorname{dist}(x,I^c)>k^{-1}\}$, $\eta_k\defeq m_n^\ast(I\setminus I_k)\to0$. Since $\operatorname{dist}(A\cap I_k,A\setminus I)>0$, Theorem \ref{thm:outer-measure-positive-distance-additivity} gives $m_n^\ast(A)\geq m_n^\ast(A\cap I_k)+m_n^\ast(A\setminus I)\geq m_n^\ast(A\cap I)+m_n^\ast(A\setminus I)-\eta_k$; let $k\to\infty$ and apply Lemma \ref{lem:caratheodory-finite-measure-test}.\\
\hlt{Structural Commentary}: Cells are the bridge between coordinate volume and Lebesgue measure.\\
\hlt{Key Insight}: Shrink, split, let the collar vanish.
\end{proof}
\end{proposition}

\begin{lemma} \label{lem:open-sets-countable-unions-rational-boxes} \hlt{Open Sets Are Countable Unions of Rational Boxes}\\
For every open $G\subseteq\R^n$, there is a countable $\Sigma\subseteq\{\prod_{i=1}^{n}(a_i,b_i):a_i,b_i\in\Q,\ a_i<b_i\}$ such that $G=\bigcup_{B\in\Sigma}B$.
\begin{proof}
Let $\mathcal{B}_{\Q}\defeq\{\prod_{i=1}^{n}(a_i,b_i):a_i,b_i\in\Q,\ a_i<b_i\}$; then $\mathcal{B}_{\Q}$ is countable since $\Q^{2n}$ is countable.\\
Set $\Sigma\defeq\{B\in\mathcal{B}_{\Q}:B\subseteq G\}$. The inclusion $\bigcup_{B\in\Sigma}B\subseteq G$ is immediate.\\
If $x\in G$, choose $r>0$ with $B_r(x)\subseteq G$, then choose $a_i,b_i\in\Q$ with $x_i-r/n<a_i<x_i<b_i<x_i+r/n$.\\
Thus $x\in B\defeq\prod_i(a_i,b_i)\subseteq B_r(x)\subseteq G$, so $x\in\bigcup_{B\in\Sigma}B$.\\
\hlt{Structural Commentary}: Rational boxes give a countable Euclidean base.\\
\hlt{Key Insight}: Openness supplies a ball; density of $\Q$ supplies rational box endpoints inside it.
\end{proof}
\end{lemma}

\begin{theorem} \label{thm:lebesgue-measure-basic-properties-checklist} \hlt{Lebesgue Measure Basic Properties}\\
Let $m=m_n^\ast|_{\mathcal{L}^n}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{B}(\R^n)\subseteq\mathcal{L}^n$, and $\mathcal{L}^n$ is closed under $E\mapsto E^c$, countable $\cup$, and countable $\cap$.
\item For $E,F,E_k\in\mathcal{L}^n$, $0\leq m(E)$, $E\subseteq F\Rightarrow m(E)\leq m(F)$, and $m(\bigcup_kE_k)\leq\sum_km(E_k)$.
\item If $(E_k)\subseteq\mathcal{L}^n$ are pairwise disjoint, then $m(\bigsqcup_kE_k)=\sum_km(E_k)$.
\item $m([0,1]^n)=1$, and $E\in\mathcal{L}^n$, $t\in\R^n$ imply $E+t\in\mathcal{L}^n$ and $m(E+t)=m(E)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By Theorems \ref{thm:rudin-measure-extension}, \ref{thm:caratheodory-measurable-sets-sigma-algebra}, $\mathcal{L}^n$ is a $\sigma$-algebra, hence closed under complements and countable unions/intersections. Open boxes are elementary, hence in $\mathcal{L}^n$; by Lemma \ref{lem:open-sets-countable-unions-rational-boxes}, each open $G=\bigcup_jB_j$ with rational open boxes $B_j$, so $G\in\mathcal{L}^n$. Thus $\mathcal{B}(\R^n)=\sigma(\mathcal{O}_{\R^n})\subseteq\mathcal{L}^n$.
\item Since $m=m_n^\ast|_{\mathcal{L}^n}$, nonnegativity follows from $m_n^\ast\geq0$. If $E\subseteq F$, then $m(E)\leq m(F)$ by outer-measure monotonicity. Also $m(\bigcup_kE_k)=m_n^\ast(\bigcup_kE_k)\leq\sum_km_n^\ast(E_k)=\sum_km(E_k)$ by outer-measure subadditivity.
\item If $(E_k)\subseteq\mathcal{L}^n$ pairwise disjoint, by Theorem \ref{thm:caratheodory-measurable-sets-sigma-algebra}, $m$ countably additive on $\mathcal{L}^n$, so $m(\bigsqcup_kE_k)=\sum_km(E_k)$.
\item By Theorem \ref{thm:outer-measure-agrees-with-cell-volume}, since $[0,1]^n$ is a cell, $m([0,1]^n)=\operatorname{vol}_n([0,1]^n)=1$.\\
For translation, let $A\subseteq\R^n$. By Theorem \ref{thm:outer-measure-translation-invariance} and\\
$E\in\mathcal{L}^n$, $m_n^\ast(A)=m_n^\ast(A-t)=m_n^\ast((A-t)\cap E)+m_n^\ast((A-t)\cap E^c)=m_n^\ast(A\cap(E+t))+m_n^\ast(A\cap(E+t)^c)$; hence $E+t\in\mathcal{L}^n$, and $m(E+t)=m_n^\ast(E+t)=m_n^\ast(E)=m(E)$.
\end{enumerate}
\hlt{Structural Commentary}: The wish list is recovered only after restricting outer measure to measurable sets.\\
\hlt{Key Insight}: The box-cover construction gives size; the measurable-set condition restores additivity.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:lebesgue-measure-unique-on-open-cells} \hlt{Uniqueness from Open Cells}\\
Let $\nu$ be a measure on $\mathcal{L}^n$ with $\nu(I)=\operatorname{vol}_n(I)$ for every open cell $I\subseteq\R^n$. Then $\nu=m$ on $\mathcal{L}^n$.
\begin{proof}
Fix $N\in\N$, $Q_N=(-N,N)^n$. If $E\in\mathcal{L}^n$ and $E\subseteq Q_N$, then any open-cell cover $E\subseteq\bigcup_jJ_j$ gives $\nu(E)\leq\sum_j\nu(J_j)=\sum_j\operatorname{vol}_n(J_j)$, hence $\nu(E)\leq m(E)$ by infimum.\\
Applying this to $E$ and $Q_N\setminus E$, and using $\nu(Q_N)=m(Q_N)<+\infty$, yields $\nu(E)+\nu(Q_N\setminus E)=m(E)+m(Q_N\setminus E)$ with each left summand $\leq$ its right counterpart; hence $\nu(E)=m(E)$.\\
For arbitrary $E\in\mathcal{L}^n$, set $A_1=E\cap Q_1$, $A_N=E\cap(Q_N\setminus Q_{N-1})$ for $N>1$.\\
Then $E=\bigsqcup_NA_N$, each $A_N\subseteq Q_N$, and countable additivity gives $\nu(E)=\sum_N\nu(A_N)=\sum_Nm(A_N)=m(E)$.\\
\hlt{Structural Commentary}: Agreement on coordinate cells determines the whole Lebesgue measure.\\
\hlt{Key Insight}: Covers give $\nu\leq m$ locally; finite complements force equality.
\end{proof}
\end{theorem}

\begin{method} \label{meth:translation-invariance-caratheodory} \hlt{Translation Invariance by Moving the Test Set}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $t\in\R^n$ and $A\subseteq\R^n$, translated box covers give $m_n^\ast(A+t)=m_n^\ast(A)$.
\item If $E\in\mathcal{L}^n$, then for all $A\subseteq\R^n$, $(A\cap(E+t))-t=(A-t)\cap E$ and $(A\cap(E+t)^c)-t=(A-t)\cap E^c$.
\item Apply the Caratheodory split for $E$ to $A-t$ to get $E+t\in\mathcal{L}^n$ and $m(E+t)=m(E)$.
\end{enumerate}
\hlt{Key Insight}: Translation invariance of outer measure transports the clean-splitting identity.
\end{method}

\begin{proposition} \label{prop:borel-hierarchy-first-examples} \hlt{First Borel Hierarchy Examples}\\
If $G=\bigcap_nU_n$ with $U_n$ open or $F=\bigcup_nC_n$ with $C_n$ closed, then $G,F\in\mathcal{B}(\R^p)\subseteq\mathcal{L}^p$. More generally, countable $\cup,\cap$ operations starting from open or closed sets stay Borel. In $\R$, $\{0\}=\bigcap_{n\geq1}(-1/n,1/n)$ is $G_\delta$ but not open, and $(0,1)=\bigcup_{n\geq2}[1/n,1-1/n]$ is $F_\sigma$ but not closed.
\begin{proof}
Since $\mathcal{B}(\R^p)=\sigma(\mathcal{O}_{\R^p})$, it contains open sets, closed sets by complements, and countable $\cup,\cap$ by $\sigma$-algebra closure; Proposition \ref{prop:sigma-algebra-closure-and-generation} gives the stated permanence, and Theorem \ref{thm:lebesgue-measure-basic-properties-checklist} gives $\mathcal{B}(\R^p)\subseteq\mathcal{L}^p$. The identities follow from $x=0\iff \abs{x}<1/n$ for all $n$ and $0<x<1\iff \exists n\geq2:1/n\leq x\leq1-1/n$; $\{0\}$ is not open and $(0,1)$ is not closed in the usual topology.\\
\hlt{Structural Commentary}: Borel sets are generated by countable topological operations.\\
\hlt{Key Insight}: Countable intersections and unions quickly leave the class of open and closed sets.
\end{proof}
\end{proposition}

\begin{perspective} \label{per:outer-measure-not-enough} \hlt{Why Not Measure Every Subset?}\\
Vitali-type sets show $m_1^\ast$ is not additive on $2^\R$: disjoint unions and refinements can break additivity.\\
Thus measure is $m^\ast$ restricted to clean splitters $E$, where $m^\ast(X)=m^\ast(X\cap E)+m^\ast(X\cap E^c)$ for every test set $X$.\\
\hlt{Key Insight}: Outer measure is universal; measure lives on clean splitters.
\end{perspective}

\begin{corollary} \label{cor:measure-continuity-up-down} \hlt{Measure Continuity}\\
Let $\mu$ be a measure. If $E_k\uparrow E$, then $\mu(E_k)\uparrow\mu(E)$. If $F_k\downarrow F$ and $\mu(F_1)<+\infty$, then $\mu(F_k)\downarrow\mu(F)$.
\begin{proof}
Disjointize $E_k\uparrow E$ by $E'_1=E_1$ and $E'_k=E_k\setminus E_{k-1}$; countable additivity makes $\mu(E_k)$ the $k$th partial sum for $\mu(E)$. For $F_k\downarrow F$, apply this to $F_1\setminus F_k\uparrow F_1\setminus F$ and subtract from finite $\mu(F_1)$.\\
\hlt{Structural Commentary}: Monotone set limits are disjoint series or complements of such series.\\
\hlt{Key Insight}: Downward continuity needs one finite anchor so subtraction is legal.
\end{proof}
\end{corollary}

\begin{definition} \label{def:null-sets-and-real-line-measure-zero} \hlt{Null Sets and Real-Line Measure Zero}\\
In a measure space, $N$ is \hlt{null} if it is measurable and $\mu(N)=0$.\\
A set $A\subseteq\R$ has \hlt{measure zero} if, for every $\varepsilon>0$, open intervals $O_n$ cover $A$ and satisfy $\sum_{n=1}^{\infty}\abs{O_n}\leq\varepsilon$.\\
For Lebesgue measure, null sets are also \hlt{zero sets}.\\
A subset of a compact interval has \hlt{content zero} if the same condition holds with finitely many intervals.
\end{definition}

\begin{proposition} \label{prop:outer-null-sets-lebesgue-measurable} \hlt{Outer-Null Sets Are Lebesgue Measurable}\\
If $Z\subseteq\R^n$ and $m_n^\ast(Z)=0$, then $Z\in\mathcal{L}^n$, $m(Z)=0$, and $\mathcal{P}(Z)\subseteq\mathcal{L}^n$ with $m(W)=0$ for every $W\subseteq Z$.
\begin{proof}
For $A\subseteq\R^n$, monotonicity gives $m_n^\ast(A\cap Z)=0$.\\
Hence $m_n^\ast(A)\leq m_n^\ast(A\cap Z)+m_n^\ast(A\cap Z^c)=m_n^\ast(A\cap Z^c)\leq m_n^\ast(A)$, $Z$ satisfies Caratheodory split and $m(Z)=0$.\\
If $W\subseteq Z$, then $m_n^\ast(W)\leq m_n^\ast(Z)=0$, and the same argument gives $W\in\mathcal{L}^n$, $m(W)=0$.\\
\hlt{Structural Commentary}: Lebesgue measure is complete because outer-null subsets split every test set trivially.\\
\hlt{Key Insight}: A zero piece cannot disturb additivity.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:finite-countable-cantor-null-sets} \hlt{Basic Measure-Zero Examples on $\R$}\\
Every finite $A\subseteq\R$ has content $0$; every countable $A\subseteq\R$ has measure $0$; Cantor set $P$ has content $0$.
\begin{proof}
Let $\varepsilon>0$. If $A=\{a_1,\ldots,a_N\}$ with $N\geq1$, choose open intervals $I_j\ni a_j$ with $\abs{I_j}<\varepsilon/N$; then $A\subseteq\bigcup_{j=1}^NI_j$ and $\sum_{j=1}^N\abs{I_j}<\varepsilon$. The case $A=\emptyset$ is trivial.\\
If $A=\{a_j:j\in\N\}$, choose open intervals $I_j\ni a_j$ with $\abs{I_j}<\varepsilon 2^{-j}$; then $A\subseteq\bigcup_jI_j$ and $\sum_j\abs{I_j}<\varepsilon$.\\
For $P$, choose $N$ with $(2/3)^N<\varepsilon/2$. Write $E_N=\bigcup_{k=1}^{2^N}J_k$, where each $J_k$ is closed and $\abs{J_k}=3^{-N}$. Choose open $I_k\supseteq J_k$ with $\abs{I_k}<3^{-N}+\varepsilon/2^{N+1}$. Since $P\subseteq E_N$, the $I_k$ cover $P$, and $\sum_{k=1}^{2^N}\abs{I_k}<\left(\frac23\right)^N+\frac{\varepsilon}{2}<\varepsilon$.\\
\hlt{Structural Commentary}: Zero size means arbitrarily small interval covers.\\
\hlt{Key Insight}: Lists split $\varepsilon$ by terms; Cantor stages have length $(2/3)^N\to0$.
\end{proof}
\end{proposition}

\begin{corollary} \label{cor:uncountable-lebesgue-null-sets} \hlt{Uncountable Null Sets Exist}\\
There exists $P\subseteq\R$ with $\abs{P}=2^{\aleph_0}$ and $m(P)=0$.
\begin{proof}
Let $P\defeq\{\sum_{k=1}^{\infty}\varepsilon_k3^{-k}:\varepsilon_k\in\{0,2\}\}$. The map $(\varepsilon_k)\mapsto\sum_k\varepsilon_k3^{-k}$ identifies $P$ with $\{0,2\}^{\N}$, so $\abs{P}=2^{\aleph_0}$.\\
At stage $N$, $P$ is covered by $2^N$ closed intervals of length $3^{-N}$, hence $m_1^\ast(P)\leq(2/3)^N$ for all $N$; letting $N\to\infty$ gives $m_1^\ast(P)=0$. By outer-null completeness, $P\in\mathcal{L}^1$ and $m(P)=0$.\\
\hlt{Structural Commentary}: Cantor coding separates cardinality from length.\\
\hlt{Key Insight}: Null need not mean countable.
\end{proof}
\end{corollary}

\begin{corollary} \label{cor:real-line-lebesgue-measure-checks} \hlt{Real-Line Lebesgue Measure Checks}\\
Let $m$ be Lebesgue measure on $\R$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $a<b\Rightarrow m((a,b))=m((a,b])=m([a,b))=m([a,b])=b-a$.
\item $A\subseteq\R$ countable $\Rightarrow A\in\mathcal{B}(\R)$ and $m(A)=0$.
\item $\emptyset\neq G\subseteq\R$ open $\Rightarrow m(G)>0$.
\item $K\subseteq\R$ compact $\Rightarrow m(K)<+\infty$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item By cell volume, $m([a,b])=b-a$; endpoint sets finite null; add or delete $\{a\},\{b\}$ does not change measure.
\item If $A=\{a_k:k\in J\}$ with $J$ finite or countable, then $A=\bigcup_{k\in J}\{a_k\}\in\mathcal{B}(\R)$; $m(A)\leq\sum_{k\in J}m(\{a_k\})=0$.
\item Choose $(a,b)\subseteq G$ with $a<b$; by monotonicity and (i), $m(G)\geq m((a,b))=b-a>0$.
\item Compact $K$ is bounded, so $K\subseteq[-R,R]$ for some $R>0$; hence $m(K)\leq m([-R,R])=2R<+\infty$.
\end{enumerate}
\hlt{Structural Commentary}: On $\R$, length, null endpoints, and compact boundedness give quick measure estimates.\\
\hlt{Key Insight}: Nonempty open sets contain intervals; compact sets fit inside one.
\end{proof}
\end{corollary}

\begin{example} \label{ex:fat-cantor-set-positive-measure-empty-interior} \hlt{A Positive-Measure Cantor-Type Set with No Interval}\\
Let $S_0=[0,1]$ and, for $n\geq1$, let $R_n$ be the union of the middle open intervals of length $2^{-2n}$ removed from the $2^{n-1}$ components of $S_{n-1}$; set $S_n\defeq S_{n-1}\setminus R_n$ and $S\defeq\bigcap_{n=0}^{\infty}S_n$.\\
Then $m(R_n)=2^{-n-1}$, $\sum_{n\geq1}m(R_n)=1/2$, so $S=[0,1]\setminus\bigcup_nR_n$ is compact with $m(S)=1/2$.\\
Since every component of $S_n$ has length $\leq2^{-n}$, no interval $(a,b)$ can lie in $S$; hence $\operatorname{int}S=\emptyset$.\\
\hlt{Structural Commentary}: Summable removals leave measure; repeated splitting kills interior.\\
\hlt{Key Insight}: Total removed length controls measure; repeated splitting controls interior.
\end{example}

\begin{remark} \label{rem:borel-null-regular-measurable-sets} \hlt{Borel Sets, Null Sets, and Regular Approximation}\\
Open, closed, and Borel subsets of $\R^p$ are measurable. Each $A\in\mathcal{M}(\mu)$ admits closed--open approximations $F\subseteq A\subseteq G$ with $\mu(G\setminus A)$ and $\mu(A\setminus F)$ arbitrarily small. In Lebesgue measure, one may take $F_\sigma\subseteq A\subseteq G_\delta$ with $\mu(G_\delta\setminus F_\sigma)=0$. Null sets form a $\sigma$-ring, and countable subsets of $\R^p$ are null.\\
\hlt{Structural Commentary}: Measurable sets are Borel sets modified by null sets.\\
\hlt{Key Insight}: Regularity explains why open and closed approximations are enough for integration.
\end{remark}

\begin{corollary} \label{cor:lebesgue-regularity-and-borel-completion-real-line} \hlt{Lebesgue Regularity and Borel Completion on $\R$}\\
For $A\in\mathcal{L}^1$,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $m(A)=\inf\{m(G):A\subseteq G,\ G$ open in $\R\}$.
\item $m(A)=\sup\{m(K):K\subseteq A,\ K$ compact in $\R\}$, finite-measure case applied to $A\cap(-n,n]$; $m(A)=+\infty$.
\item There exists $B\in\mathcal{B}(\R)$ with $A\subseteq B$ and $m(B\setminus A)=0$; equivalently, $\mathcal{L}^1$ is the completion of $\mathcal{B}(\R)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $\varepsilon>0$, regular approximation gives open $G\supseteq A$ with $m(G\setminus A)<\varepsilon$, hence $m(G)\leq m(A)+\varepsilon$.\\
Monotonicity gives the reverse inequality.
\item If $m(A)<+\infty$, choose closed $F\subseteq A$ with $m(A\setminus F)<\varepsilon$. Since $F\cap[-n,n]\uparrow F$, continuity from below gives compact $K=F\cap[-n,n]$ with $m(F)-m(K)<\varepsilon$, so $m(A)-m(K)<2\varepsilon$. If $m(A)=+\infty$, apply the finite case to $A_n=A\cap(-n,n]$ and let $n\to\infty$.
\item Choose open $G_k\supseteq A$ with $m(G_k\setminus A)<1/k$, and set $B=\bigcap_kG_k$. Then $B\in\mathcal{B}(\R)$, $A\subseteq B$, and $m(B\setminus A)=0$. Thus every Lebesgue set is Borel plus a null subset.\\
Conversely, Borel sets and null subsets are Lebesgue measurable.
\end{enumerate}
\hlt{Structural Commentary}: Lebesgue sets are Borel sets plus null-set ambiguity.\\
\hlt{Key Insight}: Open outer approximation and compact inner approximation are the usable form of regularity.
\end{proof}
\end{corollary}

\begin{proposition} \label{prop:real-line-local-measurability-unit-intervals} \hlt{Local Measurability on Unit Intervals}\\
Let $I_n\defeq(n,n+1]$, $n\in\Z$. For $E\subseteq\R$, $E\in\mathcal{L}^1$ iff $E\cap I_n\in\mathcal{L}^1$ for all $n\in\Z$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Since $I_n\in\mathcal{B}(\R)\subseteq\mathcal{L}^1$, $E\in\mathcal{L}^1$ implies $E\cap I_n\in\mathcal{L}^1$ for all $n$.
\item[$(\Leftarrow)$] If $E\cap I_n\in\mathcal{L}^1$ for all $n$, then $E=\bigcup_{n\in\Z}(E\cap I_n)\in\mathcal{L}^1$.
\end{enumerate}
\hlt{Structural Commentary}: A countable Borel exhaustion localizes measurability.\\
\hlt{Key Insight}: Local measurable pieces patch by countable unions.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:finite-measure-meets-every-unit-interval} \hlt{Finite Measure Meeting Every Unit Interval}\\
With $I_n\defeq(n,n+1]$, there exists $E\in\mathcal{B}(\R)$ such that $0<m(E\cap I_n)$ for all $n\in\Z$ and $m(E)<+\infty$.
\begin{proof}
Set $\varepsilon_n\defeq2^{-\abs{n}-3}$ and $E\defeq\bigcup_{n\in\Z}(n,n+\varepsilon_n]$. Then $(n,n+\varepsilon_n]\subseteq I_n$ are pairwise disjoint Borel intervals, so $E\in\mathcal{B}(\R)$ and $m(E\cap I_n)=\varepsilon_n>0$. Countable additivity gives $m(E)=\sum_{n\in\Z}\varepsilon_n=\frac38<+\infty$.\\
\hlt{Structural Commentary}: Finite measure can be spread over countably many cells.\\
\hlt{Key Insight}: Positive local masses can be summable globally.
\end{proof}
\end{proposition}

\begin{definition} \label{def:symmetric-difference} \hlt{Symmetric Difference}\\
For $A,B\subseteq X$, define $A\triangle B\defeq(A\setminus B)\cup(B\setminus A)=(A\cup B)\setminus(A\cap B)$.\\
Equivalently, $x\in A\triangle B$ iff $x$ belongs to exactly one of $A,B$.
\end{definition}

\begin{proposition} \label{prop:null-sets-sigma-ideal} \hlt{Null Sets Form a Sigma-Ideal}\\
Let $(X,\mathcal{M},\mu)$ be a measure space and $\mathcal{Z}\defeq\{N\in\mathcal{M}:\mu(N)=0\}$.\\
Then $\mathcal{Z}$ is a $\sigma$-ring, and $N\in\mathcal{Z}$, $E\in\mathcal{M}$ imply $N\cap E\in\mathcal{Z}$.
\begin{proof}
If $(N_k)\subseteq\mathcal{Z}$, then $\mu(\bigcup_kN_k)\leq\sum_k\mu(N_k)=0$, so $\bigcup_kN_k\in\mathcal{Z}$.\\
If $A,B\in\mathcal{Z}$, then $A\setminus B\subseteq A$, hence $A\setminus B\in\mathcal{Z}$. Finally, $N\cap E\subseteq N$ gives $N\cap E\in\mathcal{Z}$.\\
\hlt{Structural Commentary}: Null sets are stable under countable unions and measurable cuts.\\
\hlt{Key Insight}: Monotonicity makes every measurable piece of a null set null.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:completed-sigma-algebra} \hlt{Completed Sigma-Algebra}\\
Let $(X,\mathcal{M},\mu)$ be measure space, $\mathcal{Z}\defeq\{N\in\mathcal{M}:\mu(N)=0\}$; $\overline{\mathcal{M}}_\mu\defeq\{A\subseteq X:\exists E\in\mathcal{M},\ \exists N\in\mathcal{Z},\ A\triangle E\subseteq N\}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathcal{M}\subseteq\overline{\mathcal{M}}_\mu$, and $\overline{\mathcal{M}}_\mu$ is a $\sigma$-algebra;
\item $A\subseteq N\in\mathcal{Z}\Longrightarrow A\in\overline{\mathcal{M}}_\mu$;
\item $\overline{\mathcal{M}}_\mu=\{E\cup Z:E\in\mathcal{M},\ \exists N\in\mathcal{Z},\ Z\subseteq N\}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $E\in\mathcal{M}$, $E\triangle E=\emptyset$, so $\mathcal{M}\subseteq\overline{\mathcal{M}}_\mu$. If $A\triangle E\subseteq N$, then $A^c\triangle E^c=A\triangle E\subseteq N$.\\
If $A_k\triangle E_k\subseteq N_k$, then $(\bigcup_kA_k)\triangle(\bigcup_kE_k)\subseteq\bigcup_kN_k\in\mathcal{Z}$. Hence $\overline{\mathcal{M}}_\mu$ is a $\sigma$-algebra.
\item If $A\subseteq N\in\mathcal{Z}$, then $A\triangle\emptyset\subseteq N$, so $A\in\overline{\mathcal{M}}_\mu$.
\item If $A\triangle E\subseteq N$, then $A=(E\setminus N)\cup(A\cap N)$, with $E\setminus N\in\mathcal{M}$ and $A\cap N\subseteq N$.\\
Conversely, if $A=E\cup Z$ and $Z\subseteq N\in\mathcal{Z}$, then $A\triangle E\subseteq Z\subseteq N$.
\end{enumerate}
\hlt{Structural Commentary}: Completion adds all null modifications of measurable sets.\\
\hlt{Key Insight}: Equality modulo null sets is the membership criterion.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:null-ideal-and-measure-completion} \hlt{Completed Measure}\\
Let $(X,\mathcal{M},\mu)$ be a measure space, $\mathcal{Z}\defeq\{N\in\mathcal{M}:\mu(N)=0\}$,\\
and $\overline{\mathcal{M}}_\mu\defeq\{A\subseteq X:\exists E\in\mathcal{M},\ \exists N\in\mathcal{Z},\ A\triangle E\subseteq N\}$. For $A\in\overline{\mathcal{M}}_\mu$, define $\overline{\mu}(A)\defeq\mu(E)$ whenever $E\in\mathcal{M}$ and $A\triangle E\subseteq N\in\mathcal{Z}$. Then $\overline{\mu}$ is a well-defined complete measure on $\overline{\mathcal{M}}_\mu$, and $\overline{\mu}|_{\mathcal{M}}=\mu$.
\begin{proof}
If $A\triangle E\subseteq N_1$ and $A\triangle F\subseteq N_2$, with $E,F\in\mathcal{M}$ and $N_1,N_2\in\mathcal{Z}$, then $E\triangle F\subseteq N_1\cup N_2$, so $\mu(E)=\mu(F)$. Hence $\overline{\mu}$ is well defined; since $E\triangle E=\emptyset$, $\overline{\mu}|_{\mathcal{M}}=\mu$.\\
For disjoint $(A_k)\subseteq\overline{\mathcal{M}}_\mu$, choose $E_k\in\mathcal{M}$ and $N_k\in\mathcal{Z}$ with $A_k\triangle E_k\subseteq N_k$, and set $N\defeq\bigcup_kN_k$.\\
Then $E_i\cap E_j\subseteq N$ for $i\neq j$. Put $F_1\defeq E_1$ and $F_k\defeq E_k\setminus\bigcup_{j<k}E_j$.\\
Then $(F_k)$ disjoint, $A_k\triangle F_k\subseteq N$, $(\bigcup_kA_k)\triangle(\bigcup_kF_k)\subseteq N$, so $\overline{\mu}\left(\bigcup_kA_k\right)=\mu\left(\bigcup_kF_k\right)=\sum_k\mu(F_k)=\sum_k\overline{\mu}(A_k)$.\\
If $A\subseteq B\in\overline{\mathcal{M}}_\mu$ and $\overline{\mu}(B)=0$, choose $E\in\mathcal{M}$ and $N\in\mathcal{Z}$ with $B\triangle E\subseteq N$.\\
Then $\mu(E)=0$, so $A\subseteq B\subseteq E\cup N\in\mathcal{Z}$; hence $A\in\overline{\mathcal{M}}_\mu$ and $\overline{\mu}(A)=0$.\\
\hlt{Structural Commentary}: The completed measure ignores null errors.\\
\hlt{Key Insight}: Null perturbations do not change countable additivity.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:lebesgue-measurable-non-borel-sets-exist} \hlt{Lebesgue Measurable Non-Borel Sets Exist}\\
There exists $E\subseteq\R$ such that $E\in\mathcal{L}^1$ and $E\notin\mathcal{B}(\R)$; equivalently, $\mathcal{B}(\R)\subsetneq\mathcal{L}^1$.
\begin{proof}
By Lemma \ref{lem:open-sets-countable-unions-rational-boxes}, $\mathcal{B}(\R)$ is countably generated, hence $\abs{\mathcal{B}(\R)}\leq\mathfrak{c}$. Let $C$ be the Cantor set.\\
Since $\abs{C}=\mathfrak{c}$ and $m(C)=0$, Proposition \ref{prop:outer-null-sets-lebesgue-measurable} gives $\mathcal{P}(C)\subseteq\mathcal{L}^1$, so $\abs{\mathcal{L}^1}\geq2^{\mathfrak{c}}$.\\
Also $\mathcal{L}^1\subseteq\mathcal{P}(\R)$, so $\abs{\mathcal{L}^1}\leq2^{\mathfrak{c}}$; hence $\abs{\mathcal{L}^1}=2^{\mathfrak{c}}$. By Cantor's theorem, $\mathfrak{c}<2^{\mathfrak{c}}$, so $\mathcal{B}(\R)\subsetneq\mathcal{L}^1$.\\
\hlt{Structural Commentary}: Countable generation is smaller than completion by null subsets.\\
\hlt{Key Insight}: Completeness creates more measurable sets than topology can generate.
\end{proof}
\end{theorem}

\begin{definition} \label{def:charges-and-variation-measures} \hlt{Variation Candidates of a Charge}\\
Let $\lambda$ be a charge on $\mathcal{M}$. For $E\in\mathcal{M}$, define
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mathfrak{P}_{\mathcal{M}}(E)\defeq\{(A_j)_{j=1}^{n}:n\in\N,\ E=\bigsqcup_{j=1}^{n}A_j,\ A_j\in\mathcal{M}\}$, the finite measurable partitions of $E$;
\item $\pi_\lambda(E)\defeq\sup\{\lambda(A):A\subseteq E,\ A\in\mathcal{M}\}$, the \hlt{positive variation} candidate;
\item $\nu_\lambda(E)\defeq\sup\{\sum_{j=1}^{n}\abs{\lambda(A_j)}:(A_j)_{j=1}^{n}\in\mathfrak{P}_{\mathcal{M}}(E)\}$, the \hlt{total variation} candidate.
\end{enumerate}
For $E=\emptyset$, the empty partition is allowed and contributes $0$.
\end{definition}

\begin{proposition} \label{prop:charges-linear-combinations-and-variation} \hlt{Charge Operations and Variation Measures}\\
Let $\lambda,\lambda_1,\ldots,\lambda_m$ be charges on $\mathcal{M}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $a_1,\ldots,a_m\in\R$, then $\Lambda\defeq\sum_{j=1}^{m}a_j\lambda_j$ is a charge.
\item If $f\in\mathscr{L}^1(\mu)$ on $(X,\mathcal{M},\mu)$, then $\lambda_f(A)\defeq\int_A f\,d\mu$ is a charge.
\item The set functions $\pi_\lambda$ and $\nu_\lambda$ are measures on $\mathcal{M}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $E=\bigsqcup_nE_n$, then $\Lambda(E)=\sum_{j=1}^{m}a_j\sum_n\lambda_j(E_n)=\sum_n\Lambda(E_n)$.
\item By Theorem \ref{thm:integral-countably-additive-set-function}, $A\mapsto\int_A f\,d\mu$ is countably additive, and $\abs{\lambda_f(A)}\leq\int_X\abs{f}\,d\mu<+\infty$.
\item Let $E=\bigsqcup_nE_n$. For $A\subseteq E$, $A=\bigsqcup_n(A\cap E_n)$, so $\lambda(A)\leq\sum_n\pi_\lambda(E_n)$; by suprema, $\pi_\lambda(E)\leq\sum_n\pi_\lambda(E_n)$. Conversely, for $N\in\N$ choose $A_n\subseteq E_n$ with $\lambda(A_n)>\pi_\lambda(E_n)-\varepsilon2^{-n}$ for $n\leq N$.\\
Then $\pi_\lambda(E)\geq\sum_{n=1}^{N}\pi_\lambda(E_n)-\varepsilon$. Let $N\to\infty$ and $\varepsilon\downarrow0$. For $\nu_\lambda$, every $(A_j)_{j=1}^{r}\in\mathfrak{P}_{\mathcal{M}}(E)$ satisfies $\sum_{j=1}^{r}\abs{\lambda(A_j)}\leq\sum_n\nu_\lambda(E_n)$ after refining by $A_j\cap E_n$, hence $\nu_\lambda(E)\leq\sum_n\nu_\lambda(E_n)$. Conversely, combine near-optimal finite partitions of $E_1,\ldots,E_N$, add $\bigcup_{n>N}E_n$ as one part; let $N\to\infty$.
\end{enumerate}
\hlt{Structural Commentary}: Variation turns signed additivity into measure.\\
\hlt{Key Insight}: Refine partitions against a disjoint decomposition.
\end{proof}
\end{proposition}

\begin{definition} \label{def:set-liminf-limsup} \hlt{Limit Inferior and Superior of Sets}\\
For $(E_n)\subseteq\mathcal{P}(X)$, define $\limsup_nE_n\defeq\bigcap_{m=1}^{\infty}\bigcup_{n\geq m}E_n$ and $\liminf_nE_n\defeq\bigcup_{m=1}^{\infty}\bigcap_{n\geq m}E_n$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Infinitely often}: $x\in\limsup_nE_n$ iff $\{n:x\in E_n\}$ is infinite;
\item \hlt{Eventually always}: $x\in\liminf_nE_n$ iff $\{n:x\notin E_n\}$ is finite;
\item \hlt{Set limit}: if $\limsup_nE_n=\liminf_nE_n$, write $\lim_nE_n$ for the common set.
\end{enumerate}
\end{definition}

\begin{proposition} \label{prop:set-limits-basic-laws} \hlt{Basic Laws for Set Limits}\\
Let $(E_n)\subseteq\mathcal{P}(X)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\emptyset\subseteq\liminf_nE_n\subseteq\limsup_nE_n\subseteq X$.
\item $E_n\uparrow E\Longrightarrow\liminf_nE_n=\limsup_nE_n=E=\bigcup_nE_n$.
\item $E_n\downarrow E\Longrightarrow\liminf_nE_n=\limsup_nE_n=E=\bigcap_nE_n$.
\item $(E_n)\subseteq\mathcal{M}\Longrightarrow\liminf_nE_n,\limsup_nE_n\in\mathcal{M}$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $\bigcap_m\bigcup_{n\geq m}E_n\subseteq\bigcup_{n\geq1}E_n\subseteq X$ and $\bigcap_{n\geq m}E_n\subseteq\bigcup_{n\geq m}E_n$ for all $m$, the inclusions follow.
\item If $E_n\uparrow E$, then $\{n:x\in E_n\}$ is nonempty iff it is cofinite, hence $\liminf_nE_n=\limsup_nE_n=\bigcup_nE_n=E$.
\item If $E_n\downarrow E$, then $\{n:x\in E_n\}$ is infinite iff it is all of $\N$, hence $\liminf_nE_n=\limsup_nE_n=\bigcap_nE_n=E$.
\item If $(E_n)\subseteq\mathcal{M}$, then $\bigcup_{n\geq m}E_n,\bigcap_{n\geq m}E_n\in\mathcal{M}$ $\forall m$; countable closure gives $\limsup_nE_n,\liminf_nE_n\in\mathcal{M}$.
\end{enumerate}
\hlt{Structural Commentary}: Set limsup means infinitely often; set liminf means eventually always.\\
\hlt{Key Insight}: Translate the definitions into pointwise membership statements.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:measure-liminf-limsup-inequalities} \hlt{Measure Bounds for Set Limits}\\
Let $(E_n)$ be measurable in a measure space $(X,\mathcal{M},\mu)$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $\mu(\liminf_nE_n)\leq\liminf_n\mu(E_n)$.
\item $\mu(\bigcup_{n=1}^{\infty}E_n)<+\infty\Longrightarrow\limsup_n\mu(E_n)\leq\mu(\limsup_nE_n)$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Set $A_m\defeq\bigcap_{n\geq m}E_n$. Then $A_m\uparrow\liminf_nE_n$ and $A_m\subseteq E_n$ for all $n\geq m$, so $\mu(A_m)\leq\inf_{n\geq m}\mu(E_n)$.\\
By continuity from below, $\mu(\liminf_nE_n)=\lim_m\mu(A_m)\leq\sup_m\inf_{n\geq m}\mu(E_n)=\liminf_n\mu(E_n)$.
\item Set $B_m\defeq\bigcup_{n\geq m}E_n$. $B_m\downarrow\limsup_nE_n$, $\mu(B_1)<+\infty$, $E_n\subseteq B_m$ $\forall n\geq m$, so $\sup_{n\geq m}\mu(E_n)\leq\mu(B_m)$.\\
By continuity from above, $\limsup_n\mu(E_n)=\inf_m\sup_{n\geq m}\mu(E_n)\leq\lim_m\mu(B_m)=\mu(\limsup_nE_n)$.
\end{enumerate}
\hlt{Structural Commentary}: Tail intersections increase to liminf; tail unions decrease to limsup.\\
\hlt{Key Insight}: Compare each tail envelope to the corresponding tail bounds of the measures.
\end{proof}
\end{proposition}
