\subsubsection{Number Systems, Order, and Completeness}

The first pressure point in analysis is that the rational numbers are not enough. They form an ordered field, but they have gaps: sequences can point toward a number such as $\sqrt{2}$ without that number existing inside $\Q$. Completeness is the axiom that closes those gaps.

\begin{proposition} \label{prop:rational-square-root-two-gap} \hlt{The Rational Gap at $\sqrt{2}$}\\
There is no rational $p$ such that $p^2=2$.
\begin{proof}
Write $p=m/n$ in lowest terms. Then $m^2=2n^2$, so $m$ is even; write $m=2k$.\\
Hence $n^2=2k^2$, so $n$ is even, a contradiction.\\
\hlt{Structural Commentary}: This exhibits a gap in $\Q$.\\
\hlt{Key Insight}: $\sqrt{2}$ is not rational.
\end{proof}
\end{proposition}

\begin{example} \label{ex:rational-cut-around-square-root-two} \hlt{No Largest Lower Side and No Smallest Upper Side}\\
Let $A\defeq\{p\in\Q:p>0,\ p^2<2\}$, and $B\defeq\{p\in\Q:p>0,\ p^2>2\}$.\\
For every $p>0$ define $q\defeq p-\frac{p^2-2}{p+2}=\frac{2p+2}{p+2}$. Then $q^2-2=\frac{2(p^2-2)}{(p+2)^2}$.
Hence if $p\in A$, then $q\in A$ and $q>p$; if $p\in B$, then $q\in B$ and $0<q<p$. Thus $A$ has no largest element in $\Q$, and $B$ has no smallest element in $\Q$.
\end{example}

\begin{remark} \label{rem:rational-density-does-not-fill-gaps} \hlt{Density Does Not Mean Completeness}\\
Between two rationals $r<s$ there is another rational, namely $(r+s)/2$. Nevertheless, the preceding example shows that $\Q$ still has gaps. The real number system is introduced to fill these gaps.
\end{remark}

\begin{definition} \label{def:ordered-set-and-bounds} \hlt{Ordered Sets and Bounds}\\
An \hlt{order} on $S$ is a relation $<$ such that exactly one of $x<y$, $x=y$, or $y<x$ holds, and $x<y<z$ implies $x<z$. An \hlt{ordered set} is a set with an order. For $E\subseteq S$, an \hlt{upper bound} is a $\beta\in S$ with $x\leq\beta$ for all $x\in E$; then $E$ is \hlt{bounded above}. Lower bounds reverse the inequality.
\end{definition}

\begin{definition} \label{def:supremum-infimum-lub-property} \hlt{Supremum, Infimum, and Least-Upper-Bound Property}\\
For $E\subseteq S$, $\sup E$ is least upper bound of $E$, and $\inf E$ is greatest lower bound of $E$, when these exist. Ordered set $S$ has \hlt{least-upper-bound property} if every nonempty bounded-above subset of $S$ has a supremum in $S$.
\end{definition}

\begin{example} \label{ex:supremum-need-not-be-attained} \hlt{Suprema Need Not Be Elements}\\
In $\Q$, the set $E_1=\{r\in\Q:r<0\}$ and the set $E_2=\{r\in\Q:r\leq0\}$ both have supremum $0$, but $0\notin E_1$ and $0\in E_2$. If $E=\{1/n:n=1,2,3,\ldots\}$, then $\sup E=1$ while $\inf E=0$, and the infimum is not a member of $E$.
\end{example}

\begin{theorem} \label{thm:lub-property-gives-glb-property} \hlt{Least Upper Bounds Give Greatest Lower Bounds}\\
If $S$ has the least-upper-bound property, then every nonempty bounded-below set $B\subseteq S$ has an infimum.
\begin{proof}
Let $L$ be the set of lower bounds of $B$. Then $L$ is nonempty and every $x\in B$ is an upper bound of $L$, so $\alpha=\sup L$ exists. Since every $x\in B$ bounds $L$ above, $\alpha\leq x$ for all $x\in B$; hence $\alpha$ is a lower bound of $B$.\\
If $\beta>\alpha$, then $\beta\notin L$, so $\beta$ is not a lower bound. Thus $\alpha=\inf B$.\\
\hlt{Structural Commentary}: Infima are suprema of lower bounds.\\
\hlt{Key Insight}: Completeness above gives completeness below.
\end{proof}
\end{theorem}

\begin{definition} \label{def:field-and-ordered-field-analysis} \hlt{Fields and Ordered Fields}\\
A \hlt{field} is a set $F$ with addition and multiplication satisfying closure, commutativity, associativity, identities $0,1$ with $1\neq0$, additive inverses, multiplicative inverses for nonzero elements, and distributivity.\\
An \hlt{ordered field} is a field such that $y<z$ implies $x+y<x+z$, and $x>0$ and $y>0$ imply $xy>0$.
\end{definition}

\begin{proposition} \label{prop:field-axiom-consequences-analysis} \hlt{Basic Consequences of the Field Axioms}\\
Every field has cancellation, unique identities and inverses, $0x=0$, no zero divisors, and the sign rules
\begin{align}
(-x)y=-(xy)=x(-y),\qquad (-x)(-y)=xy. \nonumber
\end{align}
\begin{proof}
Cancellation comes from adding or multiplying by the relevant inverse; uniqueness follows from cancellation. Also $0x+0x=(0+0)x=0x$, so $0x=0$. If $x,y\neq0$ and $xy=0$, multiplying by $x^{-1}y^{-1}$ gives $1=0$, impossible. The sign rules follow because the displayed negatives are additive inverses of $xy$.\\
\hlt{Structural Commentary}: The field axioms license ordinary algebra.\\
\hlt{Key Insight}: Inverses force cancellation and signs.
\end{proof}
\end{proposition}

\begin{proposition} \label{prop:ordered-field-inequality-rules-analysis} \hlt{Basic Inequality Rules in Ordered Fields}\\
In every ordered field,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $x>0$ iff $-x<0$;
\item multiplying an inequality by a positive element preserves the inequality;
\item multiplying an inequality by a negative element reverses the inequality;
\item if $x\neq0$, then $x^2>0$, so in particular $1>0$;
\item if $0<x<y$, then $0<1/y<1/x$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $x>0$, then $0=-x+x>-x+0$, so $-x<0$; the converse is the same with $-x$.
\item If $y<z$ and $x>0$, then $z-y>0$, so $x(z-y)>0$ and hence $xy<xz$.
\item Apply (ii) to $-x>0$.
\item Either $x>0$ or $-x>0$, so $x^2>0$; take $x=1$ for $1>0$.
\item Since $x,y>0$, multiply $x<y$ by $(1/x)(1/y)>0$ to get $1/y<1/x$.
\end{enumerate}
\hlt{Structural Commentary}: Ordered fields make algebraic signs compatible with order.\\
\hlt{Key Insight}: Positivity decides whether multiplication preserves order.
\end{proof}
\end{proposition}
