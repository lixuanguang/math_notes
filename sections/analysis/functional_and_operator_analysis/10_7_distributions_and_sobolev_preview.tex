\subsection{Distributions and Sobolev Preview}

% **Role:** infinite-dimensional linear algebra powered by analysis.
%
% Test functions, distributions, weak derivatives, Sobolev spaces as bridge to PDE.

Distributions test against smooth compactly supported functions; Sobolev spaces package weak derivatives in normed or Hilbert spaces as a bridge to PDE.

\begin{definition} \label{def:test-functions-and-distributions} \hlt{Test Functions and Distributions}\\
Let $\Omega\subseteq\R^n$ be open. A \hlt{test function} is a smooth function $\varphi\in C_c^\infty(\Omega)$. A \hlt{distribution} on $\Omega$ is a continuous linear functional on the test-function space, written $u\in\mathcal{D}'(\Omega)$ and paired as $\langle u,\varphi\rangle$.
\end{definition}

\begin{definition} \label{def:weak-derivative} \hlt{Weak Derivative}\\
Let $u\in L_{\mathrm{loc}}^1(\Omega)$ and let $\alpha$ be a multi-index. A function $v\in L_{\mathrm{loc}}^1(\Omega)$ is the \hlt{weak derivative} $D^\alpha u$ if
\[
\int_{\Omega}u\,D^\alpha\varphi\,dx
=(-1)^{\abs{\alpha}}\int_{\Omega}v\,\varphi\,dx
\qquad\text{for every }\varphi\in C_c^\infty(\Omega).
\]
\end{definition}

\begin{definition} \label{def:sobolev-space-preview} \hlt{Sobolev Spaces}\\
For $1\leq p\leq\infty$ and an integer $k\geq0$, the Sobolev space $W^{k,p}(\Omega)$ consists of functions $u\in L^p(\Omega)$ whose weak derivatives $D^\alpha u$ lie in $L^p(\Omega)$ for every $\abs{\alpha}\leq k$. The norm is
\[
\norm{u}_{W^{k,p}}\defeq
\sum_{\abs{\alpha}\leq k}\norm{D^\alpha u}_{L^p}
\]
with the usual supremum modification when $p=\infty$.
\end{definition}
