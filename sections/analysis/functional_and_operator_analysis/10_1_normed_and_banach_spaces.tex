\subsubsection{Normed and Banach Spaces}

Normed spaces turn vector spaces into metric spaces in a way compatible with linear structure. Banach spaces are the complete normed spaces; they are the natural setting for infinite series, fixed-point arguments, bounded linear maps, and operator norms.

\begin{definition} \label{def:normed-and-banach-spaces} \hlt{Normed and Banach Spaces}\\
Let $\mathcal{X}$ be a vector space over $\mathbb{F}\in\{\R,\C\}$. A \hlt{norm} is a function $\norm{\cdot}:\mathcal{X}\to[0,\infty)$ such that, for all $x,y\in\mathcal{X}$ and $\lambda\in\mathbb{F}$,
\[
\norm{x}=0\iff x=0,\qquad
\norm{\lambda x}=\abs{\lambda}\norm{x},\qquad
\norm{x+y}\leq\norm{x}+\norm{y}.
\]
The metric $d(x,y)\defeq\norm{x-y}$ makes $\mathcal{X}$ a \hlt{normed space}. If this metric is complete, $\mathcal{X}$ is a \hlt{Banach space}.
\end{definition}

\begin{definition} \label{def:bounded-linear-maps-and-operator-norm} \hlt{Bounded Linear Maps and Operator Norm}\\
For normed spaces $\mathcal{X},\mathcal{Y}$, a linear map $T:\mathcal{X}\to\mathcal{Y}$ is \hlt{bounded} if there exists $C\geq0$ such that $\norm{Tx}_{\mathcal{Y}}\leq C\norm{x}_{\mathcal{X}}$ for all $x\in\mathcal{X}$.\\
The \hlt{operator norm} is
\[
\normop{T}\defeq \sup_{\norm{x}_{\mathcal{X}}\leq1}\norm{Tx}_{\mathcal{Y}}
=\sup_{x\neq0}\frac{\norm{Tx}_{\mathcal{Y}}}{\norm{x}_{\mathcal{X}}}.
\]
\end{definition}

\begin{theorem} \label{thm:bounded-linear-map-continuity-equivalences} \hlt{Bounded Equals Continuous for Linear Maps}\\
For a linear map $T:\mathcal{X}\to\mathcal{Y}$ between normed spaces, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $T$ is bounded.
\item $T$ is continuous at $0$.
\item $T$ is continuous everywhere.
\end{enumerate}
\begin{proof}
If $T$ is bounded, $\norm{T(x-y)}\leq\normop{T}\norm{x-y}$ gives global continuity. If $T$ is continuous at $0$, choose $\delta>0$ such that $\norm{x}<\delta$ implies $\norm{Tx}<1$. For $x\neq0$, apply this to $\delta x/(2\norm{x})$ to get $\norm{Tx}\leq 2\delta^{-1}\norm{x}$.\\
\hlt{Structural Commentary}: Linearity turns local continuity at one point into uniform norm control.\\
\hlt{Key Insight}: In normed linear spaces, continuity and boundedness are the same operator property.
\end{proof}
\end{theorem}

\begin{example} \label{ex:standard-banach-space-examples} \hlt{Standard Banach Space Examples}\\
The complete supremum spaces from Theorem \ref{thm:complete-supremum-metric-space} are Banach spaces under $\norm{\cdot}_{\infty}$. The $L^2$ space in Theorem \ref{thm:l2-schwarz-minkowski-and-completeness} is a Banach space under $\norm{\cdot}_2$ and, more specifically, a Hilbert space.
\end{example}
