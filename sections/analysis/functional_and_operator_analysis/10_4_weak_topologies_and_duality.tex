\subsection{Weak Topologies and Duality}

% **Role:** infinite-dimensional linear algebra powered by analysis.
%
% Weak convergence, weak-* convergence, dual spaces, reflexivity, Banach--Alaoglu.

Weak topologies replace norm control with scalar tests by functionals, exposing compactness and duality that strong topology can hide.

\begin{definition} \label{def:dual-space-and-weak-convergence} \hlt{Dual Space and Weak Convergence}\\
For a normed space $\mathcal{X}$, its continuous dual is $\mathcal{X}^*\defeq\LL(\mathcal{X},\mathbb{F})$. A sequence $(x_n)$ \hlt{converges weakly} to $x$, written $x_n\rightharpoonup x$, if
\[
\varphi(x_n)\to\varphi(x)\qquad\text{for every }\varphi\in\mathcal{X}^*.
\]
\end{definition}

\begin{definition} \label{def:weak-star-convergence} \hlt{Weak-* Convergence}\\
If $(\varphi_n)\subseteq\mathcal{X}^*$ and $\varphi\in\mathcal{X}^*$, then $\varphi_n$ \hlt{converges weak-*} to $\varphi$ if
\[
\varphi_n(x)\to\varphi(x)\qquad\text{for every }x\in\mathcal{X}.
\]
This is weak convergence in the dual tested only against evaluation at points of the original space.
\end{definition}

\begin{definition} \label{def:reflexive-banach-space} \hlt{Reflexivity}\\
The canonical map $J:\mathcal{X}\to\mathcal{X}^{**}$ is defined by $Jx(\varphi)=\varphi(x)$. A Banach space is \hlt{reflexive} if $J$ is onto.
\end{definition}

\begin{theorem} \label{thm:banach-alaoglu} \hlt{Banach--Alaoglu Theorem}\\
The closed unit ball of $\mathcal{X}^*$ is compact in the weak-* topology. Consequently, bounded nets of functionals have weak-* cluster points.
\end{theorem}
