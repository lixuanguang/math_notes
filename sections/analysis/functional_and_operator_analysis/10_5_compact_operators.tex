\subsubsection{Compact Operators}

Compact operators are infinite-dimensional maps with finite-dimensional behavior after taking closures. They carry bounded sets to relatively compact sets, so their spectral and solvability theory resembles matrix theory more than general bounded-operator theory.

\begin{definition} \label{def:compact-and-finite-rank-operators} \hlt{Compact and Finite-Rank Operators}\\
Let $\mathcal{X},\mathcal{Y}$ be normed spaces. A linear map $T:\mathcal{X}\to\mathcal{Y}$ is \hlt{compact} if $T(B)$ has compact closure in $\mathcal{Y}$ whenever $B\subseteq\mathcal{X}$ is bounded.\\
The map has \hlt{finite rank} if $\dim\operatorname{range}(T)<\infty$.
\end{definition}

\begin{proposition} \label{prop:finite-rank-operators-compact} \hlt{Finite Rank Operators Are Compact}\\
If $\mathcal{X}$ is normed, $\mathcal{Y}$ is Banach, and $T:\mathcal{X}\to\mathcal{Y}$ is bounded with finite-dimensional range, then $T$ is compact.
\end{proposition}

\begin{theorem} \label{thm:compact-operator-closed-under-norm-limits} \hlt{Norm Limits of Compact Operators}\\
If $T_n:\mathcal{X}\to\mathcal{Y}$ are compact operators and $T_n\to T$ in operator norm, with $\mathcal{Y}$ Banach, then $T$ is compact.
\end{theorem}

\begin{theorem} \label{thm:fredholm-alternative-compact-operator} \hlt{Fredholm Alternative}\\
Let $K:\mathcal{X}\to\mathcal{X}$ be compact on a Banach space. For $\lambda\neq0$, the equation
\[
(\lambda I-K)x=y
\]
has a solution for every $y$ exactly when the homogeneous equation $(\lambda I-K)x=0$ has only the zero solution. In this case $(\lambda I-K)^{-1}$ is bounded.
\end{theorem}
