After measurability is fixed in Definition \ref{def:measure-space-and-measurable-functions}, build Lebesgue integral from simple functions, extend it through suprema and signs, and end with null sets, convergence theorems, and $L^2$ geometry linked to Definition \ref{def:inner-product-space}.

\begin{definition} \label{def:lebesgue-product-core-source-todo}\label{def:simple-functions-and-lebesgue-integral} \hlt{Simple Functions and the Lebesgue Integral}\\
Let $(X,\mathcal{M},\mu)$ be measure space. For $E\subseteq X$, set $\mathbf{1}_E(x)\defeq
\begin{cases}
1,&x\in E,\\
0,&x\notin E.
\end{cases}$. Real-valued $s$ is \hlt{simple} if range finite.\\
With distinct values $c_i$ and sets $E_i=\{s=c_i\}$, $s=\sum\limits_{i=1}^{n}c_i\mathbf{1}_{E_i}$ and $s$ is measurable iff $E_i\in\mathcal{M}$ for all $i$.\\
For $s=\sum_{i=1}^{n}c_i\mathbf{1}_{E_i}\geq0$ simple measurable and $E\in\mathcal{M}$, define $\mathcal{I}_E(s)\defeq\sum\limits_{i=1}^{n}c_i\mu(E\cap E_i)$.\\
For $f\geq0$ measurable, let $\mathcal{S}^+(f)\defeq\{s:0\leq s\leq f,\ s\text{ simple measurable}\}$ and define $\int_E f\,d\mu\defeq\sup\limits_{s\in\mathcal{S}^+(f)}\mathcal{I}_E(s)$.\\
For measurable real $f$, put $f^+=\max(f,0)$ and $f^-=-\min(f,0)$.\\
If at least one of $\int_E f^+\,d\mu,\int_E f^-\,d\mu$ is finite, set $\int_E f\,d\mu\defeq\int_E f^+\,d\mu-\int_E f^-\,d\mu$.\\
If both are finite, $f$ is \hlt{Lebesgue integrable} on $E$, written $f\in\mathscr{L}^1(\mu)$ on $E$.
\end{definition}

\begin{theorem} \label{thm:simple-approximation-measurable-functions} \hlt{Approximation by Simple Functions}\\
Every real $f$ on $X$ is the pointwise limit of simple functions $s_n$.\\
If $f$ is measurable, choose $s_n$ measurable; if $f\geq0$, choose $0\leq s_n\uparrow f$.
\begin{proof}
For $f\geq0$, set $E_{ni}=\left\{x:\frac{i-1}{2^n}\leq f(x)<\frac{i}{2^n}\right\}$; $F_n=\{f\geq n\}$; and $s_n=\sum\limits_{i=1}^{n2^n}\frac{i-1}{2^n}\mathbf{1}_{E_{ni}}+n\mathbf{1}_{F_n}$.\\
Then $s_n$ is simple, measurable when $f$ is, and $0\leq s_n\uparrow f$. For general $f$, approximate $f^+$ and $f^-$ and subtract.\\
\hlt{Structural Commentary}: Simple functions quantize the range of $f$.\\
\hlt{Key Insight}: Dyadic range partitions give pointwise approximation while preserving measurability.
\end{proof}
\end{theorem}

\begin{proposition} \label{prop:basic-lebesgue-integral-properties} \hlt{Basic Integral Properties}\\
Let $E\in\mathcal{M}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $f$ is measurable and bounded on $E$, and $\mu(E)<+\infty$, then $f\in\mathscr{L}^1(\mu)$ on $E$.
\item If $a\leq f(x)\leq b$ on $E$ and $\mu(E)<+\infty$, then $a\mu(E)\leq\int_E f\,d\mu\leq b\mu(E)$.
\item If $f,g\in\mathscr{L}^1(\mu)$ on $E$ and $f\leq g$ on $E$, then $\int_E f\,d\mu\leq\int_E g\,d\mu$.
\item If $f\in\mathscr{L}^1(\mu)$ on $E$, then $cf\in\mathscr{L}^1(\mu)$ on $E$ for finite $c$, and $\int_E cf\,d\mu=c\int_E f\,d\mu$.
\item If $\mu(E)=0$ and $f$ is measurable, then $\int_E f\,d\mu=0$.
\item If $f\in\mathscr{L}^1(\mu)$ on $E$, $A\in\mathcal{M}$, and $A\subseteq E$, then $f\in\mathscr{L}^1(\mu)$ on $A$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\abs{f}\leq M$ on $E$, then $0\leq f^\pm\leq M\mathbf{1}_E$ on $E$, so $\int_Ef^\pm\,d\mu\leq M\mu(E)<+\infty$.
\item Since $a\mathbf{1}_E\leq f\leq b\mathbf{1}_E$ on $E$, order for the positive integral gives $a\mu(E)\leq\int_Ef\,d\mu\leq b\mu(E)$.
\item Since $g-f\geq0$ on $E$, $\int_E(g-f)\,d\mu\geq0$, hence $\int_Ef\,d\mu\leq\int_Eg\,d\mu$.
\item For $c\geq0$, scale the simple approximants. For $c<0$, use $(cf)^+=(-c)f^-$ and $(cf)^-=(-c)f^+$.
\item If $\mu(E)=0$, then $\mathcal{I}_E(s)=0$ for every $s\geq0$ simple, so $\int_Ef^\pm\,d\mu=0$.
\item Since $A\subseteq E$, $\mathcal{I}_A(s)\leq \mathcal{I}_E(s)$ for $s\geq0$ simple; hence $\int_Af^\pm\,d\mu\leq\int_Ef^\pm\,d\mu<+\infty$.
\end{enumerate}
\hlt{Structural Commentary}: Estimate $f^\pm$ by simpler functions.\\
\hlt{Key Insight}: Monotonicity makes bounds, null sets, and restrictions immediate.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:integral-countably-additive-set-function} \hlt{The Integral as a Countably Additive Set Function}\\
For $f\geq0$ measurable, or $f\in\mathscr{L}^1(\mu)$, set function $\phi_f(A)\defeq\int_A f\,d\mu$ for $A\in\mathcal{M}$ is countably additive on $\mathcal{M}$.
\begin{proof}
For $\mathbf{1}_B$, $\phi_{\mathbf{1}_B}(A)=\mu(A\cap B)$; linearity gives simple $f$.\\
If $f\geq0$ and $A=\dot\bigcup_nA_n$, then for $s\in\mathcal{S}^+(f)$, $\mathcal{I}_A(s)=\sum\limits_n\mathcal{I}_{A_n}(s)\leq\sum\limits_n\phi_f(A_n)$, so $\phi_f(A)\leq\sum\limits_n\phi_f(A_n)$.\\
Finite additivity gives $\sum\limits_{n=1}^{N}\phi_f(A_n)=\phi_f(\bigcup\limits_{n=1}^{N}A_n)\leq\phi_f(A)$; let $N\to\infty$. For signed $f$, use $\phi_f=\phi_{f^+}-\phi_{f^-}$.\\
\hlt{Structural Commentary}: Integrating over the set variable produces a new measure.\\
\hlt{Key Insight}: Countable additivity is first true for indicators, then simple functions, then suprema.
\end{proof}
\end{theorem}

\begin{definition} \label{def:almost-everywhere-and-null-set-invariance} \hlt{Almost Everywhere}\\
For measurable $E$, write $f\sim_{\mathrm{a.e.}}g$ on $E$ if $\mu(\{x\in E:f(x)\neq g(x)\})=0$. A property holds \hlt{almost everywhere} (a.e.) on $E$ if its failure set in $E$ has measure zero. Then $\sim_{\mathrm{a.e.}}$ is an equivalence relation; if $f\sim_{\mathrm{a.e.}}g$ on $E$, integrals over measurable $A\subseteq E$ agree whenever they exist. If $f\in\mathscr{L}^1(\mu)$ on $E$, then $f$ is finite a.e. on $E$.
\end{definition}

\begin{theorem} \label{thm:absolute-integrability-and-domination} \hlt{Absolute Integrability and Domination}\\
If $f\in\mathscr{L}^1(\mu)$ on $E$, then $\abs{f}\in\mathscr{L}^1(\mu)$ and $\abs{\int_E f\,d\mu}\leq\int_E\abs{f}\,d\mu$.\\
If $f$ is measurable and $\abs{f}\leq g\in\mathscr{L}^1(\mu)$ on $E$, then $f\in\mathscr{L}^1(\mu)$ on $E$.
\begin{proof}
Use $\abs{f}=f^++f^-$, $f\leq\abs{f}$, and $-f\leq\abs{f}$. If $\abs{f}\leq g$, then $0\leq f^\pm\leq g$, so $\int_Ef^\pm\,d\mu<+\infty$.\\
\hlt{Structural Commentary}: Lebesgue integrability is absolute integrability.\\
\hlt{Key Insight}: A single integrable majorant controls both signs.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:lebesgue-product-main-source-todo}\label{thm:lebesgue-convergence-theorems}\label{thm:dominated-convergence-lebesgue} \hlt{Lebesgue Convergence Theorems}\\
Let $E\in\mathcal{M}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Monotone convergence}: If $0\leq f_n\uparrow f$ on $E$, then $\int_E f_n\,d\mu\to\int_E f\,d\mu$
\item \hlt{Series form}: If $f_n\geq0$ and $f=\sum_{n=1}^{\infty}f_n$ on $E$, then $\int_E f\,d\mu=\sum_{n=1}^{\infty}\int_E f_n\,d\mu$
\item \hlt{Fatou}: If $f_n\geq0$, then $\int_E \liminf\limits_{n\to\infty}f_n\,d\mu\leq\liminf\limits_{n\to\infty}\int_E f_n\,d\mu$
\item \hlt{Dominated convergence}: If $f_n\to f$ on $E$ and $\abs{f_n}\leq g\in\mathscr{L}^1(\mu)$ on $E$, then $\lim\limits_{n\to\infty}\int_E f_n\,d\mu=\int_E f\,d\mu$
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $s\in\mathcal{S}^+(f)$ and $0<c<1$, set $E_n=\{f_n\geq cs\}$.\\
Then $E_n\uparrow E$, so Theorem \ref{thm:continuity-from-below-set-functions} applied to $\phi_s(A)=\int_A s\,d\mu$ gives $\liminf_n\int_E f_n\,d\mu\geq c\int_Es\,d\mu$.\\
Let $c\uparrow1$ and take $\sup_s$; the reverse inequality is monotonicity.
\item Apply (i) to $S_N=\sum\limits_{n=1}^{N}f_n\uparrow f$.
\item Let $g_n=\inf\limits_{i\geq n}f_i$. Then $0\leq g_n\uparrow\liminf\limits_i f_i$ and $\int_Eg_n\,d\mu\leq\inf\limits_{i\geq n}\int_Ef_i\,d\mu$; use (i).
\item Fatou applied to $f_n+g$ gives $\int_E f\,d\mu\leq\liminf\limits_n\int_E f_n\,d\mu$.\\
Fatou applied to $g-f_n$ gives $\limsup\limits_n\int_E f_n\,d\mu\leq\int_E f\,d\mu$.
\end{enumerate}
\hlt{Structural Commentary}: The major convergence theorems all reduce limit interchange to monotone convergence.\\
\hlt{Key Insight}: Monotone convergence is the engine; Fatou and domination are its comparison forms.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:riemann-lebesgue-comparison} \hlt{Comparison with the Riemann Integral}\\
On $[a,b]$ with Lebesgue measure $\mu$, $f\in\mathcal{R}$ implies $f\in\mathscr{L}^1(\mu)$ and the integrals agree. If $f$ is bounded with discontinuity set $D$, then $f\in\mathcal{R}\iff\mu(D)=0$.
\begin{proof}
For refining partitions, let $L_n\uparrow L$ and $U_n\downarrow U$ be lower and upper step envelopes.\\
Monotone convergence identifies $\int L$ and $\int U$ with the lower and upper Riemann integrals; if $f\in\mathcal{R}$, then $L=U$ a.e., so the common Lebesgue integral equals the Riemann integral.\\
For bounded $f$, put $\omega_f(x)\defeq \lim_{r\downarrow0}\diam f\bigl([a,b]\cap(x-r,x+r)\bigr)$, where $D_k\defeq\{x:\omega_f(x)\geq1/k\}$.\\
Then $f$ is continuous at $x$ iff $\omega_f(x)=0$, and its discontinuity set is $D=\bigcup_{k\geq1}D_k$.
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Fix $k$ and $\varepsilon>0$. Choose $P$ with $U(P,f)-L(P,f)<\varepsilon/k$.\\
Intervals of $P$ whose interiors meet $D_k$ have total length $<\varepsilon$; adding the finitely many partition points, null by Proposition \ref{prop:finite-countable-cantor-null-sets}, covers $D_k$ by arbitrarily short intervals.\\
Thus $\mu(D_k)=0$ by Definition \ref{def:null-sets-and-real-line-measure-zero}, and $\mu(D)=0$.
\item[$(\Leftarrow)$] Suppose $\mu(D)=0$ and $\abs{f}\leq B$. If $B=0$, there is nothing to prove. Given $\varepsilon>0$, choose $k$ with $(b-a)/k<\varepsilon/2$ and cover $D_k$ by intervals of total length $<\varepsilon/(4B)$.\\
On the compact complement, $\omega_f<1/k$, so a finite refinement makes the good intervals contribute $<\varepsilon/2$ to $U-L$; the covered intervals contribute $<2B\cdot\varepsilon/(4B)=\varepsilon/2$.\\
Hence $U(P,f)-L(P,f)<\varepsilon$, and Theorem \ref{thm:riemann-stieltjes-integrability-criterion} with $\alpha(x)=x$ gives $f\in\mathcal{R}$.
\end{enumerate}
\hlt{Structural Commentary}: Darboux gap measures oscillation sets.\\
\hlt{Key Insight}: Lebesgue measure turns ``bad discontinuities'' into a null-set condition.
\end{proof}
\end{theorem}

\begin{perspective} \label{per:lebesgue-criterion-riemann-roadmap} \hlt{Lebesgue's Criterion as a Roadmap}\\
Theorem \ref{thm:riemann-lebesgue-comparison} gives the threshold: a bounded function is Riemann integrable exactly when its discontinuities are null. Finite, countable, and Cantor-type exceptions are allowed; Dirichlet's function fails.
\end{perspective}

\begin{definition} \label{def:complex-integrable-and-l2} \hlt{Complex Integrals and $L^2$}\\
For $f=u+iv$, measurability means measurability of $u$ and $v$.\\
On $E\in\mathcal{M}$, $f\in\mathscr{L}^1(\mu)$ on $E$ $\iff$ $f$ is measurable and $\int_E\abs{f}\,d\mu<+\infty$.\\
On $X$, $f\in L^2(\mu)$ $\iff$ $f$ is complex measurable and $\int_X\abs{f}^2\,d\mu<+\infty$; $\norm{f}_2\defeq\left(\int_X\abs{f}^2\,d\mu\right)^{1/2}$.\\
Metric and inner-product structure identify functions equal a.e.
\end{definition}

\begin{theorem} \label{thm:l2-schwarz-minkowski-and-completeness} \hlt{$L^2$ Geometry}\\
For $f,g\in L^2(\mu)$, $\int_X\abs{fg}\,d\mu\leq\norm{f}_2\norm{g}_2$; $f+g\in L^2(\mu)$; and $\norm{f+g}_2\leq\norm{f}_2+\norm{g}_2$.\\
Modulo equality a.e., $L^2(\mu)$ is complete for $\norm{\cdot}_2$.
\begin{proof}
For real $\lambda$, $0\leq Q(\lambda)\defeq\int_X(\abs{f}+\lambda\abs{g})^2\,d\mu =\norm{f}_2^2+2\lambda\int_X\abs{fg}\,d\mu+\lambda^2\norm{g}_2^2$, so the discriminant gives Schwarz; Minkowski follows from $\norm{f+g}_2^2\leq\norm{f}_2^2+2\norm{f}_2\norm{g}_2+\norm{g}_2^2$.\\
For completeness, let $(f_n)$ be Cauchy and choose $n_k$ with $\norm{f_{n_{k+1}}-f_{n_k}}_2\leq2^{-k}$. Put $h_N=\sum_{k<N}\abs{f_{n_{k+1}}-f_{n_k}}$.\\
Then $\norm{h_N}_2\leq\sum_{k<N}2^{-k}$, so Fatou gives $h=\lim h_N\in L^2$; hence $f_{n_k}\to f$ a.e. for some $f$.\\
Again by Fatou, $\norm{f-f_n}_2\leq\liminf_k\norm{f_{n_k}-f_n}_2$, so $f_n\to f$ in $L^2$.\\
\hlt{Structural Commentary}: $L^2$ is Hilbert-space geometry in measure form.\\
\hlt{Key Insight}: The same Schwarz mechanism survives because the integral is positive and additive.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:l2-orthonormal-systems-riesz-fischer-parseval} \hlt{Orthonormal Systems in $L^2$}\\
Let $(\phi_n)$ be orthonormal in $L^2(\mu)$: $\int_X\phi_n\overline{\phi_m}\,d\mu=\delta_{nm}$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item For $f\in L^2(\mu)$, $c_n=\int_Xf\overline{\phi_n}\,d\mu$ defines $f\sim_{\mathrm{coef}}\sum c_n\phi_n$.
\item If $\sum\abs{c_n}^2<+\infty$, then $s_n=\sum_{k=1}^{n}c_k\phi_k$ converges in $L^2$ to some $f$.
\item If $(\phi_n)$ is complete and $f\sim_{\mathrm{coef}}\sum c_n\phi_n$, then $\int_X\abs{f}^2\,d\mu=\sum\limits_{n=1}^{\infty}\abs{c_n}^2$.
\end{enumerate}
For the trigonometric system on $[-\pi,\pi]$, this is $L^2$ Parseval and a.e. uniqueness.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Orthonormality gives finite coordinate identities behind Theorems \ref{thm:bessel-inequality-finite-orthonormal-list} and \ref{thm:parseval-identity-finite-dimensional}.\\
In particular, $\norm{s_n-s_m}_2^2=\sum\limits_{k=m+1}^{n}\abs{c_k}^2$.
\item $\sum\abs{c_n}^2<+\infty$ makes $(s_n)$ Cauchy, so Theorem \ref{thm:l2-schwarz-minkowski-and-completeness} gives $s_n\to f$ in $L^2$.\\
Schwarz passes coefficients to the limit.
\item For complete $(\phi_n)$, Bessel gives $\sum\abs{c_n}^2<+\infty$. The limit $g\sim_{\mathrm{coef}}\sum c_n\phi_n$ from (ii) satisfies $f-g\perp\phi_n$ for all $n$, hence $\norm{f-g}_2=0$; finite Parseval then gives the displayed identity.
\end{enumerate}
\hlt{Structural Commentary}: Complete orthonormal systems coordinatize $L^2$ by square-summable sequences.\\
\hlt{Key Insight}: Parseval is Pythagoras after passing from finite coordinates to an $L^2$ limit.
\end{proof}
\end{theorem}

\begin{method} \label{meth:lebesgue-product-proof-patterns-source-todo} \hlt{Lebesgue Proof Patterns}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Approximate}: for $f\geq0$, use $s\in\mathcal{S}^+(f)$ and $\int_E f\,d\mu=\sup_{s\in\mathcal{S}^+(f)}\mathcal{I}_E(s)$.
\item \hlt{Promote}: prove for $\mathbf{1}_A$, extend to $s=\sum_i c_i\mathbf{1}_{A_i}$, then pass through $s_n\uparrow f$.
\item \hlt{Ignore null sets}: if $f\sim_{\mathrm{a.e.}}g$ on $E$, replace $f$ by $g$ inside integrals over measurable $A\subseteq E$.
\item \hlt{Split signs}: use $f=f^+-f^-$ and $h=u+iv$.
\item \hlt{Exchange limits}: reduce Fatou and dominated convergence to $0\leq f_n\uparrow f$.
\end{enumerate}
