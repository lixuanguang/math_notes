Cardinality used sequences as enumerations in Definition \ref{def:sequence-as-enumeration}. Here the same object becomes a convergence probe: a sequence tests the local geometry of a metric space, while Cauchy sequences test whether the space contains the limits that its own distances predict.

\begin{definition} \label{def:sequence-convergence-metric}\label{def:sequences-core-definitions-source-todo} \hlt{Convergent Sequence in a Metric Space}\\
Let $(p_n)$ be a sequence in a metric space $X$.\\
Then $(p_n)$ \hlt{converges} to $p\in X$ if for every $\varepsilon>0$ there is an integer $N$ such that $n\geq N$ $\Longrightarrow$ $d(p_n,p)<\varepsilon$.\\
Notation is $p_n\to p$ or $\lim\limits_{n\to\infty}p_n=p$. If no such point $p\in X$ exists, the sequence \hlt{diverges}.
\end{definition}

\begin{definition} \label{def:sequence-range-and-boundedness} \hlt{Range and Boundedness of a Sequence}\\
The \hlt{range} of $(p_n)$ is the set $\{p_n:n\in\N\}$.\\
A sequence in a metric space is \hlt{bounded} if its range is bounded in the sense of Definition \ref{def:bounded-and-perfect-metric-sets}.
\end{definition}

\begin{definition} \label{def:eventually-and-frequently-in-a-set} \hlt{Eventually and Frequently in a Set}\\
Let $(p_n)$ be a sequence in $X$, and let $E\subseteq X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $(p_n)$ is \hlt{eventually in $E$} if $\exists N\in\N$ such that $n\geq N\Longrightarrow p_n\in E$.
\item $(p_n)$ is \hlt{frequently in $E$} if $\forall N\in\N$ $\exists n\geq N$ such that $p_n\in E$.
\end{enumerate}
Eventually means tail containment; frequently means arbitrarily late visits.
\end{definition}

\begin{theorem} \label{thm:metric-sequence-basic-properties} \hlt{Basic Properties of Metric Convergence}\\
Let $(p_n)$ be a sequence in a metric space $X$.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $p_n\to p$ if and only if every neighborhood of $p$ contains $p_n$ for all but finitely many $n$; equivalently, $(p_n)$ is eventually in every neighborhood of $p$.
\item If $p_n\to p$ and $p_n\to p'$, then $p=p'$.
\item If $E\subseteq X$ and $p$ is a limit point of $E$, then there is a sequence $(p_n)$ in $E$ such that $p_n\to p$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item This is the $\varepsilon$-$N$ definition, since every neighborhood of $p$ contains a ball about $p$.
\item Given $\varepsilon>0$, choose $N,N'$ so that $d(p_n,p)<\varepsilon/2$ for $n\geq N$ and $d(p_n,p')<\varepsilon/2$ for $n\geq N'$. For $n\geq\max\{N,N'\}$, the triangle inequality gives $d(p,p')<\varepsilon$. Hence $d(p,p')=0$.
\item For each $n$, choose $p_n\in E$ with $p_n\neq p$ and $d(p_n,p)<1/n$, using Definition \ref{def:metric-topology-core-definitions}. Then $p_n\to p$.
\end{enumerate}
\hlt{Structural Commentary}: Eventual containment gives convergence; triangle inequality gives uniqueness.\\
\hlt{Key Insight}: A limit point can always be approached by a sequence of points from the set.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:complex-sequence-limit-algebra} \hlt{Limit Algebra for Complex Sequences}\\
Suppose $(s_n)$ and $(t_n)$ are complex sequences with $s_n\to s$ and $t_n\to t$. Then, for every $c\in\C$,
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $s_n+t_n\to s+t$;
\item $cs_n\to cs$ and $c+s_n\to c+s$;
\item $s_nt_n\to st$;
\item if $s\neq0$ and $s_n\neq0$ for all $n$, then $1/s_n\to1/s$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Given $\varepsilon>0$, choose $N$ such that $\abs{s_n-s}<\varepsilon/2$, $\abs{t_n-t}<\varepsilon/2$ for $n\geq N$.\\
Then $\abs{(s_n+t_n)-(s+t)}\leq\abs{s_n-s}+\abs{t_n-t}<\varepsilon$.
\item Scalar multiplication uses $\abs{cs_n-cs}=\abs{c}\abs{s_n-s}$; translation is (i) with constant sequence $c$.
\item Since $s_n\to s$, choose $N\geq2$ with $\abs{s_n-s}<1$ for $n\geq N$ and set $M\defeq\max\{\abs{s_1},\ldots,\abs{s_{N-1}},\abs{s}+1\}$.\\
Then $\abs{s_nt_n-st}\leq \abs{s_n}\abs{t_n-t}+\abs{t}\abs{s_n-s}\leq M\abs{t_n-t}+\abs{t}\abs{s_n-s}\to0$.
\item If $s\neq0$, then for all sufficiently large $n$, $\abs{s_n}>\abs{s}/2$. Hence $\abs{\frac1{s_n}-\frac1s}=\frac{\abs{s_n-s}}{\abs{s_n}\abs{s}}\leq \frac{2}{\abs{s}^2}\abs{s_n-s}\to0$.
\end{enumerate}
\hlt{Structural Commentary}: Boundedness controls products; nonzero limits control reciprocals.\\
\hlt{Key Insight}: Product and reciprocal rules are estimates after the easy sum rule.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:rk-sequence-coordinate-convergence} \hlt{Coordinatewise Convergence in $\R^k$}\\
Let $\mathbf{x}_n=(\alpha_{j,n})_{j=1}^k$ and $\mathbf{x}=(\alpha_j)_{j=1}^k$ in $\R^k$. Then $\mathbf{x}_n\to\mathbf{x}$ $\Longleftrightarrow$ $\alpha_{j,n}\to\alpha_j$ $(1\leq j\leq k)$.\\
Moreover, if $\mathbf{x}_n\to\mathbf{x}$, $\mathbf{y}_n\to\mathbf{y}$, and $\beta_n\to\beta$ in $\R$, then $\mathbf{x}_n+\mathbf{y}_n\to\mathbf{x}+\mathbf{y}$; $\mathbf{x}_n\cdot\mathbf{y}_n\to\mathbf{x}\cdot\mathbf{y}$; $\beta_n\mathbf{x}_n\to\beta\mathbf{x}$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Since $\abs{\alpha_{j,n}-\alpha_j}\leq\norm{\mathbf{x}_n-\mathbf{x}}_2$ for each $j$, each coordinate converges.
\item[$(\Leftarrow)$] Given $\varepsilon>0$, choose $N$ so that $\abs{\alpha_{j,n}-\alpha_j}<\varepsilon/\sqrt{k}$ for all $j$ and $n\geq N$. Then $\norm{\mathbf{x}_n-\mathbf{x}}_2<\varepsilon$.
\end{enumerate}
The algebraic conclusions follow coordinatewise from Theorem \ref{thm:complex-sequence-limit-algebra}.\\
\hlt{Structural Commentary}: Euclidean convergence is exactly simultaneous coordinate convergence.\\
\hlt{Key Insight}: The norm packages finitely many coordinate estimates into one estimate.
\end{proof}
\end{theorem}

\begin{definition} \label{def:subsequence-and-subsequential-limit} \hlt{Subsequences and Subsequential Limits}\\
Let $(p_n)$ be a sequence. If $n_1<n_2<n_3<\cdots$ is a strictly increasing sequence of positive integers, then $(p_{n_k})$ is a \hlt{subsequence} of $(p_n)$. If $p_{n_k}\to p$, then $p$ is a \hlt{subsequential limit} of $(p_n)$.
\end{definition}

\begin{remark} \label{rem:subsequences-of-convergent-sequences} \hlt{Subsequences of Convergent Sequences}\\
If $p_n\to p$, then every subsequence of $(p_n)$ also converges to $p$. Conversely, if every subsequence of $(p_n)$ converges to $p$, then $(p_n)$ itself converges to $p$, since $(p_n)$ is one of its subsequences.
\end{remark}

\begin{theorem} \label{thm:compact-sequential-subsequence} \hlt{Compactness Produces Convergent Subsequences}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $(p_n)$ is a sequence in a compact metric space $X$, then $p_{n_i}\to p$ for some subsequence and some $p\in X$.
\item If $(\mathbf{x}_n)$ is bounded in $\R^k$, then it has a convergent subsequence.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Let $E=\{p_n:n\in\N\}$. If $E$ is finite, choose a value $p$ attained infinitely often; the corresponding constant subsequence converges. If $E$ is infinite, Theorem \ref{thm:infinite-subset-of-compact-has-limit-point} gives a limit point $p\in X$ of $E$. By Theorem \ref{thm:limit-point-neighborhoods-infinitely-many}, choose $n_i>n_{i-1}$ with $d(p_{n_i},p)<1/i$. Then $p_{n_i}\to p$.
\item The range of $(\mathbf{x}_n)$ has compact closure by Theorem \ref{thm:heine-borel-rk}; apply (i) in that closure.
\end{enumerate}
\hlt{Structural Commentary}: Compactness turns arbitrary infinite behavior into an accumulating subsequence.\\
\hlt{Key Insight}: In $\R^k$, boundedness is enough because bounded sets sit inside compact closed cells.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:subsequential-limits-closed} \hlt{Subsequential Limits Form a Closed Set}\\
For a sequence $(p_n)$ in a metric space $X$, the set of all subsequential limits is closed.
\begin{proof}
Let $E^*$ be the set of subsequential limits and let $q$ be a limit point of $E^*$. Set $n_0=0$.\\
For each $i$, choose $x_i\in E^*$ with $d(x_i,q)<2^{-i}$. Since $x_i$ is a subsequential limit, choose $n_i>n_{i-1}$ such that $d(p_{n_i},x_i)<2^{-i}$. Then $d(p_{n_i},q)<2^{1-i}$, so $p_{n_i}\to q$, and $q\in E^*$.\\
\hlt{Structural Commentary}: A limit of subsequential limits is reached by diagonalizing through those subsequences.\\
\hlt{Key Insight}: Choose later and closer terms at the same time.
\end{proof}
\end{theorem}

\begin{definition} \label{def:cauchy-sequence} \hlt{Cauchy Sequence}\\
Sequence $(p_n)$ in metric space $X$ is \hlt{Cauchy} if $\forall \varepsilon>0$  $\exists$ integer $N$ s.t. $n\geq N,\ m\geq N$ $\Longrightarrow$ $d(p_n,p_m)<\varepsilon$.
\end{definition}

\begin{definition} \label{def:diameter-of-metric-set} \hlt{Diameter}\\
Let $E$ be a nonempty subset of a metric space $X$. The \hlt{diameter} of $E$ is $\diam E\defeq\sup\{d(p,q):p,q\in E\}$, with the supremum interpreted in the extended real sense when the set of distances is unbounded.
\end{definition}

\begin{proposition} \label{prop:cauchy-tail-diameters} \hlt{Cauchy Sequences and Tail Diameters}\\
For $(p_n)$ in a metric space $X$, set $E_N\defeq\{p_j:j\geq N\}$. Then $(p_n)$ is Cauchy if and only if $\diam E_N\to0$.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] Given $\varepsilon>0$, choose $N_0$ so that $m,n\geq N_0$ implies $d(p_m,p_n)<\varepsilon/2$.\\
If $N\geq N_0$, then every pair in $E_N$ has distance $<\varepsilon/2$, so $\diam E_N<\varepsilon$.
\item[$(\Leftarrow)$] Given $\varepsilon>0$, choose $N_0$ with $\diam E_{N_0}<\varepsilon$.\\
If $m,n\geq N_0$, then $p_m,p_n\in E_{N_0}$, hence $d(p_m,p_n)\leq\diam E_{N_0}<\varepsilon$.
\end{enumerate}
\hlt{Structural Commentary}: Cauchy behavior is shrinking diameter of sequence tails.\\
\hlt{Key Insight}: The limit is not named; only mutual closeness is measured.
\end{proof}
\end{proposition}

\begin{theorem} \label{thm:nested-compact-diameters-singleton} \hlt{Nested Compact Sets with Shrinking Diameter}\\
Let $X$ be a metric space.
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\overline{E}$ is the closure of $E$ in a metric space, then $\diam\overline{E}=\diam E$.
\item If $K_1\supseteq K_2\supseteq K_3\supseteq\cdots$ is a sequence of nonempty compact sets in $X$ and $\diam K_n\to0$, then $\bigcap_{n=1}^{\infty}K_n$ consists of exactly one point.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $E\subseteq\overline{E}$, $\diam E\leq\diam\overline{E}$. For $p,q\in\overline{E}$ and $\varepsilon>0$, choose $p',q'\in E$ with $d(p,p'),d(q,q')<\varepsilon$.\\
Then $d(p,q)\leq\diam E+2\varepsilon$, so $\diam\overline{E}\leq\diam E$.
\item Put $I\defeq\bigcap_{n\geq1}K_n$. Corollary \ref{cor:nested-nonempty-compact-intersection} gives $I\neq\varnothing$.\\
If $x,y\in I$ with $x\neq y$, then $\diam K_n\geq d(x,y)>0$ for all $n$, contradicting $\diam K_n\to0$.
\end{enumerate}
\hlt{Structural Commentary}: Closure preserves $\diam$; $K_n\downarrow$ with $\diam K_n\to0$ forces a singleton.\\
\hlt{Key Insight}: Compactness gives existence; shrinking diameter gives uniqueness.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:cauchy-convergence-criterion}\label{thm:sequences-completeness-main-source-todo} \hlt{Cauchy Criterion and Completeness Results}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item In any metric space, every convergent sequence is Cauchy.
\item If $X$ is a compact metric space, then every Cauchy sequence in $X$ converges to a point of $X$.
\item In $\R^k$, every Cauchy sequence converges.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $p_n\to p$, choose $N$ with $d(p_n,p)<\varepsilon/2$ for $n\geq N$. For $m,n\geq N$, $d(p_n,p_m)<\varepsilon$.
\item Let $E_N\defeq\{p_j:j\geq N\}$ and $F_N\defeq\overline{E_N}$. Then $\diam F_N\to0$, the $F_N$ are compact and nested, and Theorem \ref{thm:nested-compact-diameters-singleton} gives $\bigcap_NF_N=\{p\}$. If $\diam F_N<\varepsilon$, then $p,p_n\in F_N$ for $n\geq N$, so $d(p_n,p)<\varepsilon$.
\item For Cauchy $(\mathbf{x}_n)\subseteq\R^k$, choose $N$ with $\diam\{\mathbf{x}_j:j\geq N\}<1$. Its range is bounded, hence has compact closure by Theorem \ref{thm:heine-borel-rk}; apply (ii).
\end{enumerate}
\hlt{Structural Commentary}: Cauchy controls $\diam E_N$; compactness turns $\bigcap_NF_N$ into the limit.\\
\hlt{Key Insight}: In $\R^k$, the Cauchy criterion is equivalent to convergence.
\end{proof}
\end{theorem}

\begin{definition} \label{def:complete-metric-space} \hlt{Complete Metric Space}\\
A metric space $X$ is \hlt{complete} if every Cauchy sequence in $X$ converges to a point of $X$.\\
Compact metric spaces and $\R^k$ are complete (by Theorem \ref{thm:cauchy-convergence-criterion}).
\end{definition}

\begin{proposition} \label{prop:basic-completeness-consequences} \hlt{Closed Subsets of Complete Spaces}\\
Every closed subset of a complete metric space is complete.
\begin{proof}
Let $E\subseteq X$ be closed, $X$ complete. If $(p_n)\subseteq E$ is Cauchy, then $p_n\to p\in X$; since $E$ is closed, $p\in E$.
\hlt{Structural Commentary}: Completeness gives the limit in $X$; closedness keeps it in $E$.\\
\hlt{Key Insight}: Closed subspaces inherit limits of their own Cauchy sequences.
\end{proof}
\end{proposition}

\begin{definition} \label{def:contraction-mapping} \hlt{Contraction}\\
Let $X$ be a metric space with metric $d$.\\
A map $\varphi:X\to X$ is a \hlt{contraction} if $\exists$ a constant $0\leq c<1$ such that $d(\varphi(x),\varphi(y))\leq c\,d(x,y)$ for $x,y\in X$.
\end{definition}

\begin{theorem} \label{thm:contraction-principle} \hlt{Contraction Principle}\\
If $X$ is a complete metric space and $\varphi:X\to X$ is a contraction, then $\exists$ a unique $x\in X$ such that $\varphi(x)=x$.
\begin{proof}
Choose $x_0\in X$ and define $x_{n+1}=\varphi(x_n)$.\\
If $c<1$ is a contraction constant, induction gives $d(x_{n+1},x_n)\leq c^n d(x_1,x_0)$.\\
For $n<m$, the triangle inequality and the finite geometric sum give $d(x_n,x_m)\leq \sum\limits_{j=n}^{m-1}d(x_j,x_{j+1}) \leq \frac{c^n}{1-c}d(x_1,x_0)$.\\
Thus $(x_n)$ is Cauchy, so completeness gives $x_n\to x\in X$.\\
Since a contraction is continuous, $\varphi(x)=\lim\varphi(x_n)=\lim x_{n+1}=x$.\\
If $\varphi(x)=x$ and $\varphi(y)=y$, then $d(x,y)\leq c\,d(x,y)$, hence $d(x,y)=0$ and $x=y$.\\
\hlt{Structural Commentary}: Iteration turns the contraction estimate into a geometric Cauchy estimate.\\
\hlt{Key Insight}: Completeness supplies the limiting point; the strict constant $c<1$ forces uniqueness.
\end{proof}
\end{theorem}

\begin{definition} \label{def:monotone-sequence} \hlt{Monotone Real Sequence}\\
A real sequence $(s_n)$ is \hlt{monotonically increasing} if $s_n\leq s_{n+1}$ for all $n$, and \hlt{monotonically decreasing} if $s_n\geq s_{n+1}$ for all $n$. A \hlt{monotonic} sequence is either increasing or decreasing.
\end{definition}

\begin{theorem} \label{thm:monotone-convergence} \hlt{Monotone Convergence Theorem for Sequences}\\
A monotonic real sequence converges if and only if it is bounded.
\begin{proof}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item[$(\Rightarrow)$] If $s_n\to s$, choose $N$ with $\abs{s_n-s}<1$ for $n\geq N$. The tail lies in $(s-1,s+1)$ and the initial set $\{s_1,\ldots,s_{N-1}\}$ is finite, so $(s_n)$ is bounded.
\item[$(\Leftarrow)$] Let $E\defeq\{s_n:n\in\N\}$. If $(s_n)$ is increasing, set $s\defeq\sup E$. Given $\varepsilon>0$, choose $N$ with $s-\varepsilon<s_N\leq s$; then $s-\varepsilon<s_n\leq s$ for $n\geq N$, so $s_n\to s$. If $(s_n)$ is decreasing, set $s\defeq\inf E$. Given $\varepsilon>0$, choose $N$ with $s\leq s_N<s+\varepsilon$; then $s\leq s_n\leq s_N<s+\varepsilon$ for $n\geq N$, so $s_n\to s$.
\end{enumerate}
\hlt{Structural Commentary}: Monotonicity traps bounded tails at $\sup E$ or $\inf E$.\\
\hlt{Key Insight}: The least upper bound is forced to be the limit.
\end{proof}
\end{theorem}

\begin{definition} \label{def:sequence-diverges-to-infinity} \hlt{Divergence to $\pm\infty$}\\
For a real sequence $(s_n)$,
\begin{align}
s_n\to+\infty &\iff \forall M\in\R,\ \exists N\in\N:\ n\geq N\Rightarrow s_n\geq M, \nonumber\\
s_n\to-\infty &\iff \forall M\in\R,\ \exists N\in\N:\ n\geq N\Rightarrow s_n\leq M. \nonumber
\end{align}
This extends $\to$ notation for divergent sequences; Definition \ref{def:sequence-convergence-metric} is unchanged.
\end{definition}

\begin{definition} \label{def:upper-and-lower-limits} \hlt{Upper and Lower Limits}\\
For real sequence $(s_n)$, let $E\defeq\{x\in\overline{\R}:\exists\, n_1<n_2<\cdots,\ s_{n_k}\to x\}$, allowing $x=\pm\infty$.\\
Set $s^*\defeq\sup E$ and $s_*\defeq\inf E$; equivalently, $\limsup_{n\to\infty}s_n=s^*$ and $\liminf_{n\to\infty}s_n=s_*$.
\end{definition}

\begin{theorem} \label{thm:limsup-characterization} \hlt{Characterization of the Upper Limit}\\
Let $(s_n)$ be a real sequence, and let $E$ and $s^*$ be as in Definition \ref{def:upper-and-lower-limits}. Then:
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item $s^*\in E$;
\item if $x>s^*$, then there is an integer $N$ such that $n\geq N$ implies $s_n<x$.
\end{enumerate}
Moreover, $s^*$ is the only extended real number with these two properties.\\
For $s_*$, the corresponding unique characterization is: $s_*\in E$, and if $x<s_*$, then $s_n>x$ eventually.
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Since $s^*=\sup E$: if $s^*=+\infty$, choose a subsequence tending to $+\infty$; if $s^*\in\R$, use Theorem \ref{thm:subsequential-limits-closed}.\\
If $s^*=-\infty$, then $s_n\to-\infty$. Hence $s^*\in E$.
\item If $s_n\geq x>s^*$ infinitely often, then some $y\in E$ satisfies $y\geq x$, contradiction.
\item If $p<q$ both satisfy (i)--(ii), choose $p<x<q$. Then $s_n<x$ eventually, contradicting $q\in E$.
\end{enumerate}
Characterization of $s_*$ follows by applying preceding argument to $(-s_n)$, multiply resulting inequalities by $-1$.\\
\hlt{Structural Commentary}: Limsup is the maximal subsequential limit and eventual ceiling.\\
\hlt{Key Insight}: Limsup combines subsequences with eventual upper bounds.
\end{proof}
\end{theorem}

\begin{example} \label{ex:upper-lower-limit-examples} \hlt{Upper and Lower Limit Examples}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $\Q\subseteq\{s_n:n\in\N\}$, then $E=\overline{\R}$; hence $\limsup s_n=+\infty$ and $\liminf s_n=-\infty$.
\item If $s_n=\frac{(-1)^n}{1+1/n}$, then $\limsup s_n=1$ and $\liminf s_n=-1$.
\item A real sequence converges to $s$ if and only if $\limsup s_n=\liminf s_n=s$.
\end{enumerate}
\end{example}

\begin{theorem} \label{thm:limsup-liminf-order-comparison} \hlt{Order Comparison for Upper and Lower Limits}\\
If $s_n\leq t_n$ for all $n\geq N$, where $N$ is fixed, then $\liminf\limits_{n\to\infty}s_n\leq \liminf\limits_{n\to\infty}t_n$ and $\limsup\limits_{n\to\infty}s_n\leq \limsup\limits_{n\to\infty}t_n$.
\begin{proof}
Discard finitely many terms, so $s_n\leq t_n$ for all $n$. Put $a=\limsup s_n$ and $b=\limsup t_n$.\\
If $x<a$, then $s_n>x$ infinitely often, hence $t_n>x$ infinitely often, so $x\leq b$ by Theorem \ref{thm:limsup-characterization}. Thus $a\leq b$. \\
Applying this limsup result to $-t_n\leq -s_n$ gives $\limsup(-t_n)\leq\limsup(-s_n)$, equivalently $\liminf s_n\leq\liminf t_n$.\\
\hlt{Structural Commentary}: Eventual order is inherited by the extremal subsequential limits.\\
\hlt{Key Insight}: Limsup and liminf ignore finitely many initial terms.
\end{proof}
\end{theorem}

\begin{theorem} \label{thm:special-sequence-limits} \hlt{Special Sequence Limits}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item If $p>0$, then $\lim\limits_{n\to\infty}n^{-p}=0$.
\item If $p>0$, then $\lim\limits_{n\to\infty}\sqrt[n]{p}=1$.
\item $\lim\limits_{n\to\infty}\sqrt[n]{n}=1$.
\item If $p>0$ and $\alpha\in\R$, then $\lim\limits_{n\to\infty}\frac{n^\alpha}{(1+p)^n}=0$.
\item If $\abs{x}<1$, then $\lim\limits_{n\to\infty}x^n=0$.
\end{enumerate}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item Given $\varepsilon>0$, choose $N>(1/\varepsilon)^{1/p}$ by Theorem \ref{thm:archimedean-and-rational-density}. Then $n\geq N$ gives $n^{-p}<\varepsilon$.
\item If $p>1$, set $x_n\defeq\sqrt[n]{p}-1$. Then $x_n>0$ and $1+nx_n\leq(1+x_n)^n=p$, so $0<x_n\leq(p-1)/n\to0$.\\
If $p=1$, then $\sqrt[n]{p}=1$ for every $n$. If $0<p<1$, then $1/p>1$, so $\sqrt[n]{1/p}\to1$ by the previous case.\\
Since $\sqrt[n]{p}=1/\sqrt[n]{1/p}$ and limits preserve reciprocals away from $0$, $\sqrt[n]{p}\to1$.
\item Put $x_n\defeq\sqrt[n]{n}-1$. Then $x_n\geq0$, and for $n\geq2$ the binomial theorem gives $n=(1+x_n)^n\geq \frac{n(n-1)}{2}x_n^2$.\\
Thus $0\leq x_n\leq\sqrt{2/(n-1)}$, so $x_n\to0$.
\item Choose an integer $k>\alpha$ with $k>0$. For $n>2k$, $(1+p)^n\geq \binom{n}{k}p^k=\frac{n(n-1)\cdots(n-k+1)}{k!}p^k>\frac{n^k p^k}{2^k k!}$.\\
Hence $0<\frac{n^\alpha}{(1+p)^n}<\frac{2^kk!}{p^k}n^{\alpha-k}$, and $n^{\alpha-k}\to0$ by (i).
\item If $x\neq0$, write $\abs{x}=(1+p)^{-1}$ with $p>0$ and apply (iv) with $\alpha=0$; $x=0$ is immediate.
\end{enumerate}
\hlt{Structural Commentary}: Each limit is reduced to an elementary comparison sequence.\\
\hlt{Key Insight}: The binomial theorem supplies the right upper bounds for roots and exponential domination.
\end{proof}
\end{theorem}

\begin{method} \label{meth:sequence-proof-patterns}\label{meth:sequence-proof-patterns-source-todo} \hlt{Sequence Proof Patterns}
\begin{enumerate}[label=(\roman*)]
\setlength{\itemsep}{0pt}
\item \hlt{Convergence}: fix $\varepsilon>0$, then find $N$ so that all later terms lie in the relevant $\varepsilon$-ball.
\item \hlt{Uniqueness}: compare two proposed limits with $\varepsilon/2$ and use the triangle inequality.
\item \hlt{Subsequences}: choose $n_k\uparrow$ recursively with distances $<1/k$.
\item \hlt{Cauchy arguments}: estimate the diameter of tails rather than guessing the limit first.
\item \hlt{Monotone sequences}: use $\sup$ or $\inf$ and then prove eventual trapping.
\end{enumerate}
\hlt{Key Insight}: Most sequence proofs are bookkeeping for quantifiers, indices, and one controlling inequality.
\end{method}
